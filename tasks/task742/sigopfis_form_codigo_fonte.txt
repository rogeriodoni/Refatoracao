private uf, da, mo, se, tnf, nnf, cnf, cnpj, lcCha, lcDig

mo = '55'

------------------------------------------------------------
* PROCEDURE digitoverificador
------------------------------------------------------------
PROCEDURE digitoverificador
parameters _key

Local i As Integer
Local j As Integer
Local k As Integer
Local DV As Double

j = 5
DV = 0.00
For i = 1 To 43 Step 1
j = Iif(j = 2, 9, j - 1)
K = j * Val(Substr(_Key , i, 1))
DV = DV + K
Release K
EndFor

DV = Mod(DV, 11)
DV = Iif(Between(DV, 0, 1), 0, 11 - DV)
Return Alltrim(Str(DV))


ENDPROC

* PROCEDURE esperaarq
------------------------------------------------------------
PROCEDURE esperaarq
parameters cArq, nTime, cMsg
*** nTime = qtde de segundos a esperar
LOCAL i, nT, j, msg

nT = nTime
i=0
j=0
DO WHILE NOT FILE(cArq) AND j < nT
	msg = IIF(VARTYPE(cMsg)<>'C','Aguardando retorno da SEFAZ', cMsg)
	WAIT window msg timeout 1
	j=j+1
ENDDO
If NOT FILE(cArq)
	Return .F.
ENDIF	
Return .T.


ENDPROC

RETURN

ENDPROC

	ENDIF

*!*		lnRet=rStatusImpressora_NFCe_Daruma()	&& verifica status da impressora
