PROCEDURE Pagina.Lista.Grupo_op.Click
LParameters opcao

EndIf

Select Count(*) as QTD from crSigCdMri a where a.SubTot100s = 1 and a.DescGrps <> crSigCdMri.DescGrps into cursor crAux

	EndWith
Endif

			    [From SigCdMri ] + ;

		.get_ds_grupos.Enabled=(optreg=1)
		.opt_analiticos.Option1.Enabled = .t.
		.opt_analiticos.Option2.Enabled = .t.
		.getTitulo.Enabled=.t.

		.opt_analiticos.Option2.Enabled = .t.
		.getTitulo.Enabled=.t.
	EndWith

------------------------------------------------------------
* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
Local lnTpregs, lnRecno1

m.lnTpregs = This.Parent.Parent.optTpregs.Value
SELECT crSigCdMri
m.lnRecno1 = RecNo()

If InList(ThisForm.pcEscolha,'INSERIR')
	SELECT crSigCdMri
	Replace CidChaves WITH fUniqueIds()
Endif


If InList(ThisForm.pcEscolha,'INSERIR','ALTERAR')
	If IsEmpty(crSigCdMri.Titulos) .and. (crSigCdMrr.Mapas = 2)
		Messagebox( 'Título Não Preenchido!', 48, '' )
		ThisForm.Pagina.Dados.getTitulo.SetFocus
		Return
	Endif

	ThisForm.podatamgr.CursorQuery('SigCdGcr','crSigCdGcr', 'Codigos', crSigCdMri.Grupos)

	Registro=ThisForm.podatamgr.ChkRegister('SigCdGcr', 'Codigos', crSigCdMri.Grupos)
	If (m.lnTpregs==1.And.(IsEmpty( crSigCdMri.Grupos ) Or !Registro))
		Messagebox( 'Grupo Inválido!', 48, '' )
		ThisForm.Pagina.Dados.Get_cd_grupos.SetFocus
		Return
	Endif
	
	DECLARE laParms[2,2]
	laParms[1,1]="Balancos"
	laParms[1,2]=crSigCdMrr.Balancos
	laParms[2,1]="Grupos"
	laParms[2,2]=This.Parent.Parent.get_cd_grupos.Value

	If (m.lnTpregs==1.And.ThisForm.podatamgr.ChkRegister('SigCdMri', @laParms))
	
		SELECT crSigCdMri
		Skip 
		
	   If crSigCdMri.Grupos==This.Parent.Parent.get_cd_grupos.Value
		
			Messagebox( 'Grupo já cadastrado!', 48, '' )
			ThisForm.Pagina.Dados.Get_cd_grupos.SetFocus
			GoTo m.lnRecno1 
			
			Return
			
		EndIf
	Endif
	
	Select crSigCdMri
	GoTo m.lnRecno1 
	
	If m.lnTpregs==3 .And. IsEmpty( crSigCdMri.Classes )
		Messagebox( 'Classe Inválida!', 48, '' )
		ThisForm.Pagina.Dados.Get_classe.SetFocus
		Return
	Endif
	
	laParms[1,1]="Balancos"
	laParms[1,2]=crSigCdMrr.Balancos
	laParms[2,1]="Classes"
	laParms[2,2]=This.Parent.Parent.get_classe.Value

	If (m.lnTpregs==3.And.ThisForm.podatamgr.ChkRegister('SigCdMri', @laParms))
	
		SELECT crSigCdMri
		Skip 
		
	   If crSigCdMri.Classes==This.Parent.Parent.get_classe.Value
		
			Messagebox( 'Classe já cadastrada!!', 48, '' )
			ThisForm.Pagina.Dados.Get_classe.SetFocus
			GoTo m.lnRecno1
			
			Return
			
		EndIf
	Endif
	
	Select crSigCdMri
	GoTo m.lnRecno1
	
	If IsEmpty( ThisForm.Pagina.Dados.Cmb_Operacaos.ListIndex )
		Messagebox( 'Operação Inválida!', 48, '' )
		ThisForm.Pagina.Dados.Cmb_Operacaos.SetFocus
		Return
	Endif
	if !Inlist( crSigCdMri.Analiticos, 1, 2 )
		Messagebox( 'Opção inválida !',16,'' )
		Thisform.Pagina.Dados.Opt_analiticos.Setfocus
		Return
	Endif
	If (Thisform.Pagina.Dados.Fator.Value<=0) or (Thisform.Pagina.Dados.Fator.Value>100)
		Messagebox('Fator Inválido !!',0+48,'')
		Thisform.Pagina.Dados.Fator.Value = 100
		Thisform.Pagina.Dados.Fator.SetFocus
		Return
	Endif
	If (m.lnTpregs==2.And.IsEmpty(This.Parent.Parent.getDescgrps.Value))
		Messagebox( 'SubTotal Inválido.',16,'' )
		This.Parent.Parent.getDescgrps.SetFocus
		Return
	EndIf
	
	DECLARE laParms[3,2]
	laParms[1,1]="Balancos"
	laParms[1,2]=crSigCdMrr.Balancos
	laParms[2,1]="Grupos"
	laParms[2,2]=This.Parent.Parent.get_cd_grupos.Value
	laParms[3,1]="Descgrps"
	laParms[3,2]=This.Parent.Parent.get_cd_grupos.Value

	If (m.lnTpregs==2.And.;
	   ThisForm.podatamgr.ChkRegister('SigCdMri', @laParms))
	   
	   SELECT crSigCdMri  
	   Skip 
	   
	   If (crSigCdMrr.Balancos+crSigCdMri.Grupos+crSigCdMri.DescGrps)==(This.Parent.Parent.get_cd_grupos.Value+This.Parent.Parent.getDescgrps.Value)
	   
			Messagebox( 'SubTotal já Cadastrado.',16,'' )
			This.Parent.Parent.getDescgrps.SetFocus
			GoTo m.lnRecno1
			Return
			
		EndIf
	EndIf
	
	If m.lnTpregs==2 and Thisform.Pagina.Dados.Fator.Value<>100
			Messagebox( 'Fator Inválido',16,'' )
			GoTo m.lnRecno1
			With Thisform.Pagina.Dados.Fator
				.Value=100
				.Refresh
				.SetFocus
			EndWith
			Return
	Endif	
	Select crSigCdMri
	GoTo m.lnRecno1

	If crSigCdMri.Tpregs==1
		Replace crSigCdMri.Classes with crSigCdGcr.Codigos in CrSigCdMri
	Endif
	If crSigCdMri.Tpregs==2
		Replace crSigCdMri.Classes with '6' in CrSigCdMri
	Endif
	ThisForm.podatamgr.CursorQuery('SigCdCss','crSigCdCss', 'Classes', crSigCdMri.Classes)
	If ThisForm.podatamgr.ChkRegister('SigCdCss', 'Classes', crSigCdMri.Classes)
		Replace crSigCdMri.Nivels with crSigCdCss.Nivels in crSigCdMri
	Else
		Replace crSigCdMri.Nivels with 0 in crSigCdMri
	Endif
	ThisForm.podatamgr.CursorQuery('SigCdGcr','crSigCdGcr', 'Codigos', crSigCdMri.Grupos)
	ThisForm.podatamgr.CursorQuery('SigCdCss','crSigCdCss', 'Classes', crSigCdMri.Classes)
	Replace crSigCdMri.DescGrps With Iif(crSigCdMri.Tpregs==1.And.ThisForm.podatamgr.ChkRegister('SigCdGcr', 'Codigos', crSigCdMri.Grupos),crSigCdGcr.Descrs,crSigCdMri.DescGrps);
	        crSigCdMri.Operacaos With Iif( Thisform.Pagina.Dados.Cmb_operacaos.Listindex = 1, '+', '-' ) ;
			crSigCdMri.DescGrps With Iif(crSigCdMri.Tpregs==3.And.ThisForm.podatamgr.ChkRegister('SigCdCss', 'Classes', crSigCdMri.Classes),crSigCdCss.Descrs,crSigCdMri.DescGrps) in crSigCdMri
Endif

=DoDefault()

ThisForm.mAtivaPagina1

ENDPROC

PROCEDURE Valid
Local lcBusca, lcRecno

lcRecno = Recno()

SELECT crSigCdMrr
LOCATE FOR Balancos = crSigCdMri.Balancos AND titulos =  This.Value

	If lcRecno = Recno()
		Skip
		If lcBusca = crSigCdMrr.Balancos + crSigCdMrr.Titulos
			Skip - 1
			Messagebox( 'Título Já Utilizado!', 48, '' )
			This.Value = Space(10)
			Return
		Else
			Skip - 1
		Endif
	Else
		Messagebox( 'Título Já Utilizado!', 48, '' )
		This.Value = Space(10)
		Return
	Endif
Endif
