* Carrega os Campos do Cabeçalho da Tela
Local lnClr, lcCpo

For lnClr = 1 To 12
	If Not InList(lnClr, 3, 4, 7, 8, 9, 11, 12)
		lcCpo = [ThisForm.getCab] + Padl(lnClr, 3, [0])
		&lcCpo..Value = []
		&lcCpo..Refresh
	EndIf 
Next

ThisForm.getCab001.Value = TprMvCab.Jobs
If Not Empty(TprMvCab.Jobs)
	ThisForm.poDataMgr.CursorQuery([SigCdCli], [LocalBus], [IClis], TprMvCab.Jobs, [RClis, DataIncs, Usuars])
	If Not Used([LocalBus])
		=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (CarregaCab - LocalBus 1)])
		Return .f.
	EndIf
	Go Top In LocalBus
	If Not Eof([LocalBus])
		ThisForm.getCab002.Value = LocalBus.RClis
*!*			ThisForm.getCab003.Value = Ttod(LocalBus.DataIncs)

*!*			If Not Empty(LocalBus.Usuars)
*!*				ThisForm.poDataMgr.CursorQuery([SigCdUsu], [LocalBus2], [Usuarios], LocalBus.Usuars, [NComps])
*!*				If Not Used([LocalBus2])
*!*					=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (CarregaCab - LocalBus 2)])
*!*					Return .f.
*!*				EndIf
*!*				Go Top In LocalBus2
*!*				If Not Eof([LocalBus2])
*!*					ThisForm.getCab004.Value = Alltrim(LocalBus.Usuars) + [ - ] + Alltrim(LocalBus2.NComps)
*!*				EndIf
*!*				Use In LocalBus2
*!*			EndIf
	EndIf
	Use In LocalBus
EndIf

If Not Empty(ThisForm.Contas)

LParameters pFrm, pCta

With ThisForm
	.ParentForm = pFrm

	Set Datasession To .ParentForm.DataSessionId

	.pcEscolha = .ParentForm.pcEscolha
	.poDataMgr = .ParentForm.poDataMgr
	.Caption   = [Complementos : ] + Alltrim(Proper(TprMvCab.Dopes))
	.Contas    = pCta
	
	.poDataMgr.CursorQuery([SigCdEmp], [LocalEmps], [Cemps], TprMvCab.Emps, [DivNotas])
	If Not Used([LocalEmps])
		=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (CursorQuery - LocalEmps)])
		Return .f.
	EndIf
	Select LocalEmps
	Go Top In LocalEmps
	lnEmp = Iif(Eof([LocalEmps]), 1, LocalEmps.DivNotas)
	Use In LocalEmps
	
	.CarregaCabs()
	.CarregaTits()
	.ChecaObrigs()
	.CompletaCpo()
	.MontaGrades()

	lcQuery = [Select a.IClis, a.RClis ] + ;
			    [From SigCdCli a, SigCdGcr b ] + ;
			   [Where a.Grupos = b.Codigos And ] + ;
			   		 [b.Coletors = 8] && Atendimento

	If (.poDataMgr.SqlExecute(lcQuery, [crEquipes]) < 1)
		=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (crEquipes)])
		Return .f.
	EndIf
	Select crEquipes
	Index On IClis Tag IClis
	Index On RClis Tag RClis
	
	With .pgfPagina.pgPagina3
		Do Case
			Case (lnEmp = 2) && Um
				.cntEmp1.Visible = .t.
				.cntEmp2.Visible = .f.
				
			Case (lnEmp = 3) && Dois
				.cntEmp2.Left = .cntEmp1.Left 
				.cntEmp1.Visible = .f.
				.cntEmp2.Visible = .t.
				
			Case (lnEmp = 4) && Nenhuma
				.cntEmp1.Visible = .f.
				.cntEmp2.Visible = .f.
		EndCase
	EndWith
		
	.pgfPagina.ActivePage = .pgfPagina.pgPagina1.PageOrder 
EndWith	
* Fim **************************************************************************************
ENDPROC
