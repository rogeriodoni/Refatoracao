PROCEDURE gerasenha
Lparameter _data
pridig = 0
segdig = 0
_var = substr(dtoc(_data),1,2)+substr(dtoc(_data),4,2)+substr(dtoc(_data),7,2)
_var = _var + _var
_var1 = transf(_var + "00" ,"@R 99999999/9999-99")
ret = fValidarCNPJ(_var1)
_senha = str(pridig,1) + str(segdig,1)
pridig = 0
segdig = 0
_var1 = padl(alltrim(str(val(_var) + val(_senha),12)),12,"0")
_var1 = transf(_var1 + "00" ,"@R 99999999/9999-99")
_senha = _senha + str(pridig,1) + str(segdig,1)
Return _senha
ENDPROC

PROCEDURE Valid
local senha,resp

If Thisform.Tipo = 'L' && 20/04/2016 - Senha livre - Checa se a senha digita é válida e já foi utilizada 
	lcQuery = [Select EmpDopNums, CidChaves From SigPrSnd Where SenhaDs = '] + Alltrim(This.Value) + [']
	Thisform.podatamgr.SqlExecute(lcQuery,[TmpPrSnd])
	If Eof([TmpPrSnd])
		=Messagebox([Senha inválida.],48,[Senha livre])
		ThisForm.Zsenha=.F.
		thisform.release()
		Return .f.
	Else
		If !Empty(TmpPrSnd.EmpDopNums)
			=Messagebox([Senha já utilizada.],48,[Senha livre])
			ThisForm.Zsenha=.F.
			thisform.release()
			Return .f.				
		Endif
	Endif
	lcEDN = TprMvCab.Emps + TprMvCab.Dopes + Str(TprMvCab.Numes,6)
	lcQuery = [Update SigPrSnd Set EmpDopNums = '] + lcEDN + [' Where CidChaves = '] + TmpPrSnd.CidChaves + [']
	Thisform.podatamgr.SqlExecute(lcQuery,[])
	ThisForm.Zsenha = Usuar
	thisform.release()		
Else
	select TmpUsu
	Senha=fCriptografar(TmpUsu.senhas)
	If Padr(this.value,10)<>Padr(senha,10)
		resp=messagebox('Senha Conflitante. Deseja Continuar?',4+256,'Aviso')
		if resp=7
			ThisForm.Zsenha=.F.
			thisform.release
		else
			return 0
		endif
	Else
		If (ThisForm.Tipo = [X])
			ThisForm.Zsenha = ThisForm.Usuario.Value 
		Else
			ThisForm.Zsenha = TmpUsu.Usuarios + ThisForm.Get_motivo.Value
		EndIf
		
		Thisform.release
	Endif
Endif
ENDPROC

PROCEDURE Valid
local lcod,lfan,codcor,xobj,totite

If Not IsEmpty(This.Value)
	lcQuery = [Select a.*, b.Emps ] + ;
				[From SigCdFs a Left Join SigCdFsI b On a.Codigos = b.Codigos ] + ;
			   [Where a.Tipos = '] + ThisForm.Tipo + [' ] + ;
					  Iif(ThisForm.Checkbin, [And a.ChkBins  <> 1 ],[ ]) + ;
					  Iif(ThisForm.Checktbd, [And a.BlqTabDs <> 1 ],[ ]) + ;
			   [Order By a.Codigos]

	If (ThisForm.poDataMgr.SqlExecute(lcQuery, [LocalMdsc1]) < 1)
		=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (LocalMdsc1)])
		Return 0
	EndIf
	Select Distinct Codigos, Descrs From LocalMdsc1 Where Nvl(Emps,Thisform.pemps) = ThisForm.pEmps Into Cursor LocalMdsc ReadWrite
	Select * From LocalMdsc Into Cursor TmpMdsc ReadWrite

	If Not Empty(ThisForm.ClaConta)
		lcSql = [Select b.* From SigCdClc a, SigCdCmt b Where a.Codigos = ']+ThisForm.ClaConta+[' ]+;
				[And a.Cidchaves = b.fkChaves ]
		If (ThisForm.poDataMgr.SqlExecute(lcSql, 'TmpClaMt') < 1)
			=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (TmpClaMt)')
			Return 0
		EndIf
		Select TmpClaMt
		Locate For Tipos = 1
		If Not Eof()
			Select a.* From LocalMdsc a, TmpClaMt b Where a.Codigos = b.codigos And b.Tipos = 1 ;
			Into Cursor TmpMdsc ReadWrite
		Else
			Select TmpClaMt
			Locate For Tipos = 2
			If Not Eof()
				Select * From LocalMdsc Where Codigos Not in (Select Codigos From TmpClaMt Where Tipos = 2 ) ;
				Into Cursor TmpMdsc ReadWrite
			EndIf
		EndIf
	Endif

	Select TmpMdsc
	Index on Codigos Tag Codigos

	If !Seek( This.Value, 'TmpMdsc', 'Codigos' )
		loBrowse = CreateObject( 'fwBuscaSel', 'TmpMdsc', 'codigos', This.Value, 'Cadastro de Motivos' )
		loBrowse.AddColuna( 'Codigos', '', 'Código', .t., 'Codigos' )
		loBrowse.AddColuna( 'Descrs', '', 'Descricao' )
		loBrowse.Show
		This.Value = TmpMdsc.Codigos
	Endif

	If Lastkey() = 27
		ThisForm.Zsenha=.F.
		Thisform.release
	Else
		If ThisForm.somotivo 
			ThisForm.Zsenha = Space(10) + This.Value
			ThisForm.Release
		Endif
	EndIf

Else

	Resp=Messagebox('Motivo não Informado!, Deseja Continuar ? (S/N)',4+256,'')
	If Resp = 7
		ThisForm.Zsenha=.F.
		Thisform.release
	Else
		return 0
	EndIf

EndIf

ENDPROC
