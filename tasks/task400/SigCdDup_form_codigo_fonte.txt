------------------------------------------------------------
* PROCEDURE Pagina.Lista.Grupo_op.Click
------------------------------------------------------------
PROCEDURE Pagina.Lista.Grupo_op.Click
LPARAMETERS opcao

Local lnCodigo, lcCodigo

If !dodefault( Opcao )
	ThisForm.mAtivapagina1
	Return
Endif
* posiciona o cursor de acordo com a opção
Do Case
	Case Inlist(ThisForm.pcEscolha, 'INSERIR', 'PROCURAR')
		thisform.pagina.dados.getseries.setfocus	

	Case ThisForm.pcEscolha = 'ALTERAR'
		Thisform.Pagina.Dados.Text5.setfocus

	Case Inlist( ThisForm.pcEscolha, 'CONSULTAR', 'EXCLUIR' )
		thisform.pagina.dados.grupo_salva.salva.setfocus

Endcase
This.Parent.Parent.Dados.Text40.value = ''

*!* VER INSTRUÇÕES DESTA FUNÇÃO NO BOTÃO SALVAR...
If Not Empty(crSigCnFDu.ctamfolhas) 
	iPosSeparador = At('/',crSigCnFDu.ctamfolhas)
	If iPosSeparador = 1
		lcTamFolhaTmp = Substr(crSigCnFDu.ctamfolhas,2)
		lcDescricao   = Substr(crSigCnFDu.ctamfolhas,At('/',lcTamFolhaTmp)+1)
		This.Parent.Parent.Dados.Text40.Value = lcDescricao
	EndIf
	If iPosSeparador = 0
		This.Parent.Parent.Dados.Text40.value = ''
	Else
*!*     Pesquisa se existe Tamanho da Folha configurado
		If At('/',substr(crSigCnFDu.ctamfolhas,iPosSeparador+1)) > 0
			This.Parent.Parent.Dados.Text40.value = Substr(substr(crSigCnFDu.ctamfolhas,iPosSeparador+1),At('/',substr(crSigCnFDu.ctamfolhas,iPosSeparador+1))+1)
		Else
			This.Parent.Parent.Dados.Text40.value = ''
		EndIf
	EndIf 
EndIf 
ThisForm.Pagina.Dados.Refresh

ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
If ThisForm.pcEscolha = 'INSERIR'
   IF IsEmpty(ThisForm.Pagina.Dados.Getseries.Value)
	  MessageBox( 'Obrigatório Informar o Código.', 48, '' )
	  ThisForm.Pagina.Dados.Getseries.SetFocus
	  Return
   ENDIF

	If ThisForm.poDataMgr.ChkRegister('SigCnFDU','series',ThisForm.pagina.dados.getseries.Value)
	  MessageBox( 'Série já cadastrada...', 48, '' )
	  ThisForm.Pagina.Dados.Getseries.SetFocus
      Return
	Endif
Endif

If ThisForm.pcEscolha = 'ALTERAR' or ThisForm.pcEscolha = 'INSERIR'

	Select crSigCnFDu

*!*		 Gravação do Nome da Fonte e Tamanho da Folha Concatenados
*!*		 Text14 = Fonte Padrao
*!*		 Text40 = Tamanho da Folha
*!*	    		Onde:  as 10 primeiras posições é reservado ao código da folha
*!*	    		       e as demais para a descrição
*!*	     concatenando fica:  Text30 + / + SubStr(Text40,1,10) + / + SubStr(Text40,11)    Times New Roman/45/Nota Fiscal
	If Not Empty(crSigCnFDu.ctamfolhas) 
		If At('/',crSigCnFDu.ctamfolhas) = 0
			lcTamFolhas = Alltrim(This.Parent.Parent.Text30.value)+'/'+SubStr(crSigCnFDu.ctamfolhas,1,10)+'/'+SubStr(crSigCnFDu.ctamfolhas,11)
		Else
			lcLocalBarraTamFolha = At('/',Substr(crSigCnFDu.ctamfolhas,At('/',crSigCnFDu.ctamfolhas)+1))
			If lcLocalBarraTamFolha > 0
				lcStringTamFolha = Substr(crSigCnFDu.ctamfolhas,At('/',crSigCnFDu.ctamfolhas)+1)
				lcCodigo    = Alltrim(Substr(lcStringTamFolha,1,At('/',lcStringTamFolha)-1))
				lcDescricao = Alltrim(Substr(lcStringTamFolha,At('/',lcStringTamFolha)+1))
				lcTamFolhas = Alltrim(This.Parent.Parent.Text30.value)+'/'+lcCodigo+'/'+lcDescricao
			Else
				lcTamFolhas = Alltrim(This.Parent.Parent.Text30.value)+'/'
			EndIf
		EndIf
	Else
		lcTamFolhas = Alltrim(This.Parent.Parent.Text30.value)+'/'
	EndIf
    Replace dDatas     With Date(),;
	        cHoras     With Time(),;
	        cUsuarios  With m.Usuar,;
	        ctamfolhas With lcTamFolhas In ('crSigCnFDu')
EndIf 

= DoDefault()

Select crSigCnFDu
This.Parent.Parent.Parent.Lista.Grade.Refresh
ThisForm.mAtivapagina1

ENDPROC
