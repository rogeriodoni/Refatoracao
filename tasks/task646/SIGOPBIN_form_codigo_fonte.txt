LParameters pPag

PROCEDURE Valid
Local lCod,lfan,codcor,xobj,totite

ThisForm.UsuLibs = []

If Empty(This.Value)
	ZSen = .f.
	ThisForm.ZSen = .f.
	ThisForm.Release
	Return .t.
EndIf

lcSql = [Select * From SigOpFpB Where Fpags = ']+ThisForm.Fpagto+[' And CodBins = ']+This.Value+[']
If Thisform.Podatamgr.Sqlexecute(lcSql,'LocalFpagB') < 1
	MessageBox('Falha na Conexão. Tente Novamente!!!',0+16,'')
	zsen = .f.
	Thisform.zsen=.F.
	Thisform.release
Else
	Select LocalFpagB
	If Reccount('LocalFpagB') > 0
		If Between(Dtos(Date()),Dtos(Nvl(LocalFpagB.Datais,Ctod('01/01/1900'))),Dtos(Nvl(LocalFpagB.DataFs,Ctod('31/12/9999'))))
			Thisform.usulibs = This.value
			Thisform.zsen=.T.
		Else
			ThisForm.zSen = .f.
		Endif
	Else
		Thisform.zsen=.F.
	EndIf
EndIf
Thisform.release
Return .t.
* Fim **************************************************************************************
ENDPROC
