PROCEDURE KeyPress
LPARAMETERS nKeyCode, nShiftAltCtrl

Do Case
	Case nKeyCode = -1 && F2
		NoDefault
		ThisForm.Imprimir.Click()

	Case nKeyCode = -2 && F3
		NoDefault
		ThisForm.Grade.Click()
EndCase

------------------------------------------------------------
* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
Parameters poform, _pEmps, _pDopes, _pNumes, _pChkImpDoc, _pBol

ThisForm.Parentform   = poform
ThisForm.pChkImpDoc   = _pChkImpDoc
ThisForm.Boleto 	  = Iif(Type([_pBol])=[N], _pBol, 0)
Thisform.oModem 	  = ThisForm.Parentform.oModemEest
ThisForm.Movimentacao = _pDopes

If (ThisForm.Boleto > 0)
	ThisForm.Caption = [Impressão Global de Boletos]
EndIf
ThisForm.cntSombra.lblSombra.Caption = ThisForm.Caption
ThisForm.cntSombra.lblTitulo.Caption = ThisForm.Caption
lcEDN = _pEmps + _pDopes + Str(_pNumes,6)

lcSql = [Select a.Emps, a.Dopes, a.Numes, a.EmpDopNums, a.Valos, a.ValDevs, b.Docus, c.VenBols ] + ;
		[From SigMvCab a, SigCdOpe b, SigOpCdc c ]+;
		[Where EmpDnCrds = ']+lcEDN+[' And a.Dopes = b.Dopes And a.Dopes = c.Dopes And b.Docus in(1,3,4)]

If ThisForm.Parentform.podatamgr.Sqlexecute(lcSql,'CrTprMvCab') > 0
	Select CrTprMvCab
	Scan
		Scatter Memvar
		m.Obs   = CrTprMvCab.dopes+' '+Str(CrTprMvCab.Numes,6)
		m.Marca = .t.
		Printed = .f.
		Insert into csPgto from Memvar
	EndScan
EndIf
Select csPgto
Go Top	
If Reccount() = 0
	ThisForm.Release
	Return .F.
EndIf

=ThisForm.Parentform.poDataMgr.CursorQuery('SigOpCdc','crSigOpCdc','Dopes',ThisForm.Movimentacao)
ThisForm.btnBoleto.Visible = Inlist(crSigOpCdc.VenBols, 1, 3)	

With ThisForm
	With .Grd 
		.RecordSource          = 'csPgto'
		.RecordSourceType      = 1
		.Column1.ControlSource = 'csPgto.marca'
		.Column2.ControlSource = 'csPgto.Obs'
		.Column3.ControlSource = 'csPgto.Valos'
		.Refresh
	EndWith	
EndWith

ENDPROC

PROCEDURE Click
Local lOk
lOk = .f.
If Reccount("csPgto") <=0
	MessageBox("Não existem observações a serem auteticadas",16,"")	
	ThisForm.Sair.Setfocus
EndIf

Select csPgto
Scan
	If csPgto.marca
		If (CsPgto.docus = 1)
			=ThisForm.Parentform.Podatamgr.cursorquery('SigCdOpe','CrSigCdOpe','dopes',TprMvCab.Dopes)
			=ThisForm.ParentForm.PodataMgr.CursorQuery('SigOpCdc','CrSigOpCdc','Dopes',TprMvCab.Dopes)
			DO SigPrIdc with CsPgto.Emps, CsPgto.Dopes, CsPgto.Numes,CsPgto.valdevs, crSigOpCdc.QtdDocs,.F.,.F.,.F.,.F.,crSigOpCdc.TpImpDocs,.F.,Thisform,ThisForm.pChkImpDoc
		EndIf

		If InList(CsPgto.Docus, 3, 4)
			Do Form SigReIfx With CsPgto.Emps, CsPgto.Dopes, CsPgto.Numes, CrSigOpCdc.TpImpDocs,ThisForm.pChkImpDoc
		EndIf

		Select csPgto
		Replace csPgto.marca with .f., csPgto.Printed with .t.
	EndIf
EndScan
ENDPROC

------------------------------------------------------------
* PROCEDURE Click
------------------------------------------------------------
PROCEDURE Click
Local lOk, lnTpB
lOk = .f.

If Reccount("csPgto") <=0
	MessageBox("Não existem observações a serem auteticadas",16,"")	
	ThisForm.Sair.Setfocus
EndIf

Select csPgto
Scan
	lnTpB = 0
	=ThisForm.Parentform.poDataMgr.CursorQuery('SigOpCdc','crSigOpCdc','Dopes',ThisForm.Movimentacao)

	Do Case 
		Case !EMPTY(substr(crSigOpCdd.ConfigGers,233,2))	&& mena - 30/06/2016 - Pega tipo de boleto para Rosana
			lnTpB = VAL(substr(crSigOpCdd.ConfigGers,233,2))
		
		Case (crSigOpCdc.BolItes = 3) && SubNiveis
			lnTpB = 11

		Case (crSigOpCdc.BolItes = 4) && Detalhado
			lnTpB = 13

		Case (crSigOpCdc.BolItes = 5) && Romaneio
			lnTpB = 14

		Case (crSigOpCdc.BolItes = 7) && Controle Interno
			lnTpB = 17

		Case (crSigOpCdc.BolItes = 6) && Vendedor
			lnTpB = 16

		Case (crSigOpCdc.BolItes = 8) && mena - 03/11/2014 - Duplicata/Carne
			lnTpB = 18

		Otherwise && Sim / Não
			lnTpB = 3
	EndCase
	
	If (lnTpB > 0)
		Do Form SigOpIbo With Thisform.Parentform, lnTpB,,csPgto.Emps, csPgto.Dopes, csPgto.Numes, 0, crSigOpCdc.qvias, crSigOpCdc.qtdbols
	EndIf
	*Do SigBlBol With lnTpB, [], csPgto.Emps, csPgto.Dopes, csPgto.Numes, .f., 1, 1
EndScan
ENDPROC
