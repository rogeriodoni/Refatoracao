PROCEDURE Init
Private pFpg

pFpg = Replicate(Chr(254), 12)

lcQryFPag = [Select Distinct a.FPags, a.Descrs, b.FPags as TemDados ] + ;
		      [From SigOpFp a ] + ;
		  	  [Left Join SigCnFBl b ] + ;
		    	[On a.FPags = b.FPags ] + ;
		     [Order By a.FPags]

lcQryCfgBl = [Select * From SigCnFBl Where FPags = ?pFpg]

With ThisForm
	.poDataMgr = CreateObject('fSqlConector', ThisForm.Name)

	If (.poDataMgr.pnIdconn > 0)
		With .poDataMgr
			.AddCursor('SigOpFp', 'FPags', 'crSigOpFp', '', ThisForm.Pagina.Lista.Grade, lcQryFPag)
			.pfSqlTabela(1).pColuna('FPags',  '', '', 'Códigos',   094, .t.)
			.pfSqlTabela(1).pColuna('Descrs', '', '', 'Descrição', 220, .t.)

			.AddCursor('SigCnFBl', 'cIdChaves', 'crSigCnFBl', '', '', lcQryCfgBl)
		EndWith

		=DoDefault()

		.Pagina.Lista.Grade.SetAll('DynamicBackColor', 'Iif(IsEmpty(crSigOpFp.TemDados), Rgb(255,255,187), RGB(255,255,255))','Column')
		.Pagina.Lista.Grade.Column1.SetFocus
	Else
		=Messagebox('Impossível Efetuar Conexão Com o Servidor de Banco de Dados...', 0+48, 'CreateObject')
		Return .f.
	EndIf
EndWith
* Fim **************************************************************************************
ENDPROC
