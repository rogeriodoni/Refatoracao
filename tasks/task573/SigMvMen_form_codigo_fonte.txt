PROCEDURE Init
Lparameters pDopes

Local llMemo 

With ThisForm
	.poDatamgr = CreateObject('fSqlConector')
	= DoDefault()
	.LockScreen = .f.
EndWith

=Thisform.PodataMgr.Cursorquery('SigOpCdd','CrSigOpCdd','Dopes',pDopes)

ThisForm.Caption = 'Mensagem da Movimentação: '+pDopes

ThisForm.cntSombra.lblSombra.Caption = ThisForm.Caption
ThisForm.cntSombra.lblTitulo.Caption = ThisForm.Caption

ThisForm.Container1.GetMemo.Value = Alltrim(Nvl(CrSigOpCdI.Mensagem,[]))
ThisForm.Ok.SetFocus 
ThisForm.Container1.Refresh

ENDPROC
