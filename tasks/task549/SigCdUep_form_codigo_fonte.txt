* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
With ThisForm
	.poDataMgr = CreateObject([fSqlConector], .Name)

	If (.poDataMgr.pnIdconn > 0)
		With .poDataMgr
			.AddCursor([SigCdUep], [cIdChaves], [crSigCdUep], [], ThisForm.Pagina.Lista.Grade)
			.pfSqlTabela(1).pColuna([Usuarios],  [], [], [Usuario],    080, .t.)
			.pfSqlTabela(1).pColuna([Emps], [], [], [Empresa], 80, .t.)
		EndWith

		=DoDefault()

		.Pagina.Lista.Grade.Column1.SetFocus
	Else
		=Messagebox([Impossível Efetuar Conexão Com o Servidor de Banco de Dados...], 0+48, [CreateObject])
		Return .f.
	EndIf
EndWith
* Fim **************************************************************************************
ENDPROC

PROCEDURE msv_procurar

EndIf
* Fim **************************************************************************************
ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Lista.Grupo_op.Click
------------------------------------------------------------
PROCEDURE Pagina.Lista.Grupo_op.Click
Parameters lnOpcao

If Not DoDefault(lnOpcao)
	ThisForm.mAtivaPagina1
	Return .f.
EndIf

Do Case
	Case InList(ThisForm.pcEscolha, [INSERIR], [PROCURAR])
		 ThisForm.Pagina.Dados.getUsuarios.SetFocus
		 
	Case InList(ThisForm.pcEscolha, [ALTERAR])
		 ThisForm.Pagina.Dados.getEmpresa.SetFocus

	Otherwise
		 ThisForm.Pagina.Dados.Grupo_Salva.Salva.SetFocus
EndCase
* Fim **************************************************************************************
ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
If InList(ThisForm.pcEscolha, [INSERIR], [ALTERAR])
	If Empty(ThisForm.Pagina.Dados.getUsuarios.Value)
		=Messagebox([Usuario Inválido!!!], 48, [Atenção!!!])
		ThisForm.Pagina.Dados.getUsuarios.SetFocus()
		Return .f.
	EndIf

	If Empty(ThisForm.Pagina.Dados.GetEmpresa.Value)
		=Messagebox([Empresa Inválida!!!], 48, [Atenção!!!])
		ThisForm.Pagina.Dados.GetEmpresa.SetFocus()
		Return .f.
	EndIf
	
	If (ThisForm.pcEscolha = [INSERIR])
		lcQuery = [Select Usuarios From SigCdUep ] + ;
				   [Where Usuarios = '] + Padr(ThisForm.Pagina.Dados.getUsuarios.Value, 3) + [']

		If (ThisForm.poDataMgr.SqlExecute(lcQuery, [LocalBus]) < 1)
			=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (LocalBus 1)])
			Return .f.
		EndIf
		Go Top In LocalBus
		If Not Eof([LocalBus])
			=Messagebox([Usuarios Já Cadastrado!!!], 48, [Atenção!!!])
			ThisForm.Pagina.Dados.getUsuarios.SetFocus()
			Return .f.
		EndIf
	EndIf
EndIf

If (ThisForm.pcEscolha = [INSERIR])
	Replace cIdChaves With fUniqueIds() in crSigCdUep 
EndIf 


=DoDefault()

ThisForm.mAtivaPagina1
* Fim **************************************************************************************
ENDPROC

Local loLista

If Not Empty(This.Value)
	loLista = CreateObject('fwBuscaExt', ThisForm.poDataMgr.pnIdConn, 'SigCdUsu', ;
						   'crListaRemota', 'Usuarios', This.Value, 'Seleção', 500,.f.,[Not cAtivos='N'])

		loLista.Show()
	EndIf
	This.Value = Iif(Lastkey()=27, '', crListaRemota.Usuarios)
	Use In crListaRemota
EndIf
