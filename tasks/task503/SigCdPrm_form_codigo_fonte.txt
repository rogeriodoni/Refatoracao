Local lcSQLProm

	.pcTipo = m.lcTipo

		EndWith

		= DoDefault()

		Select csSigPrPmi
		Set Order To CPros
		Go Top

		With .Pagina.Dados.pgGrades.pgProdutos.grd_produtos
			.RecordSource            = 'csSigPrPmi'

PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
Select crSigPrPmc

If (ThisForm.pcEscolha = 'INSERIR')
	If Empty(CrSigPrPmc.Promos)
		MessageBox('Promoção Inválida !!!',16,'Campo não preenchido')
		ThisForm.Pagina.Dados.Get_Promos.SetFocus
		Return(.F.)
	EndIf
	If Used('TmpProm')
		Use In TmpProm
	EndIf
	lStrQuery = [Select Promos,Cods From SigPrPmc Where Promos = ']+CrSigPrPmc.Promos+[' And Cods = ']+CrSigPrPmc.Cods+[' ]
	If ThisForm.poDataMgr.SqlExecute(lStrQuery,'TmpProm') < 1
		MessageBox('Favor reinicializar o processo.',16,'Falha na Conexão')
		Return(.F.)
	EndIf

	Go Top
	If !Eof()
		MessageBox('Promoção Já Cadastrada !!!',16,'')

		Return(.F.)
	EndIf

	Select CPros, Recno() as Registro ;
	  From csSigPrPmi ;
	 Where Empty(CPros) Or IsEmpty(Datas) ;
	  Into Cursor crErros
	Select crErros
	Go Top In crErros
	If Not Eof([crErros])
		If MessageBox([O Produto "] + Alltrim(crErros.CPros) + [" Está Com Informações Incompletas] + Chr(13) + ;
					  [e Será Descartado. Deseja Corrigir as Informações?], 4+32+256, [Confirmação de Gravação]) = 6

			ThisForm.Pagina.Dados.pgGrades.pgProdutos.grd_produtos.Col_cpros.Txt_cpros.SetFocus

	Select Colecoes From csSigPrPmv Into Cursor crBuscaV
	Select CPros From csSigPrPmi Into Cursor crBuscaI
	If (Reccount('crBuscaV') + Reccount('crBuscaI') < 1)
		=MessageBox('Nenhum Produto ou Grupo de Venda Foi Informado!!!', 0+16, 'Atenção!!!')
		ThisForm.Pagina.Dados.Get_Promos.SetFocus
		Return 0
	EndIf

	Update csSigPrPmi Set Promos = crSigPrPmc.Promos Where Empty(Promos)
	Update csSigPrPmi Set PromoPro = Promos + CPros Where Empty(PromoPro)
	Update csSigPrPmi Set cIdChaves = fUniqueIds()  Where Empty(cIdChaves)

	Update csSigPrPmv Set Promos = crSigPrPmc.Promos Where Empty(Promos)
	Update csSigPrPmv Set cIdChaves = fUniqueIds()  Where Empty(cIdChaves)

	Select crSigPrPmc
	If Empty(crSigPrPmc.cIdChaves)
		Replace cIdChaves With fUniqueIds() In crSigPrPmc
	EndIf

EndIf

Go Top

=DoDefault()

ThisForm.mAtivaPagina1

PROCEDURE Valid
Local loLista, lnCBar

lnCBar = 0
lcCdPro = This.Value

With ThisForm.Pagina.Dados.pgGrades.pgProdutos.grd_Produtos
	If Not Empty(This.Value)
		Set Decimals To 0
		If fVerificarBarras(This.Value)
			lStrQuery = [Select CPros, CBars From SigOpEtq Where CBars = ] + Alltrim(This.Value)
			If ThisForm.poDataMgr.SqlExecute(lStrQuery,'TmpEti') < 1
				=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão')
				Set Decimals To 2
				Return .f. 
			EndIf

			Select TmpEti
			Go Top
			If Not Eof()
				lcCdPro = TmpEti.CPros
				lnCBar  = TmpEti.CBars
			Else
				m.loLista = CreateObject('fwBuscaExt',ThisForm.poDataMgr.pnIdConn,'SigOpEtq','CrEtiquetas','CBars',Val(This.Value),'Etiquetas',1000)
				If !(m.loLista.plAchouRegistro)
					m.loLista.mAddColuna('CBars','99999999','Etiqueta')
					m.loLista.mAddColuna('CPros','','Produto')
					m.loLista.Show()
				EndIf
				If LastKey() <> 27 And Used('CrEtiquetas')
					lcCdPro = Evaluate('CrEtiquetas.CPros')
					lnCBar  = Evaluate('CrEtiquetas.CBars')
				Else
					lcCdPro = ''
					lnCBar  = 0
				EndIf
			EndIf
			Replace CBars With lnCBar In csSigPrPmi
		Else
			loLista = CreateObject('fwBuscaExt', ThisForm.poDataMgr.pnIdConn, 'SigCdPro', ;
								   'crListaRemota', 'CPros', This.Value, 'Seleção', 1000)

			If Not loLista.plAchouRegistro
				loLista.mAddColuna('CPros', '', 'Código')
				loLista.mAddColuna('DPros', '', 'Descrição')
				loLista.Show()
			EndIf
			If (Lastkey()<>27)
				This.Value = crListaRemota.CPros
				lcCdPro = This.Value
			EndIf
			Use In crListaRemota
		EndIf

		If Not Empty(lcCdPro) And (lcCdPro <> ThisForm.AntCPro)
			If ThisForm.poDataMgr.CursorQuery('SigCdPro',,'CPros',lcCdPro)
				If CqSigCdPro.situas = 2
					Messagebox('Produto Inativo !!!',32,'')
				EndIf
				.Col_CPros.Txt_CPros.Value = Evaluate('CqSigCdPro.CPros')
				.Col_DPros.Txt_DPros.Value = Evaluate('CqSigCdPro.DPros')
				.Column4.Text1.Value       = lnCBar
			EndIf
			If Empty(.Column5.Text1.Value)
				.Column5.Text1.Value = Date()
			EndIf

			Select CsSigPrPmi
			lnRec = Recno()

			Select CPros ;
			  From csSigPrPmi ;
			 Where CPros = lcCdPro And ;
			 	   Recno() <> lnRec ;
			  Into Cursor crBusca

			Go Top In crBusca
			If Not Eof('crBusca')
				=Messagebox('Produto Já Registrado Nesta Promoção!!!',32,'')
				.Col_CPros.Txt_CPros.Value = ''
				.Col_DPros.Txt_DPros.Value = ''
				.Column4.Text1.Value       = 0
				.Column5.Text1.Value       = Ctod('')
				.Column6.Check1.Value      = 1
				Set Decimals To 2
				Return .f.
			EndIf
		EndIf
	EndIf
	.Col_CPros.Txt_CPros.Refresh
	.Col_DPros.Txt_DPros.Refresh
	.Column4.Text1.Refresh
	.Column5.Text1.Refresh
	.Column6.Check1.Refresh
	.Refresh
EndWith
If Not Empty(csSigPrPmi.CPros) And Empty(csSigPrPmi.Datas)
	Replace Datas With Date() In csSigPrPmi
EndIf
Set Decimals To 2
* Fim **************************************************************************************
ENDPROC

PROCEDURE Valid
lcCdPro = ''

With ThisForm.Pagina.Dados.pgGrades.pgGrpVenda.grdColecoes
	If Not Empty(This.Value)
		loLista = CreateObject('fwBuscaExt', ThisForm.poDataMgr.pnIdConn, 'SigCdCol', ;
							   'CrListaRemota', 'Colecoes', This.Value, 'Seleção', .t.)

		If Not loLista.plAchouRegistro
			loLista.mAddColuna('Colecoes', '', 'Código')
			loLista.mAddColuna('Descs',    '', 'Descrição')
			loLista.Show()
		EndIf
		If (Lastkey()<>27)
			This.Value = crListaRemota.Colecoes
			lcCdPro = This.Value
		EndIf
		Use In crListaRemota

		If Not Empty(lcCdPro) And (lcCdPro <> ThisForm.AntColecoes)
			If Empty(.Column2.Text1.Value)
				.Column2.Text1.Value = Date()
			EndIf

			Select CsSigPrPmv
			lnRec = Recno()

			Select Colecoes ;
			  From csSigPrPmv ;
			 Where Colecoes = lcCdPro And ;
			 	   Recno() <> lnRec ;
			  Into Cursor crBusca

			Go Top In crBusca
			If Not Eof('crBusca')
				=Messagebox('Coleção Já Registrada Nesta Promoção!!!',32,'')
				.Column1.Text1.Value = []
				.Column2.Text1.Value = Ctod([])
				Set Decimals To 2
				Return .f.
			EndIf
		EndIf
	EndIf
	.Column1.Text1.Refresh
	.Column2.Text1.Refresh
	.Refresh
EndWith
Set Decimals To 2
* Fim **************************************************************************************
ENDPROC

PROCEDURE When

* Registro: 28

* PROCEDURE Valid
------------------------------------------------------------
PROCEDURE Valid
Private lcPro
If !Empty(This.Value)
	lcPro = This.Value
	If ThisForm.pcEscolha = 'INSERIR'
		If ThisForm.poDataMgr.CursorQuery('SigPrPmc',,'Promos',lcPro)
			MessageBox('Promoção Já Cadastrada !!!',16,'')
			Return(.F.)
		EndIf

		Zap In CsSigPrPmv

		lStrQuery = [Select * From SigPrPmv Where Promos = ?lcPro]
		If ThisForm.poDataMgr.SQLExecute(lStrQuery,'TmpPromV') < 1
			MessageBox('Favor reinicializar o processo !!!',16,'Falha na Conexão (TmpPromV)')
			Return(.F.)
		EndIf
		Select TmpPromV
		Scan
			Scatter MemVar Memo
			Insert Into CsSigPrPmv From MemVar
			Select TmpPromV
		EndScan
		Select CsSigPrPmv
		Go Top
		If Eof()
			Insert Into CsSigPrPmv (Promos) Values (lcPro)
		EndIf

		Zap In CsSigPrPmi

		lcPro = CrSigPrPmc.Promos
		lStrQuery = [Select a.*, b.DPros From SigPrPmi a, SigCdPro b Where Promos = ?lcPro And b.CPros = a.CPros ]
		If ThisForm.poDataMgr.SQLExecute(lStrQuery,'TmpPromI') < 1
			MessageBox('Favor reinicializar o processo !!!',16,'Falha na Conexão (TmpPromI)')
			Return(.F.)
		EndIf
		Select TmpPromI
		Go Top
		Scan
			Scatter MemVar Memo
			Insert Into CsSigPrPmi From MemVar
			Select TmpPromI
		EndScan
		Select CsSigPrPmi
		Go Top
		If Eof()
			Insert Into CsSigPrPmi (Promos, Vendavels) Values (lcPro, 1)
		EndIf
		Select CsSigPrPmi
		Go Top
		If InList(ThisForm.pcEscolha,'INSERIR','ALTERAR')
			Set Order To
		Else
			Set Order To CPros
		EndIf
		Go Bottom
	EndIf
EndIf
ENDPROC

* PROCEDURE When
------------------------------------------------------------
PROCEDURE When
Return InList(ThisForm.pcEscolha, 'INSERIR', 'PROCURAR')
ENDPROC
