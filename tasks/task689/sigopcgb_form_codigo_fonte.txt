* PROCEDURE imagem

PROCEDURE imagem
Parameter pTip

Local lcTip, lcBrJpg, lcEmp, lcBar, lcCPr, lcDPr, lcFor, lcRef

lcTip = pTip  && I=Imprimir; V=Visualizar
lcEmp = CsSigCdEmp.cemps + Iif(!Empty(CsSigCdEmp.razas),' - ' + CsSigCdEmp.razas,'')
lcBar = ThisForm.GetBarras.Value
lcCPr = ThisForm.GetCProduto.Value
lcDPr = ThisForm.GetDProduto.Value
lcFor = ThisForm.GetCFornecedor.Value
lcRef = ThisForm.GetReferencia.Value

lcBrJpg = Addbs(JustPath(ThisForm.poDataMgr.GetDatabase())) + 'BrJpg.JPG'
If !File(lcBrJpg)
	MessageBox('Está Faltando .JPG Branco !!!',0+48,'')
	Return 0
EndIf

Create Cursor CsCabecalho (empresa c(80),codbarras n(14),cpros c(14),dpros c(40),fornecedor c(10),referencia c(10))
Insert Into CsCabecalho (empresa,codbarras,cpros,dpros,fornecedor,referencia) ;
				Values (lcEmp,lcBar,lcCPr,lcDPr,lcFor,lcRef)

lcFigJpg = Addbs(JustPath(ThisForm.poDataMgr.GetDatabase())) + 'BrJpg.JPG'
Create Cursor CsBranco (branco m)
Select CsBranco
Append Blank
Append Memo Branco From (lcFigJpg) OverWrite

lcArquivo = Sys(2023) + [\] + Sys(2015) + [.jpg]

Clear Resources
If Not Empty(TmpPro.FigJpgs) And Not IsNull(TmpPro.FigJpgs)
	lcFoto=Strconv(Strtran(Strtran(Strtran(TmpPro.FigJpgs,[data:image/png;base64,],[]),[data:image/jpeg;base64,],[]),[data:image/jpg;base64,],[]),14)
	=StrToFile(lcFoto, lcArquivo)
Else
	=StrToFile(csBranco.branco, lcArquivo)
EndIf

If lcTip = 'I'
	Report Form SIGOPCGB To Print Prompt NoConsole
Else
	Report Form SIGOPCGB Preview NoConsole
EndIf
ENDPROC

* PROCEDURE Valid
------------------------------------------------------------
PROCEDURE Valid
Local lnCodBars, lcProd 
lcProd = []

If Not Empty(This.Value) And (This.peantvalue # This.Value)
	lnCodBars = This.Value
	
	lStrQuery = [Select a.CPros,a.CBars, b.cbars_old ] + ;
		[From SigOpEtq a ] + ;
		[left join sigopeta b on a.cpros = b.cpros and a.cbars = b.cbars_new ] + ;
		[Where (a.CBars = ]+Alltrim(Str(lnCodBars,14))+[ or b.cbars_old = ]+Alltrim(Str(lnCodBars,14))+[) ]
	If Thisform.poDataMgr.SqlExecute(lStrQuery,'CrEtiqs') < 1
		Messagebox('Favor reinicializar o processo !!!',16,'Falha na Conexão (CrEtiqs)')
		Return 0
	Endif
	Select CrEtiqs
	Go Top
	If Reccount([CrEtiqs])>0
		lnCodBars = CrEtiqs.CBars
		This.Value = lnCodBars
		ThisForm.GetBarra.Value = CrEtiqs.cbars_old
	Else
		lStrQuery = [Select CPros, CBars, 0 as cbars_old ] + ;
				    [From SigCdPro ] + ;
				   [Where CBars = ] + Alltrim(Str(lnCodBars,14)) + [ ] + ;
				   [Order By Cpros ]

		If ThisForm.poDataMgr.SqlExecute(lStrQuery,'CrEtiqs') < 1
			=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (CrEtiqs)')
			Return 0
		EndIf
		Select CrEtiqs
		Go Top
		If Reccount([CrEtiqs])>0
			lnCodBars = CrEtiqs.CBars
			This.Value = lnCodBars
			ThisForm.GetBarra.Value = CrEtiqs.cbars_old
			lcProd = CrEtiqs.Cpros
		EndIf 
	EndIf 
	
	If !Empty(lnCodBars)
		ThisForm.Processamento(lcProd)
	EndIf
EndIf
ENDPROC

		EndIf
	Else
		If ThisForm.Podatamgr.ChkRegister('SigCdNec','EmpDnPs',Emps+Dopes+Str(numes,10))

						'Posicione em uma Operação válida na lista !!!',0+48,'')
		EndIf
	EndIf
Else
	Messagebox('Selecione uma Etiqueta em uma das listas para visualizar o Documento !!!',0+48,'')

	.Cnt_Correcao.GrdCorrecao.RecordSource = ''
	If Used('CrSigEtHst')

			.RecordSource = 'CrSigEtHst'
