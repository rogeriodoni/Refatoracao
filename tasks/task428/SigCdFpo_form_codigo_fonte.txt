* PROCEDURE Valid
------------------------------------------------------------
PROCEDURE Valid
Local lcFpags, lnRecNo, loBrowse

If Not IsEmpty(This.Value)
	loBrowse = CreateObject('fwBuscaInt','crSigOpFp','FPags',This.Value,'Cadastro de Condição de Pagamento',.t.)
	loBrowse.mAddColuna('FPags','','Código',.t.)
	loBrowse.mAddColuna('Fparcs','99','Parc',.f.)
	loBrowse.Show		
	This.Value = Iif( Lastkey() == 27, '', crSigOpFp.FPags )
	This.Refresh
	
	lcFPags = This.Value
	If Not IsEmpty(lcFPags)
		If CrSigOpFp.Situas = [S] And Not (CrSigCdOpe.Caixas=1 And CrSigCdOpe.Copers=1)
			MessageBox('Condição de Pagamento não permitida para esta Operação!!!',0+16,'')
			This.Value = ' '
			Return .f.
		Endif
		Select crSigOpFpO
		If Not Eof() Or Bof()
			SELECT crSigOpFpo
			lnRecNo = RecNo()
		Else 
			lnRecNo = 0
		ENDIF
*!*			Set Step On 
*!*			SELECT crSigOpFpo
*!*			LOCATE FOR Dopes = crSigCdOpe.Dopes .and. FPags = lcFpags
*!*			If ! EOF()
*!*				Skip
*!*				If crSigOpFpO.Dopes + crSigOpFpO.FPags = crSigCdOpe.Dopes + lcFpags 
*!*					This.Value = ''
*!*					This.Refresh
*!*					Delete
*!*					Skip
*!*					Skip -1
*!*					MessageBox('Condição de Pagamento já Digitada ... ',48,'')
*!*				EndIf
*!*			Endif
		Select crSigOpFpO
		If lnRecNo # 0
			Goto (lnRecNo)
		Else
			Goto Bottom
		EndIf
		This.Refresh
	EndIf
EndIf
ENDPROC

PROCEDURE Click
Local m.lnErro1

Select FPags, sum(1) as TT From crSigOpFpo Where Dopes = crSigCdOpe.Dopes GROUP by 1 Into cursor Totais
Select Totais
Scan
	If TT > 1
		MessageBox('Condição de Pagamento '+Alltrim(Totais.FPags)+' Digitada em Duplicidade!!!!',0+16,'Atenção')
		Return .f.
	EndIf
EndScan

If Thisform.cmdInserir.Visible
	Select crSigOpFpo
	Scan
		If IsEmpty(crSigOpFpo.FPags)
			Delete
		Else
			If IsEmpty(crSigOpFpo.CidChaves)
				Replace CidChaves With fUniqueIds()
			EndIf
		EndIf
	EndScan
ENDIF

m.lnErro1 = ThisForm.ParentForm.poDatamgr.pnIdConn

With ThisForm.ParentForm
	If m.lnErro1>0 
		m.lnErro1 = Iif(.poDataMgr.UpDate('crSigOpFpo'),1,-1)
	EndIf 
	If m.lnErro1>0 
		m.lnErro1 = .poDataMgr.Commit()
	EndIf 
	If Not m.lnErro1>0 
		.poDataMgr.RollBack()
	EndIf 
EndWith 
If Not m.lnErro1>0 
	= MessageBox('Erro na Gravação dos Dados do Cadastro de Operações.',48)
EndIf 
Thisform.ParentForm.Enabled = .T.
Thisform.mCtrlBotoes(.T.)
ThisForm.Release
ENDPROC
