------------------------------------------------------------
* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
LParameters _GetConta, _GetMoeda, _Parent

ThisForm.cntSombra.lblSombra.Caption = ThisForm.Caption
ThisForm.cntSombra.lblTitulo.Caption = ThisForm.Caption

_GetConta = .f.

ThisForm.ParentForm = _Parent

This.Grade.Visible = _GetConta
This.Get_Grupo.Visible = _GetConta
This.Get_Conta.Visible = _GetConta
This.Get_dConta.Visible = _GetConta
This.SayGrupo.Visible = _GetConta
This.SayConta.Visible = _GetConta

This.Get_cd_moeda.Visible = _GetMoeda
This.Get_ds_moeda.Visible = _GetMoeda
This.SayMoeda.Visible = _GetMoeda

If !_GetMoeda 
	This.Height = This.Height - 25
Else
	Select TmpBaixa
	Scan for Tipos = 'X'
		This.Get_cd_moeda.value = moedas
		=Seek( moedas, 'crSigCdMoe', 'CMoes' )
		This.Get_ds_moeda.value = crSigCdMoe.dmoes
		exit
	Endscan
Endif
If !_GetConta And _GetMoeda
	This.Get_cd_moeda.Top = 50
	This.Get_ds_moeda.top = 50
	This.SayMoeda.top = 54
	This.Height = 100
*!*		This.Confirma.Top = 34
*!*		This.Cancela.Top = 34
Endif
This.Refresh

ENDPROC

PROCEDURE Valid
If !IsEmpty( This.Value ) And !Seek( This.Value, 'crSigCdMoe', 'CMoes' )
	loBrowse = CreateObject( 'fwBuscaSel', 'crSigCdMoe', 'CMoes', This.Value, 'Cadastro de Moedas' )
	loBrowse.AddColuna( 'CMoes', '', 'Código', .t., 'CMoes' )
	loBrowse.AddColuna( 'Dmoes', '', 'Descrição', .t., 'DMoes' )
	loBrowse.Show
	Thisform.Get_Cd_Moeda.Value = Iif( Lastkey() == 27, '', crSigCdMoe.CMoes )
	Thisform.Get_Ds_Moeda.Value = Iif( Lastkey() == 27, '', crSigCdMoe.DMoes )
Else
	If IsEmpty( This.Value )
		Thisform.Get_Cd_Moeda.Value = ''
		Thisform.Get_Ds_Moeda.Value = ''
	Endif
Endif

ENDPROC
