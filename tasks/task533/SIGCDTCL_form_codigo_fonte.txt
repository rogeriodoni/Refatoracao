------------------------------------------------------------
* PROCEDURE msv_procurar
------------------------------------------------------------
PROCEDURE msv_procurar
Local laCampos(2,2)

m.laCampos(1,1) = 'nivels'
m.laCampos(1,2) = thisform.pagina.dados.get_nivel.value
m.laCampos(2,1) = 'classes'
m.laCampos(2,2) = thisform.pagina.dados.get_classe.value
ThisForm.poDataMgr.LocateCursor('crSigCdCss',@m.laCampos)

ENDPROC

------------------------------------------------------------
* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
LPARAMETERS lcacesso

With ThisForm
	.poDatamgr = CreateObject('fSqlConector',ThisForm.Name)
	With .poDatamgr
		.AddCursor('SigCdCss','classes','crSigCdCss','classes',ThisForm.pagina.lista.grade)
		.pfSqlTabela(1).pColuna('classes','','','Classe',100,.t.)
		.pfSqlTabela(1).pColuna('descrs','','','Descrição',319,.t.)
		.pfSqlTabela(1).pColuna('nivels','','','Nível',83,.t.)
		.AddCursor('SigCdGcr','codigos','crSigCdGcr','','')
	EndWith
	= DoDefault()
	.LockScreen = .f.
EndWith
ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Lista.Grupo_op.Click
------------------------------------------------------------
PROCEDURE Pagina.Lista.Grupo_op.Click
LPARAMETERS opcao
select CrSigCdCss
If !dodefault( opcao )
	ThisForm.mAtivapagina1
	Return
Endif
if thisform.pcEscolha='INSERIR' or thisform.pcEscolha='PROCURAR'
	thisform.pagina.dados.get_nivel.value = 1
	thisform.pagina.dados.get_nivel.setfocus
endif

if thisform.pcEscolha='ALTERAR'	
	thisform.pagina.dados.get_descr.setfocus
endif
if thisform.pcEscolha='CONSULTAR' or ;
	thisform.pcEscolha='EXCLUIR'	
	thisform.pagina.dados.grupo_salva.salva.setfocus
EndIf



ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
if thisform.pcEscolha='INSERIR'
	if empty(thisform.pagina.dados.get_nivel.value)
		messagebox('Nível Inválido !!',0+48,'')
		thisform.pagina.dados.get_nivel.setfocus
		return
	endif
	if  .not.len(strtran(thisform.pagina.dados.get_classe.value,' ',''))==val(substr('124',thisform.pagina.dados.get_nivel.value,1))
		messagebox('Classe Inválida !!',0+48,'')
		thisform.pagina.dados.get_classe.setfocus
		return
	endif
	If ThisForm.poDataMgr.ChkRegister('SigCdCss','classes',ThisForm.Pagina.Dados.Get_classe.Value)
		messagebox('Classe já Cadastrada !!',0+48,'')
		thisform.pagina.dados.get_classe.setfocus
		return
	endif
endif

If ThisForm.pcEscolha = 'EXCLUIR'
	lcSql = [Select * From SigCdGcr Where Classes = ']+CrSigCdCss.Classes+[']
	=ThisForm.Podatamgr.Sqlexecute(lcsql,'csClasses')
	Select csClasses
	If Reccount() <> 0
		=messagebox('Classe de Conta em Uso no Cadastro de Grupos !!',0+64,'')
		thisform.pagina.dados.Grupo_Salva.Cancelar.setfocus
		return
	EndIf
EndIf

select CrSigCdCss
=DoDefault()

ThisForm.Refresh
Thisform.mAtivapagina1

ENDPROC

endif

------------------------------------------------------------
* PROCEDURE GotFocus
------------------------------------------------------------
PROCEDURE GotFocus

* MONTAGEM DE MÁSCARA DO CAMPO

MaxDig = 00

Do Case
	Case Thisform.pagina.dados.get_nivel.value=1
         Thisform.pagina.dados.get_classe.InputMask='9'
         Maxdig = 1
	Case Thisform.pagina.dados.get_nivel.value=2
	     Thisform.pagina.dados.get_classe.InputMask='99'
	     Maxdig = 2
	Case Thisform.pagina.dados.get_nivel.value=3
	     Thisform.pagina.dados.get_classe.InputMask='9999'
		 Maxdig = 4
EndCase

* SÓ LIMPA O CAMPO, CASO DIMINUI O TAMANHO DE DÍGITOS POSSÍVEIS

If Len(Alltrim(thisform.pagina.dados.get_classe.Value)) > MaxDig
	With Thisform.pagina.dados.get_classe
		.value=''
		.refresh
	Endwith
Endif

ENDPROC
