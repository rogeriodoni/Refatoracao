PROCEDURE gravardados
If InList(ThisForm.pcEscolha, [INSERIR], [ALTERAR]) And ThisForm.plAlterou	
	If (ThisForm.pcEscolha = [ALTERAR])
		lcQuery = [Delete From SigOpFin Where Dopes = '] + ThisForm.Dopes + [']
		If (ThisForm.poDataMgr.SqlExecute(lcQuery, []) < 1)
			=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (Delete - crSigOpFin)])
			Return .f.
		EndIf
	EndIf

	Zap In crSigOpFin

	lnOrd = 0
	Select crGrdValores
	Scan
		If Not Empty(crGrdValores.Codigos)
			lnOrd = lnOrd + 1
			
			Scatter Memvar Memo
			m.Dopes 	= ThisForm.Dopes
			m.cIdChaves = fUniqueIds()
			m.UsuIncs	= Iif(ThisForm.pcEscolha = [INSERIR] Or Empty(m.UsuIncs),   Usuar, 	    m.UsuIncs)
			m.DatIncs	= Iif(ThisForm.pcEscolha = [INSERIR] Or IsEmpty(m.DatIncs), Datetime(), m.DatIncs)
			m.UsuAlts	= Iif(ThisForm.pcEscolha = [ALTERAR], Usuar, 	  m.UsuAlts)
			m.DatAlts	= Iif(ThisForm.pcEscolha = [ALTERAR], Datetime(), m.DatAlts)
			m.Ordem     = lnOrd
			
			*** mena - 22/04/2014 - incluido para substituir o insert ***
			Select crSigOpFin
			Append Blank
			Gather Memvar Memo
			Select crGrdValores
			**************************************************************
*!*			Insert Into crSigOpFin From Memvar
		EndIf
	EndScan
	
	* Nota: Esse Campo Tem 20 Posições
	Replace CfgFinancs With Padr(ThisForm.chk_P1_CMVS.Value, 1) + ;
							Padr(ThisForm.chk_P2_MOBS.Value, 1) + ;
							Substr(crSigCdOpe.CfgFinancs, 3, 18) In crSigCdOpe
EndIf

Return .t.
* Fim **************************************************************************************
ENDPROC

ThisForm.getMovTits.ControlSource  = [crGrdValores.MovTits]

For I = 1 To 8

	lcSub = [ThisForm.getGrupo]  + Iif(InList(I, 1, 5), [E], Iif(InList(I, 2, 6), [P], Iif(InList(I, 3, 7), [C], [A]))) + lcTip + [s]

* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
LParameters pMov, pFrm

Private pDop

If (Type([pMov])<>[C]) Or Empty(pMov)
	=MessageBox([Não Foi Passado o Primeiro Parâmetro = Código da Movimentação!!!], 0+16, [Parâmetros Inválidos!!!])
	Return .f.
EndIf

If (Type([pFrm])<>[O])
	=MessageBox([Não Foi Passado o Segundo Parâmetro = Form!!!], 0+16, [Parâmetros Inválidos!!!])
	Return .f.
EndIf

If Not Used([crSigOpFin])
	=Messagebox([O Cursor "crSigOpFin" Não Foi Criado...], 0+16, [Cursor Inválido!!!])
	Return .f.
EndIf

With ThisForm
	.plAlterou  = .f.
	.ParentForm = pFrm
	.Dopes      = Padr(Alltrim(pMov), 20)
	.Caption    = Alltrim(.Caption) + [ - ] + Alltrim(pMov)
	.pcEscolha  = .ParentForm.pcEscolha 
	.poDataMgr  = .ParentForm.poDataMgr

	If (.poDataMgr.pnIdconn > 0)		
		pDop = .Dopes
		
		.poDataMgr.Requery([crSigOpFin])
		
		=fSigOpFin_Cria() && Função no SigOpFin.prg - Cria o Cursor de Configurações de Lançamentos Financeiros

		=fSigOpFin_Tabs(.poDataMgr) && Função no SigOpFin.prg - Cria os Arquivos Para as Listas

		If Used([crGrdValores])
			Select crGrdValores
			Go Top In crGrdValores
			Locate For Dopes = ThisForm.Dopes
			If Eof([crGrdValores])
				Zap In crGrdValores
			EndIf
		EndIf
		
		Select crGrdValores
		Go Top In crGrdValores
		If Eof([crGrdValores]) 
			lnOrd = 0
			Select crSigOpFin
			Scan
				Scatter Memvar Memo
				lnOrd     	= lnOrd + 1
				
				m.Ordem   	= lnOrd
				m.Descrs  	= fSigOpFin_Busca(crSigOpFin.Codigos, [D])
				m.Notas		= Nvl(m.Notas, [])
				m.Comandos	= Nvl(m.Comandos, [])
				m.DatIncs	= Nvl(m.DatIncs, Ctot([]))
				m.DatAlts	= Nvl(m.DatAlts, Ctot([]))
							
				Insert Into crGrdValores From Memvar
			EndScan

			If InList(ThisForm.pcEscolha, [INSERIR], [ALTERAR])
				lnOrd = lnOrd + 1
				Insert Into crGrdValores (Ordem) Values (lnOrd)
			EndIf
		EndIf
				
		.MontaGrades()
		
		.btnExcluiGrd.Visible = InList(.pcEscolha, [INSERIR], [ALTERAR])
		.btnSetaDn.Visible = .btnExcluiGrd.Visible
		.btnSetaUp.Visible = .btnExcluiGrd.Visible
		
		.chk_P1_CMVS.Value = Int(Val(Substr(crSigCdOpe.CfgFinancs, 1, 1))) && CMV de Grupos de Produtos
		.chk_P2_MOBS.Value = Int(Val(Substr(crSigCdOpe.CfgFinancs, 2, 1))) && Mão de Obra da Movimentação
		
		If InList(.pcEscolha, [INSERIR], [ALTERAR])
			.grdValores.SetFocus 
		Else
			.Sair.SetFocus 		
		EndIf
		
		=DoDefault()
		
		.ParentForm.Enabled = .f.
	Else
		=Messagebox([Impossível Efetuar Conexão Com o Servidor de Banco de Dados...], 0+48, [CreateObject])
		Return .f.
	EndIf
EndWith
* Fim **************************************************************************************1
ENDPROC
