------------------------------------------------------------
* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
lparameters loForm,Operacao,loForm2

ThisForm.cntSombra.lblSombra.Caption = ThisForm.Caption
ThisForm.cntSombra.lblTitulo.Caption = ThisForm.Caption

set datasession to loForm.DataSessionId

ThisForm.Parentform = loForm

ThisForm.PoFormCaixa = loForm2

lcSql = [Select Dopes,TipoOps, Nivels, LimCres From SigCdOpe Where Dopes = ']+Operacao+[']
If ThisForm.ParentForm.PodataMgr.Sqlexecute(lcsql,'lCredOpe') < 1
	MessageBox('Favor reinicializar o processo.',16,'Falha na Conexão (lCredOpe)')
	If Type('ThisForm.poFormCaixa') = "O"
		ThisForm.PoFormCaixa.Enabled = .t.
	Endif
	Return(.F.)
EndIf
Tipoop = lCredOpe.TipoOps
_Nivel = lCredOpe.Nivels

Select TprMvCab
If lCredOpe.limcres = 1
	_Grupo = TprMvCab.grupoos
	_Conta = TprMvCab.Contaos
	_mgrupo = 'a.grupoos'
	_mconta = 'a.contaos'
Else
	_Grupo = TprMvCab.grupods
	_Conta = TprMvCab.Contads
	_mgrupo = 'a.grupods'
	_mconta = 'a.contads'
Endif

If IsEmpty(_Grupo) Or IsEmpty(_Conta)
	If Type('ThisForm.poFormCaixa') = "O"
		ThisForm.PoFormCaixa.Enabled = .t.
	Endif
	Return 
EndIf

=ThisForm.parentForm.PoDataMgr.CursorQuery( 'SigClLcr','CrSigClLcr','GruContas',_Grupo + _Conta )

lcSql = [Select * From SigOpPed ]
If ThisForm.ParentForm.PodataMgr.Sqlexecute(lcsql,'LocalOpePd') < 1
	MessageBox('Favor reinicializar o processo.',16,'Falha na Conexão (LocalOpePd)')
	If Type('ThisForm.poFormCaixa') = "O"
		ThisForm.PoFormCaixa.Enabled = .t.
	Endif
	Return(.F.)
EndIf

Select DbOpera
Index On dopes Tag Dopes

If Reccount('LocalOpePd') > 0

	lStrQuery = [Select Dopes,OpPends From SigCdOpe ]
	If ThisForm.ParentForm.poDataMgr.SqlExecute(lStrQuery, 'CrTmpOpe') < 1
		MessageBox('Favor reinicializar o processo.',16,'Falha na Conexão (CrTmpOpe)')
		If Type('ThisForm.poFormCaixa') = "O"
			ThisForm.PoFormCaixa.Enabled = .t.
		Endif
		Return(.F.)
	EndIf

	Select CrTmpOpe
	Index On Dopes Tag dopes

	lStrQuery = [Select a.dopes, a.EmpDopNums, a.ChkPagos, a.ChkSubn, a.Utilizados From SigMvCab a, SigOpPed b ]+;
				[Where a.ContaDs = ']+_Conta+[' And a.Dopes = b.Dopes ]+ ;
				[Union all ]+;
				[Select a.dopes, a.EmpDopNums, a.ChkPagos, a.ChkSubn, a.Utilizados From SigMvCab a, SigOpPed b ]+;
				[Where a.ContaOs = ']+_Conta+[' And a.ContaDs <> ']+_Conta+[' And a.Dopes = b.Dopes ]
	If ThisForm.ParentForm.poDataMgr.SqlExecute(lStrQuery, 'CrTprMvCab') < 1
		MessageBox('Favor reinicializar o processo.',16,'Falha na Conexão (CrTprMvCab)')
		If Type('ThisForm.poFormCaixa') = "O"
			ThisForm.PoFormCaixa.Enabled = .t.
		Endif
		Return(.F.)
	EndIf

	Select a.empDopNums From CrTprMvCab a, LocalOpePd b Where a.Dopes = b.Dopes And ;
	Iif(b.Tpagos = 1,a.ChkPagos,Iif(b.Tpagos = 2,Not a.ChkPagos, 0=0 )) And ;
	Iif(b.Tutils = 1,a.Utilizados = 1,Iif(b.Tutils = 2,Utilizados = 0, 0=0 ) ) ;
	Into Cursor CrConserto2
	
	Select CrConserto2
	Scan
		lcSql = [Select Dopes, Moedas, Sum( ( Qtds-QtBaixas ) * (Units+ValRats) ) as Totas From SigMvItn ]+;
				[Where EmpDopNums = ']+CrConserto2.EmpDopNums+[' Group by Dopes, Moedas]
		ThisForm.ParentForm.PodataMgr.Sqlexecute(lcSql,'lCredEesti')
		Select lCredEesti
		Scan
			If Not Seek(lCredEesti.Dopes,'DbOpera')
				Insert Into DbOpera (Dopes,Totas) Values (lCredEesti.Dopes,0)
			EndIf
			_ttConv = Round( lCredEesti.Totas * fBuscarCotacao(lCredEesti.moedas,date()) / fBuscarCotacao( crSigClLcr.moedas,date() ), 2 )
			Replace Totas With Totas + _ttConv in DbOpera
		EndScan
	EndScan
EndIf

lcSql = [Select Rclis From SigCdCli where Iclis = ']+_Conta+[']
If ThisForm.ParentForm.PodataMgr.Sqlexecute(lcSql,'CrTmpCli') < 1
	MessageBox('Favor reinicializar o processo.',16,'Falha na Conexão (CrTmpCli)')
	If Type('ThisForm.poFormCaixa') = "O"
		ThisForm.PoFormCaixa.Enabled = .t.
	Endif
	Return(.F.)
Endif

lcRazao = CrTmpCli.Rclis

ThisForm.get_nconta.Caption = _Conta + ' ' + Alltrim(lcRazao)

Declare LaCampo(2,2)
LaCampo(1,1)='Grupos'
LaCampo(2,1)='Contas'
LaCampo(1,2)=_Grupo
LaCampo(2,2)=_Conta

=ThisForm.ParentForm.PoDataMgr.CursorQuery('SigMvSlc','CrTmpSccr',@LaCampo)

Select CrTmpSccr
_Saldocc = 0
Scan
	_Saldocc = _Saldocc + Round(Saldos * fBuscarCotacao(CrTmpSccr.moedas,date()) / fBuscarCotacao(CrSigClLcr.moedas,date()), 2 )
Endscan
ThisForm.get_contac.Value = _Saldocc

lcQuery = [Select b.grclis, b.iclis, sum(b.valors) as Valors, b.fpags ]+;
		  [From SigCqChm a, SigChe b, SigCdOpt c Where ]+;
		  [b.GrClis = ']+_Grupo+[' And b.Iclis = ']+_Conta+[' And b.umovs = c.operacaos and ]+;
		  [c.tipos='DB' and a.nums = b.nums Group by b.GrClis, b.Iclis, b.Fpags]

If ThisForm.ParentForm.PodataMgr.SqlExecute(lcQuery,'Cheques') < 1
	=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (cheques)')
	If Type('ThisForm.poFormCaixa') = "O"
		ThisForm.PoFormCaixa.Enabled = .t.
	Endif
	Return .f.
Endif
_chequec = 0
_chequep = 0
Select Cheques
Scan
	If IsEmpty(fpags)
		_chequep = _chequec + valors
	Else
		=Thisform.parentform.podatamgr.cursorquery('SigOpFp','CrSigOpFp','Fpags',Cheques.Fpags)
		If crSigOpFp.valpends # 1
			_chequec = _chequec + valors
		Else
			_chequep = _chequep + valors
		Endif
	Endif
Endscan
_Chequec = Round( _chequec / fBuscarCotacao( crSigClLcr.moedas,date() ), 2 )
_Chequep = Round( _chequep / fBuscarCotacao( crSigClLcr.moedas,date() ), 2 )

ThisForm.get_cheque.Value = _chequec 
ThisForm.get_chqpend.Value = _chequep

_ChvLCred = TprMvCab.Emps + _Nivel + Str(TprMvCab.Numes,6)

Select TprMvCab
If lCredOpe.limcres = 1
	_Grupo = TprMvCab.grupoos
	_Conta = TprMvCab.Contaos
	_mgrupo = 'a.grupoos'
	_mconta = 'a.contaos'
Else
	_Grupo = TprMvCab.grupods
	_Conta = TprMvCab.Contads
	_mgrupo = 'a.grupods'
	_mconta = 'a.contads'
EndIf

lcQuery = [Select sum( (b.Qtds - b.QtBaixas) * b.Units ) as Totas, b.Moedas, a.Dopes From SigMvCab a, SigMvItn b, SigCdOpe c ]+;
		  [Where ]+Iif(lCredOpe.LimCres=1,[a.Grupoos='],[a.Grupods='])+_Grupo+[' And ]+;
		  Iif(lCredOpe.LimCres=1,[a.Contaos='],[a.contads='])+_Conta+[' And Not a.EmpDopNums=']+_chvlCred+[' And ]+;
		  [a.EmpDopNums = b.EmpDopNums and a.Dopes = c.Dopes And c.Tipoops = ]+Str(CrSigCdPam.Oppends,2)+[ ]+;
		  [Group by b.moedas,a.Dopes]

If ThisForm.ParentForm.PodataMgr.SqlExecute(lcQuery,'Pendencia') < 1
	=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (Pendencia)')
	If Type('ThisForm.poFormCaixa') = "O"
		ThisForm.PoFormCaixa.Enabled = .t.
	Endif
	Return .f.
Endif

_Pendente = Round( TprMvCab.valos / fBuscarCotacao( crSigClLcr.moedas,date() ), 2 )
_PendOutr = 00

Select Pendencia
Scan
