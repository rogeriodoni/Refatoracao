------------------------------------------------------------
* PROCEDURE msv_procurar
------------------------------------------------------------
PROCEDURE msv_procurar
Local lcBusca

If .Not.IsEmpty(ThisForm.Pagina.Dados.Get_codigo.Value)
	ThisForm.poDataMgr.LocateCursor('crSigCdSct','codigos',ThisForm.Pagina.Dados.Get_codigo.Value)
EndIf

ENDPROC

------------------------------------------------------------
* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
LPARAMETERS lcacesso

With ThisForm
	.poDatamgr = CreateObject('fSqlConector')
	With .poDatamgr
		.AddCursor('SigCdSct','codigos','crSigCdSct','',ThisForm.pagina.lista.grade)
		.pfSqlTabela(1).pColuna('codigos','','','Código',80,.t.)
		.pfSqlTabela(1).pColuna('descs','','','Descrição',160,.t.)
	EndWith
	= DoDefault()
	.LockScreen = .f.
EndWith
ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Lista.Grupo_op.Click
------------------------------------------------------------
PROCEDURE Pagina.Lista.Grupo_op.Click
LParameters opcao

select CrSigCdSct
If !dodefault( opcao )
	ThisForm.mAtivapagina1
	Return
Endif

Do Case
	Case Inlist( ThisForm.pcEscolha, 'INSERIR', 'PROCURAR' )
		ThisForm.Pagina.Dados.Get_Codigo.SetFocus
	Case ThisForm.pcEscolha = 'ALTERAR'
		ThisForm.Pagina.Dados.Get_Descricao.SetFocus
Other
	ThisForm.Pagina.Dados.Grupo_Salva.Salva.SetFocus
Endcase

ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
If ThisForm.pcEscolha = 'INSERIR'
	If IsEmpty( CrSigCdSct.Codigos )
		Messagebox( 'Codigo não pode ficar em branco!', 48, '' )
		ThisForm.Pagina.Dados.Get_Codigo.SetFocus
		Return
	Endif
	If ThisForm.poDataMgr.ChkRegister('SigCdSct','codigos',ThisForm.Pagina.Dados.Get_Codigo.Value)
		Messagebox( 'Código já Cadastrado!', 48, '' )
		ThisForm.Pagina.Dados.Get_Codigo.SetFocus
		Return
	Endif
Endif

=DoDefault()

ThisForm.mAtivapagina1

ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Dados.Grupo_Salva.Cancelar.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Cancelar.Click
=DoDefault()
If ThisForm.plCancelar
	ThisForm.mAtivapagina1
Endif

ENDPROC
