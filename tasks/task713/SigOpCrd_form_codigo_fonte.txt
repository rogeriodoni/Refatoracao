------------------------------------------------------------
PROCEDURE inserelinha
Thisform.lHouveIns = .T.

Select crSigOpCrr
Locate for faixafs=0
If Eof()
	Insert Into crSigOpCrr (Dopes, CidChaves) Values (crSigCdOpe.Dopes,fUniqueIds())
EndIf
Select crSigOpCrr
Keyboard '{DNARROW}' 
ThisForm.GradeSubN.Refresh
Thisform.GradeSubN.Column1.SetFocus

ENDPROC

------------------------------------------------------------
* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
LParameters poForm, pnDataSes

Local lcOperacao

Set DataSession to pnDataSes

Thisform.ParentForm = poForm
Thisform.PcEscolha = IIF(EMPTY(poForm.PcEscolha),"INSERIR",poForm.PcEscolha)
Thisform.ParentForm.Enabled  = .F.
lcOperacao = crSigCdOpe.Dopes
		    
With Thisform
	.Caption = 'Faixa de Valores da Operação ' + AllTrim( Proper( crSigCdOpe.Dopes ) )
	.Top     = Thisform.ParentForm.Top + Thisform.ParentForm.Pagina.Lista.Grade.Top + 50
	.Left    = Thisform.ParentForm.Left + ;
			   Thisform.ParentForm.Pagina.Lista.Grade.Left + ;
			   Thisform.ParentForm.Pagina.Lista.Grade.Columns(1).Width + ;
			   Thisform.ParentForm.Pagina.Lista.Grade.Columns(3).Width + 16

	.cmdExcluir.Visible = fChecaAcesso('SigOpCrr','EXCLUIR') And ;
						  Iif( Type( 'Thisform.PcEscolha' ) # 'C', .T., ;
						  InList( Thisform.PcEscolha, 'INSERIR', 'ALTERAR' ) )
	.cmdExcluir.Enabled = .cmdExcluir.Visible 
EndWith

ThisForm.cntSombra.lblSombra.Caption = ThisForm.Caption
ThisForm.cntSombra.lblTitulo.Caption = ThisForm.Caption

With SigCdOpe
	With .poDatamgr
        .ReQuery('crSigOpCrr')
		SELECT crSigOpCrr
		Go Top
	EndWith
Endwith	


Select CrSigOpCrr
Goto Top
If Reccount() = 0 And InList(Thisform.PcEscolha,'INSERIR','ALTERAR')
	ThisForm.inserelinha()
Endif

With Thisform.GradeSubN
	.RecordSource          = 'crSigOpCrr'
	.Column1.ControlSource = 'crSigOpCrr.FaixaIs'
	.Column2.ControlSource = 'crSigOpCrr.FaixaFs'
	.Column3.ControlSource = 'crSigOpCrr.Percs'
	.Column4.ControlSource = 'crSigOpCrr.Multiplos'
	.Refresh
EndWith

ENDPROC

------------------------------------------------------------
* PROCEDURE Click
------------------------------------------------------------
PROCEDURE Click
If ( Thisform.lHouveExcl Or Thisform.lHouveIns )

	Select CrSigOpCrr
	Go Top
	Delete For Empty(Dopes) or Empty(faixafs)
	Replace CidChaves with fUniqueIds() For Empty(CidChaves)

	m.lnErro1 = SigCdOpe.poDatamgr.pnIdConn

	With SigCdOpe
		If m.lnErro1>0 
			m.lnErro1 = Iif(.poDataMgr.UpDate('crSigOpCrr'),1,-1)
		EndIf 
		If m.lnErro1>0 
			m.lnErro1 = .poDataMgr.Commit()
		EndIf 
		If Not m.lnErro1>0 
			.poDataMgr.RollBack()
		EndIf 
	EndWith 
	If Not m.lnErro1>0 
		= MessageBox('Erro na Gravação dos Dados do Cadastro de Operações.',48)
	EndIf 
EndIf

Thisform.ParentForm.Enabled = .T.
ThisForm.Release
ENDPROC
