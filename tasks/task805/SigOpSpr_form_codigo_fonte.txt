PROCEDURE Init
LParameters poForm

ThisForm.cntSombra.lblSombra.Caption = ThisForm.Caption
ThisForm.cntSombra.lblTitulo.Caption = ThisForm.Caption

Set DataSession To poForm.DataSessionId

With This
	.ParentForm = poForm
	.ParentForm.Enabled = .f.
	.cmdEscolha.Visible   = InList(.ParentForm.pcEscolha, [INSERIR], [ALTERAR])
	.cmdProcessar.Visible = InList(.ParentForm.pcEscolha, [INSERIR], [ALTERAR]) And (Upper(.ParentForm.Name) = [SIGCDBAL])
	.lblDias.Visible 	  = InList(.ParentForm.pcEscolha, [INSERIR], [ALTERAR]) And (Upper(.ParentForm.Name) = [SIGCDBAL])
	.getDias.Visible 	  = InList(.ParentForm.pcEscolha, [INSERIR], [ALTERAR]) And (Upper(.ParentForm.Name) = [SIGCDBAL])
	Go Top In crProdutos

	With .grdProdutos
		.RecordSource          = [crProdutos]
		.Column1.ControlSource = [crProdutos.CPros]
		.Column2.ControlSource = [crProdutos.DPros]
		.Refresh

		Select * ;
		  From crProdutos ;
		 Where Empty(CPros) ;
		  Into Cursor LocalBrancos
		Go Top In LocalBrancos

		If Eof([LocalBrancos])
			Append Blank In crProdutos
			Go Bottom In crProdutos
		EndIf

		.Column1.SetFocus
		.Refresh
	EndWith
	
	.cnpProcessando.Top = 500
Endwith	
* Fim **************************************************************************************
ENDPROC

------------------------------------------------------------
* PROCEDURE Valid
------------------------------------------------------------
PROCEDURE Valid
Local loLista

If Not Empty(This.Value)
	=Inkey(.01)
	
	loLista = CreateObject('fwBuscaExt', ThisForm.ParentForm.poDataMgr.pnIdConn, 'SigCdPro', ;
						   'crListaRemota', 'CPros', Padr(This.Value, 14), 'Seleção', 1000)

	If Not loLista.plAchouRegistro
		loLista.mAddColuna('CPros', '', 'Código')
		loLista.mAddColuna('DPros', '', 'Descrição')
		loLista.Show()
	EndIf

	ThisForm.grdProdutos.Column1.Text1.Value = crListaRemota.CPros
	ThisForm.grdProdutos.Column2.Text1.Value = crListaRemota.DPros
	ThisForm.grdProdutos.Column1.Text1.Refresh
	ThisForm.grdProdutos.Column2.Text1.Refresh
	Use In crListaRemota

	If Not Empty(This.Value)
		lnRec = Recno([crProdutos])

		Select crProdutos
		*Go Top In crProdutos

		Select * ;
		  From crProdutos ;
		 Where CPros = This.Value And Recno() <> lnRec  ;
		  Into Cursor LocalBrancos
		Go Top In LocalBrancos
		If Not Eof([LocalBrancos])
			=MessageBox([Produto Já Selecionado!!!], 0+16, [Atenção!!!])
			This.Value = []
			Return .f.
		EndIf

		Select * ;
		  From crProdutos ;
		 Where Empty(CPros) ;
		  Into Cursor LocalBrancos
		Go Top In LocalBrancos

		If Not Eof([LocalBrancos])
			Select crProdutos
			Locate For Empty(CPros)
		Else
			Append Blank In crProdutos
			Go Bottom In crProdutos
		EndIf
		ThisForm.grdProdutos.Refresh
		ThisForm.grdProdutos.Column1.Refresh
		If (Lastkey() = 13)
			Keyboard '{DNARROW}'
		EndIf
	EndIf
Else
	ThisForm.grdProdutos.Column2.Text1.Value = []
	ThisForm.grdProdutos.Column2.Text1.Refresh
EndIf
* Fim **************************************************************************************
ENDPROC

EndIf
If Not Empty(ThisForm.ParentForm.loFil.ColecoesF)
	lcWhe = lcWhe + [And p.Colecoes <= '] + ThisForm.ParentForm.loFil.ColecoesF + [' ]
EndIf
If Not Empty(ThisForm.ParentForm.loFil.LinhasI)
	lcWhe = lcWhe + [And p.Linhas >= '] + ThisForm.ParentForm.loFil.LinhasI + [' ]
EndIf
If Not Empty(ThisForm.ParentForm.loFil.LinhasF)
	lcWhe = lcWhe + [And p.Linhas <= '] + ThisForm.ParentForm.loFil.LinhasF + [' ]
EndIf
If Not Empty(ThisForm.ParentForm.loFil.IForsI)
	lcWhe = lcWhe + [And p.IFors >= '] + ThisForm.ParentForm.loFil.IForsI + [' ]
EndIf
If Not Empty(ThisForm.ParentForm.loFil.IForsF)
	lcWhe = lcWhe + [And p.IFors <= '] + ThisForm.ParentForm.loFil.IForsF + [' ]
EndIf

Do Case
	Case (This.Value = 1)
		lcQuery = [Select Distinct a.CPros, p.DPros ] + ;
					[From SigMvItn a ] + ;
				   [Inner Join SigCdPro p On a.CPros = p.CPros ] + ;
				   [Where a.Emps = '] + _Empr + [' ] + lcWhe + [ And ] + ;
				   		 [a.CPros Not In (Select l.CPros From SigPrIvl l Where l.Emps = '] + _Empr + [' And l.DtHrs >= ?pDat) ] + ;
				   [Order By a.CPros]

	Case (This.Value = 2)
		lcQuery = [Select Distinct a.CPros, p.DPros ] + ;
					[From SigMvEst a ] + ;
				   [Inner Join SigCdPro p On a.CPros = p.CPros ] + ;
				   [Where a.Emps = '] + _Empr + [' ] + lcWhe + [ And ] + ;
				   		 [Not a.SQtds = 0 And ] + ;
				   		 [a.CPros Not In (Select l.CPros From SigPrIvl l Where l.Emps = '] + _Empr + [' And l.DtHrs >= ?pDat) ] + ;
				   [Order By a.CPros]

	Case (This.Value = 3)
		lcQuery = [Select Distinct p.CPros, p.DPros ] + ;
					[From SigCdPro p ] + ;
				   [Where p.CPros Not In (Select l.CPros From SigPrIvl l Where l.Emps = '] + _Empr + [' And l.DtHrs >= ?pDat) ] + lcWhe + ;
				   [Order By p.CPros]
				   
	Otherwise
		Return .f.
EndCase

If (ThisForm.ParentForm.poDataMgr.SqlExecute(lcQuery, [LocalCPros]) < 1)
	=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (LocalCPros)])
	Return .f.
EndIf

Delete From crProdutos Where Empty(CPros)

lnCnt = 0
Select LocalCPros
Scan
	If Not Empty(LocalCPros.CPros)
		Scatter Memvar Memo
		Select crProdutos
		Go Top In crProdutos
		Locate For CPros = LocalCPros.CPros
		If Eof([crProdutos])
			lnCnt = lnCnt + 1
			Insert Into crProdutos From Memvar
		EndIf
	EndIf
EndScan

ThisForm.cnpProcessando.Top = 500

=MessageBox([Foram Selecionados ] + Alltrim(Str(lnCnt, 10)) + [ Produtos!!!], 0+64, [Seleção Concluída!!!])

Select crProdutos
Go Top In crProdutos
ThisForm.grdProdutos.Refresh
Go Bottom In crProdutos
ThisForm.grdProdutos.Refresh
* Fim **************************************************************************************
ENDPROC
