------------------------------------------------------------
* PROCEDURE importaplan
------------------------------------------------------------
PROCEDURE importaplan
PRIVATE ldIni, ldFin

lcArq = GetFile('xls*')

If Empty(lcArq)
	Return .f.
EndIf
		
If Not ImpExcelDriver(lcArq,'Cota Desconto')
	Return .F.
EndIf 
lnCelI=0

SELECT crSigCdDes
ZAP 

Select * from xlResults Into Cursor localCota ReadWrite 
SELECT localCota
GO top

Select Localcota
Go Top
Scan
	a=FIELD(1)
	lcEmp = ALLTRIM(STR(Iif(InList(VARTYPE(&a),'N'),&a,0)))
	a=FIELD(2)
	lcMes = PADL(ALLTRIM(Iif(InList(VARTYPE(&a),'C','M'),&a,[])),6,'0')
	a=FIELD(3)
	lnCota= Iif(InList(VARTYPE(&a),'N'),&a,0)
	a=FIELD(4)
	lnCotaC= Iif(InList(VARTYPE(&a),'N'),&a,0)
	
	IF lcEmp <> '0' 

		lcQuery = 	[Select * From SigcdEmp where nemps = ]+lcEmp+[ ]
		If ThisForm.Podatamgr.Sqlexecute(lcQuery,'localEmp') < 1
			Messagebox('Impossivel Efetuar Conexão com Servidor de Banco de Dados.',(0+48),'Erro')
			Return
		Endif

		lcEmp = localEmp.cemps
		*** prepara primeiro e ultimo dia do mes ***
		ldIni=CTOD('01'+'/'+SUBSTR(lcMes,1,2)+'/'+SUBSTR(lcMes,3,4))
		ldFin = ldIni+28
		do while DAY(ldFin)<>1
			ldFin=ldFin+1
		ENDDO
		ldFin=ldFin-1
		ldIni = DTOT(ldIni)
		ldFin = DTOT(ldFin)
				
		lcQuery = 	[Select * From SigcdDes where emps = ']+lcEmp+[' and dtini = ?ldIni and dtFin = ?ldFin]
		If ThisForm.Podatamgr.Sqlexecute(lcQuery,'crAux') < 1
			Messagebox('Impossivel Efetuar Conexão com Servidor de Banco de Dados.',(0+48),'Erro')
			Return
		Endif

		IF RECCOUNT('crAux') = 0
			lcDat=SUBSTR(ALLTRIM(STR(YEAR(DATE()))),3)+PADL(ALLTRIM(STR(month(DATE()))),2,'0')
			lcCod = lcEmp+lcDat+PADL(ALLTRIM(STR(fGerUniqueKey(lcEmp+lcDat))),3,'0')
			lnValdes=0.00
		ELSE
			lcCod=crAux.codigo
			lnValDes= crAux.valdes	
		ENDIF
		
		Insert Into crSigcdDes (codigo, Emps, dtini, dtfin, valmeta, valcota, valcotac,valdes, dtalts);
			             Value (lcCod, lcEmp, ldIni, ldFin, 0.00, lnCota, lnCotaC, lnValDes, DATETIME())
	
	ENDIF
		
	Select LocalCota
Endscan

llOks = .T.

If Not ThisForm.poDataMgr.Update([crSigcdDes])
	=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (Update - crSigcddes)])
	llOks = .f.
EndIf
If llOks
	If (ThisForm.poDataMgr.Commit() < 1)
		=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (Commit)])
		llOks = .f.
	EndIf
EndIf

ThisForm.Refresh 
RETURN llOks

ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
LOCAL lcQuery

If INLIST(ThisForm.pcEscolha,'INSERIR')
	If Empty( CrSigCdDes.emps )
		Messagebox( 'Empresa não pode ficar em branco!', 48, '' ) 
		ThisForm.Pagina.Dados.Getemps.SetFocus
		Return
	Endif
	If Empty( CrSigCdDes.dtini )
		Messagebox( 'Data Inicial não pode ficar em branco!', 48, '' ) 
		ThisForm.Pagina.Dados.Getdtini.SetFocus
		Return
	Endif
	If Empty( CrSigCdDes.dtFin )
		Messagebox( 'Data final não pode ficar em branco!', 48, '' ) 
		ThisForm.Pagina.Dados.Getdtfin.SetFocus
		Return
	Endif
	If CrSigCdDes.dtini >  CrSigCdDes.dtfin
		Messagebox( 'As datas estão incompatíveis!', 48, '' ) 
		ThisForm.Pagina.Dados.Getdtini.SetFocus
		Return
	Endif

	lcDat=SUBSTR(DTOS(TTOd(CrSigCdDes.dtini)),1,10)
	lcDat=SUBSTR(lcDat,1,4)+'-'+SUBSTR(lcDat,5,2)+'-'+SUBSTR(lcDat,7,2)
	lcQuery = 	[Select * From SigCdDes where emps = ']+CrSigCdDes.emps+[' and ']+lcDat+[' ]+;
				[between LEFT(CONVERT(VARCHAR, dtini, 120), 10) and  LEFT(CONVERT(VARCHAR, dtfin, 120),10)]	 && 2017-01-21
	If ThisForm.Podatamgr.Sqlexecute(lcQuery,'crAux1') < 1
		Messagebox('Impossivel Efetuar Conexão com Servidor de Banco de Dados.',(0+48),'Erro')
		Return
	Endif

	lcDat=SUBSTR(DTOS(TTOd(CrSigCdDes.dtfin)),1,10)
	lcDat=SUBSTR(lcDat,1,4)+'-'+SUBSTR(lcDat,5,2)+'-'+SUBSTR(lcDat,7,2)
	lcQuery = 	[Select * From SigCdDes where emps = ']+CrSigCdDes.emps+[' and ']+lcDat+[' ]+;
				[between LEFT(CONVERT(VARCHAR, dtini, 120), 10) and  LEFT(CONVERT(VARCHAR, dtfin, 120),10)]	 && 2017-01-21
	If ThisForm.Podatamgr.Sqlexecute(lcQuery,'crAux2') < 1
		Messagebox('Impossivel Efetuar Conexão com Servidor de Banco de Dados.',(0+48),'Erro')
		Return
	Endif

	If RECCOUNT('crAux1')>0 or RECCOUNT('crAux2') >0
		Messagebox( 'Esse intervado já está cadastrado para essa empresa!', 48, '' ) 
		ThisForm.Pagina.Dados.Getdtini.SetFocus
		Return
	Endif

	lcDat=SUBSTR(ALLTRIM(STR(YEAR(DATE()))),3)+PADL(ALLTRIM(STR(month(DATE()))),2,'0')
	replace codigo with crSigCdDes.emps+lcDat+PADL(ALLTRIM(STR(fGerUniqueKey(crSigCdDes.emps+lcDat))),3,'0') in crSigcdDes

Endif

If INLIST(ThisForm.pcEscolha,'EXCLUIR')
	If CrSigCdDes.valdes > 0
		Messagebox( 'Exclusão não permitida. Cota já utilizada!', 48, '' ) 
		ThisForm.Pagina.Dados.Getemps.SetFocus
		Return
	Endif
ENDIF

replace dtalts with DATETIME() in crSigcdDes
replace usualts with Usuar in crSigcdDes

=DoDefault()
ThisForm.mAtivapagina1

ENDPROC
