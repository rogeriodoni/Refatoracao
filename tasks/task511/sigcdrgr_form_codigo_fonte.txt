PROCEDURE Init
Lparameters pAce, pTit

With Thisform
	.poDatamgr = Createobject('fSqlConector',Thisform.Name)

	If Not .poDatamgr.pnIdconn > 0
		Messagebox('Impossivel Efetuar Conexão com Servidor de Banco de Dados.',(0+48),'Erro')
		Return(.F.)
	Endif

	lcCodigo = Space(10)
	lcSqlCdCri = [Select * From SigCdRga Where Codigo = ?lcCodigo]

	With .poDatamgr
		.AddCursor('SigCdRGR','cidchaves','crSigCdRGR','',Thisform.pagina.lista.grade)
		.pfSqlTabela(1).pColuna('Codigo','','','Código',100,.T.)
		.pfSqlTabela(1).pColuna('Descricao','','','Descrição.',250,.T.)

		.AddCursor('SigCdRga','CidChaves','crSigCdRga','','',lcSqlCdCri)
	Endwith

Endwith

=DoDefault()


ENDPROC

PROCEDURE msv_alterar
llOk = .t.

PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
lcCodigo = thisform.pagina.dados.getCodigo.value

If Inlist( ThisForm.pcEscolha,[INSERIR],[ALTERAR] )
	If Inlist( ThisForm.pcEscolha,[INSERIR]) 
		If Empty(lcCodigo)
			Messagebox([Código obrigatório.],64,[Aviso])
			thisform.pagina.dados.getCodigo.Setfocus()
			Return .f.
		Endif
		If ThisForm.podatamgr.ChkRegister('SigCdRGR', 'Codigo', lcCodigo)
			Messagebox([Código já cadastrada.],64,[Aviso])
			thisform.pagina.dados.getCodigo.Setfocus()
			Return .f.
		Endif
		
		Replace CidChaves With fUniqueIds() in crSigCdRGR
		Replace Datas With Datetime() in crSigCdRGR
		Replace UsuIncs With Usuar in crSigCdRGR
	Endif

	Select crSigCdRga
	Go Top
	
	Select csSigCdRga
	Scan For Prioridade >= 1
		Scatter Memvar
		
		Select crSigCdRga
		Append Blank 
		
		Gather Memvar 
		Select csSigCdRga
	EndScan 
	
	Select crSigCdRga
	Go Top
	Delete For (Empty(Empresa) And Empty(Grupo) And Empty(Conta) And Empty(Colecao) And Empty(GrupoPr)) And Quantidade = 0
	Go Top
	
	Replace Codigo With crSigCdRGR.Codigo For Empty(Codigo)
	Replace CidChaves With fUniqueIds() For Empty(CidChaves)

	If Inlist( ThisForm.pcEscolha,[ALTERAR])
		Replace DtaAlts With Datetime() in crSigCdRGR
		Replace UsuAlts With Usuar in crSigCdRGR
	Endif
Endif

If Inlist( ThisForm.pcEscolha,[ALTERAR])
	lcUpdate = [Update SigPrdWb set qtds = 0 where codregra = ']+lcCodigo+[' ]
	If Thisform.podatamgr.SqlExecute(lcUpdate) < 1
		MessageBox([Favor reiniciar o Processo.],0+16,[Falha na conexão (SigPrdWb)])
		Return .f.
	EndIf 
EndIf 

DoDefault()

Thisform.mAtivapagina1
