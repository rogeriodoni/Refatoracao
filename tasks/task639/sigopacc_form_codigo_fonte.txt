------------------------------------------------------------
* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
PARAMETERS pOldForm, OPER, pTpAltera

Private lcCliente

Thisform.Op_Escolha = Oper
ThisForm.parentForm = pOldForm
ThisForm.pkChaves   = crAltcc.cIdChaves
 
ThisForm.nTpAltera = Iif(Type([pTpAltera])=[N], pTpAltera, 0)

ThisForm.InvEmisCCs = .f.

ThisForm.Caption = Proper(Alltrim(Oper)) + [ Lançamento de Conta Corrente]
ThisForm.cntSombra.lblSombra.Caption = ThisForm.Caption
ThisForm.cntSombra.lblTitulo.Caption = ThisForm.Caption

ThisForm.OldContaPgs = crAltcc.ContaPgs

ThisForm.Cmd_Confirma.Enabled = .t.

If (Oper = [CONSULTAR])
	ThisForm.Cmd_Confirma.Enabled = .f.
	ThisForm.Cmd_Cancela.SetFocus
EndIf

ThisForm.ParentForm.PodataMgr.Cursorquery([SigCdPac], [crSigCdPac])
ThisForm.Sigkey = crSigCdPac.SigKeys

Thisform.ParentForm.poDataMgr.Requery([crSigCdCot])
ThisForm.Parentform.poDataMgr.Requery([crSigCdMoe])
ThisForm.ParentForm.poDataMgr.Requery([crSigCdGcr])
ThisForm.ParentForm.poDataMgr.Requery([crSigCdUsu])
ThisForm.ParentForm.poDataMgr.Requery([crSigCdCrg])

Select crSigCdCot
Index On Cmoes + Dtos(Datas) + Horas Tag Cotacaos

Select CrSigCdMoe
Index on Dmoes Tag Dmoes
Index on Cmoes tag Cmoes

Select CrSigCdUsu
Index on Usuarios Tag Usuarios

Select CrSigCdCrg
Index on cCargs Tag Ccargs

Select CrSigCdGcr
Index on Codigos Tag Codigos

If Not IsEmpty(M.grupos)
	=seek(M.grupos,'CrSigCdGcr','Codigos')
	thisform.get_dgrupo.value=CrSigCdGcr.Descrs
	thisform.get_dgrupo.refresh

	If Not Empty(crSigCdGcr.Classes)
		lcQuery = [Select InvEmisCCs ] + ;
				    [From SigCdCss ] + ;
				   [Where Classes = '] + crSigCdGcr.Classes + [']

		If (ThisForm.ParentForm.poDataMgr.SqlExecute(lcQuery, [LocalClass]) < 1)
			=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (LocalClass)])
			Return 0
		EndIf
		Select LocalClass
		If Not Eof([LocalClass])
			ThisForm.InvEmisCCs = (LocalClass.InvEmisCCs = 1)
		EndIf
	EndIf
else
	thisform.get_dgrupo.value=''
	thisform.get_dgrupo.refresh
EndIf

If ThisForm.InvEmisCCs
	m.AntSGrupos = M.SGrupos
	m.AntGrupEms = M.GrupEms
	m.AntSContas = M.SContas
	m.AntContEms = M.ContEms

	M.SGrupos = m.AntGrupEms
	M.GrupEms = m.AntSGrupos
	M.SContas = m.AntContEms
	M.ContEms = m.AntSContas
EndIf

if not IsEmpty(M.sgrupos)
	=seek(M.sgrupos,'crSigCdGcr','Codigos')
	thisform.get_sdgrupo.value=CrSigCdGcr.Descrs
	thisform.get_sdgrupo.refresh
else
	thisform.get_sdgrupo.value=''
	thisform.get_sdgrupo.refresh
endif
if not IsEmpty(M.grupems)
	=seek(M.grupems,'CrSigCdGcr','Codigos')
	thisform.get_dgrupems.value=CrSigCdGcr.Descrs
	thisform.get_dgrupems.refresh
else
	thisform.get_dgrupems.value=''
	thisform.get_dgrupems.refresh
Endif

if not IsEmpty(M.Jobs)
	LcCliente = m.Jobs
	ThisForm.ParentForm.poDataMgr.Requery('CrSigCdCli')
	Select CrSigCdCli
	Go Top
	thisform.getdJob.value=crSigCdCli.RClis
else
	thisform.getdJob.value=space(20)
endif

if not IsEmpty(M.contas)
	LcCliente = m.Contas
	ThisForm.ParentForm.poDataMgr.Requery('CrSigCdCli')
	Select CrSigCdCli
	Go Top
	thisform.get_dconta.value=crSigCdCli.RClis
else
	thisform.get_dconta.value=space(20)
endif
thisform.get_dconta.refresh
if not IsEmpty(M.scontas)
	LcCliente = m.scontas
	ThisForm.parentform.poDataMgr.Requery('CrSigCdCli')
	Select CrSigCdCli
	Go Top
	thisform.get_sdconta.value=crSigCdCli.RClis
else
	thisform.get_sdconta.value=space(20)
endif
thisform.get_sdconta.refresh
if not IsEmpty(M.contems)
	LcCliente = m.Contems
	ThisForm.ParentForm.poDataMgr.Requery('CrSigCdCli')
	Select CrSigCdCli
	Go Top
	thisform.get_dcontems.value=crSigCdCli.RClis
else
	thisform.get_dcontems.value=space(20)
endif
thisform.get_dcontems.refresh

=Seek( CrAltcc.Moedas, 'crSigCdMoe', 'CMoes' )
=Seek( Padr( Alltrim( Usuar ), 10 ), 'CrSigCdUsu', 'Usuarios' )
llAltCot = ((!Seek( CrSigCdUsu.Ccargs, 'CrSigCdCrg', 'Ccargs' ) Or CrSigCdCrg.AltCots # 'S' ) And ;
		 Thisform.op_escolha='ALTERAR' And ThisForm.nTpAltera = 0 And CrSigCdMoe.Cotas )

ThisForm.AlteraCotacao.Visible = llAltCot
ThisForm.Get_Cotacao.Visible   = CrSigCdMoe.Cotas
ThisForm.Say_Cotacao.Visible   = CrSigCdMoe.Cotas

=Seek( CrAltcc.SMoedas, 'CrSigCdMoe', 'CMoes' )
llAltcot = ( ( !Seek( CrSigCdUsu.Ccargs, 'CrSigCdCrg', 'Ccargs' ) Or CrSigCdCrg.AltCots # 'S' ) And ;
		  Thisform.op_escolha='ALTERAR' And ThisForm.nTpAltera = 0 And CrSigCdMoe.Cotas )

ThisForm.AlteraSCotacao.Visible = llAltCot
ThisForm.Get_sCotacao.Visible   = CrSigCdMoe.Cotas
ThisForm.Say_sCotacao.Visible   = CrSigCdMoe.Cotas

ENDPROC
