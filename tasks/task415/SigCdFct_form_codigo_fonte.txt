PROCEDURE verificaexcluir
Local llOks

llOks = .t.

lcGru = crFichaCab.CGrus
lcSgr = crFichaCab.SGrus

lcQuery = [Select * ] + ;
			[From SigCdFtc ] + ;
		   [Where CGrus = '] + lcGru + [' And ] + ;
		   		 [SGrus = '] + lcSGr + [']

If (ThisForm.poDataMgr.SqlExecute(lcQuery, [LocalCFTec]) < 1)
	=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (VerificaExcluir - SigCdFtc)])
	llOks = .f.
EndIf

If llOks
	Select LocalCfTec
	Scan
		Select crFichaTec
		Locate For (Not Empty(COrds) And Not Empty(CGrus) And Not Empty(CCars) And Not Empty(CTits) And ;
					Not (CLins < 1)) And (CCars = LocalCfTec.CCars And CTits = LocalCfTec.CTits)

		If Eof([crFichaTec])
			lcQuery = [Select Distinct a.CPros ] + ;
						[From SigCdPft a, SigCdPro b ] + ;
					   [Where a.CCars = '] + LocalCfTec.CCars + [' And ] + ;
					   		 [a.CTits = '] + LocalCfTec.CTits + [' And ] + ;
					   		 [a.CPros = b.CPros And ] + ;
					   		 [b.CGrus = '] + lcGru + [' And ] + ;
					   		 [b.SGrus = '] + lcSGr + [' ] + ;
					   [Order By a.CPros]

			If (ThisForm.poDataMgr.SqlExecute(lcQuery, [LocalProFT]) < 1)
				=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (VerificaExcluir - LocalProFT)])
				llOks = .f.
				Go Bottom In LocalCfTec
			EndIf
			
			If llOks
				Go Top In LocalProFT
				If (Reccount([LocalProFT]) > 0)
					If (Reccount([LocalProFT]) = 1)
						lcMsg = [O Produto "] + LocalProFT.CPros + [" Está ]
					Else
						lcMsg = [Existem Produtos Que Estão ]
					EndIf
					lcMsg = lcMsg + [Utilizando o] + Chr(13) + ;
							[Grupo de Características : ]  + Alltrim(LocalCfTec.CCars) + [ e o ] + ;
							[Título de Características : ] + Alltrim(LocalCfTec.CTits) + [.]
					
					=MessageBox(lcMsg, 0+16, [Impossível ] + Alltrim(Proper(ThisForm.pcEscolha)) + [!!!])
					llOks = .f.
					Go Bottom In LocalCfTec
				EndIf
			EndIf
			
			If llOks
				lcQuery = [Select Distinct GruCarTits ] + ;
							[From SigCdFtr ] + ;
						   [Where GruCarTits = '] + lcGru + lcSGr + LocalCfTec.CCars + LocalCfTec.CTits + [' ] + ;
						   [Order By GruCarTits]

				If (ThisForm.poDataMgr.SqlExecute(lcQuery, [LocalCFTeR]) < 1)
					=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (VerificaExcluir - LocalCFTeR)])
					llOks = .f.
					Go Bottom In LocalCfTec
				EndIf
			EndIf
							
			If llOks
				Go Top In LocalCFTeR
				If (Reccount([LocalCFTeR]) > 0)
					lcMsg = [A Característica "] + Alltrim(LocalCfTec.CCars) + ;
					        [" Com o Título "] + Alltrim(LocalCfTec.CTits) + [" ] + Chr(13) + ;
							[Possui Respostas Cadastradas!!! Exclua as Respostas Primeiro!!!]
					=MessageBox(lcMsg, 0+16, [Impossível ] + Alltrim(Proper(ThisForm.pcEscolha)) + [!!!])
					llOks = .f.
					Go Bottom In LocalCfTec
				EndIf
			EndIf
		EndIf
	EndScan
EndIf

Return llOks
* Fim **************************************************************************************
ENDPROC

PROCEDURE msv_excluir
Local llOks

llOks = .t.

* Apagando os Registros Anteriores de Ficha Técnica
lcQuery = [Delete From SigCdFtc ] + ;
		   [Where CGrus = '] + crFichaCab.CGrus + [' And ] + ;
		   		 [SGrus = '] + crFichaCab.SGrus + [']

If (ThisForm.poDataMgr.SqlExecute(lcQuery, []) < 1)
	=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (mSv_Alterar - Delete - SigCdFtc)])
	llOks = .f.
EndIf

If llOks
	* Gravando Fisicamente os Registros
	If (ThisForm.poDataMgr.Commit() < 1)
		=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (Commit)])
		llOks = .f.
	EndIf
EndIf

If llOks
	Replace TemDados With [] In crDados
	=fGravarLog('T', 'SIGCDFCT', 'EXCLUIR', Iif(ThisForm.pnTipo=2,'Movimentação : ','Grupo : ') + CrFichaCab.Cgrus + '/' +CrFichaCab.Sgrus ,Usuar)
Else
	* Cancelando as Alterações Feitas
	ThisForm.poDataMgr.RollBack()
EndIf

Return llOks
* Fim **************************************************************************************
ENDPROC

* PROCEDURE msv_alterar

PROCEDURE msv_alterar
Local llOks

llOks = .t.

If llOks
	* Checando Se Digitou Dados Válidos Na Grade de Ficha Técnica
	If (Reccount([crSigCdFtc]) < 1)
		=MessageBox([Não Há Nenhum Registro Válido Na Grade de Ficha Técnica!!!], 16, [Atenção!!!])
		llOks = .f.
	EndIf
EndIf

If llOks
	* Checando Se Digitou Dados Duplicados Na Grade de Ficha Técnica
	Select CCars, CTits, Sum(1) as Totas ;
	  From crSigCdFtc ;
	 Group By CCars, CTits ;
	  Into Cursor csItens

	Select csItens
	Scan
		If (csItens.Totas > 1)
			=MessageBox([O Grupo e o Título "] + Alltrim(csItens.CCars) + [ / ] + Alltrim(csItens.CTits) + ;
						[" Está Duplicado!!!], 48, [Atenção!!!])
			Go Bottom In csItens
			llOks = .f.
		EndIf
	EndScan
EndIf

If llOks
	* Checando Se Digitou a Ordem de Apresentação Duplicada Na Grade de Ficha Técnica
	Select COrds, Sum(1) as Totas ;
	  From crSigCdFtc ;
	 Group By COrds ;
	  Into Cursor csItens

	Select csItens
	Scan
		If (csItens.Totas > 1)
			=MessageBox([A Ordem de Apresentação "] + Alltrim(Str(csItens.COrds)) + [" Está Duplicada!!!], 48, [Atenção!!!])
			Go Bottom In csItens
			llOks = .f.
		EndIf
	EndScan
EndIf
	
If llOks and (ThisForm.pnTipo = 0)
	* Apanhando os Dados do Cadastro de Grupos de Produtos Para Serem Atualizados Pela Classe
	lcQuery = [Select * ] + ;
				[From SigCdGrp ] + ;
			   [Where CGrus = '] + crFichaCab.CGrus + [']

	If (ThisForm.poDataMgr.SqlExecute(lcQuery, [LocalGru]) < 1)
		=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (LocalGru)])
		llOks = .f.
	EndIf
EndIf

If llOks and (ThisForm.pnTipo = 0)
	* Checando Se o Grupo de Produtos Não Foi Apagado Antes do término do Processo
	Go Top In LocalGru
	If Eof([LocalGru])
		=MessageBox([Ocorreu Um Erro Ao Acessar o Cadastro de Grupos de Produtos!!!] + Chr(13) + ;
					[O Grupo "] + crFichaCab.CGrus + [" Foi Excluído Durante o Processo. Reinicie a Operação!!!], ;
					0+16, [Atenção!!!])
		llOks = .f.
	EndIf
EndIf

If llOks and (ThisForm.pnTipo = 0)
	* Passando os Dados Do Cursor Temporário Para o de Atualização do Cadastro de Grupos
	Zap In crSigCdGrp
	Select LocalGru
	Scatter Memvar Memo
	Insert Into crSigCdGrp From Memvar

	* Substituindo os Campos a Serem Atualizados
	Replace nChkDims With crFichaCab.nChkDims, ;
			nChkPess With crFichaCab.nChkPess, ;
			nChkTems With crFichaCab.nChkTems, ;
			nChkImps With crFichaCab.nChkImps, ;
			nFixFigs With crFichaCab.nFixFigs, ;
			mFicTecs With crFichaCab.mFicTecs In crSigCdGrp

	* Atualizando o Registro Alterado na Base de Dados
	If Not ThisForm.poDataMgr.Update([crSigCdGrp])
		=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (Update - crSigCdGrp)])
		llOks = .f.
	EndIf
EndIf

If llOks
	* Apagando os Registros Anteriores de Ficha Técnica
	lcQuery = [Delete From SigCdFtc ] + ;
			   [Where CGrus = '] + crFichaCab.CGrus + [' And ] + ;
			   		 [SGrus = '] + crFichaCab.SGrus + [']

	If (ThisForm.poDataMgr.SqlExecute(lcQuery, []) < 1)
	=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (mSv_Alterar - Delete - SigCdFtc)])
		llOks = .f.
	EndIf
EndIf

If llOks
	* Inserindo os Registros do Cursor No Banco de Dados
	If (Reccount([crSigCdFtc]) > 0)
		If Not ThisForm.poDataMgr.Update([crSigCdFtc])
			=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (Update - crSigCdFtc)])
			llOks = .f.
		EndIf
	EndIf
EndIf

If llOks
	* Gravando Fisicamente os Registros
	If (ThisForm.poDataMgr.Commit() < 1)
		=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (Commit)])
		llOks = .f.
	EndIf
EndIf

If llOks
	Replace TemDados With crFichaCab.CGrus + crFichaCab.SGrus In crDados
Else
	* Cancelando as Alterações Feitas
	ThisForm.poDataMgr.RollBack()
EndIf

Return llOks
* Fim **************************************************************************************
ENDPROC

* PROCEDURE Load
------------------------------------------------------------
PROCEDURE Load
Create Cursor crDados (CGrus c(20), DGrus c(30), SGrus c(6), DSGru c(20), TemDados c(20))
Index On CGrus + SGrus Tag Col1
Index On SGrus + CGrus Tag Col2
Index On DGrus + CGrus Tag Col3
Index On DSGru + SGrus Tag Col4
Set Order To Col1 In crDados

Create Cursor crFichaCab (CGrus c(20), DGrus c(30), SGrus c(6), DSGru c(20), nChkDims n(1), ;
						  nChkPess n(1), nChkTems n(1), nChkImps n(1), nFixFigs n(1), mFicTecs m Null)

Create Cursor crFichaTec (CGrus c(20), SGrus c(6), CCars c(20), CTits c(20), COrds n(3), ;
						  CLins n(2), Maxs n(3), Mins n(3), nChkCxas n(1), cSelecaos c(1), cTpDados c(1))

=DoDefault()
* Fim **************************************************************************************
ENDPROC

PROCEDURE Init
Lparameters pcTipo

** Tipo ( = 0 - FTec Produtos, = 1 - FTec Contas, = 2 - FTec Movimentaçoes ) para manter compatibilidade
ThisForm.pnTipo = Iif(Type([pcTipo])<>[C],0,Iif(pcTipo == 'C',1,2))

Do case
Case( ThisForm.pnTipo = 1 )
	ThisForm.Caption = [Cadastro de Perfil de Contas]
	ThisForm.Name    = [SIGCDFCTC]

Case( ThisForm.pnTipo = 2 )
	ThisForm.Caption = [Configuração de Movimentação]
	ThisForm.Name    = [SIGCDFCTO]
	
OtherWise
	ThisForm.Caption = [Cadastro de Ficha Técnica]
	ThisForm.Name    = [SIGCDFCT]
EndCase

With ThisForm
	loDad = .pagina.lista
	loFic = .pagina.dados.pgfPagina.pgficha
	loCom = .pagina.dados.pgfPagina.pgcompl
	
	loDad.Say5.Caption = Iif(ThisForm.pnTipo=0,[Ficha Técnica Criada],[Configuração Criada])
	
	.poDataMgr = CreateObject([fSqlConector], .Name)

	lcQryFTec = [Select * from SigCdFtc where nTipos = ]+Str(ThisForm.pnTipo,1)
	If (.poDataMgr.pnIdconn > 0)
		With .poDataMgr
			.AddCursor([SigCdFtc], [cIdChaves], [crSigCdFtc], '', '', lcQryFTec)
			
			Do Case
			Case ( ThisForm.pnTipo = 0 )
				.AddCursor([SigCdGrp], [CGrus],   [crSigCdGrp])
			Case ( ThisForm.pnTipo = 1 )
				.AddCursor([SigCdGcr], [Codigos], [crSigCdGrp])
			Case ( ThisForm.pnTipo = 2 )
				.AddCursor([SigCdope], [Dopes],   [crSigCdGrp])
			EndCase				
		EndWith 

		Do Case
		Case ( ThisForm.pnTipo = 0 ) && Produtos
			* Apanhando os Grupos de Produto
			lcQuery = [Select CGrus, DGrus, FTecSubs ] + ;
						[From SigCdGrp ] + ;
					   [Order By CGrus, DGrus]

			If (.poDataMgr.SqlExecute(lcQuery, [LocalGru]) < 1)
				=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (LocalGru)])
				Return .f.
			EndIf

			Select LocalGru
			oProg = CreateObject([fwprogressbar], [Processando Grupos de Produtos...], Reccount([LocalGru]))
			oProg.Show
			Scan
				oProg.SubTitulo.Caption = [Grupo : ] + LocalGru.CGrus
				oProg.Update(.t.)

				* Inserindo os Grupos de Produto na Grade
				Insert Into crDados (CGrus, DGrus, SGrus, DSGru, TemDados) ;
							 Values (LocalGru.CGrus, LocalGru.DGrus, [], [], [])

				* Verificando Se o Grupo Já Tem Ficha Técnica Lançada
				lcQuery = [Select Distinct CGrus ] + ;
							[From SigCdFtc ] + ;
						   [Where CGrus = '] + LocalGru.CGrus + [' And ] + ;
						   		 [SGrus = Space(6) and nTipos = 0]

				If (.poDataMgr.SqlExecute(lcQuery, [LocalCfTec]) < 1)
					=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (LocalCfTec)])
					Return .f.
				EndIf
				Go Top In LocalCfTec
				If Not Eof([LocalCfTec])
					* Se o Grupo Já Tem Ficha Técnica Lançada Marca No Cursor Para Mostrar Na Lista
					Replace TemDados With LocalGru.CGrus In crDados
				EndIf
				
				If (LocalGru.FTecSubs = 1) && Se o Grupo Trabalha Com Subgrupos Para Ficha Técnica
					* Apanhando os Subgrupos Dentro do Grupo de Produto
					lcQuery = [Select CGrus, Codigos, Descricaos ] + ;
								[From SigCdPsg ] + ;
							   [Where CGrus = '] + LocalGru.CGrus + [' ] + ;
							   [Order By CGrus, Codigos, Descricaos]

					If (.poDataMgr.SqlExecute(lcQuery, [LocalSGru]) < 1)
						=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (LocalSGru)])
						Return .f.
					EndIf

					Select LocalSGru
					Scan
						* Inserindo os Subgrupos de Produto na Grade

------------------------------------------------------------
PROCEDURE btnFigura.Click
Local lcAnt, lnCon

If InList(ThisForm.pcEscolha, 'INSERIR', 'ALTERAR')
	lcAnt = Sys(5) + Sys(2003)
	If Directory(goSistema.DirImagens)
		Set Default To &goSistema.DirImagens.
	EndIf
	lcFig = Alltrim(GetPict([Formato Jpeg:JPG;Formato Bitmap:BMP], [Nome do Arquivo:], [Selecionar]))
	Set Default To &lcAnt.

	If Not Empty(lcFig) And File(lcFig)
		If Not fTamanhoImagem(lcFig, ThisForm.poDataMgr, .t.)
			Return .f.
		EndIf

		If (MessageBox([Confirma a Transferência da Imagem?], 4+32+256, [Confirmação]) <> 6)
			ThisForm.Pagina.Dados.Grupo_Salva.Cancelar.SetFocus
			Return .f.
		EndIf

		Replace mFicTecs With FileToStr(lcFig) In crFichaCab

		lcArq = Sys(2023) + [\FicTecs.Jpg]

		With ThisForm.pagina.dados.pgfPagina.pgcompl
			Clear Resources
			.imgFigTec.Picture = []
			.imgFigTec.Visible = .f.

			* Figura no Campo Memo
			If Not Empty(crFichaCab.mFicTecs) And Not IsNull(crFichaCab.mFicTecs)
				If (StrToFile(crFichaCab.mFicTecs, lcArq) > 0)
					.imgFigTec.Visible = .t.
					.imgFigTec.Picture = (lcArq)
				EndIf
			EndIf
		EndWith

		If (MessageBox([Deseja Excluir a Imagem Original?], 4+32+256, [Confirmação]) = 6)
			Delete File (lcFig)
		EndIf
				
		If File(lcArq)
			Delete File (lcArq)
		EndIf
	EndIf
EndIf
* Fim *********************************************************************************
ENDPROC
