------------------------------------------------------------
* PROCEDURE mnovalinha
------------------------------------------------------------
PROCEDURE mnovalinha
Local lnRegs, lnRegMaxs

lnRegMaxs = 12

Insert Into CrSigCdMin (cpros,cidchaves) Values (ThisForm.Pagina.Dados.Get_Produto.Value,fUniqueIds())
Select CrSigCdMin
Count To lnRegs
Go Top

If lnRegs > lnRegMaxs
	lnI = 10
Else
	lnI = lnRegs
EndIf

With ThisForm.Pagina.Dados.GradeI
	.Refresh
	Do While lnRegs > 0
		.ActivateCell(lnI,2)
		If Empty(CrSigCdMin.emps)
			.ActivateCell(lnI,1)
			Exit
		Else
			lnRegs = lnRegs - 1
			.DoScroll(1)
		EndIf
	EndDo
EndWith
ENDPROC

------------------------------------------------------------
* PROCEDURE mrefreshget
------------------------------------------------------------
PROCEDURE mrefreshget
Select crSigCdMin
With ThisForm.Pagina.Dados
	.Get_Produto.Value  = CsCabec.cpros
	.GetDPro.Value      = CsCabec.dpros
	.GetCgru.Value      = CsCabec.cgrus
	.GetDGru.Value      = CsCabec.dgrus
	.GetIFor.Value      = CsCabec.ifors
	.GetDFor.Value      = CsCabec.rclis
	.GetRefs.Value      = CsCabec.reffs
	.GetQMin.Value      = CsCabec.qmins
	.Opc_Situacao.Value = Iif(CsCabec.Situas = 2,2,1)
	.Get_Produto.Refresh
	.GetDPro.Refresh
	.GetCgru.Refresh
	.GetDGru.Refresh
	.GetIFor.Refresh
	.GetDFor.Refresh
	.GetRefs.Refresh
	.GetQMin.Refresh
	.Opc_Situacao.Refresh
EndWith

ThisForm.lTemCor = .F.
ThisForm.lTemTam = .F.
If (!Empty(CrSigCdMin.cpros) )
	lStrQuery = [Select b.cores,b.tams,b.tipoestos From SigCdPro a, SigCdGrp b ]+;
				[Where a.cpros = ']+CrSigCdMin.cpros+[' And b.cgrus = a.cgrus ]
	If ThisForm.poDataMgr.SqlExecute(lStrQuery,'TmpProGru') < 1
		Return(.F.)
	EndIf
	Select TmpProGru
	Go Top
	If !Eof()
		ThisForm.lTemCor    = InList(TmpProGru.tipoestos,2,4) Or TmpProGru.cores = 1
		ThisForm.lTemTam    = InList(TmpProGru.tipoestos,3,4) Or TmpProGru.tams  = 1
		ThisForm.nTipoEstos = TmpProGru.tipoestos
	EndIf
EndIf
Return(.T.)
ENDPROC

			[From SigCdMin a ]+;
			[Inner Join SigCdPro b On b.cpros = a.cpros ]+;
			Iif(Empty(ThisForm.pcExterno), [], [Where a.Cpros = '] + ThisForm.pcCPros + [' ]) + ;
			[Group By a.cpros,b.dpros,b.ifors,b.reffs,b.sgrus ]
			
lcQPrEmp  = [Select a.*,b.dpros,b.ifors,b.reffs,b.cgrus,b.situas,c.rclis,g.dgrus ]+;
			[From SigCdMin a, SigCdPro b ]+;
			[Left Join SigCdCli c On b.ifors = c.iclis ]+;
			[Left Join SigCdGrp g On b.cgrus = g.cgrus ]+;
			[Where a.cpros = ?lcCPros And b.cpros = a.cpros ]+;
			[Order By a.emps,a.codtams,a.codcores ]

lcQCopia  = [Select 1 As marcas,0 As existes,a.cpros,a.emps,a.locals,a.qmins,a.qideal,a.codtams,a.codcores,a.cidchaves,a.grupos,a.contas ]+;
			[From SigCdMin a ]+;
			[Where a.emps = ?lcEmpO And (a.cpros + a.codtams + a.codcores) Not In ]+;
 				[(Select b.cpros + b.codtams + b.codcores As cprotamcores From SigCdMin b ]+;
 				[Where b.Emps = ?lcEmpD) ]+;
			[Union All ]+;
			[Select 0 As marcas,1 As existes,a.cpros,a.emps,a.locals,a.qmins,a.qideal,a.codtams,a.codcores,a.cidchaves,a.grupos,a.contas ]+;
			[From SigCdMin a ]+;
			[Where a.emps = ?lcEmpO And a.cpros + a.codtams + a.codcores In ]+;
 				[(Select b.cpros + b.codtams + b.codcores As cprotamcores From SigCdMin b ]+;
				[Where b.emps = ?lcEmpD) ]+;
			[Union All ]+;
			[Select 0 As marcas,2 As existes,a.cpros,a.emps,a.locals,a.qmins,a.qideal,a.codtams,a.codcores,a.cidchaves,a.grupos,a.contas ]+;
			[From SigCdMin a ]+;
			[Where a.emps = ?lcEmpD And a.cpros + a.codtams + a.codcores Not In ]+;
 				[(Select b.cpros + b.codtams + b.codcores As cprotamcores From SigCdMin b ]+;
				[Where b.emps = ?lcEmpO)]

lcCPros = Space(14)
lcEmpO  = Space(03)
lcEmpD  = Space(03)

With ThisForm
	.poDataMgr = CreateObject('fSqlConector','SigCdMin')
	With .poDataMgr
		.AddCursor('SigCdMin', 'cpros', 'CrProdutos', '', ThisForm.Pagina.Lista.Grade, lcQProds)
		.pfSqlTabela(1).pColuna('cpros', '', '', 'Produto'   ,  108, .T.)
		.pfSqlTabela(1).pColuna('dpros', '', '', 'Descrição' , 285, .T.)
		.pfSqlTabela(1).pColuna('ifors', '', '', 'Fornecedor',  75, .T.)
		.pfSqlTabela(1).pColuna('reffs', '', '', 'Referência', 150, .T.)
		.pfSqlTabela(1).pColuna('sgrus', '', '', 'Sub Grp'   ,  45, .F.)
		.AddCursor('SigCdMin', 'cidchaves', 'CrSigCdMin', '', '', lcQPrEmp)
		.AddCursor('SigCdMin', 'cidchaves', 'CsCopia'   , '', '', lcQCopia)
	EndWith
	=DoDefault(_Acesso)
	.lTemCor  = .F.
	.lTemTam  = .F.
	.nTeste   = 0
	.AcCopiar = fChecaAcesso('SigCdMin','COPIAR')
	With .Pagina.Lista
		.BtnCopiar.Enabled = ThisForm.AcCopiar
	EndWith
EndWith

If ThisForm.poDataMgr.SqlExecute([Select gesind From SigCdPam ],'CrSigCdPam') < 1
	MessageBox('Favor reinicializar o processo.',16,'Falha na Conexão')
	Return(.F.)
EndIf

ThisForm.poDataMgr.Requery('CrProdutos')

ThisForm.Pagina.Lista.CntCopia.Visible = .F.

ThisForm.mAtivaPagina1

If Type('_Acesso') <> 'C'
	ThisForm.LockScreen = .T.
	=fCarregaGrade(ThisForm.Pagina.Lista.Grade)
	ThisForm.LockScreen = .F.
Else
	If Reccount('CrProdutos') = 0 And ThisForm.pcExterno = 'ALTERAR'
		ThisForm.pcExterno = 'INSERIR'
	Endif
	If Empty(ThisForm.pcExterno)
		ThisForm.Pagina.Lista.Grupo_op.Click(1)
	Else

			Case (ThisForm.pcExterno = [INSERIR])

	ThisForm.LockScreen = .F.

		ThisForm.Busca = 3

	Case !Empty(lcGReffs)
		Set Order To Reffs
		Seek lcGReffs
		ThisForm.Busca = 4

EndCase

Set Near Off
ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
Private lcCPros
Local lcBcQuery, llOkBcQuery
lcCPros = This.Parent.Parent.Get_Produto.Value

If InList(ThisForm.pcEscolha,'INSERIR','ALTERAR')
	If ThisForm.pcEscolha = 'INSERIR'
		If Empty(lcCPros)
			Messagebox('Produto Inválido !!!',0+48,'')
			This.Parent.Parent.Get_Produto.SetFocus
			Return 0
		EndIf
	EndIf
	
	Select CrSigCdMin
	Go Top
	Scan
		If Empty(CrSigCdMin.emps) 
			Delete
		EndIf
	EndScan
	
	This.Parent.Parent.GradeI.Refresh
	This.Parent.Parent.GradeI.Column1.SetFocus

	Select CrSigCdMin
	Go Top
	
	If Eof()
		MessageBox('Necessário pelo menos a movimentação de uma empresa !!!',0+48,'Movimentação Inválida')
		Insert Into CrSigCdMin (cpros,cidchaves) Values (Thisform.pagina.dados.get_Produto.Value,fUniqueIds())
		Select CrSigCdMin
		thisform.Refresh
		Thisform.pagina.dados.GradeI.column1.Setfocus 
		Return
	Else
	
		** VERIFICA DUPLICADOS (Produto+empresa+grupo+conta.)
		Select CrSigCdMin
		Go Top
		Scan
			lcBcQuery = CrSigCdMin.cPros + CrSigCdMin.Emps + CrSigCdMin.Grupos + ;
						CrSigCdMin.Contas + CrSigCdMin.CodCores + CrSigCdMin.CodTams
						
			Select Cpros, Emps, Grupos, Contas, CodCores, CodTams ;
			  From CrSigCdMin ;
			 Where CPros + Emps + Grupos + Contas + CodCores + CodTams = lcBcQuery ;
			  Into Cursor crBSigCdMin

			If Reccount('crBSigCdMin')>1 && Achou Duplicado
				MessageBox('Verifique a existência de estoque mínimo duplicado no produto',16,'')
				Thisform.Pagina.Dados.Gradei.SetFocus
				Return .f.			
			EndIf 
		EndScan 


		If ThisForm.poDataMgr.SqlExecute([Select nEstMins From SigCdPac],'CrSigCdPac') < 1
			MessageBox('Favor reinicializar o processo.',16,'Falha na Conexão')
			Return(.F.)
		EndIf
		If CrSigCdPac.nEstMins=1 && Obrigado a preencher
			Select CrSigCdMin
			Scan 
				If Empty(CrSigCdMin.Grupos)
					Messagebox('Preenchimento do Grupo Obrigatório !!!',32,'')
					ThisForm.Pagina.Dados.GradeI.SetFocus 
					ThisForm.Pagina.Dados.GradeI.Column5.SetFocus
					Return .f.
				EndIf 

				If Empty(CrSigCdMin.Contas)
					Messagebox('Preenchimento da Conta Obrigatório !!!',32,'')
					ThisForm.Pagina.Dados.GradeI.SetFocus 
					ThisForm.Pagina.Dados.GradeI.Column6.SetFocus
					Return .f.
				EndIf 
			EndScan 
		EndIf 
	EndIf
EndIf

=DoDefault()

If Type('ThisForm.pProduto') <> 'C'
	ThisForm.mAtivaPagina1
Else
	ThisForm.Pagina.Lista.Grupo_saida.Sair.Click()
EndIf 
ENDPROC
