PROCEDURE acertacolunas
Lparameters pTip

With ThisForm.pgfPrincipal.pgDados.pgResults.pgRes01.grdDados1
	If (pTip = [LER])
		lcStr = ThisForm.pgfPrincipal.pgFiltros.getCfgGrade.Value 
		
		lnPos = 1 && Posição Inicial de Cada Bloco Correspondente a Uma Coluna da Grade
		For lnCnt = 1 To .ColumnCount
			lcBlo = Substr(lcStr, lnPos, 5) 		&& Bloco Com 5 Posições: 1 a 3 = Width; 	
													&&						 4     = Visible	
													&&						 5     = Uso Futuro	
			lnWid = Int(Val(Substr(lcBlo, 1, 3)))	&& Largura da Coluna						
			llVis = Not (Substr(lcBlo, 4, 1)=[0])	&& Coluna Visible=1 (default); Invisible=0	
			lcCol = Alltrim(Str(lnCnt))				&& Número da Coluna							

			.Column&lcCol..Width   = Iif(lnWid > 0, lnWid, .Column&lcCol..Width) && Se Não Estiver Gravado Mantém a Largura Original
			.Column&lcCol..Visible = llVis										 && Se Não Estiver Gravado Mantém Visible			

			lnPos = lnPos + 5 && Incrementa de 5 em 5 Posições a Cada Coluna
		Next
	Else && [GRAVAR]
		lcStr = []
		For lnCnt = 1 To .ColumnCount
			lcCol = Alltrim(Str(lnCnt))										&& Número da Coluna	
			lcStr = lcStr + Padl(.Column&lcCol..Width, 3, [0]) + ;
							Iif(.Column&lcCol..Visible, [1], [0]) + ;
							[ ] && Nota: o Branco é Para Uso Futuro
		Next
		ThisForm.pgfPrincipal.pgFiltros.getCfgGrade.Value = lcStr
	EndIf
EndWith
* Fim **************************************************************************************
ENDPROC

* PROCEDURE buscaantigos
------------------------------------------------------------
PROCEDURE buscaantigos
Private pdDtA
Local lcPro, lcTam

Select crCursor1
oProg2 = CreateObject([fwProgressBar], [Processando Dados Anteriores...], Reccount([crCursor1]))
oProg2.Top = oProg2.Top + Int(oProg2.Height/2)
oProg2.Rodape.Caption = [Buscando Produtos na Data de Movimentação]
oProg2.Show
Scan
	oProg2.Update(.t.)

	lcPro = Nvl(crCursor1.Codigo_Item, [])
	lcTam = Nvl(crCursor1.Tamanho, [])
	pdDtA = Nvl(crCursor1.Data_Venda, Datetime())
		
	lcQuery = [Select Top 1 ] + ;
					 [p.Mercs as Gde_Grupo, ] + ; 
					 [p.CGrus as Grupo, ] + ; 
					 [p.SGrus as SubGrupo, ] + ; 
					 [p.CustoFs as Custo_Cadastro, ] + ; 
					 [p.MoeCusfs as Moe_Cus_Cadastro, ] + ; 
					 [p.MarkupA as Markup_Cadastro, ] + ; 
					 [p.Reffs as Ref_Fornecedor, ] + ;
					 [p.PVens as Venda_Cadastro, ] + ; 
					 [p.PVens as Desc_Acres, ] + ; 
					 [p.PVens as Venda_CadastroR, ] + ; 
					 [p.Moevs as Moe_Vda_Cadastro, ] + ; 
					 [p.FigJpgs as Imagem, ] + ;
					 [p.IFors as Cod_Fornecedor ] + ;
				[From SigCdPrc p ] + ;
			   [Where p.CPros = '] + lcPro + [' And ] + ;
			   		 [p.DataAlts >= ?pdDtA ] + ;
			   [Order By p.DataAlts Desc]

	If Not fExecCmds(ThisForm.poDataMgr, [crAntigo1], lcQuery)
		oProg2.Complete
		Return .f.
	EndIf
	
	Select crAntigo1
	Go Top In crAntigo1
	If Not Eof([crAntigo1])
		Replace Gde_Grupo 		 With crAntigo1.Gde_Grupo, ; 
				Grupo 			 With crAntigo1.Grupo, ; 
				SubGrupo 		 With crAntigo1.SubGrupo, ; 
				Custo_Cadastro 	 With crAntigo1.Custo_Cadastro, ; 
				Moe_Cus_Cadastro With crAntigo1.Moe_Cus_Cadastro, ; 
				Markup_Cadastro  With crAntigo1.Markup_Cadastro, ; 
				Ref_Fornecedor 	 With crAntigo1.Ref_Fornecedor, ; 
				Venda_Cadastro 	 With crAntigo1.Venda_Cadastro, ; 
				Desc_Acres 		 With crAntigo1.Desc_Acres, ; 
				Venda_CadastroR  With crAntigo1.Venda_CadastroR, ; 
				Moe_Vda_Cadastro With crAntigo1.Moe_Vda_Cadastro, ; 
				Imagem 			 With crAntigo1.Imagem, ; 
				Cod_Fornecedor 	 With crAntigo1.Cod_Fornecedor In crCursor1

		lcQuery = [Select Top 1 ] + ;
						 [t.Percs as Markup_Tamanho, ] + ; 
						 [t.Valor as Valor_Tamanho, ] + ; 
						 [t.PesoMs as Peso_Tamanho ] + ;
					[From SigPrTa2 t ] + ;
				   [Where t.CPros = '] + lcPro + [' And ] + ;
				   		 [t.CodTams = '] + lcTam + [' And ] + ;
				   		 [t.DataAlts >= ?pdDtA ] + ;
				   [Order By t.DataAlts Desc]

		If Not fExecCmds(ThisForm.poDataMgr, [crAntigo1], lcQuery)
			oProg2.Complete
			Return .f.
		EndIf
		
		Select crAntigo1
		Go Top In crAntigo1
		If Not Eof([crAntigo1])
			Replace Markup_Tamanho With crAntigo1.Markup_Tamanho, ;
					Valor_Tamanho  With crAntigo1.Valor_Tamanho, ;
					Peso_Tamanho   With crAntigo1.Peso_Tamanho In crCursor1
		EndIf
	EndIf
	Use In crAntigo1
EndScan
oProg2.Complete
* Fim **************************************************************************************

ENDPROC

	If Not Empty(m.Valor_Tamanho)
		m.Venda_Cadastro = m.Valor_Tamanho
	EndIf
	
	If (lcMdC <> m.Moeda)
		lnCt1 = fBuscarCotacao(m.Moeda, m.Data_Venda, ThisForm.poDataMgr)
		m.Val_Unitario = Round((m.Val_Unitario * lnCt1) / lnCtC, 2)
		m.Total        = Round((m.Total        * lnCt1) / lnCtC, 2)

		lnCt1 = fBuscarCotacao(m.Moe_Vda_Cadastro, m.Data_Venda, ThisForm.poDataMgr)

				[From SigMvPar p, SigOpFp f ] + ;

	Select * From LocalParcs Where Ttod(Datas) = Ttod(Vencs) Or Infos = 'T' Into Cursor LocalEnts

	Use In LocalParcs

	* Fixo Aqui Pois Somente Eles Utilizam o Relatório. Caso Algum Dia Outro Cliente Vá 
	* Usar Esse Relatório Esse Trecho de Código Deve Ser Inserido Dentro de Um Campo Nos
	* Parâmetros do Sistema, Sendo Chamado e Usado Aqui Como Macro e Caso Não Seja Usado
	* o Cálculo Bastará Deixar o Campo em Branco - Consultar Richard Nesse Caso

	If (m.Gde_Grupo = [JOI])
		If (m.Parcelas > 1)
			lnEnt = Iif(llEnt, 1, 0) && Se Não Tiver Entrada Adiciona Um Mês de Juros
			For lnAdi = 1 To (m.Parcelas - lnEnt)
				lnDif = ((m.Venda_Cadastro * 2) / 100)
				m.Venda_Cadastro = m.Venda_Cadastro + lnDif
				m.Desc_Acres     = m.Desc_Acres     + lnDif
			Next
		EndIf
	EndIf

	* Caso Seja Relógio Deve Descontar 10% Se Tiver 1 Parcela OU 5% de 2 a 5 Parcelas   
	If (m.Gde_Grupo = [REL])
		If (m.Parcelas = 1)
			lnDif = ((m.Venda_Cadastro * 10) / 100)
			m.Venda_Cadastro = m.Venda_Cadastro - lnDif
			m.Desc_Acres     = m.Desc_Acres     - lnDif
		EndIf
		
		If Between(m.Parcelas, 2, 5)
			lnDif = ((m.Venda_Cadastro * 5) / 100)
			m.Venda_Cadastro = m.Venda_Cadastro - lnDif
			m.Desc_Acres     = m.Desc_Acres     - lnDif
		EndIf
	EndIf
	* Fim do Cálculo de Acréscimo e Desconto *******************************************
	
	m.Barras          = Iif(IsEmpty(m.Barra_Individual), Nvl(m.Barra_Geral, 0), Nvl(m.Barra_Individual, 0))

	Set Separator To ThisForm.goOpCge.SetSepar
	
	Insert Into crPgRes01 From Memvar
EndScan

		With .poSigSyPar
			.pGrvXMLCfg	 = ThisForm.Name

			.pGrvXMLUsu  = Usuar
			.pGrvXMLObj  = ThisForm.pgfPrincipal.pgFiltros

		.poSigSyPar.mLerCfg()

		.pgfPrincipal.pgFiltros.getCfgGrade.Visible = (Usuar = [4CONTROL])

		* Grade de Tipos de Operações
		lcQuery = [Select Codigos, Descrs, GerGdmis From SigCdTom]

		If Not fExecCmds(.poDataMgr, [crSigCdTom], lcQuery)
			Return .F.
		Endif

		Select 0 As nMarcas, ;
			Str(a.Codigos,2) As CCodigos, Codigos, ;
			Padr(Str(a.Codigos,2) + [-] + a.Descrs, 50) As CDescricaos ;
			From crSigCdTom a ;
			Order By 2 ;
			Into Cursor crTpOpe Readwrite
		Use In crSigCdTom

		* Grade de Grande Grupo de Produtos
		lcQuery = [Select Codigos, Descs From SigCdGpr]

		If Not fExecCmds(.poDataMgr, [crSigCdGpr], lcQuery)
			Return .F.
		Endif

		Select 1 As nMarcas, a.Codigos, ;
			a.Codigos + [-] + a.Descs As CDescricaos ;
			From crSigCdGpr a ;
			Order By 2 ;
			Into Cursor crGrdGpr Readwrite
		Use In crSigCdGpr

		* Grade de Grupo de Produtos
		lcQuery = [Select Mercs, CGrus as Codigos, DGrus as Descs From SigCdGrp]

		If Not fExecCmds(.poDataMgr, [crSigCdGrp], lcQuery)
			Return .F.
		Endif

		Select 1 As nMarcas, a.Mercs, a.Codigos, ;
			a.Codigos + [-] + a.Descs As CDescricaos ;
			From crSigCdGrp a ;
			Order By 2 ;
			Into Cursor crGrdGru Readwrite
		Use In crSigCdGrp
		
		If ThisForm.poDataMgr.SqlExecute([Select 0 as nMarcas, * From SigCdOpe],'CrOpera') < 1
			Messagebox('Favor reinicializar o processo !!!',16,'Falha na Conexão (CrOpera)')
			Return(.F.)
		EndIf
		Select CrOpera
		Index On Dopes Tag Dopes
		Index on TipoOps Tag IdxTipoOps
		Go Top
			
		* Carregando Cursor com as Empresas
		If Thisform.poDataMgr.SQLExecute([select Convert(bit,0) as Imps, cemps, razas from SigCdEmp Order by cEmps],'crSigCdEmp') < 1 Then
			=Messagebox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (crSigCdEmp)')
			Return(.F.)
		Endif
		Select crSigCdEmp

			* Carregando Cursor com as Empresas por acesso
			lcQuery = [Select Convert(bit,0) as Imps, a.Emps as Cemps, b.razas From SigCdAce a join SigCdEmp b on a.emps = b.cemps Where a.Usuarios = ']+Usuar+[']
			If Thisform.poDataMgr.SQLExecute(lcQuery,'crAuxCdEmp') < 1 Then
				=Messagebox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (crAuxCdEmp)')
				Return(.F.)
			Endif
			Select crAuxCdEmp
			
			If Reccount([crAuxCdEmp])>0
				Select crSigCdEmp
				Zap 
				
				Select crAuxCdEmp
				Scan
					Scatter Memvar 
					
					Select crSigCdEmp
					Append Blank 
					Gather Memvar 
					
					Select crAuxCdEmp
				EndScan 
			EndIf 
		EndIf 
		
		Select crSigCdEmp
		Index On cEmps Tag cEmps
		Go Top

		.MontaGrades()

		=DoDefault()

		lcQuery = [Select Distinct a.Grupo as GruEmps, b.Tipos ] + ;
			[From SigCdEmp a, SigCdGre b ] + ;
			[Where a.Grupo = b.Grupo And ] + ;
			[a.CEmps In (Select g.Emps From SigCdAce g Where g.Usuarios = '] + Usuar + [')]

		If Not fExecCmds(.poDataMgr, [crSigCdGre], lcQuery)
			Return .F.
		Endif

		lcQuery = [Select CMoes, DMoes, nOrdRels ] + ;
			[From SigCdMoe ] + ;
			[Order By nOrdRels]

		If Not fExecCmds(.poDataMgr, [LocalMoe], lcQuery)
			Return .F.
		Endif
		Go Top In LocalMoe
		.pgfPrincipal.pgFiltros.getCMoes.Value = LocalMoe.CMoes
		.pgfPrincipal.pgFiltros.getDMoes.Value = LocalMoe.DMoes

		.pgfPrincipal.pgFiltros.getDcI.SetFocus
		
	Else
		=Messagebox([Impossível Efetuar Conexão Com o Servidor de Banco de Dados...], 0+48, [CreateObject])
		Return .F.
	Endif
Endwith
* Fim **************************************************************************************
