* PROCEDURE chkimp
------------------------------------------------------------
PROCEDURE chkimp
LParameters Imag, nShape

Select Tmp_Imagens
If ThisForm.Primeira <= RecCount()
	Go ThisForm.Primeira
Else
	Go Bottom
	If Not Eof()
		Skip
	EndIf
EndIf
Item = ThisForm.Item - nShape
Item = Item + 1
Skip -Item

Imp = IIf(Tmp_Imagens.flag = 0,1,0)
Replace Tmp_Imagens.flag With Imp
Imag.Check1.Value = Imp

If Imp = 0
	Select CrImagens
	Locate for CrImagens.cpros = Tmp_Imagens.cpros
	If Found()
		Delete
	EndIf
Else
	lcBarra = fAltBar2de5(TransForm(Tmp_Imagens.cBars,'@L 99999999'))

	Insert Into CrImagens(cpros,dpros,linhas,colecoes,pvens,moevs,pesoms,moepvs,fvendas,pcuss,moecs,reffs,ifors,;
						datas,cgrus,cunis,figjpgs,flag,tconvs,mconvs,imppreco,custofs,moecusfs,impcusto,impcompo,;
						impreffor,cbars) ;
				Values (Tmp_Imagens.cpros,Tmp_Imagens.dpros,Tmp_Imagens.linhas,Tmp_Imagens.colecoes,;
						Tmp_Imagens.pvens,Tmp_Imagens.moevs,Tmp_Imagens.pesoms,Tmp_Imagens.moepvs,;
						Tmp_Imagens.fvendas,Tmp_Imagens.pcuss,Tmp_Imagens.moecs,Tmp_Imagens.reffs,;
						Tmp_Imagens.ifors,Nvl(Tmp_Imagens.datas,Ctod('')),Tmp_Imagens.cgrus,Tmp_Imagens.cunis,;
						Tmp_Imagens.figjpgs,Tmp_Imagens.flag,Tmp_Imagens.tconvs,Tmp_Imagens.mconvs,;
						Tmp_Imagens.imppreco,Tmp_Imagens.custofs,Tmp_Imagens.moecusfs,0,0,0,lcBarra)
EndIf
Select Tmp_Imagens
If ThisForm.Primeira <= RecCount()
	Go ThisForm.Primeira
Else
	Go Bottom
	IF Not Eof()
		Skip
	EndIf
EndIf
ENDPROC
