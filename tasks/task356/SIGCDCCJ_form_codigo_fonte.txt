PROCEDURE recalcular
Parameters pMsg

Local nOldSele
nOldSele = Select()

If Used([CsQtdDt])
	Use In CsQtdDt
Endif

If Empty(Thisform.Pagina.Dados.Get_DataBase.Value)
	If pMsg
		Messagebox([Data Base não informada.],0+48,[Aviso])
		Thisform.Pagina.Dados.Get_DataBase.SetFocus
	Endif
	Return .F.
Endif

ldDBase = Thisform.Pagina.Dados.Get_DataBase.Value
lnFator = Thisform.Pagina.Dados.Get_Fator.Value

If pMsg
	If Messagebox([Deseja Recalcular os Valores?],4+32,[Atenção]) <> 6
		Return .F.
	Endif
Endif

Select crDetalhe
Scan

	lnDias  = Ttod(crDetalhe.DataS) - Ttod(ldDBase)
	
	If (lnDias > 0) And Thisform.Pagina.Dados.optDias.Value = 2
		llSab = .T.
		llDom = .T.
		llFer = .T.

		ldDI1 = Ttod(ldDBase)
		ldDI2 = Ttod(crDetalhe.DataS)
		Do While (ldDI1 <= ldDI2)
			If fChkFeriado(Thisform.poDataMgr, ldDI1, llSab, llDom)
				lnDias = lnDias - 1
			Endif
			ldDI1 = ldDI1 + 1
		Enddo
	Endif

	lnValor = crDetalhe.Valor
	lnLiq   = Round(lnValor - (lnValor*((lnDias/30*(lnFator/100)))),2)
	
	If Abs(lnDias)>999
		=MessageBox([Quantidade de dias superior a 999 dias.],0+48,[Atenção])
		Replace Datas With Ctod('') in crDetalhe
		Return .F.
	EndIf 
	
	Select crDetalhe
	Replace Dias 	With lnDias
	Replace Liquido With lnLiq

Endscan

Select Count(*) As Qtd, Sum(Valor) As Valor, Sum(Liquido) As Liquido, Avg(Dias) As MedDias ;
	From crDetalhe ;
	Where Not Empty(Dias) ;
	Into Cursor CsQtdDt Readwrite

Select CsQtdDt
Thisform.Pagina.Dados.Get_Qtde.Value = Nvl(CsQtdDt.Qtd,0)
Thisform.Pagina.Dados.Get_Qtde.Refresh

Thisform.Pagina.Dados.Get_Media.Value = Nvl(CsQtdDt.MedDias,0)
Thisform.Pagina.Dados.Get_Media.Refresh

Thisform.Pagina.Dados.Get_TotLiq.Value = Nvl(CsQtdDt.Liquido,0)
Thisform.Pagina.Dados.Get_TotLiq.Refresh

Thisform.Pagina.Dados.Get_Total.Value = Nvl(CsQtdDt.Valor,0)
Thisform.Pagina.Dados.Get_Total.Refresh

If Used([CsQtdDt])
	Use In CsQtdDt
Endif

Select crDetalhe
Go Top

Thisform.Pagina.Dados.GradeDetalhe.Refresh

Select(nOldSele)
Return .T.

ENDPROC
