* PROCEDURE msv_procurar
------------------------------------------------------------
PROCEDURE msv_procurar
Select crSigCdGre
Index On grupo+class tag gruclass
Set Near On
=Seek(ThisForm.Pagina.Dados.Get_Grupo.Value+ThisForm.Pagina.Dados.Get_Class.Value,'crSigCdGre','gruclass')
Set Near Off

=DoDefault()
ENDPROC

------------------------------------------------------------
* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
lcQryGruEmp = [Select * From SigCdGre Order By Grupo,Class ]

ThisForm.pCol = 1

With ThisForm
	.poDataMgr = CreateObject('fSqlConector', ThisForm.Name)

	If (.poDataMgr.pnIdconn > 0)
		With .poDataMgr
			.AddCursor('SigCdGre', 'cIdChaves', 'crSigCdGre', '', ThisForm.Pagina.Lista.Grade, lcQryGruEmp)
			.pfSqlTabela(1).pColuna('grupo', '', '', 'Grupo',  115, .T.)
			.pfSqlTabela(1).pColuna('class','', '','Classificação', 115, .F.)
		EndWith

		=DoDefault()

		ThisForm.Pagina.Lista.Grade.Column1.SetFocus
	Else
		Messagebox('Impossivel Efetuar Conexão com Servidor de Banco de Dados...',48,'Create Object')
		Return(.F.)
	EndIf
EndWith
* Fim **************************************************************************************
ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Lista.Grupo_op.Click
------------------------------------------------------------
PROCEDURE Pagina.Lista.Grupo_op.Click
LParameters opcao

If !dodefault( opcao )
	ThisForm.mAtivaPagina1
	Return
Endif

Do Case
	thisform.pagina.dados.Get_Grupo.enabled = .F.
	thisform.pagina.dados.Get_Class.enabled = .F.

	Case Inlist( ThisForm.pcEscolha, 'PROCURAR', 'INSERIR' )
		thisform.pagina.dados.Get_Grupo.enabled = .T.
		thisform.pagina.dados.Get_Class.enabled = .T.
		thisform.pagina.dados.Get_Grupo.setfocus	
	Case Inlist( ThisForm.pcEscolha, 'CONSULTAR', 'EXCLUIR' )
		thisform.pagina.dados.grupo_salva.salva.setfocus
Endcase

ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
If ThisForm.pcEscolha = 'INSERIR'
   If IsEmpty(ThisForm.Pagina.Dados.Get_Grupo.Value)
	    MessageBox( 'Obrigatório Informar o Grupo.', 48, '' )
        ThisForm.Pagina.Dados.Get_Grupo.SetFocus
		Return
	Endif
   If IsEmpty(ThisForm.Pagina.Dados.Get_Class.Value)
	    MessageBox( 'Obrigatório Informar uma Classificação.', 48, '' )
        ThisForm.Pagina.Dados.Get_Class.SetFocus
		Return
	EndIf
	lcCHave=ThisForm.Pagina.Dados.Get_Grupo.Value+ThisForm.Pagina.Dados.Get_Class.Value
	If ThisForm.poDataMgr.ChkRegister('SigCdGre','gruclass',lcChave)
		MessageBox( 'Grupo + Classificação já cadastrado ...', 48, '' )
		ThisForm.Pagina.Dados.Get_Class.SetFocus
		Return
	EndIf
	Replace cIdCHaves With fUniqueIds()
	Replace gruclass  With lcChave
Endif

=DoDefault()

ThisForm.mAtivaPagina1
ENDPROC
