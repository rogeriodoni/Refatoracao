PROCEDURE impressão
LParameters lcSaida

Create Cursor TmpCabec (Titulo c(100), SubTitulo c(100), Empresa c(100))

ThisForm.poDataMgr.CursorQuery('SigCdEmp', 'crSigCdEmp', 'Cemps', _Empr, 'Razas')

lcTitulo 	= 'Falhas por Fase'
lcSubTitulo = 'Material : ' + Alltrim(This.Get_Mat.Value) + ' - ' + Alltrim(This.Get_DMat.Value) + Chr(13) + ;
			  'Período : ' + Dtoc(This.Get_Datai.Value) + ' - ' + Dtoc(This.Get_Dataf.Value)
lcEmpresa   = Iif(Eof('crSigCdEmp'), '', crSigCdEmp.Razas)

Insert Into TmpCabec (Empresa, Titulo, SubTitulo) Values (lcEmpresa, lcTitulo, lcSubTitulo)

Select TmpSaldo
Set Order to SeqFase
Go Top
If Eof() 
	=Messagebox('Nenhuma Informação Selecionada!!!', 32)
Else
	If (lcSaida = 'V')
		if thisform.optImp.Value = 3 && Grupo + Conta + Balanço
			Report Form SigOpCf2 Preview NoConsole
		Else
			Report Form SigOpCfa Preview NoConsole
		Endif
	Else
		If (lcSaida = 'I')
			if thisform.optImp.Value = 3 && Grupo + Conta + Balanço
				Report Form SigOpCf2 To Print Prompt
			Else
				Report Form SigOpCfa To Print Prompt
			Endif
		Else

			lcArq = PutFile([Nome da Planilha], [], [XLS;TXT])

			If fVerificaPasta(lcArq, 2)
				If (Lower(Alltrim(JustExt(lcArq)))=='xls')
					Copy To (lcArq) Type Xls
				Else 
					Copy To (lcArq) Delimited With "" With Character ';'
				EndIf 

				If File(lcArq)
					=MessageBox('Planilha Gerada Com Sucesso!!!', 0+48+32, '')
				EndIf
			Else
				=MessageBox('Erro na Geração da Planilha,' + Chr(10) + 'Verificar Diretório ou Planilha Já Aberta...', 0+48+32, '')
			EndIf
		Endif
	EndIf
EndIf
Use In TmpCabec
* Fim **************************************************************************************

ENDPROC

* OBJETO: SIGOPCFA.Processa
* Classe: commandbutton (Base: commandbutton)
* Registro: 14
================================================================================

------------------------------------------------------------
* PROCEDURE Click
------------------------------------------------------------
PROCEDURE Click
Private pDtI, pDtF

lcCodMat  = ThisForm.Get_Mat.Value
ldDataIni = ThisForm.Get_Datai.Value
ldDataFim = ThisForm.Get_Dataf.Value
lcGrupo   = ThisForm.Get_Grupo.Value
lcConta   = Thisform.Get_Conta.value

If Empty(ThisForm.Get_Mat.Value)
	=Messagebox('O Produto Deve Ser Informado!!!', 32, '')
	ThisForm.Get_Mat.SetFocus
	Return 0
EndIf

If (ldDataFim < ldDataIni)
	=Messagebox('Data Final Deve Ser Maior Que a Data Inicial!!!', 32, '')
	ThisForm.Get_Mat.SetFocus
	Return 0
EndIf

Select TmpSaldo 
Set Order To Fase
Zap

pDtI = fDtoSQL(ldDataIni)
pDtf = fDtoSQL(ldDataFim, '23:59:59')

lcQuery = [Select a.Emps, a.Codigos, a.Grupos, b.CPros, b.Entradas, b.Saldos, ] + ;
				 [b.Saidas, b.Pesagems, b.FReals, b.FAdmins, b.cIdChaves, a.Contas, c.Rclis, b.IndProds, a.Datas ] + ;
		    [From SigCdFcx a, SigOpCfe b, SigCdCli c ] + ;
		   [Where a.Emps = '] + _Empr + [' And ] + ;
		   		 [a.Datas Between ?pDtI And ?pDtF And ] + ;
		   		 [a.Emps = b.Emps And ] + ;
		   		 [a.Codigos = b.Codigos And ] + ;
		   		 [b.Cpros = '] + lcCodMat + [' And a.Contas = c.Iclis ] + ;
		   [Order By a.Emps, a.Codigos, a.Grupos, b.CPros, b.Entradas, b.Saldos, ] + ;
				 [b.Saidas, b.Pesagems, b.FReals, b.FAdmins, b.cIdChaves, a.contas, c.Rclis ]

If (ThisForm.poDataMgr.SqlExecute(lcQuery, 'crSigCdFcx') < 1)
	=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (crSigCdFcx)')
	Return 0
EndIf

Select crSigCdFcx
Scan
	If (Not Empty(lcGrupo) And CrSigCdFcx.Grupos <> lcGrupo ) Or (Not Empty(lcConta) And CrSigCdFcx.Contas <> lcConta )
		Loop
	Endif

	lStrQuery = [Select Setors From SigCdCle Where ClaGrupos = ']+crSigCdFcx.Grupos+[' And ClaContas = ']+CrSigCdFcx.ContaS+[']
	If (ThisForm.poDataMgr.SqlExecute(lStrQuery,'CrTmpClaes') < 1)
		=MessageBox('Favor reinicializar o processo!!!', 16, 'Falha na Conexão (CrTmpClaes)')
		Return 0
	EndIf
	
	lcGrupo2 = CrSigCdFcx.Grupos
	Select CrTmpClaes
	If Reccount() > 0
		lcGrupo2 = CrTmpClaes.Setors
	Endif

	lcChave = lcGrupo2
	Do Case
		Case thisform.optImp.Value = 2 && Grupo + Conta
			lcChave = lcGrupo2 + CrSigCdFcx.Contas
		Case thisform.optImp.Value = 3 && Grupo + Conta + Balanço
			lcChave = lcGrupo2 + CrSigCdFcx.Contas + Str(CrSigCdFcx.Codigos,6)
	EndCase

	If Not Seek(lcChave, 'TmpSaldo')
		Insert Into TmpSaldo (Fase,Conta,Rclis,Codigos,Datas) Values (lcGrupo2,;
		IIF(Inlist(thisform.optImp.Value,2,3),CrSigCdFcx.Contas,''),;
		IIF(Inlist(thisform.optImp.Value,2,3),CrSigCdFcx.Rclis,''), IIF(thisform.optImp.Value = 3,CrSigCdFcx.Codigos,0),IIF(thisform.optImp.Value = 3,CrSigCdFcx.Datas,Ctod('')))
	EndIf

	Replace Entradas With Entradas + crSigCdFcx.Entradas + crSigCdFcx.Saldos, ;
			Saidas   With Saidas   + crSigCdFcx.Saidas, ;
			FReals   With FReals   + crSigCdFcx.FReals, ;
			FAdmins  With FAdmins  + crSigCdFcx.FAdmins, ;
			Pesagems With Pesagems + crSigCdFcx.Pesagems ;
			IndProds With IndProds + (crSigCdFcx.IndProds - crSigCdFcx.FReals) In TmpSaldo

	x = Iif(TmpSaldo.Saidas = 0,0,(TmpSaldo.fReals / TmpSaldo.Saidas * 100))
	Replace pFalhas With x In TmpSaldo

EndScan

Select TmpSaldo
Append Blank
Replace Seqs With '1'
Sum FReals, FAdmins, Saidas, Entradas, Pesagems, IndProds To _FReal, _FAdm, _Saida, _Ent, _Psg, _IndProd
Append Blank
Replace Seqs     With '2', ;
		Fase     With 'Total', ;
		FReals   With _FReal, ;
		FAdmins  With _FAdm, ;
		Saidas   With _Saida, ;
		Entradas With _Ent,;
		Pesagems With _psg,;
		IndProds With _IndProd

x = Iif(TmpSaldo.Saidas = 0,0,(TmpSaldo.fReals / TmpSaldo.Saidas * 100))
Replace pFalhas With x In TmpSaldo

Replace all fTotals With fReals - fAdmins

Set Order To SeqFase
Go Top
If Eof()
	=Messagebox('Nenhuma Falha Selecionada!!!', 32)
EndIf
ThisForm.Grade.Refresh
ThisForm.Grade.SetFocus
ENDPROC
