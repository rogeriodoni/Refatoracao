PROCEDURE mctrlbotoes
LParameters plHabilita

	plHabilita = .F.

		EndIf
	EndFor

	.cmdMultiSubN.Enabled = plHabilita

------------------------------------------------------------
* PROCEDURE Click
------------------------------------------------------------
PROCEDURE Click
If Thisform.cmdInserir.Visible Or Thisform.cmdExcluir.Visible
	If ( Thisform.lHouveExcl Or Thisform.lHouveIns Or ThisForm.lHouveAlt )
		Select CrSigMvCxs
		Go Top
		Go bottom
		Select cNivels,sum(1) From CrSigMvCxs Into Cursor Selecao Group by cNivels having sum(1) > 1
		Select Selecao
		If Reccount() > 0
			MessageBox('Existem Operações Duplicadas!!!',0+16,'')
			Return .f.
		Endif

		Select [ ] as Agrupar,Sum(1) From CrSigMvCxs Into Cursor Selecao where Not Empty(cNivels) And Seqs = 0 group by 1
		Select Selecao
		If Reccount() > 0
			MessageBox('Existem Operações sem Sequência informada!!!',0+16,'')
			Return .f.
		Endif

		m.lnErro1 = SigCdOpe.poDatamgr.pnIdConn

		With SigCdOpe
			If m.lnErro1>0 
				m.lnErro1 = Iif(.poDataMgr.UpDate('crSigMvCxs'),1,-1)
			EndIf 
			If m.lnErro1>0 
				m.lnErro1 = .poDataMgr.Commit()
			EndIf 
			If Not m.lnErro1>0 
				.poDataMgr.RollBack()
			EndIf 
		EndWith 
		If Not m.lnErro1>0 
			= MessageBox('Erro na Gravação dos Dados do Cadastro de Operações.',48)
		EndIf 
	EndIf
ENDIF

Thisform.ParentForm.Enabled = .T.
Thisform.mCtrlBotoes(.T.)
ThisForm.Release
ENDPROC
