------------------------------------------------------------
PROCEDURE Init
Lparameters pcTipo

** Tipo ( = 0 - Carac Produtos, = 1 - Carac Contas, = 2 - Movimentações ) para manter compatibilidade
ThisForm.pnTipo = Iif(Type([pcTipo])<>[C],0,Iif(pcTipo == 'C',1,2))
Do case
Case( ThisForm.pnTipo = 1 )
	ThisForm.Caption = [Complementos de Características de Contas]
	ThisForm.Name    = [SIGCDMPPC]

Case( ThisForm.pnTipo = 2 )
	ThisForm.Caption = [Complementos de Características de Movimentações]
	ThisForm.Name    = [SIGCDMPPO]
	
OtherWise
	ThisForm.Caption = [Complementos de Características de Produtos]
	ThisForm.Name    = [SIGCDMPPP]
EndCase

With ThisForm
	.poDataMgr = CreateObject('fSqlConector', ThisForm.Name)

	lcQuery = [Select * from SigPrCpP where nTipos = ]+Str(ThisForm.pnTipo,1)

	If (.poDataMgr.pnIdconn > 0)
		With .poDataMgr
			.AddCursor('SigPrCpP', 'cidchaves', 'crSigPrCpP', '', ThisForm.Pagina.Lista.Grade,lcQuery)
			.pfSqlTabela(1).pColuna('Codigos', '', '', 'Código',    150, .t.)
			.pfSqlTabela(1).pColuna('Descrs',  '', '', 'Descrição', 290, .t.)
		EndWith

		=DoDefault()

		ThisForm.Pagina.Lista.Grade.Column1.SetFocus
	Else
		=Messagebox('Impossível Efetuar Conexão Com o Servidor de Banco de Dados...', 0+48, 'CreateObject')
		Return .f.
	EndIf
EndWith
* Fim **************************************************************************************
ENDPROC

PROCEDURE msv_procurar
Select crSigPrCpP
Locate for Codigos >= ThisForm.Pagina.Dados.GetCodigos.Value
* Fim **************************************************************************************
ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Lista.Grupo_op.Click
------------------------------------------------------------
PROCEDURE Pagina.Lista.Grupo_op.Click
Lparameters Opcao

If Not DoDefault(Opcao)
	ThisForm.mAtivaPagina1
	Return 0
EndIf

Do Case
	Case InList(Thisform.pcEscolha, 'INSERIR', 'PROCURAR')
		Thisform.Pagina.Dados.getCodigos.Setfocus

	Case (Thisform.pcEscolha = 'ALTERAR')
		Thisform.Pagina.Dados.getDescrs.Setfocus

	Case InList(Thisform.pcEscolha, 'CONSULTAR', 'EXCLUIR')
		Thisform.Pagina.Dados.Grupo_Salva.Salva.Setfocus
EndCase

Select crSigPrCpP
* Fim **************************************************************************************
ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
If Inlist(Thisform.Pagina.Lista.Grupo_op.Value, 1, 3)
	If Empty(Thisform.Pagina.Dados.getCodigos.Value)
		=Messagebox('Código Inválido!!!', 0+64, '')
		Thisform.Pagina.Dados.getCodigos.SetFocus
		Return 0
	Else
		If (Thisform.Pagina.Lista.Grupo_op.Value = 1)
			lcQuery = [Select * From SigPrCpP ] + ;
				[Where Codigos = '] + crSigPrCpP.Codigos + [' ] + ;
				[And nTipos = ]+Alltrim(Str(Thisform.pnTipo))
				
			If (Thisform.poDataMgr.SqlExecute(lcQuery, 'crBusca') < 1)
				=Messagebox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (crBusca)')
				Return 0
			EndIf
			
			If Not Eof('crBusca')
				=Messagebox('Complemento de Característica Já Cadastrado!!!', 0+64, '')
				Thisform.Pagina.Dados.getCodigos.SetFocus
				Return 0
			Endif
			Replace nTipos With Thisform.pnTipo In crSigPrCpP
		Endif
	Endif

	If Empty(Thisform.Pagina.Dados.GetDescrs.Value)
		=Messagebox('Descricao Inválida!!!', 0+64, '')
		Thisform.Pagina.Dados.GetDescrs.SetFocus
		Return 0
	Endif
	
Endif

If Thisform.pcEscolha = 'INSERIR'
	Replace CidChaves With fUniqueIds() In crSigPrCpP
EndIf 

If Inlist(Thisform.pcEscolha, 'EXCLUIR')
	If Thisform.pnTipo = 1
		lcQuery = [Select * From SigClCrc ] + ;
			[Where Compls = '] + Thisform.Pagina.Dados.GetDescrs.Value + [']

		If (Thisform.poDataMgr.SqlExecute(lcQuery, 'crSigClCrc') < 1)
			=Messagebox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (crSigClCrc)')
			Return 0
		Endif

		If (Reccount('crSigClCrc') > 0)
			=Messagebox('Complemento de Característica Sendo Utilizada!!!', 16, '')
			Return 0
		Endif
	Endif
	lcQuery = [Select CGrus ] + ;
		[From SigCdFtc ] + ;
		[Where CTits = '] + Thisform.Pagina.Dados.getCodigos.Value + [' ] + ;
		[And nTipos = ]+Alltrim(Str(Thisform.pnTipo))

	If (Thisform.poDataMgr.SqlExecute(lcQuery, 'SigCdFtc') < 1)
		=Messagebox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (SigCdFtc)')
		Return 0
	Endif

	If (Reccount('SigCdFtc') > 0)
		=Messagebox('Complemento Sendo Utilizado Como Título de Fichas Técnicas!!!', 16, '')
		Return 0
	Endif
Endif

Select crSigPrCpP

=DoDefault()

Thisform.mAtivaPagina1
* Fim **************************************************************************************
ENDPROC
