PROCEDURE Init
Private pIds, pNot

pIds = Replicate(Chr(254), 20)
pNot = .f.

With ThisForm
	.poDataMgr = CreateObject([fSqlConector], .Name)

	If (.poDataMgr.pnIdconn > 0)
		With .poDataMgr
			.AddCursor([SigCdFs], [Codigos], [crSigCdFs], [], ThisForm.Pagina.Lista.Grade)
			.pfSqlTabela(1).pColuna([Codigos], [], [], [Códigos],   080, .t.)
			.pfSqlTabela(1).pColuna([Descrs],  [], [], [Descrição], 290, .t.)
			.pfSqlTabela(1).pColuna([Tipos],   [], [], [Tipo],      040, .t.)

			.AddCursor([SigCdFsI], [cIdChaves], [crSigCdFsI])
			.AddCursor([SigCdFsO], [cIdChaves], [crSigCdFsO])
			.AddCursor([SigCdFsC], [cIdChaves], [crSigCdFsC])
		EndWith

		.Busca = 1

		* Carregando a Grade de Empresas
		lcQuery = [Select ?pNot as Imps, Cemps as Emps, Razas ] + ;
					[From SigCdEmp ] + ;
				   [Where Not Ativas = 2]

		If (.poDataMgr.SqlExecute(lcQuery, [csSigCdEmp]) < 1)
			=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (csSigCdEmp)])
			Return .f.
		EndIf
		Select csSigCdEmp
		Index On Emps Tag Emps
		Go Top In csSigCdEmp

		With .Pagina.Dados.grdEmps
			.ColumnCount       = 3
			.RecordSource      = [csSigCdEmp]
			.DeleteMark        = .f.
			.ReadOnly          = .f.
			.AllowHeaderSizing = .f.
			.AllowRowSizing    = .f.

			.Column1.ControlSource   = [csSigCdEmp.Imps]
			.Column1.Movable         = .f.
			.Column1.Resizable       = .f.
			.Column1.ReadOnly        = .f.
			.Column1.Sparse          = .f.
			.Column1.Check1.AutoSize = .t.

			.Column2.ControlSource   = [csSigCdEmp.Emps]
			.Column3.ControlSource   = [csSigCdEmp.Razas]

			For I=2 To .ColumnCount
				lcCol = Alltrim(Str(I))
				.Column&lcCol..Movable   = .f.
				.Column&lcCol..Resizable = .f.
				.Column&lcCol..ReadOnly  = .t.
			Next
			.SetAll([DynamicBackColor], [Iif(csSigCdEmp.Imps,Rgb(255,255,223),Rgb(255,255,255))],[Column])
		EndWith
		
		* Carregando a Grade de Operações
		lcQuery = [Select ?pNot as Imps, a.Dopes From SigCdOpe a inner join SigOpCdc b on b.Dopes = a.Dopes ] + ;
			[Where b.OpeInatvs <> 1 Order By a.Dopes ]

		If (.poDataMgr.SqlExecute(lcQuery, [csSigCdOpe]) < 1)
			=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (csSigCdOpe)])
			Return .f.
		EndIf
		Select csSigCdOpe
		Index On Dopes Tag Dopes
		Go Top In csSigCdOpe

		With .Pagina.Dados.grdOpers
			.ColumnCount       = 2
			.RecordSource      = [csSigCdOpe]
			.DeleteMark        = .f.
			.ReadOnly          = .f.
			.AllowHeaderSizing = .f.
			.AllowRowSizing    = .f.

			.Column1.ControlSource   = [csSigCdOpe.Imps]
			.Column1.Movable         = .f.
			.Column1.Resizable       = .f.
			.Column1.ReadOnly        = .f.
			.Column1.Sparse          = .f.
			.Column1.Check1.AutoSize = .t.

			.Column2.ControlSource   = [csSigCdOpe.Dopes]

			For I=2 To .ColumnCount
				lcCol = Alltrim(Str(I))
				.Column&lcCol..Movable   = .f.
				.Column&lcCol..Resizable = .f.
				.Column&lcCol..ReadOnly  = .t.
			Next
			.SetAll([DynamicBackColor], [Iif(csSigCdOpe.Imps,Rgb(255,255,223),Rgb(255,255,255))],[Column])
		EndWith
		
		* Carregando a Grade de Composições
		Select * From CrSigCdFsC Where 0=1 Into Cursor csSigPrCp ReadWrite
		Select csSigPrCp

		With .Pagina.Dados.GrdCompo
			.ColumnCount       = 1
			.RecordSource      = [CsSigPrCp]
			.DeleteMark        = .f.
			.RecordMark        = .f.
			.ReadOnly          = .f.
			.AllowHeaderSizing = .f.
			.AllowRowSizing    = .f.

			.Column1.ControlSource   = [CsSigPrCp.Compos]
			.Column1.Movable         = .f.
			.Column1.Resizable       = .f.
			.Column1.ReadOnly        = .f.
		EndWith

		** Cursor com as composições
		
		lcQuery = [Select Distinct Compos from SigCdPro]
		If (.poDataMgr.SqlExecute(lcQuery, [csCompos]) < 1)
			=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (csCompos)])
			Return .f.
		EndIf
		Select csCompos
		Index On Compos Tag Compos
		Go Top In csCompos		

		=DoDefault()

		.Pagina.Lista.Grade.Column1.SetFocus
	Else
		=Messagebox([Impossível Efetuar Conexão Com o Servidor de Banco de Dados...], 0+48, [CreateObject])
		Return .f.
	EndIf
EndWith
* Fim **************************************************************************************
ENDPROC

* PROCEDURE Pagina.Lista.Grupo_op.Click

PROCEDURE Pagina.Lista.Grupo_op.Click
Lparameters Opcao

If Not DoDefault(Opcao)
	ThisForm.mAtivaPagina1
	Return .f.
EndIf

Select crSigCdFs
Do Case
	Case Inlist(ThisForm.pcEscolha, [PROCURAR], [INSERIR])
		ThisForm.Pagina.Dados.GetCodigos.Setfocus	

	Case (ThisForm.pcEscolha = [ALTERAR])
		ThisForm.Pagina.Dados.getDescrs.Setfocus	

	Case Inlist(ThisForm.pcEscolha, [CONSULTAR], [EXCLUIR])
		ThisForm.Pagina.Dados.Grupo_Salva.Salva.Setfocus
EndCase

* Carregando a Grade de Empresas
Update csSigCdEmp Set Imps = .f.

lcQuery = [Select * ] + ;
		    [From SigCdFsI ] + ;
		   [Where Codigos = '] + crSigCdFs.Codigos + [']

If (ThisForm.poDataMgr.SqlExecute(lcQuery, [LocalEmps]) < 1)
	=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (LocalEmps)])
	Return .f.
EndIf

Select LocalEmps
Scan
	Update csSigCdEmp Set Imps = .t. Where Emps = LocalEmps.Emps
EndScan
Go Top In csSigCdEmp

** Carregando a grade de operações
Update csSigCdOpe set Imps = .F.

lcQuery = [Select * ] + ;
		    [From SigCdFsO ] + ;
		   [Where Codigos = '] + crSigCdFs.Codigos + [']

If (ThisForm.poDataMgr.SqlExecute(lcQuery, [LocalOpe]) < 1)
	=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (LocalOpe)])
	Return .f.
EndIf

Select LocalOpe
Scan
	Update csSigCdOpe Set Imps = .t. Where Dopes = LocalOpe.Dopes
EndScan
Go Top In csSigCdOpe

lcQuery = [Select * ] + ;
		    [From SigCdFsC ] + ;
		   [Where Codigos = '] + crSigCdFs.Codigos + [']

If (ThisForm.poDataMgr.SqlExecute(lcQuery, [LocalComP]) < 1)
	=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (LocalComP)])
	Return .f.
EndIf

Zap in csSigPrCp

Select LocalComP
Scan
	Insert Into CsSigPrCp (Cidchaves,Codigos,Compos) Values ;
		(LocalComP.Cidchaves,LocalComP.Codigos,LocalComP.Compos)
EndScan

Select crSigCdFs
ThisForm.Pagina.Dados.Refresh

* Fim **************************************************************************************
ENDPROC

* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
Select CsSigPrCp
Locate for .f.
Select csSigCdEmp
Locate for .f.
Select csSigCdOpe
Locate for .f.

If (ThisForm.pcEscolha = [INSERIR])
	If IsEmpty(ThisForm.Pagina.Dados.getCodigos.Value)
		MessageBox( 'Obrigatório Informar o Código!!!', 48, '' )
		ThisForm.Pagina.Dados.getCodigos.SetFocus
		Return
	EndIf

   	If ThisForm.poDataMgr.ChkRegister('SigCdFs','Codigos',ThisForm.Pagina.Dados.GetCodigos.Value)
		MessageBox( 'Código já cadastrado...', 48, '' )
		ThisForm.Pagina.Dados.GetCodigos.SetFocus
		Return
	EndIf
	
	If (ThisForm.Pagina.Dados.getPerfins.Value < ThisForm.Pagina.Dados.getPerInis.Value) 
		MessageBox('Data final deve ser maior ou igual a data inicial...',48, '')
		ThisForm.Pagina.Dados.getPerFins.SetFocus
		Return
	Endif
EndIf

If InList(ThisForm.pcEscolha, [INSERIR], [ALTERAR])
	Select Compos, Sum(1) as Tt From CsSigPrCp Group by 1 having Sum(1) > 1 Into Cursor csSelecao
	Select csSelecao
	If Reccount('csSelecao') > 0
		MessageBox('Dados Inconsistentes, composição em duplicidade!!!',0+16,'')
		Return .f.
	EndIf
Endif

If (ThisForm.pcEscolha = [EXCLUIR])
	lcQuery = [Select b.Grupos, b.Codigos ] + ;
			    [From SigCdCmt a, SigCdClc b ] + ;
			   [Where a.Codigos = '] + crSigCdFs.Codigos + [' And ] + ;
			   		 [a.fkChaves = b.cIdChaves]

	If (ThisForm.poDataMgr.SqlExecute(lcQuery, [LocalBusca]) < 1)
		=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (LocalBusca)])
		Return 0
	EndIf
	Go Top In LocalBusca
	If Not Eof([LocalBusca])
		=MessageBox([O Motivo Está Em Uso No Cadastro de Classificação!!!] + Chr(13) + ;
					[Grupo : ] + LocalBusca.Grupos + [ - Código : ] + LocalBusca.Codigos, ;
					0+16, [Exclusão Não Permitida!!!])
		Return 0
	EndIf
EndIf

If InList(ThisForm.pcEscolha, [INSERIR], [ALTERAR], [EXCLUIR])
	lcQuery = [Delete From SigCdFsI Where Codigos = '] + crSigCdFs.Codigos + [']
	If (ThisForm.poDataMgr.SqlExecute(lcQuery, []) < 1)
		=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (Delete crSigCdFsI)])
		Return .f.
	EndIf
	
	lcQuery = [Delete From SigCdFsO Where Codigos = '] + crSigCdFs.Codigos + [']
	If (ThisForm.poDataMgr.SqlExecute(lcQuery, []) < 1)
		=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (Delete crSigCdFsO)])
		Return .f.
	EndIf

	lcQuery = [Delete From SigCdFsC Where Codigos = '] + crSigCdFs.Codigos + [']
	If (ThisForm.poDataMgr.SqlExecute(lcQuery, []) < 1)
		=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (Delete crSigCdFsC)])
