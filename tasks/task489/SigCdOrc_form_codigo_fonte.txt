PROCEDURE recalculavalores
Local cArea

ThisForm.LockScreen = .T.

cArea = Alias()

Select crGrupos
Go Top in crGrupos
Scan
	If( crGrupos.Perc <> 0 ) Then
		Replace Valor With Round(ThisForm.Pagina.Dados.Get_nVlTots.Value * (crGrupos.Perc/100),2) in crGrupos		
	Endif
EndScan

Select crContas
Set Key To ('') in crContas
Go Top in crContas
Scan
	If( crContas.Perc <> 0 ) Then
		Replace Valor With Round(ThisForm.Pagina.Dados.Get_nVlTots.Value * (crContas.Perc/100),2) in crContas
	Endif
EndScan

Select (cArea)

Go Top in crGrupos
Set Key To crGrupos.Codigos in crContas
Go Top in crContas

ThisForm.Pagina.Dados.Grd_Grupos.Refresh 
ThisForm.Pagina.Dados.Grd_Contas.Refresh  

ThisForm.LockScreen = .F.
ENDPROC

PROCEDURE rettotlancado
Local nVlLancado

nVlLancado = 0

Select Sum(Valor) as TotLancado from crContas into cursor crTotContas

PROCEDURE Init
LPARAMETERS _acesso
Set Step On 
WITH Thisform
	.poDatamgr = CreateObject('fSqlConector',Thisform.Name)

	If Not .poDatamgr.pnIdconn>0
		Messagebox('Impossivel Efetuar Conexão com Servidor de Banco de Dados.',(0+48),'Erro')
		Return(.f.)
	Endif

	lcQryCdmp  = [Select cEmps, cMes, cAno, cTitulo, Moeda, nVlTots, Max(cIdChaves) as cIdChaves from SigCdMrp Group by cEmps, cMes, cAno, cTitulo, Moeda, nVlTots]
	lcQryCdmpD = [Select * From SigCdMrp Where cEmps = ?crSigCdMrp.cEmps and cMes = ?crSigCdMrp.cMes and cAno = ?crSigCdMrp.cAno and cTitulo = ?crSigCdMrp.cTitulo and Moeda = ?crSigCdMrp.Moeda]
	
	With .poDatamgr
		.AddCursor('SigCdMrp','cIdChaves','crSigCdMrp','',ThisForm.pagina.lista.grade,lcQryCdmp)
		.pfSqlTabela(1).pColuna('cEmps','','','Empresa',45,.T.)
		.pfSqlTabela(1).pColuna('cAno','','','Mês/Ano',59,.t.)
		.pfSqlTabela(1).pColuna('cTitulo','','','Titulo',290,.t.)
		.pfSqlTabela(1).pColuna('Moeda','','','Moeda',40,.f.)
		.AddCursor('SigCdMrp','cIdChaves','crCdmpDados','',.f.,lcQryCdmpD)
	EndWith

	ThisForm.Pagina.Lista.Grade.Column2.ControlSource = [Alltrim(crSigCdMrp.cMes)+Iif(!IsEmpty(crSigCdMrp.cMes),'/','')+crSigCdMrp.cAno]
	
	If .poDataMgr.SQLExecute([Select Codigos, Descrs, 999999999.99 as Valor, 999.999999 as Perc from SigCdGcr Order by Codigos],'crGrupos') < 1 Then
		=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (crGrupos)')

	ThisForm.Pagina.Dados.Grd_Grupos.Column4.ControlSource = [crGrupos.Perc]
	
Endwith	

=DoDefault( _Acesso )

PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click

Local Array laCampos(5,2)

Go Top in crGrupos
Go Top in crContas

If InList(ThisForm.pcEscolha,'INSERIR','ALTERAR')
	If Thisform.pcEscolha='INSERIR'
		m.laCampos(1,1) = 'cEmps'
		m.laCampos(1,2) = ThisForm.Pagina.Dados.Get_Emps.Value
		m.laCampos(2,1) = 'cMes'
		m.laCampos(2,2) = ThisForm.Pagina.Dados.Get_Mes.Value
		m.laCampos(3,1) = 'cAno'
		m.laCampos(3,2) = ThisForm.Pagina.Dados.Get_Ano.Value
		m.laCampos(4,1) = 'cTitulo'
		m.laCampos(4,2) = ThisForm.Pagina.Dados.Get_Titulo.Value
		m.laCampos(5,1) = 'Moeda'
		m.laCampos(5,2) = ThisForm.Pagina.Dados.Get_Cd_Moeda.Value  
	
		If 	ThisForm.podatamgr.ChkRegister('SigCdMrp',@m.laCampos)
			Messagebox( 'Previsão já cadastrada!', 48, '' )
			ThisForm.Pagina.Dados.Get_Titulo.SetFocus
			Return
		Endif
	EndIf	
	
	Select Codigos as cGrupos, Valor, Perc, 'G' as Tipo, fUniqueIds() as cIdChaves from crGrupos where Valor <> 0 or Perc <> 0 into cursor crPrevAux readwrite
	Select crPrevAux
	
	Select iClis as cGrupos, Valor, Perc, 'C' as Tipo, fUniqueIds() as cIdChaves from crContas where Valor <> 0 or Perc <> 0 into cursor crPrevAux2 
	Select crPrevAux2
	Go Top
	Scan
		Scatter MemVar
		Insert into crPrevAux from MemVar
	EndScan
	Use in ('crPrevAux2')
	
*!*		IF( crLancado.TotLancado <> ThisForm.Pagina.Dados.Get_nVlTots.Value ) Then
*!*			MessageBox('Total da Previsão diferente das previsões lançadas nos grupos/contas, verifique!',48,'')
*!*			Return(.F.)
*!*		Endif 

	* Grupos
	Select cIdChaves from crCdmpDados where cGrupos not in (Select cGrupos from crPrevAux where Tipo = 'G') into cursor crDeleta
	Select crDeleta
	Scan
		If SQLExec(ThisForm.poDataMgr.pnIdConn,[Delete from SigCdMrp where cIdChaves = ']+crDeleta.cIdChaves+[']) < 1 Then
			=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (crDeleta)')
			Return 0			
		Endif
	EndScan
	
	* Contas
	Select cIdChaves from crCdmpDados where iClis not in (Select cGrupos from crPrevAux where Tipo = 'C') into cursor crDeleta
	Select crDeleta
	Scan
		If SQLExec(ThisForm.poDataMgr.pnIdConn,[Delete from SigCdMrp where cIdChaves = ']+crDeleta.cIdChaves+[']) < 1 Then
			=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (crDeleta)')
			Return 0			
		Endif
	EndScan	

	Zap in crCdmpDados
	Select crPrevAux
	Scan
		Select cGrupos, iClis from crCdmpDados where cGrupos = Iif(crPrevAux.Tipo = 'G',crPrevAux.cGrupos,'') and ;
												    iClis = Iif(crPrevAux.Tipo = 'C', crPrevAux.cGrupos, '') into cursor crAux
		Select crAux
		If Eof()												    
			Insert into crCdmpDados(cEmps,cMes,cAno,cTitulo,Moeda,cGrupos,iClis,nValors,cIdChaves,nPercs,nVlTots) Values(;
								   ThisForm.Pagina.Dados.Get_Emps.Value,;	
								   ThisForm.Pagina.Dados.Get_Mes.Value,;
								   ThisForm.Pagina.Dados.Get_Ano.Value,;
								   ThisForm.Pagina.Dados.Get_Titulo.Value,;
								   ThisForm.Pagina.Dados.Get_cd_Moeda.Value,;
								   Iif(crPrevAux.Tipo = 'G', crPrevAux.cGrupos, ''),;
								   Iif(crPrevAux.Tipo = 'C', crPrevAux.cGrupos, ''),;
								   crPrevAux.Valor,;
   								   crPrevAux.cIdChaves,;
   								   crPrevAux.Perc,;
   								   ThisForm.Pagina.Dados.Get_nVlTots.Value)
		Else
			Update crCdmpDados set nValors = crPrevAux.Valor, nPercs = crPrevAux.Perc, nVlTots = ThisForm.Pagina.Dados.Get_nVlTots.Value ;
							   where cGrupos = Iif(crPrevAux.Tipo = 'G',crPrevAux.cGrupos,'') and iClis = Iif(crPrevAux.Tipo = 'C', crPrevAux.cGrupos, '')
		Endif   								   
	EndScan
		
EndIf

If InList(ThisForm.pcEscolha,'EXCLUIR')
	Select crCdmpDados
	Go Top
	Scan
		If SQLExec(ThisForm.podatamgr.pnIdConn,[Delete from SigCdMrp where cIdChaves = ']+crCdmpDados.cIdChaves+[']) < 1 Then
			=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (crDeleta)')
			Return 0			
		Endif
	EndScan
Endif
=DoDefault()

ThisForm.mAtivapagina1
