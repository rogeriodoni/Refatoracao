PROCEDURE Init
lcQryOpt = [Select * From SigCdPen]

With ThisForm
	.poDataMgr = CreateObject('fSqlConector', ThisForm.Name)

	If (.poDataMgr.pnIdconn > 0)
		With .poDataMgr
			.AddCursor('SigCdPen', 'cIdChaves', 'crSigCdPen', '', ThisForm.Pagina.Lista.Grade, lcQryOpt)
			.pfSqlTabela(1).pColuna('Grupos', '', '', 'Grupo', 80, .t.)
			.pfSqlTabela(1).pColuna('Contas', '', '', 'Conta', 80, .t.)
			.pfSqlTabela(1).pColuna('ddis', '', '', 'Dia Inicial', 80, .f.)
			.pfSqlTabela(1).pColuna('ddfs', '', '', 'Dia Final', 80, .f.)
			.pfSqlTabela(1).pColuna('penas', '', '', '%Penalidade', 80, .f.)
		EndWith

		=DoDefault()

		ThisForm.Pagina.Lista.Grade.Column1.SetFocus
	Else
		=Messagebox('Impossivel Efetuar Conexão com Servidor de Banco de Dados...', 0+48, 'CreateObject')
		Return .f.
	EndIf
EndWith

ENDPROC

PROCEDURE Pagina.Lista.Grupo_op.Click
LPARAMETERS opcao
If !dodefault( Opcao )
	ThisForm.mAtivapagina1
	Return
Endif

Select crSigCdPen
* posiciona o cursor de acordo com a opção
Do Case
	Case Inlist( ThisForm.pcEscolha, 'INSERIR', 'PROCURAR' )
		Thisform.pagina.dados.Get_Grupos.setfocus
	Case ThisForm.pcEscolha = 'ALTERAR'
		Thisform.pagina.dados.Get_Grupos.setfocus
	Case Inlist( ThisForm.pcEscolha, 'CONSULTAR', 'EXCLUIR' )
		Thisform.pagina.dados.grupo_salva.salva.setfocus
Endcase
ThisForm.Pagina.Dados.Refresh

ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
With ThisForm.Pagina.Dados
	If InList(ThisForm.pcEscolha ,'INSERIR','ALTERAR')
		If Empty(.get_Grupos.Value)
			MessageBox( 'Obrigatório Informar o Grupo!!!', 48, '' )
			.get_grupos.SetFocus
			Return
		Endif
		If .get_ddf.Value = 0
			MessageBox( 'Dia Final não pode ser = 0 !!!', 48, '' )
			.get_ddf.Setfocus
			Return
		Endif
	EndIf
	If InList(ThisForm.pcEscolha,'INSERIR')
		Replace CidChaves with fUniqueIds() in CrSigCdPen
	EndIf
	
	If InList(ThisForm.pcEscolha,'ALTERAR')
		lcSql = [Select * From SigCdPen where Grupos = ']+CrSigCdPen.Grupos+[' And Contas = ']+CrSigCdPen.Contas+[']
		If Thisform.Podatamgr.Sqlexecute(lcSql,'LocalPena') < 1
			MessageBox('Favor reinicializar o processo.Falha na Conexão (LocalPena)',0+48,'')
			Return .f.
		EndIf

		Select LocalPena
		Index on Grupos + Contas + Str(Ddis,4) + Str(ddfs,4) + Str(Penas,6,2) Tag GruCon
		If Seek (CrSigCdPen.Grupos+CrSigCdPen.Contas+Str(CrSigCdPen.ddis,4)+Str(CrSigCdPen.ddfs,4) + Str(CrSigCdPen.penas,6,2) )
			Messagebox('Já Existe um lançamento com estes dados !!',0+48,'')
			.Get_grupos.Setfocus		
			Return .f.
		Endif
	Endif
EndWith

=DoDefault()
ThisForm.mAtivapagina1
ENDPROC
