* PROCEDURE fazcontra
------------------------------------------------------------
PROCEDURE fazcontra
Select CrSigMvCcr
noper  = CrSigMvCcr.nopers
voper  = CrSigMvCcr.vopers
oper   = CrSigMvCcr.sopers
grupo  = CrSigMvCcr.sgrupos
conta  = CrSigMvCcr.scontas
moed   = CrSigMvCcr.smoedas
cot    = CrSigMvCcr.scotacaos
valor  = CrSigMvCcr.svalors
soper  = CrSigMvCcr.opers
sgrupo = CrSigMvCcr.grupos
sconta = CrSigMvCcr.contas
smoed  = CrSigMvCcr.moedas
scot   = CrSigMvCcr.cotacaos
svalor = CrSigMvCcr.valors
nlanc  = CrSigMvCcr.nlancs
vlanc  = CrSigMvCcr.vlancs
data   = CrSigMvCcr.datas
dataa  = CrSigMvCcr.datalts
valpag = CrSigMvCcr.valpags
emp    = CrSigMvCcr.emps
hist   = CrSigMvCcr.hists
Hist2  = CrSigMvCcr.Hist2s
dope   = CrSigMvCcr.dopes
nume   = CrSigMvCcr.numes
venc   = CrSigMvCcr.vencs
valpag = CrSigMvCcr.valpags * scot / cot

Insert Into CrSigMvCcr ( VOpers, NOpers, NLancs, VLancs, Opers, Grupos,;
		Contas, Moedas, Cotacaos, Valors, Emps, Datas, usualts, datalts,;
		sgrupos, scontas, smoedas, scotacaos, svalors, sopers, autos,;
		hists, dopes, numes, valpags, vencs, Hist2s ) ;
	Values ( vOper, nOper, nLanc, vLanc, Oper, Grupo, Conta, Moed, Cot,;
		Valor, Emp, Data, Usuar, Dataa, sGrupo, sConta, sMoed, sCot,;
		sValor, sOper, .t., hist, Dope, Nume, valpag, venc, Hist2 )

ENDPROC

PROCEDURE montatela
If Used('CrSigMvSlc')
	ThisForm.Grd_Saldo.RecordSource = ''
	Use In CrSigMvSlc
EndIf
lcGrupo = ThisForm.Container1.Get_grupo.Value

------------------------------------------------------------
* PROCEDURE Click
------------------------------------------------------------
PROCEDURE Click
With ThisForm.Container1
	If Empty(.Get_grupo.Value)
		MessageBox('Obrigatorio Informar o Grupo de Filtro!!!',48,'')
		Return 0
	EndIf
	If Empty(.Get_Grupoc.Value)
		MessageBox('Grupo de Contra-Partida não Informado!!!',48,'')
		Return 0
	EndIf
	If Empty(.Get_ContaC.Value)
		MessageBox('Conta de Contra-Partida não informada!!!',48,'')
		Return 0
	EndIf
	If Empty(.Get_Dope.Value)
		MessageBox('Operação de Mov.Estoque não cadastrada nos Parametros do Sistema!!!',48,'')
		Return 0
	EndIf
	lcGrupo  = .Get_Grupo.Value
	lcGrupoC = .Get_Grupoc.Value
	lcContaC = .Get_ContaC.Value
EndWith

Select CrSigMvSlc
Count to x For Marcas = 1
If x = 0
	MessageBox('Nenhuma Conta Selecionada!!!!',16,'')
	Return 0
Endif

lcDope = ThisForm.Container1.Get_Dope.Value

Select crSigMvCcr
ZAP

Select crSigMvCab
Zap

Select CrSigMvSlc

Scan
	If Marcas = 0
		Loop
	Endif

	Replace Flag With 1 in CrSigMvSlc
	ThisForm.Grd_saldo.Refresh

	If !Seek(_Empr + lcDope,'CrSigMvCab','EmpDopNum')
		_Inicio = fGerUniqueKey(alltrim(lcDope)+_Empr)
		_IniLet = alltrim( fGerMascara( _Inicio ) )
		Insert Into CrSigMvCab ( Emps, Dopes, Numes, Usuars, Datars, Datas, MascNum, EmpDopNums, CidChaves, DtAlts ) ;
						Values ( _Empr, lcDope, _Inicio, Usuar, Datetime(), Date(),  ;
							_IniLet, _Empr+lcDope+Str(_inicio,6), fUniqueIds(), DATETIME() )
	EndIf

	_Hist  = 'Encerramento de conta - Grupo:'+Alltrim(lcGrupo)
	_hist2 = 'com Contra-Partida:'+Alltrim(lcGrupoc)+'-'+Alltrim(lcContaC)
	
	xOper  = Iif(CrSigMvSlc.Saldos > 0,[C],[D])
	xsOper = Iif(xOper=[C],[D],[C])
	lnValor= Abs(CrSigMvSlc.Saldos)
	
	_noper = fGerUniqueKey( 'SigMvCcr' )
	_voper = CrSigMvCab.Emps + Alltrim( str( _noper, 9 ) )

	lnCota = fBuscarCotacao(CrSigMvSlc.Moedas,Date(), ThisForm.Podatamgr )

	Select CrSigMvCcr
	Append Blank
	Replace NOpers    With _noper,;
			VOpers    With _voper,;
			Emps      With _Empr,;
			Datas     With Datetime(),;
			usualts   With usuar,;
			datalts   With Datetime(),;
			Vencs     With Datetime(),;
			hists     With _hist,;
			opers     With xOper,;
			sopers    With xsOper,;
			moedas    With CrSigMvSlc.Moedas,;
			cotacaos  With lnCota,;
			autos     With .t.,;
			grupos    With CrSigMvSlc.Grupos,;
			contas    With CrSigMvSlc.Contas,;
			sgrupos   With lcGrupoc,;
			scontas   With lcContac,;
			smoedas   With CrSigMvSlc.Moedas,;
			scotacaos With lnCota,;
			dopes     With CrSigMvCab.Dopes,;
			numes     With CrSigMvCab.Numes,;
			Hist2s    With _Hist2,;
			valors    With lnValor,;
			svalors   With lnValor

	ThisForm.FazContra()

EndScan

Select CrSigMvCcr
Scan
	Replace CidChaves  With Dtos(crSigMvCcr.Datas)+Transform(fGerUniqueKey(Dtos(crSigMvCcr.datas)),"@L 999999") + ThisForm.Sigkey ,;
			gruconmoes with crSigMvCcr.grupos + crSigMvCcr.contas + crSigMvCcr.moedas,;
			EmpDopNums With crSigMvCcr.Emps + crSigMvCcr.Dopes + Str(crSigMvCcr.Numes,6)

	=fRecalculaS( CrSigMvCcr.Grupos,  CrSigMvCcr.Contas,  CrSigMvCcr.Datas, CrSigMvCcr.Moedas,  ThisForm.poDataMgr)

EndScan

llTaOk = .t.
If Not ThisForm.poDataMgr.Update('crSigMvCcr')
	=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (Update - crSigMvCcr)')
	llTaOk = .f.
EndIf

If llTaOk
	If Not ThisForm.poDataMgr.Update('crSigMvCab')
		=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (Update - crSigMvCab)')
		llTaOk = .f.
	EndIf
EndIf
If llTaOk And Not fRecalculaS( .t., ThisForm.poDataMgr)
	=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (SigOpClS)')
	llTaOk = .f.
Endif

If llTaOk
	Thisform.Podatamgr.Commit()
Else
	ThisForm.poDataMgr.RollBack()
	=MessageBox('Ocorreu Um Erro Com o Processamento!!! Favor Reinicializar o Processo!!!', 16, 'Processamento Cancelado')
EndIf

ThisForm.Iniciar()

ENDPROC
