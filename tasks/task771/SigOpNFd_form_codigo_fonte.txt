PROCEDURE atucur
****** atualiza cursor *****
LOCAL lnUltSeq

lcEDN = crSigmvNfi.empdopnums
lcQuery = [Select * From SigMvNfd Where empdopnums = '] + lcEDN + [' and nfeEve = '] + thisform.Evento + [' order by nfeseq desc]
If (ThisForm.ParentForm.poDataMgr.SqlExecute(lcQuery, [crSigMvNfd]) < 1)
	=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (crSigMvNfd)])
	llRet=.F.
EndIf
lnUltSeq=IIF(EOF('crSigMvNfd'), 1, crSigMvNfd.nfeseq+1)

SELECT crSigMvNfd
*!*	nfedte with DATETIME()

IF thisform.evento = 'CAR' or EOF('crSigMvNfd')
	APPEND BLANK
	replace cidchaves with fUniqueIds(), ;
			nfis with crsigmvnfi.nfis,;
			dopes with crsigmvnfi.dopes,;
			numes with crsigmvnfi.numes,;
			emps with crsigmvnfi.emps,;
			empdopnums with crsigmvnfi.empdopnums,;
			nfedte with CTOT(thisform.cntData.getData.Value),;
			nfedesc with '',;
			nfeeve with thisform.Evento,;
			nfeseq with lnUltSeq in crSigMvNfd
ENDIF

thisform.cntNFe.getNfe.value = crsigmvnfi.nfis
thisform.cntSeq.getSeq.value = lnUltSeq
thisform.cntObservacao.getDesc.value = crsigmvnfd.nfedesc


ThisForm.Refresh

ENDPROC

		EndIf
		If llOks
			If (ThisForm.ParentForm.poDataMgr.Commit() < 1)
				=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (Commit)])
				lOk = .f.
			EndIf
		EndIf
	ELSE
		**** verifica e mostra erros retornados
		lcDest=lcPathNFe+lcPRetorno+cId+"-"+lcSeq2+"-env-cce-ret.err"
		IF FILE(lcDest)
			cLinha = FILETOSTR(lcDest)
			cCampo="NF:" + SUBSTR(cId, 26, 9) + Chr(10) + alltrim(cLinha)
			DELETE FILE &lcDest
		EndIf
		Messagebox(STRCONV(cCampo,11), 0+64, [Retorno da SEFAZ])
	ENDIF

ENDIF

thisform.atucur()	&& atualiza cursor
RETURN lOk

ENDPROC
