PROCEDURE montagrade
Local lnOldSel

Select crItensBor 
Index on Cpros Tag Cpros
Go top

With ThisForm.grdItens
	.RecordSource = [crItensBor]
	.column1.ControlSource = [crItensBor.Cpros]
	.column2.ControlSource = [crItensBor.Dpros]
	.column3.ControlSource = [crItensBor.Marcas]
	.column4.ControlSource = [crItensBor.qtds]
	.column5.ControlSource = [crItensBor.Valor]
	.column6.ControlSource = [crItensBor.Pesos]
	.column7.ControlSource = [crItensBor.CodTams]
	.Refresh()
EndWith 

ThisForm.grdItens.Column1.SetFocus
ThisForm.grdItens.Refresh
ENDPROC

PROCEDURE Init
lParameters pCnx, pEscolha

ThisForm.pcEscolha = pEscolha
ThisForm.poDataMgr = pCnx

If Used([csProdItn])
	Use in csProdItn
EndIf 

Create Cursor csProdItn(cItens n(4),Cpros c(14), codtams c(3), Valor n(15,6), Pesos n(8,3))

ThisForm.grdItens.RecordSource = []

Select crTpmMvItn
Select 0 as Marcas, Cpros, Dpros, CodTams, Sum(Qtds) as Qtds, Max(Units) as Valor, Max(Pesos) as Pesos from crTpmMvItn ;
Where Not Empty(Cpros) Group By Cpros, CodTams, Dpros Into Cursor crItensBor  ReadWrite

PROCEDURE AfterRowColChange
LPARAMETERS nColIndex

lcArquivo2 = Sys(2023) + [\] + goSistema.Sys2015Tmp + Sys(2015) + [.jpg]

Select crItensBor 
= ThisForm.poDataMgr.CursorQuery('SigCdPro','crTPro','Cpros',crItensBor .Cpros,'FigJpgs')
If Not IsNull(crTPro.FigJpgs) And Not IsEmpty(crTPro.FigJpgs)
	This.parent.ImgJpg.Visible = .F.
	This.parent.ImgJpg.Picture = ''
	Select crTPro
	Clear Resources
	Copy Memo crTPro.FigJpgs to (lcArquivo2)
	This.parent.ImgJpg.Picture = (lcArquivo2)
	This.parent.ImgJpg.Visible = .T.
Else
	This.parent.ImgJpg.Visible = .F.
	This.parent.ImgJpg.Picture = ''
EndIf

ENDPROC

PROCEDURE btnConfirmar.Click
