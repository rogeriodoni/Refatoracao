PROCEDURE Valid
If Not Empty(This.Value) And Not Lastkey() = 27
	If ThisForm.pComis
		lcQuery = [Select * ] + ;
				    [From SigCmCab ] + ;
				   [Where Cods = '] + Padr(This.Value, 40) + [']

		If (ThisForm.poDataMgr.SqlExecute(lcQuery, [LocalBus]) < 1)
			=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (LocalBus)])
			Return 0
		EndIf
		Go Top In LocalBus
		If Not Eof([LocalBus])
			=MessageBox([Descrição do Cadastro de Comissão Já Informado!!!], 0+16, [Atenção!!!])
			Return .f.
		EndIf
	Else
		ThisForm.poDatamgr.Sqlexecute([Select Processos from SigPcOop Where Processos = '] + Padr(this.Value, 20) +['],'LocalBus')
		Select LocalBus
		If ! Eof()
			MessageBox('Processo Já Cadastrado!!!',16,'Atenção!!!')
			Return .f.
		EndIf
	EndIf 
EndIf
ENDPROC

PROCEDURE CmdCopia.Click
If Empty(ThisForm.GetDestino.Value)
	MessageBox('Destino Invalido !!!',16,'')
	Return
EndIf

lcOpeOri = ThisForm.GetOrigem.Value
lcOpeDes = ThisForm.GetDestino.Value

If ThisForm.pComis
	lcQuery = [Select * ] + ;
			    [From SigCmCab ] + ;
			   [Where Cods = '] + Padr(Thisform.getOrigem.Value, 40) + [']

	If (ThisForm.poDataMgr.SqlExecute(lcQuery, [LocalCmCab]) < 1)
		=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (LocalCmCab)])
		Return 0
	EndIf
	Go Top In LocalCmCab
	If Not Eof([LocalCmCab])
		lcQuery = [Select * ] + ;
				    [From SigCmItn ] + ;
				   [Where fkChaves = '] + Padr(LocalCmCab.pkChaves, 20) + [']

		If (ThisForm.poDataMgr.SqlExecute(lcQuery, [LocalCmItn]) < 1)
			=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (LocalCmItn)])
			Return 0
		EndIf
		Go Top In LocalCmItn

		lcKey = fUniqueIds()
		Select LocalCmCab
		Scatter Memvar Memo
		Insert Into crSigCmCab From Memvar
		Replace CODS 	 With ThisForm.getDestino.Value, ;
				PKCHAVES With lcKey, ;
				USUARS 	 With Usuars, ;
				DATAS 	 With Datetime() In crSigCmCab

		Select LocalCmItn
		Scan
			Scatter Memvar Memo
			Insert Into crSigCmItn From Memvar
			Replace FKCHAVES With lcKey, ;
					PKCHAVES With fUniqueIds() In crSigCmItn
		EndScan

		ThisForm.poDatamgr.UpDate([crSigCmCab])
		ThisForm.poDatamgr.UpDate([crSigCmItn])
	EndIf
Else
	lnNDopes = fGerUniqueKey('SigPcOoP')

	Zap In crSigPcOop
	Zap In crSigPcOoI

	ThisForm.poDatamgr.SqlExecute([Select * From SigPcOop Where Processos = ']+lcOpeOri+['],'csSigPcOop')

	Select csSigPcOop
	Scatter MemVar Memo
	m.Processos = lcOpeDes
	m.Codigos   = lnNDopes

	Select crSigPcOop
	Append Blank
	Gather MemVar Memo

	ThisForm.poDatamgr.SqlExecute([Select * From SigPcOoI Where Processos = ']+lcOpeOri+['],'csSigPcOoI')

	Select csSigPcOoI
	Scan
		Scatter MemVar Memo
		m.Processos = lcOpeDes
		m.cIdChaves = fUniqueIds()
		m.Codigos   = lnNDopes

		Select crSigPcOoI
		Append Blank
		Gather MemVar Memo
	EndScan

	fGravarLog([C], Thisform.Name, [COPIA], crSigPcOop.Processos)

	ThisForm.poDatamgr.UpDate('crSigPcOop')
	ThisForm.poDatamgr.UpDate('crSigPcOoI')
EndIf

ThisForm.poDatamgr.Commit()

ThisForm.Release
ENDPROC
