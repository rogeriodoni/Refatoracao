* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
LParameters poForm, pnDataSes

ThisForm.cntSombra.lblSombra.Caption = ThisForm.Caption
ThisForm.cntSombra.lblTitulo.Caption = ThisForm.Caption

Set DataSession to pnDataSes

Thisform.ParentForm          = poForm
Thisform.ParentForm.Enabled  = .F.

Thisform.GravaDados = .f.

With Thisform
	.cmdInserir.Visible = fChecaAcesso('SIGCDFPP','INSERIR')
	.cmdInserir.Enabled = fChecaAcesso('SIGCDFPP','INSERIR')
	.cmdExcluir.Visible = fChecaAcesso('SIGCDFPP','EXCLUIR')
	.cmdExcluir.Enabled = fChecaAcesso('SIGCDFPP','EXCLUIR')
	.label1.Caption = '[ '+AllTrim(crTSigfPag.fpags)+' ]'
EndWith

Select CrSigOpFpB
Zap

lcSql = [Select * From SigOpFpB Where Fpags = ']+crTSigfPag.Fpags+[' Order by CodBins ]
If ThisForm.ParentForm.PodataMgr.Sqlexecute(lcSql,'xFpagI') < 1
	MessageBox('Falha na Conexão. Favor Reiniciar o Processo!!!',0+16,'')
	Return .f.
EndIf

Select xFpagI
Scan
	scatter to memvar
	Insert Into CrSigOpFpB From array Memvar
EndScan

Select xFpagI
Index on CodBins Tag CodBins
Set Order to
Goto Top

With Thisform.GradeFPagto
	.RecordSource          = 'xFPagi'
	.Column1.ControlSource = 'xFPagi.CodBins'
	.Column2.ControlSource = 'xFPagi.Descs'
	.Column3.ControlSource = 'xFPagi.DataIs'
	.Column4.ControlSource = 'xFPagi.DataFs'
	.Refresh
EndWith

THisForm.GradeFPagto.Refresh
ENDPROC

------------------------------------------------------------
* PROCEDURE Click
------------------------------------------------------------
PROCEDURE Click
Retorno = .t.
If Thisform.cmdInserir.Visible And ThisForm.GravaDados
	Select CodBins,Sum(1) From xFPagi group by CodBins having sum(1) > 1;
	Into Cursor TT
	Select TT
	If Reccount('TT') > 0
		MessageBox('Dados Inconsistentes. Codigo BIN em duplicidade!!!',0+16,'')
		Return .f.
	Endif
	
	Select crSigOpFpB
	Delete all
	Select xFPagi
	Scan
		If !IsEmpty(xFPagi.CodBins)
			Scatter to memvar
			Insert Into crSigOpFpB From array memvar
			SELECT crSigOpFpB
			Replace cidChaves WITH fUniqueIds()
			SELECT xFPagi
		EndIf
	EndScan
	Retorno=TSigfpg.podatamgr.Update("crSigOpFpB")
	Retorno=TSigfpg.podatamgr.commit()
EndIf
Thisform.ParentForm.Enabled = .T.
ThisForm.Release

ENDPROC
