PROCEDURE Valid
If ! Empty(This.Value)
	
	lcQuery = [SELECT b.Cidchaves,a.Nenvs,  a.EmpDnPs, b.Datas, b.Grupoos, b.Contaos, b.grupods, b.contads, ] + ;
			  [a.servicos, a.Nops,a.cmats, c.dpros, a.Qtds, c.cunis, a.peso2s, c.cunips, ]+;
			  [a.Pesos, Space(14) as Cpros ] + ;
			  [from SigCdNei a, SigCdNec b, SigCdPro c ] + ;
			  [where a.nenvs = ]+Alltrim(Str(This.Value))+[ and a.empdnps = b.empdnps and a.cmats = c.cpros ] + ;
			  [order by b.cidchaves,a.Nops,a.cmats ]

	If ThisForm.poDataMgr.SqlExecute(lcQuery,'TmpNensi') < 1
		=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (mSv_Excluir - SigCdNei)')
		Return .f.
	EndIf
		
	Select TmpNensi
	Go Top
	If Eof()
		MessageBox('Envelope Não Localizado !!!',16,'')
		Return
	EndIf 
	Select EmpDnPs, Sum(Pesos) as Pesos From TmpNensi Where Not Servicos;
	Into Cursor TmpNens Group by 1
	
	Select a.*, b.Pesos as TotPesos From TmpNensi a, TmpNens b Where ;
	a.EmpDnPs = b.EmpDnPs Into Cursor CrSigCdNei Order by a.CidChaves,a.Nops,a.Cmats
	
	ThisForm.MakeTreeEnv()
	
Endif
ENDPROC

PROCEDURE Valid
If ! Empty(This.Value)
	
	lcQuery = [SELECT distinct b.CidChaves,a.Nops as Nenvs, a.EmpDnPs, b.Datas, ]+;
			  [b.Grupoos, b.Contaos, b.grupods, b.contads, ]+;
			  [a.servicos,  0 as Nops,a.cmats, c.dpros, a.Qtds, ]+;
			  [c.cunis, a.peso2s, c.cunips, a.Pesos, d.Cpros, d.NopMaes ]+;
			  [from SigCdNei a, SigCdNec b, SigCdPro c, SigOpPic d ]+;
			  [where a.Nops = ]+Alltrim(Str(This.Value))+[ And a.empdnps = b.empdnps And ]+;
			  [a.cmats = c.cpros And a.Nops = d.Nops ]+;
			  [Union all ]+;
			  [SELECT distinct b.CidChaves,a.Nenvs, a.EmpDnPs, b.Datas, ]+;
			  [b.Grupoos, b.Contaos, b.grupods, b.contads, ]+;
			  [a.servicos,  0 as Nops,a.cmats, c.dpros, a.Qtds, ]+;
			  [c.cunis, a.peso2s, c.cunips, a.Pesos, d.Cpros, d.NopMaes ]+;
			  [from SigCdNei a, SigCdNec b, SigCdPro c, SigOpPic d ]+;
			  [where (a.Nenvs = ]+Alltrim(Str(This.Value))+[ And a.Nops <> ]+Alltrim(Str(This.Value))+[ ) And a.empdnps = b.empdnps And ]+;
			  [a.cmats = c.cpros And a.Nenvs = d.Nops ]+;
			  [order by b.cidchaves,a.Nops,a.cmats]

	If ThisForm.poDataMgr.SqlExecute(lcQuery,'TmpNensi') < 1
		=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (TmpNensi)')
		Return .f.
	Endif	
	Select TmpNensi
	Go Top
	If Eof()
		pNot = .f.
	
		lcQuery = [SELECT b.cidchaves,a.Nops as Nenvs, a.EmpDnPs, b.Datas, ]+;
				  [b.Grupoos, b.Contaos, b.grupods, b.contads, ]+;
				  [?pNot as servicos,  0 as Nops,'Nenhum        ' as cmats, 'Nenhum Material Agregado' as dpros, ]+;
				  [0 as Qtds, ' ' as cunis, 0 as peso2s, ' ' as cunips, 0 as Pesos, a.CodPds as Cpros, 0 as NopMaes ]+;
				  [from SigPdMvf a, SigCdNec b ]+;
				  [where a.Nops = ]+Alltrim(Str(This.Value))+[ and a.empdnps = b.empdnps ]+;
				  [order by b.cidchaves,a.Nops]

		If ThisForm.poDataMgr.SqlExecute(lcQuery,'TmpNensi') < 1
			=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (TmpNensi)')
			Return .f.
		Endif
		Select TmpNensi
		Go Top
		If Eof()
			MessageBox('O.P. Não Localizada !!!',16,'')
			Return
		Endif

	EndIf
	
	lcProduto = TmpNensi.Cpros
	lcSql = [Select cpros, dpros, FigJpgs From SigCdPro Where Cpros = ']+lcProduto+[']
	If Thisform.Podatamgr.Sqlexecute(lcSql,'CrTmpPro') < 1
		=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (CrTmpPro)')
		Return .f.
	Endif
	
	Select EmpDnPs, Sum(Pesos) as TotPesos From TmpNensi Where Not Servicos ;
	Into Cursor TmpNens Group by 1
	
	Select a.*, b.TotPesos From TmpNensi a, TmpNens b Where a.EmpDnps = b.EmpDnPs ;
	Into Cursor CrSigCdNei Order by a.CidChaves, a.Nops, a.Cmats

	lcNomArq = Sys(2015)
	lcArquivo = Sys(2023)+'\'+Alltrim(lcNomArq)+'.jpg'
	Thisform.Arqfigura = lcArquivo
	Clear Resources
	ThisForm.Imagem.Picture = ''
	ThisForm.Imagem.Visible = .f.

	If Not Empty(crTmpPro.FigJpgs) And Not IsNull(crTmpPro.FigJpgs)
		If (StrToFile(crTmpPro.FigJpgs, lcArquivo) > 0)
			ThisForm.Imagem.Picture = (lcArquivo)
			ThisForm.Imagem.Visible = .t.
		EndIf
	EndIf

	ThisForm.MakeTree()
	
Endif
ENDPROC

------------------------------------------------------------
PROCEDURE When
ThisForm.Imagem.Visible = .f.
*!*	Return(.t.)
ENDPROC

PROCEDURE Click
loTree1 = ThisForm.Container1.TreeView
llExpande = .f.

For lnNode = 0 To loTree1.lIstCount-1
	If ThisForm.pnNodeExp + 1 = loTree1.NodeLevel(lnNode)
		loTree1.NodeExpanded(lnNode) = .t.
		llExpande = .t.
	Endif
EndFor

If llExpande 
	ThisForm.pnNodeExp = ThisForm.pnNodeExp + 1
Endif

ENDPROC

PROCEDURE Click
If Type('ThisForm.pnNodeExp') = 'N' And ThisForm.pnNodeExp > 0
	loTree1 = ThisForm.Container1.TreeView

	For lnNode = 0 To loTree1.lIstCount-1
		If ThisForm.pnNodeExp = loTree1.NodeLevel(lnNode)
			loTree1.NodeExpanded(lnNode) = .f.
		Endif
	EndFor

	ThisForm.pnNodeExp = ThisForm.pnNodeExp - 1
EndIf
ENDPROC

PROCEDURE DblClick
Local lcArquivo

lcArquivo = Sys(2023) + '\' + Sys(2015) + '.Jpg'

Select crTmpPro
With This.Parent
	* Figura no Campo Memo
	If Not Empty(crTmpPro.FigJpgs) And Not IsNull(crTmpPro.FigJpgs)
		=StrToFile(crTmpPro.FigJpgs, lcArquivo)
	EndIf

	If File(lcArquivo)
		Do Form SigOpZom With lcArquivo, ;
							'Produto : ' + crTmpPro.CPros + ' - ' + crTmpPro.DPros, '' 
		Delete File (lcArquivo)
	EndIf
EndWith
* Fim *********************************************************************************

ENDPROC
