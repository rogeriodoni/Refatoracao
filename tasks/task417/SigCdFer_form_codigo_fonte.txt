* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
LPARAMETERS lcacesso

With ThisForm
	.poDatamgr = CreateObject('fSqlConector')
	With .poDatamgr
		.AddCursor('SigCdFer','cIdChaves','crSigCdFer','',ThisForm.pagina.lista.grade)
		.pfSqlTabela(1).pColuna('Datas','','','Datas',80,.t.)
		.pfSqlTabela(1).pColuna('Emps','','','Empresa',80,.t.)
		.pfSqlTabela(1).pColuna('Descrs','','','Descrição',220,.t.)
	ENDWITH
	= DoDefault()
	.LockScreen = .f.
EndWith

ENDPROC

ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Lista.Grupo_op.Click
------------------------------------------------------------
PROCEDURE Pagina.Lista.Grupo_op.Click
LPARAMETERS opcao

If !DoDefault( Opcao )
	ThisForm.mAtivapagina1
	Return
Endif

Select csEmpresa
Zap

Select CrSigCdFer

*** Mena - 13/09/2017 - foi incluido campo empresa para ser possivel incluir feriados por cidades ***
lcSql = [Select * From SigCdEmp Where Cemps = ']+CrSigCdFer.Emps+[']
ThisForm.poDataMgr.Sqlexecute(lcSql,'CrSigCdEmp')
thisform.pagina.dados.get_ds_empresa.Value = CrSigCdEmp.razas   

Do Case
	Case Inlist( ThisForm.pcEscolha, 'PROCURAR', 'INSERIR' )
		ThisForm.Pagina.Dados.getDatas.Setfocus	

	Case ThisForm.pcEscolha = 'ALTERAR'
		ThisForm.Pagina.Dados.getDescrs.Setfocus	

	Case Inlist( ThisForm.pcEscolha, 'CONSULTAR', 'EXCLUIR' )
		ThisForm.Pagina.Dados.Grupo_Salva.Salva.Setfocus
Endcase

This.parent.parent.Dados.cmdCopia.Visible = .F.
This.parent.parent.Dados.CntCopia.Visible = .F.

If ThisForm.pcEscolha = 'ALTERAR' 
	This.parent.parent.Dados.cmdCopia.Visible = .T.
EndIf 

Select CrSigCdFer
ThisForm.Pagina.Dados.Refresh

ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
If INLIST(ThisForm.pcEscolha,'INSERIR','ALTERAR')
   If IsEmpty(ThisForm.Pagina.Dados.getDatas.Value)
	  MessageBox( 'Obrigatório Informar a Data!!!', 48, '' )
	  ThisForm.Pagina.Dados.getDatas.SetFocus
	  Return
   Endif
   If Year(ThisForm.Pagina.Dados.getDatas.Value) < 1900
	  MessageBox( 'Data Invalida!!!', 48, '' )
	  ThisForm.Pagina.Dados.getDatas.SetFocus
	  Return
   Endif

	*** mena - 22/09/2017 - checa se data+empresa já estão cadastradas ***
	lcDat=SUBSTR(DTOS(TTOd(thisform.Pagina.Dados.getDatas.Value)),1,10)
	lcDat=SUBSTR(lcDat,1,4)+'-'+SUBSTR(lcDat,5,2)+'-'+SUBSTR(lcDat,7,2)
	lcSql = [Select * From SigCdFer Where emps = ']+CrSigCdFer.Emps+[' and LEFT(CONVERT(VARCHAR, datas, 120), 10) = ']+lcDat+;
			[' and cidchaves <> ']+CrSigCdFer.cidchaves+[']
	ThisForm.poDataMgr.Sqlexecute(lcSql,'CrAux')
	IF RECCOUNT('crAux')>0
		Messagebox( 'Data e Empresa já cadastradas.', 48,'Aviso' )
		ThisForm.Pagina.Dados.Getdatas.SetFocus
		Return
	ENDIF
ENDIF

Select CrSigCdFer
If Empty(cidchaves)
	replace cidchaves with fUniqueIds()
EndIf

*-- Rafael - 02/05/2018 - Replica o Feriado para as empresas Selecionadas na Grade
If ThisForm.pcEscolha = 'ALTERAR'

	ldDataFer = CrSigCdFer.Datas 
	lcDescFer = CrSigCdFer.Descrs
	lnChkFer  = CrSigCdFer.chktrbesc
	
	Select csEmpresa
	Go Top
	Scan For Marcas = 1
	
		Select CrSigCdFer
		Append Blank
		Replace Datas With ldDataFer, ;
				Emps  With csEmpresa.Empresa, ;
				Descrs With lcDescFer, ;
				chktrbesc With lnChkFer, ;
				cidchaves With fUniqueIds() in CrSigCdFer
				
		Select csEmpresa
	EndScan 
EndIf 

=DoDefault()
ThisForm.mAtivapagina1

ENDPROC

PROCEDURE Click
Select csEmpresa
Zap 

Private pDt
pDt = fDtoSql(CrSigCdFer.DataS)

lcSql = [select 0 as Marcas, cemps as Empresa, razas as Nome ] + ;
		[from sigcdemp ] + ;
		[where cemps Not in( Select emps From SigCdFer Where datas = ?pDt And Emps <> space(3) ) ] + ;
		[Order By cemps ]
If Thisform.poDataMgr.Sqlexecute(lcSql,'CrAux') < 1
	Messagebox( 'Favor Reiniciar o Processo.', 0+16,'Falha de Conexão (CrAux)' )
	Return .F.
EndIf 

Select CrAux
Go top 
Scan 
	Scatter Memvar 
	
	Select csEmpresa
	Append Blank 
	Gather Memvar
	
	Select CrAux
EndScan 

Select csEmpresa
Go Top

If Reccount('csEmpresa')>0

	ThisForm.pagina.dados.cntCopia.grdEmpresa.RecordSource = [csEmpresa]
	ThisForm.pagina.dados.cntCopia.grdEmpresa.column1.ControlSource = [csEmpresa.Empresa]
	ThisForm.pagina.dados.cntCopia.grdEmpresa.column2.ControlSource = [csEmpresa.Nome]
	ThisForm.pagina.dados.cntCopia.grdEmpresa.column3.ControlSource = [csEmpresa.Marcas]
	ThisForm.pagina.dados.cntCopia.grdEmpresa.Refresh()
	
	This.parent.CntCopia.Visible = .T.
	This.Enabled = .F.
	*This.parent.CntCopia.grdEmpresa.
Else
	MessageBox([As Empresas Cadastradas Já Possuem Esse Feriado Cadastrado.],0+64,[Atenção])
	Return .F.
Endif

ENDPROC
