------------------------------------------------------------
* PROCEDURE montaestoques
------------------------------------------------------------
PROCEDURE montaestoques
Private pCod
pCod = crSigLcLo.Codigos

Zap In crSigLcLoE
Zap In crEstoqueA
Zap In crEstoqueB

ThisForm.poDataMgr.Requery([crSigLcLoE])

If Not Empty(crSigLcLo.LocalA) && Carrega Grades de Estoque do LocalA
	ThisForm.poDataMgr.CursorQuery([SigCdLoc], [TmpLocal], [Cods], crSigLcLo.LocalA)
	Go Top In TmpLocal
	If Not Eof([TmpLocal]) 
		ThisForm.DriveA = Alltrim(TmpLocal.cIdConns)

		ThisForm.poDataAux = CreateObject([fSqlConector], ThisForm.DriveA)

		lcQuery = [Select Distinct a.Emps, a.Grupos, a.Estos as Contas ] + ;
					[From SigMvEst a ] + ;
				   [Where Not a.Emps = Space(3) And ] + ;
				   		 [Not a.Grupos = Space(10) And ] + ;
				   		 [Not a.Estos = Space(10)]

		If (ThisForm.poDataAux.SqlExecute(lcQuery, [LocalEst]) < 1)
			=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (LocalEst)])
			Return .f.
		EndIf

		Select LocalEst
		Scan
			Insert Into crEstoqueA (Marca, Emps, Grupos, Contas) Values (.f., LocalEst.Emps, LocalEst.Grupos, LocalEst.Contas)
		EndScan

		Select crSigLcLoE
		Scan
			If (crSigLcLoE.Locals = crSigLcLo.LocalA)
				Update crEstoqueA ;
				   Set Marca  = .t. ;
				 Where Emps   = crSigLcLoE.Emps And ;
				 	   Grupos = crSigLcLoE.Grupos And ;
				 	   Contas = crSigLcLoE.Contas
			EndIf
		EndScan
	EndIf
	Go Top In crEstoqueA
EndIf

If Not Empty(crSigLcLo.LocalB) && Carrega Grades de Estoque do LocalB
	ThisForm.poDataMgr.CursorQuery([SigCdLoc], [TmpLocal], [Cods], crSigLcLo.LocalB)
	Go Top In TmpLocal
	If Not Eof([TmpLocal]) 
		ThisForm.DriveB = Alltrim(TmpLocal.cIdConns)

		ThisForm.poDataAux = CreateObject([fSqlConector], ThisForm.DriveB)

		lcQuery = [Select Distinct a.Emps, a.Grupos, a.Estos as Contas ] + ;
					[From SigMvEst a ] + ;
				   [Where Not a.Emps = Space(3) And ] + ;
				   		 [Not a.Grupos = Space(10) And ] + ;
				   		 [Not a.Estos = Space(10)]

		If (ThisForm.poDataAux.SqlExecute(lcQuery, [LocalEst]) < 1)
			=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (LocalEst)])
			Return .f.
		EndIf

		Select LocalEst
		Scan
			Insert Into crEstoqueB (Marca, Emps, Grupos, Contas) Values (.f., LocalEst.Emps, LocalEst.Grupos, LocalEst.Contas)
		EndScan

		Select crSigLcLoE
		Scan
			If (crSigLcLoE.Locals = crSigLcLo.LocalB)
				Update crEstoqueB ;
				   Set Marca  = .t. ;
				 Where Emps   = crSigLcLoE.Emps And ;
				 	   Grupos = crSigLcLoE.Grupos And ;
				 	   Contas = crSigLcLoE.Contas
			EndIf
		EndScan
	EndIf
	Go Top In crEstoqueB
EndIf

ThisForm.Pagina.Dados.pgComplementos.pgEstoques.grdEstoqueA.Refresh
ThisForm.Pagina.Dados.pgComplementos.pgEstoques.grdEstoqueB.Refresh
* Fim **************************************************************************************
ENDPROC

* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
Private pCod

Local lcCod, ldDat, lcCod, lnVal, llGravar, lcBuscar

Select crSigLcLo

pCod = crSigLcLo.Codigos

If InList(Thisform.pcEscolha, 'INSERIR', 'ALTERAR')
	If Empty(Thisform.Pagina.Dados.cntCodigos.getCodigos.Value)
		=MessageBox('Código Inválido!!!', 0+48, '')
		Thisform.Pagina.Dados.cntCodigos.getCodigos.SetFocus
		Return 0
	EndIf 
		
	Select crSigLcLoI
	Scan
		If Empty(crSigLcLoI.Cods) And ;
		   (Not Empty(crSigLcLoI.Emps) 		Or Not Empty(crSigLcLoI.Cods) 	 Or ;
			Not Empty(crSigLcLoI.Operacoes) Or Not Empty(crSigLcLoI.GrupoOs) Or ;
			Not Empty(crSigLcLoI.ContaOs) 	Or Not Empty(crSigLcLoI.GrupoDs) Or ;
			Not Empty(crSigLcLoI.ContaDs))
			=MessageBox('Local Inválido!!!', 0+48, '')
			ThisForm.Pagina.Dados.pgComplementos.pgOpeGerA.grdOpeGerA.Column2.SetFocus
			Return 0
		EndIf 

		If Empty(crSigLcLoI.Cods)
			Delete In crSigLcLoI
		Else
			If Empty(crSigLcLoI.EnvRecs)
				=MessageBox('Selecione Enviar ou Receber!!!', 0+48, '')
				ThisForm.Pagina.Dados.pgComplementos.pgOpeGerA.optEnvRecs.Option1.SetFocus
				Return 0
			EndIf 

			If Empty(crSigLcLoI.Codigos)
				Replace Codigos With crSigLcLo.Codigos In crSigLcLoI
			EndIf

			If Empty(crSigLcLoI.cIdChaves)
				Replace cIdChaves With fUniqueIds() In crSigLcLoI
			EndIf
		EndIf
	EndScan
EndIf

If InList(Thisform.pcEscolha, 'INSERIR', 'ALTERAR', 'EXCLUIR')
	Zap In crSigLcLoE

	lcQuery = [Delete From SigLcLoE ] + ;
			   [Where Codigos = '] + crSigLcLo.Codigos + [']

	If (ThisForm.poDatamGR.SqlExecute(lcQuery, []) < 1)
		=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (Delete - SigLcLoE)])
		Return .f.
	EndIf

	If Not Empty(crSigLcLo.LocalA)
		Select crEstoqueA
		Scan
			If crEstoqueA.Marca
				Insert Into crSigLcLoE (Codigos, Locals, Emps, Grupos, Contas, cIdChaves) ;
								Values (crSigLcLo.Codigos, crSigLcLo.LocalA, crEstoqueA.Emps, ;
										crEstoqueA.Grupos, crEstoqueA.Contas, fUniqueIds())
			EndIf
		EndScan
	EndIf

	If Not Empty(crSigLcLo.LocalB)
		Select crEstoqueB
		Scan
			If crEstoqueB.Marca
				Insert Into crSigLcLoE (Codigos, Locals, Emps, Grupos, Contas, cIdChaves) ;
								Values (crSigLcLo.Codigos, crSigLcLo.LocalB, crEstoqueB.Emps, ;
										crEstoqueB.Grupos, crEstoqueB.Contas, fUniqueIds())
			EndIf
		EndScan
	EndIf
EndIf

=DoDefault()

Thisform.mAtivapagina1
* Fim *********************************************************************************
ENDPROC
