* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
LPARAMETERS lcacesso

With ThisForm
	.poDatamgr = CreateObject('fSqlConector')
	With .poDatamgr
		.AddCursor('SigCdCnt','codigos','crSigCdCnt','',ThisForm.pagina.lista.grade)
		.pfSqlTabela(1).pColuna('codigos','','','Código',60,.t.)
		.pfSqlTabela(1).pColuna('descricaos','','','Descrição',267,.t.)
		.pfSqlTabela(1).pColuna('consts','','','Valor',177,.t.)
		.pfSqlTabela(1).pColuna('moedas','','','Moeda',42,.t.)
		.AddCursor('SigCdMoe','cmoes','SigCdMoe','','')
	EndWith
	= DoDefault()
	.LockScreen = .f.
EndWith
ENDPROC

* PROCEDURE Pagina.Lista.Grupo_op.Click
------------------------------------------------------------
PROCEDURE Pagina.Lista.Grupo_op.Click
LPARAMETERS opcao

If !DoDefault( opcao )
	ThisForm.mAtivapagina1
	Return
Endif

Thisform.podatamgr.sqlexecute("Select a.dmoes from SigCdMoe a where a.cmoes = '"+CrSigCdCnt.moedas+"'","SigCdMoe") 
This.Parent.Parent.Dados.get_ds_moeda.Value = SigCdMoe.dmoes

If (ThisForm.pcEscolha = [EXCLUIR])
	lcCon = crSigCdCnt.Codigos

	lcQuery = [Select Codigos ] + ;
				[From SigOpTdt ] + ;
			   [Where (const1t1s = '] + lcCon + [' Or ] + ;
			   		  [const1t2s = '] + lcCon + [' Or ] + ;
			   		  [const2t1s = '] + lcCon + [' Or ] + ;
			   		  [const2t2s = '] + lcCon + [' Or ] + ;
			   		  [const3t1s = '] + lcCon + [' Or ] + ;
			   		  [const3t2s = '] + lcCon + [')]

	If (ThisForm.poDataMgr.SqlExecute(lcQuery, [LocalBus]) < 1)
		=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (LocalBus)])
		Return 0
	EndIf
	Go Top In LocalBus
	If Not Eof([LocalBus])
		=MessageBox([A Constante Não Pode Ser Excluída Pois Está] + Chr(13) + ;
					[Sendo Usada Na Tabela de Descontos "] + Alltrim(LocalBus.Codigos) + ["], 0+16, ;
					[Exclusão Não Permitida])
		ThisForm.mAtivapagina1
		Return .f.
	EndIf
EndIf

Do Case
	Case Inlist( ThisForm.pcEscolha, 'INSERIR', 'PROCURAR' )
		ThisForm.Pagina.Dados.Get_Código.SetFocus
	Case Inlist( ThisForm.pcEscolha, 'CONSULTAR', 'EXCLUIR' ) 
		ThisForm.Pagina.Dados.Grupo_salva.Salva.SetFocus
	Case ThisForm.pcEscolha = 'ALTERAR'
		ThisForm.Pagina.Dados.Get_Descrição.SetFocus
Endcase

ThisForm.Pagina.Dados.Refresh
ENDPROC

------------------------------------------------------------

PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
If ThisForm.pcEscolha = 'INSERIR'		
	If ThisForm.poDataMgr.ChkRegister('SigCdCnt','codigos',ThisForm.Pagina.Dados.Get_Código.Value)
		Messagebox( 'Código da Constante já cadastrado...', 48, '' )
		ThisForm.Pagina.Dados.Get_Código.SetFocus
		Return
	Endif	
Endif		
=DoDefault()		
ThisForm.mAtivapagina1		

ENDPROC
