PROCEDURE processamento
Private pDtI, pDtF, pImg

pDtI = fDtoSQL(ThisForm.getDtIni.Value)
pDtF = fDtoSQL(ThisForm.getDtFin.Value, '23:59:59')
pImg = ThisForm.optImagem.Value
pGru = ThisForm.getGrupo.Value
pPro = ThisForm.getCProsF.Value
pLin = ThisForm.getLin.Value

lcQuery = [Select DtFilms, DtIncs, Datas, CPros, DPros, IFors, Reffs, CGrus, Linhas, FigJpgs ] + ;
			[From SigCdPro ] + ;
		   [Where DtIncs BetWeen ?pDtI And ?pDtF ] + ;
		   		  Iif(pImg=1, [And FigJpgs Is Null ], ;
		   		  Iif(pImg=2, [And Not FigJpgs Is Null ], [])) + ;
		   		  Iif(Empty(pPro), [], [And CPros = ']  + pPro + [' ]) + ;
		   		  Iif(Empty(pGru), [], [And CGrus = ']  + pGru + [' ]) + ;
		   		  Iif(Empty(pLin), [], [And Linhas = '] + pLin + [' ]) + ;
		   [Order By CPros]

If (ThisForm.poDataMgr.SqlExecute(lcQuery, 'crProdutos') < 1)
	=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (Processamento - 1)')
	Return 0
EndIf

Zap In csProdutos

If (Reccount('crProdutos') < 1)
	=MessageBox('Nenhum Produto Foi Selecionado!!!', 16, '')
Else
	Select crProdutos
	Scan
		Insert Into csProdutos (DtFilms, DtIncs, Datas, CPros, DPros, IFors, Reffs, CGrus, Linhas, FigJpgs) ;
					    Values (crProdutos.DtFilms, crProdutos.DtIncs, crProdutos.Datas, ;
					    	    crProdutos.CPros, crProdutos.DPros, crProdutos.IFors, ;
					    	    crProdutos.Reffs, crProdutos.CGrus, crProdutos.Linhas, ;
					    	    crProdutos.FigJpgs)
	EndScan
EndIf

Select crProdutos
Go Top

ThisForm.CarregaImagem(csProdutos.CPros)
ThisForm.AcertaForm(.f.)
ThisForm.grdProdutos.Refresh

If (Reccount('crProdutos') > 0)
	ThisForm.grdProdutos.Column1.SetFocus
Else
	ThisForm.cmdBtnForm.btnLimpar.SetFocus
EndIf
* Fim *********************************************************************************

ENDPROC

Set Default To &lcAntPath.

If Not Empty(lcFiguras) And File(lcFiguras)
	If Not fTamanhoImagem(lcFiguras, ThisForm.poDataMgr, .t.)
		Return 0
	EndIf
	
	If (MessageBox('Confirma a Transferência da Imagem ?', 4+32+256, '') <> 6)
		ThisForm.grdProdutos.Column1.SetFocus
		Return 0
	EndIf

	Wait Window 'Aguarde!!! Transferindo a Imagem...' NoWait

	pPro = lcProduto
	ThisForm.poDataMgr.Requery('crSigCdPro')
	Go Top In crSigCdPro

	Replace FigJpgs With FileToStr(lcFiguras), ;
			DtFilms With Datetime(), ;
			Datas   With Datetime() In crSigCdPro

	lcQuery = [Delete From SigPrPrt Where Cpros = '] + pPro + [']
	If (ThisForm.poDataMgr.SqlExecute(lcQuery, '') < 1)
		=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (Processamento - Delete)')
		Return 0
	EndIf
	
*!*		lcQuery = [Delete From SigPrWeb Where Cpros = '] + pPro + [']
*!*		If (ThisForm.poDataMgr.SqlExecute(lcQuery, '') < 1)
*!*			=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (Processamento - Delete 2)')
*!*			Return 0
*!*		EndIf
	
	If Not ThisForm.poDataMgr.Update('crSigCdPro')
		=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (Figura - 1)')
		Return 0
	EndIf

	If ThisForm.poDataMgr.Commit() < 1
		=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (Figura - 2)')
		Return 0
	EndIf

	Replace FigJpgs With FileToStr(lcFiguras), ;
			DtFilms With Datetime(), ;

	Return 0

		DtFilms With Datetime(), ;
		Datas   With Datetime() In csProdutos

Wait Clear

ThisForm.CarregaImagem(csProdutos.CPros)
