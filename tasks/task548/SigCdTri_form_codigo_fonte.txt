* PROCEDURE msv_procurar
------------------------------------------------------------
PROCEDURE msv_procurar
Local lcBusca

If .Not.IsEmpty(ThisForm.Pagina.Dados.Getcodigo.Value)
	ThisForm.poDataMgr.LocateCursor('crSigCdIcm','codigos',ThisForm.Pagina.Dados.Getcodigo.Value)
Else
	If .Not.IsEmpty(ThisForm.Pagina.Dados.Getdescricao.Value)
		ThisForm.poDataMgr.LocateCursor('crSigCdIcm','descricaos',ThisForm.Pagina.Dados.Getdescricao.Value)
	EndIf
EndIf

ENDPROC

------------------------------------------------------------
* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
With ThisForm
	.poDataMgr = CreateObject('fSqlConector', ThisForm.Name)

	If (.poDataMgr.pnIdconn > 0)
		With .poDataMgr
			.AddCursor('SigCdIcm','codigos','crSigCdIcm','',ThisForm.pagina.lista.grade)
			.pfSqlTabela(1).pColuna('codigos','','','Codigo',80,.t.)
			.pfSqlTabela(1).pColuna('descricaos','','','Descrição',337,.t.)
			.pfSqlTabela(1).pColuna('colunas','','','Coluna',60,.t.)
			.pfSqlTabela(1).pColuna('Tipos','','','Tipo',30,.t.)
		EndWith

		ThisForm.Pagina.Dados.spnColuna.Value = Val(CrSigCdIcm.Colunas)
		
		=DoDefault()

		ThisForm.Pagina.Lista.Grade.Column1.SetFocus
	Else
		=Messagebox('Impossivel Efetuar Conexão com Servidor de Banco de Dados...', 0+48, 'CreateObject')
		Return .f.
	EndIf
EndWith
* Fim **************************************************************************************
ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Lista.Grupo_op.Click
------------------------------------------------------------
PROCEDURE Pagina.Lista.Grupo_op.Click
parameters lnOpcao

If !dodefault( lnOpcao )
	ThisForm.mAtivapagina1
	Return
Endif

thisform.Pagina.Dados.spnColuna.value = Val(CrSigCdIcm.colunas)

do case
	case thisform.pcEscolha$'INSERIR PROCURAR'
		replace colunas with STR(1)
		thisform.Pagina.Dados.getcodigo.setfocus
	case thisform.pcEscolha='ALTERAR'
		thisform.pagina.dados.getdescricao.setfocus
	case thisform.pcEscolha$'CONSULTAR EXCLUIR'
		thisform.pagina.dados.grupo_salva.salva.setfocus
endcase

ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
local lnRecno,lcRetorno,Colunas

m.lnRecno = recno('CrSigCdIcm')

if inlist(thisform.pcEscolha,'INSERIR','ALTERAR')

   with thisform.Pagina.Dados
      if IsEmpty(.getCodigo.value)
         = messagebox('Código em branco !!!',0+64,goSistema.NomeSistema)
         .getCodigo.setfocus
		 return
      else

		If ThisForm.poDataMgr.ChkRegister('SigCdIcm','codigos',ThisForm.Pagina.Dados.GetCodigo.Value) and inlist(thisform.pcEscolha,'INSERIR')
	        messagebox('Código já cadastrado !!!',0+64,goSistema.NomeSistema)
			ThisForm.Pagina.Dados.GetCodigo.SetFocus
			Return
			
		Endif	

        if IsEmpty(.getDescricao.value)

	         = messagebox('Descrição em branco !!!',0+64,goSistema.NomeSistema)
	         .getDescricao.setfocus
			 return
		
		 else

		      if !between(.spnColuna.value,1,3)
		         = messagebox('Coluna Inválida, Permitido Apenas Valores 1,2 ou 3.',0+64,goSistema.NomeSistema)
		         .spnColuna.setfocus
				 return
		      endif
			
		 endif
						
		endif
		
	endwith

endif

if thisform.pcEscolha $ 'INSERIR ALTERAR'

   Select CrSigCdIcm
   Replace Colunas With Str(thisform.Pagina.Dados.spnColuna.value,1)
   
endif

=DoDefault()

select CrSigCdIcm

ThisForm.mAtivapagina1

ENDPROC
