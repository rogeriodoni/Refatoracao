PROCEDURE atustatus
lnStatus = thisform.pagina.dados.container2.opt_status.Value

_MaqTubo = ThisForm.MaqTubos

For x=1 to _MaqTubo
	With ThisForm.Pagina.Dados.Container2
		lcVar = '.Check' + Alltrim(Str(x))
		
		If &lcVar..Value = 1
			Select TmpCera
			Set Order to
			Set Order to Tubos
			If Seek(x)
				Do case
					Case lnStatus = 1 && Nenhum
						If (TmpCera.nStatus = 2 And !thisform.pagina.dados.container2.opt_status.option2.Visible) or (TmpCera.nStatus = 3 And !thisform.pagina.dados.container2.opt_status.option3.Visible) or (TmpCera.nStatus = 4 And !thisform.pagina.dados.container2.opt_status.option4.Visible)	
							thisform.pagina.dados.container2.opt_status.Value = TmpCera.nStatus
							Loop
						Endif
						&lcVar..Picture = ''	
					Case lnStatus = 2 && Cera
						If (TmpCera.nStatus = 3 And !thisform.pagina.dados.container2.opt_status.option3.Visible) or (TmpCera.nStatus = 4 And !thisform.pagina.dados.container2.opt_status.option4.Visible)	
							thisform.pagina.dados.container2.opt_status.Value = TmpCera.nStatus
							Loop
						Endif
						&lcVar..Picture = '\imagens\geral_arvore_cera_60.jpg' &&'\Vbmp\Green.bmp'
					Case lnStatus = 3 && Fundido
						If (TmpCera.nStatus = 4 And !thisform.pagina.dados.container2.opt_status.option4.Visible)
							thisform.pagina.dados.container2.opt_status.Value = TmpCera.nStatus						
							Loop
						Endif					
						&lcVar..Picture = '\imagens\geral_arvore_ouro_60.jpg' &&'\Vbmp\Yello.bmp'
					Case lnStatus = 4 && Peça
						&lcVar..Picture = '\imagens\geral_pecapronta_60.jpg' && '\Vbmp\Red.bmp'						
				EndCase
				replace nStatus with lnStatus in TmpCera
			EndIf
		EndIf
		&lcVar..Refresh()
	EndWith
Next



ENDPROC

* PROCEDURE calcmetal

PROCEDURE calcmetal
If !Empty(TmpCera.BCeras) and !Empty(TmpCera.Bases) and !Empty(CrSigCdArv.formulas)
	If TmpCera.BCeras < TmpCera.Bases
		MessageBox([Base com Cera não pode ser menor que a Base],48,[Aviso])
		replace Bases with 0 in TmpCera
		Return .f.
	EndIf

	lcFormula = CrSigCdArv.formulas
	Select TmpCera
	lnPesoMetal = &lcFormula.
	lnMetalNovo = (lnPesoMetal * CrSigCdArv.Percs) / 100
	thisform.pagina.dados.container5.get_Metal.Value = lnPesoMetal
	thisform.pagina.dados.container5.get_Mreal.Value = lnPesoMetal
	thisform.pagina.dados.container5.get_metnovo.Value = lnMetalNovo
	thisform.pagina.dados.container5.get_Mreal.Refresh()
	thisform.pagina.dados.container5.get_metnovo.Refresh()

	** Cálculo de cera
	lnBc = thisform.pagina.dados.container5.Get_Basec.Value
	lnB  = thisform.pagina.dados.container5.Get_Base.Value
	lnC  = lnBc - lnB
	thisform.pagina.dados.container5.Get_Cera.Value  = lnC
EndIf
ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
Private pDtq
Local llRetorno,lcChave,lnRecno,Codigos

m.llRetorno = .t.
m.Codigos = ThisForm.Pagina.Dados.Container1.getCodigos.Value

If Inlist(ThisForm.pcEscolha, 'INSERIR', 'ALTERAR')
	If Empty(ThisForm.Pagina.Dados.Container1.getDatas.Value)
		=MessageBox('Data Invalida!!!', 48, '')
		ThisForm.Pagina.Dados.Container1.getDatas.SetFocus
		Return 0
	EndIf
EndIf

If InList(ThisForm.pcEscolha, 'EXCLUIR', 'ALTERAR')
	If (Thisform.pcEscolha = 'EXCLUIR')
		Select crSigCdFun
		Delete

		Select TmpNop
		Set Order To
		Scan
			Nop   = TmpNop.Nops
			lnQtd = TmpNop.Qtdos

			ThisForm.poDataMgr.CursorQuery('SigOpPic', 'crSigOpPic', 'Nops', Nop, 'cIdChaves, QtdTubos')
			Select crSigOpPic
			Scan While lnQtd > 0
				If (QtdTubos <= lnQtd)
					lnQtd = lnQtd - QtdTubos

					lcQuery = [Update SigOpPic Set QtdTubos = 0 Where cIdChaves = '] + crSigOpPic.cIdChaves + [']
					If (ThisForm.poDataMgr.SqlExecute(lcQuery, '') < 1)
						=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (Update - SigOpPic - 1)')
						Return 0
					EndIf
				Else
					pDtq = Qtdtubos - lnQtd
					lcQuery = [Update SigOpPic Set QtdTubos = ?pDtq Where cIdChaves = '] + crSigOpPic.cIdChaves + [']
					If (ThisForm.poDataMgr.SqlExecute(lcQuery, '') < 1)
						=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (Update - SigOpPic - 2)')
						Return 0
					EndIf
					lnQtd = 0
				EndIf
			EndScan
			
			ThisForm.poDataMgr.CursorQuery('SigOpPic', 'crSigOpPic', 'Nops', Nop, 'cIdChaves, QtdTubos')
			Select crSigOpPic
			Sum QtdTubos To QtdTub
			If (QtdTub = 0)
				lcQuery = [Update SigOpPic Set nFunds = 0, DtFunds = Null Where Nops = ] + Str(Nop, 10)
				If (ThisForm.poDataMgr.SqlExecute(lcQuery, '') < 1)
					=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (Update - SigOpPic - 3)')
					Return 0
				EndIf
			EndIf
		EndScan
	EndIf

	lcQuery = [Delete From SigCdFud Where Codigos = ] + Str(m.Codigos, 6)
	If (ThisForm.poDataMgr.SqlExecute(lcQuery, '') < 1)
		=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (Delete - SigCdFud)')
		Return 0
	EndIf
EndIf

If InList(Thisform.pcEscolha, 'INSERIR', 'ALTERAR')
	Select TmpTrf
	Set Order To
	Scan
		If Empty(Dopes) Or (Numes = 0)
			Loop
		EndIf
		Scatter Memvar
		m.Nops  	= 0
		m.Qtds  	= 0
		m.Qtdos 	= 0
		m.cIdChaves = fUniqueIds()

		Select crSigCdFud
		Append Blank
		Gather Memvar
	EndScan

	Scatter Memvar Blank
	Select TmpNop
	Set Order to
	Scan
		If (Nops = 0) Or (Qtdos = 0)
			Loop
		EndIf
		Scatter Memvar
		m.Nnumes = 0
		m.cIdChaves = fUniqueIds()

		Select crSigCdFud
		Append Blank
		Gather Memvar
	EndScan
	Scatter Memvar Blank

	Select TmpCera
	Set Order to
	Scan
		If (bCeras = 0) Or (Bases = 0)
			Loop
		EndIf
		Scatter Memvar Memo
		m.cIdChaves = fUniqueIds()
		m.Metals    = m.pMetalRs

		Select crSigCdFud
		Append Blank
		Gather Memvar Memo
	EndScan

	*** Tiago - 10/10/2011 - Vianna - Se tiver algum tubo que não foi finalizado, salva o campo nTubos = 1, para ficar em vermelho na grade indicando que tem tubo pendente
	Select Count(1) as TotFund from TmpCera where InList(nStatus,2,3) group by CodArvs into cursor TmpTot
	replace nTubos with Iif(TmpTot.TotFund = 0,0,1) In crSigCdFun	
EndIf

If (Thisform.pcEscolha = 'INSERIR')
	Replace cIdChaves With fUniqueIds() In crSigCdFun
EndIf

=DoDefault()

Select crSigCdFun

ThisForm.mAtivaPagina1
* Fim **************************************************************************************

ENDPROC

LPARAMETERS nColIndex

_EPN = TmpNop.EmpDopNums

*** Tiago - 18/08 - Se tiver subnivel, monta o empdopnums para mostrar a operacao inicial de onde veio o pedido
lcQuery = [Select EmpSubns, Codigos from SigMvPec where EmpDopNums = '] + TmpNop.EmpDopNums + [']
If Thisform.PoDataMgr.SqlExecute(lcQuery,'CrSigMvPec') < 1
	MessageBox('Favor reinicializar o processo.',16,'Falha na Conexão (CrSigMvPec)')
	Return .F.
EndIf

Select CrSigMvPec
If !Eof()
	_Emps = crSigMvPec.EmpSubns
	_nDopes = Int( CrSigMvPec.Codigos / 1000000 )	
	_Numes = Str(Int(CrSigMvPec.Codigos - ( _nDopes  * 1000000 )),6)

	lcSql = [Select Dopes From SigCdOpe Where nDopes = ?_nDopes]
	If Thisform.PoDataMgr.SqlExecute(lcSql,'CrxOpe') < 1
		MessageBox('Favor reinicializar o processo.',16,'Falha na Conexão (CrXOpe)')
		RETURN .F.
	EndIf

	_EPN = _Emps + CrxOpe.Dopes + _Numes 
EndIf

thisform.pagina.dados.container3.get_operacao.Value = _EPN
thisform.pagina.dados.container3.get_operacao.Refresh()
ENDPROC
