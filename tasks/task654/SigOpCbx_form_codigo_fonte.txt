PROCEDURE Valid
LOCAL xBrox

If !IsEmpty( This.Value )
	xBrox = CreateObject('fwBuscaExt',Thisform.ParentForm.poDatamgr.pnIdconn,'SigCdEmp','csSigCdEmp','cEmps',This.Value,'Empresas',.t.,'cEmps')

	If  Not xBrox.plAchouRegistro
		xBrox.mAddColuna('cEmps', '', 'Cod', .t.)
		xBrox.mAddColuna('RazSocs', '', 'Razão Social', .t.)
		xBrox.Show()
		This.Value = Iif( LastKey() # 27, Evaluate('csSigCdEmp.cEmps'), '' )
	EndIf 
Endif
ENDPROC

PROCEDURE Command1.Click
If Not Empty(ThisForm.Container1.Text3.Value)
	llNum = .t.
	For I = 1 to Len(Alltrim(ThisForm.Container1.Text3.Value))
		If Not IsDigit(Substr(Alltrim(ThisForm.Container1.Text3.Value), I, 1))
			llNum = .f.
		EndIf
	Next
	If Not llNum 
		=MessageBox('O Número da Operação Digitado Não é Válido!!!', 48, '')
		ThisForm.Container1.Text3.SetFocus 
		Return 0
	EndIf
EndIf
	
If Empty(ThisForm.Container1.Text3.Value)
	=MessageBox('O Número da Operação Digitado Não é Válido!!!', 48, '')
	ThisForm.Container1.Text3.SetFocus 
	Return 0
EndIf

If Empty(ThisForm.Container1.Text1.Value)
	=MessageBox('É Necessário Preencher a Empresa !!!', 48, '')
	ThisForm.Container1.Text1.SetFocus 
	Return 0
EndIf

If Empty(ThisForm.Container1.Text2.Value)
	=MessageBox('É Necessário Preencher a Operação !!!', 48, '')
	ThisForm.Container1.Text2.SetFocus 
	Return 0
EndIf

lcEdn = Padr(ThisForm.Container1.Text1.Value,  03) + ;
		Padr(ThisForm.Container1.Text2.Value, 20) + ;
		Padl(Alltrim(ThisForm.Container1.Text3.Value), 6)
		ThisForm.ParentForm.poDataMgr.CursorQuery('SigMvCab', 'LocalEest', 'EmpDopNums', lcEdn, 'EmpDopNums')
If Eof('LocalEest')
	=MessageBox('A Operação Com o Número Digitado Não Foi Encontrada na Empresa!!!', 48, '')
	ThisForm.Container1.Text1.SetFocus 
	Return 0
EndIf

* Rotina Para Gravação dos Dados da Ficha Técnica no Cursor Para Atualizar a Base
If Used('crGrvCadRs')

	Zap In crGrvCadRs

	llOk  = .T.
		
	lnCod = ThisForm.pcCodigo

	lcVal = ThisForm.Container1.Text1.Value
	If Not Empty(lcVal)
		Insert Into crGrvCadRs (Tits, Campos, Cods, cIdChaves, Resps, Pergs, ChkApro) ;
						Values (ThisForm.ParentForm.Tits, [Emps], lnCod, fUniqueIds(), lcVal, [Empresa], llOK)
	EndIf

	lcVal = ThisForm.Container1.Text2.Value
	If Not Empty(lcVal)
		Insert Into crGrvCadRs (Tits, Campos, Cods, cIdChaves, Resps, Pergs, ChkApro) ;
						Values (ThisForm.ParentForm.Tits, [Dopes], lnCod, fUniqueIds(), lcVal, [Operação], llOK)
	EndIf

	lcVal = ThisForm.Container1.Text3.Value
	If Not Empty(lcVal)
		Insert Into crGrvCadRs (Tits, Campos, Cods, cIdChaves, Resps, Pergs, ChkApro) ;
						Values (ThisForm.ParentForm.Tits, [Numes], lnCod, fUniqueIds(), lcVal, [Código], llOK)
	EndIf

	lcVal = ThisForm.Container1.Text3.Value
	If Not Empty(lcVal)
		Insert Into crGrvCadRs (Tits, Campos, Cods, cIdChaves, Resps, Pergs, ChkApro) ;
						Values (ThisForm.ParentForm.Tits, [Baixa], lnCod, fUniqueIds(), Usuar, [Baixa], llOK)
	EndIf

	ThisForm.ParentForm.poDatamgr.UpDate('crGrvCadRs')

	ThisForm.ParentForm.poDatamgr.SqlExecute([UpDate SigSyCrs Set ChkSubn = ?llOk Where Cods = ?lnCod])
			
	ThisForm.ParentForm.poDataMgr.Commit()
	
	Bxado = 'B'

EndIf

* Fim *********************************************************************************

Release Windows

ENDPROC
