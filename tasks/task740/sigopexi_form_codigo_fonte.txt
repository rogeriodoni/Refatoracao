PROCEDURE Init
LPARAMETERS poform, pnDatSes

Set DataSession to pnDatSes

ThisForm.cntSombra.lblSombra.Caption = ThisForm.Caption
ThisForm.cntSombra.lblTitulo.Caption = ThisForm.Caption

thisform.parentform = poform

If Used([crSigOpExI])
	IF TYPE('crSigOpExI.CItens') = 'U'
		SELECT * , 0000000000 as citens, SPACE(50) as dpros from crSigOpeXi into cursor crSigOpeXi readwrite
	ENDIF
	ThisForm.grdDevols.RecordSource = [crSigOpExI]
	ThisForm.grdDevols.Column1.ControlSource = [crSigOpExI.CItens]
	ThisForm.grdDevols.Column2.ControlSource = [crSigOpExI.CBars]
	ThisForm.grdDevols.Column3.ControlSource = [crSigOpExI.CPros]
	ThisForm.grdDevols.Column4.ControlSource = [crSigOpExI.DPros]
	SELECT crSigOpExI
	SUM valors to lntotven
	COUNT to lnitn
	thisform.gettotven.value = TRANSFORM(lntotven, '999,999.99')
	thisform.gettotitens.Value = lnItn

EndIf
Set Confirm on

thisform.op_escolha = poform.op_escolha

thisform.Refresh()
ENDPROC

PROCEDURE Valid
If Not Empty(This.Value)
	lnBar = This.Value
	Select Recno() as NRec, CPros, BlqAlts, qtds, units From crTpmMvItn Where CodBarras = lnBar Into Cursor csProcura1
	Go Top In csProcura1
	If Not Eof([csProcura1])
		If csProcura1.BlqAlts
			MessageBox('Código do Produto não pode ser Alterado',0+48,'Operação com Baixa Parcial')
			RETURN .f.
		ENDIF
	
		Select Recno() as NRec From crSigOpExI Where CBars = lnBar Into Cursor csProcura
		Go Top In csProcura
		If Eof([csProcura])
			lnItn = 0
			Select crSigOpExI
			Scan
				lnItn = lnItn + 1
				Replace CItens With lnItn In crSigOpExI
			EndScan
			lnItn = lnItn + 1
			
			thisform.parentform.parentform.podatamgr.sqlexecute('select dpros from SigCdPro where cpros = ?csProcura1.CPros','LocalPro')
			
			Insert Into crSigOpExI (EmpDopNums, Datas, pkChaves, CBars, CPros, dpros, valors, CItens) ;
						    Values (TprMvCab.Emps + TprMvCab.Dopes + Str(TprMvCab.Numes,6), ;
						    		Datetime(), fUniqueIds(), lnBar, csProcura1.CPros, LocalPro.dpros, csProcura1.qtds * csProcura1.units, lnItn)
			This.Value = 0
			Use In csProcura1
			Use In csProcura
			
			thisform.gettotitens.Value = lnItn
			SELECT crSigOpExI
			SUM valors to lntotven
			thisform.gettotven.value = TRANSFORM(lntotven, '999,999.99')
			
			thisform.grdDevols.Refresh()
			Return .f.
		ENDIF
	ELSE
		MESSAGEBOX('Item inexistente nessa operação',64,'Aviso')
		this.Value = 0
		RETURN .f.
	EndIf
EndIf
* Fim **************************************************************************************
ENDPROC

ENDPROC

* PROCEDURE Click
------------------------------------------------------------
PROCEDURE Click
SELECT crSigOpeXi
SCAN
	Select crTpmMvItn 
	LOCATE for CodBarras = crSigOpeXi.cbars
	IF !EOF()
		If crTpmMvItn.BlqAlts
			MessageBox('Código do Produto não pode ser Alterado',0+48,'Operação com Baixa Parcial')
		Else
			If !IsEmpty(crTpmMvItn.cpros)
				_CItem = crTpmMvItn.CItens
				_Cpro  = crTpmMvItn.Cpros
				_Qtde  = crTpmMvItn.Qtds
				_CarregaItens = InList( CrSigCdOpe.ItAutos,1,3,4,5 ) And ThisForm.op_escolha = 'INSERIR'
				_ChecaQtde    = InList( CrSigCdOpe.ChkQtdss,1,3,4 )
				_ChecaOper    = ( Empty(CrSigOpCdc.ChkOpers) Or crTpmMvItn.Opers = CrSigOpCdc.ChkOpers )

				lnReg = Recno()
				llDel = .f.
				
				Locate For .f.
				If Between(lnReg,1,Reccount('crTpmMvItn'))
					Go lnReg
					Delete
					Delete For cItem2 = _cItem
					llDel = .t.
				EndIf

				If llDel
					Select TmpOperacao
					_SubNiv = .f.
					Scan for !IsEmpty( Codigos )
						_SubNiv = .t.
					Endscan

					If _SubNiv And _ChecaQtde And _ChecaOper And Not _CarregaItens 
						Select xBaixa
						If Seek( _Cpro )
							Replace xbaixa.qtbaixas With xbaixa.qtbaixas - _Qtde in xBaixa
						Endif
					Endif

					Select crTpmMvIts
					Set Order to
					Set Order to CItens
					Seek ( _CItem )
					Do While Not Eof('crTpmMvIts') And crTpmMvIts.CItens = _CItem
						Delete 
						Seek ( _CItem )
					EndDo

					If CrSigCdOpe.CadProds = 3
						lcSql = [Select Cpros From SigMvItn Where Cpros = ']+_cPRO+[' And EmpDopNums <> ']+TprMvCab.EmpdopNums+[']
						ThisForm.parentform.parentForm.PodataMgr.SqlExecute(lcSql,'LocEesti')

						lcSql = [Select Cpros From SigMvHst  Where Cpros = ']+_cPRO+[' And EmpDopNums <> ']+TprMvCab.EmpdopNums+[']
						ThisForm.parentForm.parentform.PodataMgr.SqlExecute(lcSql,'LocHis')
						
						lcSql = [Select Mats From SigPrCpo Where Mats = ']+_cPro+[']
						ThisForm.parentForm.parentform.PodataMgr.SqlExecute(lcSql,'LocCompo')

						If Reccount('LocEesti') = 0	And Reccount('LocHis') = 0 And Reccount('LocCompo') = 0
							If MessageBox('Deseja Excluir O Cadastro de Produto ? ',4+16+256,'') = 6
								=ThisForm.parentform.ExcluirPro(_cPro)
							Endif
						Endif
					Endif

				Endif
			Else
				Select [ ] as Agrupar,Count(1) As Total From crTpmMvItn Where IsEmpty(CPros) Into Cursor xApaga Group by 1
				If xApaga.Total > 1
					Select crTpmMvItn
					Delete
				Else
					Thisform.parentform.ApaIte()
				EndIf
				If Used('xApaga')
					Use In xApaga
				Endif
			Endif

			Select crTpmMvItn
			Set Order To Citens
			Go top
			If Eof()
				Insert Into crTpmMvItn( Emps, Dopes, Numes, CItens) ;
					Values ( TprMvCab.Emps, TprMvCab.Dopes, TprMvCab.Numes, 1 )
			Endif
			Cont = 0
			Scan
				Cont = Cont + 1
				Select crTpmMvIts
				Set Order To
				Set Order to CItens
				Seek crTpmMvItn.CItens
				Scan While CItens = crTpmMvItn.CItens
					Replace CItens With Cont
				Endscan
				Select crTpmMvItn
				_cIt = CItens
				_RecNo2 = Recno()
				Replace cItem2 With Cont For cItem2 = _cit
				Go _RecNo2
				Replace CItens With Cont
			Endscan

			Thisform.parentform.CItem = Cont

			Select crTpmMvItn
			Set Order To 
			Goto Bottom
			If !Eof()
				Select crTpmMvIts
				Set Order to CItens
				Set Key to crTpmMvItn.CItens
			EndIf
		Endif

		Select crTpmMvItn
		With Thisform.parentform.fwgrade
			If .Visible
				.Refresh
				.Produto.SetFocus
			EndIf
		EndWith

		With Thisform.parentform.fwgradeR
			If .Visible
				.Refresh
				.Produto.SetFocus
			EndIf
		EndWith	
	ENDIF
ENDSCAN

Set Confirm off
thisform.Release()
ENDPROC
