PROCEDURE montagrade
Local lnOldSel

Select crItensLoc

Go top

With ThisForm.grdItens
	.RecordSource = [crItensLoc]
	.column1.ControlSource = [crItensLoc.Cpros]
	.column2.ControlSource = [crItensLoc.Dpros]
	.column3.ControlSource = [crItensLoc.Marcas]
	.column4.ControlSource = [crItensLoc.cItens]
	.column5.ControlSource = [crItensLoc.Locals]
	.Refresh()
EndWith 

ThisForm.grdItens.Column1.SetFocus
ThisForm.grdItens.Refresh
ENDPROC

------------------------------------------------------------
* PROCEDURE reverterxml
------------------------------------------------------------
PROCEDURE reverterxml
Parameters pcCursor, pcNivAnt, pcCurDest

Local lnOldSel, lcXML

lnOldSel = Select()
lcXML = []

If Used([CrAux])
	Use in CrAux
EndIf 

Select &pcCursor.
Select cidchaves, Cods, Descs, 0 as Marcas, Space(4) as Codigo From &pcCursor. Where 1=0 Into Cursor CrAux ReadWrite

Select &pcCursor.	
Go top

Scan 			
	lcXML = Nvl(&pcCursor..Nivel,[])
	Zap in CrAux
	If Not Empty(lcXML)
		= fXmlToCursor(lcXML, [CrAux], .f.)
		Select CrAux
		Scan For CrAux.Marcas <> 0
			Select &pcCurDest.
			Append Blank 
			Replace Cods With CrAux.Cods ,;
					Descs With &pcCursor..Descs ,;
					CidChaves With CrAux.CidChaves ,;
					Codigo With &pcCursor..Cods in &pcCurDest.
			Select CrAux	
		EndScan 
	EndIf 
EndScan 

Select &pcNivAnt.
Go Top 

If Used([CrAux])
	Use in CrAux
EndIf 

Select(lnOldSel)
ENDPROC

* PROCEDURE validacao

PROCEDURE validacao
Local lnOldSel
lnOldSel = Select()

If Empty(Thisform.getBase.Value)
	Messagebox([Informar a Base da Localização.],0+48,[Atenção])
	Return .F.
Endif

Select crSigLcNv2
Go Top
If Reccount([crSigLcNv2])>0 And Empty(Thisform.getNvl2.Value)
