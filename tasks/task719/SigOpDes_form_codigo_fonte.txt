PROCEDURE calctots
LOCAL lnReg, lnSaldo, lnTDesc

SELECT localdes

*!*	IF INLIST(localDes.tpdesc,7,10)	&& updimond ou up alianca
*!*		replace valdes with IIF(localdes.selecao = 0, 000000000.00,ROUND(crProdS.pvens,2))
*!*	else
*!*	ENDIF

replace descos with IIF(localdes.selecao = 0, 00.0000, localdes.descos)
replace valdes with IIF(localdes.selecao = 0, 000000000.00, ROUND(thisform.lnvalor.value*localdes.descos/100,2))
lnReg=IIF(EOF('localdes'),0,RECNO())

GO top
SUM localdes.valdes to lnTDesc

*** soma descontos do item ***
GO top
SUM localdes.valdes to lnTDescI for !INLIST(localdes.tpdesc,1,2,6,7,10,11,12,13)	&& nao entra na cota A,V,TZ,UP,UpAlianca,extra/loja,brinde,troca preco

*** soma descontos dos outros itens ***
SELECT crSigmvitd
GO top
SUM vdes to lnTDescM for !INLIST(tpdesc,1,2,6,7,10,11,12,13) AND citens <> pcItem 	&& nao entra na cota A,V,TZ,UP

replace valdes with crValdes.vdes+lnTDescI+lnTDescM in crSigCdDes
 
SELECT localdes
GO top
lnSaldo = this.lnvalor.value-lnTDesc

thisform.lnSaldo.Value =  lnSaldo
thisform.lnTdesc.Value = lnTDesc
SELECT localdes
IF lnReg <> 0
	GO lnReg
endif
thisform.Refresh 
RETURN

ENDPROC

	SELECT top 1 * from crTpmMvItn a where a.cpros in (select b.cpros from crProdL b where b.cgrus = crGruJ.grupoJ) ;

	(select c.cpros+STR(c.citens) from crSigMvItd c where c.tpdesc = 2 and c.cdesc=crGruJ.codigos ) ;

	SELECT * from crTpmMvItn where 1=0 into cursor crItnJD
