PROCEDURE Valid

Private lcSpace

If !Empty(This.Value)
	m.loLista = CreateObject('fwBuscaExt',ThisForm.poDataMgr.pnIdConn,'SigCdUsu','CrUsuarios','NComps',This.Value,'Usuarios',.T.,.f.,[Not cAtivos='N'])
	If !(m.loLista.plAchouRegistro)
		m.loLista.mAddColuna('NComps','','Nome')
		m.loLista.mAddColuna('Usuarios','XXXXXXXXXX','Código')
		m.loLista.Show()
	EndIf
	If LastKey() <> 27 And Used('CrUsuarios')
		This.Value                  = Evaluate('CrUsuarios.NComps')
		This.Parent.GetCodigo.Value = Evaluate('CrUsuarios.Usuarios')
	Else
		This.Value                  = ''
		This.Parent.GetCodigo.Value = ''
	EndIf
EndIf
This.Refresh
This.Parent.GetCodigo.Refresh
lcSpace = Space(100)

Thisform.Grid1.RecordSource = ''

If ! Empty(This.Value)

	lcQuery = [Select Distinct a.Programas, a.Parametros, ?lcSpace As Descricaos ] + ;
			  [From SigCdAcU a ] + ;
			  [Left Outer Join SigCdAcG b On a.Grupos = b.Grupos ] + ;
			  [Where Not a.Programas = Space(10) And ] + ;
			  [(a.Usuarios = '] + This.Parent.GetCodigo.Value + [' Or b.Usuarios = '] + This.Parent.GetCodigo.Value + [') ] + ;
			  [Order By a.Programas, a.Parametros]

	ThisForm.poDatamgr.SqlExecute(lcQuery,[crSigCdAcUs])
	
	Select crSigCdAcUs

	Replace all Descricaos With Alltrim(crSigCdAcUs.Programas) + ' / ' + crSigCdAcUs.Parametros

	Scan
		lcQuery = [Select descricaos From SigCdPrg Where programas = '] + crSigCdAcUs.Programas + [' And ]+;
		     	     								    [Parametros = '] + crSigCdAcUs.Parametros + [' ]
		ThisForm.poDatamgr.SqlExecute(lcQuery,[crSigCdPrg])
		If ! Eof()
			Select crSigCdAcUs 
			Replace descricaos With crSigCdPrg.Descricaos
		EndIf
	EndScan

	lcQuery = [select a.grupos,a.usuarios,a.descricaos,a.barraforms,a.barraordem,a.programas,a.parametros,a.selbarras from SigCdAcB a]+;
               [ left join SigCdPrg b on b.programas+b.parametros=a.programas+a.parametros]+;
               [ where a.usuarios='] + This.Value + [']+;
               [ union all select a.grupos,a.usuarios,a.descricaos,a.barraforms,a.barraordem,a.programas,a.parametros,a.selbarras from SigCdAcB a]+;
               [ left join SigCdPrg b on b.programas+b.parametros=a.programas+a.parametros]+;
               [ where a.grupos in (select c.grupos from SigCdAcG c where c.usuarios='] + This.Value + [') order by 1,2]

	ThisForm.poDatamgr.SqlExecute(lcQuery,[crTmpBarra])
	
	Select crSigCdAcUs
	Index On descricaos Tag Descrs
	Set Order To Descrs
	Go Top
	
	Thisform.Grid1.RecordSource = 'crSigCdAcUs'
	Thisform.Grid1.Column1.ControlSource = 'crSigCdAcUs.Descricaos'

EndIf		     	     								    

Return(.T.)

ENDPROC

PROCEDURE Command1.Click

Local grupos, usuarios, descricaos, barraforms, barraordem, programas, parametros, selbarras, cidchaves, lnErro1

m.lnErro1 = ThisForm.poDataMgr.pnIdConn

If Empty(ThisForm.GetCodigo.Value)
	MessageBox('Usuario Invalido !!!',16, '')
	Return
Endif
If Empty(ThisForm.GetGrupo.Value)
	MessageBox('Grupo Invalido !!!',16, '')
	Return
Endif

Zap In crSigCdGrA
Append Blank In crSigCdGrA
Replace Grupos With ThisForm.GetGrupo.Value In crSigCdGrA
Replace nComps With ThisForm.GetGrupo.Value In crSigCdGrA
=fGravarLog('I', 'SIGCDGR1', '', 'Grupo : ' + ThisForm.GetGrupo.Value, Usuar)

Zap In crSigCdAcU

Select crSigCdAcUs
Scan
	Select crSigCdAcU
	Append Blank
	Replace Programas  With crSigCdAcUs.Programas
	Replace Parametros With crSigCdAcUs.Parametros
	Replace Grupos     With ThisForm.GetGrupo.Value
	Replace pkChaves  With Sys(2015)+Sys(2015)
	=fGravarLog('I', 'SIGCDGR1', crSigCdAcUs.Parametros, 'Grupo : ' + Alltrim(ThisForm.GetGrupo.Value) + ' Prog : ' + Alltrim(crSigCdAcUs.Descricaos), Usuar)
EndScan

Zap In crSigCdAcB

Select crTmpBarra

Scan
	Select crTmpBarra
	Scatter Memo Memvar

	m.grupos = ThisForm.GetGrupo.Value
	m.usuarios = ''
	m.pkchaves = fUniqueIds()
	
	Insert Into crSigCdAcB From Memvar
EndScan

If m.lnErro1>0 
	m.lnErro1 = Iif(ThisForm.poDatamgr.UpDate('crSigCdGrA'), 1, -1)
EndIf 
If m.lnErro1>0 
	m.lnErro1 = Iif(ThisForm.poDatamgr.UpDate('crSigCdAcU'), 1, -1)
EndIf 
If m.lnErro1>0 
	m.lnErro1 = Iif(ThisForm.poDatamgr.UpDate('crSigCdAcB'), 1, -1)
EndIf 
If m.lnErro1>0 
	m.lnErro1 = ThisForm.poDatamgr.Commit()
EndIf 
If Not m.lnErro1>0 
	ThisForm.poDatamgr.RollBack()
	= MessageBox('Erro Na Gravação dos Dados, Favor Tentar Novamente.', 0 + 16, '')
Else
	Release Windows
EndIf 

ENDPROC
