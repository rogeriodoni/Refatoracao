PROCEDURE Init
LParameters _Titulo, _Altera, _Arquivo

ThisForm.cntSombra.lblSombra.Caption = ThisForm.Caption
ThisForm.cntSombra.lblTitulo.Caption = ThisForm.Caption

ThisForm.Arquivo = _Arquivo

If (Type('_Arquivo') <> 'C')
	ThisForm.Arquivo = 'crSigCdCli'
EndIf

If _Altera
	This.Frase.Caption = 'Conta '+Alltrim(_Titulo)+Chr(13)+'Digite a Senha Atual : '
	This.Height = 250
	This.Altera = .t.
Else
	This.Frase.Caption = 'A Conta ' + Alltrim(_Titulo) + Chr(13) + ' Possui Senha : '
	This.Altera = .f.
EndIf

ThisForm.cntSombra.lblSombra.Caption = ThisForm.Caption
ThisForm.cntSombra.lblTitulo.Caption = ThisForm.Caption

* Fim **************************************************************************************

ENDPROC

PROCEDURE Valid
Local Senha, Arquivo

Arquivo = ThisForm.Arquivo
Senha   = fCriptografar(&Arquivo..Senhas)

If (Alltrim(This.Value) == Alltrim(Senha))
	If Not ThisForm.Altera
		Thisform.ZSen = .t.
		Set Confirm Off
		Thisform.Release
	EndIf
Else
	If (Messagebox('Deseja Digitar Novamente ?', 4+256, 'Senha Conflitante...') = 7)
		Thisform.ZSen = .f.
		Set Confirm Off
		Thisform.Release
	Else
		Return 0
	EndIf
EndIf
* Fim **************************************************************************************

ENDPROC
