------------------------------------------------------------
PROCEDURE validaregistros
Local llRetorno
llRetorno = .T.

Select crSigOpVld
Select Count(*) As Qtd, Empresa, Grupos, Contas ;
	from crSigOpVld ;
	Where Empty(Empresa) Or Empty(Grupos) Or Empty(Contas) Or Empty(Dias) ;
	Group By Empresa, Grupos, Contas ;
	Into Cursor csCount Readwrite

Select csCount
If Reccount([csCount])>0
	lnResp = Messagebox('Há Registros sem a Informação da Empresa, Grupo, Conta ou Dias serão Removidos.'+Chr(13)+'Deseja corrigir os dados?',4+32+256,'Atenção')
	If lnResp = 6
		llRetorno = .F.
	Endif
Endif

Select crSigOpVld
Delete From crSigOpVld Where Empty(Dias) Or Empty(Empresa) Or Empty(Grupos) Or Empty(Contas) Or Empty(Dopes)

Select crSigOpVld
Select Count(*) As Qtd, Empresa, Grupos, Contas ;
	from crSigOpVld ;
	Where Not Empty(Empresa) And Not Empty(Grupos) And Not Empty(Contas) ;
	Group By Empresa, Grupos, Contas ;
	Having Count(*) > 1 ;
	Into Cursor csDup Readwrite

Select csDup
If Reccount([csDup])>0
	=Messagebox('Há Registros Duplicados.'+Chr(13)+'Favor Verificar os Dados?',64,'Atenção')
	llRetorno = .F.
Endif

Return llRetorno

ENDPROC

If Thisform.cmdInserir.Visible Or Thisform.cmdExcluir.Visible
	If ( Thisform.lHouveExcl Or Thisform.lHouveIns Or ThisForm.lHouveAlt )
		
		If Not ThisForm.ValidaRegistros()

		EndIf 
				
		Select crSigOpVld

		With Thisform.ParentForm
			If m.lnErro1>0 
				m.lnErro1 = Iif(.poDataMgr.UpDate('crSigOpVld'),1,-1)
			EndIf 
			If m.lnErro1>0 
				m.lnErro1 = .poDataMgr.Commit()
			EndIf 
			If Not m.lnErro1>0 
				.poDataMgr.RollBack()
			EndIf 
		EndWith 
		If Not m.lnErro1>0 
			= MessageBox('Erro na Gravação dos Dados do Cadastro de Operações.',48)
		EndIf 
	EndIf
ENDIF

Thisform.ParentForm.Enabled = .T.
ThisForm.Release
ENDPROC
