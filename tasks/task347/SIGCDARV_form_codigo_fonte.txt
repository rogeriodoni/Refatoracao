* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
=fConfigGeral()

With ThisForm
	.poDataMgr = CreateObject('fSqlConector', ThisForm.Name)

	If (.poDataMgr.pnIdconn > 0)
        lcQryArv = [Select * From SigCdArv]
		With .poDataMgr
			.AddCursor('SigCdArv', 'codigos', 'crSigCdArv', '', ThisForm.Pagina.Lista.Grade, lcQryArv)
			.pfSqlTabela(1).pColuna('Codigos', '', '', 'Códigos', 80, .T.)
			.pfSqlTabela(1).pColuna('Descs', '', '', 'Descrição', 290, .F.)
		EndWith

		=DoDefault()
        Go Top
		ThisForm.Pagina.Lista.Grade.Column1.SetFocus()
	Else
		=Messagebox('Impossivel Efetuar Conexão com Servidor de Banco de Dados...', 0+48, 'CreateObject')
		Return .f.
	EndIf
EndWith
* Fim **************************************************************************************

ENDPROC

------------------------------------------------------------
* PROCEDURE msv_procurar
------------------------------------------------------------
PROCEDURE msv_procurar
If .Not. Empty(ThisForm.Pagina.Dados.Get_Cod.Value)
	ThisForm.poDataMgr.LocateCursor('csSigCdArv','Codigos',ThisForm.Pagina.Dados.Get_Cod.Value)
	ThisForm.Pagina.Lista.Grade.Column1.SetFocus()
EndIf
=DoDefault()

ENDPROC

------------------------------------------------------------
* PROCEDURE Release
------------------------------------------------------------
PROCEDURE Release
ThisForm.poDataMgr.Release
=DoDefault()
ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Lista.Grupo_op.Click
------------------------------------------------------------
PROCEDURE Pagina.Lista.Grupo_op.Click
LParameters opcao

If !dodefault( opcao )
	ThisForm.mAtivapagina1
	Return
Endif

Select crSigCdArv

Do Case
	Case Inlist( ThisForm.pcEscolha, 'INSERIR', 'PROCURAR' )
		ThisForm.Pagina.Dados.Get_Cod.SetFocus()
	Case ThisForm.pcEscolha = 'ALTERAR'
		ThisForm.Pagina.Dados.get_Descs.SetFocus()
Other
	ThisForm.Pagina.Dados.Grupo_Salva.Salva.SetFocus()
Endcase

ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
If ThisForm.pcEscolha = 'INSERIR'
	If IsEmpty( ThisForm.Pagina.Dados.Get_Cod.Value )
		Messagebox( 'Código não pode ficar em branco!', 48, '' )
		ThisForm.Pagina.Dados.Get_Cod.SetFocus()
		Return
	Endif
	If ThisForm.poDataMgr.ChkRegister('SigCdArv','codigos',ThisForm.Pagina.Dados.Get_Cod.Value)
		MessageBox( 'Tipo de Árvore já cadastrada!', 48, '' )
		ThisForm.Pagina.Dados.Get_Cod.SetFocus
		Return
	Endif
Endif

=DoDefault()

ThisForm.mAtivapagina1()

ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Dados.Grupo_Salva.Cancelar.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Cancelar.Click
=DoDefault()
If ThisForm.plCancelar
	ThisForm.mAtivapagina1()
Endif

ENDPROC
