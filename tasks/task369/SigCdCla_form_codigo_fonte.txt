PROCEDURE msv_procurar
Local lcBusca

	Case Not Empty(ThisForm.Pagina.Dados.Getcods.Value)

		ThisForm.poDataMgr.LocateCursor([crSigCdCls], [Descs], ThisForm.Pagina.Dados.Getdescs.Value)
EndCase
* Fim **************************************************************************************
ENDPROC

* PROCEDURE Init

PROCEDURE Init
With ThisForm
	.poDataMgr = CreateObject([fSqlConector], .Name)

	If (.poDataMgr.pnIdconn > 0)
		With .poDataMgr
			.AddCursor([SigCdCls], [Cods], [crSigCdCls], [], ThisForm.Pagina.Lista.Grade)
			.pfSqlTabela(1).pColuna([Cods],  [], [], [Código],    045, .t.)
			.pfSqlTabela(1).pColuna([Descs], [], [], [Descrição], 290, .t.)

			.AddCursor([SigCdCla], [cIdChaves], [crSigCdCom])
		EndWith

		=DoDefault()

		.Pagina.Lista.Grade.Column1.SetFocus
	Else
		=Messagebox([Impossível Efetuar Conexão Com o Servidor de Banco de Dados...], 0+48, [CreateObject])
		Return .f.
	EndIf
EndWith
* Fim **************************************************************************************
ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Lista.Grupo_op.Click
------------------------------------------------------------
PROCEDURE Pagina.Lista.Grupo_op.Click
LParameters lnOpcao

If (Type([lnOpcao]) = [N])
	This.Value = lnOpcao
EndIf

Select crSigCdCls

If Not DoDefault(lnOpcao)
	ThisForm.mAtivaPagina1
	Return .f.
EndIf

Zap In crGrdCdCom

lcQuery = [Select * ] + ;
		    [From SigCdCla ] + ;
		   [Where Cods = '] + crSigCdCls.Cods + [' ] + ;
		   [Order By CGrus]

If (ThisForm.poDataMgr.SqlExecute(lcQuery, [LocalCdCom]) < 1)
	=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (LocalCdCom)])
	Return .f.
EndIf

Select LocalCdCom
Scan
	lcQuery = [Select DGrus ] + ;
			    [From SigCdGrp ] + ;
			   [Where CGrus = '] + LocalCdCom.CGrus + [']

	If (ThisForm.poDataMgr.SqlExecute(lcQuery, [LocalGru]) < 1)
		=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (LocalGru)])
		Return .f.
	EndIf
	Go Top In LocalGru

	Insert Into crGrdCdCom (CGrus, DGrus) Values (LocalCdCom.CGrus, LocalGru.DGrus)
EndScan
Go Top In crGrdCdCom
If Eof([crGrdCdCom]) And InList(ThisForm.pcEscolha, [INSERIR], [ALTERAR])
	Append Blank In crGrdCdCom
	Go Top In crGrdCdCom
EndIf


loDad = ThisForm.Pagina.Dados

Do Case
	Case Inlist(ThisForm.pcEscolha, [PROCURAR], [INSERIR])
		loDad.getCods.Setfocus	

	Case (ThisForm.pcEscolha = [ALTERAR])
		loDad.getDescs.Setfocus	

	Case Inlist(ThisForm.pcEscolha, [CONSULTAR], [EXCLUIR])
		loDad.Grupo_Salva.Salva.Setfocus
EndCase

Select crSigCdCls

loDad.Refresh
* Fim **************************************************************************************
ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
Local loDad, lcPro

loDad = ThisForm.Pagina.Dados

Select crSigCdCls

If (ThisForm.pcEscolha = [INSERIR])
	If Empty(loDad.getCods.Value)
		=MessageBox([Código Não Informado...], 0+48, [Atenção!!!])
		loDad.getCods.SetFocus
		Return .f.
	EndIf

	lcQuery = [Select * ] + ;
			    [From SigCdCls ] + ;
			   [Where Cods = '] + loDad.getCods.Value + [']

	If (ThisForm.poDataMgr.SqlExecute(lcQuery, [LocalClas]) < 1)
		=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (LocalClas)])
		loDad.getCods.SetFocus
		Return .f.
	EndIf
	Select LocalClas
	Go Top In LocalClas
	If Not Eof([LocalClas])
		=MessageBox([Código Já Cadastrado...], 0+48, [Atenção!!!])
		loDad.getCods.SetFocus
		Return .f.
	EndIf
EndIf

If InList(ThisForm.pcEscolha, [INSERIR], [ALTERAR])

	If Empty(crSigCdCls.Tipos)
		=MessageBox([Tipo Inválido!!!], 16, [Atenção!!!])
		ThisForm.Pagina.Dados.getTipos.SetFocus
		Return .f.
	EndIf

    If Empty(ThisForm.Pagina.Dados.optSituas.Value)
		=Messagebox([Situação de Uso Em Produtos Inválida!!!], 0+48, [Atenção!!!])
		Return .f.
	EndIf
EndIf

If InList(ThisForm.pcEscolha, [EXCLUIR], [ALTERAR])
	If (ThisForm.pcEscolha = [EXCLUIR])
		If ThisForm.poDataMgr.CursorQuery([SigCdPro],,[CClass], crSigCdCls.Cods)
			=Messagebox([Existem Produtos Cadastrados Com Esta Classificacao!!!] + Chr(13) + [Impossível Executar a Exclusão!!!], 0+48, [Atenção!!!])
			ThisForm.mAtivaPagina1
			Return .f.
		EndIf
	EndIf
	
	lcQuery = [Delete From SigCdCls ] + ;
			   [Where Cods = '] + crSigCdCls.Cods + [']

	If (ThisForm.poDataMgr.SqlExecute(lcQuery, []) < 1)
		=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (Delete - LocalClas)])
		Return .f.
	EndIf

	lcQuery = [Delete From SigCdCla ] + ;
			   [Where Cods = '] + crSigCdCls.Cods + [']

	If (ThisForm.poDataMgr.SqlExecute(lcQuery, []) < 1)
		=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (Delete - LocalCdCla)])
		ThisForm.poDataMgr.RollBack()
		Return .f.
	EndIf
EndIf

Zap In crSigCdCom

Select Distinct CGrus ;
  From crGrdCdCom ;
 Where Not Empty(CGrus) ;
 Order By CGrus ;

PROCEDURE Click
Select crGrdCdCom

lcGde = ThisForm.Pagina.Dados.getGdeGrps.Value

If Empty(lcGde)
	=MessageBox([Preencha o Grande Grupo Antes de Processar!], 0+16, [Atenção!])
	ThisForm.Pagina.Dados.getGdeGrps.SetFocus
	Return .f.
EndIf

lcQuery = [Select CGrus, DGrus ] + ;
		    [From SigCdGrp ] + ;
		    [Where Mercs = '] + lcGde + [']

If (ThisForm.poDataMgr.SqlExecute(lcQuery, [LocalGru]) < 1)
	=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (LocalGru)])
	Return 0
EndIf
Go Top In LocalGru
If Not Eof([LocalGru])
	Delete From crGrdCdCom Where Empty(CGrus)
EndIf

Select LocalGru
Scan
	Select crGrdCdCom
	Go Top In crGrdCdCom
	Locate For CGrus = LocalGru.CGrus
	If Eof([crGrdCdCom])
		Insert Into crGrdCdCom (CGrus, DGrus) Values (LocalGru.CGrus, LocalGru.DGrus)
	EndIf
EndScan
Go Top In LocalGru
If Not Eof([LocalGru])
	Append Blank In crGrdCdCom
EndIf
ThisForm.Pagina.Dados.grdProdutos.Refresh 
Go Bottom In crGrdCdCom
ThisForm.Refresh
ThisForm.Pagina.Dados.grdProdutos.Column1.SetFocus
* Fim **************************************************************************************
ENDPROC
