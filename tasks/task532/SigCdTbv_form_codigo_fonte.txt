------------------------------------------------------------
* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
Select CrSigOpTds
Set Order To

Do Case
	Case InList(ThisForm.pcEscolha,'INSERIR','ALTERAR')
		If ThisForm.pcEscolha = 'INSERIR'
			If Empty(ThisForm.pagina.dados.page1.pgDados.get_codigos.Value)
				Messagebox('Código Inválido !!!',48,'')
				ThisForm.Pagina.Dados.Page1.PgDados.Get_Codigos.SetFocus
				Return
			EndIf
			If ThisForm.poDataMgr.CursorQuery('SigOpTdz',,'codigos',CrSigOpTdz.codigos)
				Messagebox('Código já cadastrado !!!',48,'')
				ThisForm.Pagina.Dados.Page1.PgDados.Get_Codigos.SetFocus
				Return
			EndIf
		Endif
		
		*-- Rafael - 29/06/2016 - Validação das Parcelas Inicial e Final
		If IsEmpty(This.parent.parent.Page1.pgDados.Get_nParcs.Value) And Not IsEmpty(This.parent.parent.Page1.pgDados.Get_nParc2s.Value)
			Messagebox('Quantidade de Parcelas Inicial Inválida !!!',48,'')
			This.parent.parent.Page1.pgDados.Get_nParcs.SetFocus
			Return	
		EndIf 
		
		If Not IsEmpty(This.parent.parent.Page1.pgDados.Get_nParcs.Value) And Not IsEmpty(This.parent.parent.Page1.pgDados.Get_nParc2s.Value)
			If This.parent.parent.Page1.pgDados.Get_nParcs.Value > This.parent.parent.Page1.pgDados.Get_nParc2s.Value
				Messagebox('Quantidade de Parcelas Inicial Maior que a Final !!!',48,'')
				This.parent.parent.Page1.pgDados.Get_nParc2s.SetFocus
				Return	
			EndIf 
		EndIf 
		*--
		
		*** Tiago - 21/01/2016 - Incluído para checar se as fórmulas estão corretas, para não dar erro depois nas movimentações
		Select crSigOpTdt
		Scan
			If crSigOpTdt.Campo1t1s > 1 And ((crSigOpTdt.Oper1t1s > 1 And Inlist(crSigOpTdt.Campo2t1s,0,1)) Or (crSigOpTdt.Oper2t1s > 1 And Inlist(crSigOpTdt.Campo3t1s,0,1)) Or ;
				((crSigOpTdt.Campo1t1s = 5 And Empty(crSigOpTdt.Const1t1s)) Or (crSigOpTdt.Campo2t1s = 5 And Empty(crSigOpTdt.Const2t1s)) ;
				Or (crSigOpTdt.Campo3t1s = 5 And Empty(crSigOpTdt.Const3t1s))))
						
				Messagebox([Fórmula 1 do Código '] + crSigOpTdt.CodTips + [' está incorreta.],48,[Aviso])
				Return .f.
			Endif
			If crSigOpTdt.Campo1t2s > 1 And ((crSigOpTdt.Oper1t2s > 1 And Inlist(crSigOpTdt.Campo2t2s,0,1)) Or (crSigOpTdt.Oper2t2s > 1 And Inlist(crSigOpTdt.Campo3t2s,0,1)) Or ;
				((crSigOpTdt.Campo1t2s = 5 And Empty(crSigOpTdt.Const1t2s)) Or (crSigOpTdt.Campo2t2s = 5 And Empty(crSigOpTdt.Const2t2s)) ;
				Or (crSigOpTdt.Campo3t2s = 5 And Empty(crSigOpTdt.Const3t2s))))			
				Messagebox([Fórmula 2 do Código '] + crSigOpTdt.CodTips + [' está incorreta.],48,[Aviso])
				Return .f.
			Endif			
		Endscan
		
		*** mena - 09/02/2017 - le empresas marcadas e grava na string
		lcEmp = []
		SELECT crSigCdemp
		GO top
		SCAN
			IF nMarca = 1
				lcEmp = lcEmp+[;]+crSigCdemp.cemps	
			ENDIF
		ENDSCAN
		replace listaemp WITH  lcEmp IN crSigOpTdz
		***************************************************************
		
		Update CrSigOpTdt Set cidchaves = Iif(Empty(cidchaves),fUniqueIds(),cidchaves)
		Update CrSigOpTdi Set cidchaves = Iif(Empty(cidchaves),fUniqueIds(),cidchaves)
		Update CrSigOpTdi Set Codigos   = CrSigOpTdz.Codigos
		Update CrSigOpTds Set cidchaves = Iif(Empty(cidchaves),fUniqueIds(),cidchaves)

		Delete From CrSigOpTdt Where Empty(tipos)
		Delete From CrSigOpTds Where Left(Campo1Is,2) = '  ' or ( descos = 0 And dFators = 0 )

		_nChv = CrSigOpTdz.nChvTbds
		Select CrSigOpTds
		Scan
			_nChv = _nChv + 1
			Replace NChvTbds with _nChv in CrSigOpTds
		EndScan
		Update CrSigOpTdz Set nChvTbds = _nChv

	Case ThisForm.pcEscolha = 'EXCLUIR'
		Delete From CrSigOpTdt
		Delete From CrSigOpTdi
		Delete From CrSigOpTds

EndCase

=DoDefault()
ThisForm.mAtivaPagina1
ENDPROC
