* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
LParameters pcForm

ThisForm.DataSessionId = pcForm.DataSessionId

Select crImpressao
Zap

Select crSigFiMpi

Scan

	Select Substr(ecfs,1,3) As cCaixas, sum(Valos) As Valos  From crCaixa ; 
	       Where Emps = pcForm.Pagina.Dados.PageFrame1.Page1.Get_emps.Value ;
	             And Datas = pcForm.Pagina.Dados.PageFrame1.Page1.Get_datas.Value ;
	             And Substr(ecfs,1,3) = crSigFiMpi.nCaixa ;
	       Group By 1 Into Cursor crTotCaixa ReadWrite
	       
	Select Substr(ecfs,1,3) As cCaixas, sum(Valos) As Valos  From crCaixaI ; 
	       Where Emps = pcForm.Pagina.Dados.PageFrame1.Page1.Get_emps.Value And ;
	       Datas = pcForm.Pagina.Dados.PageFrame1.Page1.Get_datas.Value ;
	       And Substr(ecfs,1,3) = crSigFiMpi.nCaixa Group By 1 Into Cursor crTotCaixaI

	Select Substr(ecfs,1,3) As cCaixas, sum(Valos) As Valos  From crCaixaFIN ; 
	       Where Emps = pcForm.Pagina.Dados.PageFrame1.Page1.Get_emps.Value And ;
	       Datas = pcForm.Pagina.Dados.PageFrame1.Page1.Get_datas.Value ;
	       And Substr(ecfs,1,3) = crSigFiMpi.nCaixa Group By 1 Into Cursor crTotCaixaFIN

	If round(crSigFiMpi.VlContabil,2)+round(crSigFiMpi.iss,2)+Round(crSigFiMpi.Isento, 2) <> round(crTotCaixa.Valos,2) Or round(crTotCaixa.Valos,2) <> round(crTotCaixaI.Valos,2)
		Insert Into crImpressao (Datas, Emps, Caixas, ValCxs, ValItens, ValMrs) ; 
		       Values (pcForm.Pagina.Dados.PageFrame1.Page1.Get_datas.Value,;
		               pcForm.Pagina.Dados.PageFrame1.Page1.Get_emps.Value,;
		               crSigFiMpi.nCaixa, crTotCaixa.Valos-crTotCaixaFIN.Valos, crTotCaixaI.Valos,;
		               crSigFiMpi.VlContabil+crSigFiMpi.iss)
	Endif

EndScan 

Select Substr(ecfs,1,3) As cCaixas, sum(Valos) As Valos  From crCaixa ; 
       Where Emps = pcForm.Pagina.Dados.PageFrame1.Page1.Get_emps.Value ;
             And Datas = pcForm.Pagina.Dados.PageFrame1.Page1.Get_datas.Value ;
       Group By 1 Into Cursor crTotCaixa ReadWrite

Scan
	Select  crSigFiMpi
	Locate for nCaixa = crTotCaixa.cCaixas
	If Eof()
		Select Substr(ecfs,1,3) As cCaixas, sum(Valos) As Valos  From crCaixaI ; 
		       Where Emps = pcForm.Pagina.Dados.PageFrame1.Page1.Get_emps.Value And ;
		       Datas = pcForm.Pagina.Dados.PageFrame1.Page1.Get_datas.Value ;
		       And Substr(ecfs,1,3) = crTotCaixa.cCaixas Group By 1 Into Cursor crTotCaixaI

		Insert Into crImpressao (Datas, Emps, Caixas, ValCxs, ValItens, ValMrs) ; 
		       Values (pcForm.Pagina.Dados.PageFrame1.Page1.Get_datas.Value,;
		               pcForm.Pagina.Dados.PageFrame1.Page1.Get_emps.Value,;
		               crTotCaixa.cCaixas, crTotCaixa.Valos, crTotCaixaI.Valos,0)
	Endif
EndScan 			

Select crImpressao
Index On Caixas Tag Caixas
Set Order To Caixas
Go Top

ThisForm.Grid1.RecordSource = 'crImpressao'
ThisForm.Grid1.Column1.ControlSource =   'crImpressao.ValCxs'
ThisForm.Grid1.Column2.ControlSource =   'crImpressao.ValItens'
ThisForm.Grid1.Column3.ControlSource =   'crImpressao.ValMrs'
ThisForm.Grid1.Column4.ControlSource =   'crImpressao.ValNfs'
ThisForm.Grid1.Column5.ControlSource =   'crImpressao.ValCxs-crImpressao.ValMrs'
ThisForm.Grid1.Column6.ControlSource =   'crImpressao.ValCxs-crImpressao.ValItens'
ThisForm.Grid1.Column7.ControlSource =   'crImpressao.Caixas'

ENDPROC
