PROCEDURE msv_inserir
llOk = .t.

If Not Thisform.poDataMgr.Update('crSigCdCrf')
	=Messagebox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (Update - crSigCdCrf)')
	llOks = .F.
Endif

If llOk And Not Thisform.poDataMgr.Update('crSigCdCri')
	=Messagebox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (Update - crSigCdCri)')
	llOk = .F.
Endif
If llOk
	ThisForm.poDataMgr.Commit()
Else
	Thisform.poDataMgr.Rollback()
	Return .F.
Endif
ENDPROC

		Select RecNo() As nRecNo From csSigCdCri Where CBars = Int(Val(Alltrim(lnCBars))) Into Cursor csProcura
	Else
		lcCpros = This.Value
		Select RecNo() As nRecNo From csSigCdCri Where Cpros = lcCpros Into Cursor csProcura
	EndIf
	
	Select csProcura
	Go Top
	If Not Eof()
		Select csSigCdCri
		Goto csProcura.nRecNo
		
		Select csSigCdCri
		= ThisForm.poDataMgr.CursorQuery('SigCdPro','crTPro','Cpros',csSigCdCri.Cpros,'FigJpgs')
		If Not IsNull(crTPro.FigJpgs) And Not IsEmpty(crTPro.FigJpgs)
			This.parent.ImgJpg.Visible = .F.
			This.parent.ImgJpg.Picture = ''
			Select crTPro
			lcFoto=Strtran(Strtran(Strtran(crTPro.FigJpgs,"data:image/png;base64,",""),"data:image/jpeg;base64,",""),"data:image/jpg;base64,","")
			If (StrToFile(Strconv(lcFoto,14),lcArquivo2) > 0) 
				This.parent.ImgJpg.Picture = (lcArquivo2)
				This.parent.ImgJpg.Visible = .T.
			EndIf 
		Else
			This.parent.ImgJpg.Visible = .F.
			This.parent.ImgJpg.Picture = ''
		EndIf
		
	Else
		MessageBox([Produto Não Localizado!],0+64,[Atenção])
		This.Value = []
		Return .F.
	EndIf
	 	
EndIf
	

ENDPROC
