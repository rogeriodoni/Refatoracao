* PROCEDURE processa
------------------------------------------------------------
PROCEDURE processa
Local llAtvCnt

llAtvCnt = .f.
lcSql = [Select Dopes,TipoOps,Nivels,LimCres,ObsOrigs,ObsDests,ExibSits From SigCdOpe Where Dopes = ']+ThisForm.Operacao+[']
If ThisForm.ParentForm.PodataMgr.Sqlexecute(lcsql,'LocalOpe') < 1
	MessageBox('Favor reinicializar o processo.',16,'Falha na Conexão (LocalOpe)')
	Return(.F.)
EndIf
Tipoop = LocalOpe.TipoOps
_Nivel = LocalOpe.Nivels

Thisform.Container1.Visible = .f.
ThisForm.Container2.Visible = .f.
ThisForm.Container3.Visible = .f.

If ThisForm.OriDes > 0
	ThisForm.Sit_Destino.Enabled = .f.
	ThisForm.Sit_Origem.Enabled = .f.
EndIf

ThisForm.Retorno = .t.

_GrupoO = TprMvCab.grupoos
_ContaO = TprMvCab.Contaos
_GrupoD = TprMvCab.grupods
_ContaD = TprMvCab.Contads

Select TprMvCab
Do Case
	Case ThisForm.OriDes = 1 Or ThisForm.ClickOriDes = 1
		_Grupo = TprMvCab.grupoos
		_Conta = TprMvCab.Contaos
		ThisForm.Sit_Origem.Enabled = .t.

	Case ThisForm.OriDes = 2 Or ThisForm.ClickOriDes = 2
		_Grupo = TprMvCab.grupods
		_Conta = TprMvCab.Contads
		ThisForm.Sit_Destino.Enabled = .t.

	OtherWise
	
		If InList(LocalOpe.limcres,1,3,5)
			_Grupo = TprMvCab.grupoos
			_Conta = TprMvCab.Contaos
		Else
			If InList(LocalOpe.LimCres,2,3,6)
				_Grupo = TprMvCab.grupods
				_Conta = TprMvCab.Contads
			Else
				_Grupo = ' '
				_Conta = ' '
			EndIf
		Endif
EndCase

If ( InList(LocalOpe.LimCres,1,3,5) And ( ( InList(ThisForm.OriDes,0,1) And ThisForm.ClickOriDes = 0 ) Or ThisForm.ClickOriDes = 1 ) ) Or ;
	( InList(LocalOpe.LimCres,2,3,6) And ( ( InList(ThisForm.OriDes,0,2) And ThisForm.ClickOriDes = 0 ) Or ThisForm.ClickOriDes = 2 ) )

	llAtvCnt = .t.
	Thisform.Container1.Visible = .t.

	=ThisForm.parentForm.PoDataMgr.CursorQuery( 'SigClLcr','CrSigClLcr','GruContas',_Grupo + _Conta )

	lcSql = [Select * From SigOpPed ]
	If ThisForm.ParentForm.PodataMgr.Sqlexecute(lcsql,'LocalOpePd') < 1
		MessageBox('Favor reinicializar o processo.',16,'Falha na Conexão (LocalOpePd)')
		Return(.F.)
	EndIf

	Select DbOpera
	Index On dopes Tag Dopes

	If Reccount('LocalOpePd') > 0

		lStrQuery = [Select Dopes,OpPends From SigCdOpe ]
		If ThisForm.ParentForm.poDataMgr.SqlExecute(lStrQuery, 'CrTmpOpe') < 1
			MessageBox('Favor reinicializar o processo.',16,'Falha na Conexão (CrTmpOpe)')
			Return(.F.)
		EndIf

		Select CrTmpOpe
		Index On Dopes Tag dopes

		lStrQuery = [Select a.dopes, a.EmpDopNums, a.ChkPagos, a.ChkSubn, a.Utilizados,a.Valos From SigMvCab a, SigOpPed b ]+;
					[Where a.ContaDs = ']+_Conta+[' And a.Dopes = b.Dopes ]+ ;
					[Union all ]+;
					[Select a.dopes, a.EmpDopNums, a.ChkPagos, a.ChkSubn, a.Utilizados,a.Valos From SigMvCab a, SigOpPed b ]+;
					[Where a.ContaOs = ']+_Conta+[' And a.ContaDs <> ']+_Conta+[' And a.Dopes = b.Dopes ]
		If ThisForm.ParentForm.poDataMgr.SqlExecute(lStrQuery, 'CrTprMvCab') < 1
			MessageBox('Favor reinicializar o processo.',16,'Falha na Conexão (CrTprMvCab)')
			Return(.F.)
		EndIf

		Select a.dopes,a.empDopNums,a.Valos, a.ChkSubn From CrTprMvCab a, LocalOpePd b Where a.Dopes = b.Dopes And ;
		Iif(b.Tpagos = 1,a.ChkPagos,Iif(b.Tpagos = 2,Not a.ChkPagos, 0=0 )) And ;
		Iif(b.Tutils = 1,a.Utilizados = 1,Iif(b.Tutils = 2,Utilizados = 0, 0=0 ) ) ;
		Into Cursor CrConserto2
		
		Select CrConserto2
		Scan
			lcSql = [Select Dopes, Moedas, Sum( ( Qtds-QtBaixas ) * (Units+ValRats) ) as Totas From SigMvItn ]+;
					[Where EmpDopNums = ']+CrConserto2.EmpDopNums+[' Group by Dopes, Moedas]
			ThisForm.ParentForm.PodataMgr.Sqlexecute(lcSql,'lCredEesti')
			If Reccount('lCredEestI') > 0
				Select lCredEesti
				Scan
					If Not Seek(lCredEesti.Dopes,'DbOpera')
						Insert Into DbOpera (Dopes,Totas) Values (lCredEesti.Dopes,0)
					EndIf
					_ttConv = Round( lCredEesti.Totas * fBuscarCotacao(lCredEesti.moedas,date()) / fBuscarCotacao( crSigClLcr.moedas,date() ), 2 )
					Replace Totas With Totas + _ttConv in DbOpera
				EndScan
			Else
 				If Not CrConserto2.ChkSubn
	 				If Not Seek(CrConserto2.Dopes,'DbOpera')
						Insert Into DbOpera (Dopes,Totas) Values (CrConserto2.Dopes,0)
					EndIf
					Replace Totas With Totas + CrConserto2.Valos In DbOpera
				Endif
			Endif
		EndScan
	EndIf

	lcSql = [Select Rclis From SigCdCli where Iclis = ']+_Conta+[']
	If ThisForm.ParentForm.PodataMgr.Sqlexecute(lcSql,'CrTmpCli') < 1
		MessageBox('Favor reinicializar o processo.',16,'Falha na Conexão (CrTmpCli)')
		Return(.F.)
	Endif

	lcRazao = CrTmpCli.Rclis

	ThisForm.Container1.get_nconta.Caption = _Conta + ' '+Alltrim(lcRazao)

	lcSql = [Select distinct GrupoDs From SigOpFp Where Situas = 'S' And GrupoDs <> ' ' ]
	If ThisForm.ParentForm.PodataMgr.SqlExecute(lcSql,'LocFpg') < 1
		=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (LocFpg)')
		Return .f.
	EndIf
	_GrupoCc = _Grupo
	If Reccount('LocFpg') > 0 And .f. &&desabilitado para ajuste no adiantamento a fornecedores
		Go Top in LocFpg
		_GrupoCc = LocFpg.GrupoDs
	Endif

	Declare LaCampo(2,2)
	LaCampo(1,1)='Grupos'
	LaCampo(2,1)='Contas'
	LaCampo(1,2)=_GrupoCc
	LaCampo(2,2)=_Conta

	=ThisForm.ParentForm.PoDataMgr.CursorQuery('SigMvSlc','CrTmpSccr',@LaCampo)

	Select CrTmpSccr
	_Saldocc = 0
	Scan
		_Saldocc = _Saldocc + Round(Saldos * fBuscarCotacao(CrTmpSccr.moedas,date()) / fBuscarCotacao(CrSigClLcr.moedas,date()), 2 )
	Endscan
	ThisForm.Container1.get_contac.Value = _Saldocc

	lcQuery = [Select b.grclis, b.iclis, sum(b.valors) as Valors, b.fpags ]+;
			  [From SigCqChm a, SigChe b, SigCdOpt c Where ]+;
			  [b.GrClis = ']+_Grupo+[' And b.Iclis = ']+_Conta+[' And b.umovs = c.operacaos and ]+;
			  [c.tipos='DB' and a.nums = b.nums Group by b.GrClis, b.Iclis, b.Fpags]

	If ThisForm.ParentForm.PodataMgr.SqlExecute(lcQuery,'Cheques') < 1
		=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (cheques)')
		Return .f.
	Endif
	_chequec = 0
	_chequep = 0
	Select Cheques
	Scan
		If IsEmpty(fpags)
			_chequep = _chequec + valors
		Else
			=Thisform.parentform.podatamgr.cursorquery('SigOpFp','CrSigOpFp','Fpags',Cheques.Fpags)
			If crSigOpFp.valpends # 1
				_chequec = _chequec + valors
			Else
				_chequep = _chequep + valors
			Endif
		Endif
	Endscan
	_Chequec = Round( _chequec / fBuscarCotacao( crSigClLcr.moedas,date() ), 2 )
	_Chequep = Round( _chequep / fBuscarCotacao( crSigClLcr.moedas,date() ), 2 )

	ThisForm.Container1.get_cheque.Value = _chequec 
	ThisForm.Container1.get_chqpend.Value = _chequep

	_ChvLCred = TprMvCab.Emps + _Nivel + Str(TprMvCab.Numes,6)

	lcQuery = [Select sum( (b.Qtds - b.QtBaixas) * b.Units ) as Totas, b.Moedas, a.Dopes From SigMvCab a, SigMvItn b, SigCdOpe c ]+;
			  [Where ]+Iif(InList(LocalOpe.LimCres,1,3,5),[a.Grupoos='],[a.Grupods='])+_Grupo+[' And ]+;
			  Iif(InList(LocalOpe.LimCres,1,3,5),[a.Contaos='],[a.contads='])+_Conta+[' And Not a.EmpDopNums=']+_chvlCred+[' And ]+;
			  [a.EmpDopNums = b.EmpDopNums and a.Dopes = c.Dopes And c.Tipoops = ]+Str(CrSigCdPam.Oppends,2)+[ ]+;
			  [Group by b.moedas,a.Dopes]

	If ThisForm.ParentForm.PodataMgr.SqlExecute(lcQuery,'Pendencia') < 1
		=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (Pendencia)')
		Return .f.
	Endif

	_Pendente = Round( TprMvCab.valos / fBuscarCotacao( crSigClLcr.moedas,date() ), 2 )
	_PendOutr = 00

	Select Pendencia
	Scan
		_ttConv = Round( totas * fBuscarCotacao(moedas,date()) / fBuscarCotacao( crSigClLcr.moedas,date() ), 2 )
		_PendOutr = _PendOutr + _ttConv
	EndScan

	Select DbOpera
	Go Top
	With ThisForm.container1.Grade
		.RecordSource = "DbOpera"
		.Column1.ControlSource = "DbOpera.Dopes"
		.Column2.ControlSource = "DbOpera.Totas"
		.Refresh
	EndWith

	ThisForm.Parentform.poDataMgr.Cursorquery('SigCdTom', 'LocalTOpe', 'Codigos', TipoOp)

	_SaldoFinal = crSigClLcr.limcres - Iif(LocalTOpe.GerGdmis=3,(_Pendente + _PendOutr),0) - ;
				  _Saldocc - Iif(LocalTOpe.GerGdmis=3 Or crSigCdPam.ChqlCreds,_Chequec,0)

	ThisForm.Container1.get_limcred.Value = crSigClLcr.limcres
	ThisForm.Container1.get_moeda.Value = crSigClLcr.moedas
	ThisForm.Container1.get_cotacao.Value = fBuscarCotacao( crSigClLcr.moedas, date() )
	ThisForm.Container1.get_pendente.Value = _Pendoutr
	ThisForm.Container1.get_pendoutr.Value = _Pendente
	ThisForm.Container1.get_disponivel.forecolor=Iif(_SaldoFinal < 0,RGB(255,0,0),RGB(0,0,255))
	ThisForm.Container1.get_disponivel.Value = _SaldoFinal

EndIf

If ( ( ( InList(ThisForm.OriDes,0,1) And Thisform.ClickOriDes=0 ) Or ThisForm.ClickOriDes = 1 ) And LocalOpe.ObsOrigs = 1 ) Or ;
	( ( ( InList(ThisForm.OriDes,0,2) And Thisform.ClickOriDes=0 )Or ThisForm.ClickOriDes = 2 ) And LocalOpe.ObsDests = 1 )
	
	If ( ( ( InList(ThisForm.OriDes,0,1) And Thisform.ClickOriDes=0 ) Or ThisForm.ClickOriDes = 1 ) And LocalOpe.ObsOrigs = 1 )
		lcSql = [Select Obs,Situas,VultComps,UltComps,Nascs,Codigos  From SigCdCli Where Iclis = ']+_contaO+[']
	Else
		lcSql = [Select Obs,Situas,VultComps,UltComps,Nascs,Codigos  From SigCdCli Where Iclis = ']+_contaD+[']
	Endif
	=ThisForm.ParentForm.Podatamgr.Sqlexecute(lcSql,'CrTmpcli')
	If Not IsEmpty( CrTmpCli.Obs )
		llAtvCnt = .t.
		ThisForm.Container2.Visible = .t.
		ThisForm.container2.fwmemo1.Value = CrTmpCli.Obs
		ThisForm.container2.fwmemo1.Refresh
		ThisForm.Container2.Get_nconta.Caption = _Conta
	Endif
EndIf

If ( ( ( InList(ThisForm.OriDes,0,1) And Thisform.ClickOriDes=0 ) Or ThisForm.ClickOriDes = 1) And LocalOpe.ExibSits = 3 ) Or ;
	( ( ( InList(ThisForm.OriDes,0,2) And Thisform.ClickOriDes=0 ) Or ThisForm.ClickOriDes = 2 ) And LocalOpe.ExibSits = 1 )

	If ( ( ( InList(ThisForm.OriDes,0,1) And Thisform.ClickOriDes=0 ) Or ThisForm.ClickOriDes = 1) And LocalOpe.ExibSits = 3 )
		lcSql = [Select Obs,Situas,VultComps,UltComps,Nascs,Codigos,DtnCons,DtCasas  From SigCdCli Where Iclis = ']+_contaO+[']
	Else
		lcSql = [Select Obs,Situas,VultComps,UltComps,Nascs,Codigos,DtnCons,DtCasas  From SigCdCli Where Iclis = ']+_contaD+[']
	Endif
	=ThisForm.ParentForm.Podatamgr.Sqlexecute(lcSql,'CrTmpcli')
	=ThisForm.parentForm.PodataMgr.Cursorquery('SigCdCst','CrTmpScli','Codigos',CrTmpCli.Situas)

	If CrtmpScli.DispMovs = 1 And Not Empty(_Conta)
	
		lcdSit = Iif(Empty(CrTmpScli.Descr2s),CrTmpScli.Descrs,CrTmpScli.Descr2s)

		llAtvCnt = .t.
		ThisForm.Container3.Visible = .t.
		ThisForm.Container3.Get_nconta.Caption = _Conta
		ThisForm.container3.lb_msg1.Caption = CrtmpCli.Situas + ' - ' + Alltrim(lcdSit)
		ThisForm.Container3.lb_msg2.Caption = Alltrim(Transform(CrTmpCli.VultComps,"@Z 999,999,999.99")) + ' / ' + Iif(!IsEmpty(CrTmpCli.UltComps),Dtoc(CrTmpCli.UltComps),'')
		ThisForm.Container3.lb_msg3.Caption = Iif(!IsEmpty(CrTmpCli.Nascs),Dtoc(CrTmpCli.Nascs),'')
		ThisForm.Container3.lb_msg4.Caption = CrTmpCli.Codigos + ' ' + Iif(!IsEmpty(CrTmpCli.DtnCons),'j - '+Dtoc(CrTmpCli.DtnCons),'') + ;
											' ' + Iif(!IsEmpty(CrTmpCli.DtCasas),'c - '+Dtoc(CrTmpCli.Dtcasas),'' )
		ThisForm.container3.lb_MSG5.Visible = .f.
			        
		If Thisform.OriDes > 0
			If CrTmpScli.BlqMovs = 1
				ThisForm.container3.lb_MSG5.Visible = .t.
				Thisform.Container3.lb_msg5.Caption = 'Movimentação Bloqueada pela Situacao da Conta'
				ThisForm.Retorno = .f.
			Else
				If TprMvCab.ValVars < 0 And CrTmpScli.BlqDescs = 1
					ThisForm.container3.lb_MSG5.Visible = .t.
					Thisform.Container3.lb_msg5.Caption = 'Desconto Bloqueado Pela Situação da Conta'
					ThisForm.Retorno = .f.
				EndIf
			EndIf
		EndIf
	Endif
EndIf

Thisform.Refresh

Return ( llAtvCnt )

ENDPROC

			ThisForm.PoFormCaixa.Enabled = .t.
		Endif
		ThisForm.Release
	Else
		ThisForm.OriDes = 2

			If Type('ThisForm.poFormCaixa') = "O"
				ThisForm.PoFormCaixa.Enabled = .t.
			Endif
			ThisForm.Release
		EndIf
	Endif
EndIf
