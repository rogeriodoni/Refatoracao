* PROCEDURE imprimir
------------------------------------------------------------
PROCEDURE imprimir
Private pTip, pLot

Local lcChv

If Not Eof('crSigPrLit')
	lcChv = crSigPrLit.NumLotes

	pTip  = ThisForm.Tipos
	pLot  = lcChv
	
	ThisForm.Pagina.Dados.cntGradeCheques.GradeCheques.RecordSource = ''
	ThisForm.Pagina.Dados.cntGradeTitulos.GradeTitulos.RecordSource = ''
	
	Thisform.poDataMgr.Requery('crSigPrLii')

	Select crSigPrLii
	Scan
		If IsNull(DtVencs)
			Replace DtVencs With Ctod('') In crSigPrLii
		EndIf

		If IsNull(PrVencs)
			Replace PrVencs With Ctod('') In crSigPrLii
		EndIf
		
		ThisForm.poDatamgr.CursorQuery('SigCdCli', 'crSigCdCli', 'IClis', crSigPrLii.IClis)
		If Used('crSigCdCli') And (Reccount('crSigCdCli') > 0)
			Replace Cpfs  With crSigCdCli.Cpfs, ;
					RClis With crSigCdCli.RClis In crSigPrLii
		EndIf
	EndScan

	Select a.Emps, a.Datas, a.Contas, b.Bancos, b.Agencias, b.NContas, ;
		   b.NCheques, b.Titulos, b.IClis, b.DtVencs, b.PrVencs, b.Valors, ;
		   b.Moedas, b.Borderos, c.Mensas ;
	  From crSigPrLit a, crSigPrLii b, crSigOpIns c ;
	 Where a.NumLotes = lcChv And ;
	 	   a.NumLotes = b.NumLotes And ;
	 	   b.Cods = c.Cods ;
	  Into Cursor crInstrucao

	Select Emps, Datas, Contas, Bancos, Agencias, NContas, NCheques, Titulos, IClis, Borderos, ;
		   Space(20) as Cpfs, Space(50) as RClis, DtVencs, PrVencs, Valors, Moedas, Mensas ;
	  From crInstrucao ;
	  Into Cursor crClientes ReadWrite

	Select crClientes
	Scan
		ThisForm.poDatamgr.CursorQuery('SigCdCli', 'crSigCdCli', 'IClis', crClientes.IClis)
		If Used('crSigCdCli') And (Reccount('crSigCdCli') > 0)
			Replace Cpfs  With crSigCdCli.Cpfs, ;
					RClis With crSigCdCli.RClis In crClientes
		EndIf
	EndScan
	
	Select Emps, Datas, Space(50) as Envio, Contas, Bancos, Agencias, NContas, NCheques, ;
		   Titulos, IClis, Cpfs, DtVencs, PrVencs, Valors, Moedas, Borderos, Mensas ;
	  From crClientes ;
	  Into Cursor crEnvios ReadWrite
	  
	Select crEnvios
	Scan
		ThisForm.poDatamgr.CursorQuery('SigCdCli', 'crSigCdCli', 'IClis', crEnvios.Contas)
		If Used('crSigCdCli') And (Reccount('crSigCdCli') > 0)
			Replace Envio With crSigCdCli.RClis In crEnvios
		EndIf
	EndScan

	Select Nvl(b.RazSocs, Space(50)) as Remetes, a.Envio, a.Datas, a.Mensas, a.Bancos, ;
		   a.Agencias, a.NContas, a.NCheques, a.Titulos, a.Cpfs, a.DtVencs, a.PrVencs, ;
		   a.Valors, a.Moedas, a.Borderos, ThisForm.Tipos as Tipos, pLot as NumLotes ;
	  From crEnvios a Left Join crSigCdEmp b On a.Emps = b.Cemps ;
	  Into Cursor csImpressao

	Select csImpressao
	Go Top In csImpressao

	If .f.  && Não Retirar Estas Linhas Pois Elas Existem Para Anexar o Report ao Projeto
		Report Form SigOpLin Preview NoConsole
	EndIf 

	Do Form SigMvDoc With 'SigOpLin', 'Impressão do Lote de Instruções'
	
EndIf 
* Fim **************************************************************************************

ENDPROC

PROCEDURE validacheque
ThisForm.FocarBanco = .t.

SELECT crSigPrLii
GO bott
lnRec = Recno('crSigPrLii')

If (ThisForm.Tipos = 1) && Cheques
	Select Tipos ;
	  From crSigPrLii ;
	 Where Empty(Bancos) Or Empty(Agencias) Or Empty(NContas) Or Empty(NCheques) ;
	  Into Cursor crBrancos

	lcChq = crSigPrLii.Bancos + crSigPrLii.Agencias + crSigPrLii.NContas + crSigPrLii.NCheques

	Select Bancos, Agencias, NContas, NCheques ;
	  From crSigPrLii ;
	 Where Bancos + Agencias + NContas + NCheques = lcChq And ;
	 	   Recno() <> lnRec ;
	  Into Cursor crCheques

	If (RecCount('crCheques') > 0)
		=MessageBox('Cheque Já Digitado Nesta Instrução!!!', 0+48, '')
		With ThisForm.Pagina.Dados.cntGradeCheques
			.getBanco.Value   = Space(03)
			.getAgencia.Value = Space(04)
			.getConta.Value   = Space(10)
			.getCheque.Value  = Space(06)
			.getDtVencs.Value = Ctod('  /  /  ')
			.getPrVencs.Value = Ctod('  /  /  ')
			.getValor.Value   = 0.00
			.getMoeda.Value   = Space(03)
			.getBordero.Value = Space(10)
			.getIClis.Value   = Space(10)
			.getCpf.Value     = Space(20)
			.getDConta.Value  = Space(40)

			.GradeCheques.Column1.SetFocus
			.GradeCheques.Refresh
			.getBanco.SetFocus
		EndWith
	EndIf 
Else
	Select Tipos ;
	  From crSigPrLii ;
	 Where Empty(Titulos) ;
	  Into Cursor crBrancos

	lcChq = crSigPrLii.Titulos

	Select Titulos ;
	  From crSigPrLii ;
	 Where Titulos = lcChq And ;
	 	   Recno() <> lnRec ;
	  Into Cursor crCheques

	If (RecCount('crCheques') > 0)
		=MessageBox('Titulo Já Digitado Nesta Instrução!!!', 0+48, '')
		With ThisForm.Pagina.Dados.cntGradeTitulos
			.getTitulo.Value  = Space(10)
			.getDtVencs.Value = Ctod('  /  /  ')
			.getPrVencs.Value = Ctod('  /  /  ')
			.getValor.Value   = 0.00
			.getMoeda.Value   = Space(03)
			.getBordero.Value = Space(10)
			.getIClis.Value   = Space(10)
			.getCpf.Value     = Space(20)
			.getDConta.Value  = Space(40)

			.GradeTitulos.Column1.SetFocus
			.GradeTitulos.Refresh
			.getTitulos.SetFocus
		EndWith
	EndIf 
EndIf 

If ((Not Empty(crSigPrLii.Bancos) And Not Empty(crSigPrLii.Agencias) And ;
	 Not Empty(crSigPrLii.NContas) And Not Empty(crSigPrLii.NCheques)) Or ;
	 Not Empty(crSigPrLii.Titulos)) And ;
   ((Not Empty(crSigPrLii.DtVencs) And Not Empty(crSigPrLii.Valors) And ;
     Not Empty(crSigPrLii.Moedas) And Not Empty(crSigPrLii.IClis)) And ;
     (RecCount('crBrancos') < 1) And (RecCount('crCheques') < 1))

	Insert Into crSigPrLii (Tipos, NumLotes, Cods, cIdChaves) ;
					Values (crSigPrLit.Tipos, crSigPrLit.NumLotes, crSigPrLit.Cods, fUniqueIds())
					
	Go Bottom In crSigPrLii

		ThisForm.Pagina.Dados.cntGradeCheques.GradeCheques.Column1.SetFocus

	Replace Bancos   With Space(03), ;

			Titulos  With Space(10), ;

			Moedas   With Space(03), ;

	If (ThisForm.Tipos = 1) && Cheques
		ThisForm.Pagina.Dados.cntGradeCheques.getBanco.SetFocus
	Else
		ThisForm.Pagina.Dados.cntGradeTitulos.getTitulo.SetFocus
	EndIf 
	ThisForm.FocarBanco = .f.
Else
	ThisForm.FocarBanco = .f.
	Return 0
EndIf 

If (ThisForm.Tipos = 1) && Cheques
	With ThisForm.Pagina.Dados.cntGradeCheques
		.getBanco.Refresh
		.getAgencia.Refresh
		.getConta.Refresh
		.getCheque.Refresh

		.getPrVencs.Refresh
		.getValor.Refresh

		.getCpf.Refresh
		.getDConta.Refresh
	EndWith
EndIf

* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
Local lcCod, ldDat, lcCod, lnVal, llGravar, lcBuscar

llGravar = .f.
lcBuscar = ''
Select crSigPrLit
lcBuscar = crSigPrLit.NumLotes

If InList(Thisform.pcEscolha, 'INSERIR', 'ALTERAR')
	If Empty(ThisForm.Pagina.Dados.getInstrucao.Value)
		=MessageBox('Instrução Inválida!!!', 0+48, '')
		ThisForm.Pagina.Dados.getInstrucao.SetFocus
		Return 0
	EndIf

	Select crSigPrLii
	Scan
		If (ThisForm.Tipos = 1) && Cheques
			If (Empty(Bancos) Or Empty(Agencias)  Or Empty(NContas) Or Empty(NCheques) Or ;
				Empty(IClis)  Or IsEmpty(DtVencs) Or Empty(Valors)  Or Empty(Moedas)) ;
			    And ;
			   (Not Empty(Bancos) Or Not Empty(Agencias)  Or Not Empty(NContas) Or Not Empty(NCheques) Or ;
			    Not Empty(IClis)  Or Not IsEmpty(DtVencs) Or Not Empty(Valors)  Or Not Empty(Moedas))

				=MessageBox('Cheque Com Dados Inválidos!!! Corrija Antes de Gravar!!!', 0+48, '')
				ThisForm.Pagina.Dados.cntGradeCheques.getBanco.SetFocus
				Return 0
			EndIf 
		Else
			If (Empty(Titulos) Or  Empty(IClis) Or IsEmpty(DtVencs) Or Empty(Valors) Or Empty(Moedas)) ;
			    And ;
			   (Not Empty(Titulos) Or Not Empty(IClis) Or Not IsEmpty(DtVencs) Or Not Empty(Valors) Or Not Empty(Moedas))

				=MessageBox('Titulo Com Dados Inválidos!!! Corrija Antes de Gravar!!!', 0+48, '')
				ThisForm.Pagina.Dados.cntGradeTitulos.getTitulo.SetFocus
				Return 0
			EndIf 
		EndIf 

		If (Empty(Bancos) And Empty(Agencias) And Empty(NContas) And Empty(NCheques) And Empty(Titulos))
			Delete In crSigPrLii
		EndIf 

		Replace cIdChaves With fUniqueIds() In crSigPrLii
	EndScan

	ldDat = crSigPrLit.Datas
	lcCod = crSigPrLit.Cods
	lnVal = 0

	Select crSigPrLii
	Scan
		If ((Not Empty(Bancos) And Not Empty(Agencias) And Not Empty(NContas) And Not Empty(NCheques)) Or Not Empty(Titulos)) ;
	         And ;
	        (Not Empty(IClis) And Not IsEmpty(DtVencs) And Not Empty(Valors) And Not Empty(Moedas))

			Replace Cods With lcCod In crSigPrLii

			lnCot = fBuscarCotacao(crSigPrLii.Moedas, ldDat, Thisform.podatamgr ) / fBuscarCotacao(ThisForm.Moeda, ldDat , Thisform.podatamgr)
			lnVal = lnVal + (crSigPrLii.Valors * lnCot)
		EndIf
	EndScan

	Replace Valors With lnVal, ;
			Moedas With ThisForm.Moeda In crSigPrLit
EndIf

=DoDefault()

ThisForm.mAtivaPagina1
* Fim **************************************************************************************

Private lcBus, lnLot, lcChv

With ThisForm.Pagina.Lista.cntBuscaCheques
	If Empty(.getIClis.Value)
		lcBus   = .getBanco.Value + .getAgencia.Value + .getConta.Value + .getCheque.Value
		lcQuery = 'Select * From SigPrLii Where Bancos + Agencias + NContas + NCheques = ?lcBus'
	Else
		lcBus   = .getIClis.Value
		lcQuery = 'Select * From SigPrLii Where IClis = ?lcBus'
	EndIf

	ThisForm.poDataMgr.SqlExecute(lcQuery, 'crTmp')
	
	If Used('crTmp') And (Reccount('crTmp') > 0)
		Go Top In crTmp
		ThisForm.NumLotes = crTmp.NumLotes

		Select crTmp
		Scan
			lcChv = crTmp.NumLotes

			ThisForm.poDataMgr.SqlExecute('Select * From SigPrLit Where NumLotes = ?lcChv', 'crTmp2')
			If Used('crTmp2') And (Reccount('crTmp2') > 0)
				If Not Seek(lcChv, 'crProcurar', 'NumLotes')
					Select crTmp2
					Scatter MemVar Memo
				 	Insert Into crProcurar From MemVar
					ThisForm.NumLotes = crTmp2.NumLotes
				EndIf 

				If Not Seek(lcChv, 'crSigPrLit', 'NumLotes')
					Select crTmp2
					Scatter MemVar Memo
				 	Insert Into crSigPrLit From MemVar
					ThisForm.NumLotes = crTmp2.NumLotes
				EndIf 
			EndIf
	 		Use In crTmp2
		EndScan
		* Mostrar Uma Lista Com Todos os Cheques Encontrados
		If Not Empty(.getIClis.Value)
			Select crTmp
			Index On IClis + Str(NumLotes, 8) Tag IClis
			loBrowse = CreateObject('fwBuscaSel', 'crTmp', 'IClis', lcBus, 'Seleção')
			loBrowse.AddColuna('NumLotes', '99999999', 		 'Lote')
