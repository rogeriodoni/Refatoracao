* PROCEDURE impressao

PROCEDURE impressao
Local lcRep

lcRep = Alltrim(ThisForm.NomeRepo)

If (Type([ThisForm.pcRep])=[C]) And (ThisForm.pcRep = [SIGCDENT])
	Select * From crDados Where nFis Between ThisForm.GetNotaI.Value And ThisForm.GetNotaF.Value Into Cursor crDados1
	Select crDados
	Use
	Select * From crDados1 Into Cursor crDados
	Go Top
EndIf 

Report Form &lcRep. To Printer Prompt NoConsole
**********************************************************************
ENDPROC

------------------------------------------------------------
* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
LParameters pRep, pTit

Local lnOldW, lnBotW

ThisForm.pcRep = pRep

If (Type([pRep]) <> [C]) Or (pRep <> [SIGCDENT])
	ThisForm.label1.Visible   = .f.
	ThisForm.label2.Visible   = .f.
	ThisForm.Shape1.Visible   = .f.
	Thisform.GetNotaI.Visible = .f.
	Thisform.GetNotaF.Visible = .f.
	ThisForm.btnReport.Top = 4
Else
	ThisForm.label1.Visible   = .t.
	ThisForm.label2.Visible   = .t.
	ThisForm.Shape1.Visible   = .t.
	Thisform.GetNotaI.Visible = .t.
	Thisform.GetNotaF.Visible = .t.
	Go Top 
	Thisform.GetNotaI.Value = nFis
	Go Bott
	Thisform.GetNotaF.Value = nFis
EndIf

=DoDefault()

With Thisform
	lnOldW = .Width	
	lnBotW = .btnReport.Width
	If .Width > ( lnBotW + 20 )
	Else
		.btnReport.Left = 10
		.Width = lnBotW + 20
		.Left = .Left - Int( ( .Width - lnOldW ) / 2 )
	EndIf
EndWith	

If .f.
	Report Form SigMvDoc Preview NoConsole
EndIf

ThisForm.NomeRepo = Iif(Type('pRep')='C', pRep, 'SigMvDoc')
Thisform.Caption  = Iif(Type('pTit')='C', pTit, 'Emissão do Documento')
**********************************************************************
ENDPROC

* PROCEDURE processamento
------------------------------------------------------------
PROCEDURE processamento
Thisform.AlwaysOnTop = .f.

Go Top

=DoDefault()
**********************************************************************
ENDPROC

------------------------------------------------------------
* PROCEDURE btnReport.Click
------------------------------------------------------------
PROCEDURE btnReport.Click
If (This.Value <> 3) And (Type([ThisForm.pcRep])=[C]) And (ThisForm.pcRep = [SIGCDENT])
	If Empty(ThisForm.GetNotaI.Value)
		MessageBox('Nota Inicial Invalida !!!',16,'')
		Return .f.
	EndIf
	If Empty(ThisForm.GetNotaF.Value)
		MessageBox('Nota Final Invalida !!!',16,'')
		Return .f.
	EndIf
EndIf

=DoDefault()
**********************************************************************
ENDPROC
