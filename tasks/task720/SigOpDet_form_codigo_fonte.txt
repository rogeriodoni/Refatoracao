* PROCEDURE validacao
------------------------------------------------------------
PROCEDURE validacao
Select crSigMvFtc

If Used([CarsObrigs])
	Use in CarsObrigs
EndIf 

Select LocalFtc 
Select Distinct cCars, Desccars From LocalFtc Where chkobrig = 1 Into Cursor CarsObrigs ReadWrite 

lcMsg = []
lcCar = []
llOk  = .T.

Select CarsObrigs
Scan 
	Select crSigMvFtc
	Locate For Ccars = CarsObrigs.CCars
	If Not Found()
		lcCar = lcCar + Alltrim(CarsObrigs.Ccars) + Chr(13)+Chr(10)
	EndIf 
	
	If Not Empty(lcCar)
		lcMsg = [Existem Caracteristicas Obrigatórias e não Foram Informadas.] + Chr(13)+Chr(10)
		lcMsg = lcMsg + Chr(13)+Chr(10) + lcCar
		lcMsg = lcMsg + Chr(13)+Chr(10) + [Verifique.]
		MessageBox(lcMsg,0+64,[Atenção])
		llOk  = .F.
	EndIf 
	If !llOk
		Exit 
	EndIf 
	Select CarsObrigs
EndScan 

If llOk
	Delete From crSigMvFtc where Empty(CCars)
EndIf 
ENDPROC

* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
lParameters pCnx, pEscolha, pDopes, pProd, pEDN

ThisForm.cntSombra.lblSombra.Caption = ThisForm.Caption
ThisForm.cntSombra.lblTitulo.Caption = ThisForm.Caption

ThisForm.pcDopes   = pDopes
ThisForm.pcEscolha = pEscolha
ThisForm.poDataMgr = pCnx
ThisForm.pcEDN 	   = pEDN
ThisForm.pcPro 	   = pProd

=Thisform.Podatamgr.CursorQuery([SigCdPro],[crProduto],[Cpros],ThisForm.pcPro,[Mercs, Cpros])

If Used([crSigMvFtc])
	Zap in crSigMvFtc
EndIf 

If Reccount([crSigMvFtc])=0 or Not Used([crSigMvFtc])
	lcQuery = [Select * from SigMvFtc where EmpDopNums = ']+ThisForm.pcEDN+[' And Cpros = ']+crProduto.Cpros+['] 
	If (ThisForm.poDataMgr.SqlExecute(lcQuery, [crSigMvFtc]) < 1)
		=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (crSigMvFtc)])
		Return .f.
	EndIf
EndIf 

Select crSigMvFtc
Go Top 

*-- Carrega as informações do detalhe
lcQuery = [Select a.ccars, b.descrs as Desccars, a.cgrus as Dopes,a.ctits, b.chkobrig, b.cgrus, c.descrs, a.cselecaos, space(20) as Resps ] + ;
		[from SigCdFtc a ] + ;
		[join SigCrRap b on a.ccars = b.codigos ] + ;
		[join SigPrCpP c on a.ctits = c.codigos ] + ;
		[where a.ntipos = 2 ] + ;
		[And a.cgrus = ']+ThisForm.pcDopes+[' ] + ;
		[And b.cgrus = ']+crProduto.Mercs+[' ]

If (ThisForm.poDataMgr.SqlExecute(lcQuery, [LocalFtc]) < 1)
	=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (LocalFtc)])
	Return .f.
EndIf

Select LocalFtc
Index on cCars Tag cCars
Index on cTits Tag cTits
Go Top In LocalFtc

Select crSigMvFtc
Go Top 

This.Grade.RecordSource          = [crSigMvFtc]
This.Grade.Column1.ControlSource = [crSigMvFtc.ccars]
This.Grade.Column2.ControlSource = [crSigMvFtc.ctits]
This.Grade.Column3.ControlSource = [crSigMvFtc.Resps]

Select LocalFtc
Select Distinct cCars, Desccars From LocalFtc Where chkobrig = 1 Into Cursor CarsObrigs ReadWrite 

Select CarsObrigs
Scan 
	Select crSigMvFtc
	Locate for cCars = CarsObrigs.cCars
	If Not Found()
		Insert Into crSigMvFtc (EmpDopNums, Cpros, cIdChaves, cCars) Values (ThisForm.pcEdn, ThisForm.pcPro, fUniqueIds(), CarsObrigs.cCars)
	EndIf 	
	Select CarsObrigs
EndScan 

Select crSigMvFtc
Go Top 

This.Refresh
This.Grade.Column1.SetFocus

* Fim **************************************************************************************
ENDPROC

PROCEDURE Valid
Local loLista

If Not Empty(This.Value)
	Select LocalFtc
	Select Distinct cCars, Desccars From LocalFtc Into Cursor LocalCars ReadWrite 
	
	Select LocalCars
	Go Top In LocalCars
	Locate For cCars = This.Value
	
	If Eof([LocalCars])
		loLista = CreateObject([fwbuscaint], [LocalCars], [cCars], This.Value, [Caracteristicas], .t.)
		loLista.mAddColuna([cCars], 	[], [Código], 	  .t.)
		loLista.mAddColuna([Desccars],  [], [Descrição], .t.)
		loLista.Show()
	EndIf
	This.Value = LocalCars.cCars
Else
	This.Value = []	
EndIf

Replace CCars With This.Value in crSigMvFtc

This.Refresh
ENDPROC

Local loLista, lcCCars

	Select LocalFtc
	Select Distinct ctits, Descrs From LocalFtc Where CCars = lcCCars Into Cursor LocalTits ReadWrite 
	
	Select LocalTits
	Go Top In LocalTits
	Locate For ctits = This.Value
	
	If Eof([LocalTits])
		loLista = CreateObject([fwbuscaint], [LocalTits], [ctits], This.Value, [Complementos], .t.)
		loLista.mAddColuna([ctits],  [], [Código], 	  .t.)
		loLista.mAddColuna([Descrs], [], [Descrição], .t.)
		loLista.Show()
	EndIf
	This.Value = LocalTits.ctits
Else
	This.Value = []	
EndIf

Replace cTits With This.Value in crSigMvFtc

If Not Empty(This.Value)
	Replace Resps With [] in crSigMvFtc
EndIf 
This.Refresh
ENDPROC
