* PROCEDURE Init

PROCEDURE Init
parameters pAut, pUf, pCodUf

Private pUFs

pUFs = Replicate(Chr(254), 2)

lcQryCep = [Select * From Cep Where Estados = ?pUFs]


With ThisForm
	.poDataMgr = CreateObject([fSqlConector], [cep])
	.poDataOri = CreateObject([fSqlConector], .Name)

	If (.poDataMgr.pnIdconn > 0) And (.poDataOri.pnIdconn > 0)
		With .poDataMgr
			.AddCursor([Cep], [cIdChaves], [crSigCdCep], [], ThisForm.Pagina.Lista.Grade, lcQryCep)
			.pfSqlTabela(1).pColuna([Ceps], [], [], [CEP], 80, .t.)
			.pfSqlTabela(1).pColuna([TipoNomes], [], [], [Tipo], 80, .t.)
			.pfSqlTabela(1).pColuna([Nomes], [], [], [Endereço], 250, .t.)
			.pfSqlTabela(1).pColuna([Cidades], [], [], [Cidade], 150, .t.)
			.pfSqlTabela(1).pColuna([Estados], [], [], [UF], 24, .t.)
		EndWith

		ThisForm.pcTpSaida = '0'	&& mena 01/10/2016 - seta saida normal

		.CodUFs = 0
		
		IF pAut
			= DoDefault()
			thisform.WindowType = 1
			ThisForm.pcTpSaida = '1'	&& seta saida automatica
			ThisForm.pcEscolha = [INSERIR]
			thisform.pagina.lista.getEstados.Value = pUf 
			thisform.codUfs = pCodUf
			thisform.pagina.lista.grupo_Op.Click(1)
		else		
			=DoDefault()
			.Pagina.Lista.Grade.Column1.SetFocus
		ENDIF
	Else
		=Messagebox([Impossível Efetuar Conexão Com o Servidor de Banco de Dados...], 0+48, [CreateObject])
		Return .f.
	EndIf
ENDWITH




* Fim **************************************************************************************
ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Lista.Grupo_op.Click
------------------------------------------------------------
PROCEDURE Pagina.Lista.Grupo_op.Click
LParameters Opcao

Private pUFs
If Not DoDefault(Opcao)
	ThisForm.mAtivaPagina1
	Return .f.
EndIf

* Checando se Preencheu o Grupo de UF
If Empty(ThisForm.Pagina.Lista.getEstados.Value)
	=MessageBox([A UF Não Foi Preenchida!!!], 0+48, [Atenção!!!])
	ThisForm.mAtivaPagina1
	Return .f.
Else
	ThisForm.Pagina.Dados.getEstados.Value = ThisForm.Pagina.Lista.getEstados.Value
EndIf

If (ThisForm.pcEscolha = [INSERIR])
	Replace cIdChaves With fUniqueIds() In crSigCdCep
EndIf

Do Case
	Case Inlist(ThisForm.pcEscolha, [INSERIR], [PROCURAR])
		ThisForm.Pagina.Dados.getCEPS.SetFocus 

	Case (ThisForm.pcEscolha = [ALTERAR])
		ThisForm.Pagina.Dados.getNomes.SetFocus 

	Otherwise
		ThisForm.Pagina.Dados.Grupo_Salva.Salva.SetFocus
EndCase
* Fim **************************************************************************************
ENDPROC

* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
Local lcCEP

lcCEP = ThisForm.Pagina.Dados.getCEPS.Value 

If InList(ThisForm.pcEscolha, [INSERIR], [ALTERAR])
	If Empty(crSigCdCep.Ceps)
		=Messagebox([O CEP Não Pode Ficar Em Branco!!!], 48, [Erro!!!]) 
		ThisForm.Pagina.Dados.getCeps.SetFocus
		Return 0
	EndIf

	If Empty(crSigCdCep.TipoNomes)
		=Messagebox([O Tipo de Endereço Não Pode Ficar Em Branco!!!], 48, [Erro!!!]) 
		ThisForm.Pagina.Dados.getTipoNomes.SetFocus
		Return 0
	EndIf

	If Empty(crSigCdCep.Nomes)
		=Messagebox([O Endereço Não Pode Ficar Em Branco!!!], 48, [Erro!!!]) 
		ThisForm.Pagina.Dados.getNomes.SetFocus
		Return 0
	EndIf

	If Empty(crSigCdCep.Cidades)
		=Messagebox([A Cidade Não Pode Ficar Em Branco!!!], 48, [Erro!!!]) 
		ThisForm.Pagina.Dados.getCidades.SetFocus
		Return 0
	EndIf

	If Empty(crSigCdCep.Estados)
		=Messagebox([A UF Não Pode Ficar Em Branco!!!], 48, [Erro!!!]) 
		ThisForm.Pagina.Dados.getEstados.SetFocus
		Return 0
	Endif
EndIf

=DoDefault()

If InList(ThisForm.pcEscolha, [INSERIR], [ALTERAR])
	If Empty(crSigCdCep.cbarceps)
		=fBarCepSedex(crSigCdCep.Ceps)	
	Endif
Endif

*** mena 01/10/2016 - verifica se saida automatica ou normal  ***
If ThisForm.pcTpSaida == '0'	&& saida normal
	ThisForm.mAtivaPagina1()
Else 
	This.Parent.Parent.Parent.Lista.Grupo_Saida.Sair.Click && fecha o formulario...
*!*	thisform.Release()
EndIf 

*!*ThisForm.mAtivaPagina1
* Fim **************************************************************************************
ENDPROC

PROCEDURE LostFocus
Local loLista

If Not Empty(This.Value)
	loLista = CreateObject([fwBuscaExt], ThisForm.poDataOri.pnIdConn, [SigCdUfs], ;
						   [crListaRemota], [Estados], This.Value, [Seleção], .t.)

	If Not loLista.plAchouRegistro
		loLista.mAddColuna([Estados], [], [UF])
		loLista.mAddColuna([Descrs],  [], [Descrição])
		loLista.mAddColuna([UFIBGES], [], [IBGE])
		loLista.Show()
	EndIf
	
	ThisForm.Pagina.Lista.getEstados.Value = Iif(Lastkey()=27, [], crListaRemota.Estados)

	ThisForm.CodUFs = Iif(Lastkey()=27, 0, crListaRemota.UFIBGES)
	Use In crListaRemota
EndIf
ThisForm.Pagina.Lista.getEstados.Refresh

If (Padr(This.Tag, 2) <> Padr(This.Value, 2))
	ThisForm.mAtivaPagina1
EndIf
* Fim **************************************************************************************
ENDPROC
