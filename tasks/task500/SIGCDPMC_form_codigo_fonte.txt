PROCEDURE Init
LPARAMETERS lcacesso

		.pfSqlTabela(1).pColuna('valmin','','','Valor minimo',75,.t.)
		.pfSqlTabela(1).pColuna('dtini','','','Dt Inicio',75,.t.)
		.pfSqlTabela(1).pColuna('dtfin','','','Dt final',75,.t.)
		.pfSqlTabela(1).pColuna('usualts','','','Usuário',75,.t.)
		.pfSqlTabela(1).pColuna('dtalts','','','Alteracao',75,.t.)
	EndWith
	= DoDefault()
	.LockScreen = .f.

* PROCEDURE Pagina.Lista.Grupo_op.Click
------------------------------------------------------------
PROCEDURE Pagina.Lista.Grupo_op.Click
LParameters opcao

If !dodefault( Opcao )
	ThisForm.mAtivapagina1
	Return
Endif

Do Case
	Case Inlist( ThisForm.pcEscolha, 'INSERIR', 'PROCURAR' )
		ThisForm.Pagina.Dados.Getcpros.SetFocus
	Case ThisForm.pcEscolha = 'ALTERAR'
		ThisForm.Pagina.Dados.Getcpros.SetFocus
	Case Inlist( ThisForm.pcEscolha, 'EXCLUIR', 'CONSULTAR' )
		ThisForm.Pagina.Dados.Grupo_Salva.Salva.SetFocus
Endcase

ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
LOCAL lcQuery

If INLIST(ThisForm.pcEscolha,'INSERIR')
	If Empty( CrSigCdPmc.cpros )
		Messagebox( 'Produto não pode ficar em branco!', 48, '' ) 
		ThisForm.Pagina.Dados.Getcpros.SetFocus
		Return
	Endif
	If Empty( CrSigCdPmc.dtini )
		Messagebox( 'Data Inicial não pode ficar em branco!', 48, '' ) 
		ThisForm.Pagina.Dados.Getdtini.SetFocus
		Return
	Endif
	If Empty( CrSigCdPmc.dtFin )
		Messagebox( 'Data final não pode ficar em branco!', 48, '' ) 
		ThisForm.Pagina.Dados.Getdtfin.SetFocus
		Return
	Endif
	If CrSigCdPmc.dtini >  CrSigCdPmc.dtfin
		Messagebox( 'As datas estão incompatíveis!', 48, '' ) 
		ThisForm.Pagina.Dados.Getdtini.SetFocus
		Return
	Endif

	lcDat=SUBSTR(DTOS(TTOd(CrSigCdPmc.dtini)),1,10)
	lcDat=SUBSTR(lcDat,1,4)+'-'+SUBSTR(lcDat,5,2)+'-'+SUBSTR(lcDat,7,2)
	lcQuery = 	[Select * From SigCdPmc where cpros = ']+CrSigCdPmc.cpros+[' and ']+lcDat+[' ]+;
				[between LEFT(CONVERT(VARCHAR, dtini, 120), 10) and  LEFT(CONVERT(VARCHAR, dtfin, 120),10)]	 && 2017-01-21
	If ThisForm.Podatamgr.Sqlexecute(lcQuery,'crAux1') < 1
		Messagebox('Impossivel Efetuar Conexão com Servidor de Banco de Dados.',(0+48),'Erro')
		Return
	Endif

	lcDat=SUBSTR(DTOS(TTOd(CrSigCdPmc.dtfin)),1,10)
	lcDat=SUBSTR(lcDat,1,4)+'-'+SUBSTR(lcDat,5,2)+'-'+SUBSTR(lcDat,7,2)
	lcQuery = 	[Select * From SigCdPmc where cpros = ']+CrSigCdPmc.cpros+[' and ']+lcDat+[' ]+;
				[between LEFT(CONVERT(VARCHAR, dtini, 120), 10) and  LEFT(CONVERT(VARCHAR, dtfin, 120),10)]	 && 2017-01-21
	If ThisForm.Podatamgr.Sqlexecute(lcQuery,'crAux2') < 1
		Messagebox('Impossivel Efetuar Conexão com Servidor de Banco de Dados.',(0+48),'Erro')
		Return
	Endif

	If RECCOUNT('crAux1')>0 or RECCOUNT('crAux2') >0
		Messagebox( 'Esse intervado já está cadastrado para esse produto!', 48, '' ) 
		ThisForm.Pagina.Dados.Getdtini.SetFocus
		Return
	Endif

	lcDat=SUBSTR(ALLTRIM(STR(YEAR(DATE()))),3)+PADL(ALLTRIM(STR(month(DATE()))),2,'0')
	replace codigo with 'PMC'+lcDat+PADL(ALLTRIM(STR(fGerUniqueKey('PMC'+lcDat))),3,'0') in crSigCdPmc

Endif

replace dtalts with DATETIME() in crSigCdPmc
replace usualts with Usuar in crSigCdPmc

=DoDefault()
ThisForm.mAtivapagina1

ENDPROC
