PROCEDURE Init
LPARAMETERS lcacesso
LOCAL lcCod

With ThisForm
	.poDatamgr = CreateObject('fSqlConector',ThisForm.Name)
	With .poDatamgr
		.AddCursor('SigCdClf','codigos','crSigCdClf','',ThisForm.pagina.lista.grade)
		.pfSqlTabela(1).pColuna('Codigos','','','Codigo',94,.t.)
		.pfSqlTabela(1).pColuna('descricaos','','','Descrição',290,.t.)
		.pfSqlTabela(1).pColuna('aipis','','','Alíq. IPI',45,.t.)

		.pfSqlTabela(1).pColuna('cest','','','CEST',70,.t.)
		.AddCursor('SigCdIva','cidChaves','crSigCdIva','','')
		.AddCursor('SigCdRed','cidChaves','crSigCdRed','','')
	EndWith
	= DoDefault()
	.LockScreen = .f.
ENDWITH

lcCod = crSigCdClf.codigos
SELECT * from crSigcdIva where codigos = lcCod into cursor crIva readwrite
SELECT crIva

SELECT crRed

	.Column4.controlsource = 'crIva.indivastpi'

	.column1.ControlSource = 'crRed.Estas'
	.column2.ControlSource = 'crRed.Areds'

EndWith

	Return
Endif

	Select crRed

			estas    with tmpRed.estas, ;
			cidChaves with tmpRed.cidchaves, ;
			aReds with TmpRed.aReds, ;
			bReds with TmpRed.bReds

	Append Blank

	case thisform.pcEscolha='INSERIR' or ;
		thisform.pcEscolha='PROCURAR'
		thisform.Pagina.Dados.getcodigo.setfocus
	case thisform.pcEscolha='ALTERAR'
		lcSql = [Select Count(*) as QtProd From SigCdPro Where ClFiscals = ']+CrSigCdClf.codigos+[' And ]+;

		Endif
		Thisform.pagina.dados.getdescricao.setfocus

		thisform.pcEscolha='EXCLUIR'
		thisform.pagina.dados.grupo_salva.salva.setfocus
endcase

* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
local lnRecno,llRetorno,llOks

if thisform.pcEscolha $ 'INSERIR ALTERAR'
   
   with thisform.Pagina.Dados
      if IsEmpty(.getCodigo.value)
         = messagebox('Código em branco !!!',0+64,goSistema.NomeSistema)
         .getCodigo.setfocus
		 return
      else
		If ThisForm.poDataMgr.ChkRegister('SigCdClf','codigos',ThisForm.Pagina.Dados.GetCodigo.Value) and thisform.pcEscolha = 'INSERIR'

	         = messagebox('Código já cadastrado !!!',0+64,goSistema.NomeSistema)
	         .getCodigo.setfocus
	         return
             
         endif
         
      endif    
         
	endwith
    
	m.Ipis     = padr(upper(alltrim(thisform.Pagina.Dados.cmbIpi.value)),1)
	m.basereds = padr(upper(alltrim(thisform.Pagina.Dados.cmbBred.value)),1)
	m.aliqreds = padr(upper(alltrim(thisform.Pagina.Dados.cmbAred.value)),1)
	m.IpiProds = padr(upper(alltrim(thisform.Pagina.Dados.CmbIpip.value)),1)
	
	select CrSigCdClf
	gather memvar fields basereds,Aliqreds,ipis,IpiProds

	**** Mena - 28/11/2013 salva IVAs ****
	SET delete off
	If InList(ThisForm.pcescolha,'INSERIR','ALTERAR')
		Select CrSigCdIva
		ZAP
		Select crIVA
		GO top
		Scan
			If Not Empty(estasO) and Not Empty(estasD)
				Select CrSigCdIva
				Append Blank
				replace codigos with CrSigcdclf.codigos,;
						estasO with crIva.estasO,;
						estasD with crIva.estasD,;
						indivast with crIva.indivast,;
						indivastPi with crIva.indivastPi,;
						cidchaves with  IIF(EMPTY(crIva.cidChaves), fUniqueIds(), crIva.cidChaves)
				IF DELETED('crIva')
					DELETE 
				ENDIF
			ENDIF
			SELECT crIva
		EndScan
	ENDIF
	SET delete on
	********************************************************
	**** Mena - 07/05/2017 salva Reduções ****
	SET delete off
	If InList(ThisForm.pcescolha,'INSERIR','ALTERAR')
		Select CrSigCdRed
		ZAP
		Select crRed
		GO top
		Scan
			If Not Empty(estas)
				Select CrSigCdRed
				Append Blank
				replace codigos with CrSigcdclf.codigos,;
						estas with crRed.estas,;
						aReds with crRed.aReds,;
						bReds with crRed.bReds,;
						cidchaves with  IIF(EMPTY(crRed.cidChaves), fUniqueIds(), crRed.cidChaves)
				IF DELETED('crRed')
					DELETE 
				ENDIF
			ENDIF
			SELECT crRed
		EndScan
	ENDIF
	SET delete on
	********************************************************
endif

llOks = .T.
If Not ThisForm.poDataMgr.Update('crSigCdIva')
	=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (Update - crSigCdIva)')
	llOks = .f.
EndIf
If Not ThisForm.poDataMgr.Update('crSigCdRed')
	=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (Update - crSigCdRed)')
	llOks = .f.
EndIf
If !llOks
	ThisForm.poDataMgr.Rollback()
	Return .f.
EndIf

If ThisForm.pcEscolha = [EXCLUIR]
	lcSQL = [Select top 1 cpros from SigCdPro where clfiscals = ']+ThisForm.Pagina.Dados.getCodigo.Value+[' ] + ;
				[UNION ALL ] + ;
			[Select top 1 cpros from SigCdPrc where clfiscals = ']+ThisForm.Pagina.Dados.getCodigo.Value+[']	
	ThisForm.poDataMgr.SQLExecute(lcSQL,[crChecaPro])	
	If Used([crChecaPro]) and not Eof([crChecaPro])
		MessageBox('Não É Possivel Excluir, Classificação Fiscal Já Informada em Produtos.', 0 + 16, '')
		Return 0
	Endif		
Endif

select CrSigCdClf
=DoDefault()
ThisForm.mAtivapagina1

ENDPROC
