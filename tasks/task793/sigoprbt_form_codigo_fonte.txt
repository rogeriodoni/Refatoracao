lcQuery = [SELECT B.DATAS,B.EMPS,CONVERT(CHAR(2),B.DIA) AS DIA,CONVERT(CHAR(2),B.MES) AS MES,A.COLECOES,B.VENDA,A.DESCONTO ] + ;
	[FROM ( ] + ;
	[SELECT A.EMPDOPNUMS,B.COLECOES,SUM(VDES) AS 'DESCONTO' FROM SIGMVITD A ] + ;
	[LEFT OUTER JOIN SIGCDPRO B ON A.CPROS=B.CPROS ] + ;
	[WHERE DDESC IN ('DESCONTO ELO/OUROCARD','DESCONTO ELO') ] + ;
	[GROUP BY A.EMPDOPNUMS,B.COLECOES ) A ] + ;
	[INNER JOIN ( ] + ;
	[SELECT A.EMPS,A.DATAS,REPLICATE('0',2-LEN(DAY(A.DATAS))) +CONVERT(VARCHAR(2),DAY(A.DATAS)) AS DIA, ] + ;
	[REPLICATE('0',2-LEN(MONTH(A.DATAS))) +CONVERT(VARCHAR(2),MONTH(A.DATAS)) AS MES, ] + ;
	[A.EMPDOPNUMS,D.COLECOES,SUM(C.TOTAS) AS 'VENDA' FROM SIGMVCAB A ] + ;
	[INNER JOIN SIGCDOPE B ON A.DOPES=B.DOPES ] + ;
	[INNER JOIN SIGMVITN C ON A.EMPDOPNUMS=C.EMPDOPNUMS ] + ;
	[LEFT OUTER JOIN SIGCDPRO D ON C.CPROS=D.CPROS ] + ;
	[LEFT OUTER JOIN SIGMVITD E ON C.EMPDOPNUMS=E.EMPDOPNUMS AND C.CITENS=E.CITENS AND E.CPROS=D.CPROS ] + ;
	[WHERE B.TIPOOPS=4 AND A.DOPES NOT IN ('ENTREGA CONSERTO CLI') ] + ;
	[GROUP BY A.DATAS,A.EMPS,DAY(A.DATAS),MONTH(A.DATAS),A.EMPDOPNUMS,D.COLECOES)  B ON A.EMPDOPNUMS=B.EMPDOPNUMS ] + ;
	[AND A.COLECOES=B.COLECOES ] + ;
	[AND LTRIM(RTRIM(B.EMPS))+LTRIM(RTRIM(B.DIA))+LTRIM(RTRIM(B.MES)) NOT IN(SELECT LTRIM(RTRIM(EMPS))+LTRIM(RTRIM(DIA))+LTRIM(RTRIM(MES)) FROM SIGOPRBT) ] + ;
	[ORDER BY 4,3,5,2]

If Thisform.poDataMgr.SqlExecute(lcQuery,[crSigMvRbt]) < 1
	Return .F.
Endif

	EndIf

		Select CrTmpObs
		Go Top
		If (Reccount() > 0) And ;
				Not ([#]  $ CrTmpObs.Observas) And ;

				Not ([!]  $ CrTmpObs.Observas) And ;
				Not ([^$] $ CrTmpObs.Observas) And ;

					Moedas,DPros,CUnis,CidChaves,DtAlts,EmpDopNums,Pesos,codbarras,Obs) ;
					Values (crSigCdPro.cPros,lcDopes,lcEmps,_Inicio,lcOpers,1,lnItem,lnValor,lnValor2,0,(lnDesc * (-1)),;

			Endfor

		Endif

		*lnItem = lnItem + 1
		Select csItnRbt
	Endscan


	If Inlist(crSigCdOpe.DtEntrs, 1, 4, 6)
		If (crSigCdOpe.Situas > 1) && Adicionar na Entrega: 1-Nenhum; 2-Cor; 3-Componente; 4-Ambos
			lnSum = 0
			If Inlist(crSigCdOpe.Situas, 2, 4)
				If (Thisform.poDataMgr.SqlExecute([Select Cods, DiasEnts From SigCdCor], [LocalCor]) < 1)
					If Not Thisform.Automatico
						=Messagebox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (LocalCor)])
					Endif
					Return 0
				Endif

				Select Max(b.DiasEnts) As Dias ;
					From crSigMvIts a, LocalCor b ;
					Where a.CodCors = b.Cods ;
					Into Cursor LocalDias

				Select LocalDias
				Go Top In LocalDias
				If Not Eof([LocalDias]) And (LocalDias.Dias > 0)
					lnSum = lnSum + LocalDias.Dias
				Endif
				Use In LocalCor
				Use In LocalDias
			Endif

			If Inlist(crSigCdOpe.Situas, 3, 4)
				lnMax = 0
				If (lnMax > 0)
					lnSum = lnSum + lnMax
				Endif
			Endif

			*** mena - 07/11/2016 - busca qtde de dias para calculo de prazo de entrega de serviço - montecarlo
			If Inlist(crSigCdOpe.Situas, 5)
				lnMax = 0
				Select crSigMvIts
				Scan
					lcQuery = [Select b.DiasPrz From SigCdPro a, SigCdprz b where a.cpros = ']+crSigMvIts.cPros+['] +;
						[ and a.CGrus = b.codigos And b.Emps =']+crSigcdemp.Cemps+[']

					If (Thisform.poDataMgr.SqlExecute(lcQuery, [crPrz]) < 1)
						If Not Thisform.Automatico
							=Messagebox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (crPrz)])
						Endif
						Return 0
					Endif

					Select crPrz
					Go Top In crPrz
					If Not Eof([crPrz]) And (crPrz.DiasPrz > lnMax)
						lnMax = crPrz.DiasPrz
					Endif
					Use In crPrz
				Endscan

				If (lnMax > 0)
					lnSum = lnMax
				Endif
			Endif
			**********************************************************

			If (lnSum > 0)
				If Not Inlist(crSigOpCdd.chkUteis, 0, 4) && Dias Úteis
					llSab = Inlist(crSigOpCdd.chkUteis, 1)
					llDom = Inlist(crSigOpCdd.chkUteis, 1, 3)
					llFer = Inlist(crSigOpCdd.chkUteis, 1, 2, 3)

					ldDI1 = Ttod(crSigMvCab.DataS)
					ldDI2 = Ttod(crSigMvCab.DataS) + lnSum
					Do While (ldDI1 <= ldDI2)
						If fChkFeriado(Thisform.poDataMgr, ldDI1, llSab, llDom)
							lnSum = lnSum + 1
						Endif
						ldDI1 = ldDI1 + 1
						ldDI2 = Ttod(crSigMvCab.DataS) + lnSum
					Enddo
				Endif

				ldEnt = Ttod(crSigMvCab.DataS) + lnSum

				If (Ttod(crSigMvCab.PrazoEnts) <> ldEnt)
					Replace PrazoEnts With ldEnt In crSigMvCab
				Endif
			Endif
		Endif
	Endif

	*************************************************************************************************************************************************************
	*--- Lança as informações da Parcela
	*************************************************************************************************************************************************************
	_Noper = fGerUniqueKey('SigMvCcr')
	lcMoe = crSigCdPam.Moedaps

	Select crSigMvPar
	Index On EmpDopNums Tag EmpDopNums


	Store [] To lcBandeira, lcAdquirente, lcPagto, lcFilEmp

	lnPar   = 1
	ldDatas = crSigMvCab.DataS
	lnValor = crSigMvCab.Valos
	ldDtV   = ldDatas

	lcPagto = crSigCdOpe.FpagOpCrs
	lcFpags = crSigCdOpe.cMoes

	lcQuery = [Select * from SigOpFp Where FPags = ']+lcPagto+[']
	If Thisform.poDataMgr.SqlExecute (lcQuery,[LocalFPag]) < 1
		If Not Thisform.Automatico
			=Messagebox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (CursorQuery - LocalFPag)])
		Endif
		Return .F.
	Endif


	Thisform.ContaPar = 0

	totpar	= 0
	totdsc  = 0
	nump 	= 1
	nparc   = lnPar

	Select LocalFPag
	If Not Eof([LocalFPag])

		lcMdf = Iif(Empty(LocalFPag.MoeFpgs), lcMoe, LocalFPag.MoeFpgs)
		lcPagto = LocalFPag.FPags
		ldDat   = ldDatas

		ldDtV = Iif(Type([ldDtV])=[T],Ttod(ldDtV),ldDtV)
		ldDtV = ldDtV + LocalFPag.Dias

		For i = 1 To lnPar

			valpar = Round(lnValor / lnPar,2)

			totpar = totpar + valpar

			Insert Into crSigMvPar (Emps, Dopes, Numes, EmpDopNums, DataS, CidChaves, CotFPgs, FPags, MoeFpgs, NOpers, Valos, Vencs, Parcs, Pagos, DtAlts, Nsu, Autoriza) ;
				Values (lcEmps, lcDopes, _Inicio, _EDN, ldDatas, fUniqueIds(), 1, lcPagto, lcMdf, _Noper, valpar, ldDtV, i, [1], Datetime(), _Trans,[.])

			ldDtV = ldDtV + LocalFPag.Dias

			If nparc = nump
				If totpar # ( lnValor )
					_dIf = ( lnValor -  totpar )
					Replace crSigMvPar.Valos With crSigMvPar.Valos + _dIf
				Endif
			Endif

			*************************************************************************************************************************************************************
			*--- Lança as informações da Conta Corrente
			*************************************************************************************************************************************************************
			Thisform.ContaPar = Thisform.ContaPar + 1

			If ( crSigCdOpe.COpers = 1 Or crSigCdOpe.ctipos = 2 ) And Inlist(crSigCdOpe.ParContas,1,3)
				=Thisform.LancaContaCorrente()
			Endif

			nparc = nparc - 1

		Next
	Endif

Endscan

With Thisform.poDataMgr
	If lnErro1>0
		lnErro1 = Iif(.Update('crSigMvCab'), 1, -1)
	Endif
	If lnErro1>0
		lnErro1 = Iif(.Update('crSigMvItn'), 1, -1)
	Endif
	If lnErro1>0
		lnErro1 = Iif(.Update('crSigMvIts'), 1, -1)
	Endif
	If lnErro1>0
		lnErro1 = Iif(.Update('crSigMvPar'), 1, -1)
	Endif
	If lnErro1>0
		lnErro1 = Iif(.Update('crSigMvCCr'), 1, -1)
	Endif
	If lnErro1>0
		lnErro1 = Iif(.Update('crSigMvHst'), 1, -1)
	Endif
	If lnErro1>0
		lnErro1 = Iif(.Update('CrSigOpCoc'), 1, -1)
	Endif
	If lnErro1>0
		lnErro1 = Iif(.Update('crSigChe'), 1, -1)
	Endif
	If lnErro1>0
		lnErro1 = Iif(.Update('crSigChLcq'), 1, -1)
	Endif
	If lnErro1>0
		lnErro1 = Iif(.Update('crSigCqChm'), 1, -1)
	EndIf
	If lnErro1>0
		lnErro1 = Iif(.Update('crSigOpRbt'), 1, -1)
	Endif
	If lnErro1>0
		lnErro1 = .Commit()
	Endif
	If Not lnErro1>0
		= .Rollback()
	Endif
Endwith

Select(lnOldSel)
Return .T.

ENDPROC

			_ContaOrig = crSigMvCab.Contaos
		EndIf
		If (crSigOpCdd.LanFinMats = 2) And !Empty(_ContaOrig) And ThisForm.poDataMgr.CursorQuery('SigCdCli','CrtmpCli','Iclis',_ContaOrig, [grupomats, contamats])
			If !IsEmpty( crtmpCli.GrupoMats ) And !IsEmpty( crtmpCli.ContaMats )
				_GrupoOrig = CrTmpCli.GrupoMats
				_ContaOrig = CrTmpCli.ContaMats
			EndIf
		EndIf
	EndIf
	
	If Not IsEmpty(_ContaOrig) And ThisForm.poDataMgr.CursorQuery('SigCdCli','CrtmpCli','Iclis',_ContaOrig, [rclis])
		lcRclis = crtmpCli.Rclis
	EndIf

	Select crSigMvCab
	_Hist = ThisForm.MontaHists(_Titulo, crSigOpFp.FPags, lcRClis)

	_noper = fGerUniqueKey( 'SigMvCcr' )
	If llVOper
		_voper = crSigMvCab.Emps + Alltrim( Str( lnRetVOper, 9 ) )
	Else
		_voper = crSigMvCab.Emps + _TIPOE + Alltrim( str( _noper, 9 ) )
	EndIf

	lnValor = Abs( crSigMvPar.Valos )
	If crSigOpFp.Trocos = 1
		xsoper = iif(crSigCdOpe.copers=1,'C','D')
		xoper = iif(crSigCdOpe.copers=1,'D','C')
	Else
		If crSigOpFp.OriDes = 1 And crSigOpFp.ValPends = 1
			xoper = iif(crSigCdOpe.copers=2,'C','D')
			xsoper = iif(crSigCdOpe.copers=2,'D','C')
		Else
			xoper = iif(crSigCdOpe.copers=1,'C','D')
			xsoper = iif(crSigCdOpe.copers=1,'D','C')
		EndIf
	EndIf

	_Hist2 = Alltrim( crSigCdOpe.Hist2s )
	
	DO Case
		Case crSigCdOpe.tpHist2s = 1
			_hist2 = _hist2 + ' ' + _Titulo

		Case crSigCdOpe.tpHist2s = 2
			If crSigOpFp.Infos = [C]
				_Hist2 = _Hist2 + ' Ch:'+crSigMvPar.Bancos+'-'+crSigMvPar.Agencias+'-'+crSigMvPar.Contas+'-'+crSigMvPar.numeros
			Else
				If crSigOpFp.Infos = [T]
					_Hist2 = _Hist2 + ' Ct:'+crSigMvPar.cnnsuparcs
				EndIf
			EndIf

		Case crSigCdOpe.tpHist2s = 3
			_hist2 = _hist2 + ' ' + Alltrim(CrTmpCli.Rclis)

		Case crSigCdOpe.tpHist2s = 4
			_hist2 = _hist2 + ' ' + Alltrim(_ContaOrig)

		Case crSigCdOpe.TpHist2s = 5
			_hist2 = _hist2 + ' ' + Iif(!IsNull(crSigMvCab.Obses),Alltrim(crSigMvCab.Obses),'')
	EndCase

	If (crSigOpFp.TpHists = 4)
		Do Case
			Case (crSigOpFp.Infos = [C]) && Cheque
				_Hist = crSigMvCab.Notas + [ CHQ: ] + crSigMvPar.Bancos + [/] + crSigMvPar.Agencias + [/] + ;
						crSigMvPar.Contas + [/] + crSigMvPar.Numeros
				_Hist2 = [CPF: ] + crSigMvPar.Cpfs		

			Case (crSigOpFp.Infos = [D]) && Duplicata
				_Hist = crSigMvCab.Notas + [ (] + Alltrim(Str(crSigMvPar.Parcs, 2)) + [/] + ;
						Alltrim(Str(crSigMvPar.TParcs, 2)) + [)]
				_Hist2 = lcRclis
						
			Case (crSigOpFp.Infos = [O]) && Outros
				_Hist = crSigMvCab.Notas
				_Hist2 = lcRclis
						
			Case (crSigOpFp.Infos = [R]) && Conta Corrente
				_Hist = crSigMvCab.Notas
				_Hist2 = lcRclis
						
			Case (crSigOpFp.Infos = [T]) && Cartão
				_Hist = crSigMvCab.Notas + [ - ] + Alltrim(crSigMvPar.Outros) + [ - (] + Alltrim(Str(crSigMvPar.Parcs, 2)) + [/] + ;
						Alltrim(Str(crSigMvPar.TParcs, 2)) + [)]
				_Hist2 = lcRclis

			Otherwise
				_Hist = crSigMvCab.Notas
				_Hist2 = lcRclis
		EndCase 
	EndIf
	
	_CidChave = Dtos(ldDtContab)+Transform(fGerUniqueKey(Dtos(ldDtContab)),"@L 999999")+ThisForm.Sigkey 

	lcMd1 = Iif(Empty(crSigOpFp.MoeFpgs), crSigCdOpe.CMoes, crSigOpFp.MoeFpgs)

	Insert Into crSigMvCcr ( NOpers, VOpers, Docus, Emps, Datas, usualts, datalts, Vencs, hists,;
			opers, sopers, moedas, cotacaos, valors, autos, grupos, contas, ;
			sgrupos, scontas, smoedas, scotacaos, svalors, Dopes, numes, valpags, Hist2s, CidChaves, EmpDopNums, Jobs ) ;
		Values ( _NOper, _VOper, _Doc, crSigMvCab.emps, ldDtContab, usuar, _pData, crSigMvPar.Vencs, ;
				_hist, xoper, xsoper, lcMd1, 1, lnValor, .t., ;
				_GrupoOrig, _ContaOrig, _GrupoDest, _ContaDest, ;
				lcMd1, 1, lnValor, ;
				crSigMvCab.Dopes, crSigMvCab.numes, crSigMvPar.VPags, _Hist2, _CidChave, ;
				crSigMvCab.Emps + crSigMvCab.Dopes + Str(crSigMvCab.Numes,6), crSigMvCab.Jobs )
	Replace crSigMvCcr.gruconmoes with crSigMvCcr.grupos + crSigMvCcr.contas + crSigMvCcr.moedas,;
			crSigMvCcr.EmpDopncs  With crSigMvCcr.Emps + crSigMvCcr.Dopcs + Str(crSigMvCcr.Numcs,6),;
			crSigMvCcr.Pagos      With Iif(crSigMvCcr.Valors=crSigMvCcr.valpags,'0','1') ,;
			crSigMvCcr.DopoTps    with crSigMvCcr.Dopcs + crSigMvCcr.Opers + crSigMvCcr.Tipos + crSigMvCcr.Pagos in crSigMvCcr
			
	=fRecalculaS( crSigMvCcr.Grupos, crSigMvCcr.Contas, crSigMvCcr.Datas, crSigMvCcr.Moedas, ThisForm.poDataMgr  )

	ThisForm.FazContra

	Select crSigMvPar

	If Not InList(crSigOpFp.OriDes,1,2)
		Replace NOpers With _NOper, DtAlts With _pData
	EndIf
EndIf

* Taxas de Cartão de Crédito
If Not Empty(crSigCdOpe.Grupos + crSigCdOpe.Contas) And (crSigOpFp.Infos = [T]) And Not Empty(crSigOpFp.GrupoTs) And ;
   Not Empty(crSigOpFp.ContaTs) And (crSigOpFp.Desc2s > 0) And (_Numc > 0) And Not Eof([crSigMvPar])
   
	lcSql = [Select * From SIGOPOCO Where Grupos = ']+crSigOpFp.GrupoTs+[' And Contas = ']+crSigOpFp.ContaTs+[' ]+;
			Iif(crDOpe.Opers=[CR],[And Opers = 'DB' ],[And Opers = 'CR'] )

	ThisForm.poDataMgr.Sqlexecute(lcSql,'CrSIGOPOCO')
	
	Select CrSIGOPOCO
	If Reccount('CrSIGOPOCO') = 0
		MessageBox('Ocorrencia não Cadastrada para Taxa de Cartão da Condição de Pagamento '+crSigOpFp.Fpags,0+16,'')
		If llvOper
			Return -1
		Else
			Return .f.
		EndIf
	EndIf
	xsOper = CrSIGOPOCO.Opers
	xOper  = Iif(xsOper=[DB],[CR],[DB])
	
	lnvTx  = Round( lnValor * crSigOpFp.Desc2s / 100 , 2 ) * crSigMvPar.Cotfpgs
		
	_noper = fGerUniqueKey( 'SigMvCcr' )

	Insert Into crSigOpCoc (Emps,Dopcs,Dataos,Notitulos,Moedas,Cotacaos, EmpDopNcs,CidChaves,Nopers,Valors,Opers,;
							Codigos,ValConvs, Autos ) Values ;
						   (crSigMvCab.Emps,_Dopc,Date(),_Numc,crSigMvPar.MoeFpgs, crSigMvPar.CotFpgs,;
						   crSigMvCab.Emps + _Dopc + Str(_Numc,6), fUniqueIds(), _nOper, lnvTx, CrSIGOPOCO.Opers, ;
						   CrSIGOPOCO.Codigos, Abs(lnvTx), .t.)

	_Hist = Alltrim(crSIGOPOCO.Descrs) + [-] + Alltrim(crSigOpCoc.codigos) + [-] + _Titulo

	For lnCont=1 to 2

		lcGOri = Iif(lnCont=1,crSigOpFp.GrupoTs,_GrupoTx)
		lcCOri = Iif(lnCont=1,crSigOpFp.ContaTs,_ContaTx)
		lcGDes = Iif(lnCont=1,_GrupoTx,crSigOpFp.GrupoTs)
		lcCDes = Iif(lnCont=1,_ContaTx,crSigOpFp.ContaTs)
		lcodc  = Iif(lnCont=1,xOper,xsoper)
		lcSdc  = Iif(lnCont=1,xsoper,xoper)
		lcGPem = ' '
		lcCPem = ' '

			_voper = crSigMvCab.Emps + Alltrim( str( _noper, 9 ) )
		Else
			_voper = crSigMvCab.Emps + _TIPOE + Alltrim( str( _noper, 9 ) )
		EndIf
		
		If lnvTx = 0
			Loop
		EndIf
		
		_CidChave = Dtos(ldDtContab)+Transform(fGerUniqueKey(Dtos(ldDtContab)),"@L 999999")+ThisForm.Sigkey 

		lcMd1 = Iif(Empty(crSigOpFp.MoeFpgs), crSigCdOpe.CMoes, crSigOpFp.MoeFpgs)

		Insert Into &_ArqMccr ( NOpers, VOpers, Docus, Emps, Datas, usualts, datalts, Vencs, hists,;
				opers, sopers, moedas, cotacaos, valors, autos, grupos, contas, sgrupos, scontas, ;
				smoedas, scotacaos, svalors, Dopes, numes, valpags,	dopcs, numcs, nfs, titulos,;

				bContas,TitBans,Tipos, Jobs ) ;

					_hist, lcodc, lcsdc, lcMd1, 1, lnvTx, .t., lcGOri, lcCOri, ;
					lcGDes,lcCDes, lcMd1, 1, lnvTx, ;
					crSigMvCab.Dopes, crSigMvCab.numes, lnvTx ,_dopc, _numc, ;
					_lcNota, Alltrim( _Titulo ) + '/' + Alltrim( Str( ThisForm.ContaPar ) ),;
					lcGPem, lcCPem, lcGrupoCob, lcContaCob, 0, crSigMvCab.DtEmis, _Hist2, _CidChave, ;
					crSigMvCab.Emps + crSigMvCab.Dopes + Str(crSigMvCab.Numes,6), _Departamento, lcTitBan,'O', crSigMvCab.Jobs )

			EndIf
			=fRecalculaS( crSigMvCcr.Grupos, crSigMvCcr.Contas, crSigMvCcr.Datas, crSigMvCcr.Moedas, ThisForm.poDataMgr  )

	Next

	Replace VPags    With lnvTx,;
			ValoCurs With lnvTx,;
			Pagos    With Iif(Valos = Vpags, '0', '1') in crSigMvPar
EndIf

Else
	Return .t.
EndIf

PROCEDURE montahists
Lparameters pTit, pFpg, pCli

Local lcHis, lcTit, lcFpg, lcCli

lcTit = Iif(Type([pTit])=[C], pTit, Titulos)
lcFpg = Iif(Type([pFpg])=[C], pFpg, [])
lcCli = Iif(Type([pCli])=[C], pCli, [])

lcHis = []
If Not Empty(crSigCdOpe.Hists)
	lcHis = lcHis + Alltrim(crSigCdOpe.Hists) + [ ]
EndIf

If (crSigCdOpe.TpHists = 5) And ThisForm.MultEmps
	lcHis = lcHis + Alltrim(Emps) + [ ]
EndIf

Do Case
	Case (crSigCdOpe.TpHists = 1) && NF
		lcHis = lcHis + Iif(Empty(Notas), [SEM DOCUMENTO], Alltrim(Notas))

	Case (crSigCdOpe.TpHists = 2) && Título
		lcHis = lcHis + Iif(Empty(lcTit), [SEM DOCUMENTO], Alltrim(lcTit)) + [/] + Alltrim(Str(ThisForm.ContaPar))

	Case (crSigCdOpe.TpHists = 3) && Movimentação
		lcHis = lcHis + fGerMascara(Numes)

	Case (crSigCdOpe.TpHists = 5) && Movimentação Detalhada
		lcHis = lcHis + Alltrim(Dopes) + [ ] + Alltrim(fGerMascara(Numes)) + [ ] + ;
						Iif(Empty(lcFpg), [], [(] + Alltrim(crSigOpFp.FPags) + [) ]) + ;
						Alltrim(lcCli)

	Case (crSigCdOpe.TpHists = 6) && Conta
		lcCta = Iif(crSigCdOpe.Origems = 1, crSigMvCab.ContaOs, crSigMvCab.ContaDs)

		ThisForm.poDataMgr.CursorQuery('SigCdCli', 'crTmpCli', 'Iclis', lcCta, [IClis, RClis])
		lcHis = lcHis + Alltrim(crTmpCli.RClis) + [ (] + Alltrim(crTmpCli.IClis) + [)]
EndCase

Return lcHis
* Fim **************************************************************************************
ENDPROC
