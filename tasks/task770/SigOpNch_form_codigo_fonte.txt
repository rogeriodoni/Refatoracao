PROCEDURE Init

ThisForm.cntSombra.lblSombra.Caption = ThisForm.Caption
ThisForm.cntSombra.lblTitulo.Caption = ThisForm.Caption

Private lcNew, lcChave

lcChave = NULL

Thisform.nAntChqVal = 0

WITH Thisform
	.poDatamgr = CreateObject('fSqlConector',thisform.Name)

	If Not .poDatamgr.pnIdconn>0
		Messagebox('Impossivel Efetuar Conexão com Servidor de Banco de Dados.',(0+48),'Erro')
		Return(.f.)
	Endif

	With .poDatamgr
		.AddCursor('SigOpFp','FPags','crSigOpFp','FPags',.f.)
		.AddCursor('SigSySeq','Valor','crAutoriza','Valor',.f.)
		.Requery("CrSigOpFp")
		.Requery("CrAutoriza")
	EndWith
Endwith	

Select crSigOpFp
Scan
	If crSigOpFp.Infos = 'C' And crSigOpFp.EmiChqs = 1 And Not IsEmpty(crSigOpFp.Bancos) And ;
			Not IsEmpty(crSigOpFp.Agencias) And Not IsEmpty(crSigOpFp.NContas)
		lcChave = PADR(crSigOpFp.Bancos + crSigOpFp.Agencias + crSigOpFp.NContas,25)
	  	Select crAutoriza
	  	LOCATE FOR valor = lcChave
		IF ! EOF()
			Insert Into csCheques (FPags, Bancos, Agencias, NContas, Cheques) Values ;
								  (crSigOpFp.FPags, crSigOpFp.Bancos, crSigOpFp.Agencias, ;
								   crSigOpFp.NContas, crAutoriza.Conteudo + 1)
		Else
			Insert Into csCheques (FPags, Bancos, Agencias, NContas, Cheques) Values ;
								  (crSigOpFp.FPags, crSigOpFp.Bancos, crSigOpFp.Agencias, ;
								   crSigOpFp.NContas, 1)
		Endif
	Endif
	Select crSigOpFp
EndScan

Select csCheques
Go Top

With ThisForm.Grade
	.Refresh
	.Column5.SetFocus
EndWith

ENDPROC

------------------------------------------------------------
* PROCEDURE Load
------------------------------------------------------------
PROCEDURE Load

=fConfigGeral()

Create Cursor csCheques ( FPags C(12), Bancos C(3), Agencias C(4), NContas C(10), ;
						  Cheques N(6), Marcas L(1) )
Index On Fpags Tag FPags
ENDPROC

* PROCEDURE Click
------------------------------------------------------------
PROCEDURE Click
Local lcChave

Select csCheques
Scan For Marcas
	If !IsEmpty(csCheques.Cheques)
		lcChave = PADR(csCheques.Bancos + csCheques.Agencias + csCheques.NContas,25)
		SELECT crAutoriza
		LOCATE FOR valor = lcChave
		If ! EOF()
			Do While !Rlock()
			Enddo
			Replace Conteudo With csCheques.Cheques - 1
			Unlock
		Else
			Insert Into crAutoriza (Valor, Conteudo) Values (lcChave, csCheques.Cheques-1)
		Endif
	Endif
	Select csCheques
EndScan

SELECT crAutoriza

Local m.lnErro1

m.lnErro1 = ThisForm.poDatamgr.pnIdConn

With ThisForm
	If m.lnErro1>0 
		m.lnErro1 = Iif(.poDataMgr.UpDate('crAutoriza'),1,-1)
	EndIf 
	If m.lnErro1>0 
		m.lnErro1 = .poDataMgr.Commit()
	EndIf 
	If Not m.lnErro1>0 
		.poDataMgr.RollBack()
	EndIf 
EndWith 
If Not m.lnErro1>0 
	= MessageBox('Erro na Gravação dos Dados do Cadastro de Operações.',48)
EndIf 

ThisForm.Release
ENDPROC
