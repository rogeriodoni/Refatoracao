PROCEDURE Init
lcQryCanf  = [Select * From SigCdCnf Where emps = '] + _Empr + [']
lcQryICanf = [Select * From SigCdCaf ]
lcQryOpi   = [Select * From SigOpPic ]
With ThisForm
	.poDataMgr = CreateObject('fSqlConector', ThisForm.Name)

	If (.poDataMgr.pnIdconn > 0)
		With .poDataMgr
			.AddCursor('SigCdCnf', 'Codigos', 'crSigCdCnf', '', ThisForm.Pagina.Lista.Grade, lcQryCanf)
				.pfSqlTabela(1).pColuna('codigos', '', '', 'Códigos', 55, .T.)
				.pfSqlTabela(1).pColuna('datas', '', '', 'Datas', 80, .F.)
				.pfSqlTabela(1).pColuna('usuars', '', '', 'usuars', 80, .F.)
				.pfSqlTabela(1).pColuna('obss', '', '', 'Observações',500, .F.)
			.AddCursor('SigCdCaf', 'cIdChaves', 'crSigCdCaf', '', , lcQryICanf)
			.AddCursor('SigOpPic', 'cIdChaves', 'crSigOpPic', '', , lcQryOpi)
		EndWith

		Select crSigOpPic
		Index On nops tag nops

		Create Cursor TmpNop (Nops n(10), Cpros c(14), Qtds n(10,3), Qtdos n(10,3), nFunds n(6), DtFunds d(8))
		Index on Nops Tag Nops

		With ThisForm.Pagina.Dados.GradeNop
			.recordsource        = 'TmpNop'
			.column1.ControlSource = 'TmpNop.Nops'
			.Column2.ControlSource = 'TmpNop.Cpros'
			.Column3.ControlSource = 'TmpNop.Qtdos'
			.Column4.ControlSource = 'TmpNop.Qtds'
		EndWith

		=DoDefault()

		ThisForm.Pagina.Lista.Grade.SetFocus
		ThisForm.Refresh
		ThisForm.LockScreen = .f.
	Else
		=Messagebox('Impossivel Efetuar Conexão com Servidor de Banco de Dados...', 0+48, 'CreateObject')
		Return .f.
	EndIf
EndWith
* Fim **************************************************************************************
ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Lista.Grupo_op.Click
------------------------------------------------------------
PROCEDURE Pagina.Lista.Grupo_op.Click
LPARAMETERS opcao

Local Codigos,llFechado

If Type('m.Opcao')=='N'
	This.Value = m.Opcao 
Endif

Select TmpNop
Zap
Select CrSigOpPic
Zap
Select CrSigCdCaf
Zap

If ( DoDefault(This.Value) )

	ThisForm.LockScreen = .t.
	
	If InList(ThisForm.pcEscolha,'ALTERAR','CONSULTAR','EXCLUIR')
	
		lcQuery = [Select * ] + ;
			    [ From SigCdCaf] + ;
			    [ Where codigos=]+Str(crSigCdCnf.codigos,6)
		If (ThisForm.poDataMgr.SqlExecute(lcQuery, 'crCarregaSigCdCaf') < 1)
			=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (crCarregaSigCdCaf)')
			Return 0
		EndIf
		Select crCarregaSigCdCaf
		Scan 
			lcQuery = [Select * ] + ;
				    [ From SigOpPic] + ;
				    [ Where nops=]+Str(crCarregaSigCdCaf.Nops)+[]
			If (ThisForm.poDataMgr.SqlExecute(lcQuery, 'crCarregaSigOpPic') < 1)
				=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (crCarregaSigOpPic)')
				Return 0
			EndIf
			Select crCarregaSigOpPic
			lcPro = crCarregaSigOpPic.Cpros
			Insert Into TmpNop (Nops, Cpros, Qtds, nFunds, DtFunds ) ;
					Value (crCarregaSigCdCaf.Nops, lcPro, crCarregaSigCdCaf.Qtds, crCarregaSigCdCaf.nFunds, crCarregaSigCdCaf.DtFunds )

			Select crCarregaSigOpPic
			Scan
				Scatter Memvar
				Select crSigOpPic
				Append Blank
				Gather Memvar
			EndScan
		EndScan	
		
	Endif
	
	Do Case
		Case InList(ThisForm.pcEscolha,'INSERIR','ALTERAR')

			If ThisForm.pcEscolha=='INSERIR'
	
				Select crSigCdCnf
				
				m.Codigos = fGerUniqueKey('SigCdCnf')
				m.Emps    = _Empr
				Gather Memvar Fields Codigos, emps

				ThisForm.Pagina.Dados.GetUsu.Value = Usuar
				
			EndIf
			
			Thisform.Pagina.Dados.getDatas.Value = Iif(ThisForm.pcEscolha=='INSERIR',Date(),This.Parent.Parent.Dados.getDatas.Value)
			Thisform.Pagina.Dados.getDatas.SetFocus
			
		Case ThisForm.pcEscolha=='CONSULTAR'
	
			  
		Case ThisForm.pcEscolha=='EXCLUIR'
			Thisform.Pagina.Dados.Grupo_Salva.Salva.SetFocus
			  
		Case ThisForm.pcEscolha=='PROCURAR'
			ThisForm.Pagina.Dados.Visible = .f.
			Thisform.Pagina.Dados.getCodigos.SetFocus

	EndCase
	
	Select crSigCdCnf
	
	ThisForm.Refresh
	ThisForm.LockScreen = .f.
	
Else

	Select crSigCdCnf
	ThisForm.mAtivapagina1
	
EndIf
ENDPROC

------------------------------------------------------------
PROCEDURE Valid
If This.Value > 0
	Nop = This.Parent.Get_Nop.Value
	=Seek(Nop,'TmpNop','Nops')
	If This.Value > TmpNop.Qtdos
		MessageBox('Quantidade não disponivel!!!',0+48,'')
		Return .f.
	EndIf
	
	lnQtd = This.Value
	Replace TmpNop.Qtds With lnQtd in TmpNop
	
	Select crSigOpPic
	Set Order to Nops
	Seek Nop

	Scan While Nops = Nop And lnQtd > 0
	
		If QtdTubos <= lnQtd
			lnQtd = lnQtd - QtdTubos
			Replace QtdTubos with 0
		Else
			Replace Qtdtubos With Qtdtubos - lnQtd
			lnQtd = 0
		Endif
			
	EndScan

	Select crSigOpPic
	Set Order to Nops
	Seek Nop
	Sum QtdTubos to QtdTub While Nops = Nop
	If QtdTub = 0
		Seek Nop
		Replace nFunds With 0, DtFunds With Ctod('') While Nops = Nop
	Endif
		
	This.Parent.GradeNop.Refresh
	
	This.Value = 0
	This.Refresh

EndIf

ENDPROC

PROCEDURE When
Return (!Empty(This.Parent.Get_nop.Value) )

	Set Order to Nops

		This.Refresh
		Return .f.
	Endif

	Insert Into TmpNop (Nops, Cpros, Qtds, Qtdos, nFunds, DtFunds ) ;
		Value (LnOp, lcPro, 0, lnQtdT, lnFund, ldDtf )
