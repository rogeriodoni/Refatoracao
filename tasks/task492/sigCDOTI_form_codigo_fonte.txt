* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
LParameters lcAcesso

WITH Thisform
	.poDatamgr = CreateObject('fSqlConector',.Name)

	If Not .poDatamgr.pnIdconn>0
		Messagebox('Impossivel Efetuar Conexão com Servidor de Banco de Dados.',(0+48),'Erro')
		Return(.f.)
	Endif

	With .poDatamgr
		.AddCursor('SigOpOpe','Dopes','crSigOpOpe','Dopes',ThisForm.pagina.lista.grade)
		.pfSqlTabela(1).pColuna('Dopes','','','Descrição',200,.t.)
		.pfSqlTabela(1).pColuna('Opers','','','Operação',80,.t.)
		.pfSqlTabela(1).pColuna('NDopes','','','Número',80,.t.)
		.AddCursor('SigCdPrg','pkChaves','crSigCdPrg','programas',.f.)
	EndWith
Endwith	

=DoDefault()

ENDPROC

* PROCEDURE msv_alterar
------------------------------------------------------------
PROCEDURE msv_alterar
Local lnErro1

m.lnErro1 = ThisForm.poDatamgr.pnIdConn

With ThisForm
	If m.lnErro1>0 
		m.lnErro1 = Iif(.poDataMgr.UpDate('crSigOpOpe'),1,-1)
	EndIf 
	If m.lnErro1>0 
		m.lnErro1 = Iif(.poDataMgr.UpDate('crSigCdPrg'),1,-1)
	EndIf 
	If m.lnErro1>0 
		m.lnErro1 = .poDataMgr.Commit()
	EndIf 
	If Not m.lnErro1>0 
		.poDataMgr.RollBack()
	EndIf 
EndWith 
If Not m.lnErro1>0 
	= MessageBox('Erro na Alteração dos Dados do Cadastro de Operações.',48)
EndIf 

ENDPROC
