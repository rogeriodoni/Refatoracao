PROCEDURE Init
LPARAMETERS lcacesso
ThisForm.cntSombra.lblSombra.Caption = ThisForm.Caption
ThisForm.cntSombra.lblTitulo.Caption = ThisForm.Caption
With ThisForm
	.LockScreen = .T.
	ThisForm.grid1.RecordSource=''		&&Conexao com a tabela de Dados
	
	.poDatamgr = CreateObject('fSqlConector',.name)	&&Criacao do Objeto de conexao
	m.lnErro1 = .poDatamgr.pnIdconn						&&Status da Conexao
	
	lcQuery = [select carqicones,marqicones,']+Space(128)+[' as ctmpicones from SigSyIco order by 1]  &&Tras Dados da Tabela
	With .podatamgr
		.AddCursor('SigSyIco','carqicones','crTmpcdico','carqicones',.F.,lcQuery)
		.Requery('crTmpcdico')
	EndWith
	
	If Used('crTmpcdico')
		m.lcDirTmp1 = Addbs(Sys(2023))				&&Cria Diretorio Temporario
		Select ('crTmpcdico')
		Index On carqicones Tag carqicones
		Set Order To carqicones In crTmpcdico
		
		Locate 
				
		Scan While m.lnErro1>0
			m.lcArqIcone1 = (m.lcDirTmp1+'ArqIcone'+Sys(2015)+'.ico')		&&Caminho para Acessar os Arquivos
			m.lnTamIcone1 = StrToFile(Nvl(crTmpcdico.marqicones,''),m.lcArqIcone1)
			If m.lnTamIcone1>0
				Replace ctmpicones With LOWER(m.lcArqIcone1) In ('crTmpcdico')
			EndIf 
		EndScan 
		Go Top In crTmpcdico 
	EndIf 
	

	= DoDefault()
	With .Grid1
		.SetAll("FontName","Courier New","Column")
		.SetAll("FontName","Arial","Header")
		.SetAll("FontName","Courier New","TextBox")
		.SetAll("FontName","Courier New","CommandButton")
		.SetAll("FontSize",9,"Column")
		.SetAll("FontSize",8,"Header")
		.SetAll("FontSize",9,"TextBox")
		.SetAll("FontSize",9,"CommandButton")
		.SetAll("Format","K","Column")
		.SetAll("Format","K","TextBox")
		.SetAll("Movable",.f.,"Column")
		.SetAll("ReadOnly",.t.,"Column")
		.SetAll("ReadOnly",.t.,"TextBox")
		.SetAll("Resizable",.f.,"Column")
		.SetAll("Alignment",3,"Column")
		.SetAll("Alignment",3,"TextBox")
		.SetAll("Alignment",2,"Header")
		.SetAll("BorderStyle",0,"TextBox")
		.SetAll("Margin",0,"TextBox")
		.SetAll("Visible",.t.,"TextBox")
		.HeaderHeight = 15
		.RowHeight = 15
		.LinkMaster = ''
		.RelationalExpr = ''
		.RecordSource = 'crTmpcdIco' 
		.Column1.ControlSource = 'crTmpcdIco.carqicones'
		.Refresh
		.Visible = .t.
	EndWith 
	.LockScreen = .F.
EndWith


Local lcarquivo

lcarquivo = Alltrim(crTmpcdIco.ctmpicones)

If Not IsEmpty(lcarquivo)
	ThisForm.Icone.picture = lcarquivo
	ThisForm.Icone.visible = .t.
else
	ThisForm.Icone.Visible = .f.
EndIf
ThisForm.grid1.SetFocus


ENDPROC

PROCEDURE cmdInserir.Click
m.lcArqIcone = GetPict('ico;gif;jpg;bmp;pic','','OK')

**Obs: Os icones devem esta em Minusculo e com Underline

If Not Empty(m.lcArqIcone) And File(m.lcArqIcone)							&&Caso seja Selecionado uma Imagem
	ThisForm.LockScreen = .t.

	m.lcDirTmp1 = Addbs(Sys(2023))								&&Cria Diretorio Temporario
	m.lcArqIcone1 = (m.lcDirTmp1+'ArqIcone'+Sys(2015)+'.ico')		&&Caminho para Acessar os Arquivos

	If Not Seek(JustFname(LOWER(m.lcArqIcone)),'crTmpcdIco','carqicones')  &&Caso nao seja encontrado no Banco de Dados
		Append Blank in crTmpcdico									&&Adiciona no Cursor
			
		Replace carqicones With  LOWER(JustFname(m.lcArqIcone)) 			In crTmpcdico
		Replace marqicones With  FILETOSTR(m.lcArqIcone)					In crTmpcdico
		Replace ctmpicones With  m.lcArqIcone1	 			    			In crTmpcdico
		
		m.lnTamIcone1 = StrToFile(crTmpcdico.marqicones,m.lcArqIcone1)
	Else														
		Replace carqicones With  LOWER(JustFname(m.lcArqIcone)) In crTmpcdico
		Replace marqicones With  FILETOSTR(m.lcArqIcone)		In crTmpcdico
		Replace ctmpicones With  m.lcArqIcone1	 			    In crTmpcdico
		
		m.lnTamIcone1 = StrToFile(crTmpcdico.marqicones,m.lcArqIcone1)
		
	    Update crTmpcdIco set carqicones=JustFname(LOWER(m.lcArqIcone)), marqicones=FILETOSTR(m.lcArqIcone ) where carqicones=JustFname(m.lcArqIcone)  &&Atualiza tabela	
	EndIf
	
	Thisform.grid1.Refresh
    m.lnErro1 = ThisForm.podatamgr.pnIdconn

	If m.lnErro1>0 
		m.lnErro1 = Iif(ThisForm.podatamgr.Update('crTmpcdico'),1,-1)
	EndIf 
	If m.lnErro1>0 
		m.lnErro1 = ThisForm.podatamgr.Commit()
	EndIf 
	If Not m.lnErro1>0 
		ThisForm.podatamgr.RollBack()
	EndIf 

	Select crTmpcdIco
	Go top
	Locate for cArqIcones = LOWER(JustFname(m.lcArqIcone))

	ThisForm.Refresh
	ThisForm.LockScreen = .f.
	Thisform.Grid1.SetFocus
Else 
	= MessageBox('Nome do Arquivo Invalido.',0+48+0)
EndIf

ENDPROC

* PROCEDURE cmdExcluir.Click
------------------------------------------------------------
PROCEDURE cmdExcluir.Click
If MessageBox("Tem certeza que deseja Excluir?",4+32,"Excluir")=6 &&Confirma
	ThisForm.LockScreen = .t.
	m.registro=Alltrim(crTmpcdico.carqicones)
	
	If ThisForm.poDataMgr.ChkRegister('SigCdPrg','barrapict',m.registro)
		Messagebox( 'Existem ítens cadastros que estão utilizando esse Registro!', 48, 'Exclusão não permetida' )
		ThisForm.LockScreen = .f.
		Return
	Endif

	Delete from crTmpcdico where carqicones=m.registro
	
    m.lnErro1 = ThisForm.podatamgr.pnIdconn

	If m.lnErro1>0 
		m.lnErro1 = Iif(ThisForm.podatamgr.Update('crTmpcdico'),1,-1)
	EndIf 
	
	If m.lnErro1>0 
		m.lnErro1 = ThisForm.podatamgr.Commit()
	Else
		ThisForm.podatamgr.RollBack()
	EndIf 

	ThisForm.LockScreen = .f.
	
	Thisform.grid1.Refresh
	Select crTmpcdIco
	Go top
	Locate for carqicones > m.registro
	If Eof()
		Go Bott
	Endif
	ThisForm.Refresh	
	ThisForm.grid1.SetFocus
EndIf
ENDPROC
