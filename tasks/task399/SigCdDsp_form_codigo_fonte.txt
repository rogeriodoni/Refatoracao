* PROCEDURE msv_procurar
------------------------------------------------------------
PROCEDURE msv_procurar
IF ! EMPTY(ThisForm.Pagina.Dados.get_cd_conta.Value)
	SELECT crSigCdDis
	SET order to contas
	SET NEAR on
	SEEK ALLTRIM(ThisForm.Pagina.Dados.get_cd_conta.Value)
	SET NEAR off
	ThisFOrm.Busca=2
ENDIF
IF ! EMPTY(ThisForm.Pagina.Dados.get_cd_grupo.Value)
	SELECT crSigCdDis
	SET order to Grupos
	SET NEAR on
	SEEK ALLTRIM(ThisForm.Pagina.Dados.get_cd_grupo.Value)
	SET NEAR off
	ThisFOrm.Busca=1
ENDIF
ThisForm.Pagina.Lista.Grade.Refresh
ENDPROC

------------------------------------------------------------
* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
LParameters lcAcesso

ThisFOrm.Busca=1

WITH Thisform
	.poDatamgr = CreateObject('fSqlConector',ThisForm.Name)

	If Not .poDatamgr.pnIdconn>0
		Messagebox('Impossivel Efetuar Conexão com Servidor de Banco de Dados.',(0+48),'Erro')
		Return(.f.)
	Endif

	With .poDatamgr
		.AddCursor('SigCdDis','CidChaves','crSigCdDis','Grupos',ThisForm.pagina.lista.grade)
		.pfSqlTabela(1).pColuna('Grupos','','','Grupo',80,.t.)
		.pfSqlTabela(1).pColuna('Contas','','','Conta',80,.t.)
	EndWith
Endwith	

=DoDefault()


ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Lista.Grupo_op.Click
------------------------------------------------------------
PROCEDURE Pagina.Lista.Grupo_op.Click
LPARAMETERS opcao

If !dodefault( OPCAO )
	ThisForm.mAtivapagina1
	Return
Endif

* inicializa os campos para inserção e procura
do case
case thisform.pcEscolha $ 'INSERIR/PROCURAR'
	with thisform.pagina.dados
		.get_ds_grupo.value = ''
		.get_ds_conta.value = ''
		.get_ds_grupo.refresh
		.get_ds_conta.refresh
	endwith
case thisform.pcEscolha $ 'ALTERAR/CONSULTAR/EXCLUIR'
	if ThisForm.podatamgr.CursorQuery('SigCdGcr','crSigCdGcr', 'Codigos', crSigCdDis.grupos)
		with thisform.pagina.dados.get_ds_grupo
			.value = evaluate('crSigCdGcr.descrs')
			.refresh
		endwith
	ENDIF
	DECLARE laCCli[2,2]
	laCCli[1,1]="Grupos"
	laCCli[1,2]=padr(crSigCdDis.grupos,10)
	laCCli[2,1]="IClis"
	laCCli[2,2]=crSigCdDis.contas
	if ThisForm.podatamgr.CursorQuery('SigCdCli','crSigCdCli', @LaCCli)
		with thisform.pagina.dados.get_ds_conta
			.value = evaluate('crSigCdCli.RClis')
			.refresh
		endwith
	endif
endcase

* posiciona o cursor no primeiro campo de acordo com a opção
do case
case thisform.pcEscolha $ 'ALTERAR'
	thisform.pagina.dados.get_cd_conta.setfocus
case thisform.pcEscolha $ 'INSERIR/PROCURAR' 
	thisform.pagina.dados.get_cd_grupo.setfocus
case thisform.pcEscolha $ 'CONSULTAR/EXCLUIR'
	thisform.pagina.dados.grupo_salva.salva.setfocus
endcase

ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
local lcCdGrupo,lcDsGrupo,lcCdConta,lcDsConta,lnRegistro

* inicializa as variáveis de campos
lcCdGrupo = thisform.pagina.dados.get_cd_grupo.value
lcDsGrupo = thisform.pagina.dados.get_ds_grupo.value
lcCdConta = thisform.pagina.dados.get_cd_conta.value
lcDsConta = thisform.pagina.dados.get_ds_conta.value

if thisform.pcEscolha $ 'INSERIR/ALTERAR'
	* consiste o campo grupo
	if IsEmpty(lcCdGrupo) or not ThisForm.podatamgr.ChkRegister('SigCdGcr', 'Codigos', lcCdGrupo)
		messagebox('Grupo inválido !',16,'')
		thisform.pagina.dados.get_cd_grupo.setfocus
		return(.f.)
	endif
	
	* consiste o campo conta
	if IsEmpty(lcCdConta) or not ThisForm.podatamgr.ChkRegister('SigCdCli', 'IClis', lcCdConta)
		messagebox('Conta inválida !',16,'')
		thisform.pagina.dados.get_cd_conta.setfocus
		return(.f.)
	ENDIF
	
	IF ThisForm.pcEscolha = "INSERIR"
		SELECT crSigCdDis
		Replace CidChaves WITH fUniqueIds()
	Endif
endif

	
* consiste a existência de um código já cadastrado
do case
case thisform.pcEscolha = 'INSERIR'

	DECLARE laCCli[2,2]
	laCCli[1,1]="Grupos"
	laCCli[1,2]=padr(lcCdGrupo,10)
	laCCli[2,1]="Contas"
	laCCli[2,2]=lcCdConta
	if ThisForm.podatamgr.ChkRegister('SigCdDis', @laCCli)
		messagebox('Disponibilidade já está cadastrada !',16,'')
		thisform.pagina.dados.get_cd_grupo.setfocus
		return(.f.)
	endif
case thisform.pcEscolha = 'ALTERAR'

	SELECT crSigCdDis
	lnRegistro = recno()
	DECLARE laCCli[2,2]
	laCCli[1,1]="Grupos"
	laCCli[1,2]=padr(lcCdGrupo,10)
	laCCli[2,1]="Contas"
	laCCli[2,2]=lcCdConta
	if ThisForm.podatamgr.ChkRegister('SigCdDis', @laCCli)
		SELECT crSigCdDis
		if lnRegistro <> recno()
			messagebox('Disponibilidade já está cadastrada !',16,'')
			thisform.pagina.dados.get_cd_conta.setfocus
			go lnRegistro
			return(.f.)
		endif
	else
		go lnRegistro
	endif
endcase

DoDefault()
Thisform.mAtivapagina1
ENDPROC
