PROCEDURE Init
LParameters poForm, pnDataSes

Set DataSession to pnDataSes

Thisform.ParentForm          = poForm
Thisform.ParentForm.Enabled  = .F.

Thisform.GravaDados = .f.

With Thisform
	.Caption = 'Grupos de Produtos ' + AllTrim(crSigCdOpe.Dopes)
EndWith

ThisForm.cntSombra.lblSombra.Caption = ThisForm.Caption
ThisForm.cntSombra.lblTitulo.Caption = ThisForm.Caption

lcOperacao = crSigCdOpe.Dopes

With Thisform
	.poDatamgr = CreateObject('fSqlConector')

	If Not .poDatamgr.pnIdconn>0
		Messagebox('Impossivel Efetuar Conexão com Servidor de Banco de Dados.',(0+48),'Erro')
		Return(.f.)
	Endif

	With .poDatamgr
		.AddCursor('SigOpGpo','cIdChaves','CrSigOpGpo','')
	EndWith
Endwith	

Create Cursor xOpe ( Dopes c(20), Cgrus c(3), Dgrus c(30) )
Index on Dopes Tag Dopes
Set Order to

With Thisform.GradeOpe
	.RecordSource          = 'xOpe'
	.Column1.ControlSource = 'xOpe.Cgrus'
	.Column2.ControlSource = 'xOpe.Dgrus'
	.Refresh
EndWith

lcSql = [Select a.*, b.Dgrus From SigOpGpo a, SigCdGrp b Where a.Dopes = ']+CrSigCdOpe.Dopes+[' And a.Cgrus = b.Cgrus ]
ThisForm.Podatamgr.Sqlexecute(lcSql,'CsSelecao')

Select CsSelecao
Scan
	Scatter Memvar
	Select xOpe
	Append Blank
	Gather Memvar
EndScan

Select xOpe
Set Order to
Go Top

ThisForm.GradeOpe.Refresh


ENDPROC

PROCEDURE Load
=fConfigGeral()

ENDPROC

* OBJETO: SIGCDGRP.cmdSair
* Classe: fwbtng (Base: commandbutton)
* Registro: 16
================================================================================

------------------------------------------------------------
* PROCEDURE Click
------------------------------------------------------------
PROCEDURE Click
If ThisForm.GravaDados

	Select Cgrus, Sum(1) as Qt From xOpe Group by 1 Into Cursor Selecao where cgrus <> ' ' having Sum(1) > 1
	If Reccount('Selecao') > 0
		MessageBox('Existem lancamentos de Grupos Em Duplicidade!!!',0+16,'')
		Return .f.
	EndIf
	
	Select CrSigOpGpo
	Zap
	
	Select xOpe
	Scan
		If Not Empty(Cgrus)
			Insert Into crSigOpGpo (Dopes, Cgrus, cIdChaves) ;
							Values (crSigCdOpe.Dopes, xOpe.Cgrus, fUniqueIds())
		Endif
	EndScan
	
	m.lnErro1 = ThisForm.poDatamgr.pnIdConn

	With ThisForm

		lcSql = [Delete From SigOpGpo Where dopes = ']+CrSigCdOpe.dopes+[']
		m.lnErro1 = ( .PodataMgr.SqlExecute(lcSql,'') )

		If m.lnErro1>0 
			m.lnErro1 = Iif(.poDataMgr.UpDate('crSigOpGpo'),1,-1)
		EndIf 
		If m.lnErro1>0 
			m.lnErro1 = .poDataMgr.Commit()
		EndIf 
		If Not m.lnErro1>0 
			.poDataMgr.RollBack()
		EndIf 
	EndWith 
	If Not m.lnErro1>0 
		= MessageBox('Erro na Gravação dos Dados!!!',48)
	EndIf 
EndIf
Thisform.ParentForm.Enabled = .T.
ThisForm.Release

ENDPROC

------------------------------------------------------------
* PROCEDURE Click
------------------------------------------------------------
PROCEDURE Click
Select xOpe

lcGde = ThisForm.getGdeGrps.Value

If Empty(lcGde)
	=MessageBox([Preencha o Grande Grupo Antes de Processar!], 0+16, [Atenção!])
	ThisForm.Pagina.Dados.getGdeGrps.SetFocus
	Return .f.
EndIf

ThisForm.GravaDados = .t.

lcQuery = [Select CGrus, DGrus ] + ;
		    [From SigCdGrp ] + ;
		    [Where Mercs = '] + lcGde + [']

If (ThisForm.poDataMgr.SqlExecute(lcQuery, [LocalGru]) < 1)
	=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (LocalGru)])
	Return 0
EndIf
Go Top In LocalGru
If Not Eof([LocalGru])
	Delete From xOpe Where Empty(CGrus)
EndIf

Select LocalGru
Scan
	Select xOpe
	Go Top
	Locate For CGrus = LocalGru.CGrus
	If Eof([xOpe])
		Insert Into xOpe(CGrus, DGrus) Values (LocalGru.CGrus, LocalGru.DGrus)
	EndIf
EndScan
Go Top In LocalGru
If Not Eof([LocalGru])
	Append Blank In xOpe
EndIf
ThisForm.gradeOpe.Refresh() 
Go Bottom In xOpe
ThisForm.gradeOpe.Column1.SetFocus()
* Fim **************************************************************************************
ENDPROC
