PROCEDURE msv_procurar
Local lcBusca

If Not Empty(ThisForm.Pagina.Dados.getDescs.Value)
	ThisForm.poDataMgr.LocateCursor([crSigCdMod], [Descs], ThisForm.Pagina.Dados.getDescs.Value)
EndIf
* Fim **************************************************************************************
ENDPROC

------------------------------------------------------------
* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
With ThisForm
	.poDataMgr = CreateObject([fSqlConector], .Name)

	If (.poDataMgr.pnIdconn > 0)
		With .poDataMgr
			.AddCursor([SigCdMod], [pkchave], [crSigCdMod], [], ThisForm.Pagina.Lista.Grade)
			.pfSqlTabela(1).pColuna([Codigos],  [], [], [Código],    080, .t.)
			.pfSqlTabela(1).pColuna([Descs], [], [], [Descrição], 150, .t.)
		EndWith

		=DoDefault()

		.Pagina.Lista.Grade.Column1.SetFocus
	Else
		=Messagebox([Impossível Efetuar Conexão Com o Servidor de Banco de Dados...], 0+48, [CreateObject])
		Return .f.
	EndIf
EndWith
* Fim **************************************************************************************
ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Lista.Grupo_op.Click
------------------------------------------------------------
PROCEDURE Pagina.Lista.Grupo_op.Click
Parameters lnOpcao

If Not DoDefault(lnOpcao)
	Thisform.mAtivaPagina1
	Return .F.
Endif

Do Case
Case Inlist(Thisform.pcEscolha, [INSERIR])
	
	lnCodigo = fGerUniqueKey('SIGCDMOD')
	lcCodigo = Alltrim(Str(lnCodigo))

	Replace Codigos with lcCodigo in crSigCdMod
	
	Thisform.pagina.dados.getCodigos.Value = lcCodigo
	Thisform.pagina.dados.getCodigos.Refresh
	
	Thisform.pagina.dados.getDescs.SetFocus

Case Inlist(Thisform.pcEscolha, [INSERIR], [PROCURAR], [ALTERAR])
	Thisform.pagina.dados.getDescs.SetFocus

Otherwise
	Thisform.pagina.dados.Grupo_Salva.Salva.SetFocus
Endcase
* Fim **************************************************************************************

ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
If InList(ThisForm.pcEscolha, [INSERIR], [ALTERAR])
	If Empty(ThisForm.Pagina.Dados.getDescs.Value)
		=Messagebox([Descrição Inválida!!!], 48, [Atenção!!!])
		ThisForm.Pagina.Dados.getDescs.SetFocus()
		Return .f.
	EndIf
	Set Step On 
	If (ThisForm.pcEscolha = [INSERIR])
		lcQuery = [Select Descs ] + ;
				    [From SigCdMod ] + ;
				   [Where descs = '] + Alltrim(ThisForm.Pagina.Dados.getDescs.Value) + [']

		If (ThisForm.poDataMgr.SqlExecute(lcQuery, [LocalBus]) < 1)
			=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (LocalBus 1)])
			Return .f.
		EndIf
		Go Top In LocalBus
		If Not Eof([LocalBus])
			=Messagebox([Descrição Já Cadastrada!!!], 48, [Atenção!!!])
			ThisForm.Pagina.Dados.getCodigos.SetFocus()
			Return .f.
		EndIf
	EndIf
EndIf

If (ThisForm.pcEscolha = [INSERIR])
	Replace pkchave With fUniqueIds() in crSigCdMod 
EndIf 


=DoDefault()

ThisForm.mAtivaPagina1
* Fim **************************************************************************************
ENDPROC
