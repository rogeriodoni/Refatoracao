PROCEDURE KeyPress
LParameters nKeyCode, nShiftAltCtrl

If nKeyCode==60
	If !Thisform.LeCheque
		Thisform.ChqLido = ""
	Endif
	Thisform.LeCheque = .T.
Endif
If Thisform.LeCheque
	NoDefault
	Thisform.ChqLido = Thisform.ChqLido + CHR(nKeyCode)
Endif
If nKeyCode==58
	This.Valid
	ThisForm.LeCheque = .F.
Endif
ENDPROC

PROCEDURE Valid
Local leitor, contador, tecla

leitor = Iif(Thisform.LeCheque, Thisform.ChqLido, '')

	Keyboard"{CTRL+TAB}"

	Endif

------------------------------------------------------------
* PROCEDURE Click
------------------------------------------------------------
PROCEDURE Click
Local llOk, lcWher
Set Step On 
llOk = .f.
lcWher = ''
Select crSigChe
Set Near On
Do Case
	Case !IsEmpty( _ProcVen )
		Set Order To VENC
		_Busca = ( DTOS( _PROCVEN ) + Iif(IsEmpty(_PROCBCO),'',_ProcBco) + Iif(IsEmpty(_PROCAGE),'',_ProcAge) + Iif(IsEmpty(_PROCCON),'',_ProcCon) + Iif(IsEmpty(_PROCCHE),'',_ProcChe) )
		llOk = Seek( _Busca )
		If !llOk
			llVenc = fDtoSQL(_ProcVen)
			lcWher = [Vencs = ?llVenc ]+Iif(!IsEmpty(_ProcBco),[And Bancos=?_ProcBco ],[])+;
					Iif(!IsEmpty(_ProcAge),[And Agencias=?_ProcAge ],[])+Iif(!IsEmpty(_ProcCon),[And nContas=?_ProcCon ],[])+;
					Iif(!IsEmpty(_PROCCHE),[And nCheques=?_ProcChe ],[])
		Endif

	Case !IsEmpty( _ProcVal )
		Set Order To VALOR
		_Busca = ( STR( _PROCVAL, 12 , 2 ) + Iif(IsEmpty(_PROCBCO),'',_ProcBco) + Iif(IsEmpty(_PROCAGE),'',_ProcAge) + Iif(IsEmpty(_PROCCON),'',_ProcCon) + Iif(IsEmpty(_PROCCHE),'',_ProcChe) )
		llOk = Seek( _Busca )
		If !llOk
			lcWher = [Valors = ?_ProcVal ]+Iif(!IsEmpty(_ProcBco),[And Bancos=?_ProcBco ],[])+;
					Iif(!IsEmpty(_ProcAge),[And Agencias=?_ProcAge ],[])+Iif(!IsEmpty(_ProcCon),[And nContas=?_ProcCon ],[])+;
					Iif(!IsEmpty(_PROCCHE),[And nCheques=?_ProcChe ],[])
		Endif		

	Case !IsEmpty( _ProcBco )
		Set Order To CHEQUE
		_Busca = ( _PROCBCO + Iif(IsEmpty(_PROCAGE),'',_ProcAge) + Iif(IsEmpty(_PROCCON),'',_ProcCon) + Iif(IsEmpty(_PROCCHE),'',_ProcChe) )
		llOk = Seek( _Busca )
		If !llOk
			lcWher = [Bancos=?_ProcBco ]+;
					Iif(!IsEmpty(_ProcAge),[And Agencias=?_ProcAge ],[])+Iif(!IsEmpty(_ProcCon),[And nContas=?_ProcCon ],[])+;
					Iif(!IsEmpty(_PROCCHE),[And nCheques=?_ProcChe ],[])
		Endif		

	Case !IsEmpty( _ProcAge )
	
		Set Order To AGENCIA
		
		If Empty(_PROCCON) and Not Empty(_PROCCHE)
			Set Order To AgenciaChe
		EndIf 
		
		If Empty(_PROCCHE) and Not Empty(_PROCCON)
			Set Order To AgenciaCta
		EndIf 
		
		_Busca = ( _PROCAGE + Iif(IsEmpty(_PROCCON),'',_ProcCon) + Iif(IsEmpty(_PROCCHE),'',_ProcChe) )
		llOk = Seek( _Busca )
		If !llOk
			lcWher = [Agencias=?_ProcAge ]+;
					Iif(!IsEmpty(_ProcCon),[And nContas=?_ProcCon ],[])+;
					Iif(!IsEmpty(_PROCCHE),[And nCheques=?_ProcChe ],[])
		Endif

	Case !IsEmpty( _ProcCon )
		Set Order To CONTA
		_Busca = ( _PROCCON + Iif(IsEmpty(_PROCCHE),'',_ProcChe) )
		llOk = Seek( _Busca )
		If !llOk
			lcWher = [nContas=?_ProcCon ]+;
					Iif(!IsEmpty(_PROCCHE),[And nCheques=?_ProcChe ],[])
		Endif

	Case !IsEmpty( _ProcChe )
		Set Order To NCHEQUE
		_Busca = ( _ProcChe )
		llOk = Seek( _Busca )
		If !llOk
			lcWher = [Ncheques = ?_ProcChe ]
		Endif
	Case !IsEmpty( _ProcControle )
		Set Order To Nums
		llOk=Seek( _PROCCONTROLE )
		If !llOk
			lcWher = [Nums = ?_ProcControle ]
		EndIf
		
	Case !IsEmpty( _ProcProtocolo )
		If ThisForm.Parentform.Podatamgr.CursorQuery('SigCqChm','CrChm','Protocolos',Trans(Val(_ProcProtocolo),'@L 99999999'))
			Set Order To Nums
			llOk=Seek( CrChm.Nums )
			If !llOk
				lcWher = [Nums = ']+CrChm.Nums+[']
			Endif
		Endif
	Case !IsEmpty( _ProcArquivo )
		Set Order To NUMEROS
		llOk=Seek( Trans( _ProcArquivo, '@L 999999' ) )
		If !llOk
			_ProcNumero = Trans( _ProcArquivo, '@L 999999' ) 
			lcWher = [Numeros = ?_ProcNumero ]
		Endif
EndCase
Set Near Off

If !llOk And !IsEmpty(lcWher)
	lnErro = ThisForm.ParentForm.podatamgr.SqlExecute([Select * From SigChe Where ]+lcWher,'CrCheque')
	If lnErro > 0
		Select CrCheque
		Scan
			Scatter to Memvar Memo
			Insert Into CrSigChe From Array Memvar
		EndScan
	Endif
Endif
ThisForm.Release
ENDPROC
