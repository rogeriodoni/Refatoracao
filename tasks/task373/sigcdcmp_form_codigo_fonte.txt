------------------------------------------------------------
* PROCEDURE msv_procurar
------------------------------------------------------------
PROCEDURE msv_procurar
Local lcBusca

If .Not.IsEmpty(ThisForm.Pagina.Dados.Get_codigo.Value)
	ThisForm.poDataMgr.LocateCursor('CrSigCdCmp','cods',ThisForm.Pagina.Dados.Get_codigo.Value)
EndIf


ENDPROC

------------------------------------------------------------
* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
LPARAMETERS lcacesso

With ThisForm
	.poDatamgr = CreateObject('fSqlConector')
	With .poDatamgr
		.AddCursor('SigCdCmp','cods','CrSigCdCmp','',ThisForm.pagina.lista.grade)
		.pfSqlTabela(1).pColuna('cods','','','Código',90,.t.)
		.pfSqlTabela(1).pColuna('descs','','','Descrição',337,.t.)
	EndWith
	= DoDefault()
	.LockScreen = .f.
EndWith
ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Lista.Grupo_op.Click
------------------------------------------------------------
PROCEDURE Pagina.Lista.Grupo_op.Click
LPARAMETERS opcao

If !dodefault( opcao )
	ThisForm.mAtivapagina1
	Return
Endif

Select CrSigCdCmp
Do Case
	Case Inlist( ThisForm.pcEscolha, 'INSERIR', 'PROCURAR' )
		ThisForm.Pagina.Dados.Get_Codigo.SetFocus
	Case ThisForm.pcEscolha = 'ALTERAR'
		ThisForm.Pagina.Dados.Get_Desc.SetFocus
Other
	ThisForm.Pagina.Dados.Grupo_Salva.Salva.SetFocus
Endcase

ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
Select CrSigCdCmp
If ThisForm.pcEscolha = 'INSERIR'
	If IsEmpty( CrSigCdCmp.Cods )
		=Messagebox( 'Código Inválido!', 48, '' )
		ThisForm.Pagina.Dados.Get_Codigo.SetFocus
		Return
	Endif
	If ThisForm.poDataMgr.ChkRegister('SigCdCmp','cods',ThisForm.Pagina.Dados.Get_Codigo.Value)
		=Messagebox( 'Código já cadastrado!', 48, '' )
		ThisForm.Pagina.Dados.Get_Codigo.SetFocus
		Return
	Endif
EndIf

Select CrSigCdCmp
If (ThisForm.pcEscolha = [INSERIR])
	Replace cIdChaves With fUniqueIds() in CrSigCdCmp
EndIf 

=DoDefault()
ThisForm.mAtivapagina1

ENDPROC

PROCEDURE Pagina.Dados.Grupo_Salva.Cancelar.Click
=DoDefault()
If ThisForm.plCancelar
	ThisForm.mAtivapagina1
Endif

ENDPROC
