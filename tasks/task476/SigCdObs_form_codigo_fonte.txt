------------------------------------------------------------
* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
LPARAMETERS lcacesso

With ThisForm
	.poDatamgr = CreateObject('fSqlConector')
	With .poDatamgr
		.AddCursor('SIGCDOBS','codigos','crSigCdObs','',ThisForm.pagina.lista.grade)
		.pfSqlTabela(1).pColuna('Codigos','','','Códigos',50,.t.)
		.pfSqlTabela(1).pColuna('Descrs','','','Descriçãos',450,.t.)
		.pfSqlTabela(1).pColuna("Iif(CrSigCdObs.comiss=2, 'NÃO', 'SIM')",'','','Pg.Comissão',90,.t.)
		.pfSqlTabela(1).pColuna('Grupos','','','Grupo',80,.t.)
		.pfSqlTabela(1).pColuna('Agrupas','','','Agrupamento',87,.t.)
	EndWith
	= DoDefault()
	
	.LockScreen = .f.
	
	.podatamgr.sqlexecute([Select Dopes, Cmoes, nFiscals, Abrevs From SigCdOpe Order by Dopes],'nivel')
EndWith

ENDPROC

------------------------------------------------------------
* PROCEDURE msv_procurar
------------------------------------------------------------
PROCEDURE msv_procurar
Local lcBusca

If !IsEmpty(ThisForm.Pagina.Dados.get_codigos.Value)
	ThisForm.poDataMgr.LocateCursor('crSigCdObs','codigos',ThisForm.Pagina.Dados.Get_codigos.Value)
EndIf

ENDPROC
