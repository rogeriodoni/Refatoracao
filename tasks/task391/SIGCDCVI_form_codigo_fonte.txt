Private pMsI, pMsF, pAnI, pAnF
Set Step On 
With ThisForm
	.poDataMgr = CreateObject('fSqlConector', ThisForm.Name)

			.pfSqlTabela(1).pColuna('Adics',   '999.99', '', 'Adicional', 080, .t.)
			.pfSqlTabela(1).pColuna('Minimos', '999,999,999.99', '', 'Mínimo', 108)

		.Pagina.Lista.cntFiltro.getMess.Value = pMsI
		.Pagina.Lista.cntFiltro.getAnos.Value = pAnI

		=MessageBox('Obrigatório Informar o Mes de Referência!!!', 48, '')
		loDad.getMess.SetFocus
		Return 0
	EndIf

	If Not (Int(Val(loDad.getMess.Value)) > 0)
		=MessageBox('Mês de Referência Inválido!!!', 48, '')
		loDad.getMess.SetFocus
		Return 0
	EndIf

	If Empty(loDad.getAnos.Value)
		=MessageBox('Obrigatório Informar o Ano de Referência!!!', 48, '')
		loDad.getAnos.SetFocus
		Return 0
	EndIf

	If Not Between(Int(Val(loDad.getAnos.Value)), 1, 9999)
		=MessageBox('Ano de Referência Inválido!!!', 48, '')
		loDad.getAnos.SetFocus
		Return 0
	EndIf

	If Empty(loDad.getUsuars.Value)
		=MessageBox('Obrigatório Informar o Usuário!!!', 48, '')
		loDad.getUsuars.SetFocus
		Return 0
	EndIf

	If Empty(loDad.getIndivs.Value)
		=MessageBox('Obrigatório Informar o Status de Tipo de Vendedor!!!', 48, '')
		loDad.getIndivs.SetFocus
		Return 0
	EndIf

	If Not Inlist(loDad.getIndivs.Value, [I], [R], [S])
		=MessageBox('Observe a Informação na Tela ao Lado do Tipo de Vendedor!!!' + Chr(13) + ;
					'Você Está Tentando Usar Um Tipo de Vendedor Que Não Existe!!! ', 48, 'Atenção!!!')
		loDad.getIndivs.SetFocus
		Return 0
	EndIf

	If Empty(loDad.getMinimos.Value)
		=MessageBox('Obrigatório Informar o Valor Mínimo!!!', 48, '')
		loDad.getMinimos.SetFocus
		Return 0
	EndIf
	
*********RETIRADO A PEDIDO DO CF 
*!*		ThisForm.poDataMgr.CursorQuery('SigCcCmV', 'LocalCComV', 'Usuars', loDad.getUsuars.Value, ;
*!*									   'Usuars, Mess, Anos, Indivs, cIdChaves')
*!*		Select LocalCComV
*!*		Scan
*!*			If ((LocalCComV.Mess = loDad.getMess.Value) And ;
*!*				(LocalCComV.Anos = loDad.getAnos.Value) And ;
*!*				(LocalCComV.Indivs = loDad.getIndivs.Value) And ;
*!*				(LocalCComV.cIdChaves <> crSigCcCmV.cIdChaves))
*!*				=MessageBox('Usuário Já Cadastrado No Período!!!', 48, '')
*!*				loDad.getUsuars.SetFocus
*!*				Return 0
*!*			EndIf
*!*		EndScan

	ThisForm.poDataMgr.CursorQuery('SigCcCmI', 'LocalCComI', 'Usuars', loDad.getUsuars.Value, 'Usuars, Mess, Anos, cIdChaves')
	Select LocalCComI
	Scan
		If ((LocalCComI.Mess = loDad.getMess.Value) And ;
			(LocalCComI.Anos = loDad.getAnos.Value))
			=MessageBox('Usuário Já Cadastrado No Período Como Vendedor Por Cargo!!!', 48, '')
			loDad.getUsuars.SetFocus
			Return 0
		EndIf
	EndScan

	* Checando Campos Obrigatórios ao Vendedor Individual
	If (loDad.getIndivs.Value = 'I') And Not Empty(loDad.getAdics.Value)
		=MessageBox('O Vendedor Não Pode Ser Individual e Receber Adicional!!!', 48, '')
		loDad.getSetors.SetFocus
		Return 0
	EndIf

	If (loDad.getIndivs.Value = 'S') And Not Empty(loDad.getAdics.Value)
		=MessageBox('O Vendedor Não Pode Ser de Setor e Receber Adicional!!!', 48, '')
		loDad.getSetors.SetFocus
		Return 0
	EndIf

	* Checando Campos Obrigatórios ao Responsável
	If (loDad.getIndivs.Value = 'R') And Not Empty(loDad.getSupervs.Value)
		=MessageBox('O Responsável Não Pode Ter Um Responsável Cadastrado!!!', 48, '')
		loDad.getSupervs.SetFocus
		Return 0
	EndIf

	If Not Empty(loDad.getSetors.Value)
		lcQuery = [Select Usuars ] + ;
				    [From SigCcCmV ] + ;
				   [Where Supervs = '] + Padr(crSigCcCmV.Usuars, 10) + [' And ] + ;
				   		 [Mess = '] + loDad.getMess.Value + [' And ] + ;
				   		 [Anos = '] + loDad.getAnos.Value + [' And ] + ;
				   		 [Not cIdChaves = '] + crSigCcCmV.cIdChaves  + [']

		If (ThisForm.poDataMgr.SqlExecute(lcQuery, 'LocalCComV') < 1)
			=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (LocalCComV)')
			Return 0
		EndIf
		Go Top In LocalCComV
		If Not Eof('LocalCComV')
			=MessageBox('Usuário Cadastrado Como Vendedor Responsável!!!', 48, '')
			loDad.getSetors.SetFocus
			Return 0
		EndIf
	EndIf

	* Checando Campos Obrigatórios ao Vendedor Responsável
	If (loDad.getIndivs.Value = 'R') And Empty(loDad.getSetors.Value) And Empty(loDad.getAdics.Value)
		=MessageBox('O Vendedor Responsável Tem Que Ter Um Adicional Cadastrado!!!', 48, '')
		loDad.getSupervs.SetFocus
		Return 0
	EndIf
EndIf

=DoDefault()

ThisForm.mAtivaPagina1

------------------------------------------------------------
* PROCEDURE Click
------------------------------------------------------------
PROCEDURE Click
Local lcThisRec, lcNewPros, lcLocPros, lCreated

lcOMess = Padl(Alltrim(This.Parent.getOMess.Value), 2, '0')
lcDMess = Padl(Alltrim(This.Parent.getDMess.Value), 2, '0')
lcOAnos = Padl(Alltrim(This.Parent.getOAnos.Value), 4, '0')
lcDAnos = Padl(Alltrim(This.Parent.getDAnos.Value), 4, '0')
lcChave = crSigCcCmV.cIdChaves

If Empty(lcOMess)
	=MessageBox('Obrigatório Informar o Mes de Referência da Origem!!!', 48, '')
	This.Parent.getOMess.SetFocus
	Return 0
EndIf

If Not (Int(Val(lcOMess)) > 0)
	=MessageBox('Mês de Referência da Origem Inválido!!!', 48, '')
	This.Parent.getOMess.SetFocus
	Return 0
EndIf

If Empty(lcOAnos)
	=MessageBox('Obrigatório Informar o Ano de Referência da Origem!!!', 48, '')
	This.Parent.getOAnos.SetFocus
	Return 0
EndIf

If Not Between(Int(Val(lcOAnos)), 1, 9999)
	=MessageBox('Ano de Referência da Origem Inválido!!!', 48, '')
	This.Parent.getOAnos.SetFocus
	Return 0
EndIf

If Empty(lcDMess)
	=MessageBox('Obrigatório Informar o Mes de Referência do Destino!!!', 48, '')
	This.Parent.getDMess.SetFocus
	Return 0
EndIf

If Not (Int(Val(lcDMess)) > 0)
	=MessageBox('Mês de Referência do Destino Inválido!!!', 48, '')
	This.Parent.getDMess.SetFocus
	Return 0
EndIf

If Empty(lcDAnos)
	=MessageBox('Obrigatório Informar o Ano de Referência do Destino!!!', 48, '')
	This.Parent.getDAnos.SetFocus
	Return 0
EndIf

If Not Between(Int(Val(lcDAnos)), 1, 9999)
	=MessageBox('Ano de Referência do Destino Inválido!!!', 48, '')
	This.Parent.getDAnos.SetFocus
	Return 0
EndIf

If (lcOMess = lcDMess) And (lcOAnos = lcDAnos)
	=MessageBox('Período de Referência da Origem e do Destino Não Podem Ser Iguais!!!', 48, '')
	This.Parent.getOMess.SetFocus
	Return 0
EndIf

* Apanha Todos os Registros No Mes/Ano de Origem
lcQuery = [Select * ] + ;
		    [From SigCcCmV ] + ;
		   [Where Mess = '] + lcOMess + [' And ] + ;
		   		 [Anos = '] + lcOAnos + [' ] + ; 
		   [Order By Usuars]

If (ThisForm.poDataMgr.SqlExecute(lcQuery, 'crOrigem') < 1)
	=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (crOrigem)')
	Return 0
EndIf

Select crOrigem
Go Top
If Eof('crOrigem')
	=MessageBox('Nenhum Vendedor Está Cadastrado no Período de Origem Informado!!!', 48, '')
	This.Parent.getOMess.SetFocus
	Return 0
EndIf

If MessageBox('Serão Gerados Registros Para Todos os Vendedores no Período ' + lcDMess + '/' + lcDAnos + Chr(13) + ;
              'Os Registros Já Cadastrados Não Serão Alterados! Confirma a Geração?', 4+32+256, ;
              'Solicitação de Confirmação...') <> 6
	Wait Window 'Operação Cancelada Pelo Usuário...' NoWait
	Return 0
EndIf

llIns = .f.
llOks = .t.
lcMsg = []

Select crOrigem
oProg = CreateObject('fwprogressbar', 'Processando Vendedores...', Reccount() + 1)
oProg.Show
Scan 
	oProg.Update(.t.)
	oProg.SubTitulo.Caption = 'Processando o Vendedor : ' + crOrigem.Usuars

	* Procura a Condição Atual do Usuário na Folha de Pagamento
	lcQuery = [Select Funcs, Conds ] + ;
			    [From FopCff ] + ;
			   [Where Funcs = '] + crOrigem.Usuars + [' ] + ; 
			   [Order By Funcs]

	If (ThisForm.poDataFP2.SqlExecute(lcQuery, 'crFopCff') < 1)
		=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (crFopCff)')
		llOks = .f.
		Go Bottom In crOrigem
	EndIf
	If llOks
		Select crFopCff
		Go Top
		If Eof([crFopCff]) Or InList(crFopCff.Conds, [D], [Q])
			* Se o Usuário Não Está Ativo na Folha de Pagamento
			lcMsg = lcMsg + Alltrim(crFopCff.Funcs) + [(] + crFopCff.Conds + [); ]
		Else
			* Se o Usuário Está Ativo na Folha de Pagamento
			lcQuery = [Select * ] + ;
					    [From SigCcCmV ] + ;
					   [Where Usuars = '] + crOrigem.Usuars + [' And ] + ;
					   		 [Indivs = '] + crOrigem.Indivs + [' And ] + ;
					   		 [Setors = '] + crOrigem.Setors + [' And ] + ;
					   		 [Mess = '] + lcDMess + [' And ] + ;
					   		 [Anos = '] + lcDAnos + [' ] + ; 
					   [Order By Usuars]

			If (ThisForm.poDataMgr.SqlExecute(lcQuery, 'crBusca') < 1)
				=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (crBusca)')
				llOks = .f.
				Go Bottom In crOrigem
			EndIf
			If llOks
				Select crBusca
				Go Top
				If Eof('crBusca')
					Select crOrigem
					Scatter Memvar Memo
					m.cIdChaves = fUniqueIds()
					m.Mess = lcDMess
					m.Anos = lcDAnos

					Zap In crSigCcCmV

					Insert Into crSigCcCmV From Memvar
					llIns = .t.

					If Not ThisForm.poDataMgr.Update('crSigCcCmV')
						=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (Update - crSigCcCmV - ' + ;
									'Registro : ' + Alltrim(Str(Recno('crOrigem'))) + ')')
						llOks = .f.
						Go Bottom In crOrigem
					EndIf
				EndIf
			EndIf
		EndIf
	EndIf
EndScan

If llIns
	If llOks
		oProg.Update(.t.)
		oProg.SubTitulo.Caption = 'Atualizando Os Registros'

		If (ThisForm.poDataMgr.Commit() < 1)
			=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (Commit)')
			llOks = .f.
		EndIf
	EndIf

	If Not llOks
		ThisForm.poDataMgr.RollBack()
		=MessageBox('Houve Uma Falha na Gravação e Os Registros NÃO Foram Gerados!!!', 16, 'Erro!!!')
	Else
		If Not Empty(lcMsg)
			lcMsg = [Não Gerados: ] + lcMsg
			=MessageBox(lcMsg, 0+48, 'Cópia Parcialmente Concluída')
		Else
			=MessageBox('Os Registros Foram Gerados Com Sucesso!!!', 0+48, 'Cópia Concluída')
		EndIf
	EndIf
Else
	=MessageBox('Nenhum Registro Foi Gerado!!!', 0+48, 'Cópia Concluída Sem Sucesso')
EndIf
oProg.Complete

With This.Parent 
	.Visible           = .f.
	.cmdOk.Enabled     = .f.
	.cmdCancel.Enabled = .f.
	.getOMess.Enabled  = .f.
	.getDMess.Enabled  = .f.
	.getOAnos.Enabled  = .f.
	.getDAnos.Enabled  = .f.
EndWith

With ThisForm.Pagina.Lista
	.Grade.Enabled 		 = .t.
	.Grupo_Saida.Enabled = .t.
	.Grupo_op.Enabled    = .t.
	.cmdCopia.Enabled    = .f.
	.Grade.SetFocus
EndWith

ThisForm.Pagina.Dados.Enabled = .t.
This.Parent.Parent.cmdCopia.Enabled = .t.

ThisForm.mAtivaPagina1

Select crSigCcCmV
Go Top
Locate For cIdChaves = lcChave

ThisForm.Refresh
ThisForm.Pagina.Lista.Grade.SetFocus
* Fim *********************************************************************************
ENDPROC

If Not Empty(This.Value)
	lcQuery = [Select Distinct a.Usuars, b.NComps ] + ;
			    [From SigCcCmV a, SigCdUsu b ] + ;
			   [Where a.Indivs  = 'R' And ] + ;
			   		 [a.Supervs = Space(10) And ] + ;
			   		 [a.Setors  = Space(10) And ] + ;
			   		 [a.Usuars  = b.Usuarios ] + ;
			   [Order By a.Usuars]

	If (ThisForm.poDataMgr.SqlExecute(lcQuery, 'LocalSupervs') < 1)
		=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (LocalSupervs)')
		Return 0
	EndIf
	Select LocalSupervs
	Index On Usuars Tag Usuars
	Index On NComps + Usuars Tag NComps

	If Not Seek(Padr(This.Value, 10), 'LocalSupervs', 'Usuars')
		xBrow = CreateObject('fwBuscaSel', 'LocalSupervs', 'Usuars', This.Value, 'Responsável')
		xBrow.AddColuna('Usuars', '', 'Usuário', .t.)
		xBrow.AddColuna('NComps', '', 'Nome', .t.)
		xBrow.Show
	EndIf
	If (Lastkey() <> 27)
		This.Value = LocalSupervs.Usuars
	Else
		This.Value = ''
	EndIf
EndIf
This.Refresh
* Fim *********************************************************************************
ENDPROC
