							lcSql = [Select b.Nenvs,b.CidChaves,a.cpros,a.dpros from SigPdMvf b, SigOpPic a Where b.EmpDnPs = ']+lcEdn+[' ]+;
									[And b.Nenvs = ]+Str(m.nenvs)+[ And a.nops = b.nops ]
							=Thisform.Podatamgr.Sqlexecute(lcsql,'TmpOpi')
							_nVezes = 1
							Select TmpOpi
							Scan
								If _nVezes > 1
									Select dbImpressao
									Append Blank
									replace cmats  with m.cmats,;
											Nenvs  With m.nenvs,;
											Datas  With m.datas,;
											Opers  With m.Opers,;
											Ordems With 2
								EndIf

	EndIf

PROCEDURE inclui_histórico
LParameters _Arquivo, _TipoConta, _Tipo, _Valor, _Peso, _PesoFabr, _Grupoo, _Contao, _Grupod, _Contad
Local _Entrada, _Saida
Store 0 To _Entrada, _Saida, _pEntrada, _pSaida, _pfEntrada, _pfSaida

If (_Tipo = 'E')
	_Entrada  = _Valor
	_pEntrada = _Peso
	_pfEntrada = _PesoFabr
Else
	_Saida  = _Valor
	_pSaida = _Peso
	_pfSaida = _PesoFabr
EndIf

If (_Arquivo = 'Nens')
	Select LocalNens
	If (_TipoConta = 'O')
		_Grupo  = LocalNens.Grupoos
		_Conta  = LocalNens.Contaos
		_Grupo2 = LocalNens.Grupods
		_Conta2 = LocalNens.Contads
	Else
		_Grupo  = LocalNens.Grupods
		_Conta  = LocalNens.Contads
		_Grupo2 = LocalNens.Grupoos
		_Conta2 = LocalNens.Contaos
	EndIf
Else
	Select LocalEest
	_Grupo  = Iif(Type('_Grupoo')='C', _Grupoo, Space(10))
	_Conta  = Iif(Type('_Contao')='C', _Contao, Space(10))
	_Grupo2 = Iif(Type('_Grupod')='C', _Grupod, Space(10))
	_Conta2 = Iif(Type('_Contad')='C', _Contad, Space(10))
EndIf

If _Entrada = 0 And _Saida = 0 And _pEntrada = 0 And _pSaida = 0
	Return
Endif

If (_Arquivo = 'Nens')
	Insert Into TmpHist (Grupo, Conta, CMats, TpOps, Datas, QtdeEnts, QtdeSais, Dopps, Numps, cUnis, ;
						 Grupo2, Conta2, PesoEnts, PesoSais, Nenvs, Obss, Opers, PesoFabre, PesoFabrs, cUnips, Emps, nLotes) ;
				 Values (_Grupo, _Conta, LocalNensI.CMats, LocalNensI.TpOps, LocalNens.Datas, ;
				 		 _Entrada, _Saida, LocalNens.Dopps, LocalNens.Numps, LocalNensI.CUnis, ;
				 		 _Grupo2, _Conta2, _pEntrada, _pSaida, LocalNensI.Nenvs, ;
				 		 Nvl(LocalNens.Obss,''), _Tipo, _pfEntrada, _pfSaida, TmpPro.cUnips, LocalNens.Emps, LocalNens.nLotes)

	If Not Seek(LocalNens.EmpDnPs,'TmpHistC','EmpDnPs')
		Insert Into TmpHistC (EmpDnPs, Obss ) Values (LocalNens.EmpDnps,TmpHist.Obss)
	Endif
Else
	Insert Into TmpHist (Grupo, Conta, CMats, Datas, QtdeEnts, QtdeSais, Dopps, Numps, cUnis, ;
						 Grupo2, Conta2, PesoEnts, PesoSais, Obss, Opers, PesoFabre, PesoFabrs, cUniPs, Emps) ;
				 Values (_Grupo, _Conta, LocalEestI.CPros, LocalEest.Datas, _Entrada, _Saida, ;
				 		 LocalEest.Dopes, LocalEest.Numes, LocalEestI.CUnis, _Grupo2, _Conta2, ;
				 		 _pEntrada, _pSaida, Nvl(LocalEest.Obses,''), _Tipo,_pfEntrada, _pfSaida, ;
				 		 TmpPro.cuniPs, LocalEest.Emps)

	If Not Seek(LocalEest.Emps + LocalEest.Dopes + Str(LocalEest.Numes,10),'TmpHistC','EmpDnPs')
		Insert Into TmpHistC (EmpDnPs, Obss ) Values ;
					(LocalEest.Emps + LocalEest.Dopes + Str(LocalEest.Numes,10),TmpHist.Obss)
	Endif
EndIf
