------------------------------------------------------------
* PROCEDURE Click
------------------------------------------------------------
PROCEDURE Click
LOCAL _ORDEM

Declare LaCampo4(3,2)

If IsEmpty(ThisForm.Get_Venc.Value)
	Wait Window "Vencimento não pode ficar em branco!" Nowait
	ThisForm.Get_Venc.SetFocus
	Return
EndIf

llVenc = fDtoSQL(ThisForm.Get_Venc.Value)
lDtAlt = Datetime()
lcUpDate = [Update SigChe Set Vencs = ?llVenc, DtAlts = ?lDtAlt Where Nums = ']+CrSigChe.Nums+[']
lnErro   = ThisForm.ParentForm.Podatamgr.SqlExecute(lcUpDate,'')
If lnErro < 1
	MessageBox('Favor reinicializar o processo.',16,'Falha na Conexão')
	Return
Endif

SELECT CrSigChe
_OVenc = Vencs
Replace Vencs With ThisForm.Get_Venc.Value, DtAlts With Datetime()

m.Nums = Nums
m.Vencs = _OVenc
valjuro=ThisForm.Get_ValJuros.Value 
perjuro=ThisForm.Get_Juros.Value
maqusuar=SUBSTR(SYS(0),1,ATCC('#',SYS(0)))+usuar
pCidC = fUniqueIds()
lcNum    = ' '
CidChav  = ' '

ThisForm.ParentForm.Podatamgr.Requery('CrSigChea')
ThisForm.ParentForm.Podatamgr.Requery('CrSigMvCcr')
ThisForm.ParentForm.Podatamgr.Requery('CrSigClLan')

Select CrSigChea
Zap
Select CrSigMvCcr
Zap
Select CrSigClLan
Zap

Insert Into CrSigChea (nums,vencs,usuars,vjuros,pjuros,datas,CidChaves) ;
			values (m.Nums,m.Vencs,maqusuar,valjuro,perjuro,date(),fUniqueIds())

ThisForm.ParentForm.podatamgr.Update('CrSigChea')

Select crSigCdPam
Go Top
If ThisForm.Get_ValJuros.Value > 0
	pCidC = Dtos(Datetime())+Transform(fGerUniqueKey(Dtos(Datetime())),"@L 999999")+ThisForm.sigkey 
	Select crSigMvCcr
	Append Blank
	Replace nopers    with fGerUniqueKey('SigMvCcr') ;
			emps      with _empr ;
			datas     with datetime() ; 
			vencs     with DATETIME() ;
			usualts   with usuar ;
			datalts   with datetime() ;
			hists     with _empr + " JUROS PRORROG. CHEQUE " + crSigChe.Bancos + "/" + ;
							crSigChe.Agencias + "/" + crSigChe.NContas + "/" + crSigChe.NCheques ;
			opers     with "C" ;
			sopers    with "D" ;
			moedas    with Iif(Empty(CrSigCdPam.MoedaCheqs), [R], CrSigCdPam.MoedaCheqs) ;
			cotacaos  with 1 ;
			valors    with ThisForm.Get_ValJuros.Value ;
			autos     with .t. ;
			contas    with crSigCdPam.contajuros ;
			grupos    with crSigCdPam.grupojuros ;
			scontas   with crSigChe.IClis ;
			sgrupos   with crSigChe.GrClis ;
			smoedas   with Iif(Empty(CrSigCdPam.MoedaCheqs), [R], CrSigCdPam.MoedaCheqs) ;
			scotacaos with 1 ;
			svalors   with ThisForm.Get_ValJuros.Value ;
			CidChaves With pCidc ;
			Pagos     with '1' ;
			GruConMoes With crSigMvCcr.grupos + crSigMvCcr.contas + crSigMvCcr.moedas

	=fRecalculaS( CrSigMvCcr.Grupos, CrSigMvCcr.Contas, CrSigMvCcr.Datas, CrSigMvCcr.Moedas,,ThisForm.parentform.Podatamgr  )
			
	Select crSigMvCcr
	noper  = crSigMvCcr.nopers
	voper  = crSigMvCcr.vopers
	oper   = crSigMvCcr.sopers
	grupo  = crSigMvCcr.sgrupos
	conta  = crSigMvCcr.scontas
	moed   = crSigMvCcr.smoedas
	cot    = crSigMvCcr.scotacaos
	valor  = crSigMvCcr.svalors
	soper  = crSigMvCcr.opers
	sgrupo = crSigMvCcr.grupos
	sconta = crSigMvCcr.contas
	smoed  = crSigMvCcr.moedas
	scot   = crSigMvCcr.cotacaos
	svalor = crSigMvCcr.valors
	nlanc  = crSigMvCcr.nlancs
	vlanc  = crSigMvCcr.vlancs
	data   = crSigMvCcr.datas
	hist   = crSigMvCcr.hists
	pCidC = Dtos(Datetime())+Transform(fGerUniqueKey(Dtos(Datetime())),"@L 999999")+ThisForm.Sigkey 

	*** INSERIR CONTRA MOVIMEN//
	select crSigMvCcr
	append blank
	replace vopers    with voper ;
			nopers    with noper ;
			nlancs    with nlanc ;
			vlancs    with vlanc ;
			opers     with oper  ;
			grupos    with grupo ;
			contas    with conta ;
			moedas    with moed  ;
			cotacaos  with cot   ;
			valors    with valor ;
			emps      with _empr ;
			datas     with data  ;
			usualts   with usuar ;
			datalts   with date();
			sgrupos   with sgrupo;
			scontas   with sconta;
			smoedas   with smoed ;
			scotacaos with scot  ;
			svalors   with svalor;
			sopers    with soper ;
			autos     with .t.   ;
			hists     with hist  ;
			vencs     with DATETIME() ;
			CidChaves With pcidc;
			pagos     with '1';
			GruConMoes With crSigMvCcr.grupos + crSigMvCcr.contas + crSigMvCcr.moedas

	=fRecalculaS( CrSigMvCcr.Grupos, CrSigMvCcr.Contas, CrSigMvCcr.Datas, CrSigMvCcr.Moedas,ThisForm.ParentForm.podatamgr )

	ThisForm.ParentForm.podatamgr.Update('CrSigMvCcr')
	
EndIf

_Ndia = 0
If ThisForm.Parentform.podatamgr.CursorQuery('SigChea','CrChea','Nums',CrSigChe.Nums)
	Select CrChea
	Go Top
	_Ndia = crSigChe.Vencs - crChea.Vencs
Endif

If _Ndia >= crSigCdPam.DiaPrors
	laCampo4(1,1)='Grupos'
	laCampo4(1,2)=CrSigChe.GrClis
	laCampo4(2,1)='Contas'
	laCampo4(2,2)=CrSigChe.Iclis
	laCampo4(3,1)='Cocos'
	laCampo4(3,2)='05'
	If !ThisForm.parentForm.podatamgr.ChkRegister('SigClLan',@laCampo4)
		Select CrSigClLan
		Append Blank
		Replace Grupos With crSigChe.GrClis,;
				Contas With crSigChe.Iclis,;
				Cocos  With '05',;
				CidChaves With fUniqueIds(),;
				GruContas With Grupos + Contas

		If Not ThisForm.ParentForm.podatamgr.Update('CrSigClLan')
			MessageBox('Favor reinicializar o processo.',16,'Falha na Conexão')
			Return
		Endif
	Endif

EndIf

llOk = fRecalculaS( .t.,ThisForm.ParentForm.PodataMgr)
If llOk
	llOk = ThisForm.Parentform.podatamgr.commit() > 0
EndIf
If Not llOk
	ThisForm.parentform.rollback()
	MessageBox('Favor reinicializar o processo.',16,'Falha na Conexão')
Endif

ThisForm.Release

ENDPROC
