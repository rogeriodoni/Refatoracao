------------------------------------------------------------
* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
LOCAL lcQuery, lcCheCod

lcCheCod = []
If INLIST(ThisForm.pcEscolha,'ALTERAR')
	lcChecod = [ and codigo <> ']+CrSigCdVen.codigo+[' ]
ENDIF

If INLIST(ThisForm.pcEscolha,'INSERIR','ALTERAR')
	If Empty( CrSigCdVen.emps )
		Messagebox( 'Empresa não pode ficar em branco!', 48, '' ) 
		ThisForm.Pagina.Dados.Getemps.SetFocus
		Return
	Endif
	If Empty( CrSigCdVen.Codvens )
		Messagebox( 'Vendedor não pode ficar em branco!', 48, '' ) 
		ThisForm.Pagina.Dados.Getemps.SetFocus
		Return
	Endif
	If Empty( CrSigCdVen.dtini )
		Messagebox( 'Data Inicial não pode ficar em branco!', 48, '' ) 
		ThisForm.Pagina.Dados.Getdtini.SetFocus
		Return
	Endif
	If CrSigCdVen.situas <> 'ATIVO' and Empty( CrSigCdVen.dtFin )
		Messagebox( 'Data final não pode ficar em branco!', 48, '' ) 
		ThisForm.Pagina.Dados.Getdtfin.SetFocus
		Return
	Endif
	If Empty( CrSigCdVen.Situas )
		Messagebox( 'Situação não pode ficar em branco!', 48, '' ) 
		ThisForm.Pagina.Dados.Getdtfin.SetFocus
		Return
	Endif
	If !EMPTY(CrSigCdVen.dtfin) and CrSigCdVen.dtini >  CrSigCdVen.dtfin
		Messagebox( 'As datas estão incompatíveis!', 48, '' ) 
		ThisForm.Pagina.Dados.Getdtini.SetFocus
		Return
	ENDIF

	*** mena 12/03/2017 - verifica se intervalo atual já foi cadastrado para o vendedor e Situacao ***
	lcDat=SUBSTR(DTOS(TTOd(CrSigCdVen.dtini)),1,10)
	lcDatI=SUBSTR(lcDat,1,4)+'-'+SUBSTR(lcDat,5,2)+'-'+SUBSTR(lcDat,7,2)
	lcDat=SUBSTR(DTOS(TTOd(CrSigCdVen.dtfin)),1,10)
	lcDatF=SUBSTR(lcDat,1,4)+'-'+SUBSTR(lcDat,5,2)+'-'+SUBSTR(lcDat,7,2)
	IF CrSigCdVen.situas <> 'ATIVO'
		lcQuery = 	[Select * From SigCdVen where codvens = ']+CrSigCdVen.codvens+[' ]+;
					[and situas <> 'ATIVO' ]+;
					[and (']+lcDatI+[' between LEFT(CONVERT(VARCHAR, dtini, 120), 10) and  LEFT(CONVERT(VARCHAR, dtfin, 120),10) ]+; 
					[or ']+lcDatF+[' between LEFT(CONVERT(VARCHAR, dtini, 120), 10) and  LEFT(CONVERT(VARCHAR, dtfin, 120),10) ]+; 
					[or LEFT(CONVERT(VARCHAR, dtini, 120), 10) between '] + lcDatI + [' and ']+lcDatF + [' ]+;
					[or LEFT(CONVERT(VARCHAR, dtfin, 120), 10) between '] + lcDatI + [' and ']+lcDatF + [') ]+lcChecod
		If ThisForm.Podatamgr.Sqlexecute(lcQuery,'crAux1') < 1
			Messagebox('Impossivel Efetuar Conexão com Servidor de Banco de Dados.',(0+48),'Erro')
			Return
		Endif
	ELSE
		lcQuery = 	[Select * From SigCdVen where codvens = ']+CrSigCdVen.codvens+[' ]+;
					[and situas = 'ATIVO' ]+;
					[and (']+lcDatI+[' between LEFT(CONVERT(VARCHAR, dtini, 120), 10) and  LEFT(CONVERT(VARCHAR, dtfin, 120),10) ]+; 
					[or ']+lcDatF+[' between LEFT(CONVERT(VARCHAR, dtini, 120), 10) and  LEFT(CONVERT(VARCHAR, dtfin, 120),10) ]+; 
					[or LEFT(CONVERT(VARCHAR, dtini, 120), 10) between '] + lcDatI + [' and ']+lcDatF + [' ]+;
					[or LEFT(CONVERT(VARCHAR, dtfin, 120), 10) between '] + lcDatI + [' and ']+lcDatF + [' ]+;
					[or dtfin is null ) ]+lcChecod
		If ThisForm.Podatamgr.Sqlexecute(lcQuery,'crAux1') < 1
			Messagebox('Impossivel Efetuar Conexão com Servidor de Banco de Dados.',(0+48),'Erro')
			Return
		Endif
	ENDIF
	
	If RECCOUNT('crAux1')>0		&& or RECCOUNT('crAux2') >0
		Messagebox( 'Esse intervado já está cadastrado para esse vendedor!', 48, '' ) 
		ThisForm.Pagina.Dados.Getdtini.SetFocus
		Return
	Endif

	If INLIST(ThisForm.pcEscolha,'INSERIR')
		lcDat=SUBSTR(ALLTRIM(STR(YEAR(DATE()))),3)+PADL(ALLTRIM(STR(month(DATE()))),2,'0')
		replace codigo with crSigCdVen.codvens+lcDat+PADL(ALLTRIM(STR(fGerUniqueKey(crSigCdVen.codvens+lcDat))),5,'0') in crSigCdVen
	ENDIF

Endif

replace dtalts with DATETIME() in crSigCdVen
replace usualts with Usuar in crSigCdVen

=DoDefault()
ThisForm.mAtivapagina1

ENDPROC
