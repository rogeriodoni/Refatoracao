* PROCEDURE baixabarra

PROCEDURE baixabarra
Private pBar, pDat

Local lcDop, lcEmp, lnNum, llTaOk

Select CrSigMvCab
Zap
Select CrSigMvHst
Zap

pBar = crSigOpEtq.Cbars
pDat = Datetime()

lcDop = Padr('BAIXA ETIQUETA', 20)
lcEmp = crSigOpEtq.Empos
lnNum = fGerUniqueKey(Alltrim(lcDop) + lcEmp)

Insert Into crSigMvCab (Emps, Dopes, Numes, MascNum, GrupoOs, ContaOs, Datas, Usuars, EmpDopNums, cIdChaves, DtAlts) ;
			   Values (lcEmp, lcDop, lnNum, fGerMascara(lnNum), crSigOpEtq.Grupos, crSigOpEtq.Contas, ;
			   		   pDat, Usuar, lcEmp + lcDop + Str(lnNum,6), fUniqueIds(), pDat)

pCidC = Dtos(pDat)+'S'+Transform(fGerUniqueKey(Dtos(pDat)),"@L 999999")+ThisForm.Sigkey 
pSeq  = fGerUniqueKey('HISTBAR')
Insert Into crSigMvHst (CodBarras, Usuars, Opers, Cpros, Datas, Datars, Dopes, Empos, ;
					  Emps, Grupos, Estos, Numes, Cidchaves, EmpDopNums, EmpGruEsts, DtAlts, OriDopNums, Seqs) ;
			  Values (crSigOpEtq.Cbars, Usuar, 'S', crSigOpEtq.Cpros, pDat, pDat, lcDop, crSigOpEtq.Empos, crSigOpEtq.Empos, ;
			  		  crSigOpEtq.Grupos, crSigOpEtq.Contas, lnNum, pCidC, crSigOpEtq.Empos + lcDop + Str(lnNum,6), ;
			  		  crSigOpEtq.Empos + crSigOpEtq.Grupos + crSigOpEtq.Contas, pDat, CrSigOpEtq.Empos + lcDop + Str(lnNum,6), pSeq)

=fRecalculaP(crSigMvHst.Emps, crSigMvHst.Grupos, crSigMvHst.Estos, crSigMvHst.CPros, crSigMvHst.Datas, CrSigMvHst.Codcors, CrSigMvHst.CodTams, ThisForm.Podatamgr )
=fRecalculaC(crSigMvHst.Emps, crSigMvHst.CPros, crSigMvHst.Datas, ThisForm.Podatamgr )

lcQuery = [Update SigOpEtq ] + ;
		     [Set Grupos = '', ] + ;
		     	 [Contas = '', ] + ;
		     	 [DtMovs = ?pDat, ] + ;
		     	 [Usuars = '] + Usuar + [' ] + ;
		   [Where CBars = ?pBar]

llTaOk = .t.
If (ThisForm.poDataMgr.SqlExecute(lcQuery, '') < 1)
	=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (Update - crSigOpEtq)')
	llTaOk = .f.
EndIf

If llTaOk And Not ThisForm.poDataMgr.Update('crSigMvCab')
	=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (Update - crSigMvCab)')
	llTaOk = .f.
EndIf

If llTaOk And Not ThisForm.poDataMgr.Update('crSigMvHst')
	=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (Update - crSigMvHst)')
	llTaOk = .f.
EndIf
If llTaOk And Not fRecalculaP(.t., Thisform.Podatamgr )
	=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (Saldo Produto)')
	llTaOk = .f.
EndIf
If Not fRecalculaC(.t., .f., .f., ThisForm.poDataMgr)
	=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (Custo Produto)')
	llTaOk = .f.
EndIf
If llTaOk
	If (ThisForm.poDataMgr.Commit() < 1)
		=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (Commit)')
		llTaOk = .f.
	Endif
EndIf
If Not llTaOk
	ThisForm.Podatamgr.rollback()
Endif

With ThisForm
	.get_Codigo.Value  = ''
	.Get_Grupo.Value   = ''
	.Get_Conta.Value   = ''
	.Get_Produto.Value = ''
	.Refresh
Endwith
* Fim **************************************************************************************

ENDPROC

PROCEDURE Init
Private pIds
ThisForm.cntSombra.lblSombra.Caption = ThisForm.Caption
ThisForm.cntSombra.lblTitulo.Caption = ThisForm.Caption
=DoDefault()

pIds = Replicate(Chr(254), 20)

lcQryEest = [Select * From SigMvCab Where cIdChaves = '] + pIds + [']
lcQryHis  = [Select * From SigMvHst  Where cIdChaves = '] + pIds + [']

With ThisForm
	.poDataMgr = CreateObject('fSqlConector', ThisForm.Name)

	If (.poDataMgr.pnIdconn > 0)
		With .poDataMgr
			.AddCursor('SigMvCab', 'cIdChaves', 'crSigMvCab', '', '', lcQryEest)
			.AddCursor('SigMvHst',  'cIdChaves', 'crSigMvHst',  '', '', lcQryHis)
		EndWith
		
		ThisForm.poDataMgr.CursorQuery('SigCdPac', 'crSigCdPac')
		ThisForm.SigKey = CrSigCdPac.SigKeys

		ThisForm.get_Codigo.Value = Space(14)
		ThisForm.Refresh
	Else
		=Messagebox('Impossivel Efetuar Conexão com Servidor de Banco de Dados!!!', 0+48, 'Erro')
		Return(.f.)
	EndIf
EndWith
* Fim **************************************************************************************

ENDPROC

PROCEDURE Valid
With ThisForm
	.Get_Grupo.Value   = ' '
	.Get_Conta.Value   = ' '
	.Get_Produto.Value = ' '
	.Get_Dpro.Value    = ' '
	.Get_Qtde.Value    = 0
	.Get_peso.Value    = 0
	.Say5.Caption      = ' '
	.Say6.Caption      = ' '
	.Refresh
Endwith

lnBar = Iif(Type('This.Value')='C', Int(Val(This.Value)), This.Value)

If (lnBar > 0)
	ThisForm.poDataMgr.CursorQuery('SigOpEtq', 'crSigOpEtq', 'CBars', lnBar)

	If Eof('crSigOpEtq')
		=Messagebox('O Código [' + Padl(lnBar, 14) + '] ' + ;
					'Não Localizado... ' + Chr(13) + Chr(13) + Space(10) + ;
					'Verifique ou Utilize Outro Código!!!', 48 , 'Código Não Localizado' )
		This.Value = Space(8)
		This.Refresh
		Return 0
	EndIf
	
	If Empty(crSigOpEtq.Grupos) And Empty(crSigOpEtq.Contas)
		=Messagebox('O Código [' + Padl(lnBar, 8) + '] ' + ;
					'Já Foi Baixado... ' + Chr(13) + Chr(13) + Space(10) + ;
					'Verifique ou Utilize Outro Código!!!', 48 , 'Código Já Baixado')
		This.Value = Space(8)
		This.Refresh
		Return 0
	Endif
	Set Confirm Off
	
	ThisForm.poDataMgr.CursorQuery('SigCdPro', 'LocalPro', 'Cpros', CrSigOpEtq.cpros,[ Dpros, cUnis, cUniPs ])
	
	With ThisForm
		.Get_Grupo.Value   = crSigOpEtq.Grupos
		.Get_Conta.Value   = crSigOpEtq.Contas
		.Get_Produto.Value = crSigOpEtq.CPros
		.Get_dpro.Value    = LocalPro.Dpros
		.Get_Qtde.Value    = CrSigOpEtq.Qtds
		.Get_Peso.Value    = CrSigOpEtq.Pesos
		.Say5.Caption      = '( '+LocalPro.Cunis+')'
		.Say6.Caption      = '( '+LocalPro.Cunips+')'
		.Refresh
	Endwith
EndIf
* Fim **************************************************************************************

ENDPROC
