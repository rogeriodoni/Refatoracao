PROCEDURE msv_procurar
Local lcBusca

If .Not.IsEmpty(ThisForm.Pagina.Dados.GetExpressao.Value)
	ThisForm.poDataMgr.LocateCursor('crSigCdDic','Expressao',ThisForm.Pagina.Dados.GetExpressao.Value)
EndIf

ENDPROC

------------------------------------------------------------
* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
LPARAMETERS lcacesso

With ThisForm
	.poDatamgr = CreateObject('fSqlConector')
	With .poDatamgr
		.AddCursor('SigCdDic','cIdChaves','crSigCdDic','',ThisForm.pagina.lista.grade)
		.pfSqlTabela(1).pColuna('Idioma','','','Idioma',70,.t.)
		.pfSqlTabela(1).pColuna('Expressao','','','Expressão',280,.t.)
		.pfSqlTabela(1).pColuna('Traducao','','','Tradução',280,.t.)
	EndWith
	= DoDefault()
	.LockScreen = .f.
EndWith

ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Lista.Grupo_op.Click
------------------------------------------------------------
PROCEDURE Pagina.Lista.Grupo_op.Click
parameters lcOpcao

If !dodefault( lcopcao )
	ThisForm.mAtivapagina1
	Return
Endif

do case
case Thisform.pcEscolha='INSERIR' or ;
	Thisform.pcEscolha='PROCURAR'
	Thisform.Pagina.Dados.GetExpressao.setfocus
case Thisform.pcEscolha='ALTERAR'
	Thisform.pagina.dados.GetTraducao.setfocus
case Thisform.pcEscolha='CONSULTAR' or ;
	Thisform.pcEscolha='EXCLUIR'
	Thisform.pagina.dados.grupo_salva.salva.setfocus
endcase
	

ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
if inlist(thisform.pcEscolha,'INSERIR','ALTERAR')
    if IsEmpty(thisform.Pagina.Dados.GetExpressao.value)
		Messagebox('Expressão Inválida !!',0+48,'')
		Thisform.Pagina.Dados.GetExpressao.setfocus
		Return 	
	endif	
    if IsEmpty(thisform.Pagina.Dados.GetIdioma.value)
		Messagebox('Idioma Inválido !!',0+48,'')
		Thisform.Pagina.Dados.getiDescs.setfocus
		Return 	
	endif	
    if IsEmpty(thisform.Pagina.Dados.GetTraducao.value)
		Messagebox('Tradução Inválida !!',0+48,'')
		Thisform.Pagina.Dados.GetTraducao.setfocus
		Return 	
	endif	
endif

If inlist(thisform.pcEscolha,'INSERIR')
	lcQuery = [Select * From SigCdDic Where Expressao = '] + ThisForm.Pagina.Dados.GetExpressao.Value + [' ] + ;
	                                 [And Idioma    = '] + ThisForm.Pagina.Dados.GetIdioma.Value + [']
    ThisForm.poDataMgr.SqlExecute(lcQuery, 'csSigCdDic')
    If ! Eof()
		Messagebox('Código já Cadastrado !!',0+48,'')
		Thisform.Pagina.Dados.GetExpressao.setfocus
		Return
	Endif
	Select crSigCdDic
	Replace cIdChaves With fUniqueIds()
endif			

DoDefault()
Thisform.mAtivapagina1

ENDPROC
