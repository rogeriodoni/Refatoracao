* PROCEDURE calcularateioauto
------------------------------------------------------------
PROCEDURE calcularateioauto
Select csSigCdRci
lnReg = Recno()
Go Top
Count For Marcas = 1 To lnQtde
If lnQtde > 0
	lnRat = Round(100/lnQtde,2)
	Select csSigCdRci
	Scan For Marcas = 1
		Replace PercRats With lnRat In csSigCdRci
	Endscan
EndIf
Select csSigCdRci
Sum PercRats To lnPTotal

If lnPTotal > 100
	Select csSigCdRci
	Locate for Marcas = 1
	replace PercRats With PercRats - (lnPTotal - 100) in csSigCdRci
	lnPTotal = 0
	Sum PercRats To lnPTotal
EndIf 

If Not Empty(lnPTotal) And lnPTotal < 100
	Select csSigCdRci
	Locate for Marcas = 1
	replace PercRats With PercRats + ((lnPTotal - 100)*-1) in csSigCdRci
	lnPTotal = 0
	Sum PercRats To lnPTotal
EndIf 

Go lnReg
Thisform.pagina.dados.getPTotal.Value = lnPTotal
Thisform.pagina.dados.getPTotal.Refresh()

ENDPROC

* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click

lcDescRat = thisform.pagina.dados.getDesc.value

If Inlist( ThisForm.pcEscolha,[INSERIR],[ALTERAR] )
	If Inlist( ThisForm.pcEscolha,[INSERIR]) 
		If Empty(lcDescRat)
			Messagebox([Descrição de rateio obrigatório.],64,[Aviso])
			thisform.pagina.dados.getDesc.Setfocus()
			Return .f.
		Endif
		If ThisForm.podatamgr.ChkRegister('SigCdRcc', 'DescRats', lcDescRat)
			Messagebox([Descrição de rateio já cadastrada.],64,[Aviso])
			thisform.pagina.dados.getDesc.Setfocus()
			Return .f.
		Endif
		
		Replace CidChaves With fUniqueIds(), Datas With Datetime() in crSigCdRcc
	Endif

	Select crSigCdRci
	Go Top
	
	Select Sum(PercRats) as TotRats From csSigCdRci Where Marcas = 1 Into Cursor TmpItens
	If Nvl(TmpItens.TotRats,0) <> 100
		Messagebox([Total de rateio diferente de 100%.],48,[Lançamento incorreto])
		Return .f.
	Endif		
	Use In TmpItens
	
	Select csSigCdRci
	Scan For Marcas = 1
		Scatter Memvar
		
		Select crSigCdRci
		Append Blank 
		
		Gather Memvar 
		Select csSigCdRci
	EndScan 
	
	Select crSigCdRci
	Go Top
	Delete For (Empty(Emps) And Empty(Grupos) And Empty(Contas)) And PercRats = 0
	Go Top
	
	Replace DescRats With crSigCdRcc.DescRats For Empty(DescRats)
	Replace CidChaves With fUniqueIds() For Empty(CidChaves)

	If Inlist( ThisForm.pcEscolha,[ALTERAR])
		Replace DtAlts With Datetime() in crSigCdRcc
	Endif
Endif

DoDefault()
Thisform.mAtivapagina1
ENDPROC
