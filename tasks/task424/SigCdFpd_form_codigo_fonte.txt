PROCEDURE mctrlbotoes
LParameters plHabilita

Local lnI

If Type('plHabilita') # 'L'
	plHabilita = .F.
Endif

With Thisform.ParentForm.Pagina.Lista
	For lnI=1 to .Grupo_Op.ButtonCount
		If .Grupo_Op.Buttons(lnI).BaseClass = 'Commandbutton' And .Grupo_Op.Buttons(lnI).Visible
			.Grupo_Op.Buttons(lnI).Enabled = plHabilita
		EndIf
	EndFor
	.cmdFPOper.Enabled = plHabilita
	.Grupo_Saida.Sair.Enabled = plHabilita
EndWith



ENDPROC

PROCEDURE Init
LParameters poForm, pnDataSes

ThisForm.cntSombra.lblSombra.Caption = ThisForm.Caption
ThisForm.cntSombra.lblTitulo.Caption = ThisForm.Caption

Set Datasession To pnDataSes

With ThisForm
	With .poDatamgr
		.AddCursor('SigOpFpD','CidChaves','crSigOpFpD')
	EndWith
EndWith

lcSql = [Select * From SigOpFpD Where TabDs = ']+CrSigOpTdz.Codigos+[']
ThisForm.Podatamgr.SqlExecute(lcSql,'LocalFpagD')

lcSql = [Select Fpags,Descrs From SigOpFp ]
ThisForm.Podatamgr.SqlExecute(lcSql,'LocalFpag')

Select Localfpag
Index On fpags tag fpags

Select CrSigOpFpD

	.ColumnCount = 2
	.RecordSource = 'crSigOpFpD'
	.DeleteMark = .f.
	.ReadOnly = .f.

	.AllowRowSizing = .f.

	.Column1.ControlSource = 'crSigOpFpD.Fpags'

	.Column1.Movable = .f.
	.Column1.Resizable = .f.
	.Column1.ReadOnly = .f.

	.Column2.ControlSource = 'crSigOpFpD.nParcs'
	.Column2.FontName = 'Courier New'
	.Column2.Movable = .f.
	.Column2.Resizable = .f.
	.Column2.ReadOnly = .f.

------------------------------------------------------------
* PROCEDURE Click
------------------------------------------------------------
PROCEDURE Click
Local m.lnErro1

If Thisform.cmdInserir.Visible And (ThisForm.Inseriu Or Thisform.Excluiu )
	Select crSigOpFpD
	Locate For .f.
	Delete For Empty(Fpags)
	Delete For Empty(nParcs)
	Update CrSigOpFpD set CidChaves = fUniqueIds()

	Select Fpags,nParcs,Sum(1) as TT From CrSigOpFpD Group by 1,2 having Sum(1) > 1;
	Into Cursor CsSelecao ReadWrite
	If Reccount('CsSelecao') > 0
		MessageBox('Dados em Duplicidade!!!',0+16,'')
		Return .f.
	Endif

	m.lnErro1 = ThisForm.poDatamgr.pnIdConn

	With ThisForm

		lcSql = [Delete From SigOpFpD Where TabDs = ']+CrSigOpTdz.Codigos+[']
		If .Podatamgr.SqlExecute(lcSql,'') < 1
			MessageBox('Favor reinicializar o processo.',16,'Falha na Conexão (1)')
			.poDataMgr.RollBack()
			Return(.F.)
		EndIf

		If m.lnErro1>0 
			m.lnErro1 = Iif(.poDataMgr.UpDate('crSigOpFpD'),1,-1)
		EndIf 
		If m.lnErro1>0 
			m.lnErro1 = .poDataMgr.Commit()
		EndIf 
		If Not m.lnErro1>0 
			.poDataMgr.RollBack()
		EndIf 
	EndWith 
	If Not m.lnErro1>0 
		= MessageBox('Erro na Gravação dos Dados!!!',48)
	EndIf 

EndIf

Thisform.ParentForm.Enabled = .T.
Thisform.mCtrlBotoes(.T.)
ThisForm.Release
ENDPROC
