PROCEDURE impressao
parameters pcTipo

lcCodigo = crSigCdRom.Codigo
lcQuery = [Select a.*, IsNull(c.razaos,'') as razaos, IsNull(d.rclis,'') as rclis, IsNull(d.cpfs,'') as cpfs, b.chave, ] + ;
	[convert(varchar(4),replicate('0',4-len(ROW_NUMBER() OVER(ORDER BY b.chave ASC)))+convert(varchar(3),ROW_NUMBER() OVER(ORDER BY b.chave ASC))) AS Ids, ] + ; 
	[isnull(e.empdopnums,'') as empdopnums, isnull(f.pedidoweb,'') as pedidoweb, isnull(f.cdrastreio,'') as cdrastreio, isnull(f.notas,'') as notas, ] + ; 
	[isnull(e.series,'') as series, isnull(g.rclis,'') as nome, isnull(h.descricaos,'') as dDepOrig, isnull(i.descricaos,'') as dDepDest ] + ;
	[from sigcdrom a ] + ;   
	[join sigcdroi b on a.codigo = b.codigo ] + ;   
	[left join sigprtrp c on a.codtransp = c.itras ] + ;  
	[left join sigcdcli d on a.contas = d.iclis ] + ; 
	[left join sigmvnfi e on b.chave = e.nfechv ] + ;
	[left join sigmvcab f on e.empdopnums = f.empdopnums ] + ;
	[left join sigcdcli g on f.contads = g.iclis ] + ;
	[left join sigcddpt h on a.deporig = h.codigos ] + ;
	[left join sigcddpt i on a.depDest = i.codigos ] + ;
	[Where a.codigo = ']+lcCodigo+[' ]
	
Thisform.podatamgr.SqlExecute(lcQuery,[csRel])

Select csRel
Index on Ids Tag Ids
Go Top

If pcTipo = 'V'
	Report Form SIGCDROM Preview NoConsole
Else
	Report Form SIGCDROM To Printer Prompt NoConsole
EndIf 
ENDPROC

PROCEDURE Init
LParameters pTipo

If Substr(Alltrim(pTipo),1,1) = 'R'
	This.Caption = "Romaneio Recebimento XML"
	ThisForm.Name = [SIGCDROMR]
	ThisForm.Pagina.Dados.cntRec.Visible = .T.
	ThisForm.Pagina.Dados.cntRom.Visible = .F.
Else
	This.Caption = "Romaneio XML"
	ThisForm.Name = [SIGCDROM]
	ThisForm.Pagina.Dados.cntRec.Visible = .F.
	ThisForm.Pagina.Dados.cntRom.Visible = .T.
Endif

ThisForm.Busca=1
ThisForm.Tiporomaneio = Substr(Alltrim(pTipo),1,1)

With Thisform
	.poDatamgr = Createobject('fSqlConector',Thisform.Name)

	ThisForm.Pagina.Lista.Cnt_periodo.Dt_inicial.Value = Date()
	ThisForm.Pagina.Lista.Cnt_periodo.Dt_Final.Value = Date()
	
	llDatai = ThisForm.Pagina.Lista.Cnt_periodo.Dt_inicial.Value
	llDataf = ThisForm.Pagina.Lista.Cnt_periodo.Dt_final.Value
	llDatai = fDtoSQL(llDatai)
	llDataf = fDtoSQL(llDataf,'23:59:59')
	lcTipo  = ThisForm.Tiporomaneio
	
	lcQryRom = [Select a.*, IsNull(b.razaos,'') as Razaos, IsNull(c.rclis,'') as RClis, ] + ;
		[isnull(h.descricaos,'') as dDepOrig, isnull(i.descricaos,'') as dDepDest, ] + ;
		[Left(a.Obs,20) as ObsRed, case when a.nAceite = 1 then 'S' else 'N' end as aceite ] + ;
		[from sigcdrom a ] + ;
		[left join sigprtrp b on a.codtransp = b.itras ] + ;
		[Left Join SigCdCli c on a.Contas = c.iclis ] + ;
		[left join sigcddpt h on a.deporig = h.codigos ] + ;
		[left join sigcddpt i on a.depDest = i.codigos ] + ;
		[Where a.Datas Between ?llDatai And ?llDataf ] + ;
		[And a.Tipo = ']+lcTipo+[' ]
		
	If Not .poDatamgr.pnIdconn > 0
		Messagebox('Impossivel Efetuar Conexão com Servidor de Banco de Dados.',(0+48),'Erro')
		Return(.F.)
	Endif
		
	lcCodigo = Space(10)
	lcSqlCdCri = [Select * From SigCdRoI Where Codigo = ?lcCodigo]

	With .poDatamgr
		.AddCursor('SigCdRom','cidchaves','crSigCdRom','',Thisform.pagina.lista.grade, lcQryRom)
		.pfSqlTabela(1).pColuna('Codigo','','','Código',60,.T.)
		If ThisForm.Tiporomaneio = 'R'
			.pfSqlTabela(1).pColuna('DepOrig','','','Origem',100,.T.)
			.pfSqlTabela(1).pColuna('dDepOrig','','','Descrição',150,.T.)
			.pfSqlTabela(1).pColuna('DepDest','','','Destino',100,.T.)
			.pfSqlTabela(1).pColuna('dDepDest','','','Descrição',150,.T.)
		Else
			.pfSqlTabela(1).pColuna('CodTransp','','','Transportadora.',100,.T.)
			.pfSqlTabela(1).pColuna('razaos','','','Nome',150,.T.)
		EndIf 
		.pfSqlTabela(1).pColuna('Datas','','','Data',73,.T.)
		If ThisForm.Tiporomaneio = 'R'
			.pfSqlTabela(1).pColuna('ObsRed','','','Obs',150,.T.)
			.pfSqlTabela(1).pColuna('aceite','','','Aceite',60,.T.)
		EndIf 
		.pfSqlTabela(1).pColuna('Contas','','','Conta',80,.T.)
		.pfSqlTabela(1).pColuna('RClis','','','Nome',150,.T.)
		.pfSqlTabela(1).pColuna('Valor','','','Valor',80,.T.)
		.pfSqlTabela(1).pColuna('Usuarios','','','Usuario',80,.T.)
		.pfSqlTabela(1).pColuna('dataact','','','Aceite',75,.T.)

		.AddCursor('SigCdRoI','CidChaves','crSigCdRoI','','',lcSqlCdCri)
	Endwith

Endwith

ThisForm.Pagina.Lista.Aceite.Visible = fChecaAcesso(ThisForm.Name,'ACEITE')
ThisForm.Pagina.Lista.Aceite.Enabled = fChecaAcesso(ThisForm.Name,'ACEITE')

If Inlist( ThisForm.pcEscolha,[INSERIR],[ALTERAR] )
	If Inlist( ThisForm.pcEscolha,[INSERIR]) 
		If Empty(lcCodigo)
			Messagebox([Código obrigatório.],64,[Aviso])

			Return .f.
		EndIf
		
		If Empty(This.parent.parent.GetDatas.Value)
			Messagebox([Data Obrigatória.],64,[Aviso])
			thisform.pagina.dados.GetDatas.Setfocus()
			Return .f.

			If Empty(This.parent.parent.cntRec.getDDepOrig.Value)
				Messagebox([Origem Obrigatória.],64,[Aviso])
				This.parent.parent.cntRec.getDDepOrig.Setfocus()

			EndIf

				Messagebox([Destino Obrigatório.],64,[Aviso])
				This.parent.parent.cntRec.getDDepDest.Setfocus()
				Return .f.
			EndIf
		Else
			
			If Empty(This.parent.parent.cntRom.GetCodTransp.Value)
				Messagebox([Transportadora Obrigatória.],64,[Aviso])
				This.parent.parent.cntRom.GetCodTransp.Setfocus()
				Return .f.
			EndIf 
		EndIf 
		
		
		If ThisForm.podatamgr.ChkRegister('SigCdRom', 'Codigo', lcCodigo)
			Messagebox([Código já cadastrado.],64,[Aviso])
			thisform.pagina.dados.getCodigo.Setfocus()
			Return .f.
		Endif
		
		Replace CidChaves With fUniqueIds() in crSigCdRom

	Endif

	Select crSigCdRoI
	Go Top
	
	Select csSigCdRoI
	Scan 
		If Empty(Chave)
			loop
		EndIf 
		Scatter Memvar
		
		Select crSigCdRoI
		Append Blank 
		
		Gather Memvar 
		Select csSigCdRoI
	EndScan 
	
	Select crSigCdRoI
	Go Top
	Delete For (Empty(Chave)) 
	Go Top

=== OBJECT: Get_cpf (parent: SIGCDROM.Pagina.Dados.cntRom, class: textbox) ===
PROCEDURE When
Return (InList(ThisForm.pcEscolha, 'INSERIR', 'ALTERAR') And Empty(This.Parent.Get_Conta.Value))

ENDPROC
PROCEDURE Valid
Grupo = crSigCdPam.GrPadClis
If !Empty(This.Value)
	lnVerCpfCgc = 0
	lcCgc = This.Value
	lcCgc = StrTran(StrTran(StrTran(lcCgc,'.',''),'-',''),'/','')
	If Len(Alltrim(lcCgc)) <> 14
		lcCgc1 = TransForm(lcCgc,'@R 999.999.999-99')
		If Len(Alltrim(lcCgc)) = 11
			lnVerCpfCgc = Iif(fValidarCPF(lcCgc1),1,2)
		EndIf
	Else
		lcCgc1 = TransForm(lcCgc,'@R 99.999.999/9999-99')
		lnVerCpfCgc = Iif(fValidarCNPJ(lcCgc1),1,2)
	EndIf
	
	If lnVerCpfCgc = 2
		MessageBox('CPF / CGC Incorreto !!!',48,'')
		Return 0
	EndIf

	If !(ThisForm.poDataMgr.CursorQuery('SigCdCli',,'cpfs',Padr(Alltrim(lcCgc1),20)))
		If lnVerCpfCgc = 1
			MessageBox('CPF / CGC não encontrado !!!',48,'')
			Return 0
		EndIf
	EndIf
	
	lcIclis = CqSigCdCli.iclis
	If !fAcessoContas(Usuar,Grupo,'C',lcIclis,This.Parent.Get_Conta.Value,This.Parent.Get_DConta.Value)
		Messagebox('Acesso Negado !!',0+48,'')
		This.Parent.Get_Conta.Value  = ''
		This.Parent.Get_DConta.Value = ''
		This.Parent.Get_Cpf.Value    = ''
		Return 0
	Else
		Select CqSigCdCli
		This.Parent.Get_Conta.Value  = CqSigCdCli.iclis
		This.Parent.Get_DConta.Value = CqSigCdCli.rclis
		This.Parent.Get_Cpf.Value    = CqSigCdCli.cpfs
		This.Parent.Get_Conta.Refresh
		This.Parent.Get_DConta.Refresh
		This.Parent.Get_Cpf.Refresh
	EndIf
Else
	This.Parent.Get_DConta.Value = ''
EndIf

Return(.T.)
ENDPROC

	=ThisForm.Atualizatotal()
	Keyboard '{DNARROW}'
