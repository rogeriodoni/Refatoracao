PROCEDURE acertacolunas
Lparameters pTip

With ThisForm.pgfPrincipal.pgDados.pgResults.pgRes01.grdDados1
	If (pTip = [LER])
		lcStr = ThisForm.pgfPrincipal.pgFiltros.getCfgGrade.Value 
		
		lnPos = 1 && Posição Inicial de Cada Bloco Correspondente a Uma Coluna da Grade
		For lnCnt = 1 To .ColumnCount
			lcBlo = Substr(lcStr, lnPos, 5) 		&& Bloco Com 5 Posições: 1 a 3 = Width; 	
													&&						 4     = Visible	
													&&						 5     = Uso Futuro	
			lnWid = Int(Val(Substr(lcBlo, 1, 3)))	&& Largura da Coluna						
			llVis = Not (Substr(lcBlo, 4, 1)=[0])	&& Coluna Visible=1 (default); Invisible=0	
			lcCol = Alltrim(Str(lnCnt))				&& Número da Coluna							

			.Column&lcCol..Width   = Iif(lnWid > 0, lnWid, .Column&lcCol..Width) && Se Não Estiver Gravado Mantém a Largura Original
			.Column&lcCol..Visible = llVis										 && Se Não Estiver Gravado Mantém Visible			

			lnPos = lnPos + 5 && Incrementa de 5 em 5 Posições a Cada Coluna
		Next
	Else && [GRAVAR]
		lcStr = []
		For lnCnt = 1 To .ColumnCount
			lcCol = Alltrim(Str(lnCnt))										&& Número da Coluna	
			lcStr = lcStr + Padl(.Column&lcCol..Width, 3, [0]) + ;
							Iif(.Column&lcCol..Visible, [1], [0]) + ;
							[ ] && Nota: o Branco é Para Uso Futuro
		Next
		ThisForm.pgfPrincipal.pgFiltros.getCfgGrade.Value = lcStr
	EndIf
EndWith
* Fim **************************************************************************************
ENDPROC

* PROCEDURE buscaantigos
------------------------------------------------------------
PROCEDURE buscaantigos
Private pdDtA
Local lcPro, lcTam

Select crCursor1
oProg2 = CreateObject([fwProgressBar], [Processando Dados Anteriores...], Reccount([crCursor1]))
oProg2.Top = oProg2.Top + Int(oProg2.Height/2)
oProg2.Rodape.Caption = [Buscando Produtos na Data de Movimentação]
oProg2.Show
Scan
	oProg2.Update(.t.)

	lcPro = Nvl(crCursor1.Codigo_Item, [])
	lcTam = Nvl(crCursor1.Tamanho, [])
	pdDtA = Nvl(crCursor1.Data_Venda, Datetime())
		
	lcQuery = [Select Top 1 ] + ;
					 [p.Mercs as Gde_Grupo, ] + ; 
					 [p.CGrus as Grupo, ] + ; 
					 [p.SGrus as SubGrupo, ] + ; 
					 [p.CustoFs as Custo_Cadastro, ] + ; 
					 [p.MoeCusfs as Moe_Cus_Cadastro, ] + ; 
					 [p.MarkupA as Markup_Cadastro, ] + ; 
					 [p.Reffs as Ref_Fornecedor, ] + ;
					 [p.PVens as Venda_Cadastro, ] + ; 
					 [p.PVens as Desc_Acres, ] + ; 
					 [p.PVens as Venda_CadastroR, ] + ; 
					 [p.Moevs as Moe_Vda_Cadastro, ] + ; 
					 [p.FigJpgs as Imagem, ] + ;
					 [p.IFors as Cod_Fornecedor ] + ;
				[From SigCdPrc p ] + ;
			   [Where p.CPros = '] + lcPro + [' And ] + ;
			   		 [p.DataAlts >= ?pdDtA ] + ;
			   [Order By p.DataAlts Desc]

	If Not fExecCmds(ThisForm.poDataMgr, [crAntigo1], lcQuery)
		oProg2.Complete
		Return .f.
	EndIf
	
	Select crAntigo1
	Go Top In crAntigo1
	If Not Eof([crAntigo1])
		Replace Gde_Grupo 		 With crAntigo1.Gde_Grupo, ; 
				Grupo 			 With crAntigo1.Grupo, ; 
				SubGrupo 		 With crAntigo1.SubGrupo, ; 
				Custo_Cadastro 	 With crAntigo1.Custo_Cadastro, ; 
				Moe_Cus_Cadastro With crAntigo1.Moe_Cus_Cadastro, ; 
				Markup_Cadastro  With crAntigo1.Markup_Cadastro, ; 
				Ref_Fornecedor 	 With crAntigo1.Ref_Fornecedor, ; 
				Venda_Cadastro 	 With crAntigo1.Venda_Cadastro, ; 
				Desc_Acres 		 With crAntigo1.Desc_Acres, ; 
				Venda_CadastroR  With crAntigo1.Venda_CadastroR, ; 
				Moe_Vda_Cadastro With crAntigo1.Moe_Vda_Cadastro, ; 
				Imagem 			 With crAntigo1.Imagem, ; 
				Cod_Fornecedor 	 With crAntigo1.Cod_Fornecedor In crCursor1

		lcQuery = [Select Top 1 ] + ;
						 [t.Percs as Markup_Tamanho, ] + ; 
						 [t.Valor as Valor_Tamanho, ] + ; 
						 [t.PesoMs as Peso_Tamanho ] + ;
					[From SigPrTa2 t ] + ;
				   [Where t.CPros = '] + lcPro + [' And ] + ;
				   		 [t.CodTams = '] + lcTam + [' And ] + ;
				   		 [t.DataAlts >= ?pdDtA ] + ;
				   [Order By t.DataAlts Desc]

		If Not fExecCmds(ThisForm.poDataMgr, [crAntigo1], lcQuery)
			oProg2.Complete
			Return .f.
		EndIf
		
		Select crAntigo1
		Go Top In crAntigo1
		If Not Eof([crAntigo1])
			Replace Markup_Tamanho With crAntigo1.Markup_Tamanho, ;
					Valor_Tamanho  With crAntigo1.Valor_Tamanho, ;
					Peso_Tamanho   With crAntigo1.Peso_Tamanho In crCursor1
		EndIf
	EndIf
	Use In crAntigo1
EndScan
oProg2.Complete
* Fim **************************************************************************************

ENDPROC

LParameters pVal, pTip

lcTip = Iif(Type([pTip])=[C], pTip, [N])

Do Case
	Case (lcTip = [C])
		lcRet = Strtran(Strtran(Strtran(Nvl(pVal, []), Chr(9), []), Chr(10), []), Chr(13), [])

	Case (lcTip = [N])

			(lnValor * lnQtds),crSigCdPro.MoeVs,crSigCdPro.DPros,crSigCdPro.CUnis,fUniqueIds(),Date(),_EDN,lnPesos,lnBarras,crTmpItem.PedidoWeb)

		*--Lanca o Barra no SIGMVITS
		If (Not Empty(crSigCdPro.CodCors) Or Not Empty(crTmpItem.CodTams))
			Insert Into crSigMvIts (Emps,Dopes,Numes,cPros,CodTams,CodCors,qtds,codbarras,Pesos,cidchaves,EmpDopNums,chksubn,CItens) ;
				Values (_Emps,_Dopes,_Inicio,crSigCdPro.cPros,crTmpItem.CodTams,crSigCdPro.CodCors,lnQtds,lnBarras,0,fUniqueIds(),_EDN,.F.,lnItem)
		Endif

		If crSigCdOpe.Estoqs = 1

			pCidC = Dtos(crTmpItem.DataS)+lcOpers+Transform(fGerUniqueKey(Dtos(crTmpItem.DataS)),"@L 999999")+Thisform.Sigkey
			pSeq  = fGerUniqueKey('HISTBAR')

			Insert Into crSigMvHst (cPros,Datars,DataS,DtAudits,Dopes,EmpOs,Emps,Opers,Numes,qtds,Units,Totas,Grupos,;

		If (crSigCdOpe.Situas > 1) && Adicionar na Entrega: 1-Nenhum; 2-Cor; 3-Componente; 4-Ambos
			lnSum = 0
			If Inlist(crSigCdOpe.Situas, 2, 4)

						=Messagebox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (LocalCor)])
					Endif
					Return 0
				Endif

				Select Max(b.DiasEnts) As Dias ;
					From crSigMvIts a, LocalCor b ;
					Where a.CodCors = b.Cods ;
					Into Cursor LocalDias

				Select LocalDias
				Go Top In LocalDias
				If Not Eof([LocalDias]) And (LocalDias.Dias > 0)
					lnSum = lnSum + LocalDias.Dias
				Endif
				Use In LocalCor
				Use In LocalDias
			Endif

			If Inlist(crSigCdOpe.Situas, 3, 4)
				lnMax = 0
				If (lnMax > 0)
					lnSum = lnSum + lnMax
				Endif
			Endif

			*** mena - 07/11/2016 - busca qtde de dias para calculo de prazo de entrega de serviço - montecarlo
			If Inlist(crSigCdOpe.Situas, 5)
				lnMax = 0
				Select crSigMvIts
				Scan
					lcQuery = [Select b.DiasPrz From SigCdPro a, SigCdPrz b where a.cpros = ']+crSigMvIts.cPros+['] +;
						[ and a.CGrus = b.codigos And b.Emps =']+crSigcdemp.Cemps+[']

					If (Thisform.poDataMgr.SqlExecute(lcQuery, [crPrz]) < 1)
						If Not Thisform.Automatico
							=Messagebox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (crPrz)])
						Endif
						Return 0
					Endif

					Select crPrz
					Go Top In crPrz
					If Not Eof([crPrz]) And (crPrz.DiasPrz > lnMax)
						lnMax = crPrz.DiasPrz
					Endif
					Use In crPrz
				Endscan

				If (lnMax > 0)
					lnSum = lnMax
				Endif
			Endif
			**********************************************************

			If (lnSum > 0)
				If Not Inlist(crSigOpCdd.chkUteis, 0, 4) && Dias Úteis
					llSab = Inlist(crSigOpCdd.chkUteis, 1)
					llDom = Inlist(crSigOpCdd.chkUteis, 1, 3)
					llFer = Inlist(crSigOpCdd.chkUteis, 1, 2, 3)

					ldDI1 = Ttod(crSigMvCab.DataS)
					ldDI2 = Ttod(crSigMvCab.DataS) + lnSum
					Do While (ldDI1 <= ldDI2)
						If fChkFeriado(Thisform.poDataMgr, ldDI1, llSab, llDom)
							lnSum = lnSum + 1
						Endif
						ldDI1 = ldDI1 + 1
						ldDI2 = Ttod(crSigMvCab.DataS) + lnSum
					Enddo
				Endif

				ldEnt = Ttod(crSigMvCab.DataS) + lnSum

				If (Ttod(crSigMvCab.PrazoEnts) <> ldEnt)
					Replace PrazoEnts With ldEnt In crSigMvCab
				Endif
			Endif
		Endif
	Endif

	lcUpdate = [Update SigMvCab Set pStatus = 'D' Where EmpDopNums = ']+lcEdn+[' ]
	If (Thisform.poDataMgr.SqlExecute(lcUpdate) < 1)
		If Not Thisform.Automatico
			=Messagebox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (Update)])
		Endif
		Return .F.
	Endif

	With Thisform.poDataMgr
		If lnErro1>0
			lnErro1 = Iif(.Update('crSigMvCab'), 1, -1)
		Endif
		If lnErro1>0
			lnErro1 = Iif(.Update('crSigMvItn'), 1, -1)
		Endif
		If lnErro1>0
			lnErro1 = Iif(.Update('crSigMvIts'), 1, -1)
		Endif

		If lnErro1>0
			lnErro1 = Iif(.Update('crSigMvHst'), 1, -1)
		Endif
		If lnErro1>0
			lnErro1 = .Commit()
		Endif
		If Not lnErro1>0
			= .Rollback()
		Endif
	Endwith

	Select crRecolhe
Endscan

Select(lnOldSel)
Go Top

ENDPROC

PROCEDURE seltoprint
Dimension laArray(2)
Set Step On 
Select crPgRes01
Select pedidoweb, Nvl(Count(*),0) As qtdP From  crPgRes01 b Where ckPresente <> 0 Group By pedidoweb Into Cursor crPres

Select crPgRes01
Select pedidoweb, Sum(quantidade) As qtdT From  crPgRes01 b Group By pedidoweb Into Cursor crPg

Select a.*, b.qtdT, Nvl(qtdP,0) as qtdP, Space(60) As endes,Space(10) As nums,Space(25) As compls,Space(40) As bairs, ;
	Space(30) As cidas,Space(2) As estas, Space(9) As Ceps, 'N' as cGNRE  ;
	from crPgRes01 a ;
	join crPg b on a.pedidoweb=b.pedidoweb ;
	left join crPres c on a.pedidoweb=c.pedidoweb ;
	Order By a.pedidoweb, a.empresa, a.Ref_Fornecedor ;
	into Cursor crPg1 Readwrite
Select crPg1
Go Top

*** mena - 02/07/2017 - le os dados do cliente ***
Scan
	lcQuery = [select endes, nums, compls, bairs, cidas, estas, ceps from sigcdcli where iclis = ']+crPg1.cod_cliente+[']
	If Thisform.podatamgr.SqlExecute(lcQuery,[crCli]) < 1
		Messagebox([Favor Reiniciar o Processo],0+16,[])
	Endif

	Replace endes With crCli.endes In crPg1
	Replace nums With crCli.nums In crPg1
	Replace compls With crCli.compls In crPg1
	Replace cidas With crCli.cidas In crPg1
	Replace bairs With crCli.bairs In crPg1
	
	Replace Ceps With crCli.Ceps In crPg1
	
	lcUFD = crCli.estas
	
	
	If Not Empty(crPg1.IDEntrega)
		laArray(1) = crPg1.ContaDs
		laArray(2) = []
	Endif

	If fEndEntrega(@laArray, Thisform.poDataMgr,,crPg1.IDEntrega)
		lcUFD = Alltrim(laArray(11)) 							&& Estas
	EndIf
	
	If Not Empty(lcUFD)
		lcQuery = [select estados, chkgnre from sigcdufs where estados = ']+lcUFD+[']
		If Thisform.podatamgr.SqlExecute(lcQuery,[crEstas]) < 1
			Messagebox([Favor Reiniciar o Processo],0+16,[])
		Endif
		If Reccount([crEstas])>0
			replace cGNRE With Iif(crEstas.chkgnre = 1,'S','N') In crPg1 	
		EndIf 
	EndIf 
	Replace estas With lcUFD In crPg1
	Select crPg1
Endscan

Select crPg1
Go Top

Select crPg1
Select Space(10) as Quebra, * from crPg1 ;
	Where 1=0 ;
	Into Cursor crPg2 ReadWrite

Select crPg1
Select distinct pedidoweb, empresa, contaOrigem ;
	From crPg1 ;
	Into Cursor crPg3 ReadWrite

Select crPg3
Select pedidoweb, Count(*) As qtdPedido ;
	From crPg3 ;
	group by pedidoweb ;
	Into Cursor crPg4 ReadWrite

Select crPg4
Select a.* ;
	from crPg1 a ;
	join crPg4 b on a.pedidoweb = b.pedidoweb ;
	where b.qtdPedido = 1 ;
	Into cursor crPg5 ReadWrite

Select crPg5
Scan 
	Scatter memvar 
	m.quebra = Alltrim(Iif(Empty(m.pstatus),Chr(254),m.pstatus)) + m.contaOrigem
	Select crPg2
	Append Blank 
	Gather memvar 
	Select crPg5
EndScan


Select a.* from crPg1 a join crPg4 b on a.pedidoweb = b.pedidoweb where b.qtdPedido <> 1 Into cursor crPg6 ReadWrite
Select crPg6
Scan 
	Scatter memvar 
	m.quebra = Alltrim(Iif(Empty(m.pstatus),Chr(254),m.pstatus)) + Replicate(Chr(254),9)
	Select crPg2
	Append Blank 
	Gather memvar 
	Select crPg6
EndScan

If Used([crPg1])
	use in crPg1
EndIf 

If Used([crPg3])
	use in crPg3
EndIf 

If Used([crPg4])
	use in crPg4
EndIf 

If Used([crPg5])
	use in crPg5
EndIf 

If Used([crPg6])
	use in crPg6
EndIf 

Select crPg2
Select * from crPg2 Into cursor crPg1 ReadWrite 

Set Step On 
Select crPg1
index on quebra + pedidoweb + empresa + contaOrigem + Codigo_Item Tag EmpCtaOri
Go Top
