PROCEDURE Init
Lparameters pArquivo, pTexto1, pTexto2, pTimer
* Função : Mostrar Uma Imagem Passada Como Parâmetro (Todos os Formatos Válidos)
* pArquivo = Nome Completo do Arquivo de Imagem a Ser Mostrado (Incluindo Path) 
* pTexto1  = Texto Apresentado Abaixo da Imagem no 1º Quadro (Opcional)
* pTexto2  = Texto Apresentado Abaixo da Imagem no 2º Quadro (Opcional)

ThisForm.cntSombra.lblSombra.Caption = ThisForm.Caption
ThisForm.cntSombra.lblTitulo.Caption = ThisForm.Caption

If Empty(pArquivo) Or Not File(pArquivo)
	ThisForm.Release
EndIf

If pTimer
	ThisForm.Timer.Interval = 3000
Else
	ThisForm.Timer.Interval = 0
	Thisform.timer.Enabled = .F.
EndIf 

If Not InList(Lower(Right(Alltrim(pArquivo),4)), '.bmp', '.dib', '.gif', '.jpg', '.cur', '.ani', '.ico')
	=MessageBox('O Arquivo Passado Não é Um Arquivo de Imagem Válido!!!' + Chr(13) + ;
				'Use Um Arquivo de Extensão : .bmp, .dib, .gif, .jpg, .cur, .ani ou .ico', 16, '')
	ThisForm.Release
EndIf

ThisForm.lblTexto1.Caption = ''
If (Type('pTexto1') = 'C')
	ThisForm.lblTexto1.Caption = pTexto1
EndIf

ThisForm.lblTexto2.Caption = ''
If (Type('pTexto2') = 'C')
	ThisForm.lblTexto2.Caption = pTexto2
EndIf

Clear Resources

ThisForm.ImgFigJpg.Picture = (pArquivo)
* Fim ********************************************************************************* 

ENDPROC
