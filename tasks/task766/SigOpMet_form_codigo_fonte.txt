------------------------------------------------------------
* PROCEDURE importaplan
------------------------------------------------------------
PROCEDURE importaplan
Private ldDataD
Local alMes

Declare alMes(12)
alMes[1] = 'JANEIRO'
alMes[2] = 'FEVEREIRO'
alMes[3] = 'MARCO'
alMes[4] = 'ABRIL'
alMes[5] = 'MAIO'
alMes[6] = 'JUNHO'
alMes[7] = 'JULHO'
alMes[8] = 'AGOSTO'
alMes[9] = 'SETEMBRO'
alMes[10] = 'OUTUBRO'
alMes[11] = 'NOVEMBRO'
alMes[12] = 'DEZEMBRO'

lcArq = Getfile('xls*')

If Empty(lcArq)
	Return .F.
Endif

If Not ImpExcelDriver(lcArq,'Orç.Meta Dia')
	Return .F.
Endif
lnCelI=0

Select * From xlResults Into Cursor localMeta Readwrite
Select localMeta
Go Top
*** produra o numero da coluna onde esta a data do orcamento e parte desse posicionamento para calcular as outras colunas ***
For j=1 To 10
	lcCampo=[LocalMeta.F]+Alltrim(Str(j))
	If &lcCampo='ORÇAMENTO'
		lnCelI=j	&& celula inicial onde está o mes da meta
		Exit
	Endif
Endfor


If lnCelI <> 0
	lcCampo=[LocalMeta.F]+Alltrim(Str(lnCelI))
	lcData = Alltrim(&lcCampo)+' '
	lcData =Alltrim(Strextract(lcData,'DE '))
	lcMes = Upper(fLimpaTexto(Alltrim(Strextract(lcData,'',' ',1))))
	lcAno = Right(lcData,4)
	For I=1 To 12
		If lcMes=alMes[i]
			lcMes = Padl(I,2,'0')
			Exit
		Endif
	Endfor
Endif

If Not ImpExcelDriver(lcArq,'Orç.Desafio Dia')
	Return .F.
Endif
Select * From xlResults Into Cursor localMetaD Readwrite

Select crSigopMet
Zap

*Append From &lcArq. delimited with tab
Select localMeta
Go Top
Scan
	lcCampo=[LocalMeta.F]+Alltrim(Str(lnCelI+0))
	lcUf   = Alltrim(Iif(Inlist(Vartype(&lcCampo),'C','M'),Nvl(&lcCampo,[]),[]))
	lcCampo=[LocalMeta.F]+Alltrim(Str(lnCelI+2))
	lcCEmp = Alltrim(Iif(Inlist(Vartype(&lcCampo),'C','M'),Nvl(&lcCampo,[]),[]))
	lcCampo=[LocalMeta.F]+Alltrim(Str(lnCelI+3))
	lcEmp  = Alltrim(Iif(Inlist(Vartype(&lcCampo),'C','M'),Nvl(&lcCampo,[]),[]))
	lcCampo=[LocalMeta.F]+Alltrim(Str(lnCelI+4))
	lcNomeL= Alltrim(Iif(Inlist(Vartype(&lcCampo),'C','M'),Nvl(&lcCampo,[]),[]))

	lcQuery = 	[Select * From SigcdEmp where cemps = ']+lcEmp+[' ]
	If Thisform.Podatamgr.Sqlexecute(lcQuery,'localEmp') < 1
		Messagebox('Impossivel Efetuar Conexão com Servidor de Banco de Dados.',(0+48),'Erro')
		Return
	Endif

	lnCel = lnCelI+6
	If Reccount('localEmp') > 0
		Select localMeta
		For I=1 To 31
			lnDay=I
			ldDataD=Ctot(Padl(lnDay,2,'0')+'/'+lcMes+'/'+lcAno)
			lcCampo=[LocalMeta.F]+Alltrim(Str(lnDay+lnCel))
			lnMeta =Iif(Inlist(Vartype(&lcCampo),'N','M','C'),Nvl(&lcCampo,0),0)
			
			If Type([lnMeta])=[C]
				lnMeta = Val(lnMeta)
			EndIf 
			
			If lnMeta <> 0
				lcQuery = 	[Select * From SigopMet where emps = ']+lcEmp+[' and datas = ?ldDataD]
				If Thisform.Podatamgr.Sqlexecute(lcQuery,'crAux') < 1
					Messagebox('Impossivel Efetuar Conexão com Servidor de Banco de Dados.',(0+48),'Erro')
					Return
				Endif
				If Reccount('crAux') = 0
					lcCod = lcEmp+Padl(Alltrim(Str(fGerUniqueKey(lcEmp))),7,'0')
				Else
					lcCod = crAux.codigo
				Endif

				Select localMetaD
				lcCpoUfd=[LocalMetaD.F]+Alltrim(Str(lnCelI+0))
				lcCpoEmpd=[LocalMetaD.F]+Alltrim(Str(lnCelI+3))

				Locate For &lcCpoUfd=lcUf And &lcCpoEmpd=lcEmp
				lcCampo=[LocalMetaD.F]+Alltrim(Str(lnDay+lnCel))
				lnMetaD =Iif(Inlist(Vartype(&lcCampo),'N','M','C'),Nvl(&lcCampo,0),0)

				If Type([lnMetaD])=[C]
				
					lnMetaD = Val(lnMetaD)
				EndIf 
			
				Insert Into crSigopMet (ufs, Emps, codemps, nomeloja, DataS, valmeta, valmetad, codigo, usualts, dtalts);
					Value (lcUf, lcEmp, Val(lcCEmp), lcNomeL, ldDataD, lnMeta, lnMetaD, lcCod, Usuar, Datetime())

			Endif

		Endfor
	Endif
	Select localMeta
Endscan

llOks = .T.

If Not Thisform.Podatamgr.Update([crSigOpMet])
	=Messagebox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (Update - crSigPrNfe)])
	llOks = .F.
Endif
If llOks
	If (Thisform.Podatamgr.Commit() < 1)
		=Messagebox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (Commit)])
		llOks = .F.
	Endif
Endif

Thisform.Refresh
Return llOks

ENDPROC
