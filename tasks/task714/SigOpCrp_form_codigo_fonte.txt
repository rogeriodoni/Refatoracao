PROCEDURE btnOK.Click
Local lcChave, llFinal

Set Near On
Select CrSigMvCcr

llFinal = Eof()
If !llFinal 
	Skip
	llFinal = Eof()
	Skip - 1
Endif

Do Case
	Case !IsEmpty(ThisForm.getLancamento.Value)
		ThisForm.Continuar = ''
		lcChave = Dtos(ThisForm.getLancamento.Value)
		Locate For Dtos(Datas) = lcChave

	Case ThisForm.getValor.Value # 0
		lcChave = ThisForm.getValor.Value
		If lcChave = Val(ThisForm.Continuar) .and. !llFinal
			Continue
		Else
			ThisForm.Continuar = Str(lcChave,12,2)
			Go Top
			Locate For lcChave = Valors
		Endif
		
	Case !IsEmpty(ThisForm.getHistorico.Value)
		lcChave = Alltrim(ThisForm.getHistorico.Value)
		If Alltrim(ThisForm.Continuar) = lcChave .and. !llFinal
			Continue
		Else
			ThisForm.Continuar = lcChave
			Go Top 
			Locate For lcChave $ Hists
		Endif
EndCase

ThisForm.FormPai.Page.Hist.grd_lancamento.Refresh
ThisForm.FormPai.Page.Hist.grd_lancamento.SetFocus

Do Case
	Case !IsEmpty(ThisForm.getLancamento.Value)
		ThisForm.Release
	Case ThisForm.getValor.Value # 0
		ThisForm.getValor.SetFocus
	Case !IsEmpty(ThisForm.getHistorico.Value)
		ThisForm.getHistorico.SetFocus
Otherwise
	ThisForm.Release
EndCase

Set Near Off

ENDPROC
