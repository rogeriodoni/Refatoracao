------------------------------------------------------------
* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
LParameters pBar

=fConfigGeral()

ThisForm.cntSombra.lblSombra.Caption = ThisForm.Caption
ThisForm.cntSombra.lblTitulo.Caption = ThisForm.Caption

Private lnBar

If (Type('pBar')<>'N') Or Empty(pBar)
	ThisForm.Release
EndIf

lnBar = pBar

ThisForm.poDataMgr = CreateObject('fSqlConector',ThisForm.Name)

lcQuery   = [Select datas,horas,locatus,locnews,usuars,empdopnums ]+;
			[From SigCdMlc ]+;
			[Where barras = ?lnBar ]+;
			[Order By datas,horas,locatus,locnews,usuars,empdopnums ]

If ThisForm.poDataMgr.SqlExecute(lcQuery,'CsSigMvHst') < 1
	MessageBox('Favor reinicializar o processo !!!',16,'Falha na Conexão - (CsSigMvHst)')
	Return 0
EndIf

With ThisForm.GrdHistorico
	.ColumnCount  = 6
	.DeleteMark   = .F.
	.RecordMark   = .F.
	.RecordSource = 'CsSigMvHst'
	.Column1.ControlSource  = 'CsSigMvHst.datas'
	.Column2.ControlSource  = 'CsSigMvHst.horas'
	.Column3.ControlSource  = 'CsSigMvHst.locatus'
	.Column4.ControlSource  = 'CsSigMvHst.locnews'
	.Column5.ControlSource  = 'CsSigMvHst.usuars'
	.Column6.ControlSource  = 'CsSigMvHst.empdopnums'
	For lnCol = 1 To .ColumnCount
		lcCol = Alltrim(Str(lnCol))
		.Column&lcCol..Movable   = .F.
		.Column&lcCol..Resizable = .F.
		.Column&lcCol..ReadOnly  = .T.
	Next
EndWith

ThisForm.CmdBtnForm.BtnSair.SetFocus
ENDPROC

------------------------------------------------------------
* PROCEDURE Release
------------------------------------------------------------
PROCEDURE Release
ThisForm.poDataMgr.Release

=DoDefault()
* Fim **************************************************************************************
ENDPROC
