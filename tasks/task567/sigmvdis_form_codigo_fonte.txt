* PROCEDURE calcula_saldo
------------------------------------------------------------
PROCEDURE calcula_saldo
*!*	**
If crDistribui.QtdDistr <= 0
	Replace crDistribui.QtdDistr With 0
EndIf

*!*	*!*	a.CPros, b.Dpros, a.CodCors, a.CodTams, a.SQtds, 0 As QtdSaida, a.SQtds As QtdSaldo
*!*	*!*		c.cEmps, 0 As QtdSaldo, 0 As QtdDistr, a.cPros 

*!*	Local lnQtdDistr As Integer
*!*	lnQtdDistr = crDistribui.QtdDistr

*!*	If crDisponivel.QtdSaldo > 0

*!*		If crDisponivel.QtdSaldo >= crDistribui.QtdDistr

*!*			Select crDistribui
*!*			Replace crDistribui.QtdSaldo With crDistribui.QtdSaldo + lnQtdDistr
*!*	**		Replace crDistribui.QtdDistr With 0

*!*			Select crDisponivel
*!*			Replace crDisponivel.QtdSaida With crDisponivel.QtdSaida + lnQtdDistr
*!*			Replace crDisponivel.QtdSaldo With crDisponivel.QtdSaldo - lnQtdDistr

*!*			This.Pagina.dados.pageframe1.page2.grdDisponivel.Refresh()
*!*			This.Pagina.dados.pageframe1.page2.grdDistribui.Refresh()
*!*			This.Pagina.dados.pageframe1.page2.grdDistribui.Col03_Distribuido.Refresh()
*!*			This.Pagina.dados.pageframe1.page2.grdDistribui.Col03_Distribuido.Text1.Refresh()
*!*	**		This.Pagina.dados.pageframe1.page2.grdDistribui.Col03_Distribuido.Text1.SetFocus()

*!*		EndIf
*!*		
*!*	EndIf

Select crDistribui
lnReg = Recno()
Sum QtdDistr for Cpros = crDisponivel.Cpros to lnQtdDistr
Go lnReg

If crDisponivel.Sqtds - lnQtdDistr < 0
	Messagebox('Saldo insuficiente para movimentar.',48,'Aviso')
	Replace QtdDistr With 0 in crDistribui
	Select crDistribui
	lnReg = Recno()
	Sum QtdDistr for Cpros = crDisponivel.Cpros to lnQtdDistr
	Go lnReg
Endif 

*!*	Replace QtdSaldo With crDistribui.QtdDistr in crDistribui
Replace QtdSaida With lnQtdDistr in crDisponivel
Replace QtdSaldo With Sqtds - lnQtdDistr in crDisponivel

This.Pagina.dados.pageframe1.page2.grdDisponivel.Refresh()
This.Pagina.dados.pageframe1.page2.grdDistribui.Refresh()



ENDPROC

	Return

Zap in dbimpressao

		If ThisForm.poDataMgr.SqlExecute(lcQuery,'CrEesti') < 0
			MessageBox('Favor Reiniciar o Processo!!!',0+48,'Erro Conexao')
			Return 0
		EndIf
		lcquery = [Select * From SigMvIts Where 0=1 Order By CItens]
		If ThisForm.poDataMgr.SqlExecute(lcQuery,'CrEsti2') < 1
			MessageBox('Favor Reiniciar o Processo!!!',0+48,'Erro Conexao')
			Return 0
		EndIf

	Else
		lcbusca = lcEmps + lcDopes + Str(lnNumes,6)
		lcQuery = [Select * From SigMvItn Where EmpDopNums = ']+lcbusca+[' Order By CItens]
		If ThisForm.poDataMgr.SqlExecute(lcQuery,'CrEesti') < 0
			MessageBox('Favor Reiniciar o Processo!!!',0+48,'Erro Conexao')
			Return 0

Count For nMarca = 1 To lnSel1
If lnSel1 = 0
	=Messagebox('Nenhum estoque foi selecionado.', 16, 'Aviso')
	Return .F.
Endif

Select 0
logrdDisponivel.RecordSourceType = 1 	&& Alias
logrdDisponivel.RecordSource = ""

logrdDistribui.RecordSourceType = 1		&& Alias
logrdDistribui.RecordSource = ""

*** Filtro de estoques de distribuição
lcEgc = []
lcEmps = []
lcEdn = []
Select crSigMvDis
Scan
	If crSigMvDis.nMarca = 1
		lcEgc = lcEgc + Iif(Empty(lcEgc),[('],[',']) + crSigMvDis.Emps + crSigMvDis.Grupos + crSigMvDis.Contas
		lcEmps = lcEmps + Iif(Empty(lcEmps),[('],[',']) + crSigMvDis.Emps
		lcEdn = lcEdn + Iif(Empty(lcEdn ),[('],[',']) + crSigMvDis.OriDopNums
	Endif
Endscan
lcEgc = Iif(Empty(lcEgc),'',lcEgc + [') ])
lcEmps = Iif(Empty(lcEmps),'',lcEmps + [') ])
lcEdn = Iif(Empty(lcEdn),'',lcEdn + [') ])

TEXT tO lcSQL NoShow TextMerge
	Select a.CPros, b.Dpros, a.CodCors, a.CodTams, Sum(a.SQtds) as Sqtds, 0000000 As QtdSaida, Sum(a.SQtds) As QtdSaldo, b.Cgrus, b.Sgrus, c.OriDopNums
	From SigMvEst a
	Join SigCdpro b on a.Cpros = b.Cpros
	Join (Select distinct a.Cpros, Isnull(b.codcors,'') as CodCors, Isnull(b.codtams,'') as CodTams, a.EmpDopNums as OriDopNums
						from SigmvItn a
						Left Join SigMvIts b on a.empdopnums = b.EmpDopNums And a.cItens = b.cItens And a.Cpros = b.Cpros
						join SigMvCab c on a.EmpDopNums = c.EmpDopNums
						join sigcdope d on c.dopes = d.dopes
						join SigOpCdd e on d.dopes = e.dopes
						Where e.Distribui = 1 And c.GrupoDs <> Space(10) And c.ContaDs <> Space(10) And a.EmpDopNums in <<lcEdn>> ) c
							on a.Cpros = c.Cpros And a.CodTams = c.CodTams And a.CodCors = c.CodCors
	Where a.EmpGruEsts in <<lcEgc>>
	  And a.SQtds > 0
	Group By c.OriDopNums, a.CPros, b.Dpros, a.CodCors, a.CodTams, b.Cgrus, b.Sgrus
ENDTEXT

If Thisform.podatamgr.SqlExecute(lcSQL, [crDisponivel] ) < 1
	Messagebox( [Problemas no Select dos Produtos com Saldo], 48, "Atenção" )
	Return(0)
Endif

TEXT tO lcSQL NoShow TextMerge
	Select b.cEmps, Case When d.SQtds is null Then 0000000 Else d.Sqtds End As QtdSaldo, 0 As QtdDistr, Space(10) as Locals, a.CPros, a.CodCors, a.CodTams, c.Dopes, e.OriDopNums
	from (Select Distinct a.CPros, a.CodCors, a.CodTams From SigMvEst a
	Where a.EmpGruEsts in <<lcEgc>> And a.SQtds > 0) a
	Inner Join SigCdEmp b On b.cEmps not in <<lcEmps>>
	Inner Join SigCdOpe c On b.DirColOuts = c.Dopes
	Left Join SigMvEst d On d.EmpGruEsts = b.cEmps + c.GruDests + c.ConDests And d.CPros + d.CodCors + d.CodTams = a.CPros + a.CodCors + a.CodTams
	Join (Select distinct a.Cpros, Isnull(b.codcors,'') as CodCors, Isnull(b.codtams,'') as CodTams, a.EmpDopNums as OriDopNums
						from SigmvItn a
						Left Join SigMvIts b on a.empdopnums = b.EmpDopNums And a.cItens = b.cItens And a.Cpros = b.Cpros
						join SigMvCab c on a.EmpDopNums = c.EmpDopNums
						join sigcdope d on c.dopes = d.dopes
						join SigOpCdd e on d.dopes = e.dopes
						Where e.Distribui = 1 And c.GrupoDs <> Space(10) And c.ContaDs <> Space(10) And a.EmpDopNums in <<lcEdn>> ) e
							on a.Cpros = e.Cpros And a.CodTams = e.CodTams And a.CodCors = e.CodCors
	Order By a.CPros, a.CodCors, a.CodTams, b.cEmps
ENDTEXT

If Thisform.podatamgr.SqlExecute(lcSQL, [crDistribui] ) < 1
	Messagebox( [Problemas no Select dos Produtos Distribuidos], 48, "Atenção" )
	Return(0)
Endif

logrdDistribui.RecordSource = [crDistribui]

			Contas    With crSigMvDis.Contas,;

			Emps      With crSigMvDis.Emps  ,;

			CidQuerys With lccidquerys
	Endscan

	Select CrSigTempd
	Append Blank
	Replace Grupos    With crSigCdPam.GrupoEsts,;
		Contas    With crSigCdPam.contaEsts,;
		codObs    With 1,;

		CidQuerys With lccidquerys

		_Contad = crSigCdOpe.ConDests &&TempEest.Contads

		_Nume   = fGerUniqueKey(TmpEstoque.Emps + TmpEstoque.Dopes)
		_citens = 1
		_pData = Iif(Thisform.pcescolha = [ALTERAR],TmpSigPrDis.DataS,Datetime())

		Insert Into crSigMvCab (Emps, Dopes, Numes, MascNum, DataS, Datars, Usuars, Grupoos, Contaos, ;
			Grupods, Contads, CidChaves, Dtalts, EmpDopNums, EmpDs, Utilizados, codObs, Obses) ;

			TmpEstoque.Grupos, TmpEstoque.Estos, _Grupod, _Contad, ;
			fUniqueIds(), _pData, TmpEstoque.Emps+TmpEstoque.Dopes+Str(_Nume,6), TmpEstoque.EmpDs, 0, lcCodObs, lcObs)

		_GrvMvCab = .T.
	Endif

	llGeraEti = (crSigCdOpe.Estoqs = 1) And (crSigOpCdc.IEtiqs = 1) And (crTmpUni.Etiqs = 'S')
	lnLoop = Iif(llGeraEti,TmpEstoque.Estoque,1)

	For i=1 To lnLoop

		lnQtds = Iif(llGeraEti,1,TmpEstoque.Estoque)
		
		Insert Into crSigMvItn (Emps, Dopes, Numes, EmpDopNums, CPros, Qtds, Cunis, Pesos, CuniPs, DPros, Opers, cItens, CidChaves, Dtalts, Locals, Moedas, SitTrIcms) ;
			Values (TmpEstoque.Emps, TmpEstoque.Dopes, _Nume, crSigMvCab.EmpDopNums, TmpEstoque.CPros, lnQtds, crSigCdPro.Cunis, ;
			crSigCdPro.PesoMs, crSigCdPro.CuniPs, crSigCdPro.DPros, 'S', _citens, fUniqueIds(), _pData, TmpEstoque.Locals, crSigCdPro.moecusfs, crSigCdPro.SitTrIcms)

		Select crSigMvItn
		Scatter Memo Memvar
		m.CidChaves = fUniqueIds()
		m.Datas = _pData
		m.CodTams = TmpEstoque.CodTams
		m.CodCors = TmpEstoque.CodCors

		Select crDisponivel
		Go Top
		Locate For CPros + CodCors + CodTams = TmpEstoque.CPros + TmpEstoque.CodCors + TmpEstoque.CodTams
		m.QtdOs = crDisponivel.SQtds
		m.Datas = Ttod(_pData)
		m.EmpDs = crSigMvCab.EmpDs
		m.Dtalts = Iif(Thisform.pcescolha = [ALTERAR],Datetime(),Ctod([]))
		m.UsuAlts = Iif(Thisform.pcescolha = [ALTERAR],m.Usuar,[])

		Insert Into crSigPrDis From Memvar

		Select TmpEstoque

		If (crSigCdGrp.TipoEstos > 1)
			Insert Into crSigMvIts (cItens, Emps, Dopes, Numes, EmpDopNums, CPros, Qtds, Pesos, CodCors, CodTams, QtdEmbs, CidChaves, Dtalts, Locals) ;
				Values (_citens, TmpEstoque.Emps, TmpEstoque.Dopes, _Nume, crSigMvCab.EmpDopNums, TmpEstoque.CPros, lnQtds, crSigCdPro.PesoMs, ;
				TmpEstoque.CodCors, TmpEstoque.CodTams, 1, fUniqueIds(), _pData, TmpEstoque.Locals)
		Endif

		llGeraEti = (crSigCdOpe.Estoqs = 1) And (crSigOpCdc.IEtiqs = 1) And (crTmpUni.Etiqs = 'S')

		CodBarra = 0
		If llGeraEti
			cbar  = Transform(fGerUniqueKey('SigOpEtq',Thisform.Podatamgr2.PnidConn,.F.,.F.,.T.),"@L 9999999999999")
			cbar1 = fDigVerificador(cbar)
			cbar2 = cbar + Transform(cbar1,"@L 9")
			Set Decimals To 0
			CodBarra = Val( cbar2 )
			Set Decimals To 2
			If (crSigCdGrp.TipoEstos > 1)
				Select crSigMvIts
			Else
				Select crSigMvItn
			Endif
			Replace CodBarras With CodBarra

			llUpd = (Int(Val(Substr(crSigOpCdi.lbltxidCs, 18, 1))) = 1)
			lcCodAmss = Iif(llUpd,crSigCdOpe.Abrevs,[])

			Insert Into crSigOpEtq (cbars,CPros,Pesos,Qtds,Emps,Dopes,Numes,DataS,empos,obsetiqs, DtMovs, DtBals, DtIncs, Localizas, Grupos, Contas, CodAmss,EmpDopNums, ;
				CodCors, CodTams, Usuars) ;
				values (CodBarra, crSigMvItn.CPros, crSigMvItn.Pesos, ;
				crSigMvItn.Qtds, crSigMvItn.Emps, crSigMvItn.Dopes, ;
				crSigMvItn.Numes, crSigMvCab.DataS, Iif(crSigCdOpe.Transs = 1,crSigMvCab.EmpDs,crSigMvCab.Emps),Nvl(crSigMvItn.obs, ''), ;
				_pData, _pData, _pData, crSigMvItn.Locals, crSigMvCab.Grupods, crSigMvCab.Contads, lcCodAmss,crSigMvCab.EmpDopNums, ;
				TmpEstoque.CodCors, TmpEstoque.CodTams,m.Usuar)

			If crSigOpCdc.MovLocs <> 2
				lncodigo = fGerUniqueKey('SigCdMlc' + crSigOpEtq.empos)
				lclocAtu = crSigOpEtq.Localizas
				Insert Into crSigCdMlc (Barras, Codigos, Contas, CPros, DataS, Emps, Grupos, Horas, ;
					LocAtus, LocNews, Usuars, EmpDopNums, CidChaves, EmpCods) ;
					Values (crSigOpEtq.cbars, lncodigo, crSigOpEtq.Contas, crSigOpEtq.CPros, _pData, ;
					crSigOpEtq.empos, crSigOpEtq.Grupos, Time(), lclocAtu, crSigMvItn.Locals, Usuar, ;
					crSigMvCab.Emps + crSigMvCab.Dopes + Str(crSigMvCab.Numes,6), fUniqueIds(), ;
					crSigOpEtq.empos + Str(lncodigo, 10))
			Endif
		Endif

		_citens = (_citens + 1)

		Thisform.PodataMgr.CursorQuery('SigCdOpe', 'crSigCdOpe', 'Dopes', TmpEstoque.Dopes)

		If (crSigCdOpe.Estoqs = 1)
			pSeq  = 0
			Insert Into crSigMvHst (Usuars, DataS, Datars, Emps, Dopes, Numes, empos, CPros, Qtds, Opers, Grupos, ;
				Estos, CodCors, CodTams, EmpDopNums, EmpGruEsts, OriDopNums, CidChaves, Seqs, CodBarras) ;
				Values (Usuar, _pData, _pData, TmpEstoque.Emps, TmpEstoque.Dopes, _Nume, TmpEstoque.Emps, TmpEstoque.CPros, ;
				crSigMvItn.Qtds, 'S', TmpEstoque.Grupos, TmpEstoque.Estos, ;
				TmpEstoque.CodCors, TmpEstoque.CodTams, TmpEstoque.Emps + TmpEstoque.Dopes + Str(_Nume,6), ;
				TmpEstoque.Emps + TmpEstoque.Grupos + TmpEstoque.Estos, ;
				TmpEstoque.Emps + TmpEstoque.Dopes + Str(_Nume,6), [], pSeq, 0)

			_Inicio    = fGerUniqueKey(Dtos(crSigMvHst.DataS))
			_IniSeq    = fGerUniqueKey('HISTBAR')
			m.NewOpe = crSigMvHst.Opers
			*** Tiago - 23/10/2015 - Definido com o Richard que vamos gravar no cidchaves a mesma letra da movimentação para padronizar, assim a ordem do histórico fica correta
			*!*			If 	crSigCdOpe.Estoqs = 1 and crSigCdOpe.Origems = 1 and InList(crSigCdOpe.EstOrigs,1,2) And ;
			*!*					crSigCdOpe.destinos = 1 and InList(crSigCdOpe.EstDests,1,2)
			*!*				m.NewOpe = Iif(crSigMvHst.Opers = [E],[H],[K])
			*!*			EndIf
			Replace CidChaves With Dtos(DataS)+m.NewOpe+Transform(_Inicio,"@L 999999")+Thisform.Sigkey ,;
				Seqs      With _IniSeq In crSigMvHst

			=SqlCalcp2(crSigMvHst.Emps, crSigMvHst.Grupos, crSigMvHst.Estos, crSigMvHst.CPros, crSigMvHst.DataS, crSigMvHst.CodCors, crSigMvHst.CodTams, Thisform.PodataMgr  )
			=fRecalculaC(crSigMvHst.Emps, crSigMvHst.CPros, crSigMvHst.DataS, Thisform.PodataMgr  )

			pSeq  = 0
			Insert Into crSigMvHst (Usuars, DataS, Datars, Emps, Dopes, Numes, empos, CPros, Qtds, Opers, Grupos, ;
				Estos, CodCors, CodTams, EmpDopNums, EmpGruEsts, OriDopNums, CidChaves, Seqs, CodBarras) ;
				Values (Usuar, _pData, _pData, TmpEstoque.EmpDs, TmpEstoque.Dopes, _Nume, TmpEstoque.Emps, TmpEstoque.CPros, ;
				crSigMvItn.Qtds, 'E', _Grupod, _Contad, TmpEstoque.CodCors, TmpEstoque.CodTams, ;
				TmpEstoque.EmpDs + TmpEstoque.Dopes + Str(_Nume,6), TmpEstoque.EmpDs + _Grupod + _Contad, ;
				TmpEstoque.Emps + TmpEstoque.Dopes + Str(_Nume,6), [], pSeq, CodBarra)

			_Inicio    = fGerUniqueKey(Dtos(crSigMvHst.DataS))
			_IniSeq    = fGerUniqueKey('HISTBAR')
			m.NewOpe = crSigMvHst.Opers
			*** Tiago - 23/10/2015 - Definido com o Richard que vamos gravar no cidchaves a mesma letra da movimentação para padronizar, assim a ordem do histórico fica correta
			*!*			If 	crSigCdOpe.Estoqs = 1 and crSigCdOpe.Origems = 1 and InList(crSigCdOpe.EstOrigs,1,2) And ;
			*!*					crSigCdOpe.destinos = 1 and InList(crSigCdOpe.EstDests,1,2)
			*!*				m.NewOpe = Iif(crSigMvHst.Opers = [E],[H],[K])
			*!*			EndIf
			Replace CidChaves With Dtos(DataS)+m.NewOpe+Transform(_Inicio,"@L 999999")+Thisform.Sigkey ,;
				Seqs      With _IniSeq In crSigMvHst

			=SqlCalcp2(crSigMvHst.Emps, crSigMvHst.Grupos, crSigMvHst.Estos, crSigMvHst.CPros, crSigMvHst.DataS, crSigMvHst.CodCors, crSigMvHst.CodTams, Thisform.PodataMgr  )
			=fRecalculaC(crSigMvHst.Emps, crSigMvHst.CPros, crSigMvHst.DataS, Thisform.PodataMgr  )
		Endif
	Endfor
Endscan

If _GrvMvCab
	lcCodigo  = Iif(Thisform.pcescolha = [INSERIR],Alltrim(Str(fGerUniqueKey([SIGPRDIS]),10)),TmpSigPrDis.Codigos)
	Select crSigPrDis

llErro = .F.
If Not Thisform.PodataMgr.Update('crSigPrDis')
	=Messagebox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (Update - crSigPrDis)')

	=Messagebox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (SigOpClP)')
