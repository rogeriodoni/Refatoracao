PROCEDURE excluiitens
lcQuery = [Delete From SigCdDpr Where nome=']+ThisForm.pagina.dados.getnome.Value+[']
If (ThisForm.poDataMgr.SqlExecute(lcQuery, 'crDelSigCdDpr') < 1)
	=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (crDelSigCdDpr)')
	Return 0
EndIf
If (ThisForm.poDataMgr.Commit() < 1)
	=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (Commit)')
	Return 0
EndIf



ENDPROC

* PROCEDURE iniciadesc
------------------------------------------------------------
PROCEDURE iniciadesc
Chave = crSigPrDmo.Grupos
If not IsEmpty(Chave)
	thisform.podatamgr.cursorquery('SigCdGcr','crCursorPesq','Codigos',Chave)
	ThisForm.Pagina.Dados.Get_DGrupo.Value = crCursorPesq.DesCrs
Else
	ThisForm.Pagina.Dados.Get_DGrupo.Value = ''
EndIf
Chave = crSigPrDmo.Contas
If not IsEmpty(Chave)
	thisform.podatamgr.cursorquery('SigCdCli','crCursorPesq','iclis',Chave)
	ThisForm.Pagina.Dados.Get_DConta.Value = crCursorPesq.rclis
Else
	ThisForm.Pagina.Dados.Get_DConta.Value = ''
EndIf
ENDPROC

------------------------------------------------------------
* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
Private sNome
sNome=Space(20)
lcQryDmPro = [Select * From SigPrDmo ]
lcQryDmPri = [Select * From SigCdDpr Where nome=?sNome Order By Ordems]
With ThisForm
	.poDataMgr = CreateObject('fSqlConector', ThisForm.Name)

	If (.poDataMgr.pnIdconn > 0)
		With .poDataMgr
			.AddCursor('SigPrDmo', 'cIdChaves', 'crSigPrDmo', '', ThisForm.Pagina.Lista.Grade, lcQryDmPro)
			.pfSqlTabela(1).pColuna('Nome', '', '', 'Nome do Demonstrativo', 150, .t.)
			.AddCursor('SigCdDpr', 'cIdChaves', 'crSigCdDpr', '', , lcQryDmPri)
		EndWith
		Select crSigPrDmo
		Index On nome tag ix_nome

		lcQuery = [Select * ] + ;
		    [ From SigOpOpt]
		If (ThisForm.poDataMgr.SqlExecute(lcQuery, 'crSigOpOpt') < 1)
			=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (crSigOpOpt)')
			Return 0
		EndIf
		Select crSigOpOpt
		Index On cods tag cods

		lcQuery = [Select codigos,Descrs ] + ;
		    [ From SigCdGcr] + ;
		    [ Where gerbals=1 ]
		If (ThisForm.poDataMgr.SqlExecute(lcQuery, 'crSigCdGcr') < 1)
			=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (crSigCdGcr)')
			Return 0
		EndIf
		Select crSigCdGcr
		Index On codigos tag BALCODIGO
		Index On descrs tag BALDESCRS

		lcQuery = [Select Grupos, IClis, RClis ] + ;
				    [From SigCdCli ] + ;
				   [Where GerBals = 1]
		If (ThisForm.poDataMgr.SqlExecute(lcQuery, 'crSigCdCli') < 1)
			=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (crSigCdCli)')
			Return 0
		EndIf
		Select crSigCdCli
		Index On IClis Tag BalCodigo
		Index On RClis Tag BalNome

		ThisForm.pagina.dados.grade.RecordSource=''
		ThisForm.pagina.dados.grade.column1.ControlSource=''
		ThisForm.pagina.dados.grade.column2.ControlSource=''
		ThisForm.pagina.dados.grade.column3.ControlSource=''
		ThisForm.pagina.dados.grade.column4.ControlSource=''
		ThisForm.pagina.dados.grade.column5.ControlSource=''

		=DoDefault()

		ThisForm.Pagina.Lista.Grade.Column1.SetFocus
	Else
		=Messagebox('Impossivel Efetuar Conexão com Servidor de Banco de Dados...', 0+48, 'CreateObject')
		Return .f.
	EndIf
EndWith
* Fim **************************************************************************************
ENDPROC

* PROCEDURE Pagina.Lista.Grupo_op.Click
------------------------------------------------------------
PROCEDURE Pagina.Lista.Grupo_op.Click
LPARAMETERS opcao

If !DoDefault( opcao )
	ThisForm.mAtivapagina1
	Return
Endif

ThisForm.Pagina.Dados.Grade.Refresh
Do Case
	Case Inlist( ThisForm.pcEscolha, 'INSERIR', 'PROCURAR' )
		ThisForm.Pagina.Dados.GetNome.SetFocus
	Case Inlist( ThisForm.pcEscolha, 'EXCLUIR', 'CONSULTAR' )
		ThisForm.Pagina.Dados.Grupo_Salva.Salva.SetFocus
	Case ThisForm.pcEscolha = 'ALTERAR'
		ThisForm.Pagina.Dados.Get_Grupo.SetFocus
Endcase

ThisForm.pagina.dados.grade.RecordSource = ''

If not Inlist( ThisForm.pcEscolha, 'INSERIR', 'PROCURAR' )
	ThisForm.IniciaDesc()
	sNome=ThisForm.pagina.dados.getNome.value
	ThisForm.podatamgr.Requery('crSigCdDpr')
Else
	Zap in ('crSigCdDpr')
EndIf

ThisForm.pagina.dados.grade.RecordSource='crSigCdDpr'
ThisForm.pagina.dados.grade.column1.ControlSource='crSigCdDpr.Ordems'
ThisForm.pagina.dados.grade.column1.ReadOnly=.F.
ThisForm.pagina.dados.grade.column2.ControlSource='crSigCdDpr.grupos'
ThisForm.pagina.dados.grade.column2.ReadOnly=.F.
ThisForm.pagina.dados.grade.column3.ControlSource='crSigCdDpr.contas'
ThisForm.pagina.dados.grade.column3.ReadOnly=.F.
ThisForm.pagina.dados.grade.column4.ControlSource='crSigCdDpr.tpops'
ThisForm.pagina.dados.grade.column4.ReadOnly=.F.
ThisForm.pagina.dados.grade.column5.ControlSource='crSigCdDpr.operas'
ThisForm.pagina.dados.grade.column5.ReadOnly=.F.

Go Top in ('crSigCdDpr')
Thisform.Pagina.Dados.Refresh

ENDPROC

------------------------------------------------------------
* PROCEDURE When
------------------------------------------------------------
PROCEDURE When
Do Case 
	Case ThisForm.pcEscolha = 'PROCURAR'
		Return .t.
	Case ThisForm.pcEscolha <> 'INSERIR'
		Return .f.
	Case ThisForm.pcEscolha = 'INSERIR'
		Select crSigCdDpr
		Count to lnLinhas
		If lnLinhas > 1
			Return .f.
		Endif
		Go Top
		If ( lnLinhas = 1 And IsEmpty( crSigCdDpr.Grupos ) And IsEmpty( crSigCdDpr.Contas ) ) Or lnLinhas = 0
			Return .t.
		Else
			Return .f.
		Endif
Other
	Return .f.
Endcase

ENDPROC

------------------------------------------------------------
* PROCEDURE Valid
------------------------------------------------------------
PROCEDURE Valid
If IsEmpty( This.Value ) And LastKey() # 27
	Messagebox( 'O Nome deve ser informado...', 32, '' )
	Return 0
Endif

Select crSigCdDpr
Go Top

If Eof()
	ThisForm.InsiraRegGrade()
	Go Top
Endif

ENDPROC
