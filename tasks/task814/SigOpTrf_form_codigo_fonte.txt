PROCEDURE Init
LParameters poForm, pnDataSes

Set DataSession to pnDataSes

Thisform.ParentForm          = poForm
Thisform.ParentForm.Enabled  = .F.

Thisform.GravaDados = .f.

With Thisform
	.Caption = 'Operações Automáticas P/ ' + AllTrim(crSigCdOpe.Dopes)
EndWith

ThisForm.cntSombra.lblSombra.Caption = ThisForm.Caption
ThisForm.cntSombra.lblTitulo.Caption = ThisForm.Caption

lcOperacao = crSigCdOpe.Dopes

WITH Thisform
	.poDatamgr = CreateObject('fSqlConector')

	If Not .poDatamgr.pnIdconn>0
		Messagebox('Impossivel Efetuar Conexão com Servidor de Banco de Dados.',(0+48),goSistema.TituloSistema)
		Return(.f.)
	Endif

	With .poDatamgr
		.AddCursor('SigOpTrf','cIdChaves','crSigOpTrf','Dopes',.f.,[Select * From SigOpTrf Where Dopes = '] + crSigCdOpe.Dopes + [' Order by Dopes])
		.AddCursor('SigCdOpe','Dopes','TmpOpe','Dopes',.f.,[Select Dopes,nDopes From SigCdOpe Where Dopes <> '] + crSigCdOpe.Dopes + [' order by Dopes])
        .ReQuery('crSigOpTrf')
        .ReQuery('TmpOpe')
		SELECT crSigOpTrf
		GO top
	EndWith
Endwith	

Create Cursor xTrf (Dopes c(20), Dopeds c(20), Cpros c(14), Dpros c(60))
Index on Cpros Tag Cpros
Index on Dopes + Dopeds Tag Dopes
Set Order to

Create Cursor xOpe ( Dopes c(20) )
Index on Dopes Tag Dopes
Set Order to

SELECT TmpOpe 
Index on Dopes Tag Dopes

With Thisform.GradeOpe
	.RecordSource          = 'xOpe'
	.Column2.ControlSource = 'xOpe.Dopes'
	.Refresh
EndWith

With Thisform.GradePro
	.RecordSource         = 'xTrf'
	ThisForm.GradePro.col_cpros.ControlSource = 'xTrf.Cpros'
	ThisForm.GradePro.col_dpros.Controlsource = 'xTrf.Dpros'
ENDWITH

Select crSigOpTrf
Scan While Dopes = crSigCdOpe.Dopes
	Scatter Memvar
	
	ThisForm.podatamgr.SqlExecute([Select cPros, dPros from SigCdPro Where Cpros = ?m.cPros],'CrSigCdPro')

	Insert Into xTrf ( Dopes, Dopeds, Cpros, Dpros ) ;
		Value (m.Dopes, m.Dopeds, m.Cpros, crSigCdPro.Dpros)

	Select xOpe
	Set Order to Dopes
	
	LOCATE FOR Dopes = m.Dopeds
	IF EOF()
		Insert Into xOpe ( Dopes ) Value ( m.Dopeds )
	Endif

EndScan

Select xOpe
Set Order to
Go Top

Select xTrf
Set Order to Dopes
Go Top

SET KEY TO crSigCdOpe.Dopes+xOpe.Dopes
Go Top
ThisForm.GradePro.Refresh

ENDPROC

PROCEDURE Valid
Local lcFpags, lnRecNo, loBrowse

If IsEmpty(This.Value)
	Return .f.
Endif

If Not IsEmpty(This.Value) And Thisform.AntValue # This.Value
	
	ThisForm.GravaDados = .t.

	If Not Seek(This.Value, 'TmpOpe', 'Dopes')
		loBrowse = CreateObject('fwBuscaInt','TmpOpe','Dopes',This.Value,'Operações',.t.)
		loBrowse.mAddColuna('Dopes', '', 'Operação', .t.)
		loBrowse.mAddColuna('nDopes', '9999', ' Cod ',.f.)
		loBrowse.Show		
		This.Value = Iif( Lastkey() == 27, '', TmpOpe.Dopes )
		This.Refresh
	EndIf
	lcdope = This.Value
	If !IsEmpty(lcDope)
		Select xOpe
		If Not Eof() Or Bof()
			lnRecNo = RecNo('xope')
		Else 
			lnRecNo = 0
		EndIf
		Set Order to Dopes
		Seek (lcDope)
		If Not Eof()
			Skip
			If xOpe.Dopes = lcDope
				This.Value = ''
				This.Refresh
				Delete
				Skip
				Skip -1
				MessageBox('Operação já Digitada ... ',48,'')
			EndIf
		Endif
		Select xOpe
		Set Order to Dopes
		If lnRecNo # 0
			Goto (lnRecNo)
		Else
			Goto Bottom
		EndIf
		This.Refresh
	EndIf
EndIf
ENDPROC

PROCEDURE When
Thisform.AntValue = This.Value

PROCEDURE Click
If ThisForm.GravaDados

	SELECT crSigOpTrf
	DELETE ALL

	Select xTrf
	SET KEY TO 
	Set Order To
	Scan
		If !IsEmpty(xTrf.Dopes) And !IsEmpty(xTrf.Dopeds) And !IsEmpty(xTrf.Cpros)
			Scatter memvar
			Insert Into crSigOpTrf From memvar
			Replace crSigOpTrf.cIdChaves WITH fUniqueIds()
		EndIf
	EndScan
	
	If (Reccount([crSigOpTrf]) = 0) 
		Select xOpe
		Scan
			Insert Into crSigOpTrf (Dopes, DopeDs, cIdChaves) ;
							Values (crSigCdOpe.Dopes, xOpe.Dopes, fUniqueIds())
		EndScan
	EndIf
	
	m.lnErro1 = ThisForm.poDatamgr.pnIdConn

	With ThisForm
		If m.lnErro1>0 
			m.lnErro1 = Iif(.poDataMgr.UpDate('crSigOpTrf'),1,-1)
		EndIf 
		If m.lnErro1>0 
			m.lnErro1 = .poDataMgr.Commit()
		EndIf 
		If Not m.lnErro1>0 
			.poDataMgr.RollBack()
		EndIf 
	EndWith 
	If Not m.lnErro1>0 
		= MessageBox('Erro na Gravação dos Dados do Cadastro de Operações.',48)
	EndIf 

EndIf
Thisform.ParentForm.Enabled = .T.
ThisForm.Release

ENDPROC

PROCEDURE Valid
If !IsEmpty(this.value)
	ThisForm.GravaDados = .t.

	If !ThisForm.podatamgr.CursorQuery('SigCdPro','crSigCdPro', 'cPros', This.Value)
		xBrox = CreateObject('fwBuscaExt',Thisform.poDatamgr.pnIdconn,'SigCdPro','crSigCdPro','cPros',This.Value,'Cadastro de Produtos',.t.,'cPros')

			xBrox.mAddColuna('DPros', '', 'Descrição', .t.)
		xBrox.Show()
		EndIf 
	Endif
	This.Value = evaluate('crSigCdPro.CPROS')
	This.Parent.Parent.col_dpros.txt_dpros.Value = crSigCdPro.Dpros
	This.Parent.Parent.col_dpros.txt_dpros.Refresh

	cPro = This.Value
	
	Select xTrf
	_Recno = Recno()
	Set Order to Cpros
	Seek cPro
	If !Eof()
		lcdope = xTrf.Dopeds
		Skip
		If Cpros = Cpro
			This.Value = ''
			This.Refresh
			Messagebox('Produto já Digitado na Operação '+lcDope,48,'')
		Endif
	Endif
	Set Order to Dopes
	Set Key to crSigCdOpe.Dopes + xOpe.Dopes
	If _Recno # 0
		go _Recno
	Endif
Endif

ENDPROC
