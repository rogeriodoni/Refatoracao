PROCEDURE impetiqueta
lnPrinters = APrinters(laPrinters)
If (lnPrinters <> 0)
	For lnI = 1 to lnPrinters
		laPrinters(lnI, 1) = Upper(laPrinters(lnI, 1))
	EndFor
	=ASort(laPrinters)
	lnTam = Len(laPrinters(1, 1))
Else
	=MessageBox('Não Foi Encontrada Nenhuma Impressora!!!', 48, '')
	Return 0
EndIf

lcCpo = 'OcoEtiqs, Ouros, nMaxTpEtis, TpEtiPads, nMaxImpEti, ImpEtis, AjVerts, AjHorzs, LibObss, TpInstalas'
ThisForm.poDataMgr.CursorQuery('SigCdPam', 'LocalParam', .F., .F., lcCpo)

ThisForm.poDataMgr.CursorQuery('SigCdmp', 'LocalCImp')
Select LocalCImp
Index On Impres Tag skImpres

lnImpres = 1
If (lnPrinters <> 0)
	Select LocalCImp
	Set Order To skImpres
	Scan For LocalCImp.nTpImpres = 2
		Dimension laImpOk(lnImpres)
		lnOk = AScan(laPrinters, Alltrim(Upper(LocalCImp.Impres)))
		If (lnOk <> 0)
			laImpOk(lnImpres) = laPrinters(lnOk)
			lnImpres = lnImpres + 1
		EndIf
	EndScan
EndIf
lnImpres = lnImpres - 1

With ThisForm.Pagina.Lista.cntEtiquetas.optImpressora
	lnTop 	 = .Top
	lnHeight = 15
	If (lnImpres <> 0)
		.ButtonCount = Min(lnImpres, Max(LocalParam.nMaxImpEti, 3))
		For lnI = 1 To lnImpres
			lcI = Alltrim(Str(lnI))
			With .Option&lcI.
				.AutoSize = .F.
				.Caption  = ' \<' + lcI + '. ' + laImpOk(lnI)
				.FontSize = 8
				.Tag      = Upper(laImpOk(lnI))
				.Top      = lnTop
				.Width    = 180
			EndWith
			lnTop 	 = lnTop + 20
			lnHeight = lnHeight + 20
		EndFor
	Else
		.ButtonCount = 1
		lnOk = AScan(laPrinters, 'GENÉRICO/SOMENTE TEXTO')
		With .Option1
			.AutoSize = .F.
			.Caption  = ' \<0. Genérico/Somente Texto'
			.Enabled  = (lnOk <> 0)
			.FontSize = 8
			.Tag      = 'GENÉRICO/SOMENTE TEXTO'
			.Width    = 180
			lnImpres  = Iif(.Enabled, 1, 0)
		EndWith
		lnHeight = lnHeight + 20
	EndIf
	.Enabled = (lnImpres > 1)
	.Height  = lnHeight
EndWith

ThisForm.Pagina.Lista.cntEtiquetas.spnVertical.Value   = LocalParam.AjVerts
ThisForm.Pagina.Lista.cntEtiquetas.spnHorizontal.Value = LocalParam.AjHorzs

ThisForm.Pagina.Lista.cntEtiquetas.Visible 	= .t.
ThisForm.Pagina.Lista.cntEtiquetas.Enabled 	= .t.
ThisForm.Pagina.Lista.Grade.Enabled 		= .f.
ThisForm.Pagina.Lista.Grupo_op.Enabled 		= .f.
ThisForm.Pagina.Lista.Grupo_saida.Enabled 	= .f.
ThisForm.Pagina.Lista.cmdEtiquetas.Enabled 	= .f.

ThisForm.Pagina.Lista.cntEtiquetas.getCodigos.SetFocus
ThisForm.Pagina.Lista.cntEtiquetas.getQtde.Value = 1
* Fim **************************************************************************************
ENDPROC

Use In crBusca

PROCEDURE Click
* Impressão das Etiquetas
lcCod = ThisForm.Pagina.Lista.cntEtiquetas.getCodigos.Value
lnQtd = ThisForm.Pagina.Lista.cntEtiquetas.getQtde.Value
lnTip = ThisForm.Pagina.Lista.cntEtiquetas.optTipo.Value
lnDen = Int(ThisForm.Pagina.Lista.cntEtiquetas.spnDensidade.Value)
lnHor = Int(ThisForm.Pagina.Lista.cntEtiquetas.spnHorizontal.Value)
lnVer = Int(ThisForm.Pagina.Lista.cntEtiquetas.spnVertical.Value)
lnImp = ThisForm.Pagina.Lista.cntEtiquetas.optImpressora.Value
lcImp = ThisForm.Pagina.Lista.cntEtiquetas.optImpressora.Buttons(lnImp).Tag

If Empty(lcCod)
	ThisForm.poDataMgr.CursorQuery('SigPrLcl', 'LocalPrLoc', 'Etiquetas', 1, 'Codigos')
	If (Reccount('LocalPrLoc') < 1)
		=MessageBox('Não Existem Localizações a Serem Impressas!!!', 0+48 , 'Atenção!!!')
		Return 0
	EndIf
Else
	ThisForm.poDataMgr.CursorQuery('SigPrLcl', 'LocalPrLoc', 'Codigos', Padr(lcCod, 6), 'Codigos')
	If (Reccount('LocalPrLoc') < 1)
		=MessageBox('Não Existem Localizações a Serem Impressas!!!', 0+48 , 'Atenção!!!')
		Return 0
	EndIf
	Zap In LocalPrLoc
	For I = 1 To lnQtd
		Insert Into LocalPrLoc (Codigos) Values (lcCod)
	Next
	lnQtd = 1
EndIf

Insert Into LocalPrLoc (Codigos) Values ('*FIM*')

Set Printer To Name (lcImp)
Set Device To Printer

For lnVezes = 1 To lnQtd
	lnCnt = 1
	Select LocalPrLoc
	Scan
		If (lnCnt = 1)
			??? Chr(02) + 'L' + Chr(13)
			??? 'H' + Alltrim(Str(lnDen)) + Chr(13)
			??? 'D11' + Chr(13)
		EndIf
		
		Do Case
			Case (lnTip = 1) && Pequena
				lnInc = ((85 * lnCnt) - 85)

				If (LocalPrLoc.Codigos <> [*FIM* ])
					lcLin = 015 + lnHor
					lcCol = 010 + lnVer + lnInc
					??? '1e42034' + Padl(lcLin, 4, '0') + Padl(lcCol, 4, '0') + 'B' + Padr(LocalPrLoc.Codigos, 3) + Chr(13)

					lcLin = 005 + lnHor
					lcCol = 036 + lnVer + lnInc
					??? '1911001' + Padl(lcLin, 4, '0') + Padl(lcCol, 4, '0') + Padr(LocalPrLoc.Codigos, 3) + Chr(13)
				Else
					lcLin = 005 + lnHor
					lcCol = 036 + lnVer + lnInc
					??? '1911001' + Padl(lcLin, 4, '0') + Padl(lcCol, 4, '0') + Padr(LocalPrLoc.Codigos, 5) + Chr(13)
				EndIf

			Case (lnTip = 2) && Grande
				lnInc = ((89 * lnCnt) - 89)

				If (LocalPrLoc.Codigos <> [*FIM* ])
					lcLin = 022 + lnHor
					lcCol = 050 + lnVer + lnInc
					??? '4e42034' + Padl(lcLin, 4, '0') + Padl(lcCol, 4, '0') + 'B' + Padr(LocalPrLoc.Codigos, 3) + Chr(13)

					lcLin = 045 + lnHor
					lcCol = 065 + lnVer + lnInc
					??? '4911002' + Padl(lcLin, 4, '0') + Padl(lcCol, 4, '0') + Padr(LocalPrLoc.Codigos, 3) + Chr(13)
				Else
					lcLin = 045 + lnHor
					lcCol = 065 + lnVer + lnInc
					??? '1911001' + Padl(lcLin, 4, '0') + Padl(lcCol, 4, '0') + Padr(LocalPrLoc.Codigos, 5) + Chr(13)
				EndIf
		EndCase
		lnCnt = lnCnt + 1
		If (lnCnt > 4) Or (LocalPrLoc.Codigos = [*FIM* ])
			lnCnt = 1
			??? 'Q0001' + Chr(13)
			??? 'E' + Chr(13)
		EndIf
	EndScan
Next

Set Printer To Default
Set Device To Screen

* Retornando Para a Grade
ThisForm.Pagina.Lista.cntEtiquetas.Visible 	= .f.
ThisForm.Pagina.Lista.cntEtiquetas.Enabled 	= .f.
ThisForm.Pagina.Lista.Grade.Enabled 		= .t.
ThisForm.Pagina.Lista.Grupo_op.Enabled 		= .t.
ThisForm.Pagina.Lista.Grupo_saida.Enabled 	= .t.
ThisForm.Pagina.Lista.cmdEtiquetas.Enabled 	= .t.
ThisForm.Refresh
ThisForm.Pagina.Lista.Grade.SetFocus
* Fim *********************************************************************************

ENDPROC
