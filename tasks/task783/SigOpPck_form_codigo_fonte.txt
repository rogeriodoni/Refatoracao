PROCEDURE acertacolunas
Lparameters pTip

With ThisForm.pgfPrincipal.pgDados.pgResults.pgRes01.grdDados1
	If (pTip = [LER])
		lcStr = ThisForm.pgfPrincipal.pgFiltros.getCfgGrade.Value 
		
		lnPos = 1 && Posição Inicial de Cada Bloco Correspondente a Uma Coluna da Grade
		For lnCnt = 1 To .ColumnCount
			lcBlo = Substr(lcStr, lnPos, 5) 		&& Bloco Com 5 Posições: 1 a 3 = Width; 	
													&&						 4     = Visible	
													&&						 5     = Uso Futuro	
			lnWid = Int(Val(Substr(lcBlo, 1, 3)))	&& Largura da Coluna						
			llVis = Not (Substr(lcBlo, 4, 1)=[0])	&& Coluna Visible=1 (default); Invisible=0	
			lcCol = Alltrim(Str(lnCnt))				&& Número da Coluna							

			.Column&lcCol..Width   = Iif(lnWid > 0, lnWid, .Column&lcCol..Width) && Se Não Estiver Gravado Mantém a Largura Original
			.Column&lcCol..Visible = llVis										 && Se Não Estiver Gravado Mantém Visible			

			lnPos = lnPos + 5 && Incrementa de 5 em 5 Posições a Cada Coluna
		Next
	Else && [GRAVAR]
		lcStr = []
		For lnCnt = 1 To .ColumnCount
			lcCol = Alltrim(Str(lnCnt))										&& Número da Coluna	
			lcStr = lcStr + Padl(.Column&lcCol..Width, 3, [0]) + ;
							Iif(.Column&lcCol..Visible, [1], [0]) + ;
							[ ] && Nota: o Branco é Para Uso Futuro
		Next
		ThisForm.pgfPrincipal.pgFiltros.getCfgGrade.Value = lcStr
	EndIf
EndWith
* Fim **************************************************************************************
ENDPROC

* PROCEDURE buscaantigos
------------------------------------------------------------
PROCEDURE buscaantigos
Private pdDtA
Local lcPro, lcTam

Select crCursor1
oProg2 = CreateObject([fwProgressBar], [Processando Dados Anteriores...], Reccount([crCursor1]))
oProg2.Top = oProg2.Top + Int(oProg2.Height/2)
oProg2.Rodape.Caption = [Buscando Produtos na Data de Movimentação]
oProg2.Show
Scan
	oProg2.Update(.t.)

	lcPro = Nvl(crCursor1.Codigo_Item, [])
	lcTam = Nvl(crCursor1.Tamanho, [])
	pdDtA = Nvl(crCursor1.Data_Venda, Datetime())
		
	lcQuery = [Select Top 1 ] + ;
					 [p.Mercs as Gde_Grupo, ] + ; 
					 [p.CGrus as Grupo, ] + ; 
					 [p.SGrus as SubGrupo, ] + ; 
					 [p.CustoFs as Custo_Cadastro, ] + ; 
					 [p.MoeCusfs as Moe_Cus_Cadastro, ] + ; 
					 [p.MarkupA as Markup_Cadastro, ] + ; 
					 [p.Reffs as Ref_Fornecedor, ] + ;
					 [p.PVens as Venda_Cadastro, ] + ; 
					 [p.PVens as Desc_Acres, ] + ; 
					 [p.PVens as Venda_CadastroR, ] + ; 
					 [p.Moevs as Moe_Vda_Cadastro, ] + ; 
					 [p.FigJpgs as Imagem, ] + ;
					 [p.IFors as Cod_Fornecedor ] + ;
				[From SigCdPrc p ] + ;
			   [Where p.CPros = '] + lcPro + [' And ] + ;
			   		 [p.DataAlts >= ?pdDtA ] + ;
			   [Order By p.DataAlts Desc]

	If Not fExecCmds(ThisForm.poDataMgr, [crAntigo1], lcQuery)
		oProg2.Complete
		Return .f.
	EndIf
	
	Select crAntigo1
	Go Top In crAntigo1
	If Not Eof([crAntigo1])
		Replace Gde_Grupo 		 With crAntigo1.Gde_Grupo, ; 
				Grupo 			 With crAntigo1.Grupo, ; 
				SubGrupo 		 With crAntigo1.SubGrupo, ; 
				Custo_Cadastro 	 With crAntigo1.Custo_Cadastro, ; 
				Moe_Cus_Cadastro With crAntigo1.Moe_Cus_Cadastro, ; 
				Markup_Cadastro  With crAntigo1.Markup_Cadastro, ; 
				Ref_Fornecedor 	 With crAntigo1.Ref_Fornecedor, ; 
				Venda_Cadastro 	 With crAntigo1.Venda_Cadastro, ; 
				Desc_Acres 		 With crAntigo1.Desc_Acres, ; 
				Venda_CadastroR  With crAntigo1.Venda_CadastroR, ; 
				Moe_Vda_Cadastro With crAntigo1.Moe_Vda_Cadastro, ; 
				Imagem 			 With crAntigo1.Imagem, ; 
				Cod_Fornecedor 	 With crAntigo1.Cod_Fornecedor In crCursor1

		lcQuery = [Select Top 1 ] + ;
						 [t.Percs as Markup_Tamanho, ] + ; 
						 [t.Valor as Valor_Tamanho, ] + ; 
						 [t.PesoMs as Peso_Tamanho ] + ;
					[From SigPrTa2 t ] + ;
				   [Where t.CPros = '] + lcPro + [' And ] + ;
				   		 [t.CodTams = '] + lcTam + [' And ] + ;
				   		 [t.DataAlts >= ?pdDtA ] + ;
				   [Order By t.DataAlts Desc]

		If Not fExecCmds(ThisForm.poDataMgr, [crAntigo1], lcQuery)
			oProg2.Complete
			Return .f.
		EndIf
		
		Select crAntigo1
		Go Top In crAntigo1
		If Not Eof([crAntigo1])
			Replace Markup_Tamanho With crAntigo1.Markup_Tamanho, ;
					Valor_Tamanho  With crAntigo1.Valor_Tamanho, ;
					Peso_Tamanho   With crAntigo1.Peso_Tamanho In crCursor1
		EndIf
	EndIf
	Use In crAntigo1
EndScan
oProg2.Complete
* Fim **************************************************************************************

ENDPROC

Private pDtI, pDtF

With Thisform.pgfPrincipal
	ldDtI = .pgFiltros.getDcI.Value
	ldDtF = .pgFiltros.getDcF.Value
	lcEmp = .pgFiltros.getEmpOs.Value
	lcMoe = []

	pDtI = fDtoSQL(ldDtI)
	pDtF = fDtoSQL(ldDtF, [23:59:59])
	
	lcMov = .pgFiltros.getOperacao.Value
	
	Go Top In crTpOpe
	.pgFiltros.grdTipo.Refresh
Endwith

Select CCodigos ;
	From crTpOpe ;
	Where nMarcas = 1 ;
	Into Cursor LocalTpOpe

If Empty(lcMov)
	Select LocalTpOpe
	Go Top In LocalTpOpe
	If (Reccount([LocalTpOpe]) = 1)
		lcTpO = [o.TipoOps = ] + LocalTpOpe.CCodigos
	Else
		lcVir = []
		lcTpO = [o.TipoOps In (]
		Scan
			lcTpO = lcTpO + lcVir + LocalTpOpe.CCodigos
			lcVir = [, ]
		Endscan
		lcTpO = lcTpO + [)]
	Endif
Else
	lcTpO = [ c.Dopes = ']+lcMov+[' ]
EndIf 
oProg1 = Createobject([fwProgressBar], [Filtrando os Dados Selecionados...], 6)
oProg1.Top = oProg1.Top - Int(oProg1.Height/2)
oProg1.Titulo.FontBold = .T.
oProg1.Rodape.Caption = [Aguarde os Processamentos...]
oProg1.Show

This.Refresh

oProg1.SubTitulo.Caption = [01/06: Seleção de Dados]
oProg1.Update(.T.)

lcQuery = [Select c.empds as Destino, c.emps as Empresa, c.Dopes as Movimentacao, c.Numes as Numero, c.Datas as Data_Venda, i.CPros as Codigo_Item, ] + ;
	[IsNull(p.cbars,0) as Barra_Geral, case when isnull(i.codbarras,0) = 0 then isnull(s.CodBarras,0)  else isnull(i.codbarras,0) end as Barra_Individual, ] + ;
	[rtrim(isnull(p.SGrus,''))+' - '+rtrim(IsNull(sg.Descricaos,'')) as SubGrupo,  p.Reffs as Ref_Fornecedor, isnull(s.CodTams,'') as Tamanho, isnull(s.CodCors, IsNull(p.codcors,'')) as Cor, ] + ;
	[isnull(x.iclis,'') as Cod_Cliente, isnull(x.RClis,'') as Nome_Cliente, isnull(i.Pedidoweb,'') as PedidoWeb, isnull(e.local,'') as Local, ] + ;
	[isnull(p.dpro2s,'') as Descritivo, i.DPros as Desc_Item, i.Qtds as Quantidade, isnull(e.qtde,0) as Qtde, ] + ;
	[rtrim(isnull(p.colecoes,''))+' - '+isnull(co.descs,'') as Colecoes, p.figjpgs as Imagem, c.ContaOs as ContaOrigem ] + ;
	[From SigMvCab c ] + ;
	[Left Join SigCdOpe o On c.Dopes = o.Dopes ] + ;
	[Left Join SigWbPed w On c.PedidoWeb = w.Pedido ] + ;
	[left join SigCdCli x on w.cpfcli = replace(replace(replace(x.cpfs,'-',''),'.',''),'/','') ] + ;
	[LEFT JOIN (select max(datapgt) as DtPagto, Pedido from sigwbpar group by pedido ) n on c.pedidoweb = n.pedido ] + ;
	[Left Join SigMvItn i On c.EmpDopNums = i.EmpDopNums ] + ;
	[Left Join SigMvIts s On i.EmpDopNums = s.EmpDopNums And i.CItens = s.CItens and i.cpros = s.cpros ] + ;
	[Left Join SigPrTam t On s.CPros = t.CPros And s.CodTams = t.CodTams ] + ;
	[Left Join SigCdPro p On i.CPros = p.CPros ] + ;
	[Left Join SigCdCli c1 On c1.ICLis = c.ContaDs ] + ;
	[Left Join SigCdCli c2 On c2.ICLis = p.IFors ] + ;
	[Left Join SigCdPsg sg On sg.CGruCods = p.CGrus + p.SGrus ] + ;
	[left join SigCdCol Co on p.colecoes = co.colecoes ] + ;
	[left join (select a.cpros, a.codtams, a.codcors, a.empos as emps, a.localizas as local, sum(qtde) as qtde ] + ;
	[from (Select a.cpros, a.codtams, a.codcors, a.empos, localizas, Count(1) As qtde From SigOpEtq A  ] + ;
	[left join (Select Cpros,Cgrus,Mercs,Colecoes,SGrus from SigCdPro) B on b.cpros = a.cpros ] + ;
	[Where a.empos = '] + lcEmp + ['  And a.grupos <> space(10) and a.contas <> space(10) ] + ;
	[Group By a.cpros, a.codtams, a.codcors, a.empos, localizas having count(1) > 0 ] + ;
	[union all ] + ;
	[Select a.cpros, '' as codtams, '' as codcors, a.emps, a.locals, sum(sqtds) as qtde From SigMvesl a, SigCdpro b, SigCdUni c ] + ;
	[Where a.CPros = b.Cpros And b.Cunis = c.Cunis  and c.Etiqs <> 'S' And  a.emps = '] + lcEmp + [' ] + ;
	[group by a.cpros, a.locals, a.grupos, a.estos, a.emps having sum(sqtds) > 0) a ] + ;
	[group by a.cpros, a.codtams, a.codcors, a.empos, a.localizas) e on i.cpros = e.cpros and isnull(s.CodTams,'') = e.codtams ] + ;
	[Where ] + lcTpO + [ And c.Emps = '] + lcEmp + [' And c.Datas Between ?pDtI And ?pDtF ] + ;
	[And c.pStatus IN( '','N') ]

If Not fExecCmds(Thisform.poDataMgr, [crCursor1], lcQuery)

	Return .F.
Endif

With ThisForm
	.poDataMgr = CreateObject([fSqlConector], .Name)

		.goOpCge = CreateObject([Relation])
		.goOpCge.AddProperty([SetPoint], Set([Point]))
		.goOpCge.AddProperty([SetSepar], Set([Separator]))
		
		.poObjExt1 = Null
		.poObjExt2 = Null

		.pgfPrincipal.pgFiltros.getDcI.Value = Date()
		.pgfPrincipal.pgFiltros.getDcF.Value = Date()
		
		.pgfPrincipal.pgFiltros.getOperacao.Value = []
		
		If Not fChecaAcesso([SIGOPPCK],[CONPORTIPO]) And (Usuar <> [4CONTROL])
			.pgfPrincipal.pgFiltros.Label1.Visible = .F.
			.pgfPrincipal.pgFiltros.grdTipo.Visible = .F.
			.pgfPrincipal.pgFiltros.btnMarca.Visible = .F.
			.pgfPrincipal.pgFiltros.btnDesmarca.Visible = .F.
			
			.pgfPrincipal.pgFiltros.Say2.Visible = .T.
			.pgfPrincipal.pgFiltros.getOperacao.Visible = .T.
		Else
			.pgfPrincipal.pgFiltros.Label1.Visible = .T.
			.pgfPrincipal.pgFiltros.grdTipo.Visible = .T.
			.pgfPrincipal.pgFiltros.btnMarca.Visible = .T.
			.pgfPrincipal.pgFiltros.btnDesmarca.Visible = .T.
			
			.pgfPrincipal.pgFiltros.Say2.Visible = .F.
			.pgfPrincipal.pgFiltros.getOperacao.Visible = .F.
		EndIf 

		
		* Empresa
		lcQuery = [Select * From SigCdEmp where Cemps= ']+_Empr+[' ]
		If Not fExecCmds(.poDataMgr, [crSigCdEmp], lcQuery)
			Return .f.
		EndIf
		
		.pgfPrincipal.pgFiltros.getEmpOs.Value = _Empr
		.pgfPrincipal.pgFiltros.getDEmpOs.Value = crSigCdEmp.Razas
		
		.AddObject([poSigSyPar], [fGravaUserXML])
		With .poSigSyPar
			.pGrvXMLCfg	 = ThisForm.Name
			.pGrvXMLDsc  = []
			.pGrvXMLUsu  = Usuar
			.pGrvXMLObj  = ThisForm.pgfPrincipal.pgFiltros
			lcIds = Upper(Alltrim(Nvl(.pGrvXMLCfg, [])))
		EndWith
		.poSigSyPar.mLerCfg()
		
		.pgfPrincipal.pgFiltros.getCfgGrade.Visible = (Usuar = [4CONTROL])
		
		* Grade de Tipos de Operações
		lcQuery = [Select * From SigCdPac]
		If Not fExecCmds(.poDataMgr, [crSigCdPac], lcQuery)
			Return .f.
		EndIf
		
		* Grade de Tipos de Operações
		lcQuery = [Select Codigos, Descrs, GerGdmis From SigCdTom]
		If Not fExecCmds(.poDataMgr, [crSigCdTom], lcQuery)
			Return .f.
		EndIf

		Select Iif(GerGdmis = 1, 1, 0) as nMarcas, ;
			   Str(a.Codigos,2) as CCodigos, ;
			   Padr(Str(a.Codigos,2) + [-] + a.Descrs, 50) as CDescricaos ;
		  From crSigCdTom a ;
		 Order By 2 ;
		  Into Cursor crTpOpe ReadWrite
		Use In crSigCdTom
		
		* Grade de Grande Grupo de Produtos
		lcQuery = [Select Codigos, Descs From SigCdGpr]
		If Not fExecCmds(.poDataMgr, [crSigCdGpr], lcQuery)
			Return .f.
		EndIf

		Select 1 as nMarcas, a.Codigos, ;
			   a.Codigos + [-] + a.Descs as CDescricaos ;
		  From crSigCdGpr a ;
		 Order By 2 ;
		  Into Cursor crGrdGpr ReadWrite
		Use In crSigCdGpr

		* Grade de Grupo de Produtos
		lcQuery = [Select Mercs, CGrus as Codigos, DGrus as Descs From SigCdGrp]
		If Not fExecCmds(.poDataMgr, [crSigCdGrp], lcQuery)
			Return .f.
		EndIf

		Select 1 as nMarcas, a.Mercs, a.Codigos, ;
			   a.Codigos + [-] + a.Descs as CDescricaos ;
		  From crSigCdGrp a ;
		 Order By 2 ;
		  Into Cursor crGrdGru ReadWrite
		Use In crSigCdGrp
		
		.MontaGrades()

		=DoDefault()

		lcQuery = [Select Distinct a.Grupo as GruEmps, b.Tipos ] + ;
					[From SigCdEmp a, SigCdGre b ] + ;
				   [Where a.Grupo = b.Grupo And ] + ;
				   		 [a.CEmps In (Select g.Emps From SigCdAce g Where g.Usuarios = '] + Usuar + [')]
		If Not fExecCmds(.poDataMgr, [crSigCdGre], lcQuery)
			Return .f.
		EndIf

		.pgfPrincipal.pgFiltros.getDcI.SetFocus
	Else
		=Messagebox([Impossível Efetuar Conexão Com o Servidor de Banco de Dados...], 0+48, [CreateObject])
		Return .f.
	EndIf
EndWith
