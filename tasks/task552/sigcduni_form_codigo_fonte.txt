------------------------------------------------------------
* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click

If InList(ThisForm.pcEscolha,'INSERIR','ALTERAR')
	If IsEmpty( ThisForm.Pagina.Dados.Get_Unid.Value )
		Messagebox( 'Unidade não pode ficar em branco!', 48, '' )
		ThisForm.Pagina.Dados.Get_Unid.SetFocus
		Return
	Endif
	If (ThisForm.pcEscolha = 'INSERIR' And ThisForm.poDataMgr.ChkRegister('SigCdUni','cunis',ThisForm.Pagina.Dados.Get_Unid.Value))
		Messagebox( 'Unidade já Cadastrada!', 48, '' )
		ThisForm.Pagina.Dados.Get_Unid.SetFocus
		Return
	Endif
	If IsEmpty( ThisForm.Pagina.Dados.Get_Desc.Value )
		Messagebox( 'Descrição Não pode ficar em branco.', 48, '' )
		ThisForm.Pagina.Dados.Get_Desc.SetFocus
		Return
	Endif
Else
	If ThisForm.pcEscolha = 'EXCLUIR'
		If ThisForm.poDataMgr.ChkRegister('SigCdPro','cunis',ThisForm.Pagina.Dados.Get_Unid.Value)
			Messagebox( 'Existe Produto Cadastrado com essa Unidade !', 48, 'Impossível Exclusão' )
			ThisForm.Pagina.Dados.Grupo_Salva.Cancelar.SetFocus
			Return
		Endif

		If ThisForm.poDataMgr.ChkRegister('SigCdGrp','cUniPs',ThisForm.Pagina.Dados.Get_Unid.Value)
			Messagebox( 'Existe Um Grupo de Produto Cadastrado com essa Unidade !', 48, 'Impossível Exclusão' )
			ThisForm.Pagina.Dados.Grupo_Salva.Cancelar.SetFocus
			Return
		Endif

		lcQuery = [Select CfgGerGprs ] + ;
				    [From SigCdGrp]

		If (ThisForm.poDataMgr.SqlExecute(lcQuery, [LocalBus]) < 1)
			=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (LocalBus)])
			Return 0
		EndIf
		Go Top In LocalBus
		Locate For Substr(LocalBus.CfgGerGprs, 11, 3) = ThisForm.Pagina.Dados.Get_Unid.Value
		If Not Eof([LocalBus])
			Use In LocalBus
			Messagebox( 'Existe Um Grupo de Produto Cadastrado com essa Unidade !', 48, 'Impossível Exclusão' )
			ThisForm.Pagina.Dados.Grupo_Salva.Cancelar.SetFocus
			Return .f.
		EndIf
		Use In LocalBus
	EndIf
Endif

= DoDefault()

ThisForm.mAtivaPagina1()

ENDPROC
