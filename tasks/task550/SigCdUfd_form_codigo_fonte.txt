PROCEDURE Init
LPARAMETERS lcacesso

With ThisForm
	.poDatamgr = CreateObject('fSqlConector')
	With .poDatamgr
		.AddCursor('SigCdUfd','cidchaves','crSigCdUfd','codigos,destinos,cfos',ThisForm.pagina.lista.grade)
		.pfSqlTabela(1).pColuna('codigos','','','UF Origem',80,.t.)
		.pfSqlTabela(1).pColuna('destinos','','','UF Destino',80,.t.)
		.pfSqlTabela(1).pColuna('aliqs','','','Aliq. ICMS',80,.t.)
		.pfSqlTabela(1).pColuna('AliqSbs','','','Aliq.Sub.Trib',80,.t.)
		.pfSqlTabela(1).pColuna('AliqPImps','','','Aliq.Prod.Importado',80,.t.)
		.pfSqlTabela(1).pColuna('Aliqfcp','','','Aliq.FECP',80,.t.)
		.pfSqlTabela(1).pColuna('Aliqredb','','','Aliq.Reducao Base',80,.t.)
		.pfSqlTabela(1).pColuna('iest','','','IE Subst.Trib.Dest.',100,.t.)
		.AddCursor('SigCdUfs','estados','crSigCdUfs','','')
		.AddCursor('SigCdCfo','codigos','crSigCdCfo','','')
		.AddCursor('SigCdAlo','pkChave','crSigCdAlo','','')
		.AddCursor('SigCdAlc','pkChave','crSigCdAlc','','')
	EndWith
	= DoDefault()
	.LockScreen = .f.
ENDWITH

lcCod = crSigCdUFd.cidchaves
SELECT * from crSigcdAlo where fkchave = lcCod order by CodClasFiscal into cursor crAlo readwrite
SELECT crAlo

With ThisForm.Pagina.Dados.gradAlo 
	.RecordSource = 'crAlo'
	.column1.ControlSource = 'crAlo.CodClasFiscal'
	.column2.ControlSource = 'crAlo.AliqLocal'
	.column3.ControlSource = 'crAlo.Aliqfcp'
	.column4.ControlSource = 'crAlo.AliqIcms'
	.column5.ControlSource = 'crAlo.sitribs'
	.column6.ControlSource = 'crAlo.cbenef'
	.Refresh
EndWith

lcCod = crSigCdUFd.cidchaves
SELECT * from crSigcdAlc where fkchave = lcCod order by codcfop into cursor crAlocfop readwrite
SELECT crAlocfop

With ThisForm.Pagina.Dados.gradAlocfop 
	.RecordSource = 'crAlocfop'
	.column1.ControlSource = 'crAlocfop.Codcfop'
	.column3.ControlSource = 'crAlocfop.Aliqfcp'
	.Refresh
EndWith

ENDPROC

* PROCEDURE Pagina.Lista.Grupo_op.Click

PROCEDURE Pagina.Lista.Grupo_op.Click
LPARAMETERS opcao

If !dodefault( opcao )
	ThisForm.mAtivapagina1
	Return
Endif

*** Mena - 14/01/2018 - incluido para ter aliquotas local por classificação fiscal ***
Select crAlo
Zap
Select crAlocfop
Zap

*** pega chave secundária se houver ***
lcCod = IIF(ThisForm.PcEscolha='INSERIR', '', CrSigcdUfd.cidchaves)

lcSql = [Select * From SigCdAlo Where fkchave = '] + lcCod + [' order by codClasFiscal]

=ThisForm.PodataMgr.SqlExecute(lcSql,'TmpAlo')
Select TmpAlo
Scan
	Select crAlo
	Append Blank
	replace	codClasFiscal  with tmpAlo.codClasFiscal, ;
			pkChave with tmpAlo.pkchave, ;
			FkChave with tmpAlo.fkchave, ;
			aliqLocal with TmpAlo.AliqLocal, ;
			aliqIcms with TmpAlo.AliqIcms, ;
			sitribs with TmpAlo.sitribs, ;
			cbenef with TmpAlo.cbenef, ;
			aliqfcp with TmpAlo.aliqfcp
EndScan

Select crAlo
If Reccount() = 0
	Append Blank
EndIf
ThisForm.Pagina.Dados.GradAlo.Refresh

***************************************************************************************************
lcSql = [Select * From SigCdAlc Where fkchave = '] + lcCod + [' order by codcfop]

=ThisForm.PodataMgr.SqlExecute(lcSql,'TmpAlocfop')
Select TmpAlocfop
Scan
	Select crAlocfop
	Append Blank
	replace	codcfop  with TmpAlocfop.codcfop, ;
			pkChave with TmpAlocfop.pkchave, ;
			FkChave with TmpAlocfop.fkchave, ;
			aliqfcp with TmpAlocfop.aliqfcp
EndScan

Select crAlocfop
If Reccount() = 0
	Append Blank
EndIf
ThisForm.Pagina.Dados.GradAlocfop.Refresh
***************************************************************************************************


if inlist(thisform.pcEscolha,'INSERIR','ALTERAR','PROCURAR')
	thisform.Pagina.Dados.Get_OCodigo.setfocus
else
	thisform.Pagina.Dados.Grupo_salva.Salva.setfocus
endif   

If Thisform.PoDatamgr.SQLExecute([Select Descrs from SigCdUfs where Estados = '] + CrSigCdUfd.codigos + [' ],'CrSigCdUfs') < 0
	MessageBox('Favor reinicializar o processo.',16,'Falha na Conexão')
	Return(.F.)
Else
	ThisForm.Pagina.Dados.Get_ONome.Value = CrSigCdUfs.Descrs
Endif

lcqueryok = Thisform.PoDatamgr.SQLExecute([Select Descrs from SigCdUfs where Estados = '] + CrSigCdUfd.Destinos + [' ],'CrSigCdUfs') 
If  lcqueryok < 0
	MessageBox('Favor reinicializar o processo.',16,'Falha na Conexão')
	Return(.F.)
Else
	ThisForm.Pagina.Dados.Get_DNome.Value = CrSigCdUfs.Descrs
Endif	

ENDPROC

* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
Local laCampos(3,2)

If Inlist( Thisform.pcEscolha,'INSERIR','ALTERAR')

	If IsEmpty( Thisform.Pagina.Dados.Get_OCodigo.value )
		=messagebox('U.F. Origem Inválido.',0+64,'Atenção!')
		Thisform.Pagina.Dados.Get_OCodigo.setfocus
		Return
	Endif
	If IsEmpty( Thisform.Pagina.Dados.Get_DCodigo.value )
		=messagebox('U.F. Destino Inválido.',0+64,'Atenção!')
		Thisform.Pagina.Dados.Get_DCodigo.setfocus
		Return
	Endif
	
	m.laCampos(1,1) = 'codigos'
	m.laCampos(1,2) = Thisform.Pagina.Dados.Get_OCodigo.value
	m.laCampos(2,1) = 'destinos'
	m.laCampos(2,2) = Thisform.Pagina.Dados.Get_DCodigo.value
	m.laCampos(3,1) = 'cfos'
	m.laCampos(3,2) = Thisform.Pagina.Dados.GetCfo.value

	If ThisForm.poDataMgr.ChkRegister('SigCdUfd',@m.laCampos) and Inlist( Thisform.pcEscolha,'INSERIR')
		Messagebox( 'ICMS Interestadual já cadastrado.', 48, '' )
		ThisForm.Pagina.Dados.Get_OCodigo.SetFocus
		Return
	Endif
	
	If Inlist( Thisform.pcEscolha,'INSERIR')
		Select CrSigCdUfd
		Replace cidchaves with fUniqueIds()
	Endif
	
	*************************************************************
	**** Mena - 14/01/2018 salva aliquotas por classificacao ****
	SET delete off
	If InList(ThisForm.pcescolha,'INSERIR','ALTERAR')
		Select CrSigCdAlo
		ZAP
		Select crAlo
		GO top
		Scan
			If Not Empty(codclasfiscal)
				Select CrSigCdAlo
				Append Blank
				replace fkchave with CrSigcdUfd.cidchaves,;
						codclasfiscal with crAlo.codclasfiscal,;
						aliqLocal with crAlo.aliqLocal,;
						aliqfcp with crAlo.aliqfcp,;
						aliqIcms with crAlo.aliqIcms,;
						sitribs wITH crAlo.sitribs,;
						cbenef wITH crAlo.cbenef,;
						pkchave with  IIF(EMPTY(crAlo.pkChave), fUniqueIds(), crAlo.pkChave)
				IF DELETED('crAlo')
					DELETE 
				ENDIF
			ENDIF
			SELECT crAlo
		EndScan
	ENDIF
	SET delete on
	********************************************************

	*************************************************************
	**** Mena - 09/09/2020 salva aliquotas por CFOP ****
	SET delete off
	If InList(ThisForm.pcescolha,'INSERIR','ALTERAR')
		Select CrSigCdAlc
		ZAP
		Select crAlocfop
		GO top
		Scan
			If Not Empty(codcfop)
				Select CrSigCdAlc
				Append Blank
				replace fkchave with CrSigcdUfd.cidchaves,;
						codcfop with crAlocfop.codcfop,;
						aliqfcp with crAlocfop.aliqfcp,;
						pkchave with  IIF(EMPTY(crAlocfop.pkChave), fUniqueIds(), crAlocfop.pkChave)
				IF DELETED('crAlocfop')
					DELETE 
				ENDIF
			ENDIF
			SELECT crAlo
		EndScan
	ENDIF
	SET delete on
	********************************************************


Endif
If Not ThisForm.poDataMgr.Update('crSigCdAlo')
	=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (Update - crSigCdAlo)')
	llOks = .f.
EndIf
If Not ThisForm.poDataMgr.Update('crSigCdAlc')
	=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (Update - crSigCdAlc)')
	llOks = .f.
EndIf

=DoDefault()

ThisForm.mAtivapagina1
ENDPROC
