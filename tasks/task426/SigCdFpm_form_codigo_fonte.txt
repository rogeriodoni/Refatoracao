PROCEDURE Init
LParameters poForm, pnDataSes

Set DataSession to pnDataSes

Thisform.ParentForm          = poForm
Thisform.ParentForm.Enabled  = .F.

Thisform.GravaDados = .f.

With Thisform
	.Caption = 'Valor Mínimo/Máximo Cond.Pagto ' + AllTrim(crTSigfPag.fpags)
*	.Caption = 'Valor Mínimo/Máximo Cond.Pagto ' + AllTrim(crSigOpFp.fpags)
	.cmdInserir.Visible = fChecaAcesso('SIGCDFPM','INSERIR')
	.cmdInserir.Enabled = fChecaAcesso('SIGCDFPM','INSERIR')
	.cmdExcluir.Visible = fChecaAcesso('SIGCDFPM','EXCLUIR')
	.cmdExcluir.Enabled = fChecaAcesso('SIGCDFPM','EXCLUIR')
EndWith
ThisForm.cntSombra.lblSombra.Caption = ThisForm.Caption
ThisForm.cntSombra.lblTitulo.Caption = ThisForm.Caption

lcSql = [Select * From SigOpFpI Where Fpags = ']+crTSigfPag.Fpags+[']
If ThisForm.ParentForm.PodataMgr.Sqlexecute(lcSql,'xFpagI') < 1
	MessageBox('Falha na Conexão. Favor Reiniciar o Processo!!!',0+16,'')
	Return .f.
EndIf

Select xFpagI
Scan
	scatter to memvar
	Insert Into CrSigOpFpI From array Memvar
EndScan

Select xFpagI
Index on Fpags + Str(nParcs,2) Tag fPags
Index on nParcs TAG nParcs

Set Order to
Goto Top

With Thisform.GradeFPagto
	.RecordSource          = 'xFPagi'
	.Column2.ControlSource = 'xFPagi.nparcs'
	.Column3.ControlSource = 'xFPagi.valmins'
	.Column1.ControlSource = 'xFPagi.ValMaxs'
	.Column4.ControlSource = 'xFPagi.Emps'
	.Refresh
EndWith

THisForm.GradeFPagto.Refresh
ENDPROC

------------------------------------------------------------
PROCEDURE Valid
Local lcFpags, lnRecNo, loBrowse

If Thisform.peantvalue # This.Value Or This.Value = 0
	
	ThisForm.GravaDados = .t.

	lcFPags = xfPagi.fpags + Str(This.Value,2)

	Select xFPagI
	lnRecNo = RecNo('xFPagi')
	Set Order to FPags
	Seek (xfPagi.fpags + Str(This.Value,2))
	If Not Eof()
		Skip
		If xFPagi.fpags + Str(xFPagi.NParcs,2) = lcFPags
			MessageBox('Número de Parcelas já Digitada para esta Condição de Pagamento ... ',48,'')
			This.Value = 0
			Return .F.
		EndIf
	Endif
	Select xFPagI
	Set Order to
	If lnRecNo # 0
		Goto (lnRecNo)
	Else
		Goto Bottom
	EndIf
	This.Refresh
EndIf

ENDPROC

------------------------------------------------------------
* PROCEDURE Click
------------------------------------------------------------
PROCEDURE Click
Retorno = .t.
If Thisform.cmdInserir.Visible And ThisForm.GravaDados
Set Step On 
	Select nParcs,sum(1) From xFPagi Where nParcs > 0 Or xFPagi.ValMaxs > 0 ;
	group by nparcs having sum(1) > 1;
	Into Cursor TT
	Select TT
	If Reccount('TT') > 0
		MessageBox('Dados Inconsistentes!!!',0+16,'')
		Return .f.
	Endif
	
	Select crSigOpFpi
	LOCATE FOR Fpags = crTSigfPag.fpags
	Dele While crSigOpFpi.fpags = crTSigfPag.fpags
	Select xFPagi
	Scan
		If !IsEmpty(xFPagi.nparcs) Or Not Empty(xFPagi.ValMaxs)
			Scatter to memvar
			Insert Into crSigOpFpi From array memvar
			SELECT crSigOpFpi
			Replace cidChaves WITH fUniqueIds()
			SELECT xFPagi
		EndIf
	EndScan
	Retorno=TSigfpg.podatamgr.Update("crSigOpFpi")
	Retorno=TSigfpg.podatamgr.commit()
EndIf

Thisform.ParentForm.Enabled = .T.
ThisForm.Release

ENDPROC
