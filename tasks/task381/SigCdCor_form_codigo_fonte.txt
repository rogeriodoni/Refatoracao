* PROCEDURE msv_procurar
------------------------------------------------------------
PROCEDURE msv_procurar
Local lcBusca

If .Not.Empty(ThisForm.Pagina.Dados.Getcods.Value)
	ThisForm.poDataMgr.LocateCursor('crSigCdCor','cods',ThisForm.Pagina.Dados.Getcods.Value)
EndIf

ENDPROC

------------------------------------------------------------
* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
LPARAMETERS lcacesso

With ThisForm
	.poDatamgr = CreateObject('fSqlConector', THISFORM.NAME)
	With .poDatamgr
		.AddCursor('SigCdCor','cods','crSigCdCor','',ThisForm.pagina.lista.grade)
		.pfSqlTabela(1).pColuna('cods','','','Código',75,.t.)
		.pfSqlTabela(1).pColuna('descs','','','Descrição',301,.t.)
		.pfSqlTabela(1).pColuna('varias','','','Variação Preço',100,.t.)
		.pfSqlTabela(1).pColuna('Pesos','','','Variação Peso',100,.t.)
	EndWith
	= DoDefault()
	.LockScreen = .f.
EndWith
ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Lista.Grupo_op.Click
------------------------------------------------------------
PROCEDURE Pagina.Lista.Grupo_op.Click
LParameters opcao

If !dodefault( Opcao )
	ThisForm.mAtivapagina1
	Return
Endif

Do Case
	Case Inlist( ThisForm.pcEscolha, 'INSERIR', 'PROCURAR' )
		ThisForm.Pagina.Dados.GetCods.SetFocus
	Case ThisForm.pcEscolha = 'ALTERAR'
		ThisForm.Pagina.Dados.GetDescs.SetFocus
	Case Inlist( ThisForm.pcEscolha, 'EXCLUIR', 'CONSULTAR' )
		ThisForm.Pagina.Dados.Grupo_Salva.Salva.SetFocus
Endcase

ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
If ThisForm.pcEscolha = 'INSERIR'
	If Empty( CrSigCdCor.Cods )
		Messagebox( 'Código não pode ficar em branco!', 48, '' ) 
		ThisForm.Pagina.Dados.GetCods.SetFocus
		Return
	Endif
	If ThisForm.poDataMgr.ChkRegister('SigCdCor','cods',ThisForm.Pagina.Dados.Getcods.Value)
		Messagebox( 'Código já Cadastrado!', 48, '' ) 
		ThisForm.Pagina.Dados.GetCods.SetFocus
		Return
	Endif
Endif

If (ThisForm.pcEscolha = 'EXCLUIR')
	lcQuery = [Select Count(*) As nNreg1s ] + ;
			    [From SigPrPrr a Inner Join SigCdPro b On b.cPros = a.cPros] + ;
			   [ Where a.CodCors = '] + ThisForm.Pagina.Dados.getCods.Value + [']

	If (ThisForm.poDataMgr.SqlExecute(lcQuery, 'crSigPrPrr') < 1)
		=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (crSigPrPrr)')
		Return 0
	EndIf

	If Nvl(crSigPrPrr.nNreg1s, 0) > 0
		=MessageBox('Cor Sendo Utilizada Em Fichas Técnicas!!!', 16, '')
		Return 0
	EndIf
EndIf

=DoDefault()
ThisForm.mAtivapagina1

ENDPROC

If ThisForm.plCancelar
	ThisForm.mAtivapagina1
Endif
ENDPROC
