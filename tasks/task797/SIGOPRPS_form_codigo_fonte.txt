------------------------------------------------------------
* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
Lparameters pcEmp, pcTipoRps, pcSerieRps, pnNumRps, pcDopes, pnNumes

With Thisform
	.poDatamgr = CreateObject('fSqlConector',ThisForm.Name)
	If Not .poDatamgr.pnIdconn>0
		Messagebox('Impossivel Efetuar Conexão com Servidor de Banco de Dados.',(0+48),'Erro')
		Return(.f.)
	EndIf
Endwith	

ThisForm.get_Emps.Value   = pcEmp
ThisForm.get_Tipos.Value  = pcTipoRps
ThisForm.get_Series.Value = pcSerieRps
ThisForm.get_NumRps.Value = pnNumRps
ThisForm.get_Dopes.Value  = pcDopes
ThisForm.get_Numes.Value  = pnNumes

lcQuery = [Select Datas, Transacaos, usuars, nfis from SigCdFrp where Emps = ']+pcEmp+[' ] + ;
			[and Tipos = ']+pcTipoRps+[' and Series = ']+pcSerieRps+[' and NumRps = ] + ;
			Str(pnNumRps,6) + [ Order by Datas Desc]
			
If ThisForm.poDataMgr.SQLExecute(lcQuery,[crSigCdFrp]) < 1
	=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (crSigCdFrp)')
	Return
Endif

With ThisForm.grdMov
	.RecordSource = [crSigCdFrp]
	.Column1.ControlSource = [crSigCdFrp.Datas]
	.Column2.ControlSource = [crSigCdFrp.Transacaos]
	.Column3.ControlSource = [crSigCdFrp.Usuars]
	.Column4.ControlSource = [crSigCdFrp.NFis]
EndWith

ThisForm.Refresh()



ENDPROC
