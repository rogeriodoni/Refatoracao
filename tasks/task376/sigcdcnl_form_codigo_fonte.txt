------------------------------------------------------------
* PROCEDURE reverterxml
------------------------------------------------------------
PROCEDURE reverterxml
Parameters pcCursor, pcNivAnt, pcCurDest

Local lnOldSel, lcXML

lnOldSel = Select()
lcXML = []

If Used([CrAux])
	Use in CrAux
EndIf 

Select &pcCursor.
Select cidchaves, Cods, Descs, 0 as Marcas, Space(4) as Codigo From &pcCursor. Where 1=0 Into Cursor CrAux ReadWrite

Select &pcCursor.	
Go top

Scan 			
	lcXML = Nvl(&pcCursor..Nivel,[])
	Zap in CrAux
	If Not Empty(lcXML)
		= fXmlToCursor(lcXML, [CrAux], .f.)
		Select CrAux
		Scan For CrAux.Marcas <> 0
			Select &pcCurDest.
			Append Blank 
			Replace Cods With CrAux.Cods ,;
					Descs With &pcCursor..Descs ,;
					CidChaves With CrAux.CidChaves ,;
					Codigo With &pcCursor..Cods in &pcCurDest.
			Select CrAux	
		EndScan 
	EndIf 
EndScan 

Select &pcNivAnt.
Go Top 

If Used([CrAux])
	Use in CrAux
EndIf 

Select(lnOldSel)
ENDPROC

PROCEDURE Init
With ThisForm
	.poDataMgr = CreateObject([fSqlConector], .Name)
	
	lcQryCnl = [select a.*, b.rclis as desvends, c.descricaos as dlocals ] + ;
				[from sigcdcnl a ] + ;
