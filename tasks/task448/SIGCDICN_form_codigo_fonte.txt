PROCEDURE Init

Local lnErro1,lcMcrSql1,lcDirTmp1,lcArqIcone1,lnTamIcone1

With ThisForm
	.poDatamgr1 = CreateObject('fSqlConector',.Name)
	.Caption = Prompt() && 'Faixa p/ Cor'
	m.lnErro1 = .poDatamgr1.pnIdconn
		
	.cntSombra.lblSombra.Caption = .Caption
	.cntSombra.lblTitulo.Caption = .Caption
	
	If m.lnErro1>0 
		.Grid1.Visible = .f.
		.Botão_okcancel.Visible = .f.
		.Refresh()
		.Show()
		.Draw()
		.LockScreen = .t.
		m.lcMcrSql1 = [select a.descricaos,a.programas,a.barrapict,a.barraforms,a.pkChaves from sigcdprg a order by 1]
		If m.lnErro1>0
			m.lnErro1 = .poDatamgr1.SQLExecute(m.lcMcrSql1,'crProg1')
		EndIf
		If Used('crProg1')
			Select ('crProg1')
			Index On pkChaves Tag pkChaves
			Set Order To ('') In ('crProg1')
			Update crProg1 Set barrapict=Lower(Alltrim(JustFname(barrapict)))
			If Used('crTmpProg2')
				USE In ('crTmpProg2')
			EndIf 
			Select a.* From crProg1 a Where Not IsEmpty(a.descricaos) Order By a.descricaos Into Cursor crTmpProg2
			Select ('crTmpProg2')
			Index On pkChaves Tag pkChaves
			If Used('crProg2')
				USE In ('crProg2')
			EndIf 
			USE Dbf('crTmpProg2') Again Alias ('crProg2') In 0
			Select ('crProg2')
			Set Order To ('') In ('crProg2')
			Locate 
			With .Grid1
				.SetAll("FontName","Courier New","Column")
				.SetAll("FontName","Arial","Header")
				.SetAll("FontName","Courier New","TextBox")
				.SetAll("FontName","Courier New","CommandButton")
				.SetAll("FontSize",9,"Column")
				.SetAll("FontSize",8,"Header")
				.SetAll("FontSize",9,"TextBox")
				.SetAll("FontSize",9,"CommandButton")
				.SetAll("Format","K","Column")
				.SetAll("Format","K","TextBox")
				.SetAll("Movable",.f.,"Column")
				.SetAll("ReadOnly",.t.,"Column")
				.SetAll("ReadOnly",.t.,"TextBox")
				.SetAll("Resizable",.f.,"Column")
				.SetAll("Alignment",3,"Column")
				.SetAll("Alignment",3,"TextBox")
				.SetAll("Alignment",2,"Header")
				.SetAll("BorderStyle",0,"TextBox")
				.SetAll("Margin",0,"TextBox")
				.SetAll("Visible",.t.,"TextBox")
				.HeaderHeight = 15
				.RowHeight = 15
				.LinkMaster = ''
				.RelationalExpr = ''
				.RecordSource = 'crProg2' 
				.Column1.ControlSource = 'crProg2.descricaos'
				.Column2.ControlSource = 'crProg2.barrapict'
				.Refresh
				.Visible = .t.
			EndWith 
		EndIf 
		m.lcMcrSql1 = ([select a.carqicones,a.marqicones,']+Space(128)+[' as ctmpicones from SigSyIco a order by 1])
		If m.lnErro1>0
			m.lnErro1 = .poDatamgr1.SQLExecute(m.lcMcrSql1,'crSigSyIco1')
		EndIf
		If Used('crSigSyIco1')
			m.lcDirTmp1 = Addbs(Sys(2023))
			Select ('crSigSyIco1')
			Index On carqicones Tag carqicones
			Set Order To ('') In ('crSigSyIco1')
			Locate 
			
			If Not Eof('crSigSyIco1')
				Insert Into crSigSyIco1 (carqicones) Values (' < Nenhum >')
			EndIf 
			Locate 
			
			Scan While m.lnErro1>0
				m.lcArqIcone1 = (m.lcDirTmp1+'ArqIcone'+Sys(2015)+'.ico')
				m.lnTamIcone1 = StrToFile(crSigSyIco1.marqicones,m.lcArqIcone1)
				If m.lnTamIcone1>0
					Replace ctmpicones With m.lcArqIcone1 In ('crSigSyIco1')
				EndIf 
			EndScan 
		EndIf 
		= DoDefault(.Name)
		.Botão_okcancel.Visible = .t.
	Else 
		= Messagebox('Impossivel Efetuar Conexão com Servidor de Banco de Dados.',(0+48),'Erro')
		Return(.f.)
	EndIf 
	.LockScreen = .f.
EndWith

ENDPROC

PROCEDURE Release
Local lcDirTmp1,lcArqIcone1

With ThisForm
	If Used('crSigSyIco1')
		.Grid1.Visible = .f.
		.Botão_okcancel.Visible = .f.

		m.lcDirTmp1 = Addbs(Sys(2023))

		Select ('crSigSyIco1')
		Set Order To ('') In ('crSigSyIco1')
		Locate 
		
		Scan
			m.lcArqIcone1 = Alltrim(crSigSyIco1.ctmpicones)
			If (Not IsEmpty(m.lcArqIcone1) And File(m.lcArqIcone1))
				Delete File (m.lcArqIcone1)
			EndIf 
		EndScan 
	EndIf 
	.poDatamgr1.Release
EndWith 
= DoDefault()


ENDPROC

PROCEDURE ok.Click

Local lnErro1,lcMcrSql1, lcBar
With ThisForm
	m.lnErro1 = .poDataMgr1.pnIdConn

	If Used('crProg1')
		.Grid1.Visible = .f.
		.Botão_okcancel.Visible = .f.

		Select ('crProg1')
		Set Order To ('') In ('crProg1')
		Locate 
		
		Scan While m.lnErro1>0
			*** mena - 10/11/2014 - substitui aspa simples por aspa dupla para não dar erro no update ***
			lcBar = STRTRAN(Alltrim(crProg1.barraforms),['],["])	
			m.lcMcrsql1 = ([update sigcdprg set barrapict=']+Alltrim(crProg1.barrapict)+[',barraforms=']+lcBar+[']+;
							[ where pkChaves=']+crProg1.pkChaves+['])
			If m.lnErro1>0
				m.lnErro1 = .poDatamgr1.SQLExecute(m.lcMcrSql1)
			EndIf
		EndScan 
		If m.lnErro1>0 
			m.lnErro1 = .poDatamgr1.Commit()
		EndIf 
		If Not m.lnErro1>0 
			= .poDatamgr1.RollBack()
		EndIf 
	EndIf 
	.Release
EndWith

ENDPROC
