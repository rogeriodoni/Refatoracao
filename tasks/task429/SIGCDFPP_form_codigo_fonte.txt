PROCEDURE Init
LParameters poForm, pnDataSes

ThisForm.cntSombra.lblSombra.Caption = ThisForm.Caption
ThisForm.cntSombra.lblTitulo.Caption = ThisForm.Caption

Set DataSession to pnDataSes

Thisform.ParentForm          = poForm
Thisform.ParentForm.Enabled  = .F.

Thisform.GravaDados = .f.

With Thisform
	.cmdInserir.Visible = fChecaAcesso('SIGCDFPP','INSERIR')
	.cmdInserir.Enabled = fChecaAcesso('SIGCDFPP','INSERIR')
	.cmdExcluir.Visible = fChecaAcesso('SIGCDFPP','EXCLUIR')
	.cmdExcluir.Enabled = fChecaAcesso('SIGCDFPP','EXCLUIR')
	.label1.Caption = '[ '+AllTrim(crSigOpFp.fpags)+' ]'
EndWith

Select CrSigOpFpP
Zap

lcSql = [Select * From SigOpFpP Where Fpags = ']+CrSigOpFp.Fpags+[' Order by nparcs ]
If ThisForm.ParentForm.PodataMgr.Sqlexecute(lcSql,'xFpagI') < 1
	MessageBox('Falha na Conexão. Favor Reiniciar o Processo!!!',0+16,'')
	Return .f.
EndIf

Select xFpagI
Scan
	scatter to memvar
	Insert Into CrSigOpFpP From array Memvar
EndScan

Select xFpagI
Index on nParcs TAG nParcs
Set Order to
Goto Top

With Thisform.GradeFPagto
	.RecordSource          = 'xFPagi'
	.Column1.ControlSource = 'xFPagi.nparcs'
	.Column2.ControlSource = 'xFPagi.Descs'
	.Column3.ControlSource = 'xFPagi.ValMins'
	.Column5.ControlSource = 'xFPagi.ValMaxs'
	.Column4.ControlSource = 'xFPagi.parcDes'
	.Refresh
EndWith

THisForm.GradeFPagto.Refresh
ENDPROC

------------------------------------------------------------
* PROCEDURE Click
------------------------------------------------------------
PROCEDURE Click
Retorno = .t.
Set Step On 
If Thisform.cmdInserir.Visible And ThisForm.GravaDados
	Select nParcs,Descs,sum(1) From xFPagi Where nParcs > 0 And (xFPagi.Descs > 0 Or xFpagi.ParcDes > 0 ) ;
	group by nparcs,Descs having sum(1) > 1;
	Into Cursor TT
	Select TT
	If Reccount('TT') > 0
		MessageBox('Dados Inconsistentes. No. de Parcelas/%Desconto em duplicidade!!!',0+16,'')
		Return .f.
	Endif
	
	Select crSigOpFpP
	Delete all
	Select xFPagi
	Scan
		If !IsEmpty(xFPagi.nparcs) And (Not Empty(xFPagi.Descs) Or Not Empty(xFPagi.ParcDes))
			Scatter to memvar
			Insert Into crSigOpFpp From array memvar
			SELECT crSigOpFpP
			Replace cidChaves WITH fUniqueIds()
			SELECT xFPagi
		EndIf
	EndScan
	Retorno=ThisForm.ParentForm.podatamgr.Update("crSigOpFpP")
	Retorno=ThisForm.ParentForm.podatamgr.commit()
EndIf
Thisform.ParentForm.Enabled = .T.
ThisForm.Release

ENDPROC
