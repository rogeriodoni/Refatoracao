* PROCEDURE msv_procurar

PROCEDURE msv_procurar
Local laCampos(1,2)

m.laCampos(1,1) = 'Cods'
m.laCampos(1,2) = ThisForm.Pagina.Dados.getCods.Value

ThisForm.poDataMgr.LocateCursor('crSigCdNcf', @m.laCampos)
* Fim **************************************************************************************
ENDPROC

------------------------------------------------------------
* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
lcQuerLista = [Select * From SigCdNcf Where Emps = ']+_Empr+[' Or Emps = ' ' ]

With ThisForm
	.poDataMgr = CreateObject('fSqlConector', ThisForm.Name)
	
	If (.poDataMgr.pnIdconn > 0)
		With .poDataMgr
			.AddCursor('SigCdNcf', 'Cods', 'crSigCdNcf', '', ThisForm.Pagina.Lista.Grade,lcQuerLista)
			.pfSqlTabela(1).pColuna('Cods',  '', '', 'Códigos',   080, .t.)
			.pfSqlTabela(1).pColuna('Descs', '', '', 'Descrição', 290, .t.)
		EndWith

		=DoDefault()

		ThisForm.Pagina.Lista.Grade.Column1.SetFocus
	Else
		=Messagebox('Impossível Efetuar Conexão Com o Servidor de Banco de Dados...', 0+48, 'CreateObject')
		Return .f.
	EndIf
EndWith
* Fim **************************************************************************************
ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Lista.Grupo_op.Click
------------------------------------------------------------
PROCEDURE Pagina.Lista.Grupo_op.Click
Lparameters Opcao

If Not DoDefault(Opcao)
	ThisForm.mAtivaPagina1
	Return 0
EndIf

Select crSigCdNcf

Do Case
	Case Inlist(ThisForm.pcEscolha, 'INSERIR', 'PROCURAR')
		ThisForm.Pagina.Dados.getCods.Setfocus

	Case Inlist(ThisForm.pcEscolha, 'ALTERAR')
		ThisForm.Pagina.Dados.getDescs.Setfocus

	Otherwise
		ThisForm.Pagina.Dados.Grupo_Salva.Salva.Setfocus
EndCase 

Select crSigCdNcf

ThisForm.Pagina.Dados.Activate
* Fim **************************************************************************************
ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
If InList(ThisForm.pcEscolha,'INSERIR','ALTERAR')
	If Empty(ThisForm.Pagina.Dados.getCods.Value)
		=MessageBox('Obrigatório Informar o Código!!!', 48, '')
		ThisForm.Pagina.Dados.getCods.SetFocus
		Return 0
	EndIf
	
	If Empty(ThisForm.Pagina.Dados.getDescs.Value)
		=MessageBox('Obrigatório Informar a Descrição!!!', 48, '')
		ThisForm.Pagina.Dados.getDescs.SetFocus
		Return 0
	EndIf

	If (ThisForm.pcEscolha = 'INSERIR')
		ThisForm.poDataMgr.CursorQuery('SigCdNcf', 'LocalNConf', 'Cods', Padr(ThisForm.Pagina.Dados.getCods.Value, 10))
		If Not Eof('LocalNConf')
			=MessageBox('Código Já Cadastrado!!!', 48, '')
			ThisForm.Pagina.Dados.getCods.SetFocus
			Return 0
		EndIf
	EndIf
EndIf

=DoDefault()

ThisForm.mAtivapagina1
* Fim **************************************************************************************
ENDPROC

PROCEDURE Valid
LOCAL xBrox

If !IsEmpty( This.Value )
	xBrox = CreateObject('fwBuscaExt',Thisform.poDatamgr.pnIdconn,'SigCdEmp','crSigCdEmp','cEmps',This.Value,'Empresas',.t.,'cEmps')

	If  Not xBrox.plAchouRegistro
		xBrox.mAddColuna('CEmps', 'XXXXXX', 'Código', .t.)
		xBrox.mAddColuna('Razas', '', 'Descrição', .t.)
		xBrox.Show()
		This.Value = Iif( LastKey() # 27, Evaluate('crSigCdEmp.CEmps'), '' )
	EndIf 
Endif
ENDPROC
