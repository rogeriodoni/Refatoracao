PROCEDURE atuatotas
Select [ ] as Agrupar, Sum(Qtds) as Qtds, Sum(Totas) as Totas ;
  From crGrade ;
 Where Marcas ;
 Group By 1 ;
  Into Cursor LocalTot
Go Top In LocalTot

ThisForm.grdRodape.Column3.Text1.Value = LocalTot.Totas
ThisForm.grdRodape.Column3.Refresh

ThisForm.cntBotoes.btnConfirmar.Enabled = (LocalTot.Qtds > 0)

=DoDefault()
* Fim **************************************************************************************
ENDPROC

------------------------------------------------------------
* PROCEDURE selecionadados
------------------------------------------------------------
PROCEDURE selecionadados
Local lcQuery

Zap In crGrade
Zap In crRodape

lcQuery = [Select a.CPros, a.Mats, b.DPros, a.Qtds, a.UniCompos, b.PVens as PCompos, b.Moevs as Moeds, ] + ;
				 [(a.Qtds * b.PVens) as Totas, a.Etiqs, b.Reffs, b.FigJpgs, a.Tipos ] + ;
		    [From SigPrCpo a, SigCdPro b ] + ;
		   [Where a.CPros = '] + ThisForm.lpKey + [' And ] + ;
		   		 [a.Mats = b.CPros ] + ;
		   [Order By a.Mats]

If (ThisForm.poDataMgr.SqlExecute(lcQuery, [LocalCompo]) < 1)
	=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (LocalCompo)])
	Return .f.
EndIf

llBrc = .f.
Select LocalCompo
Scan
	Insert Into crGrade (Marcas, Mats, DPros, Qtds, UniCompos, PCompos, Moeds, Totas, Reffs, FigJpgs, Tipos) ;
				 Values (LocalCompo.Etiqs = [S], LocalCompo.Mats, LocalCompo.DPros, LocalCompo.Qtds, ;
				 		 LocalCompo.UniCompos, LocalCompo.PCompos, LocalCompo.Moeds, LocalCompo.Totas, ;
				 		 LocalCompo.Reffs, Nvl(LocalCompo.FigJpgs, []), LocalCompo.Tipos)

	If Not llBrc And Empty(LocalCompo.Tipos)
		llBrc = .t.
	EndIf
EndScan
Select crGrade

ThisForm.cmbTipos.Clear
If llBrc
	ThisForm.cmbTipos.AddItem(Space(20))
EndIf

lcQuery = [Select Tipos ] + ;
		    [From SigCdTpc ] + ;
		   [Where Custos = 2 ] + ;
		   [Order By Tipos]

If (ThisForm.poDataMgr.SqlExecute(lcQuery, [LocalTpCom]) < 1)
	=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (LocalTpCom)])
	Return .f.
EndIf

Select LocalTpCom
Scan
	ThisForm.cmbTipos.AddItem(LocalTpCom.Tipos)
EndScan
Go Top In LocalTpCom
ThisForm.cmbTipos.Value = Iif(llBrc, Space(20), LocalTpCom.Tipos)
ThisForm.cmbTipos.Refresh
Set Filter To Tipos = ThisForm.cmbTipos.Value In crGrade
Go Top In crGrade
ThisForm.grdItens.Refresh
ThisForm.grdItens.Column1.SetFocus

ThisForm.cntBotoes.btnConfirmar.Enabled = .f.
Insert Into crRodape (Totas) Values (0)
If (Reccount([crGrade]) > 0)
	Select [ ] as Agrupar, Sum(Qtds) as Qtds, Sum(Totas) as Totas ;
	  From crGrade ;
	 Where Marcas ;
	 Group By 1 ;
	  Into Cursor LocalTot
	Go Top In LocalTot
	Replace Totas With LocalTot.Totas In crRodape
	ThisForm.cntBotoes.btnConfirmar.Enabled = (LocalTot.Qtds > 0)
