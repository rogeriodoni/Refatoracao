------------------------------------------------------------
PROCEDURE Init
lcQryCnq = [Select * From SigCdCnq]

With ThisForm
	.poDataMgr = CreateObject('fSqlConector', ThisForm.Name)

	If (.poDataMgr.pnIdconn > 0)
		With .poDataMgr
			.AddCursor('SigCdCnq', 'pkChaves', 'crSigCdCnq', '', ThisForm.Pagina.Lista.Grade, lcQryCnq)
			.pfSqlTabela(1).pColuna('conquilhas', '', '', 'Conquilha', 100, .t.)
			.pfSqlTabela(1).pColuna('codtams', '', '', 'Aro', 80, .t.)
			.pfSqlTabela(1).pColuna('valors', '', '', 'Valor', 100, .t.)
		EndWith

		Select crSigCdCnq
		Index On conquilhas tag CON
		Index On codtams tag ARO
		Set Order To CON
		ThisForm.Pagina.Lista.Grade.Column1.SetFocus
		=DoDefault()
	Else
		=Messagebox('Impossivel Efetuar Conexão com Servidor de Banco de Dados...', 0+48, 'CreateObject')
		Return .f.
	EndIf
EndWith
* Fim **************************************************************************************



ENDPROC

PROCEDURE Pagina.Lista.Grupo_op.Click
LPARAMETERS opcao
If !dodefault( Opcao )
	ThisForm.mAtivapagina1
	Return
Endif

Select crSigCdCnq

* posiciona o cursor de acordo com a opção
Do Case
	Case Inlist( ThisForm.pcEscolha, 'INSERIR', 'PROCURAR', 'ALTERAR' )
		thisform.pagina.dados.get_conquilha.setfocus	
	Case Inlist( ThisForm.pcEscolha, 'CONSULTAR', 'EXCLUIR' )
		thisform.pagina.dados.grupo_salva.salva.setfocus
Endcase

Select crSigCdCnq
=DoDefault()
ThisForm.Pagina.Dados.Refresh

ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
LOCAL lcQuery, lcCon, lcCod

If ThisForm.pcEscolha = 'INSERIR'
	If IsEmpty(ThisForm.Pagina.Dados.Get_conquilha.Value)
		MessageBox( 'Obrigatório Informar a Conquilha.', 48, '' )
		ThisForm.Pagina.Dados.Get_conquilha.SetFocus
		Return
	ENDIF  

	*** Verifica se registro ja cdastrado ***
	lcCon = ThisForm.Pagina.Dados.Get_conquilha.Value
	lcCod = ThisForm.Pagina.Dados.Get_codtam.Value
	lcQuery = [Select * From SigCdCnq Where conquilhas = '] + lcCon + [' And codtams = '] + lcCod + [']
	ThisForm.poDataMgr.SqlExecute(lcQuery, [crLocal])
	SELECT crLocal
	GO top
	IF !EOF()
		MessageBox( 'Conquilha já cadastrada...', 48, '' )
		ThisForm.Pagina.Dados.Get_conquilha.SetFocus
		SELECT crSigCdCnq
		Return
	EndIf
	Replace pkChaves With fUniqueIds() in crSigCdCnq
ENDIF
SELECT crSigCdCnq
=DoDefault()
ThisForm.mAtivapagina1

ENDPROC

Local loLista

	loLista = CreateObject('fwBuscaExt', ThisForm.poDataMgr.pnIdConn, 'SigCdTam', ;
						   'CrListaRemota', 'cods', This.Value, 'Seleção', .t.)

	If Not loLista.plAchouRegistro
		loLista.mAddColuna('cods',  '', 'Cod Tam')
		loLista.mAddColuna('descs',  '', 'Descrição')
		loLista.Show()
	EndIf
	
	This.Value = CrListaRemota.cods
	Use In CrListaRemota
EndIf
