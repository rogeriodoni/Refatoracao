------------------------------------------------------------
* PROCEDURE cadauto
------------------------------------------------------------
PROCEDURE cadauto
LParameters pcCadGrupo, pcCpfs, pcNome

Local lcRetCli, lcMsg, lcCaption, lcCpfs, lnTpClis, lnIClis, lnResposta

lcRetCli  = ''
lcMsg     = ''
lcCaption = ''
lnTpClis  = 0
lcCpfs = Strt( Strt( Strt( pcCpfs, ".", "" ), "-", "" ), "/", "" )

If ThisForm.poDataMgr.Cursorquery('SigCdGcr','CrTmpGccr','Codigos',pcCadGrupo) And CrTmpGccr.TpCods = 1
	If Len(Alltrim(lcCpfs))<>14
		If fValidarCPF( lcCpfs )
			lnTpClis = 1
			lcCpfs   = Transf(lcCpfs, '@R 999.999.999-99')
		Else
			MessageBox('CPF ' + AllTrim(lcCpfs) + ' Inválido!!! ',0+16,'Cadastro Automático de Contas')
		EndIf
	Else
		If fValidarCNPJ( lcCpfs )
			lnTpClis = 2
			lcCpfs = Transf( lcCpfs, '@R 99.999.999/9999-99' )
		Else
			MessageBox('CNPJ ' + AllTrim(lcCpfs) + ' Inválido!!! ',0+16,'Cadastro Automático de Contas')
		EndIf
	EndIf

	If InList( lnTpClis, 1, 2 )

		IF !EOF('crTmpCli')	&& se o cpf já está cadastrado atualiza com os dados digitados
			lnIclis = crTmpCli.NClis
			lcIclis = crTmpCli.IClis
		ELSE
			If CrTmpGccr.TpEmps = 1
				lnIclis = fGerUniqueKey( AllTrim( CrTmpGccr.Codigos ) + _empr )
				lcIclis = AllTrim( CrTmpGccr.DgCods + _empr + PadL( lnIclis, 6, '0' ) )
			Else
				lnIclis = fGerUniqueKey( AllTrim( CrTmpGccr.Codigos ) )
				lcIclis = AllTrim( CrTmpGccr.DgCods + PadL( lnIclis, 9, '0' ) )
			EndIf
		ENDIF
		
		_pData = Datetime() &&ThisForm.poDataMgr.GetDatetime()
		Insert Into crSigCdCli (Iclis, Cpfs, Usuars, TpCads, Grupos, DataIncs, TpClis, NClis, DtAlts, rclis, razaos) ;
				Values ( lcIclis, lcCpfs, Usuar, CrTmpGccr.TpCads, pcCadGrupo, _pData, ;
						 lnTpClis, lnIClis, _pData, ALLTRIM(pcNome), ALLTRIM(pcNome) )
		lcRetCli = lcIclis

		m.Retorno = ThisForm.poDataMgr.UpDate('crSigCdCli')
		IF !m.Retorno
			MessageBox('Favor reinicializar o processo.',16,'Falha na Conexão (crSigCdCli ')
			ThisForm.poDataMgr.RollBack()
			lcRetCli = ' '
		EndIf
		If ThisForm.poDataMgr.Commit() < 1
			MessageBox('Favor reinicializar o processo.',16,'Falha na Conexão (Commit 3)')
			ThisForm.poDataMgr.RollBack()
			lcRetCli = ' '
		EndIf
	EndIf
Else
	If Eof('crTmpGccr')
		lcMsg     = 'Grupo de Conta ' + AllTrim(pcCadGrupo) + ' Inválido ... ' + Chr(13) + ;
			        'Contate o Suporte Técnico.'
		lcCaption = 'Erro: Cadastro Inconsistente'
	Else
		lcMsg     = 'Configuração do Grupo de Conta ' + AllTrim(pcCadGrupo) + ' Incorreto ... ' + Chr(13) + ;
			        'Contate o Suporte Técnico.'
		lcCaption = 'Erro: Código de Conta Precisa Ser Automático'
	EndIf
EndIf

If Not IsEmpty(lcMsg)
	MessageBox(lcMsg, 48, lcCaption)
EndIf
Return( !EMPTY(lcRetCli) )

***************** fim *****************
ENDPROC

PROCEDURE calcvalors

Local RecChq, lnVJuros
Select crSigCdPam
Goto top
select tempchq
set order to

RecChq=recno('tempchq')
Thisform.VTotal=0
Thisform.VTJuros = 0 

If Thisform.Pagina.Dados.Get_Opera.Value='S' or pnRefaz
	scan
		Thisform.VTotal=Thisform.VTotal+tempchq.valors
		ldVen = Nvl(TempChq.Vencs, CtoD(''))
		LdDat = Ttod(fDtoSQL(Thisform.Pagina.Dados.Get_Data.Value))
		lnDias = IIF(crSigCdOpt.devolvidos<>'S',Iif(LdDat>=Ldven,0,Ldven-LdDat),LdDat-ldVen)
		If crSigCdPam.JuroCarts = 2 &&&&& Juros Compostos
			lnVJuros = round(TempChq.Valors*(((1+crSigCqLcm.taxdias/100)^(lnDias))-1),2)
		Else &&&&&& Juros Simples
			lnVJuros = round(TempChq.Valors*(crSigCqLcm.taxcarts/100)*(lnDias/30),2)
		Endif
		*** mena 26042015 incluido iof ***
		lnVJuros = lnVJuros + round(TempChq.Valors*(crSigCqLcm.taxiofs/100)*(lnDias/30),2)
		Select TempChq
		Replace TempChq.ValJuros with lnVJuros
		Thisform.VTJuros = Thisform.VTJuros + lnVJuros
	endscan
Else
	lnnche = 0
	lnTTaxa = 0
	lnTTaxaDia = 0
	lnVJuros = 0
	Select TempChq
	scan

		lcQuery = [Select nums,ValJuros ] + ;
		    [From SigCcChm] + ;
		    [ Where nums=']+TempChq.Nums+[' Order By nums,ValJuros]
		If (ThisForm.poDataMgr.SqlExecute(lcQuery, 'csSigCcChm') < 1)
			=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (csSigCcChm)')
			Return 0
		EndIf
		Select csSigCcChm
		If not eof()
			lnVJuros = 0	&&csSigCcChm.ValJuros
		EndIf
		USE in csSigCcChm

*!*			*** mena - 08/04/2015 - usar para calculo os juros do campo para permitir outra taxa de juros ***
*!*			IF pnRefaz
*!*				ldVen = Nvl(TempChq.Vencs, CtoD(''))
*!*				LdDat = Ttod(fDtoSQL(Thisform.Pagina.Dados.Get_Data.Value))
*!*				lnDias = Iif(LdDat>=Ldven,0,Ldven-LdDat)
*!*				If crSigCdPam.JuroCarts = 2 &&&&& Juros Compostos
*!*					lnVJuros = round(TempChq.Valors*(((1+crSigCqLcm.taxdias/100)^(lnDias))-1),2)
*!*				Else &&&&&& Juros Simples
*!*					lnVJuros = round(TempChq.Valors*(crSigCqLcm.taxcarts/100)*(lnDias/30),2)
*!*				Endif
*!*				lnVJuros = lnVJuros + round(TempChq.Valors*(crSigCqLcm.taxiofs/100)*(lnDias/30),2)
*!*			ENDIF
*!*			***************************************************************************************************

		Select TempChq
		Replace TempChq.ValJuros with lnVJuros
		ldVen = Nvl(TempChq.Vencs, CtoD(''))
		LdDat = Ttod(fDtoSQL(Thisform.Pagina.Dados.Get_Data.Value))
		lnDias = Iif(LdDat>=LdVen,0,LdVen-LdDat)
		Thisform.VTotal=Thisform.VTotal+tempchq.valors
		Thisform.VTJuros = Thisform.VTJuros + lnVJuros

		If TempChq.Valors<>0 and lnDias <> 0
			lnnche = lnnche + 1
			If crSigCdPam.JuroCarts = 2 &&&&& Juros Compostos
				lnTaxaDia = round(((((lnVJuros/TempChq.Valors)+1)^(1/lnDias))-1)*100,4)
				lnTaxa = round((((1+lnTaxaDia /100)^(30))-1)*100,2)
			else
				lnTaxa = round((3000*lnVJuros)/(TempChq.Valors*lnDias),2)
			Endif
			lnTTaxa = lnTTaxa + lnTaxa
		Endif
	endscan
	
	lnTTaxa = iif(lnnche=0,0,lnTTaxa /lnnche)
	With Thisform.Pagina.Dados.Get_Taxa
		.Value = lnTTaxa
		.Refresh
	EndWith
	If crSigCdPam.JuroCarts = 2 &&&&& Juros Compostos
		lnTTaxaDia = round((((1+lnTTaxa/100)^(1/30)) - 1 ) * 100,4)
	Else &&&&&& Juros Simples
		lnTTaxaDia = Round(lnTTaxa / 30 ,2)
	Endif
	With Thisform.Pagina.Dados.Get_TaxaDia
		.Value = lnTTaxaDia
		.Refresh
	EndWith
EndIf
Select TempChq
Locate For Recno() = RecChq

ENDPROC

		ENDIF
	ENDIF
	thisform.LeCheque = .F.

		thisform.pagina.dados.get_DContaOs.Value = crTmpSigCdCli.rclis
	Else
		thisform.pagina.dados.get_DContaOs.Value = ''
	EndIf
ENDIF

IF TYPE('pcDado') = 'L' or pcDado = '2'
	**************************************
	***********		DESTINO    ***********
	**************************************
	iGrupo = thisform.pagina.dados.get_GrupoDs.Value
	If not IsEmpty(iGrupo)
		thisform.podatamgr.cursorquery('SigCdGcr','crTmpgccr','codigos',iGrupo)
		thisform.pagina.dados.get_DGrupoDs.Value = crTmpgccr.Descrs
	Else
		thisform.pagina.dados.get_DGrupoDs.Value = ''
	EndIf

	iContas = thisform.pagina.dados.get_ContaDs.Value
	If not IsEmpty(iContas)
		thisform.podatamgr.cursorquery('SigCdCli','crTmpSigCdCli','iclis',iContas)
		thisform.pagina.dados.get_DContaDs.Value = crTmpSigCdCli.rclis
	Else
		thisform.pagina.dados.get_DContaDs.Value = ''
	EndIf
ENDIF

IF TYPE('pcDado') = 'L' or pcDado = '3'
	**************************************
	***********		JUROS	   ***********
	**************************************
	iGrupo = thisform.pagina.dados.get_GruJuro.Value
	If not IsEmpty(iGrupo)
		thisform.podatamgr.cursorquery('SigCdGcr','crTmpgccr','codigos',iGrupo)
		thisform.pagina.dados.get_DGruJuro.Value = crTmpgccr.Descrs
	Else
		thisform.pagina.dados.get_DGruJuro.Value = ''
	EndIf

	iContas = thisform.pagina.dados.get_ConJuro.Value
	If not IsEmpty(iContas)
		thisform.podatamgr.cursorquery('SigCdCli','crTmpSigCdCli','iclis',iContas)
		thisform.pagina.dados.Get_djuro.Value = crTmpSigCdCli.rclis
	Else
		thisform.pagina.dados.Get_djuro.Value = ''
	EndIf
ENDIF

IF TYPE('pcDado') = 'L' or pcDado = '4'
	**************************************
	***********		TERCEIRO	   ***********
	**************************************
	iGrupo = thisform.pagina.dados.get_GrupoT.Value
	If not IsEmpty(iGrupo)
		thisform.podatamgr.cursorquery('SigCdGcr','crTmpgccr','codigos',iGrupo)
		thisform.pagina.dados.get_DGrupoT.Value = crTmpgccr.Descrs
	Else
		thisform.pagina.dados.get_DGrupoT.Value = ''
	EndIf
ENDIF

IF TYPE('pcDado') = 'L' or pcDado = '5'
	**************************************
	***********		TERCEIRO	   ***********
	**************************************
	IF !EOF('TempChq')
		iContas = TempChq.ContaT
		If not IsEmpty(iContas)
			thisform.podatamgr.cursorquery('SigCdCli','crTmpSigCdCli','iclis',iContas)
			replace nomet with crTmpSigCdCli.rclis in TempChq
			replace cpft with crTmpSigCdCli.cpfs in TempChq
		Else
			replace nomet with '' in TempChq
			replace cpft with '' in TempChq
		EndIf
	ENDIF
ENDIF

thisform.pagina.dados.refresh

ENDPROC

	SELECT tempchq
	GO top
	ThisForm.Enabled = .f.

*!*		SELECT * from tempchq into cursor  crGraCart		&& salva cursor da grade 
*!*		SELECT * from crSigche into cursor crCheCart		&& salva cursor de cheques

*!*		lcQuery = [Select * From SigChe ]+;
*!*				  [ Where Bancos=']+tempchq.bancos+[']+;
*!*				  [ and agencias=']+tempchq.agencias+[']+;
*!*				  [ and  ncontas=']+tempchq.ncontas+[']+;
*!*				  [ and ncheques=']+tempchq.ncheques+[']
*!*		If (ThisForm.poDataMgr.SqlExecute(lcQuery, 'crTmpChe') < 1)
*!*			=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (SigChe)')
*!*			Return 0
*!*		EndIf

*!*		SELECT * from crTmpChe into cursor tempchq 

	DO FORM SigOpChi WITH 'CARTEIRA',.t.,tempchq.bancos,tempchq.agencias,tempchq.ncontas,tempchq.ncheques, tempchq.vencs

*!*		SELECT * from crGraCart into cursor tempchq readwrite	&& recupera cursor da grade 
*!*		SELECT * from crCheCart into cursor crSigche readwrite	&& recupera cursor de cheques
*!*		SELECT tempchq
*!*		GO lnRec
	ThisForm.Enabled = .T.
	thisform.Refresh
EndIf 

ENDPROC
