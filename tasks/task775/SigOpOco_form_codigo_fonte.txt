PROCEDURE Valid
If !isEmpty(This.Value)
	lStrQuery = [Select * From SigBaOco Where Tipos = 'O' And Autos <> 'S' ]
	If ThisForm.ParentForm.poDataMgr.SqlExecute(lStrQuery,'CsOcorr') < 1
		MessageBox('Favor reinicializar o processo !!!',16,'Falha na Conexão (CsOcorr)')
		Return(.F.)
	EndIf
	Select CsOcorr
	Index On Codigos Tag Codigos
	Index On Descrs  Tag Descrs
	Set Order To Codigos
	Go Top
	If !Seek(This.Value)
		m.loLista = CreateObject('fwBuscaInt','CsOcorr','Codigos',This.Value,'Ocorrências',.T.)
		m.loLista.mAddColuna('codigos','XXXXXXXXXX','Código')
		m.loLista.mAddColuna('descrs','','Descrição')
		m.loLista.Show()
		If LastKey() <> 27
			This.Value = Evaluate('CsOcorr.codigos')
			Replace Usuars With Usuar In Historico
		Else
			This.Value = ''
		EndIf	
	EndIf
	This.Refresh
EndIf
Return(.T.)
ENDPROC
