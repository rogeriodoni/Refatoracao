PROCEDURE Init
LParameters lcAcesso

lcSqlTTri = [Select a.*, b.Descrs As Origemd, c.Descrs As Destinod, d.Descricaos As CFOPd From SigPrTri a ]+;
                         [Left Join SigCdUfs b on a.Origems = b.Estados ]+;
                         [Left Join SigCdUfs c on a.Destinos = c.Estados ]+;
                         [Left Join SigCdCfo d On a.Cfops = d.Codigos]

With ThisForm
	.poDataMgr = CreateObject('fSqlConector', ThisForm.Name)

	If (.poDataMgr.pnIdconn > 0)
		With .poDataMgr
			.AddCursor('SigPrTri', 'cIdChaves', 'crSigPrTri', '', ThisForm.Pagina.Lista.Grade, lcSqlTTri)
			.pfSqlTabela(1).pColuna('Tipos', '!', '', 'Tipo', 40, .t.)
			.pfSqlTabela(1).pColuna('Descs', '', '', 'Descrição', 130, .t.)
			.pfSqlTabela(1).pColuna('CFOPs', '', '', 'CFOP', 100, .t.)
			.pfSqlTabela(1).pColuna('Origems', '', '', 'Origem', 50, .t.)
			.pfSqlTabela(1).pColuna('Destinos', '', '', 'Destino', 50, .t.)
			.pfSqlTabela(1).pColuna('Movs', '', '', 'Movs', 30, .t.)
			.pfSqlTabela(1).pColuna('Aliquotas', '', '', 'Alíquota', 70, .t.)
			.pfSqlTabela(1).pColuna('Bases', '', '', 'B. Reduzida', 70, .t.)
			.pfSqlTabela(1).pColuna('Reduzidas', '', '', 'Aliq. Red.', 70, .t.)
		EndWith

		* Aqui Vai o Código Deste Método

		=DoDefault()

		ThisForm.Pagina.Lista.Grade.Column1.SetFocus
	Else
		=Messagebox('Impossível Efetuar Conexão Com o Servidor de Banco de Dados...', 0+48, 'CreateObject')
		Return .f.
	EndIf
EndWith
* Fim **************************************************************************************

If ! fChecaAcesso('SIGCDICT','DUPLICA')
	ThisForm.Pagina.Lista.Shape1.Visible = .f.
	ThisForm.Pagina.Lista.Command1.Visible = .f.
Endif
ENDPROC

PROCEDURE msv_procurar
Declare laTTri[1,2]

lnTTri=1

If ! Empty(ThisForm.pagina.dados.Container1.GetTipo.Value)
	laTTri[1,1] = 'Tipos'
	laTTri[1,2] = ThisForm.pagina.dados.Container1.GetTipo.Value
	lnTTri      = lnTTri+1
Endif

If ! Empty(ThisForm.pagina.dados.Container1.GetDesc.Value)
	If lnTTri > 1
		Declare laTTri[Alen(laTTri,1)+1,2] 
	Endif
	laTTri[lnTTri,1] = 'Descs'
	laTTri[lnTTri,2] = ThisForm.pagina.dados.Container1.GetDesc.Value
	lnTTri      = lnTTri+1
Endif

If ! Empty(ThisForm.pagina.dados.Container1.GetCFOP.Value)
	If lnTTri > 1
		Declare laTTri[Alen(laTTri,1)+1,2] 
	Endif
	laTTri[lnTTri,1] = 'CFOPs'
	laTTri[lnTTri,2] = ThisForm.pagina.dados.Container1.GetCFOP.Value
	lnTTri      = lnTTri+1
Endif

If ! Empty(ThisForm.pagina.dados.Container1.GetOrigem.Value)
	If lnTTri > 1
		Declare laTTri[Alen(laTTri,1)+1,2] 
	Endif
	laTTri[lnTTri,1] = 'Origems'
	laTTri[lnTTri,2] = ThisForm.pagina.dados.Container1.GetOrigem.Value
