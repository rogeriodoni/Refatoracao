LParameters poForm

ThisForm.ParentForm = poForm

With Thisform.CrtFiltro
	.Get_GrVenda.Value   = ''
	.Get_GrProduto.Value = ''
	.Get_MoeVenda.Value  = ''
	.Get_AcreDesc.Value  = 0
	.Get_Arrend.Value    = 0
	.Opc_Situacao.Value  = 1
EndWith

If ThisForm.ParentForm.poDataMGr.SqlExecute([Select 0 As lMarcas,linhas,descs From SigCdLin ],'CsSigCdLin') < 1
	MessageBox('Favor reinicializar o processo.',16,'Falha na Conexão')
	ThisForm.Release()
EndIf

With ThisForm.CrtFiltro.GradeLinhas
	.RecordSource = 'CsSigCdLin'
	.Column1.ControlSource = 'CsSigCdLin.lMarcas'
	.Column2.ControlSource = 'CsSigCdLin.linhas'
	.Column3.ControlSource = 'CsSigCdLin.descs'
	.Refresh
EndWith

ThisForm.GradeItens.SetAll('DynamicForeColor','RGB(0,0,Iif(vlCdPrLp.Marcas = 0,0,255))','Column')

ThisForm.GradeItens.RecordSource = ''

Create Cursor CsSigCdPro (flagctabs n(1), cpros c(14), dpros c(40), pvens n(12,2), pliqs n(12,2), moevs c(3), situas n(1),;
						cgrus c(3), colecoes c(10), linhas c(10), lMarcas n(1))
Index On cpros + cgrus Tag CProGrus

With ThisForm.GradeItens
	.RecordSource = 'CsSigCdPro'
	.Column1.ControlSource = 'CsSigCdPro.flagctabs'
	.Column2.ControlSource = 'CsSigCdPro.cpros'
	.Column3.ControlSource = 'CsSigCdPro.dpros'
	.Column4.ControlSource = 'CsSigCdPro.pvens'
	.Column5.ControlSource = 'CsSigCdPro.moevs'
	.Column6.ControlSource = 'CsSigCdPro.pliqs'
	.SetAll('DynamicForeColor','RGB(0,0,Iif(CsSigCdPro.lMarcas = 0,0,255))','Column')
	.Refresh
EndWith

ThisForm.CopiaLista.Enabled = .F.
ENDPROC

PROCEDURE Click
Delete From CrSigCdLpi Where Empty(cpros)

Select CsSigCdPro
Go Top
Scan 
	If CsSigCdPro.flagctabs = 1 And CsSigCdPro.lMarcas = 0 And !Seek(CsSigCdPro.cpros,'CrSigCdLpi','CPros')
		lnSituacao  = Thisform.CrtFiltro.Opc_Situacao.Value
		If lnSituacao <> 3 And lnSituacao <> CsSigCdPro.situas
			Loop
		EndIf
		Scatter MemVar
		m.lprecos   = CrSigCdLpc.lprecos
		m.flagutabs = .T.
		m.cidchaves = fUniqueIds()
		m.vencis    = {}
		m.vencfs    = {}
		m.pvens     = m.pliqs

		If (ThisForm.ParentForm.CompVenda <> [C])
			m.pvens = fArredondamento(m.pvens, ThisForm.ParentForm.poDataMgr)
		EndIf

	    Insert Into CrSigCdLpi From MemVar
	EndIf
	Select CsSigCdPro
EndScan

ThisForm.Release()
ENDPROC

* Registro: 48
================================================================================

------------------------------------------------------------
* PROCEDURE Click
------------------------------------------------------------
PROCEDURE Click
Local vCond, lnMarcas, lnVariacao, lnArred, lnPLiqs

If Used('CsSigCdPro')
	Zap In CsSigCdPro
EndIf

ThisForm.GradeItens.Refresh

Wait Window 'Aguarde !!! Selecionando Produtos...' NoWait

lnAjuste = 1 + (ThisForm.CrtFiltro.Get_AcreDesc.Value / 100)

With ThisForm.CrtFiltro
	lcCGrus    = Iif(!Empty(.Get_GrProduto.Value),.Get_GrProduto.Value,'')
	lcColecoes = Iif(!Empty(.Get_GrVenda.Value)  ,.Get_GrVenda.Value  ,'')
	lcMoevs    = Iif(!Empty(.Get_MoeVenda.Value) ,.Get_MoeVenda.Value ,'')
	lnSitua    = .Opc_Situacao.Value
Endwith	

lWherePro = [0 = 0 ]+;
			Iif(Empty(lcCGrus)   ,'',[And cgrus = ']+lcCGrus+[' ])+;
			Iif(Empty(lcColecoes),'',[And colecoes = ']+lcColecoes+[' ])+;
			Iif(Empty(lcMoevs)   ,'',[And moevs = ']+lcMoevs+[' ])

lcLinhas  = ''
Select CsSigCdLin
Go Top
Scan
	If CsSigCdLin.lMarcas = 1
		lcLinhas = lcLinhas + [linhas = ']+CsSigCdLin.linhas+[' ] + [Or ]
	EndIf
	Select CsSigCdLin
EndScan
If !Empty(lcLinhas)
	lWherePro = lWherePro + [And (]+Substr(lcLinhas,1,Len(lcLinhas)-3)+[) ]
EndIf

lStrQuery = [Select flagctabs,cpros,dpros,pvens,moevs,situas,cgrus,colecoes,linhas ]+;
			[From SigCdPro ]+;
			[Where ]+lWherePro+;
			[Order By cpros,cgrus,flagctabs,dpros,pvens,moevs,situas,colecoes,linhas ]
If ThisForm.ParentForm.poDataMgr.SqlExecute(lStrQuery,'TmpPro') < 1
	MessageBox('Favor reinicializar o processo.',16,'Falha na Conexão')
	ThisForm.Release()
EndIf
		
lnVariacao = 1 + (Thisform.CrtFiltro.Get_AcreDesc.Value / 100)
lnArred    = Thisform.CrtFiltro.Get_Arrend.Value
Select TmpPro
Go Top
Scan
	Scatter MemVar
	m.flagctabs = Iif(Seek(TmpPro.cpros,'CrSigCdLpi','CPros'),1,0)
	m.lmarcas   = m.flagctabs
	m.pliqs = Round(TmpPro.pvens * lnVariacao,2)
	If lnArred # 0
		m.pliqs = Round(m.pliqs / lnArred,0) * lnArred
	EndIf
	Insert Into CsSigCdPro From MemVar
	Select TmpPro
EndScan

Select CsSigCdPro
ThisForm.GradeItens.Refresh
Go Top
If Eof()
	MessageBox('Não existem produtos com esses parâmetros !!!',16,'')
	ThisForm.CrtFiltro.Get_GrVenda.SetFocus
	Return(.T.)
Else
	ThisForm.CopiaLista.Enabled = .T.
	ThisForm.GradeItens.Column1.SetFocus
Endif
ENDPROC
