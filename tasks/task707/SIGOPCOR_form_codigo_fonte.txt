PROCEDURE Load
=fConfigGeral()

Create Cursor XCorNf (cods n(3),descs c(30),retifs m NULL)
ENDPROC

------------------------------------------------------------
* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
LParameters pForm,Emp,Serie,EmisDe,EmisAte
Private pcEmps,pcSeries,pcNFisDe,pcNFisAte

ThisForm.poDataMgr = CreateObject('fSqlConector',.Name)

ThisForm.ParentForm = pForm

pcEmps    = ''
pcSeries  = ''
pcNFisDe  = Null
pcNFisAte = Null

lcQryCorNf = [Select * From SigNfCor ]+;
 			  [Where cemps = ?pcEmps And cseries = ?pcSeries And cnfis between ?pcNFisDe And ?pcNFisAte ]+;
		   [Order By cemps,cseries,cnfis,noccors ]
lcQryEmp   = [Select * From SigCdEmp Where cemps = ?pcEmps ]

With ThisForm.poDataMgr
	.AddCursor('SigNfCor','cidchaves','CrSigNfCor','','',lcQryCorNf)
	.AddCursor('SigCcOco','ncodigos' ,'CrSigCcOco','','')
	.AddCursor('SigCdEmp'  ,'cemps'    ,'CrSigCdEmp'  ,'','',lcQryEmp)
EndWith

ThisForm.poDataMgr.Requery('CrSigNfCor')
ThisForm.poDataMgr.Requery('CrSigCcOco')

ThisForm.Emp     = Emp
ThisForm.Serie   = Serie
ThisForm.EmisDe  = EmisDe
ThisForm.EmisAte = EmisAte

pcEmps = Emp

ThisForm.poDataMgr.Requery('CrSigCdEmp')

Select CrSigCcOco
Scan
	Insert Into XCorNf (cods,descs,retifs) ;
	Values (CrSigCcOco.ncodigos,CrSigCcOco.cdescrs,'')
EndScan
With ThisForm.Grade
	.RecordSource = 'XCorNf'
	.Column1.ControlSource = 'XCorNf.cods'
	.Column2.ControlSource = 'XCorNf.descs'
	.Column3.ControlSource = 'XCorNf.retifs'
	.Refresh
EndWith
Select XCorNf
Go Top
ENDPROC

PROCEDURE KeyPress
LParameters nKeyCode, nShiftAltCtrl
Do Case
	Case nKeyCode = 13 or nKeyCode = 24
		Select XCorNf
		If Not Eof()
			Skip
		EndIf
		ThisForm.Grade.Refresh
	Case nKeyCode = 5 
		Select XCorNf
		If Not Bof()
			Skip -1
		EndIf
		ThisForm.Grade.Refresh
	Case nKeyCode = 9
		KeyBoard "{CTRL+TAB}"
EndCase
ENDPROC
