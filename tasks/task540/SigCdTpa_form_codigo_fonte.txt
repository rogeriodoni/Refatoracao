------------------------------------------------------------
* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
LPARAMETERS _acesso

ThisForm.Busca=1

WITH Thisform
	.poDatamgr = CreateObject('fSqlConector',ThisForm.Name)

	If Not .poDatamgr.pnIdconn>0
		Messagebox('Impossivel Efetuar Conexão com Servidor de Banco de Dados.',(0+48),goSistema.TituloSistema)
		Return(.f.)
	Endif

	With .poDatamgr
		.AddCursor('SigCdAca','Cods','crSigCdAca','Cods',ThisForm.pagina.lista.grade)
		.pfSqlTabela(1).pColuna('Cods','','','Código',80,.t.)
		.pfSqlTabela(1).pColuna('Descrs','','','Descrição',150,.t.)
	EndWith
Endwith	
=DoDefault()

ENDPROC

------------------------------------------------------------
* PROCEDURE mativapagina1
------------------------------------------------------------
PROCEDURE mativapagina1
=DoDefault()

lcCol = Padr(ThisForm.Busca,1)
ThisForm.Pagina.Lista.Grade.SetFocus
ThisForm.Pagina.Lista.Grade.Column&lcCol..SetFocus
ThisForm.Pagina.Lista.Grade.Refresh
ENDPROC

------------------------------------------------------------
* PROCEDURE msv_procurar
------------------------------------------------------------
PROCEDURE msv_procurar
WITH ThisForm
	WITH .pagina.dados
		lcCodigo=.GetCodigos.Value
		lcDescr =.GetDescrs.value
		If Not Empty(lcCodigo)
			Select CrSigCdAca
			Set Order To Cods
			Set Near On
			Seek lccodigo
			Set Near Off
			Thisform.pagina.lista.grade.columns(1).setfocus
			ThisForm.Busca=1
		ELSE
			SELECT crSigCdAca
			SET ORDER TO Descrs
			SET NEAR on
			SEEK lcDescrs
			SET NEAR off
			Thisform.pagina.lista.grade.columns(2).setfocus
			ThisForm.Busca=2
		Endif
	ENDWITH
EndWith

ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Lista.Grupo_op.Click
------------------------------------------------------------
PROCEDURE Pagina.Lista.Grupo_op.Click
LPARAMETERS opcao

If !DoDefault( Opcao )
	ThisForm.mAtivaPagina1
	Return
Endif

Select crSigCdAca

Do Case
	Case Inlist( ThisForm.pcEscolha, 'PROCURAR', 'INSERIR' )
		ThisForm.Pagina.Dados.GetCodigos.Setfocus	

	Case ThisForm.pcEscolha = 'ALTERAR'
		ThisForm.Pagina.Dados.getDescrs.Setfocus	

	Case Inlist( ThisForm.pcEscolha, 'CONSULTAR', 'EXCLUIR' )
		ThisForm.Pagina.Dados.Grupo_Salva.Salva.Setfocus
Endcase

Select crSigCdAca
ThisForm.Pagina.Dados.Refresh

ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
If ThisForm.pcEscolha = 'INSERIR'
	If IsEmpty(ThisForm.Pagina.Dados.getCodigos.Value)
		MessageBox( 'Obrigatório Informar o Código!!!', 48, '' )
		ThisForm.Pagina.Dados.getCodigos.SetFocus
		Return
	EndIf
   	If ThisForm.poDataMgr.ChkRegister('SigCdAca','Cods',ThisForm.Pagina.Dados.GetCodigos.Value)
		MessageBox( 'Código já cadastrado...', 48, '' )
		ThisForm.Pagina.Dados.GetCodigos.SetFocus
		Return
	Endif
EndIf

=DoDefault()
ThisForm.mAtivaPagina1

ENDPROC

=DoDefault()

	ThisForm.mAtivaPagina1
Endif

ENDPROC
