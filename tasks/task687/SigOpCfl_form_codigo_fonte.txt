PROCEDURE Init
LParameters pConta, pData

ThisForm.cntSombra.lblSombra.Caption = ThisForm.Caption
ThisForm.cntSombra.lblTitulo.Caption = ThisForm.Caption

If Type('pConta')#'C'
	pNum = Space(10)
EndIf

ThisForm.conta = pConta
ThisForm.Tipo = ''
lcNum = pConta

lcQueryCheh  = [Select * From SigCheh Where Nums = ?lcNum ]

With ThisForm
	.poDataMgr = CreateObject('fSqlConector','SigChefl')
	With .poDataMgr
		.AddCursor('SigCheh','CidChaves','CrSigCheh','','',lcQueryCheh)
	EndWith
	= DoDefault()
EndWith

=ThisForm.Podatamgr.Requery('CrSigCheh')

=ThisForm.Podatamgr.Cursorquery('SigChe','CrTmpChe','Nums',lcNum)
=ThisForm.Podatamgr.Cursorquery('SigCdCli','CrTmpCli','Iclis',CrTmpChe.Iclis)

ThisForm.GetConta.Value  = CrTmpCli.Iclis
ThisForm.GetdConta.Value = CrTmpCli.Rclis
Thisform.CntDadoschq.Get_bco.Value = CrTmpChe.Bancos
Thisform.CntDadoschq.Get_agc.Value = CrTmpChe.agencias
Thisform.CntDadoschq.Get_cta.Value = CrTmpChe.nContas
Thisform.CntDadoschq.Get_Nch.Value = CrTmpChe.nCheques

lcPos = 5
lcLrg = 80
With ThisForm.cmdApagaItem
	.cmdInserir.Visible = fChecaAcesso('SIGOPCFL','INSERIR')
	.cmdAlterar.Visible = fChecaAcesso('SIGOPCFL','ALTERAR')
	.cmdApagar.Visible = fChecaAcesso('SIGOPCFL','EXCLUIR')
	.cmdListar.Visible = fChecaAcesso('SIGOPCFL','LISTAR')

	If .cmdInserir.Visible
		.cmdInserir.Left = lcPos
		lcPos = lcPos + lcLrg
	Endif
	If .cmdAlterar.Visible
		.cmdAlterar.Left = lcPos
		lcPos = lcPos + lcLrg
	Endif
	If .cmdApagar.Visible
		.cmdApagar.Left = lcPos
		lcPos = lcPos + lcLrg
	Endif

	If .cmdListar.Visible
		.cmdListar.Left = lcPos
		lcPos = lcPos + lcLrg
	Endif
	
	If lcPos > 5
		.Width = lcPos + 5
		.Refresh
	Else
		.Visible = .f.
	Endif	
EndWith

Select crSigCheH
Go Top

With ThisForm.gridFollowup
	.ColumnCount = 2
	.DeleteMark = .f.
	.RecordSource = 'crSigCheh'
	.Scrollbars = 2
	.Width = 667
	
	.Column1.Controlsource = 'crSigCheH.Datas'
	.Column1.Movable = .f.
	.Column1.Resizable = .f.
	.Column1.Width = 144
	.Column1.Header1.Alignment = 2
	.Column1.Header1.Caption = 'Data / Hora'
	
	.Column2.InputMask = 'XXXXXXXXXXXXXXXXXXXXXXXXXXXXXXXXXXXXXXXXXXXXXXXXXXXXXXXXXXXXXXXXXXXXXXXXXXXX'
	.Column2.Controlsource = 'crSigCheH.Hists'
	.Column2.Movable = .f.
	.Column2.Resizable = .f.
	.Column2.Width = 490
	.Column2.Header1.Alignment = 2
	.Column2.Header1.Caption = 'Observação'

	.SetAll('DynamicForeColor','Iif(crSigCheh.Autos,RGB(0,128,64),RGB(0,0,0))','Column')

EndWith

ENDPROC

PROCEDURE cmdApagar.Click
Select CrSigCheh
If !Eof()
	If !CrSigCheH.Autos .or. IsEmpty(CrSigCheh.Hists) Or IsEmpty(CrSigCheh.Datas)
		pchave = CrSigCheh.CidChaves
		lcDelete = [ Delete From SigCheh Where CidChaves = ']+pChave+[']
		lnErro = ThisForm.poDatamgr.SqlExecute(lcDelete,'')
		If lnErro > 0
			ThisForm.poDatamgr.Commit()
			Select CrSigCheh
			Delete
			Skip
			Skip -1
		Else
			MessageBox('Favor Reiniciar o Processo !!!',0+48,'Falha na Conexão')
		Endif
	Else
		MessageBox('Histórico automático !!!',0+48,'Impossível Excluir')
	EndIf
Endif
Select CrSigCheH
Thisform.GridFollowUp.Refresh
Thisform.GridFollowUp.Column1.Text1.Setfocus

ENDPROC

PROCEDURE LostFocus
Local lcConta
lcConta = ThisForm.Conta
Select CrSigCheh

If !Empty(This.Value)
	If ThisForm.Tipo = 'INSERIR'
		pChave = fUniqueIds()
		Select CrSigCheh
		Append Blank
		Replace Nums      With lcConta,;
				Datas     With This.Parent.GetData.Value ,;
				Emps      With _Empr,;
				CidChaves with pChave
	Endif
	Replace Hists  With This.Parent.GetHist.Value

	ThisForm.podatamgr.Update('CrSigCheh')
	ThisForm.podatamgr.commit()
Endif
ThisForm.cnt_inserir.Visible = .f.
ThisForm.gridFollowup.Refresh
ThisForm.GridFollowup.Column1.SetFocus
ENDPROC
