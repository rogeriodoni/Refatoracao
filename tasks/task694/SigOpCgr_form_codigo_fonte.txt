PROCEDURE processa
Local lcArquivo1

lcArquivo1 = Sys(2023) + '\Temp3.jpg'

* Caso a Consulta Já Esteja Executada, Não Processa Novamente
If Not Thisform.ConsultaExecutada
	Wait Window 'Processando...' NoWait

	pcMensagem = .f.
	
	*-[ Limpa os Arquivos ]-*
	Select DbProduto
	Zap
	
	lnOp   = ThisForm.fld_principal.pag_consulta.Origem.Pedido.Get_Op.Value
	lnEnv  = ThisForm.fld_principal.pag_consulta.Origem.Pedido.Get_Env.Value
	lcPed  = ThisForm.fld_principal.pag_consulta.Origem.Pedido.Get_Pedido.Value
	lnPed  = ThisForm.fld_principal.pag_consulta.Origem.Pedido.Get_Numero.Value
	lcEDN  = _Empr + lcPed + Str(lnPed,6)
	
	If (lnOp <> 0)
		ThisForm.poDataMgr.CursorQuery('SigOpPic', 'crSigOpPic', 'Nops', lnOp, 'Dopes, Numes, Qtds')
		Select crSigOpPic
		Scan
			lcQuery = [Select Top 1 Nops, Nenvs, CodPds, Grupods, Contads ] + ;
					    [From SigPdMvf ] + ;
					   [Where Nops = ] + Str(lnOp, 10) + [ ] + ;
					   [Order By Nops Desc, Nenvs Desc, CodPds Desc, Grupods Desc, Contads Desc]

			If (ThisForm.poDataMgr.SqlExecute(lcQuery, 'crSigPdMvf') < 1)
				=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (crSigPdMvf 1)')
				Return 0
			EndIf
			Select crSigPdMvf
			Go Top

			Select DbProduto
			Append Blank
			Replace Nenvs   With crSigPdMvf.Nenvs, ;
					Nops    With crSigPdMvf.Nops, ;
					DopNums With crSigOpPic.Dopes + Str(crSigOpPic.Numes, 6), ;
					Cpros   With crSigPdMvf.CodPds, ;
					Qtds    With crSigOpPic.Qtds, ;
					Grupods With crSigPdMvf.Grupods, ;
					Contads With crSigPdMvf.Contads
		EndScan
	Else
		If (lnEnv > 0)
			lcQuery = [Select Top 1 EmpDNps ] + ;
					    [From SigPdMvf ] + ;
					   [Where Nenvs = ] + Str(lnEnv, 10) + [ ] + ;
					   [Order By Nenvs Desc, EmpDNps Desc]

			If (ThisForm.poDataMgr.SqlExecute(lcQuery, 'crTopMfas') < 1)
				=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (crTopMfas)')
				Return 0
			EndIf
			Select crTopMfas
			Go Top
			If Not Eof()
				_Chave = crTopMfas.EmpDNps

				lcQuery = [Select Nenvs, Nops, CodPds, Grupods, Contads, Emps, Dopps, Numps ] + ;
						    [From SigPdMvf ] + ;
						   [Where EmpDNps = '] + _Chave + [' ] + ;
						   [Order By Nenvs, Nops, CodPds, Grupods, Contads, Emps, Dopps, Numps]

				If (ThisForm.poDataMgr.SqlExecute(lcQuery, 'crSigPdMvf') < 1)
					=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (crSigPdMvf 2)')
					Return 0
				EndIf
				Select crSigPdMvf
				Scan
					ThisForm.poDataMgr.CursorQuery('SigOpPic', 'crSigOpPic', 'Nops', crSigPdMvf.Nops, 'Dopes, Numes, Qtds')
					Select crSigOpPic
					Scan
						Select DbProduto
						Append Blank
						Replace Nenvs   With crSigPdMvf.Nenvs, ;
								Nops    With crSigPdMvf.Nops, ;
								DopNums With crSigOpPic.Dopes + Str(crSigOpPic.Numes, 6), ;
								Cpros   With crSigPdMvf.CodPds, ;
								Qtds    With crSigOpPic.Qtds, ;
								Grupods With crSigPdMvf.Grupods, ;
								Contads With crSigPdMvf.Contads
					EndScan
				EndScan
			EndIf
		Else
			ThisForm.poDataMgr.CursorQuery('SigOpPic', 'crSigOpPic', 'EmpDopNums', lcEDN, 'Nops, Dopes, Numes, Qtds')
			Select crSigOpPic
			Scan
				lcQuery = [Select Top 1 Nops, Nenvs, CodPds, Grupods, Contads ] + ;
						    [From SigPdMvf ] + ;
						   [Where Nops = ] + Str(crSigOpPic.Nops, 10) + [ ] + ;
						   [Order By Nops Desc, Nenvs Desc, CodPds Desc, Grupods Desc, Contads Desc]

				If (ThisForm.poDataMgr.SqlExecute(lcQuery, 'crSigPdMvf') < 1)
					=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (crSigPdMvf 3)')
					Return 0
				EndIf
				Select crSigPdMvf
				Go Top

				Select DbProduto
				Append Blank
				Replace Nenvs   With crSigPdMvf.Nenvs, ;
						Nops    With crSigPdMvf.Nops, ;
						DopNums With crSigOpPic.Dopes + Str(crSigOpPic.Numes,6), ;
						Cpros   With crSigPdMvf.CodPds, ;
						Qtds    With crSigOpPic.Qtds, ;
						Grupods With crSigPdMvf.Grupods, ;
						Contads With crSigPdMvf.Contads
			EndSCan
		EndIf
	EndIf

	Thisform.ConsultaExecutada = .t.
EndIf

Wait Clear
* Fim **************************************************************************************

ENDPROC

PROCEDURE Click
Local llRetorno

m.llRetorno =.t.

If (Empty(This.Parent.Origem.Pedido.Get_Env.Value) And ;
	Empty(This.Parent.Origem.Pedido.Get_Op.Value) And ;
	Empty(This.Parent.Origem.Pedido.Get_Pedido.Value))
	      
	=Messagebox('Nenhum Opção de Filtro Foi Informada!!!', 16, '')
	This.Parent.Origem.Pedido.Get_OP.SetFocus
	m.llRetorno = .f.
EndIf

If m.llRetorno
	m.pcMensagem = .f.
	ThisForm.LockScreen = .t.
	ThisForm.Processa()
	
	If Not Eof('DbProduto')
		ThisForm.Fld_Principal.ActivePage = 2
		ThisForm.Refresh()
	Else
		=Messagebox('Nenhum Registro Encontrado!!!', 16, '')
	EndIf

	ThisForm.LockScreen = .f.
	
	With This.Parent.Parent.pag_resultado.Container1
		.Chk_op.Value  = Iif(Empty(This.Parent.Origem.Pedido.Get_Op.Value), 0, 1)
		.Chk_Env.Value = Iif(Empty(This.Parent.Origem.Pedido.Get_Env.Value), 0, 1)
		.Chk_Dop.Value = Iif(Empty(This.Parent.Origem.Pedido.Get_Pedido.Value), 0, 1)
	EndWith
EndIf
* Fim **************************************************************************************

ENDPROC
