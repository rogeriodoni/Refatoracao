------------------------------------------------------------
* PROCEDURE validarmd5
------------------------------------------------------------
PROCEDURE validarmd5
Lparameters pMov

Local lcMd5

lcQry = [Select * From SigCdPaf Where Dopes = '] + fDecriptografar(pMov) + [']
If ThisForm.poDataMgr.SqlExecute(lcQry, [LocalMd5]) < 1
	=MessageBox([Favor Reinicializar o Processo!!! Falha na Conexão (LocalMd5)], 0+16, [Erro!!!])
	Return .f.
EndIf
Select LocalMd5
Go Top

lcMd5 = fGerarMD52016([LocalMd5], [cMd5s])

If (LocalMd5.cMd5s <> lcMd5)
	Return .f.
EndIf

Return .t.
ENDPROC

------------------------------------------------------------
* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
LParameters lcAcesso

With ThisForm
	.poDatamgr = CreateObject('fSqlConector',ThisForm.Name)

	lcSql = [Select a.Dopes,b.OpeInatvs,b.cnCaixas,c.CupFis,c.nFiscals,c.Agrupas,c.Docus,c.TrNfis From SigOpCdd a, SigOpCdc b, SigCdOpe c ]+;
			[Where a.pafEcfs = 1 And a.Dopes = b.dopes And a.Dopes = c.Dopes ]

	ThisForm.Podatamgr.SqlExecute(lcSql,'LocalOpe2')
	Select LocalOpe2
	Index on Dopes Tag Dopes
	Locate for .f.

	lcSql = [Select Dopes,cncaixas,trnfis From SigCdPaf ]
	ThisForm.Podatamgr.SqlExecute(lcSql,'CrTmpPafEc')
	Select CrTmpPafEc
	Locate for .f.
	Update CrTmpPafEc Set Dopes = fCriptografar(Dopes), cncaixas = fCriptografar(cnCaixas), trnfis= fCriptografar(trNFis)
	Locate for .f.

	Select * From LocalOpe2 WHERE Dopes NOt in (select dopes from CrTmpPafEc) ;
	Into Cursor CsSelecao
	If Reccount('CsSelecao') > 0

		Create Cursor TmpPafEc (Dopes c(20), Inativas c(1), Docs c(1), cnCaixas c(3), TrNfis c(2), cmd5s c(32))

		Select CsSelecao
		Scan
			MessageBox('Atenção!!!, Operação :'+CsSelecao.Dopes+' Esta Inconsistente. O sistema irá assumir as Configurações Padrões!!!!',0+16,'')
			
			lcDoc = '0'
			Do Case
				Case CsSelecao.CupFis = 1
					lcDoc = '1'
				Case CsSelecao.CupFis = 3
					lcDoc = '2'
				Case CsSelecao.CupFis = 2 And CsSelecao.Docus = 3
					lcDoc = '3'
				Case CsSelecao.CupFis = 2 And CsSelecao.Docus = 2
					lcDoc = '4'
			EndCase

			Insert Into TmpPafEc (Dopes,Inativas,Docs,cnCaixas,TrNfis) Values ;
					(fDecriptografar(CsSelecao.Dopes),fDecriptografar(Str(CsSelecao.OpeInatvs,1)),;
					 fDecriptografar(lcDoc),fDecriptografar(CsSelecao.cnCaixas),fDecriptografar(Str(CsSelecao.TrNfis,2)))
			
			lcMd5 = fGerarMD52016('TmpPafEc', 'cMd5s')

			Replace cMd5s With lcMd5 In TmpPafEc

			lcSql = [Insert Into SigCdPaf (Dopes, Inativas, Docs,cMd5s, cnCaixas, TrNfis ) Values (]+;
					[']+TmpPafEc.Dopes+[',']+TmpPafEc.Inativas+[',']+TmpPafEc.docs+[',]+;
					[']+TmpPafEc.cMd5s+[',']+TmpPafEc.cnCaixas+[',']+TmpPafEc.TrNfis+[')]

			ThisForm.Podatamgr.SqlExecute(lcSql,'')
		EndScan
		
		Select Dopes From CrTmpPafEc Where dopes Not in (Select Dopes From LocalOpe2) ;
		Into Cursor CsErro
		Select CsErro
		Scan
			lcDopePaf = fDecriptografar(CsErro.Dopes)
			lcDelete = [Delete From SigCdPaf Where Dopes = ']+lcDopePaf+[']
			ThisForm.Podatamgr.SqlExecute(lcDelete,'')
		EndScan

		Thisform.Podatamgr.Commit()
		
	EndIf

	Create Cursor TmpPafEc (Dopes c(20), Inativas c(1), Docs c(1), cnCaixas c(3), TrNfis c(2), cmd5s c(32))

	Select CrTmpPafEc
	Scan
		If Not ThisForm.ValidarMd5(CrTmpPafEc.Dopes) And Seek(CrTmpPafEc.dopes,'LocalOpe2','Dopes')
			MessageBox('Atenção!!!, Operação :'+CrTmpPafEc.Dopes+' Esta Inconsistente. O sistema irá assumir as Configurações Padrões!!!!',0+16,'')
			lcDoc = '0'
			Do Case
				Case LocalOpe2.CupFis = 1
					lcDoc = '1'
				Case LocalOpe2.CupFis = 3
					lcDoc = '2'
				Case LocalOpe2.CupFis = 2 And LocalOpe2.Docus = 3
					lcDoc = '3'
				Case LocalOpe2.CupFis = 2 And LocalOpe2.Docus = 2
					lcDoc = '4'
			EndCase

			Insert Into TmpPafEc (Dopes,Inativas,Docs,cnCaixas,TrNfis) Values ;
					(fDecriptografar(CrTmpPafEc.Dopes),fDecriptografar(Str(LocalOpe2.OpeInatvs,1)),;
					 fDecriptografar(lcDoc),fDecriptografar(LocalOpe2.cnCaixas),fDecriptografar(Str(LocalOpe2.TrNfis,2)) )

			lcMd5 = fGerarMD52016('TmpPafEc', 'cMd5s')
			Replace cMd5s With lcMd5 In TmpPafEc
		
		Endif
	EndScan
	
	Select TmpPafEc
	Scan
		Scatter Memvar
		lcSql = [Update SigCdPaf Set Inativas = ']+TmpPafEc.Inativas+[', ]+;
				[Docs = ']+TmpPafEc.docs+[', cMd5s = ']+TmpPafEc.cMd5s+[', cnCaixas = ']+TmpPafEc.cnCaixas+[', ]+;
				[TrNfis = ']+TmpPafEc.TrNfis+[' Where Dopes=']+m.Dopes+[' ]

		ThisForm.Podatamgr.SqlExecute(lcSql,'')

		Thisform.Podatamgr.Commit()

	EndScan

	lcQueryPAF = [Select *, Dopes as DopePafEcfs, cnCaixas as CX, trNFis as TRNF From SigCdPaf ]
	With .poDatamgr
		.AddCursor('SigCdPaf','Dopes','CrSigCdPaf','',ThisForm.Pagina.Lista.Grade,lcQueryPaf)
		.pfSqlTabela(1).pColuna('DopePafEcfs','','','Função'  , 200,.T.)
		.pfSqlTabela(1).pColuna('CX','','','Nº do caixa',90,.T.)
		.pfSqlTabela(1).pColuna('TRNF','','','Finalizadora Não Fiscal',161,.T.)
		.AddCursor('SlvSerie','EmpCods','CqSlvSerie','','')
	EndWith
	
	Update CrSigCdPaf Set DopePafEcfs = fCriptografar(Dopes), cx = fCriptografar(cnCaixas), trnf = fCriptografar(trNFis)

	ThisForm.placexcluir = .f.
	ThisForm.plAcInserir = .f.
	
	=DoDefault()
	.LockScreen = .F.
EndWith

ThisForm.mAtivaPagina1

ThisForm.LockScreen = .T.
=fCarregaGrade(ThisForm.Pagina.Lista.Grade)
ThisForm.LockScreen = .F.
ENDPROC

			lcUpdate = [Update SigOpCdc Set verests = 2, opeinatvs = ]+lnInt+[, cncaixas = ']+lcCaixa+[' Where Dopes = ']+lcDopPaf+[']
			ThisForm.Podatamgr.SqlExecute(lcUpdate,'')

			lcUpdate = [Update SigCdOpe Set CupFis = 2, estoqs = 2, Docus = 3 Where Dopes = ']+lcDopPaf+[']
			ThisForm.Podatamgr.SqlExecute(lcUpdate,'')

			lcUpdate = [Update SigOpCdc Set verests = 3 , opeinatvs = ]+lnInt+[, cncaixas = ']+lcCaixa+[' Where Dopes = ']+lcDopPaf+[']

			ThisForm.Podatamgr.SqlExecute(lcUpdate,'')

			ThisForm.Podatamgr.SqlExecute(lcUpdate,'')

			lcUpdate = [Update SigOpCdc Set verests = 1, opeinatvs = ]+lnInt+[, cncaixas = ']+lcCaixa+[' Where Dopes = ']+lcDopPaf+[']
			ThisForm.Podatamgr.SqlExecute(lcUpdate,'')
 
		Case lnDoc = '5'

			lcUpdate = [Update SigCdOpe Set CupFis = 2, estoqs = 1, nfiscals = 2, Agrupas = 1, Docus = 2 Where Dopes = ']+lcDopPaf+[']
			ThisForm.Podatamgr.SqlExecute(lcUpdate,'')

			ThisForm.Podatamgr.SqlExecute(lcUpdate,'')
		
		Otherwise
			MessageBox('Configuração Invalida de Documento!!! ',0+16,'')

	EndCase

EndIf
