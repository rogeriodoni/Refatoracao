PROCEDURE montagrade
lnOldSele = Select()

Select CrSigCdVca
Go top

With ThisForm.Grade
	.RecordSource = [CrSigCdVca]	
	.Column1.ControlSource   = [CrSigCdVca.Iclis]
	.Column2.ControlSource   = [CrSigCdVca.rclis]
	.Column3.ControlSource   = [CrSigCdVca.Resp]
	.Column4.ControlSource   = [CrSigCdVca.Usuario]
	.Column5.ControlSource   = [CrSigCdVca.Deptos]
	.Refresh
EndWith 	

Select(lnOldSele)
Return .T.
ENDPROC

* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
ThisForm.cntSombra.lblSombra.Caption = ThisForm.Caption
ThisForm.cntSombra.lblTitulo.Caption = ThisForm.Caption

lcQryVca = [select a.*, c.rclis, b.deptos, b.ncomps as usuario ] + ;
	[from sigcdvca a ] + ;
	[inner join sigcdusu b on a.resp = b.usuarios ] + ; 
	[inner join sigcdcli c on a.iclis = c.iclis  ]   

ThisForm.Grade.RecordSource = []

With ThisForm
	.poDataMgr = CreateObject([fSqlConector], .Name)
	If (.poDataMgr.pnIdconn > 0)
		.poDataMgr.AddCursor('SigCdVca', 'pkChaves', 'CrSigCdVca','','', lcQryVca)
		.poDataMgr.AddCursor('SigCdGcr', 'Codigos',	 'CrSigCdGcr','','')
	Else
		=Messagebox([Impossível Efetuar Conexão Com o Servidor de Banco de Dados...], 0+48, [CreateObject])
		Return .f.
	EndIf	
EndWith 

ThisForm.poDataMgr.SqlExecute([Select * From SigCdPac],[crSigCdPac])
ThisForm.SigKey = CrSigCdPac.SigKeys

ThisForm.poDataMgr.SqlExecute([Select * From SigCdPaM],[crSigCdPaM])

ThisForm.poDataMgr.Requery([CrSigCdVca])
ThisForm.poDataMgr.Requery([CrSigCdGcr])

Select CrSigCdVca
Index on iclis Tag iclis additive 
Index on resp Tag resp additive
Index on rclis Tag rclis additive
Index on deptos Tag deptos additive
Index on usuario Tag usuario additive
Go Top

ThisForm.MontaGrade()








ENDPROC

PROCEDURE cmdok.Click
Local lnOldSel
lnOldSel = Select()

Select CrSigCdVca
Go Top

If Reccount([CrSigCdVca])=0
	Messagebox([Nenhum Registro para Atualização.],0+64,[Atenção])
	Return .F.
Endif

Select CrSigCdVca
Select Iclis, rClis, Count(*) as TT From CrSigCdVca Where Not Empty(Iclis) group by iclis, rClis having Count(*) > 1 Into Cursor cdCtaDup ReadWrite

Select cdCtaDup
Go Top
If Reccount([cdCtaDup]) > 0
	lcCta = [] 
	Scan 
		lcCta = lcCta + Iif(Empty(lcCta),[],[,])+ Alltrim(Iclis) + [ - ] + Alltrim(rClis)+Chr(13)
	EndScan 
	
	MessageBox([Existem Contas Duplicadas, Favor Verificar.] + Chr(13) + [Contas : ] + Chr(13) + lcCta ,0+64,[Atenção])
	Return .F.
EndIf 

lnResp=Messagebox([Deseja Gravar as Informações?.],4+32,[Atenção])
If lnResp<>6
	Return .F.
Endif

Select CrSigCdVca
Delete From CrSigCdVca Where Empty(IClis) or Empty(Resp)
Go Top

Select CrSigCdVca
Scan 
	Replace pkChaves With Sys(2015)+Sys(2015) in CrSigCdVca
EndScan 

lcDelete = [Delete From SigCdVca]
If Thisform.PodataMgr.SqlExecute(lcDelete) < 1
	MessageBox([Favor reiniciar o Processo.],0+16,[Falha de Conexão])
	Return .F.
EndIf 

Select CrSigCdVca
Locate for .F.
Go Top

llErro1 = Thisform.PodataMgr.Update('CrSigCdVca')

If !llErro1
	Messagebox('Favor reinicializar o processo.',16,'Falha na Conexão')
	ThisForm.poDataMgr.RollBack()
Else
	ThisForm.poDataMgr.Commit()
	Messagebox('Gravação Realizada.',64,'Atenção')
Endif

Select(lnOldSel)

thisform.Release
ENDPROC
