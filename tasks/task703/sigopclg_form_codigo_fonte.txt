PROCEDURE processamento
ThisForm.PgfPagina.PgResultado.GrdLog.RecordSource = ''

Private lcUserRede, ldDtIni, ldDtFin, lcHrIni, lcHrFin, lcPrg
Private lcOperacao, lcUserSis, lcParam, lcTrans 

With ThisForm.PgfPagina.PgDados
	lcUserRede	= .GetUserRede.Value
	ldDtIni		= fDtoSQL(.GetDtIni.Value)
	ldDtFin		= fDtoSQL(.GetDtFin.Value,'23:59:59')
	lcHrIni		= .GetHrIni.Value
	lcHrFin		= .GetHrFin.Value
	lcPrg		= .GetPrg.Value
	lcOperacao	= .GetOperacao.Value
	lcUserSis	= .GetUserSis.Value
	lcParam		= .GetParam.Value
	lcTrans 	= .GetTrans.Value
EndWith

If  IsEmpty(lcUserRede) And ;
    IsEmpty(ldDtIni)    And ;
	IsEmpty(ldDtFin)    And ;
	IsEmpty(lcHrIni)    And ;
	IsEmpty(lcHrFin)    And ;
	IsEmpty(lcPrg)      And ;
	IsEmpty(lcOperacao) And ;
	IsEmpty(lcUserSis)  And ;
	IsEmpty(lcParam)    And ;
	IsEmpty(lcTrans) 
	MessageBox("Pesquisa não pode ser realizada,"+Chr(13)+"Nenhum Parametro de Seleção foi Preenchido",16,"")
	Return
Endif

lcSqlLog = [Select '          'As cOpers,a.* From SigOpLog a Where ]
		   
lcWhere = []
lcWhere = lcWhere + Iif(!IsEmpty(lcUserRede),[ UsuaRedes = ']+lcUserRede+[' ],[])
lcWhere = lcWhere + Iif(!IsEmpty(ldDtIni),Iif(!Empty(lcWhere),[ And ],[])+[ Datars >= ?ldDtIni ],[])
lcWhere = lcWhere + Iif(!IsEmpty(ldDtFin),Iif(!Empty(lcWhere),[ And ],[])+[ Datars <= ?lddtFin ],[])
lcWhere = lcWhere + Iif(!IsEmpty(lcHrIni),Iif(!Empty(lcWhere),[ And ],[])+[ Horas >= ']+Substr(lcHrIni,1,2)+[:]+Substr(lcHrIni,3,2)+[:00' ],[])
lcWhere = lcWhere + Iif(!IsEmpty(lcHrFin),Iif(!Empty(lcWhere),[ And ],[])+[ Horas <= ']+Substr(lcHrFin,1,2)+[:]+Substr(lcHrFin,3,2)+[:00' ],[])
lcWhere = lcWhere + Iif(!IsEmpty(lcPrg),Iif(!Empty(lcWhere),[ And ],[])+[ Progs = ']+lcPrg+[' ],[])
lcWhere = lcWhere + Iif(!IsEmpty(lcOperacao),Iif(!Empty(lcWhere),[ And ],[])+[ Opers ]+Iif(Upper(lcOperacao)="ENTRADA",[= 'E'],Iif(Upper(lcOperacao)="SAIDA",[= 'S'],Iif(Upper(lcOperacao)="LIBERAÇÃO",[= 'L'],[<> 'E' And Opers <> 'S' And Opers <> 'L'])))+[ ],[])
lcWhere = lcWhere + Iif(!IsEmpty(lcUserSis),Iif(!Empty(lcWhere),[ And ],[])+[ Usuasis = ']+lcUserSis+[' ],[])
lcWhere = lcWhere + Iif(!IsEmpty(lcParam),Iif(!Empty(lcWhere),[ And ],[])+[ Parametros Like ']+Alltrim(lcParam)+[%' ],[])
lcWhere = lcWhere + Iif(!IsEmpty(lcTrans),Iif(!Empty(lcWhere),[ And ],[])+[ Transacaos Like ']+Alltrim(lcTrans)+[%' ],[])

ThisForm.poDatamgr.SqlExecute(lcSqlLog+lcWhere,"crSigOpLog")

Select crSigOpLog
Replace All cOpers With Iif(Opers='E','Entrada',Iif(Opers='S','Saida',Iif(Opers='L','Liberação','Transação')))
Go Top

If Eof()
	MessageBox("Nenhum Registro Encontrado para Seleção",64,"")
	Return
Endif

ThisForm.PgfPagina.PgResultado.GrdLog.RecordSource = 'crSigOpLog'
ThisForm.PgfPagina.PgResultado.GrdLog.Column1.ControlSource = 'crSigOpLog.Usuaredes'
ThisForm.PgfPagina.PgResultado.GrdLog.Column2.ControlSource = 'crSigOpLog.Datars'
ThisForm.PgfPagina.PgResultado.GrdLog.Column3.ControlSource = 'crSigOpLog.Horas'
ThisForm.PgfPagina.PgResultado.GrdLog.Column4.ControlSource = 'crSigOpLog.Progs'
ThisForm.PgfPagina.PgResultado.GrdLog.Column5.ControlSource = 'crSigOpLog.cOpers'
ThisForm.PgfPagina.PgResultado.GrdLog.Column6.ControlSource = 'crSigOpLog.Usuasis'
ThisForm.PgfPagina.PgResultado.GrdLog.Column7.ControlSource = 'crSigOpLog.Parametros'
ThisForm.PgfPagina.PgResultado.fwget1.ControlSource = 'crSigOpLog.Transacaos'


ThisForm.PgfPagina.PgResultado.Enabled=.t.
ThisForm.PgfPagina.ActivePage = 2

ENDPROC
