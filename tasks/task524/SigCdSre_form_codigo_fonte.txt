PROCEDURE msv_procurar

If !Empty(ThisForm.Pagina.Dados.getcods.Value)
	ThisForm.poDataMgr.LocateCursor('crSigCdSer','cods',ThisForm.Pagina.Dados.Getcods.Value)

	EndIf

------------------------------------------------------------
* PROCEDURE Pagina.Lista.Grupo_op.Click
------------------------------------------------------------
PROCEDURE Pagina.Lista.Grupo_op.Click
parameters lnOpcao

Select CrSigCdSer

If !dodefault( lnOpcao )
	ThisForm.mAtivapagina1
	Return
Endif

ThisForm.Pagina.Dados.getDEmps.Value = ''
If InList(ThisForm.pcEscolha, 'ALTERAR', 'CONSULTAR', 'EXCLUIR')
	ThisForm.poDataMgr.CursorQuery('SigCdEmp', 'crSigCdEmp', 'Cemps', crSigCdSer.Emps)
	ThisForm.Pagina.Dados.getDEmps.Value = crSigCdEmp.Razas
EndIf

do case
	case thisform.pcEscolha='INSERIR'
		thisform.Pagina.Dados.getemps.setfocus
	case thisform.pcEscolha='PROCURAR'
		thisform.Pagina.Dados.getcods.setfocus
	case thisform.pcEscolha='ALTERAR'
		thisform.pagina.dados.getdescs.setfocus
	case thisform.pcEscolha='CONSULTAR' or ;
		thisform.pcEscolha='EXCLUIR'
		thisform.pagina.dados.grupo_salva.salva.setfocus
endcase

Select CrSigCdSer

ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
local lnRecno,llRetorno

m.llRetorno = .t.

if inlist(thisform.Pagina.Lista.Grupo_op.value,1,3)
   with thisform.Pagina.Dados
      if empty(.getcods.value)
         = messagebox('Série inválida.',0+64,goSistema.NomeSistema)
         .getcods.setfocus
		Return
      else
      
         if thisform.Pagina.Lista.Grupo_op.value=1 

			lcKey = ThisForm.Pagina.Dados.GetEmps.Value + ThisForm.Pagina.Dados.GetCods.Value

			if ThisForm.poDataMgr.ChkRegister('SigCdSer','EmpCods',lcKey)
                 = messagebox('Série já cadastrada.',0+64,goSistema.NomeSistema)
		        .getcods.setfocus
				Return
		  	endif
		
	    endif	
	 EndIf
	 If Empty(This.Parent.Parent.gettiponf.Value)
         = messagebox('Tipo Nota Fiscal.',0+64,goSistema.NomeSistema)
         This.Parent.Parent.gettiponf.setfocus
         Return
	EndIf 		
  endwith

	Replace EmpCods With Emps + Cods In crSigCdSer
ENDIF

IF inlist(thisform.pcEscolha,'EXCLUIR')
	IF ThisForm.poDataMgr.ChkRegister('SigMvNfi','series',ThisForm.Pagina.Dados.GetCods.Value)
		MESSAGEBOX('Série não pode ser excluida pois está sendo usada em Notas Fiscais!',0+64,goSistema.NomeSistema)
		RETURN .f.
	ENDIF
ENDIF


Dodefault()
thisform.mAtivapagina1

ENDPROC
