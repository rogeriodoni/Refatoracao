* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
lcQryCat = [Select * From SigCdCat ]
With ThisForm
	.poDataMgr = CreateObject('fSqlConector', ThisForm.Name)

	If (.poDataMgr.pnIdconn > 0)
		With .poDataMgr
			.AddCursor('SigCdCat', 'cIdChaves', 'crSigCdCat', '', ThisForm.Pagina.Lista.Grade, lcQryCat)
			.pfSqlTabela(1).pColuna('Cods', '', '', 'Códigos', 52, .t.)
			.pfSqlTabela(1).pColuna('descs', '', '', 'Descrição', 115, .t.)
			.pfSqlTabela(1).pColuna('gdegrps', '', '', 'Gde. Grp', 50, .t.)
			.pfSqlTabela(1).pColuna('prazos', '', '', 'Prazos', 30, .t.)
		EndWith
		Select crSigCdCat
		Index On cods tag cods
		Index On descs tag descs
		Set Order To cods
		=DoDefault()

		ThisForm.Pagina.Lista.Grade.Column1.SetFocus
	Else
		=Messagebox('Impossivel Efetuar Conexão com Servidor de Banco de Dados...', 0+48, 'CreateObject')
		Return .f.
	EndIf
EndWith
* Fim **************************************************************************************

ENDPROC

PROCEDURE msv_procurar
Set Near On
IF !IsEmpty(thisform.pagina.dados.get_Cods.value)
   SET ORDER TO TAG "Cods" IN crSigCdCat
   Seek thisform.pagina.dados.get_Cods.value
ENDIF

IF !IsEmpty(thisform.pagina.dados.get_Descs.value)
   SET ORDER TO TAG "Descs" IN crSigCdCat
   Seek thisform.pagina.dados.get_Descs.value
ENDIF
Set Near Off

ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Lista.Grupo_op.Click
------------------------------------------------------------
PROCEDURE Pagina.Lista.Grupo_op.Click
LPARAMETERS opcao

If !dodefault( Opcao )
	ThisForm.mAtivapagina1
	Return
Endif

Select crSigCdCat

* posiciona o cursor de acordo com a opção
Do Case
	Case Inlist( ThisForm.pcEscolha, 'PROCURAR', 'INSERIR' )
		thisform.pagina.dados.get_Cods.setfocus	

	Case ThisForm.pcEscolha = 'ALTERAR'
		thisform.pagina.dados.get_Descs.setfocus	

	Case Inlist( ThisForm.pcEscolha, 'CONSULTAR', 'EXCLUIR' )
		thisform.pagina.dados.grupo_salva.salva.setfocus
Endcase

Select crSigCdCat
ThisForm.Pagina.Dados.Refresh

ENDPROC

PROCEDURE Valid
if not IsEmpty(this.value)
	loLista = CreateObject('fwBuscaExt', ThisForm.poDataMgr.pnIdConn, 'SigCdGpr', ;
						   'crListaRemota', 'Codigos', This.Value, 'Grandes Grupos de Produtos', .T.)
	If Not loLista.plAchouRegistro
		loLista.mAddColuna('Codigos', '', 'Código')
		loLista.mAddColuna('Descs', '', 'Descrição')
		loLista.Show()
	EndIf
	If Lastkey() = 27
		This.Value = ''
	Else
		This.Value = crListaRemota.Codigos
	Endif
	Use In crListaRemota
	this.refresh
EndIf
ENDPROC
