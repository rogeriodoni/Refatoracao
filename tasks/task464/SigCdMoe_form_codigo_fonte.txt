------------------------------------------------------------
* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
Select CrSigCdMoe
If ThisForm.pcEscolha = 'INSERIR'
	If Empty(CrSigCdMoe.cmoes )
		Messagebox('Moeda Inválida !!!',48,'')
		ThisForm.Pagina.Dados.Get_Moeda.SetFocus
		Return(.F.)
	EndIf
	If ThisForm.poDataMgr.CursorQuery('SigCdMoe',,'cmoes',CrSigCdMoe.cmoes)
		Messagebox('Moeda já cadastrada !!!',48,'')
		ThisForm.Pagina.Dados.Get_Moeda.SetFocus
		Return(.F.)
	EndIf
	If !BetWeen(This.Parent.Parent.SpnNordRels.Value,0,5)
		Messagebox('Ordem de Impressão nos Relatórios Inválido !!!',48,'')
		This.Parent.Parent.SpnNordRels.SetFocus
		Return(.F.)
	EndIf
EndIf

If ThisForm.pcEscolha = 'EXCLUIR'
	lcMoe = CrSigCdMoe.cmoes
	ThisForm.poDataMgr.Requery('CrSigCdCot')
	Select CrSigCdCot
	Go Top
	If !Eof()
		MessageBox('Impossível excluir moeda com cotação cadastrada !!!',16,'Exclusão Inválida')
		Return(.F.)
	EndIf
EndIf

Select CrSigCdMoe
If InList(ThisForm.pcEscolha,'INSERIR','ALTERAR') And Empty(CrSigCdMoe.disfpres)
	Replace CrSigCdMoe.disfpres With 'S'
EndIf

=DoDefault()
ThisForm.mAtivaPagina1
ENDPROC
