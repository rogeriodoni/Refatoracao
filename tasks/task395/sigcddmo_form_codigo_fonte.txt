------------------------------------------------------------
* PROCEDURE msv_inserir
------------------------------------------------------------
PROCEDURE msv_inserir
Private pTru, pDay, pBai

pTru = .t.
pDay = Datetime()
pBai = fDtoSQL(ThisForm.Pagina.Dados.Dt_data.Value)

Create Cursor TmpLanca (CMats c(14), Qtds n(12,3), CGrus c(3))
Index On Cmats Tag Cmats

Create Cursor TmpMat (Cmats c(14), Qtdes n(12,3), Varia n(1), Fators n(12,4), CGrus c(3))
Index On Cmats Tag CMats

Create Cursor crCompo (CPros c(14), CMats c(14), Qtds n(9, 3), CUnis c(3), CGrus c(3), Varias n(1))
Index On CPros + CMats Tag CPros

Select CPros, Space(3) as CUnis, Space(3) as CGrus, 0 as Varias ;
  From crLclMvItn ;
 Group By CPros ;
  Into Cursor crProdutos ReadWrite
Select crProdutos
Index On CPros Tag CPros

Select crProdutos
Scan
	ThisForm.poDataMgr.CursorQuery('SigCdPro', 'crPro', 'CPros', crProdutos.CPros, 'CUnis, CGrus, Varias, CproEqs')
	Replace CUnis  With crPro.CUnis, ;
			CGrus  With crPro.CGrus, ;
			Varias With crPro.Varias In crProdutos

	ThisForm.poDataMgr.CursorQuery('SigPrCpo', 'crTCompo', 'CPros', crProdutos.CPros, 'Mats, Qtds')
	Select crTCompo
	If Reccount() = 0
		ThisForm.poDataMgr.CursorQuery('SigPrCpo', 'crTCompo', 'CPros', crPro.CproEqs, 'Mats, Qtds')
		Select CrTCompo
	Endif
	Scan
		ThisForm.poDataMgr.CursorQuery('SigCdPro', 'crPro', 'CPros', crTCompo.Mats, 'CUnis, CGrus, Varias')
		Insert Into crCompo (CPros, CMats, Qtds, CUnis, CGrus, Varias) ;
					 Values (crProdutos.CPros, crTCompo.Mats, crTCompo.Qtds, ;
					 		 crPro.CUnis, crPro.CGrus, crPro.Varias)
	EndScan
EndScan

Delete From crLclMvItn Where Empty(Cpros) Or Empty(Qtds)

Select crLclMvItn
oProg = CreateObject('fwprogressbar', 'Processando Ítens Selecionados...', Reccount() * 2)
oProg.Show
Scan
	oProg.Update(.t.)

	Zap In TmpMat

	Select crCompo
	Seek crLclMvItn.CPros
	Scan While crCompo.CPros = crLclMvItn.CPros
		=Seek(crCompo.CUnis, 'TempUni', 'CUnis')
		=Seek(crCompo.CGrus, 'TempGru', 'CGrus')

		If (TempGru.CEstoqs <> 2 And CrCompo.Qtds <> 0 )

			If Not Seek(crCompo.CMats, 'TmpMat', 'CMats')
				Insert Into TmpMat (CMats, Varia, Fators, CGrus) ;
							Values (crCompo.CMats, crCompo.Varias, TempUni.Fators, crCompo.CGrus)
			EndIf
			Replace Qtdes With TmpMat.Qtdes + Round(crLclMvItn.Qtds * crCompo.Qtds, 2) In TmpMat
		EndIf
	EndScan
	
	lnPeso = (crLclMvItn.Pesos * crLclMvItn.Qtds)
	
	* Apanha as Composições Em Que o Peso NÃO Varia (Ex.: Brilhante e Rubi)
	Select TmpMat
	Scan For (Varia <> 1)
		lnPeso = lnPeso - (TmpMat.Qtdes * TmpMat.Fators)

		If Not Seek(TmpMat.CMats, 'TmpLanca', 'CMats')
			Insert Into TmpLanca (CMats, Qtds, CGrus) Values (TmpMat.CMats, TmpMat.Qtdes, TmpMat.CGrus)
		Else
			Replace Qtds With TmpLanca.Qtds + TmpMat.Qtdes In TmpLanca
		EndIf
	EndScan
	
	Store 0 to lnPesoRat, lnRecnoRat

	* Apanha as Composições Em Que o Peso Varia (Ex.: Ouro e Prata)
	Select TmpMat
	Sum Qtdes To lnPesoTot For (Varia = 1)
	Scan For (Varia = 1)
		lnPesoLanc = Iif(lnPesoTot = 0,0,Round(Qtdes / lnPesoTot * lnPeso, 2))
		lnPesoRat  = lnPesoRat + lnPesoLanc

		If Not Seek(TmpMat.CMats, 'TmpLanca', 'CMats')
			Insert Into TmpLanca (CMats, Qtds, CGrus) Values (TmpMat.CMats, lnPesoLanc, TmpMat.CGrus)
		Else
			Replace Qtds With TmpLanca.Qtds + lnPesoLanc In TmpLanca
		EndIf
		Select TmpLanca
		If Empty(lnRecnoRat)
			lnRecnoRat = Recno()
		EndIf
	EndScan
	
	If (lnPeso <> lnPesoRat) And Not Empty(lnRecnoRat)
		Go lnRecnoRat In TmpLanca
		Replace Qtds With TmpLanca.Qtds + (lnPeso - lnPesoRat) In TmpLanca
	EndIf
EndScan

Select crLclMvItn
Scan
	oProg.Update(.t.)

	=Seek(crLclMvItn.CPros, 'crProdutos', 'CPros')
	=Seek(crProdutos.CGrus, 'TempGru', 'CGrus')

	Insert Into crSigMvItn (Aqtds, Cpros, Dpros, Cunis, Datatrans, Dopes, Emps, NrCons, Numes, Qtds, Totas, ;
							Units, NumOlds, Notas, Opers, Moedas, Pesos, Fators, MoeFats, FatVals, MoeVals, ;
							DescVals, UniVals, Iconfs, cItens, UnitEmbs, TPesos, Obs, CodBarras, ValIpis, ;
							TpIpis, Aliqs, Sitribs, EmpDopNums, DtAlts, cIdChaves) ;
					values (crLclMvItn.Qtds, crLclMvItn.Cpros, crLclMvItn.Dpros, crLclMvItn.Cunis, crLclMvItn.Datatrans, ;
							crSigMvCab.Dopes, _Empr, crLclMvItn.NrCons, crSigMvCab.Numes, crLclMvItn.Qtds, crLclMvItn.Totas, ;
							crLclMvItn.Units, crLclMvItn.NumOlds, crLclMvItn.Notas, crLclMvItn.Opers, crLclMvItn.Moedas, ;
							crLclMvItn.Pesos, crLclMvItn.Fators, crLclMvItn.Moefats, crLclMvItn.FatVals, crLclMvItn.MoeVals, ;
							crLclMvItn.DescVals, crLclMvItn.UniVals, crLclMvItn.Iconfs, crLclMvItn.cItens, crLclMvItn.UnitEmbs, ;
							crLclMvItn.TPesos, crLclMvItn.Obs, crLclMvItn.CodBarras, crLclMvItn.ValIpis, crLclMvItn.TpIpis, ;
							crLclMvItn.Aliqs, crLclMvItn.Sitribs, _Empr + crSigMvCab.Dopes + Str(crSigMvCab.Numes,6), ;
							Datetime(), fUniqueIds())

	pCidC = Dtos(crSigMvCab.Datas)+'S'+Transform(fGerUniqueKey(Dtos(crSigMvCab.Datas)),"@L 999999")+ThisForm.Sigkey 
	pSeq  = fGerUniqueKey('HISTBAR')
	Insert Into crSigMvHst (Usuars, Datars, Emps, Opers, Dopes, Numes, Datas, DtAudits, Cpros, Moedas, Totas, Datatrans, ;
						  EmpOs, Qtds, Grupos, Estos, cIdChaves, EmpDopNums, EmpGruEsts, DtAlts, OriDopNums, Seqs,;
						  Units, Pesos) ;
				  values (Usuar, Datetime(), _Empr, 'S', ThisForm.Dopes, crSigMvItn.Numes, crSigMvCab.Datas, Null, ;
				  		  crSigMvItn.Cpros, TempGru.MoeCusts, crSigMvItn.Totas, crSigMvItn.Datatrans, _Empr, ;
				  		  crSigMvItn.Qtds, crSigMvCab.GrupoOs, crSigMvCab.ContaOs, pCidC, ;
				  		  _Empr + ThisForm.Dopes + Str(crSigMvItn.Numes, 6), ;
				  		  _Empr + crSigMvCab.GrupoOs + crSigMvCab.ContaOs, Datetime(), ;
				  		  _Empr + ThisForm.Dopes + Str(crSigMvItn.Numes, 6), pSeq, ;
				  		  crLclMvItn.Units, crLclMvItn.Pesos)

	=fRecalculaP(crSigMvHst.Emps, crSigMvHst.Grupos, crSigMvHst.Estos, crSigMvHst.CPros, crSigMvHst.Datas, crSigMvHst.CodCors, crSigMvHst.CodTams, ThisForm.poDataMgr)
	=fRecalculaC(crSigMvHst.Emps, crSigMvHst.CPros, crSigMvHst.Datas, ThisForm.Podatamgr )
EndScan

lcIds = Dtos(CrSigMvCab.Datas) + Transform(fGerUniqueKey(Dtos(CrSigMvCab.Datas)), '@L 999999')+ThisForm.Sigkey 

Insert Into crSigCdNec (Emps, Dopps, Numps, Usuars, Datas, Datars, Grupoos, Contaos, ;
					   Grupods, Contads, cIdChaves, EmpDNps) ;
			   Values (_Empr, crSigMvCab.Dopes, crSigMvCab.Numes, Usuar, crSigMvCab.Datas, ;
			   		   Datetime(), crSigMvCab.GrupoOs, crSigMvCab.ContaOs, crSigMvCab.GrupoDs, ;
			   		   crSigMvCab.ContaDs, lcIds, _Empr + crSigMvCab.Dopes + Str(crSigMvCab.Numes, 10))
Delete From TmpLanca Where Empty(CMats) Or Empty(Qtds)
oProg.Complete

lnCnt = Reccount('TmpLanca') + Reccount('Tmpoperacao')
Select TmpLanca
oProg = CreateObject('fwprogressbar', 'Processando Históricos e Subníveis...', lnCnt)
oProg.Show
Scan
	oProg.Update(.t.)

	Insert Into crSigCdNei (Emps, Dopps, Numps, CMats, Qtds, cIdChaves, EmpDnPs) ;
					Values (crSigMvCab.Emps, crSigMvCab.Dopes, crSigMvCab.Numes, ;
							TmpLanca.Cmats, TmpLanca.Qtds, fUniqueIds(), ;
							crSigMvCab.Emps + crSigMvCab.Dopes + Str(crSigMvCab.Numes, 10))

	=Seek(TmpLanca.CGrus, 'TempGru', 'CGrus')

	If (TempGru.CEstoqs <> 2)
		pCidC = Dtos(crSigMvCab.Datas)+'E'+Transform(fGerUniqueKey(Dtos(crSigMvCab.Datas)),"@L 999999")+Thisform.Sigkey 
		pSeq  = fGerUniqueKey('HISTBAR')
		Insert Into crSigMvHst (Usuars, Datars, Emps, Opers, Dopes, Numes, Datas, DtAudits, CPros, Empos, Qtds, ;
							  Grupos, Estos, CidChaves, EmpDopNums, EmpGruEsts, DtAlts, OriDopNums, Seqs) ;
					  Values (Usuar, Datetime(), _Empr, 'E', crSigMvCab.Dopes, crSigMvCab.Numes, crSigMvCab.Datas, Null, ;
					  		  TmpLanca.CMats, _Empr, TmpLanca.Qtds, crSigMvCab.Grupods, crSigMvCab.Contads, pCidC, ;
					  		  _Empr + crSigMvCab.Dopes + Str(crSigMvCab.Numes, 6), _Empr + crSigMvCab.Grupods + crSigMvCab.Contads, ;
					  		  Datetime(), _Empr + crSigMvCab.Dopes + Str(crSigMvCab.Numes, 6), pSeq)
		=fRecalculaP(crSigMvHst.Emps, crSigMvHst.Grupos, crSigMvHst.Estos, crSigMvHst.CPros, crSigMvHst.Datas, crSigMvHst.CodCors, crSigMvHst.CodTams, ThisForm.poDataMgr)
		=fRecalculaC(crSigMvHst.Emps, crSigMvHst.CPros, crSigMvHst.Datas, Thisform.Podatamgr )
	EndIf
EndScan

Select Tmpoperacao
Scan For Not Empty(Codigos)
	oProg.Update(.t.)
	Insert Into crSigMvPec (Emps, Dopes, Numes, Codigos, cIdChaves, EmpDopNums, EmpSubns, Datas) ;
					Values (crSigMvCab.Emps, crSigMvCab.Dopes, crSigMvCab.Numes, TmpOperacao.Codigos, ;
							fUniqueIds(), crSigMvCab.Emps + crSigMvCab.Dopes + Str(crSigMvCab.Numes, 6), CrSigMvCab.Emps, Datetime())

	lnNuc = Int(Val(Left(Str(TmpOperacao.Codigos, 10), 4)))
	ThisForm.poDataMgr.CursorQuery('SigCdOpe', 'crOpe', 'NDopes', lnNuc, 'Dopes')

	lcEdn = crSigMvCab.Emps + crOpe.Dopes + Str(Int(Val(Right(Str(TmpOperacao.Codigos, 10), 6))), 6)

	lcQuery = [Update SigMvCab ] + ;
				 [Set ChkSubn = ?pTru, ] + ;
				 	 [DtAlts = ?pDay, ] + ;
				 	 [DtBaixas = ?pBai ] + ;
			   [Where EmpDopNums = '] + lcEdn + [']

	If (ThisForm.poDataMgr.SqlExecute(lcQuery, '') < 1)
		=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (mSv_Inserir - Update - SigMvCab)')
		Return .f.
	EndIf
EndScan
oProg.Complete

Wait Window 'Aguarde!!! Gravando Arquivos (01/06)...' NoWait
If Not ThisForm.poDataMgr.Update('crSigMvItn')
	=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (Update - crSigMvItn)')
	Return .f.
EndIf

Wait Window 'Aguarde!!! Gravando Arquivos (02/06)...' NoWait
If Not ThisForm.poDataMgr.Update('crSigMvHst')
	=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (Update - crSigMvHst)')
	Return .f.
EndIf

Wait Window 'Aguarde!!! Gravando Arquivos (03/06)...' NoWait
If Not ThisForm.poDataMgr.Update('crSigMvPec')
	=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (Update - crSigMvPec)')
	Return .f.
EndIf

Wait Window 'Aguarde!!! Gravando Arquivos (04/06)...' NoWait
If Not ThisForm.poDataMgr.Update('crSigCdNec')
	=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (Update - crSigCdNec)')
	Return .f.
EndIf

Wait Window 'Aguarde!!! Gravando Arquivos (05/06)...' NoWait
If Not ThisForm.poDataMgr.Update('crSigCdNei')
	=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (Update - crSigCdNei)')
	Return .f.
EndIf

Wait Window 'Aguarde!!! Gravando Arquivos (06/06)...' NoWait
If Not ThisForm.poDataMgr.Update('crSigMvCab')
	=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (Update - crSigMvCab)')
	Return .f.
EndIf

NoDefault

Wait Clear

Return .t.
* Fim *********************************************************************************

ENDPROC

* PROCEDURE msv_excluir
------------------------------------------------------------
PROCEDURE msv_excluir
Private pNot, pDay, pEdn, pNul

pNot = .f.
pDay = Datetime()
pNul = Null

Select crSigMvCab
lcChaveDel  = Emps + Dopes + Str(Numes, 06)
lcChaveDel2 = Emps + Dopes + Str(Numes, 10)

lcQuery = [Delete From SigMvCab ] + ;
		   [Where EmpDopNums = '] + lcChaveDel + [']

If (ThisForm.poDataMgr.SqlExecute(lcQuery, '') < 1)
	=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (Delete - SigMvCab)')
	Return .f.
EndIf

lcQuery = [Delete From SigMvItn ] + ;
		   [Where EmpDopNums = '] + lcChaveDel + [']

If (ThisForm.poDataMgr.SqlExecute(lcQuery, '') < 1)
	=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (Delete - SigMvItn)')
	Return .f.
EndIf

pEdn = lcChaveDel
ThisForm.poDataMgr.Requery('crSigMvPec')

Select crSigMvPec
Scan
	nDope = Val(Left(Str(Codigos, 10), 4))
	ThisForm.poDataMgr.CursorQuery('SigCdOpe', 'crOpe', 'NDopes', NDope, 'Dopes')
	Dope = crOpe.Dopes
	Nume = Val(Right(Str(Codigos,10), 6))
	Set Decimals To 2

	lcQuery = [Update SigMvCab ] + ;
				 [Set ChkSubn = ?pNot, ] + ;
				 	 [DtAlts = ?pDay, ] + ;
				 	 [DtBaixas = ?pNul ] + ;
			   [Where EmpDopNums = '] + _Empr + Dope + Str(Nume, 6) + [']

	If (ThisForm.poDataMgr.SqlExecute(lcQuery, '') < 1)
		=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (mSv_Excluir - Update - SigMvCab)')
		Return .f.
	EndIf

	lcQuery = [Delete From SigMvPec ] + ;
			   [Where EmpDopNums = '] + crSigMvPec.Emps + crSigMvPec.Dopes + Str(crSigMvPec.Numes, 6) + [']

	If (ThisForm.poDataMgr.SqlExecute(lcQuery, '') < 1)
		=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (Delete - SigMvCab)')
		Return .f.
	EndIf
EndScan
Zap In crSigMvPec

lcQuery = [Delete From SigCdNec ] + ;
		   [Where EmpDNps = '] + lcChaveDel2 + [']

If (ThisForm.poDataMgr.SqlExecute(lcQuery, '') < 1)
	=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (Delete - SigCdNec)')
	Return .f.
EndIf

lcQuery = [Delete From SigCdNei ] + ;
		   [Where EmpDNps = '] + lcChaveDel2 + [']

If (ThisForm.poDataMgr.SqlExecute(lcQuery, '') < 1)
	=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (Delete - SigCdNei)')
	Return .f.
EndIf

lcCpo = 'Emps, Grupos, Estos, CPros, Datas, CodCors, CodTams, cIdChaves'
ThisForm.poDataMgr.CursorQuery('SigMvHst', 'crHis', 'EmpDopNums', lcChaveDel, lcCpo)
Select crHis
Scan
	lcQuery = [Delete From SigMvHst Where cIdChaves = '] + crHis.cIdChaves + [']

	If (ThisForm.poDataMgr.SqlExecute(lcQuery, '') < 1)
		=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (Delete - SigMvHst)')
		Return .f.
	EndIf

			.AddCursor('SigCdNei', 'cIdChaves', 'crSigCdNei', '', '', lcQryNensI)

		ThisForm.SqlDados

		* Criando um Cursor Vazio Igual ao SigMvItn
		Select * From crSigMvItn Into Cursor crLclMvItn ReadWrite
		
		Select crLclMvItn

			.RecordSource           = 'crLclMvItn'

		Select TempUni
		Index On CUnis Tag CUnis
		
		ThisForm.poDataMgr.CursorQuery('SigCdGrp', 'TempGru', .f., .f., 'CGrus, CEstoqs, MoeCusts')

		If Eof('crSigCdOpe')

		ThisForm.Refresh
		ThisForm.Pagina.Lista.Grade.Column1.SetFocus
	Else
		=Messagebox('Impossivel Efetuar Conexão com Servidor de Banco de Dados...', 0+48, 'CreateObject')

* OBJETO: SIGCDDMO.Pagina.Dados.GradeOperação.Operação.Text1
* Classe: textbox (Base: textbox)

================================================================================

------------------------------------------------------------
* PROCEDURE When
------------------------------------------------------------
PROCEDURE When
ThisForm.OldCodigo = This.Value
Return (ThisForm.pcEscolha = 'INSERIR')
* Fim *********************************************************************************

ENDPROC

------------------------------------------------------------
* PROCEDURE Valid
------------------------------------------------------------
PROCEDURE Valid
Private pSim, pDay, pBai
Local ndope, nume, codconf, opeatu

pBai = fDtoSQL(ThisForm.Pagina.Dados.Dt_data.Value)

lcPosicao = crSigMvCab.Emps + crSigMvCab.Dopes + Str(crSigMvCab.Numes, 6)
lnPuxaOrigem  = crSigCdOpe.ConOSubs
lnPuxaDestino = crSigCdOpe.ConDSubs

If (This.Value <> 0) And (This.Value <> ThisForm.OldCodigo)
	If IndexSeek(This.Value, .f., 'TmpOperacao', 'Codigos')
		=Messagebox('Movimentação Já Digitada!!!', 0+48, '')
		This.Value = ThisForm.OldCodigo
		Return 0
	EndIf

	* Reseta o Nivel 
	Set Decimals To 0
	nDope = Val(Left(Str(This.Value, 10), 4))
	ThisForm.poDataMgr.CursorQuery('SigCdOpe', 'TmpOpe', 'NDopes', nDope, 'Dopes')
	Select TmpOpe
	Dope = TmpOpe.Dopes
	Nume = Val(Right(Str(This.Value, 10), 6))
	Set Decimals To 2

	Select crSigMvCab
Else
	If Empty(This.Value) And Not Empty(ThisForm.OldCodigo)
		Set Decimals To 0
		OldNDope = Val(Left(Str(ThisForm.OldCodigo, 10), 4))
		Select crSigCdOpe
		ThisForm.poDataMgr.CursorQuery('SigCdOpe', 'TmpOpe', 'NDopes', OldNDope, 'Dopes')
		OldDope = TmpOpe.Dopes
		ThisForm.poDataMgr.CursorQuery('SigCdOpe', 'TmpOpe', 'Dopes', OldDope)
		OldNume = Val(Right(Str(ThisForm.OldCodigo, 10), 6))
		Set Decimals To 2

		Select crLclMvItn
		Delete All For Dopes + Str(Numes, 6) = OldDope + Str(OldNume, 6)

		pSim = .t.
		pDay = Datetime()

		lcQuery = [Update SigMvCab ] + ;
					 [Set ChkSubn = ?pSim, ] + ;
					 	 [DtAlts = ?pDay, ] + ;
					 	 [DtBaixas = ?pBai ] + ;
				   [Where EmpDopNums = '] + _Empr + OldDope + Str(OldNume, 6) + [']

		If (ThisForm.poDataMgr.SqlExecute(lcQuery, '') < 1)
			=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (Update - 1)')
			Return 0
		EndIf

		Select TmpOperacao
		Skip
		If Eof()
		 	Insert Into TmpOperacao (Emps, Dopes, Numes, Codigos) ;
		 			 		 Values (_Empr, crSigMvCab.Dopes, crSigMvCab.Numes, 0)
		EndIf
	EndIf
	Return .T.
EndIf

lcNivel = crSigCdOpe.Nivels

Set Decimals To 0
nDope = Val(Left(Str(This.Value, 10), 4))
ThisForm.poDataMgr.CursorQuery('SigCdOpe', 'TmpOpe', 'NDopes', nDope)
Select TmpOpe
Dope = TmpOpe.Dopes
Nume = Val(Right(Str(This.Value, 10), 6))
Set Decimals To 2
CodConf = This.Value
Ok = .t.

If Not Empty(This.Value) And (This.Value <> ThisForm.OldCodigo)
	If (lcNivel <> Dope)
		=Messagebox('SubNivel Inválido!!!', 0+48, '')
		This.Value = 0 
		Ok = .f.
		Return 0
	EndIf
	
	Select crSigMvCab

	lcEdn = _Empr + Dope + Str(Nume, 6)
	ThisForm.poDataMgr.CursorQuery('SigMvCab', 'LocalEest', 'EmpDopNums', lcEdn)

	If Eof('LocalEest')
		=Messagebox('Movimentação Não Localizada!!!', 0+48, '')
		This.Value = 0 
		Ok = .f.
		Return 0
	EndIf

	If LocalEest.ChkSubn
		=Messagebox('Movimentação Já Utilizada Em Outra Operação!!!', 0+48, '')
		This.Value = 0 
		Ok = .f.
		Return 0
	EndIf

	If Not Empty(crSigMvCab.ContaDs) And (LocalEest.ContaDs <> crSigMvCab.ContaDs)
		=Messagebox('Movimentação De Outra Conta !!!', 0+48, '')
		This.Value = 0 
		Ok = .f.
		Return 0
	EndIf

	If Not Empty(crSigMvCab.ContaOs) And (LocalEest.ContaOs <> crSigMvCab.ContaOs)
		=Messagebox('Movimentação de Outra Conta!!!', 0+48, '')
		This.Value = 0 
		Ok = .f.
		Return 0
	EndIf
	
	Select LocalEest
	If Ok	
		pSim = .t.
		pDay = Datetime()

		lcQuery = [Update SigMvCab ] + ;
					 [Set ChkSubn = ?pSim, ] + ;
					 	 [DtAlts = ?pDay, ] + ;
					 	 [DtBaixas = ?pBai ] + ;
				   [Where EmpDopNums = '] + lcEdn + [']

		If (ThisForm.poDataMgr.SqlExecute(lcQuery, '') < 1)
			=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (Update - 2)')
			Return 0
		EndIf

		If Not Empty(ThisForm.OldCodigo)
			Set Decimals To 0
			OldNDope = Val(Left(Str(ThisForm.OldCodigo, 10), 4))
			Select crSigCdOpe
			ThisForm.poDataMgr.CursorQuery('SigCdOpe', 'TmpOpe', 'NDopes', OldNDope, 'Dopes')
			OldDope = TmpOpe.Dopes
			ThisForm.poDataMgr.CursorQuery('SigCdOpe', 'TmpOpe', 'Dopes', OldDope)
			OldNume = Val(Right(Str(ThisForm.OldCodigo, 10), 6))
			Set Decimals To 2

			Select crLclMvItn
			Delete All For Dopes + Str(Numes, 6) = OldDope + Str(OldNume, 6)
		EndIf

		Select crLclMvItn
		Delete All For Empty(Cpros)
		Go Bottom

		If InList(lnPuxaOrigem,1,2)
			If Empty( crSigMvCab.Grupoos ) Or ( Empty( crSigMvCab.Contaos ) And !crSigCdOpe.FixGOrigs )	
				ThisForm.Pagina.Dados.Origem.Get_Grupo.Value = Iif( lnPuxaOrigem=1, LocalEest.Grupoos, LocalEest.Grupods )
			Endif
			If Empty( crSigMvCab.Contaos ) 
				ThisForm.Pagina.Dados.Origem.Get_Conta.Value = Iif( lnPuxaOrigem=1, LocalEest.Contaos, LocalEest.Contads )
			Endif
		Endif
		
		If InList(lnPuxaDestino,1,2)
			If Empty( crSigMvCab.Grupods ) Or ( Empty( crSigMvCab.Contads ) And !crSigCdOpe.FixGDests )
				ThisForm.Pagina.Dados.Destino.Get_Grupo.Value = Iif( lnPuxaDestino=1, LocalEest.Grupoos, LocalEest.Grupods )
			Endif
			If Empty( crSigMvCab.Contads ) 
				ThisForm.Pagina.Dados.Destino.Get_Conta.Value = Iif( lnPuxaDestino=1, LocalEest.Contaos, LocalEest.Contads )
			Endif
		Endif

		If Empty( crSigMvCab.GrVends )
			ThisForm.Pagina.Dados.responsavel.Get_Grupo.Value = LocalEest.GrVends
		Endif
		If Empty( crSigMvCab.Vends ) And crSigMvCab.GrVends = LocalEest.GrVends
			ThisForm.Pagina.Dados.responsavel.Get_Conta.Value = LocalEest.Vends
		Endif
		
		If Empty( crSigMvCab.GrResps )
			Replace crSigMvCab.GrResps With LocalEest.GrResps
		Endif
		If Empty( crSigMvCab.Resps ) And crSigMvCab.GrResps = LocalEest.GrResps
			Replace crSigMvCab.Resps With LocalEest.Resps
		Endif
				
		ThisForm.Pagina.Dados.Refresh

		ThisForm.poDataMgr.CursorQuery('SigMvItn', 'LocalEestI', 'EmpDopNums', lcEdn)
		Select LocalEestI
		Scan
			If Empty(CPros)
				Loop
			EndIf
			Scatter Memvar Memo
			If (m.Qtds <> 0)
				ThisForm.CItem = ThisForm.CItem + 1
				m.CItens = ThisForm.CItem
				m.Pesoms = Round(m.Pesos / m.Qtds, 2)
				Insert Into crLclMvItn From Memvar
				Replace Obs with LocalEestI.Obs In crLclMvItn
			EndIf
		EndScan
		
		Select crLclMvItn
		lnItem = 0
		Scan
			lnItem = lnItem + 1
			Replace CItens 	   With lnItem, ;
					EmpDopNums With crLclMvItn.Emps + crLclMvItn.Dopes + Str(crLclMvItn.Numes, 6), ;
					cIdChaves  With Iif(Empty(crLclMvItn.cIdChaves), fUniqueIds(), crLclMvItn.cIdChaves), ;
					DtAlts     With Datetime()
		EndScan
		Go Top

		If Eof()
		 	Insert Into crLclMvItn (Emps, Dopes, Numes, CItens, EmpDopNums, cIdChaves, DtAlts) ;
	 			 			Values (crSigMvCab.Emps, crSigMvCab.Dopes, crSigMvCab.Numes, 1, ;
	 			 					crSigMvCab.Emps + crSigMvCab.Dopes + Str(crSigMvCab.Numes, 6), ;
	 			 					fUniqueIds(), Datetime())
		EndIf
		Select crSigMvCab

		With ThisForm.Pagina.Dados
			ThisForm.poDataMgr.CursorQuery('SigCdCli', 'crSigCdCli', 'IClis', crSigMvCab.Contaos, 'RClis')
			If Not Eof('crSigCdCli')
				.origem.get_dconta.value = crSigCdCli.Rclis
			Else
				.origem.get_dconta.Value = ''
			EndIf
			
			ThisForm.poDataMgr.CursorQuery('SigCdCli', 'crSigCdCli', 'IClis', crSigMvCab.Contads, 'RClis')
			If Not Eof('crSigCdCli')
				.destino.get_dconta.value = crSigCdCli.Rclis
			Else
				.destino.get_dconta.Value = ''
			EndIf

			ThisForm.poDataMgr.CursorQuery('SigCdCli', 'crSigCdCli', 'IClis', crSigMvCab.Vends, 'RClis')
			If Not Eof('crSigCdCli')
				.responsavel.get_dconta.value = crSigCdCli.Rclis
			Else
				.responsavel.get_dconta.Value = ''
			EndIf
		EndWith
		ThisForm.Pagina.Dados.Refresh
		
		Select TmpOperacao
		Skip
		If Eof()
		 	Insert Into TmpOperacao (Emps, Dopes, Numes, Codigos) ;
		 			 		 Values (_Empr, crSigMvCab.Dopes, crSigMvCab.Numes, 0)
			Keyboard "{DNARROW}"
		EndIf

		Select [ ] as Agrupar,Sum(Qtds) as Qtds, Sum(Qtds * Pesos) as Pesos ;
		  From crLclMvItn ;
		  Into Cursor crTotais group by 1

		Go Top In crTotais
		ThisForm.Pagina.Dados.getTQtde.Value = crTotais.Qtds
		ThisForm.Pagina.Dados.getTPeso.Value = crTotais.Pesos

		ThisForm.Pagina.Dados.Refresh
		Return 1
	Else
		Select crSigMvCab
		Return 0
	EndIf
EndIf

Select crSigMvCab
ThisForm.Pagina.Dados.Refresh
* Fim *********************************************************************************

ENDPROC

Local loLista

If Not Empty(This.Value)
	lcQuery = [Select IClis, RClis ] + ;
			    [From SigCdCli ] + ;
			   [Where IClis = '] + This.Value + [' ]

	If (ThisForm.poDataMgr.SqlExecute(lcQuery, 'TempCli') < 1)
		=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (TempCli)')
		Return 0
	EndIf

	Dimension laFiltro(1,2)
	laFiltro(1,1) = 'GerBals'
	laFiltro(1,2) = 1

	loLista = CreateObject('fwBuscaExt', ThisForm.poDataMgr.pnIdConn, 'SigCdCli', ;
						   'crListaRemota', 'IClis', This.Value, 'Seleção', 500, .f.)

	If Not loLista.plAchouRegistro
		loLista.mAddColuna('IClis',  '', 'Código')
		loLista.mAddColuna('RClis',  '', 'Descrição')
		loLista.mAddColuna('Grupos', '', 'Grupo')
		loLista.mAddColuna('Inativas ', .f., .f., .f., .f., .f., .t.)
		loLista.gridProc.SetAll('DynamicForeColor', [Iif(crListaRemota.Inativas == 1, Rgb(255, 0, 0), Rgb(0,0,0))], 'Column')
		loLista.Show()
	EndIf
	
	This.Parent.Get_Conta.Value  = crListaRemota.IClis
	This.Parent.Get_dConta.Value = crListaRemota.RClis

	If Not fAcessoContas(Usuar, This.Parent.Get_Grupo.Value, 'C', This.Value)
		=Messagebox('Acesso Negado!!!', 48, '')
		This.Parent.Get_Conta.Value  = ''
		This.Parent.Get_dConta.Value = ''
	EndIf

	If Empty(This.Parent.get_grupo.Value)
		This.Parent.get_grupo.Value = crListaRemota.Grupos
	EndIf
	This.Parent.get_conta.Refresh
	This.Parent.get_dconta.Refresh
	This.Parent.get_grupo.Refresh
	Use In crListaRemota
Else
	This.Parent.Get_Conta.Value  = ''
	This.Parent.Get_dConta.Value = ''
EndIf
This.Refresh
* Fim **************************************************************************************

ENDPROC
