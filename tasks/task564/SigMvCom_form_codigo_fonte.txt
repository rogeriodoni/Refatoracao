------------------------------------------------------------
* PROCEDURE calculos
------------------------------------------------------------
PROCEDURE calculos
Local lnCal, lnBas, lnRed

lnCal = 0
lnBas = TmpComiss.Bases
If Not Empty(TmpComiss.Impostos)
	lnRed = Round(lnBas * TmpComiss.Impostos / 100, 2)
	lnBas = lnBas - lnRed
EndIf

If Not Empty(TmpComiss.Juros)
	lnRed = Round(lnBas * TmpComiss.Juros / 100, 2)
	lnBas = lnBas - lnRed
EndIf

lnCal = Round(lnBas * TmpComiss.Percs / 100, 2)

Return lnCal
* Fim **************************************************************************************
ENDPROC

PROCEDURE Init
Lparameters pEdicao, poForm

ThisForm.Edicao = pEdicao
ThisForm.Parentform = poForm

With This.Grade
	_Alias = Alltrim( Alias() )

	ThisForm.Areadados = _Alias

	lcSql = [Select * From SigCcCpe Where Dopes = ']+TprMvCab.Dopes+[']
	=Thisform.ParentForm.PodataMgr.Sqlexecute(lcSql,'xCcOpe')
	Select xCcOpe
	Index On Grupos Tag Grupos

	Select ( _Alias )
	Count to lnTot
	If pEdicao
		If Reccount('xCcOpe') > 0
			Select xCcOpe
			Scan
				Select ( _Alias )
				Locate For Grupos = xCcOpe.Grupos
				If Eof()
					Append Blank
					Replace Flags   With .t.,;
							Grupos  With xCcOpe.Grupos,;
							Bases   With TprMvCab.Valos,;
							GruCads With xCcOpe.GruCads,;
							CtFils  With xCcOpe.CtFils
				Endif
			EndScan
		Else
			If lnTot = 0
				Append Blank
				Replace Flags with .t.,;
						Bases With TprMvCab.Valos
			Endif
		Endif
	EndIf
	
	*** Tiago - 04/02/2011	- Incluido descricao da conta para facilitar a procura da mesma
	Select ( _Alias )
	Scan
		lcQuery = [Select rclis from SigCdCli where iclis = ']+ &_Alias..contas +[']
		If Thisform.ParentForm.PodataMgr.Sqlexecute(lcQuery,'xConta') < 1
			=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (CursorQuery - xConta)])
			Return .f.
		EndIf
		replace &_Alias..rclis with xConta.rclis
	EndScan
	*****
	
	Select ( _Alias )
	If pEdicao
		Set Order To
		Go Top
	Endif
	.RecordSource = _Alias
	.Grupo.ControlSource = _Alias + ".Grupos"
	.Grupo.text1.ReadOnly = Not pEdicao

	.Conta.Text1.ReadOnly = Not pEdicao

	.Base.Text1.ReadOnly = Not pEdicao
	.Comissão.ControlSource = _Alias + ".Comissaos"
	.Comissão.Text1.ReadOnly = Not pEdicao
	.Moeda.ControlSource = _Alias + ".Moedas"
	.Moeda.Text1.ReadOnly = Not pEdicao
	.Impostos.ControlSource = _Alias + ".Impostos"
	.Impostos.Text1.ReadOnly = Not pEdicao
	.Juros.ControlSource = _Alias + ".Juros"
	.Juros.Text1.ReadOnly = Not pEdicao
	.Percentual.ControlSource = _Alias + ".Percs"
	.Percentual.Text1.ReadOnly = Not pEdicao
	.descricao.ControlSource = _Alias + ".rclis"
	.descricao.Text1.ReadOnly = Not pEdicao	
	.ReadOnly = Not pEdicao

	ThisForm.Grade.SetAll('DynamicForeColor',[Iif(] +_Alias + [.ComiOrgs <> 0 And ] + _Alias + [.ComiOrgs <> ] + _Alias + [.Comissaos,Rgb(255,0,0),RGB(0,0,0))],'Column')	
EndWith
ThisForm.Cancela.Visible = pEdicao
ThisForm.CmdExcluir.Visible = pEdicao

ThisForm.Cancela.Cancel = pEdicao
This.Refresh

lnValTit = 0

ThisForm.Container1.Text1.Value = 0
ThisForm.Container1.Text2.Value = 0

If ThisForm.Edicao And CrSigCdOpe.EdComis = 3 And Not Empty(CrSigCdOpe.Dopcs) And Not Empty(CrSigCdOpe.Grupos) And Not Empty(CrSigCdOpe.contas)
	Select xPar
	Scan
		lcSql = [Select ValPends From SigOpFp Where Fpags = ']+xPar.Fpags+[']
		=ThisForm.ParentForm.PodataMgr.Sqlexecute(lcSql,'LocalFpag')
		
		lnValTit = lnValTit + ( xPar.Valos * xPar.CotFpgs )
	ENDSCAN
	lnValTit = lnValTit + (crSigmvnfi.vcofins+crSigmvnfi.vcsll+crSigmvnfi.vinss+crSigmvnfi.virf+crSigmvnfi.vpis+crSigmvnfi.vtotiss)	&& mena 03/06/2014 - soma impostos para não dar erro
	ThisForm.Container1.Text1.Value = lnValTit
	ThisForm.Container1.Text2.Value = 0
	ThisForm.Container1.Visible = .t.

EndIf
