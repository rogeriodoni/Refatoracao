	Use in TmpItePen

Dodefault()

------------------------------------------------------------
* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
LParameters pform, datses, pVisual 

Set Datasession To DatSes
Thisform.parentform = pform
pVisual = Iif(Type([pVisual]) = [L],[P],pVisual)

If pVisual = [P]
	thisform.Caption = [Total Previsão]

	Select crSigMvOrp
	Set Key To
	Go top

	Thisform.grdTotPrev.RecordSource = []

	Select Datas, Sum(Valos) as Valos, Sum(VlBaixas) as VlBaixas, Sum(Valos - VlBaixas) as VlPends From crSigMvOrp Group By Datas Into Cursor crTmpTotOrp

	Select crSigMvOrp
	Set Order To Tag Citens
	Set Key To crSigMvOrc.Citens

	Select crTmpTotOrp
	Go Top
	If !Eof('crTmpTotOrp')
		With Thisform.grdTotPrev
			.RecordSource = 'crTmpTotOrp'
			.Column1.ControlSource = 'crTmpTotOrp.Datas'
			.Column2.ControlSource = 'crTmpTotOrp.Valos'
			.Column3.ControlSource = 'crTmpTotOrp.VlBaixas'
			.Column4.ControlSource = 'crTmpTotOrp.VlPends'
		EndWith
	EndIf

	Select Sum(Valos) as Valos, Sum(VlBaixas) as VlBaixas, Sum(Valos - VlBaixas) as VlPends From crSigMvOrp Into Cursor crTmpTotOrp2

	thisform.getTotPrev.Value = Nvl(crTmpTotOrp2.Valos,0)
	thisform.getTotReal.Value = Nvl(crTmpTotOrp2.VlBaixas,0)
	thisform.getTotDif.Value = Nvl(crTmpTotOrp2.VlPends,0)

	Select Citens, Dpros, Sum(tota1s + tota2s + tota3s) as Tota1s ;
		from crSigMvOrc ;
		where Inlist(cBlocos,[SUBTOTAIS],[TAXA],[TOTAL],[IMPOSTO],[TOTALGERAL]) Group By Citens, Dpros into cursor crTmpTotal

	Select crTmpTotal
	Go Top
	If !Eof('crTmpTotal')
		With Thisform.grdTotal
			.RecordSource = 'crTmpTotal'
			.Column1.ControlSource = 'crTmpTotal.Dpros'
			.Column2.ControlSource = 'crTmpTotal.Tota1s'
		EndWith
	EndIf				

	thisform.grdItem.Visible = .f.
	thisform.say2.Visible = .f.
	thisform.getDProd.Visible = .f.
	thisform.grdTotPrev.Visible = .t.
	thisform.getTotPrev.Visible = .t.
	thisform.getTotReal.Visible = .t.
	thisform.getTotDif.Visible = .t.
Else && Tiago - 14/04/2014 - SRCOM - Busca todos os subníveis que o item passou que são iguais ao DOPES atual
	thisform.Caption = [Itens Subnível]
	
	Create Cursor TmpItePen ( DopeNums c(26), CPros c(14), Dpros c(40), Qtds n(12,3), Cunis c(3), Pesos n(12,3), CuniPs c(3), Fators n(12,3), Units n(12,3), ;
		Moedas c(3), Totas n(15,2), Datas D, Dcunis c(20), Dcunips c(20))
	Index On Dtos(Datas) Tag Datas

	lnCodigo = crSigMvOrc.nCodigos
	llAchou = .t.
	Do While llAchou
		Set decimals to 0
		ndope = Val( Left( Str(lnCodigo, 10 ), 4 ) )
		=ThisForm.Parentform.ParentForm.poDataMgr.cursorquery('SigCdOpe','TmpOpe','ndopes',nDope,[Dopes])
		lcDope = TmpOpe.Dopes
*!*			If TprMvCab.Dopes <> lcDope
*!*				Exit
*!*			Endif
		lnNume = Val( Right( Str(lnCodigo, 10 ), 6 ) )
		lcEDN = TprMvCab.Emps + lcDope + Str( lnNume, 6 )
	
		lcSqlEstPe = [Select b.Datas, a.nCodigos, c.citens, c.cpros, c.dpros, c.qtds, c.cunis, c.pesos, c.cunips, c.fators, c.units, c.moedas, c.totas, ] +;
						 [d.Dunis as Dcunis, e.Dunis as Dcunips ] +;
		             [From SigMvItn a Inner Join SigMvCab b On a.EmpDopNums = b.EmpDopNums ] +;
		             [Inner Join SigMvOrc c On a.EmpDopNums = c.EmpDopNums And a.Citens = c.Citens ] +;
		             [Inner Join SigCdUni d On a.Cunis = d.Cunis ] +;
		             [Inner Join SigCdUni e On a.Cunips = e.Cunis ] +;
			             [Where a.EmpDopNums = '] + lcEDN + [' ] + ;
			             [And a.Cpros = '] + crSigMvOrc.Cpros + [' Order By c.cItens]																
		If (ThisForm.Parentform.ParentForm.poDataMgr.SqlExecute(lcSqlEstPe, 'crMvPec') < 1)
			=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (crMvPec)')
			Return .f.
		Endif
		If Eof('crMvPec')
			Exit
		Endif
		Select crMvPec
		Scatter Memo Memvar
		m.DopeNums = lcDope + Str( lnNume, 6 )

		Insert into TmpItePen from Memvar
		lnCodigo = m.nCodigos
	Enddo
	Select TmpItePen
	Go top
	If Eof([TmpItePen])
		ThisForm.ParentForm.Enabled = .t.	
		Return .f.
	Endif

	With ThisForm.grdItem
		.RecordSource = [TmpItePen]
		.Column1.ControlSource = [TmpItePen.DopeNums]
		.Column6.ControlSource = [TmpItePen.Datas]		

		.Column2.ControlSource = [TmpItePen.Qtds]
		lcMask = '999999' + Iif(crSigOpCdd.CasQtds > 0, '.' + Repl( '9', crSigOpCdd.CasQtds ),'')
		.Column2.Inputmask = lcMask
		.Column2.Text1.Inputmask = lcMask
		.Column8.ControlSource = [TmpItePen.Dcunis]	
	
		.Column3.ControlSource = [TmpItePen.Pesos]
		lcMask = '999999' + Iif(crSigCdOpe.NCPesos > 0, '.' + Repl( '9', crSigCdOpe.NCPesos ),'')
		.Column3.Inputmask = lcMask
		.Column3.Text1.Inputmask = lcMask		
		.Column9.ControlSource = [TmpItePen.DcuniPs]

		.Column7.ControlSource = [TmpItePen.Fators]
		.Column7.Inputmask           = lcMask
		.Column7.Text1.Inputmask     = lcMask

		.Column4.ControlSource = [TmpItePen.Units]
		.Column10.ControlSource = [TmpItePen.Totas]
	Endwith
				
	thisform.grdItem.Visible = .t.
	thisform.say2.Visible = .t.
	thisform.getDProd.Visible = .t.
	thisform.grdTotal.Visible = .f.
	thisform.grdTotPrev.Visible = .f.
	thisform.getTotPrev.Visible = .f.
	thisform.getTotReal.Visible = .f.
	thisform.getTotDif.Visible = .f.	
Endif 


ENDPROC
