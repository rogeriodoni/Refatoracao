PROCEDURE lerarquivos
Local lcDir, lcDirReq, lcDirResp, laDir, lnDir
LOCAL lnFab, lcArqResp, lcArqReq, lcTexto, lcCmd, lcsessao, lcCodAti, lcChave, lcXml

lcDir = Alltrim(crsigcdpac.dirresta)
lcDir = Iif(Right(lcDir, 1)=[\], lcDir, lcDir + [\])

lcDirReq= lcDir+[Req\]
lcDirResp=lcDir+[Resp\]

*** Declara as dlls  ***
Declare String AtivarSAT             IN "SAT.dll" Integer numeroSessao, String subComando, String codigoDeAtivacao, String CNPJ, Integer cUF
Declare String EnviarDadosVenda      IN "SAT.dll" Integer numeroSessao, String codigoDeAtivacao, String dadosVenda
Declare String CancelarUltimaVenda   IN "SAT.dll" Integer numeroSessao, String codigoDeAtivacao, String chave, String dadosCancelamento
Declare String ConsultarSAT          IN "SAT.dll" Integer numeroSessao
Declare String ConsultarNumeroSessao IN "SAT.dll" Integer numeroSessao, String numeroSessaoAConsultar
Declare String DesbloquearSAT        IN "SAT.dll" Integer numeroSessao, String codigoDeAtivacao
Declare String AssociarAssinatura    IN "SAT.dll" Integer numeroSessao, String codigoDeAtivacao, String CNPJs, String assinaturaCNPJs
Declare String TesteFimAFim          IN "SAT.dll" Integer numeroSessao, String codigoDeAtivacao, String dadosVenda
SET STEP ON 
*** mena 07/07/2016 - associar assinatura antes de entrar no loop ***
lcCnpjAC = ALLTRIM(SUBSTR(crCdEmp.textoPaf,1,14))		&& AC de teste cadastrar no texto do ECF '16716114000172'	&&'11458830000183'
lcAssi   = ALLTRIM(SUBSTR(crCdEmp.textoPaf,16,344))	&& cadastrar no texto do ECF a partir da posicao 16	'SGR-SAT SISTEMA DE GESTAO E RETAGUARDA DO SAT'
lnSessao = fGerUniqueKey('SAT')	&& mena 26/06/2016 ajusta tipo do campo sessao
lcCodAti = '12345678'
lcRetorno = AssociarAssinatura(lnSessao, lcCodAti, lcCnpjAC, lcAssi)


llParar = .F.
Set Escape On
On Escape llParar = .T.

ThisForm.Label1.Caption = '<ESC> para interromper...'

*** espera xml de resposta
do while !llParar
	lnArq = ADir(laArq, lcDirReq + [*.XML])
	If (lnArq > 0)
		For lnPrc = 1 To lnArq
			lcArqResp = lcDirResp+Alltrim(laArq(lnPrc, 1))
			lcArqRespT= lcDirResp+Alltrim(laArq(lnPrc, 1))
			lcArqRespT= LEFT(lcArqRespT,LEN(lcArqRespT)-3)+[tmp]	&& prepara mesmo nome .tmp para depois renomear
			lcArqReq = lcDirReq + Alltrim(laArq(lnPrc, 1))
			lcTexto = FILETOSTR(lcArqReq)
			lcCmd	= STREXTRACT(lcTexto,'<Comando>', '</Comando>',1)
			lcsessao= STREXTRACT(lcTexto,'<Sessao>', '</Sessao>',1)
			lcFab   = STREXTRACT(lcTexto,'<Fabricante>', '</Fabricante>',1)
			lcCodAti= STREXTRACT(lcTexto,'<CodAti>', '</CodAti>',1)
			lcChave = STREXTRACT(lcTexto,'<Chave>', '</Chave>',1)
			lcXml= STREXTRACT(lcTexto,'<Xml>', '</Xml>',1)

			DELETE FILE &lcArqReq
			IF lcCmd = 'ConsultarSAT'
				lcSessao = VAL(lcSessao)	&& mena 26/06/2016 ajusta tipo do campo sessao
				lcRetorno = ConsultarSAT(lcSessao)
			ENDIF
			IF lcCmd = 'EnviarDadosVenda'
				lcSessao = VAL(lcSessao)	&& mena 26/06/2016 ajusta tipo do campo sessao
				lcRetorno = EnviarDadosVenda(lcSessao, lcCodAti, lcXml)
			ENDIF
			IF lcCmd = 'CancelarUltimaVenda'
				lcSessao = VAL(lcSessao)	&& mena 26/06/2016 ajusta tipo do campo sessao
				lcRetorno = CancelarUltimaVenda(lcSessao, lcCodAti, lcChave, lcXml)
			ENDIF
			STRTOFILE(lcRetorno,lcArqRespT,.f.)
			RENAME &lcArqRespT to &lcArqResp	
		ENDFOR
	endif		
EndDo

Set Escape Off
On Escape
If llParar
	MessageBox( 'Processamento Interrompido...', 32 )
	=fGravarLog('S', [SIGOPSAT], [SAIR], 'Processamento Interrompido...' )
	Do pEncerrar
EndIf

ENDPROC
