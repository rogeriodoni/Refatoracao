PROCEDURE msv_procurar
SELECT crSigCdMun
If not Empty(ThisForm.Pagina.Dados.get_Codigos.Value) 
	LOCATE FOR Codigos = ThisForm.Pagina.Dados.get_Codigos.Value
Else
	LOCATE FOR Descs = ThisForm.Pagina.Dados.get_Descs.Value 
Endif	
* Fim **************************************************************************************

ENDPROC

------------------------------------------------------------
* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
Lparameters lcAcesso

With ThisForm
	.poDatamgr = CreateObject('fSqlConector', 'SigCdMun')
	With .poDataMgr
		lcQry = [Select * from SigCdMun]
		.AddCursor('SigCdMun', 'cIdChaves', 'crSigCdMun', '', ThisForm.Pagina.Lista.Grade,lcQry)
		.pfSqlTabela(1).pColuna('Codigos', '', '', 'Código',    060, .t.)
		.pfSqlTabela(1).pColuna('Descs',    '', '', 'Município', 290, .t.)
	EndWith	
EndWith

=DoDefault()

ThisForm.Pagina.Lista.Grade.Column1.SetFocus
* Fim **************************************************************************************

ENDPROC

PROCEDURE Pagina.Lista.Grupo_op.Click
Lparameters Opcao

If Not DoDefault(Opcao)
	ThisForm.mAtivapagina1
	Return 0
EndIf

Select crSigCdMun
Do Case
	Case Inlist(ThisForm.pcEscolha, 'CONSULTAR', 'EXCLUIR')
		ThisForm.Pagina.Dados.Grupo_Salva.Salva.Setfocus
	Otherwise 
		This.Parent.Parent.Dados.get_Codigos.Setfocus
EndCase 

Select crSigCdMun

ThisForm.Pagina.Dados.Activate
* Fim **************************************************************************************
ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
Local lnRecno1,lnNreg1,lcGrupo,lcDope,laCampos(2,2)

If InList(ThisForm.pcEscolha,'INSERIR')
	If Empty(ThisForm.Pagina.Dados.get_Codigos.Value)
		=MessageBox('Obrigatório Informar o Código do Município!!!', 48, '')
		This.Parent.Parent.Get_Codigos.SetFocus
		Return 0
	EndIf
	
	If Empty(This.Parent.Parent.Get_Descs.Value)
		=MessageBox('Obrigatório Informar o Nome do Município!!!', 48, '')
		This.Parent.Parent.Get_Descs.SetFocus
		Return 0
	EndIf
	
	lcquery = [select codigos from SigCdMun where codigos = '] + ThisForm.Pagina.Dados.Get_Codigos.Value + ;
				[' and descs = '] + This.Parent.Parent.Get_Descs.Value + [']
*!*		ThisForm.poDataMgr.ChkRegister('SigCdMun','Codigos',ThisForm.Pagina.Dados.Get_Codigos.Value )
	thisform.podatamgr.sqlexecute(lcquery,'Tmpmunic')
	If Reccount('Tmpmunic') > 0
		=MessageBox('Município já cadastrado!!!', 48, '')
		This.Parent.Parent.Get_Codigos.SetFocus
		Return 0
	EndIf 

	Select crSigCdMun
	If Empty(crSigCdMun.cIdChaves)
		Replace cIdChaves With fUniqueIds() In crSigCdMun
	EndIf
EndIf

=DoDefault()

ThisForm.mAtivapagina1
* Fim **************************************************************************************

ENDPROC
