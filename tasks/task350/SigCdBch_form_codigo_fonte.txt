pcCodigos = pCodigos
pcForms   = pForms
pcArqs    = pArqs
pcCursor  = pCursor
plExcluir = pExcluir

llOk = .T.

If ThisForm.poDataMg2.Requery('CrSigCdGrl') < 1
	MessageBox('Favor reinicializar o processo !!!',16,'Falha na Conexão (Requery SigCdGrl)')
	llOk = .F.
Else
	Select CrSigCdGrl
	Index On Arqs + Progs + Str(Codigos,6) Tag Arqs
	Go Top
EndIf

loBarraGrv = CreateObject('fwprogressbar','Atualizando Arquivos...',2)
loBarraGrv.Show

Select (pcCursor)
Go Top
Scan
	loBarraGrv.Update(.T.,'SigCdBch...')
	lnRegs = Reccount(pcCursor)
	If lnRegs > 0
		If llOk
			Select CrSigCdGrl
			Go Top
			If !Seek(pcArqs + pcForms + Str(pcCodigos,6),'CrSigCdGrl','Arqs')
				m.Emps      = _Empr
				m.Codigos   = pcCodigos
				m.Arqs      = pcArqs
				m.Progs     = pcForms
				m.AtuRegs   = 0
				m.Datas     = Datetime()
				m.CIdChaves = fUniqueIds()
				Insert Into CrSigCdGrl From MemVar
			EndIf
			Replace Regs With lnRegs In CrSigCdGrl
			lnAtuRegs = (CrSigCdGrl.AtuRegs + 1)
			lcCursor  = 'Cg' + Alltrim(pcCursor)
			Select (pcCursor)
			Go Top
			If lnAtuRegs > lnRegs
				Go Bottom
				Skip
			Else
				Go lnAtuRegs
			EndIf
			lnCnt = 0
			Do While !Eof()
				Scan While !Eof() And lnCnt < 100
					loBarraGrv.Rodape.Caption = 'Atualizando Registros... ' + Alltrim(Str(((Recno() / lnRegs) * 100),3)) + ' %'
					Scatter MemVar Memo
					Insert Into (lcCursor) From MemVar
					lnCnt = lnCnt + 1
					Select (pcCursor)
				EndScan
				If !ThisForm.poDataMg2.Update(lcCursor)
					llOk = .F.
					MessageBox('Favor reinicializar o processo!!!',16,'Falha na Conexão (Update - ' + lcCursor + ')')
				Else
					Select (lcCursor)
					lnAtualizados = Reccount()
					Select CrSigCdGrl
					Replace AtuRegs With AtuRegs + lnAtualizados , ;
							Datas With Datetime() In CrSigCdGrl
					lnCnt = 0
					If !ThisForm.poDataMg2.Update('CrSigCdGrl')
						llOk = .F.
						MessageBox('Favor reinicializar o processo!!!',16,'Falha na Conexão (Update - CrSigCdGrl')
					Else
						If ThisForm.poDataMg2.Commit() < 1
							llOk = .F.
						Else
							Zap In (lcCursor)
						EndIf
					EndIf
				EndIf
				Select (pcCursor)
			EndDo
		EndIf
	EndIf
EndScan
loBarraGrv.Complete(.T.)

If plExcluir
	lnRegs   = Reccount(pcCursor)
	lcCursor = 'Cg' + Alltrim(pcCursor)
	
	loBarraExc = CreateObject('fwprogressbar','Atualizando Arquivos...',lnRegs)
	loBarraExc.Show
	
	Set Deleted Off

	Select (pcCursor)
	Go Top
	Scan
		loBarraExc.Update(.T.)
		If Deleted()
			Scatter MemVar Memo
			Insert Into (lcCursor) From MemVar
		EndIf
		Select (pcCursor)
	EndScan
	loBarraExc.Complete(.T.)
	
	Set Deleted On
	
	Select (lcCursor)
	Delete From (lcCursor)
	
	If !ThisForm.poDataMg2.Update(lcCursor)
		llOk = .F.
		MessageBox('Favor reinicializar o processo!!!',16,'Falha na Conexão (Delete - ' + lcCursor + ')')
	Else
		If ThisForm.poDataMg2.Commit() < 1
			llOk = .F.
		Else
			Zap In (lcCursor)
		EndIf
	EndIf
EndIf

Return(llOk)
ENDPROC

------------------------------------------------------------
PROCEDURE verifcheq
LParameters lcBanco, lcAgencia, lcConta, lcNumero

lcTipo   = Str(ThisForm.Pagina.Leitura.TipoLeitura.Value,1)
lcCheque = lcBanco + lcAgencia + lcConta + lcNumero
lcMsg = ''
If !fChkDigitos(lcBanco) Or !fChkDigitos(lcAgencia) Or !fChkDigitos(lcConta) Or !fChkDigitos(lcNumero)
	lcMsg = 'Leitura de Cheque Incorreta !!!'
Else
	Select Historico
	Set Order To TpCheque &&Cheque
	If Seek(lcTipo + lcCheque)
		lcMsg = 'Cheque já lançado !!! <ENTER> para continuar...'
	Else
		Select Balanco
		Set Order To Cheque
		=Seek(lcCheque)
		If Eof('Balanco')
			If !ThisForm.ReadComm.Ole_Comm.PortOpen And !ThisForm.ReadComm.PortaError
				ThisForm.ReadComm.Ole_Comm.PortOpen = .T.
			EndIf
			ThisForm.ReadComm.Ole_Comm.OutPut = 'WEBCD'
			Wait Window 'Cheque não Cadastrado !!! <ENTER> para continuar...'
		EndIf
	EndIf
EndIf
If !Empty(lcMsg)
	Wait Window lcMsg
	Return(.F.)
Else
	Wait Window 'Passe o Cheque pela Leitora, ou ' + Chr(13) + '<ENTER> para Digitar...' NoWait
	Return(.T.)
EndIf
ENDPROC

PROCEDURE msv_inserir
Local lnCodigos, lcGrupo, lcConta, lcOpeCh, lcCond

With ThisForm.Pagina.Dados
	lnCodigos = .Get_Codigo.Value
	lcGrupo   = .GetGrupo.Value
	lcConta   = .GetConta.Value
	lcOpeCh   = .Get_Oper.Value
	lnTipo    = .Cmb_Operacao.Value
EndWith

lcWhere = ''

Do Case
	Case lnTipo = 1 And Empty(lcOpeCh)
		lcWhere = [And b.Tipos = 'DB' And b.Ordens = 1 ]
	Case lnTipo = 2 And Empty(lcOpeCh)
		lcWhere = [And b.Tipos = 'DB' And b.Devolvidos <> 'S' ]
	Case lnTipo = 3 And Empty(lcOpeCh)
		lcWhere = [And b.Tipos = 'DB' And b.Devolvidos = 'S' ]
	Case !Empty(lcOpeCh)
		lcWhere = ''
	Other
		lcWhere = [And b.Tipos = 'DB' ]
Endcase

Wait Window 'Aguarde! Selecionando Cheques...' NoWait
If Used('Balanco')
	Zap In Balanco
EndIf

lStrQuery = [Select a.Bancos,a.Agencias,a.NContas,a.NCheques,a.Nums,a.UMovs,a.Valors,a.Vencs,a.GrClis,a.IClis,]+;
				[0 As Lidos,0 As Relidos ]+;
			[From SigChe a, SigCdOpt b ]+;
			[Where a.Umovs = b.Operacaos ] + Iif(Empty(lcWhere),'',lcWhere)+;
				Iif(Empty(lcGrupo),'',[And a.Grupos = ']+lcGrupo+[' ])+;
				Iif(Empty(lcConta),'',[And a.Contas = ']+lcConta+[' ])+;
				Iif(Empty(lcOpeCh),'',[And a.Umovs = ']+lcOpeCh+[' ])

If ThisForm.poDataMgr.SqlExecute(lStrQuery,'CsBalanco') < 1
	MessageBox('Favor renicializar o processo !!!',16,'Falha na Conexão (CsBalanco)')
	Return(.F.)
EndIf

Wait Clear

loBarra = CreateObject('fwprogressbar','Aguarde! Atualizando o Balanço...',Reccount('CsBalanco'))
loBarra.Show

Select CsBalanco
Go Top
Scan
	loBarra.Update(.T.)
	Scatter MemVar
	m.Codigos   = lnCodigos
	m.CIdChaves = fUniqueIds()
	Insert Into Balanco From MemVar
EndScan
loBarra.Complete(.T.)

	lnOk = ThisForm.poDataMgr.Commit()
Else
	MessageBox('Favor reinicializar o processo !!!',16,'Falha na Conexão (Update SigCqBal)')
	Return(.F.)
EndIf

	llOk = ThisForm.Grava(lnCodigos,ThisForm.Name,'SigCqBal','Balanco',.F.)
	If llOk
		Select CrSigCdBch
		Replace Grvs With .T. In CrSigCdBch
	Else
		MessageBox('Favor reinicializar o processo !!!',16,'Falha na Conexão (Grava - Inserir)')
		Return(.F.)
	EndIf
Else
	MessageBox('Favor reinicializar o processo !!!',16,'Falha na Conexão (Commit - Inserir)')
	Return(.F.)
