PROCEDURE Click
ThisForm.Podatamgr6.cursorquery('SigCdOpd','CrSigCdOpd','Dopps',Upper(ThisForm.pDope),[Dopps,Pccs])

If (ThisForm.pTipo <> 'C') And Not WExist('SigPdMp6')
	DO form SigPdMp6 with ThisForm.pDope,ThisForm
Else
	If (ThisForm.pTipo = 'C') And Not WExist('SigPdMp6')
		DO form SigPdMp6 with ThisForm.pDope,ThisForm,Thisform.pNume,ThisForm.pEmp,.t.
	EndIf
EndIf

ENDPROC
