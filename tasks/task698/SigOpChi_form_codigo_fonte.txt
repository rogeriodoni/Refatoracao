With ThisForm.Pagina.Filtro.Origem
	If !Empty(This.value)
		Cgc = This.Value
		Cgc = Strt( Strt( Strt( Cgc, ".", "" ), "-", "" ), "/", "" )
		If Len(Alltrim(Cgc))<>14
			Cgc1 = TransForm( Cgc, "@R 999.999.999-99" )
		Else
			Cgc1 = TransForm( Cgc, "@R 99.999.999/9999-99" )
		EndIf
		
		lcQuery = [Select Iclis, Cpfs, Rclis From SigCdCli Where Cpfs = ?Cgc1 ]

		lnErro = ThisForm.podatamgr.SqlExecute(lcQuery,'CrSigCdCli')
		
		If lnErro < 1
			MessageBox('Favor reinicializar o processo.',16,'Falha na Conexão')
			Return 0
		EndIf
		Select CrSigCdCli
		Go Top
		If Reccount() = 0
			m.loLista = CreateObject('fwBuscaExt',Thisform.poDatamgr.pnIdconn,'SigCdCli','crSigCdCli','cpfs',This.Value,'Clientes',1000)
			If .Not.m.loLista.plAchouRegistro
				m.loLista.mAddColuna('Cpfs' ,'', 'C.P.F. / C.N.P.J.' )
				m.loLista.mAddColuna('Rclis',Replicate('x',50), 'Nome' )
				m.loLista.mAddColuna('Iclis','', 'Código' )
				m.loLista.mAddColuna('Inativas ', .f., .f., .f., .f., .f., .t.)
				m.loLista.gridProc.SetAll('DynamicForeColor', [Iif(crSigCdCli.Inativas == 1, Rgb(255, 0, 0), Rgb(0,0,0))], 'Column')
				m.loLista.Show()
			EndIf
		Endif

		If !fAcessoContas(Usuar, '', 'C', crSigCdCli.Iclis, .get_conta.value, .get_dconta.value)
			=Messagebox('Acesso Negado!!!', 0+48, '')
			.get_conta.Value  = ''
			.get_dconta.Value = ''
			.get_cpf.Value    = ''
			Return(.f.)
		Else
			.get_conta.value  = crSigCdCli.Iclis
			.get_dconta.value = crSigCdCli.Rclis
			.get_cpf.value    = crSigCdCli.Cpfs
		Endif
		.get_cpf.Refresh
		.get_conta.Refresh
		.get_dconta.Refresh

		If Len(Alltrim(This.Value))=14
			If !fValidarCPF(Alltrim(This.Value))
				=MessageBox('CPF Inválido!!!', 0+64, 'Atenção')
				Return(.f.)
			Endif
		Else
			If !fValidarCNPJ(Alltrim(This.Value))
				=MessageBox('CGC Inválido!!!', 0+64, 'Atenção')
				Return(.f.)
			Endif
		Endif
	Else
		.Get_DConta.Value = ''
	Endif
EndWith
ENDPROC

* PROCEDURE Valid

PROCEDURE Valid
Local lcCgc

With ThisForm.Pagina.historico.emitente
	If !Empty(This.value)
		Cgc = This.Value
		Cgc = Strt( Strt( Strt( Cgc, ".", "" ), "-", "" ), "/", "" )
		If Len(Alltrim(Cgc))<>14
			Cgc1 = TransForm( Cgc, "@R 999.999.999-99" )
		Else
			Cgc1 = TransForm( Cgc, "@R 99.999.999/9999-99" )
		EndIf
		
		lcQuery = [Select Iclis, Cpfs, Rclis From SigCdCli Where Cpfs = ?Cgc1 ]

		lnErro = ThisForm.podatamgr.SqlExecute(lcQuery,'CrSigCdCli')
		
		If lnErro < 1
			MessageBox('Favor reinicializar o processo.',16,'Falha na Conexão')
			Return 0
		EndIf
		Select CrSigCdCli
		Go Top
		If Reccount() = 0
			m.loLista = CreateObject('fwBuscaExt',Thisform.poDatamgr.pnIdconn,'SigCdCli','crSigCdCli','cpfs',This.Value,'Clientes',1000)
			If .Not.m.loLista.plAchouRegistro
				m.loLista.mAddColuna('Cpfs' ,'', 'C.P.F. / C.N.P.J.' )
				m.loLista.mAddColuna('Rclis',Replicate('x',50), 'Nome' )
				m.loLista.mAddColuna('Iclis','', 'Código' )
				m.loLista.mAddColuna('Inativas ', .f., .f., .f., .f., .f., .t.)
				m.loLista.gridProc.SetAll('DynamicForeColor', [Iif(crSigCdCli.Inativas == 1, Rgb(255, 0, 0), Rgb(0,0,0))], 'Column')
				m.loLista.Show()
			EndIf
		Endif

		If !fAcessoContas(Usuar, '', 'C', crSigCdCli.Iclis, .get_conta.value, .get_dconta.value)
			=Messagebox('Acesso Negado!!!', 0+48, '')
			.get_conta.Value  = ''
			.get_dconta.Value = ''
			.get_cpf.Value    = ''
			Return(.f.)
		Else
			.get_conta.value  = crSigCdCli.Iclis
			.get_dconta.value = crSigCdCli.Rclis
			.get_cpf.value    = crSigCdCli.Cpfs
		Endif
		.get_cpf.Refresh
		.get_conta.Refresh
		.get_dconta.Refresh

		If Len(Alltrim(This.Value))=14
			If !fValidarCPF(Alltrim(This.Value))
				=MessageBox('CPF Inválido!!!', 0+64, 'Atenção')
				Return(.f.)
			Endif
		Else
			If !fValidarCNPJ(Alltrim(This.Value))
				=MessageBox('CGC Inválido!!!', 0+64, 'Atenção')
				Return(.f.)
			Endif
		Endif
	Else
		.Get_DConta.Value = ''
	Endif
ENDWITH

ENDPROC

	If !Empty(This.Value)

			lcTmp = '%'+Alltrim(This.Value) + '%'
			ThisForm.poDataMgr.SqlExecute([Select Rclis,Iclis,Cpfs,Conjuges,Inativas From SigCdCli Where Rtrim(RClis) Like ?lcTmp ], 'CrSigCdCli')
			Select CrSigCdCli

			xBrow=CreateObject('fwBuscaSel','CrSigCdCli','RClis',This.value,'Seleção')

			xBrowCol=xBrow.AddColuna('CPFS','','C.P.F.',.F.)

			xBrow.Show
			lfan = CrSigCdCli.Rclis
			lcod = CrSigCdCli.Iclis
		Else
			lfan = crSigCdCli.Rclis
			lcod = crSigCdCli.Iclis
		Endif

		.get_conta.value  = crSigCdCli.Iclis
		.get_dconta.value = crSigCdCli.Rclis
		.get_cpf.value  = crSigCdCli.Cpfs

		If !fAcessoContas(Usuar, '', 'D', lfan, .get_conta.value, This )
			Messagebox('Acesso Negado !!',0+48,'')
			.get_conta.value  = ''
			.get_dconta.value = ''
			.get_cpf.value  = ''
			.get_cpf.refresh
			.get_conta.refresh
			.get_dconta.refresh
			return 0 
		Else
			.get_cpf.refresh
			.get_conta.refresh
			.get_dconta.refresh
		Endif
	else
		.Get_Conta.Value = ''
		.get_cpf.value  = ''
		.get_cpf.refresh
		.get_conta.refresh
	endif

EndWith
Return(.t.)

ENDPROC
