* PROCEDURE atualizamarkup
------------------------------------------------------------
PROCEDURE atualizamarkup
Lparameters pPro

lnMkp = 0

lcQuery = [Select CustoFs, MoeCusFs, PVens, MoeVs ] + ;
		    [From SigCdPro ] + ;
		   [Where CPros = '] + pPro + [']

If (ThisForm.poDataMgr.SqlExecute(lcQuery, 'LocalPro') < 1)
	ThisForm.Podatamgr.RollBack()
	MessageBox('Favor reinicializar o processo.',16,'Falha na Conexão')
	Return .f.
EndIf
Go Top In LocalPro

lcQuery = [Select CMoes, MoeQs, QtdEqs ] + ;
		    [From SigCdMoe ] + ;
		   [Where CMoes In ('] + LocalPro.MoeCusFs + [', '] + LocalPro.MoeVs + [')]

If (ThisForm.poDataMgr.SqlExecute(lcQuery, 'LocalMoe') < 1)
	ThisForm.Podatamgr.RollBack()
	MessageBox('Favor reinicializar o processo.',16,'Falha na Conexão')
	Return .f.
EndIf
Select LocalMoe
Index On CMoes Tag CMoes

lcMoe = LocalPro.Moecusfs
lnQeq = 1
If Seek(LocalPro.MoeCusFs, [LocalMoe], [CMoes]) And Not Empty(LocalMoe.MoeQs)
	lcMoe = LocalMoe.MoeQs
	lnQeq = Iif(LocalMoe.QtdEqs = 0, 1, LocalMoe.QtdEqs)
EndIf
lnMof = fBuscarCotacao(lcMoe, DateTime(), ThisForm.poDataMgr) * lnQeq

lcMoe = LocalPro.Moevs
lnQeq = 1
If Seek(LocalPro.MoeVs, [LocalMoe], [CMoes]) And Not Empty(LocalMoe.MoeQs)
	lcMoe = LocalMoe.Moeqs
	lnQeq = Iif(LocalMoe.Qtdeqs=0, 1, LocalMoe.Qtdeqs)
EndIf
lnMov = fBuscarCotacao(lcMoe, DateTime(), ThisForm.poDataMgr) * lnQeq

lnMkp = Iif(LocalPro.CustoFs = 0, 0, Round((LocalPro.PVens * lnMov) / (LocalPro.CustoFs * lnMof), 3))

lcUpdate = [Update SigCdPro Set MarkupA = ] + Str(lnMkp,12,3) + [ Where Cpros = '] + pPro + [']
If ThisForm.PodataMgr.SqlExecute(lcUpdate,'') < 1
	ThisForm.Podatamgr.RollBack()
	MessageBox('Favor reinicializar o processo.',16,'Falha na Conexão')
	Return .f.
EndIf

Return .t.
* Fim **************************************************************************************
ENDPROC

------------------------------------------------------------
PROCEDURE carregaitens
Local vCond, lnMarcas, lnVariacao, lnArred, lnPLiqs, lnIndice

lnIndice = ThisForm.Pagina.Dados.Cnt_Compra.GetIndice.Value

Wait Window 'Aguarde !!! Selecionando Produtos...' NoWait

lStrQuery = [Select flagctabs,cpros,dpros,pvens,precode,moevs,situas,cgrus,colecoes,linhas,Pcuss,Moecs,Reffs,Ean13 ]+;
			[From SigCdPro ]+;
			[Where Ifors = ']+ThisForm.Pagina.Dados.Cnt_Compra.Get_conta.Value+[' ]+;
			[Order By cpros ]
If ThisForm.poDataMgr.SqlExecute(lStrQuery,'TmpPro') < 1
	MessageBox('Favor reinicializar o processo.',16,'Falha na Conexão')
	RETURN 0
EndIf
		
SELECT CrSigCdLpi
Zap

Select TmpPro
Scan 
	Scatter MemVar
	m.lprecos   = CrSigCdLpc.lprecos
	m.flagutabs = .T.
	m.cidchaves = fUniqueIds()
	m.vencis    = ThisForm.Pagina.Dados.Cnt_Compra.fweditdata_VENC.Value
	m.vencfs    = CTOT('31/12/9999')
	m.pvens     = m.pcuss
	m.Moevs     = m.Moecs
	m.precode	= m.precode
	m.pcuss     = Iif(lnIndice <> 0,m.pcuss * lnIndice,m.pcuss)

	If (ThisForm.CompVenda <> [C])
		m.pvens = fArredondamento(m.pvens, ThisForm.poDataMgr)
	EndIf

    Insert Into CrSigCdLpi From MemVar
EndScan
ENDPROC

* PROCEDURE checagrpvenda

PROCEDURE checagrpvenda
Lparameters pPro

If (ThisForm.CompVenda <> [C])
	lcQuery = [Select a.Colecoes, b.AltProds ] + ;
				[From SigCdPro a, SigCdCol b ] + ;
			   [Where a.CPros = '] + Padr(pPro, 14) + [' And ] + ;
			   		 [a.Colecoes = b.Colecoes]

	If (ThisForm.poDataMgr.SqlExecute(lcQuery, [crBusCol]) < 1)
		=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (Grupo_Op - crBusCol)')
		Return .f.
	EndIf

	Select crBusCol
	Go Top In crBusCol
	If Not Eof([crBusCol]) And Not Empty(crBusCol.Colecoes) And (crBusCol.AltProds = 1)
		lcMsg = [Produto "] + Alltrim(pPro) + [" no Grupo de Venda "] + Alltrim(crBusCol.Colecoes) + ;
				[". Deseja Prosseguir?]
		If (MessageBox(lcMsg, 4+32+256, [Atenção!!!]) <> 6)
			Return .f.
		Else
			=fGravarLog('T', 'SigCdLpc', Usuar, 'O Produto [' + Alltrim(pPro) + '] ' + ;
						  'do Grupo de Venda [' + Alltrim(crBusCol.Colecoes) + '] Teve Liberação Para Alteração.')
		EndIf
	EndIf
EndIf

Return .t.
* Fim **************************************************************************************
ENDPROC

* PROCEDURE gravarlog
------------------------------------------------------------
PROCEDURE gravarlog
If Used([crOriginal])
	Select crOriginal
	Scan
		lcDtI = Dtoc(Ttod(fDtoSQL(crOriginal.VencIs)))
		lcDtF = Dtoc(Ttod(fDtoSQL(crOriginal.VencFs)))
		lcTip = []
		
		Select crSigCdLpi
		Go Top In crSigCdLpi
		Locate For CPros = crOriginal.CPros
		If Eof([crSigCdLpi])
			lcTip = [EXCLUIR]
		Else
			If ((crOriginal.PVens  <> crSigCdLpi.PVens) Or ;
				(crOriginal.VencIs <> crSigCdLpi.VencIs) Or ;
				(crOriginal.VencFs <> crSigCdLpi.VencFs))
				lcTip = [ALTERAR]
			EndIf
		EndIf

		If Not Empty(lcTip)
			lcTra = [Lista:] 	+ Alltrim(crOriginal.LPrecos) + ;
					[/Produto:] + Alltrim(crOriginal.CPros) + ;
					[/Valor:]   + Alltrim(Str(crOriginal.PVens, 12, 2)) + ;

* Classe: checkbox (Base: checkbox)
* Registro: 15
================================================================================

------------------------------------------------------------
* PROCEDURE Click
------------------------------------------------------------
PROCEDURE Click
This.Value = 0
If CrSigCdLpc.Vencis > DATE()
	MESSAGEBOX('Lista de Preço com Data de Vencimento para '+DTOC(CrSigCdLpc.VencIs),0+48,'')
	RETURN .f.
EndIf
If CrSigCdLpc.Flags
	MessageBox('Lista de Preço já Atualizada !!!',0+48,'')
	Return .f.
EndIf

lcquery = [Select * From SigCdLpi Where Lprecos = ']+CrSigCdLpc.Lprecos+[']
If ThisForm.PodataMgr.SqlExecute(lcQuery,'TmpLprei') < 1
	MessageBox('Favor reinicializar o processo.',16,'Falha na Conexão')
	Return(.F.)
EndIf

Select VencFs ;
  From TmpLprei ;
 Where (Ttod(Nvl(VencFs, Ctot([]))) >= Date()) Or ;
 	   Empty(Ttod(Nvl(VencFs, Ctot([])))) ;
  Into Cursor crAlerta

If (Reccount([crAlerta]) > 0)
	If MessageBox([Existe Produto Com Vencimento Maior Ou Igual a Hoje] + Chr(13) + ;
				  [Deseja Prosseguir Com a Atualização Mesmo Assim?],4+32+512, [Atenção!!!]) <> 6
		Return .f.
	EndIf
EndIf

Select TmpLpreI
Scan
	If Not ThisForm.ChecaGrpVenda(TmpLpreI.CPros)
		Return .f.
	EndIf
EndScan

If MessageBox('Confirma a Atualização dos Preços ',4+32,'Confirmação') = 6
	Zap In crSigCdPrc
	Zap In crSigPrCp2

	Select TmpLprei
	Scan
		ThisForm.poDataMgr.CursorQuery('SigCdPro', 'LocalPro', 'CPros', TmpLPreI.CPros)
		If Used('LocalPro') And Not Eof('LocalPro')
			Select LocalPro
			Scatter MemVar Memo
			m.DataAlts  = Datetime()
			m.HoraAlts  = Substr(Ttoc(m.DataAlts),12,8)
			m.UsuaAlts  = Usuar
			m.CIdChaves = fUniqueIds()
			m.Origem    = Ttoc(Datetime()) + [ SigCdLpc] 
			Insert Into crSigCdPrc From MemVar
		EndIf

		ThisForm.poDataMgr.CursorQuery('SigPrCpo', 'LocalCompo', 'CPros', TmpLPreI.CPros)
		If Used('LocalCompo') And Not Eof('LocalCompo')
			Select LocalCompo
			Scan
				Scatter MemVar Memo
				m.DataAlts = Datetime()
				m.HoraAlts = Substr(Ttoc(m.DataAlts),12,8)
				m.UsuaAlts = Usuar
				Insert Into crSigPrCp2 From MemVar
			EndScan
		EndIf
		
		Select TmpLprei
		Scatter Memvar

		If (ThisForm.CompVenda = [C])
			lcUpdate = [Update SigCdPro Set Pcuss = ] + Str(m.PCuss,15,3) + [, CustoFs = ]+STR(m.Pcuss,15,3)+[ Where Cpros = ']+m.Cpros+[']
		Else
			lcUpdate = [Update SigCdPro Set PVens = ] + Str(m.PVens,15,3) + [, PrecoDe = ] + Str(m.PrecoDe,15,3) + [ Where Cpros = '] + m.Cpros + [']
		EndIf
		
		If ThisForm.PodataMgr.SqlExecute(lcUpdate,'') < 1
			ThisForm.Podatamgr.RollBack()
			MessageBox('Favor reinicializar o processo.',16,'Falha na Conexão')
			Return .f.
		EndIf

		lcDelete = [Delete From SigPrPrt Where Cpros = '] + m.Cpros + [']
		If ThisForm.PodataMgr.SqlExecute(lcDelete,'') < 1
			ThisForm.Podatamgr.RollBack()
			MessageBox('Favor reinicializar o processo.',16,'Falha na Conexão - Delete - SigPrPrt')
			Return .f.
		EndIf
		
*!*			lcDelete = [Delete From SigPrWeb Where Cpros = '] + m.Cpros + [']
*!*			If ThisForm.PodataMgr.SqlExecute(lcDelete,'') < 1
*!*				ThisForm.Podatamgr.RollBack()
*!*				MessageBox('Favor reinicializar o processo.',16,'Falha na Conexão - Delete - SigPrWeb')
*!*				Return .f.
*!*			EndIf
		
		If Not ThisForm.AtualizaMarkup(m.Cpros)
			Return .f.
		EndIf
	EndScan

	lcFlag = .t.
	lcUpdate = [Update SigCdLpc Set Flags = ?lcFlag Where Lprecos = ']+CrSigCdLpc.Lprecos+[']
	If ThisForm.PodataMgr.SqlExecute(lcUpdate,'') < 1
		ThisForm.Podatamgr.RollBack()
		MessageBox('Favor reinicializar o processo.',16,'Falha na Conexão')
		Return(.F.)
	EndIf 

	If Not ThisForm.poDataMgr.Update('crSigCdPrc')
		ThisForm.Podatamgr.RollBack()
		=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (Update - crSigCdPrc)')
		Return 0
	EndIf

	If Not ThisForm.poDataMgr.Update('crSigPrCp2')
		ThisForm.Podatamgr.RollBack()
		=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (Update - crSigPrCp2)')
		Return 0
	EndIf

	If ThisForm.Podatamgr.Commit() < 1
		ThisForm.Podatamgr.RollBack()
		MessageBox('Favor reinicializar o processo.',16,'Falha na Conexão')
		Return(.F.)
	EndIf

	lcMsg = [Atualização de preço dos produtos com a lista de preço : ] + CrSigCdLpc.Lprecos

	=fGravarLog([T], ThisForm.Name, Usuar, lcMsg)

	ThisForm.mativapagina1 
EndIf
ENDPROC

PROCEDURE Valid
Local Grupo
Grupo = crSigCdPam.GrPadFors
If !IsEmpty(this.value)
	If !fAcessoContas(Usuar,grupo,'C',this.value,This,this.parent.get_dconta)
		Messagebox('Acesso Negado !!',0+48,'')
		This.Value=''
		This.Refresh
		this.parent.get_dconta.Value = ''
		this.parent.get_dconta.Refresh
		return 0 
	Endif
Else
	this.parent.get_dconta.Value = ''
	this.parent.get_dconta.Refresh
	return(.t.)
ENDIF

ENDPROC
