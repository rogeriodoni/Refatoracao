* PROCEDURE carregapedidos
------------------------------------------------------------
PROCEDURE carregapedidos
Local lcLet, lnUni, lnTot

lcQuery = [Select a.Codigos, b.DtCotas as Datas, b.Situas, a.QtdAdqs, a.CUnis, a.ValAdqs, a.Marca01, a.Unit01, ] + ;
				 [a.Tota01, a.Marca02, a.Unit02, a.Tota02, a.Marca03, a.Unit03, a.Tota03, a.Marca04, a.Unit04, ] + ;
				 [a.Tota04, a.Marca05, a.Unit05, a.Tota05, a.Marca06, a.Unit06, a.Tota06, b.Comprador, b.Aprovador, ] + ;
				 [b.DtAprovs, b.ObsAprovs ] + ;
		    [From SigCtTCt a, SigCtTDc b ] + ;
		   [Where a.CPros = '] + ThisForm.CPros + [' And ] + ;
		   		 [Not a.Codigos = ] + Str(ThisForm.Codigos, 6) + [ And ] + ;
		   		 [a.Codigos = b.Codigos ] + ;
		   [Order By b.DtCotas Desc, a.Codigos Asc] 

If (ThisForm.poDataMgr.SqlExecute(lcQuery, [LocalCotCt]) < 1)
	=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (CarregaPedidos - LocalCotCt)])
	Return .f.
EndIf

Select LocalCotCt
Scan
	lcLet = [ ]
	lnUni = 0
	lnTot = 0
	Do Case
		Case LocalCotCt.Marca01
			lcLet = [A]
			lnUni = LocalCotCt.Unit01
			lnTot = LocalCotCt.Tota01
		Case LocalCotCt.Marca02
			lcLet = [B]
			lnUni = LocalCotCt.Unit02
			lnTot = LocalCotCt.Tota02
		Case LocalCotCt.Marca03
			lcLet = [C]
			lnUni = LocalCotCt.Unit03
			lnTot = LocalCotCt.Tota03
		Case LocalCotCt.Marca04
			lcLet = [D]
			lnUni = LocalCotCt.Unit04
			lnTot = LocalCotCt.Tota04
		Case LocalCotCt.Marca05
			lcLet = [E]
			lnUni = LocalCotCt.Unit05
			lnTot = LocalCotCt.Tota05
		Case LocalCotCt.Marca06
			lcLet = [F]
			lnUni = LocalCotCt.Unit06
			lnTot = LocalCotCt.Tota06
	EndCase

	If Not Empty(lcLet)
		Scatter Memvar Memo
		m.Datas 	= Nvl(m.Datas, 	   Ctod([]))
		m.DtAprovs	= Nvl(m.DtAprovs,  Ctod([]))
		m.ObsAprovs = Nvl(m.ObsAprovs, [])
		Insert Into crHistPeds From Memvar
		
		lcQuery = [Select a.IClis, a.Fornecs ] + ;
				    [From SigCtTFn a ] + ;
				   [Where a.Codigos = ] + Alltrim(Str(m.Codigos, 6)) + [ And ] + ;
				   		 [a.Grades = '] + lcLet + [']

		If (ThisForm.poDataMgr.SqlExecute(lcQuery, [LocalCotFn]) < 1)
			=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (CarregaPedidos - LocalCotFn)])
			Return .f.
		EndIf
		Select LocalCotFn
		Go Top In LocalCotFn
		If Not Eof([LocalCotFn])
			Replace IClis   With LocalCotFn.IClis, ;
					Fornecs With LocalCotFn.Fornecs, ;
					Units   With lnUni, ;
					Totas   With lnTot In crHistPeds
		EndIf
	EndIf
EndScan

Select crHistPeds
Go Top In crHistPeds

ThisForm.grdCotacoes.Refresh
* Fim **************************************************************************************
ENDPROC

* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
Lparameters pCnx, pPro, pCod

If (Type([pCnx]) <> [O])
	=Messagebox([O Parâmetro de Conexão Passado Não É Valido...], 0+48, [Parâmetros Inválidos!!!])
	Return .f.
EndIf

With ThisForm
	.poDataMgr = pCnx
	If (.poDataMgr.pnIdconn > 0)
		.CPros = Padr(pPro, 10)
		.Codigos = pCod

		.grdCotacoes.SetAll([DynamicForeColor],[Iif(crHistPeds.Situas > 0, Rgb(0,0,255), Rgb(0,0,0))], [Column])
		
		.CarregaPedidos				
		
		=DoDefault()
	Else
		=Messagebox('Impossível Efetuar Conexão Com o Servidor de Banco de Dados...', 0+48, 'CreateObject')
		Return .f.
	EndIf
EndWith
* Fim **************************************************************************************
ENDPROC

=DoDefault()
