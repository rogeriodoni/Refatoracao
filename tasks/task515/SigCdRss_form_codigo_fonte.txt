------------------------------------------------------------
* PROCEDURE msv_procurar
------------------------------------------------------------
PROCEDURE msv_procurar
Local lcBusca

If .Not.IsEmpty(ThisForm.Pagina.Dados.Get_Origem.Value)
	ThisForm.poDataMgr.LocateCursor('crSigCdRps','grupoos',ThisForm.Pagina.Dados.Get_Origem.Value)
Endif
ENDPROC

------------------------------------------------------------
* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
LPARAMETERS lcacesso

With ThisForm
	.poDatamgr = CreateObject('fSqlConector')
	With .poDatamgr
		.AddCursor('SigCdRps','cidchaves','crSigCdRps','',ThisForm.pagina.lista.grade)
		.pfSqlTabela(1).pColuna('grupoos','','','Grupo Origem',100,.t.)
		.pfSqlTabela(1).pColuna('grupods','','','Grupo Destino',100,.t.)
		.AddCursor('SigCdGcr','codigos','crSigCdGcr','','')
	EndWith
	= DoDefault()
	.LockScreen = .f.
EndWith
ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Lista.Grupo_op.Click
------------------------------------------------------------
PROCEDURE Pagina.Lista.Grupo_op.Click
LParameters opcao

If !dodefault( opcao )
	ThisForm.mAtivapagina1
	Return
Endif

If InList(ThisForm.pcEscolha,'INSERIR','PROCURAR')
	ThisForm.Pagina.Dados.Get_dOrigem.Value = ' '
	ThisForm.Pagina.Dados.Get_ddestino.Value = ' '
Else
	If Thisform.PoDatamgr.SQLExecute([Select Descrs from SigCdGcr where Codigos = '] + CrSigCdRps.Grupoos + [' ],'CrSigCdGcr') < 0
		MessageBox('Favor reinicializar o processo.',16,'Falha na Conexão')
		Return(.F.)
	Else
		ThisForm.Pagina.Dados.Get_dOrigem.Value = CrSigCdGcr.Descrs
	Endif	
	
	If Thisform.PoDatamgr.SQLExecute([Select Descrs from SigCdGcr where Codigos = '] + CrSigCdRps.Grupods + [' ],'CrSigCdGcr') < 0
		MessageBox('Favor reinicializar o processo.',16,'Falha na Conexão')
		Return(.F.)
	Else
		ThisForm.Pagina.Dados.Get_ddestino.Value = CrSigCdGcr.Descrs
	Endif	
Endif

Do Case
	Case Inlist( ThisForm.pcEscolha, 'INSERIR', 'PROCURAR' )
		ThisForm.Pagina.Dados.Get_Origem.SetFocus
	Case ThisForm.pcEscolha = 'ALTERAR'
		ThisForm.Pagina.Dados.Get_Destino.SetFocus
Other
	ThisForm.Pagina.Dados.Grupo_Salva.Salva.SetFocus
Endcase

ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
Local laCampos(2,2)

get_origem  = thisform.pagina.dados.get_Origem.value
get_destino = thisform.pagina.dados.get_Destino.value

If ThisForm.pcEscolha = 'INSERIR'

	m.laCampos(1,1) = 'grupoos'
	m.laCampos(1,2) = get_origem
	m.laCampos(2,1) = 'grupods'
	m.laCampos(2,2) = get_destino

	If ThisForm.poDataMgr.ChkRegister('SigCdRps',@m.laCampos)
		Messagebox( 'Grupo Origem , Grupo Destino já Cadastrado!', 48, '' )
		ThisForm.Pagina.Dados.Get_Origem.SetFocus
		Return
	Endif
	Select CrSigCdRps
	Replace cidchaves with fUniqueIds(),;
			Datars    With DateTime(),;
			Usuars    With Usuar

Endif

=DoDefault()

ThisForm.mAtivapagina1

ENDPROC

	ThisForm.mAtivapagina1
Endif

ENDPROC

return( !thisform.pcEscolha $ 'EXCLUIR/CONSULTAR/PROCURAR' )

------------------------------------------------------------
* PROCEDURE Valid
------------------------------------------------------------
PROCEDURE Valid
local lcod,lfan,codcor,xobj,grupo,retorno

m.llRetorno = (IsEmpty(This.Value))

If .Not.m.llRetorno

	m.loLista = CreateObject('fwBuscaExt',Thisform.podatamgr.pnidconn,'SigCdGcr','crSigCdGcr','Codigos',This.Value,'Grupos',1000)
	If .Not.m.loLista.plAchouRegistro
		m.loLista.mAddColuna('Codigos','','Código')
		m.loLista.mAddColuna('Descrs',Replicate('x',60),'Descrição')
		m.loLista.Show()
	EndIf

	this.value = lcod
	this.parent.get_Destino.value = Iif(LastKey()==27 or !used('CrSigCdGcr'),Space(15),crSigCdGcr.codigos)
	ThisForm.Pagina.Dados.Get_dDestino.Value = CrSigCdGcr.Descrs
	This.Refresh
	
	m.llRetorno = (.Not.IsEmpty(This.Value))
	
EndIf
m.llRetorno = .t.

Return(m.llRetorno)

ENDPROC
