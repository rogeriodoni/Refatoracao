PROCEDURE msv_excluir
Private pChave, llOK

pChave= crSigPrTrp.Itras
llOK = .t.

* Eliminando os Registros
Select crSigPrTrp
Go top
Delete For Itras = pChave

Select crSigPrTrv
Go top
Delete For Itras = pChave

* Excluindo os Registros
If llOK
	If Not ThisForm.poDataMgr.Update('crSigPrTrp')
		=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (msv_excluir - 3)')
		llOK = .f.
	EndIf
EndIf
If llOK
	If Not ThisForm.poDataMgr.Update('crSigPrTrv')
		=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (msv_excluir - 4)')
		llOK = .f.
	EndIf
EndIf

If llOK
	ThisForm.poDataMgr.Commit()
Else
	ThisForm.poDataMgr.Rollback()
	Return 0
EndIf
NoDefault
* Fim *********************************************************************************
ENDPROC

* PROCEDURE Init

PROCEDURE Init
LPARAMETERS lcacesso

lcItra = []
lcSqlSigPrTrv = [Select a.*, d.estados as uf, c.descs as munici From SigPrTrv a Inner Join SigCdUfs d On d.ufibges = a.ufibges ] +;
				[Left Join SigCdMun c on c.codigos = a.codibges Where a.Itras = ?lcItra]

With ThisForm
	.poDatamgr = CreateObject('fSqlConector')
	With .poDatamgr
		.AddCursor('SigPrTrp','itras','crSigPrTrp','',ThisForm.pagina.lista.grade)
		.pfSqlTabela(1).pColuna('itras','','','Código',85,.t.)
		.pfSqlTabela(1).pColuna('razaos','','','Razão Social',322,.t.)
		.pfSqlTabela(1).pColuna('tel1s','','','Telefone',120,.t.)	
		.AddCursor('SigPrTrv','CidChaves','crSigPrTrv','','',lcSqlSigPrTrv)
		.AddCursor('SigCdCli','iclis','crSigCdCli','','')
		.Requery('crSigPrTrv')
	EndWith
	
	
	Select crSigPrTrv
	Go top
	Replace all Munici with Nvl(Munici,'')
	
	With Thisform.pagina.Dados.gradEstado
		.RecordSource = 'crSigPrTrv'
		.column1.ControlSource = 'crSigPrTrv.uf'
		.Column2.ControlSource = 'crSigPrTrv.valfretes'
		.Column3.ControlSource = 'crSigPrTrv.pesomins'
		.Column4.ControlSource = 'crSigPrTrv.pesomaxs'
		.Column5.ControlSource = 'crSigPrTrv.Munici'
		.Refresh
	Endwith

	If Not fChecaAcesso([SIGCDTRA],[SENHAFTP])
		ThisForm.Pagina.Dados.Get_senha2.Visible = .f.
		ThisForm.Pagina.Dados.Get_ftpsenha.Visible = .f.
		ThisForm.Pagina.Dados.Say22.Visible = .f.
		ThisForm.Pagina.Dados.Say24.Visible = .f.
	EndIf 


	Select crSigPrTrp
	= DoDefault()	
	.LockScreen = .f.
EndWith
ENDPROC
