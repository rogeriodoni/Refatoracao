LParameters poForm, pnTpBol, pcTitEnvelope, pcEmp, pcDope, pnNum, pnDev, pnVias, pnQBols, pCursor

	Messagebox('Impossivel Efetuar Conexão com Servidor de Banco de Dados.',(0+48),'Erro')
	return(.f.)
EndIf

lcErro = ''
llSair = .F.
With Thisform
	.vcTitEnvelope = Iif(Type('pcTitEnvelope') = 'C', pcTitEnvelope, '')
	.vnTpBol  = Iif(Type('pnTpBol') = 'N', pnTpBol, 0)
	.vcEmp    = Iif(Type('pcEmp')   = 'C', pcEmp, '')
	.vcDope   = Iif(Type('pcDope')  = 'C', pcDope, '')
	.vnNum    = Iif(Type('pnNum')   = 'N', pnNum, 0)
	.vnDev    = Iif(Type('pnDev')   = 'N', pnDev, 0)
	.vnVias   = Iif(Type('pnVias')  = 'N' And pnVias > 0, pnVias, 0)
	.vnQBols  = Iif(Type('pnQBols') = 'N' And pnQBols > 0, pnQBols, 1)
	.vnCursor = pCursor
	
	.getQtde.Value = .vnQBols
	
	Do Case
		Case .vnTpBol = 1
			.Caption = Iif( Type('pcTitEnvelope') = 'C', pcTitEnvelope, ;
					   'Impressão de Protocolo de Cheque' )
			llSair = fChecaAcesso('SIGCHE','PROTSEMPRE') And ( Type('poForm.Op_Escolha') = 'C') And ;
					 poForm.Op_Escolha = 'INSERIR'
			
		Case InList( .vnTpBol, 2, 3, 4, 5, 7, 9 )
			.Caption = 'Impressão de Boleto'

		Case .vnTpBol = 6
			.Caption = 'Impressão de Protocolo de Cartão de Crédito'
			
		Case .vnTpBol = 8
			.Caption = 'Impressão de Boleto de Conserto'
			
		Case .vnTpBol = 12

			.Caption = 'Impressão de Boleto de Vendas'

			.Caption = 'Impressão de Duplicata/Carne'

	*** Mena - 03/11/2014 - incluido check box para escolher duplicata ou carnê

	Thisform.chkAgrupa.Visible = (.vnTpBol = 3)

	ThisForm.chkObss.Visible = !(.vnTpBol = 18)	
	ThisForm.chkCarne.Visible = (.vnTpBol = 18)	
EndWith

Create Cursor crImpre (impres c(50))
If lnPrinters # 0
	For lnI = 1 to lnPrinters
		Insert Into crImpre	(impres) Values(Upper(laPrinters(lnI, 1)))
*!*			laPrinters( lnI, 2 ) = laPrinters( lnI, 1 )
*!*			laPrinters( lnI, 1 ) = Padr(Upper( laPrinters( lnI, 1 ) ),30)
	EndFor
*!*		=ASort( laPrinters )
EndIf

lcSql = [Select b.* From SigSyImp a, SigCdmp b Where a.UsuAcess = ']+Usuar+[' And a.Cimps = b.Impres ]+;
		[Union all ]+;
		[Select c.* From SigCdAcG a, SigSyImp b, SigCdmp c Where a.Usuarios=']+Usuar+[' ]+;
		[And a.Grupos = b.GrAcess And b.Cimps = c.Impres]

If poForm.podatamgr.sqlexecute(lcSql,'CrTmpCimp') <= 0

EndIf

Select CrTmpCimp
If Reccount() > 0
	Select Distinct Impres, nTpImpres, TpBarra, MaxCols ;
	  From crTmpCimp ;
	 Order By Impres ;
	  Into Cursor crSigCdmp
Endif

*!*	lnImpres = 1
*!*	lnCimp = 0
*!*	If lnPrinters # 0
*!*		Select CrSigCdmp
*!*		Scan For ( Empty( CrSigCdmp.nTpImpres ) Or ( CrSigCdmp.nTpImpres = 1 ) ) And ;
*!*				 ( CrSigCdmp.TpBarra = 0 ) And ( CrSigCdmp.MaxCols > 39 )
*!*			Dimension laImpOk( lnImpres )
*!*			lnOk = AScan( laPrinters, ( Upper( CrSigCdmp.Impres ) ) )
*!*			If lnOk # 0
*!*				laImpOk( lnImpres ) = laPrinters( lnOk + 1 )
*!*				lnImpres = lnImpres + 1
*!*			EndIf
*!*			lnCImp = lnCImp + 1
*!*		EndScan
*!*	Else
*!*		lcErro = 'Seu Computador Não Apresenta Impressora Configurada. Contate o Suporte Técnico.'
*!*	EndIf
*!*	lnImpres = lnImpres - 1

If Empty(lcErro) And ( Thisform.vnTpBol = 3 ) And ( Thisform.vnVias = 0 )
	lcErro = 'Boleto Tipo 3 Sem Configuração do Tipo de Via. Contate o Suporte Técnico.'
EndIf

*** mena - 05/03/2015 - prepara cursor com as impressoras para colocar num combobox ***
SELECT PADR(ALLTRIM(a.Impres),15)+' '+ALLTRIM(b.impres) as IDupla, b.impres, a.impres as impresS from crSigCdmp a, crImpre b ;
where (ALLTRIM(UPPER(b.impres)) like '%'+ALLTRIM(UPPER(a.impres))+'%' or ALLTRIM(UPPER(a.impres)) like '%'+ALLTRIM(UPPER(b.impres))+'%') ;
group by b.impres, a.impres into cursor crImp READWRITE

*!*	SELECT b.Impres from crSigCdmp a, crImpre b where a.nTpImpres=1 and (ALLTRIM(UPPER(a.impres)) like '%'+ALLTRIM(UPPER(b.impres))+'%' ;
*!*			or ALLTRIM(UPPER(b.impres)) like '%'+ALLTRIM(UPPER(a.impres))+'%') group by b.impres order by 1 into cursor crImpreV

SELECT * FROM crImp order by 1 into cursor crImpreV

SELECT crImpreV
GO top

*** mena - 05/03/2015 - a lista de impressoras sairam dos radio para o combobox *** 
If lnCImp = 0

*!*			.Opt_Impressora.Visible = .F.
*!*			.Lbl_Impressora.Visible = .F.
*!*	*!*			.Lbl_Erro.Top = .Opt_Impressora.Top

		EndIf
	EndWith

PROCEDURE Cmd_Imprimir.Click
Local lnTpBols, lcTitEnv, lcEmps, lcDopes, lnNumes, lnValDevs, lnTpVias, lnQtdBols, lcCursor, llAgrupa

With Thisform
	lnTpBols  = .vnTpBol
	lcTitEnv  = .vcTitEnvelope
	lcEmps    = .vcEmp
	lcDopes   = .vcDope
	lnNumes   = .vnNum
	lnValDevs = .vnDev
	lnTpVias  = .vnVias
	lnQtdBols = .vnQBols
	*** mena - 05/03/2015 - pega o nome da impressora que está no combobox *** 
	lnImpres  = RECNO('crImpreV')
	lcImp     = crImpreV.IDupla	&& impres
*!*		lnImpres  = .Opt_Impressora.Value
*!*		lcImp     = .Opt_Impressora.Buttons(lnImpres).Tag
	lcCursor  = .vnCursor
	llObs     = IIF(lnTpbols=18,.chkCarne.Value,.chkObss.Value )
	llAgrupa  = (.chkagrupa.Value)
	.btnReport.Cmd_Sair.Enabled = .T.
EndWith

Do SigBlBol With lnTpBols, lcTitEnv, lcEmps, lcDopes, lnNumes, lnValDevs, lnTpVias, lnQtdBols, lcImp, lcCursor, llObs, llAgrupa

* Parâmetros :
* 01. nTpbol, Tipo de Boleto
* 02. TitEnvelope, Título
* 03. pEmp, Empresa
* 04. pDop, Nome da Operação de Estoque / Operação de Produção
