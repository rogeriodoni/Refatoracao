PROCEDURE alterar_rpt
**
**Thisform.Pagina.ActivePage = 3

Local lcNmRPT As String, lcRPT_ID As String, lcSqlQuery As String, lcArqReport As String, lcLblCaption As String

If Used( "crSIGCDRLC" ) And RecCount( "crSIGCDRLC" ) > 0
	** Ok, há relatórios na Lista
Else
	Select crSIGCDRLC
	MessageBox( "Selecione um Relatório para " + Iif( Thisform.pcEscolha = [ALTERAR], "Alterar", "Visualizar" ), 48, "Atenção" )
	Return
EndIf

lcNmRPT = AllTrim( crSIGCDRLC.NmReport )

If Used( "Crs_RPT_C" )
	** Ok, Cursor com os Relatórios Carregado
Else
	Select crSIGCDRLC
	MessageBox( "Problemas no cursor Crs_RPT_C - Alterar_RPT()", 48, "Atenção" )
	Return
EndIf

Select Crs_RPT_C
Locate For Upper( Crs_RPT_C.NmReport ) = Upper( lcNmRPT )

If Found( "Crs_RPT_C" ) 
	** Ok, Encontrei o Cabeçalho do Relatório
Else
	Select crSIGCDRLC
	MessageBox( "Não foi Encontrado o cabeçalho do Rel. - Alterar_RPT() " +Chr(13)+ lcNmRPT, 48, "Atenção" )
	Return
EndIf

PROCEDURE calc_tamcampo
**
Lparameters tcTpCampo As String, tcFonte As String

Local lnTam As Integer, lcTipo As String, lnFieldWidth As Number
Local lcFontName As String, lnFontSize As Integer, lcFontStyle As String

lcTipo = Upper( Left( AllTrim( tcTpCampo ), 1 ) )
tcTpCampo = StrTran( tcTpCampo, lcTipo, "" )
tcTpCampo = StrTran( tcTpCampo, "(", "" )
tcTpCampo = StrTran( tcTpCampo, ")", "" )

lnTam = Int( Val( tcTpCampo ) )

lcFontName = Left( tcFonte, At( ",", tcFonte, 1 )-1 )
tcFonte    = StrTran( tcFonte, lcFontName+",", "" )

lnFontSize = Left( tcFonte, At( ",", tcFonte, 1 )-1 )
tcFonte    = StrTran( tcFonte, lnFontSize+",", "" )

lcFontStyle = tcFonte
lnFontSize  = Val( lnFontSize )


Local lnFACTOR As Number, lcAVG_CHAR As String, lnMax As Number, lnAvg As Number
* lnFACTOR is the number of report units per pixel. It's calculated as report
* units per inch (10,000) divided by pixels per inch (96). 10000/96 = 104.166.
  lnFACTOR = 104.166
* 
lnMax = FontMetric( 7, lcFontName, lnFontSize ) 	&& 7 = Maximum character width in pixels
lnAvg = FontMetric( 6, lcFontName, lnFontSize ) 	&& 6 = Average character width in pixels

If lnMax - lnAvg > 1
	* lcAVG_CHAR is the "average" character in a font used to determine object sizing.
	lcAVG_CHAR = lcFontStyle 
	lnFACTOR = TxtWidth( lcAVG_CHAR, lcFontName, lnFontSize ) * lnAvg * lnFACTOR
Else 
	lnFACTOR = lnMax * lnFACTOR
EndIf 


If lcTipo $ "N,F,I,B,Y"
	lnFieldWidth = Max( lnTam, 18 )
Else 
	lnFieldWidth = lnTam
	Do Case
	Case lcTipo = 'D'
		lnFieldWidth = Iif( lnFieldWidth < 15, 15, lnFieldWidth )

	Case lcTipo = 'T'
		lnFieldWidth = Iif( lnFieldWidth < 20, 20, lnFieldWidth )

	EndCase

EndIf 

**lnFieldWidth = lnFieldWidth * lnFACTOR

Return( lnFieldWidth )


ENDPROC

* PROCEDURE ctrl_v
------------------------------------------------------------
PROCEDURE ctrl_v
** 
Local lcChr13 As String, lcChr10 As String
lcChr13 = Chr(13)
lcChr10 = Chr(10)

Store "" To _oRTF.cLetra, _oRTF.cPalavra, _oRTF.cCMD 
Store 0 To _oRTF.nSelStart, _oRTF.nSelPosA, _oRTF.nSelPosI, _oRTF.nSelPosF
Store 0 To _oRTF.nLineI, _oRTF.nLineFromChar

_oRTF.lMudaCor 	= .F.
_oRTF.nForeColor = 0
_oRTF.nRet		= 0

For _oRTF.nSelPosF = 0 To Len( This.Pagina.Dados.pageframe1.Page1.OleRTF_SqlCMD.Text )

	_oRTF.cLetra = SubStr( This.Pagina.Dados.pageframe1.Page1.OleRTF_SqlCMD.Text, _oRTF.nSelPosF, 1 )

	If Len( _oRTF.cPalavra ) < 2 And ( _oRTF.cLetra = lcChr10 OR _oRTF.cLetra = lcChr13 )
		_oRTF.nSelPosI = _oRTF.nSelPosF
		Loop
	EndIf

	If _oRTF.cLetra == " "
	
		_oRTF.cPalavra = StrTran( _oRTF.cPalavra, lcChr13, "" )
		_oRTF.cPalavra = StrTran( _oRTF.cPalavra, lcChr10, "" )
		_oRTF.lMudaCor 	= .F.
		_oRTF.nForeColor = 0
		_oRTF.nRet		= 0

		ThisForm.Color_CMD()

		This.Pagina.Dados.pageframe1.Page1.OleRTF_SqlCMD.SelStart  = _oRTF.nSelPosI
		This.Pagina.Dados.pageframe1.Page1.OleRTF_SqlCMD.SelLength = _oRTF.nLen + Iif( _oRTF.nSelPosI=0, 0, 1 )
		This.Pagina.Dados.pageframe1.Page1.OleRTF_SqlCMD.SelColor  = _oRTF.nForeColor 

		Store "" To _oRTF.cLetra, _oRTF.cPalavra, _oRTF.cCMD 
		Store _oRTF.nSelPosF To _oRTF.nSelStart, _oRTF.nSelPosI

	EndIf
	
	_oRTF.cPalavra = _oRTF.cPalavra + _oRTF.cLetra

Next _oRTF.nSelPosF

This.Pagina.Dados.pageframe1.Page1.OleRTF_SqlCMD.SelLength = 0
This.Pagina.Dados.pageframe1.Page1.OleRTF_SqlCMD.Refresh()
This.Pagina.Dados.pageframe1.Page1.OleRTF_SqlCMD.SetFocus()

ENDPROC

* PROCEDURE detalhar_fonte
------------------------------------------------------------
PROCEDURE detalhar_fonte
** 
Parameters tcFonte As String, tcFontName As String, tnFontSize As Integer, tcFontStyle As String, tnFontStyle As Integer

If Type( "tnFontStyle" ) # "N"
	tnFontStyle = 0
EndIf

tcFontName = Left( tcFonte, At( ",", tcFonte, 1 )-1 )
tcFonte    = AllTrim( StrTran( tcFonte, tcFontName+",", "" ) )

tnFontSize = Left( tcFonte, At( ",", tcFonte, 1 )-1 )
tcFonte    = AllTrim( StrTran( tcFonte, tnFontSize+",", "" ) )

tcFontStyle = tcFonte
tnFontSize  = Val( tnFontSize )

**
Do Case
Case tcFontStyle = "N"		&& Normal
	tnFontStyle = 0

Case tcFontStyle = "B"		&& Bold
	tnFontStyle = 1

Case tcFontStyle = "I"		&& Italic
	tnFontStyle = 2

Case tcFontStyle = "BI"		&& Bold And Italic
	tnFontStyle = 3

Case tcFontStyle = "U"		&& Underlined
	tnFontStyle = 4

Case tcFontStyle = "BU"		&& Bold And Underlined		-  These values may be added to one another for combined effects. For example, a value of 5 = Bold and Underlined.
	tnFontStyle = 5

Case tcFontStyle = "S"		&& Strikethrough
	tnFontStyle = 128

EndCase

Return

ENDPROC

* PROCEDURE excluir_rpt
------------------------------------------------------------
PROCEDURE excluir_rpt
**
Local lcNmRPT As String, lcRPT_ID As String, lnConnHnd As Integer

If Used( "crSIGCDRLC" ) And RecCount( "crSIGCDRLC" ) > 0
	** Ok, há relatórios na Lista
Else
	Select crSIGCDRLC
	MessageBox( "Selecione um Relatório para Excluir", 48, "Atenção" )
	Return
EndIf
This.poDataMgr.Commit()

lcNmRPT = AllTrim( crSIGCDRLC.NmReport )

If Used( "Crs_RPT_C" )
	** Ok, Cursor com os Relatórios Carregado
Else
	Select crSIGCDRLC
	MessageBox( "Problemas no cursor Crs_RPT_C - Excluir_RPT()", 48, "Atenção" )
	Return
EndIf


Select Crs_RPT_C
Locate For Upper( Crs_RPT_C.NmReport ) = Upper( lcNmRPT )

If Found( "Crs_RPT_C" ) 
	** Ok, Encontrei o Cabeçalho do Relatório
Else
	Select crSIGCDRLC
	MessageBox( "Não foi Encontrado o cabeçalho do Rel. - Excluir_RPT() " +Chr(13)+ lcNmRPT, 48, "Atenção" )
	Return
EndIf

If MessageBox( "Confirma Exclusão do Relatório:" + Chr(13) + lcNmRPT, 32+4+256, "Atenção" ) == 7 	&& 32=Question Mark / 4=Yes and No Buttons / 256=2nd Button Default / 7=No
	** No Delete
	Return
EndIf


lcRPT_ID  = AllTrim( Crs_RPT_C.RPT_ID )
lnConnHnd = ThisForm.nConnHnd 

If Thisform.podatamgr.SqlExecute([Delete From SIGCDRLC Where RPT_ID = ?lcRPT_ID]) <= 0
	AError(laError)
	MessageBox( "Problemas na Deleção do ID " + lcRPT_ID +" SIGCDRLC - Excluir_RPT()." +Chr(13)+ laError[2], 48, "Atenção" )
	Return
EndIf
*!*	SQLCommit( lnConnHnd )

If Thisform.podatamgr.SqlExecute([Delete From SIGCDRLD Where RPT_ID = ?lcRPT_ID]) <= 0
	AError(laError)
	MessageBox( "Problemas na Deleção do ID " + lcRPT_ID +" SIGCDRLD - Excluir_RPT()." +Chr(13)+ laError[2], 48, "Atenção" )
	Return
EndIf
*!*	SQLCommit( lnConnHnd )

If Thisform.podatamgr.SqlExecute([Delete From SIGCDPRG Where PkChaves = '3RPTCUSTOM] + lcRPT_ID + ['] ) <= 0
	AError(laError)
	MessageBox( "Problemas na Deleção do ID " + lcRPT_ID +" SIGCDPRG - Excluir_RPT()." +Chr(13)+ laError[2], 48, "Atenção" )
	Return
EndIf
*!*	SQLCommit( lnConnHnd )

If Thisform.podatamgr.SqlExecute([Delete From SIGCDACU Where Programas = ?lcRPT_ID]) <= 0
	AError(laError)
	MessageBox( "Problemas na Deleção do ID " + lcRPT_ID +" SIGCDACU - Excluir_RPT()." +Chr(13)+ laError[2], 48, "Atenção" )
	Return
EndIf
SQLCommit( lnConnHnd )


*!*	This.Preenche_Lista()

*******************************************************
** Buscando a Tabela de Cabeçalho dos Reports
**
If Used( "CrsTmp_RPT" )
	Use IN CrsTmp_RPT
EndIf

If Thisform.podatamgr.SqlExecute([Select * From SIGCDRLC], [CrsTmp_RPT]) <= 0
	AError(laError)
	Select crSIGCDRLC
	MessageBox( "Problemas no Select do cursor CrsTmp_RPT:"+Chr(13)+ laError[2], 48, "Atenção" )
	Return
EndIf

If Used( "CrsTmp_RPT" )
	Select * From CrsTmp_RPT Into Cursor Crs_RPT_C ReadWrite

	Use IN CrsTmp_RPT

Else

	Select crSIGCDRLC
	MessageBox( "Problemas no Select do cursor CrsTmp_RPT (2)", 48, "Atenção" )
	Return

EndIf

Select crSIGCDRLC

ThisForm.Refresh()

MessageBox( "Deleção efetuada com Sucesso!", 64, "Atenção" )

ENDPROC

* PROCEDURE retorna_fontstyle
------------------------------------------------------------
PROCEDURE retorna_fontstyle
**
Lparameters tvFontStyle As Variant 

Local lcType As String, lvFontStyle As Variant 
lcType = Type( "tvFontStyle" )

Do Case
Case (lcType = "C" And tvFontStyle = "N") OR (lcType = "N" And tvFontStyle = 0)		&& Normal
	lvFontStyle = Iif( lcType = "C", 0, "N" )

Case (lcType = "C" And tvFontStyle = "B") OR (lcType = "N" And tvFontStyle = 1) 	&& Bold
	lvFontStyle = Iif( lcType = "C", 1, "B" )

Case (lcType = "C" And tvFontStyle = "I") OR (lcType = "N" And tvFontStyle = 2) 	&& Italic
	lvFontStyle = Iif( lcType = "C", 2, "I" )

Case (lcType = "C" And tvFontStyle = "BI") OR (lcType = "N" And tvFontStyle = 3) 	&& Bold And Italic
	lvFontStyle = Iif( lcType = "C", 3, "BI" )

Case (lcType = "C" And tvFontStyle = "U") OR (lcType = "N" And tvFontStyle = 4) 	&& Underlined
	lvFontStyle = Iif( lcType = "C", 4, "U" )

Case (lcType = "C" And tvFontStyle = "BU") OR (lcType = "N" And tvFontStyle = 5) 	&& Bold And Underlined		-  These values may be added to one another for combined effects. For example, a value of 5 = Bold and Underlined.
	lvFontStyle = Iif( lcType = "C", 5, "BU" )

Case (lcType = "C" And tvFontStyle = "S") OR (lcType = "N" And tvFontStyle = 128) 	&& Strikethrough
	lvFontStyle = Iif( lcType = "C", 128, "S" )

EndCase

Return( lvFontStyle )

ENDPROC

* PROCEDURE seleciona_fonte
------------------------------------------------------------
PROCEDURE seleciona_fonte
** 
Parameters tnLen, toCmpFonte As Object, toCmpTamFt As Object

Local lcFonte As String, lcFonteSelec As String, lcFontName As String, lnFontSize As Integer, lcFontStyle As String
Local lnTamSp As Number, lnTamAt As Number, lnMtrcAt As Number, lnTamNv As Number, lnMtrcNv As Number

lcFonte = AllTrim( toCmpFonte.Value )

lcFontName = Left( lcFonte, At( ",", lcFonte, 1 )-1 )
lcFonte    = StrTran( lcFonte, lcFontName+",", "" )

lnFontSize = Left( lcFonte, At( ",", lcFonte, 1 )-1 )
lcFonte    = StrTran( lcFonte, lnFontSize+",", "" )

lcFontStyle = lcFonte
lnFontSize  = Val( lnFontSize )

lnTamSp = Iif( tnLen=0, 0, toCmpTamFt.Value )

lnTamAt  = tnLen
lnMtrcAt = FontMetric( 7, lcFontName, lnFontSize, lcFontStyle )

lcFonteSelec = Iif(Empty(lcFontName),GetFont(),GetFont( lcFontName, lnFontSize, lcFontStyle ))

If IsNull( lcFonteSelec ) Or Empty( lcFonteSelec )
	Return
EndIf

lcFonte = AllTrim( lcFonteSelec )

lcFontName = Left( lcFonte, At( ",", lcFonte, 1 )-1 )
lcFonte    = StrTran( lcFonte, lcFontName+",", "" )

lnFontSize = Left( lcFonte, At( ",", lcFonte, 1 )-1 )
lcFonte    = StrTran( lcFonte, lnFontSize+",", "" )

lcFontStyle = lcFonte
lnFontSize  = Val( lnFontSize )


lnMtrcNv = FontMetric( 7, lcFontName, lnFontSize, lcFontStyle )
lnTamNv  = Round( ( lnTamAt / lnMtrcAt ) * lnMtrcNv , 0 )

If tnLen > 0
	toCmpTamFt.Value = Max( lnTamSp, lnTamAt, lnTamNv )
	toCmpTamFt.Refresh()
	toCmpTamFt.Parent.Refresh()
EndIf

toCmpFonte.Value = lcFonteSelec
toCmpFonte.Refresh()
toCmpFonte.Parent.Refresh()

ENDPROC

*** ActiveX Control Event ***
LPARAMETERS tnkeyCode, Shift

*!* CTRL+A, CTRL+C, CTRL+V, CTRL+X, CTRL+Z 
If Shift = 2 AND InList( tnkeyCode, 65, 67, 86, 88, 90 )
	**Return
EndIf


If tnkeyCode = 13	&& 	13 -> Enter

	If This.SelColor # 0

		Store "" To _oRTF.cLetra, _oRTF.cPalavra, _oRTF.cCMD 
		Store This.SelStart To _oRTF.nSelStart, _oRTF.nSelPosA, _oRTF.nSelPosI, _oRTF.nSelPosF
		Store This.GetLineFromChar( _oRTF.nSelPosA ) To _oRTF.nLineI, _oRTF.nLineFromChar

		Local i As Integer, lnLen As Integer

		_oRTF.nForeColor = 0		&& Preto

		If _oRTF.nSelPosF > 0

			For i = _oRTF.nSelPosI To Len( This.Text )

*!*					This.SelStart = _oRTF.nSelPosF
*!*					This.SelLength = 1
*!*					_oRTF.cLetra = This.SelText 

				_oRTF.cLetra = SubStr( This.Text, _oRTF.nSelPosF, 1 )

				_oRTF.nLineFromChar = This.GetLineFromChar( _oRTF.nSelPosF )
				
				If _oRTF.cLetra == " " OR _oRTF.cLetra == "(" OR _oRTF.cLetra == "[" OR _oRTF.cLetra == "{" OR _oRTF.nLineI # _oRTF.nLineFromChar 
					_oRTF.nSelPosF = _oRTF.nSelPosF -1
					Exit
				EndIf
				
				_oRTF.nSelPosF = _oRTF.nSelPosF +1

			Next i

			If _oRTF.nSelPosF > _oRTF.nSelPosI

				_oRTF.nSelPosI = _oRTF.nSelPosF

				For i = _oRTF.nSelPosF To 0 Step -1

*!*						This.SelStart = _oRTF.nSelPosI
*!*						This.SelLength = 1
*!*						_oRTF.cLetra = This.SelText 

					_oRTF.cLetra = SubStr( This.Text, _oRTF.nSelPosI, 1 )

					If _oRTF.cLetra = " "
						_oRTF.nSelPosI = _oRTF.nSelPosI +1
						Exit
					EndIf
					
					_oRTF.cPalavra = _oRTF.cLetra + _oRTF.cPalavra 

					If _oRTF.nSelPosI > 0
						_oRTF.nSelPosI = _oRTF.nSelPosI -1
					EndIf

				Next i
				
				This.SelStart  = _oRTF.nSelPosI
				This.SelLength = ( _oRTF.nSelPosF - _oRTF.nSelPosI ) +1
				This.SelColor  = 0
				
				This.SelStart  = _oRTF.nSelPosA
				This.SelLength = 0
			EndIf			
		EndIf
	EndIf
EndIf

PROCEDURE KeyUp
*** ActiveX Control Event ***
LPARAMETERS tnkeyCode, shift

** Wait Window "tnkeyCode-> " + TransForm( tnkeyCode ) + Chr(13) + "shift-> " + TransForm( shift ) NoWait

*!* tnKeyCode -> table lists the return values for INKEY()
*!*  8 -> BackSpace
*!* 13 -> Enter
*!* 16 -> Shift
*!* 17, 21 e 22 -> CRTL
*!*	32 -> Space
*!*	33 -> PgUp
*!*	34 -> PgDown
*!*	35 -> End
*!*	36 -> Home
*!*	37 -> Left
*!*	38 -> Up
*!*	39 -> Right
*!*	40 -> Down
*!* 46 -> Del
*!* 119-> F8
*!* 192-> '

*!* 57 -> (
*!* 48 -> )
*!* 220-> ]
*!* 221-> [

If InList( tnkeyCode, 13, 16, 17, 21, 22, 33, 34, 35, 36, 37, 38, 39, 40, 119, 48,57,220,221 )
	Return
EndIf


*!* CTRL+A, CTRL+C, CTRL+V, CTRL+X, CTRL+Z 
If Shift = 2 AND InList( tnkeyCode, 65, 67, 86, 88, 90 )

	Do Case
	Case tnkeyCode = 65	&& CTRL+A
		Return	
	Case tnkeyCode = 86	&& CTRL+V
		ThisForm.Ctrl_V()
		Return
	EndCase

	**Return
EndIf

Store "" To _oRTF.cLetra, _oRTF.cPalavra, _oRTF.cCMD 
Store This.SelStart To _oRTF.nSelStart, _oRTF.nSelPosA, _oRTF.nSelPosI, _oRTF.nSelPosF
Store This.GetLineFromChar( _oRTF.nSelPosA ) To _oRTF.nLineI, _oRTF.nLineFromChar

_oRTF.lMudaCor 	= .F.
_oRTF.nForeColor = 0
_oRTF.nRet		= 0

If tnkeyCode = 8 And _oRTF.nSelPosA > 0 And _oRTF.nSelPosA < _oRTF.nSelPosF
	_oRTF.nSelPosA = _oRTF.nSelPosA -1
EndIf

If tnkeyCode = 32

	If _oRTF.nSelStart > 1
		Store _oRTF.nSelStart-1 To _oRTF.nSelPosA, _oRTF.nSelPosI, _oRTF.nSelPosF
	EndIf

Else

	If tnkeyCode = 192 	&& ' aspa simples
		
	EndIf
	

	For _oRTF.nSelPosF = (_oRTF.nSelPosA+1) To Len( This.Text )

*!*			This.SelStart  = _oRTF.nSelPosF
*!*			This.SelLength = 1
*!*			_oRTF.cLetra = This.SelText 

		_oRTF.cLetra = SubStr( This.Text, _oRTF.nSelPosF, 1 )

		_oRTF.nLineFromChar = This.GetLineFromChar( _oRTF.nSelPosF )

		If _oRTF.cLetra == " "
			_oRTF.nSelPosF = _oRTF.nSelPosF -1
			Exit
		EndIf
		
		_oRTF.cPalavra = _oRTF.cPalavra + _oRTF.cLetra

	Next _oRTF.nSelPosF
EndIf

For _oRTF.nSelPosI = _oRTF.nSelPosA To 0 Step -1

*!*		This.SelStart = _oRTF.nSelPosI
*!*		This.SelLength = 1
*!*		_oRTF.cLetra = This.SelText 

	_oRTF.cLetra = SubStr( This.Text, _oRTF.nSelPosI, 1 )

	_oRTF.nLineFromChar = This.GetLineFromChar( _oRTF.nSelPosI )

	If _oRTF.nSelPosI > 0 And ( _oRTF.cLetra == " " OR _oRTF.cLetra == "(" OR _oRTF.cLetra == "[" OR _oRTF.cLetra == "{" )
		Exit
	EndIf

	If _oRTF.nLineFromChar # _oRTF.nLineI 
		_oRTF.nSelPosI = _oRTF.nSelPosI +1
		Exit
	EndIf

	_oRTF.cPalavra = _oRTF.cLetra + _oRTF.cPalavra

Next _oRTF.nSelPosI
_oRTF.nSelPosI = Iif( _oRTF.nSelPosI < 0, 0, _oRTF.nSelPosI )


ThisForm.Color_CMD()


This.SelStart  = _oRTF.nSelPosI
This.SelLength = _oRTF.nLen + Iif( _oRTF.nSelPosI=0, 0, 1 )
This.SelColor  = _oRTF.nForeColor 


*=======================================================================================
This.SelStart  = _oRTF.nSelStart 
This.SelLength = 0
This.SelColor  = 0


*!*	Wait Window "tnkeyCode-> " + TransForm( tnkeyCode ) +Chr(13)+ ;
*!*				"lnSelPosA-> " + TransForm( lnSelPosA ) +Chr(13)+ ;
*!*				"lnSelPosI-> " + TransForm( lnSelPosI ) +Chr(13)+ ;
