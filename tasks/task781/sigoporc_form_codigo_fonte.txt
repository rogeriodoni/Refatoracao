* PROCEDURE calc_valite
------------------------------------------------------------
PROCEDURE calc_valite
Lparameters pArqItem, pArqCab

Local lcpArqItem, lcpArqCab, lcMoeVs
Local lnCasas, lnValIpi, lnNParcs, llAtualiza, _xPrecoParc, _xPrecoNovo, llMoePreco, lcMoeFormula, llChkCot

If Type('pArqItem')='C'
	lcpArqItem = Alltrim(pArqItem)
Else
	lcpArqItem = 'crSigMvOrc'
EndIf

If Type('pArqCab')='C'
	lcpArqCab = Alltrim(pArqCab)
Else
	lcpArqCab = 'TprMvCab'
EndIf

Select ( lcpArqItem )

lcQuery = [Select a.Cpros, a.Moevs, a.Moecs, a.Moedas, a.ClFiscals, a.Cgrus, a.Linhas, a.Colecoes, a.CUnis, ] + ;
				 [a.CUniPs, a.CProEqs, a.vUltComps, a.CFtios, a.Sgrus, a.Mercs, a.CodFinP, a.OrigMercs, ] + ;
				 [a.SitTrIcms, a.AliqIpis, a.mUltComps, b.CfgGerGprs, a.moecusfs ] + ;
		    [From SigCdPro a, SigCdGrp b ] + ;
		    [Where a.CPros = '] + &lcpArqItem..Cpros + [' And ] + ;
		    	  [a.CGrus = b.CGrus]

If (ThisForm.ParentForm.poDataMgr.SqlExecute(lcQuery, [crTmpPro]) < 1)
	=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (crTmpPro)])
	Return 0
EndIf

Select CrTmpPro
Go Top In CrTmpPro
If Empty(CrTmpPro.SitTrIcms)
	Replace SitTrIcms With [00] In CrTmpPro
EndIf
If Empty(CrTmpPro.OrigMercs)
	Replace OrigMercs With [0] In CrTmpPro
EndIf
If Empty(CrTmpPro.ClFiscals)
	Replace ClFiscals With [0] In CrTmpPro
EndIf

lcWhere = [Select Arreds, TipoEstos, Tams, Cores, Embs, chkInstalas From SigCdGrp Where Cgrus = ']+CrTmpPro.Cgrus+[']
lnErro = ThisForm.ParentForm.poDataMgr.SqlExecute(lcWhere,'crSigCdGrp')
If lnErro < 1
	MessageBox('Favor reinicializar o processo.',16,'Falha na Conexão (crSigCdGrp)')
	Return .f.
EndIf

If InList(crSigCdOpe.PrecoPs,8,9)
	=fCustoMedio(&lcpArqCab..Emps,CrTmpPro.Cpros,&lcpArqCab..Datas,ThisForm.ParentForm.poDataMgr,(crSigCdOpe.PrecoPs=8))
EndIf

lcMoeVs = CrTmpPro.Moecs
If InList(crSigOpCdd.chkIfors,1,2)
	_Ct = Iif(crSigOpCdd.ChkIfors=1,TprMvCab.ContaOs,TprMvCab.Contads)
	lcQuery = [Select Moevs,Pvens From SigPrFnc Where Cpros = ']+CrTmpPro.Cpros+[' And Ifors = ']+_Ct+[']
	=ThisForm.ParentForm.poDataMgr.Sqlexecute(lcQuery,'CrTmpFpro')
	If Not Eof('CrTmpfPro') And CrTmpfPro.pvens > 0 and Not Empty(CrTmpfPro.Moevs)
		lcMoeVs   = crtmpfPro.Moevs
	EndIf
EndIf

lcMoeFormula = Iif( crSigCdOpe.PrecoPs = 1, PadR( CrTmpPro.MoeVs, 10 ), ;
			   Iif( crSigCdOpe.PrecoPs = 2, PadR( lcMoeVs, 10 ), ;
			   Iif( crSigCdOpe.PrecoPs = 5, Padr( CrTmpPro.Moedas,10 ), ;
			   Iif( crSigCdOpe.PrecoPs = 7, Padr( CrTmpPro.mUltComps,10),;
			   Iif( crSigCdOpe.PrecoPs = 8, Padr( TmpCusMed.Moedas,10), .F. )))))

If Empty( &lcpArqItem..Utilizas ) And ( Not Empty( &lcpArqItem..Cpros ) )
	Select &lcpArqItem.

	lnCasas   = Iif( crSigCdOpe.CupFis = 1 And crSigCdOpe.Casas > 2, 2, crSigCdOpe.Casas )
	xTabela = &lcpArqCab..Tabds

	_xMoeOper  = crSigCdOpe.Cmoes &&Iif(Empty(crSigCdOpe.CMoes),&lcpArqCab..MoeIts,crSigCdOpe.Cmoes)
	llMoePreco = .F.
	If InList(crSigCdOpe.PrecoAuts,1,3)
		If !Empty( &lcpArqCab..LPrecos ) 
			lcwhere = [Select * From SigCdLpi Where Cpros = ']+&lcpArqItem..Cpros+[' And LPrecos = ']+&lcpArqCab..lPrecos+[']
			lnErro = ThisForm.ParentForm.poDataMgr.SqlExecute(lcWhere,'crSigCdLpi')
			If lnErro < 1
				MessageBox('Favor reinicializar o processo.',16,'Falha na Conexão (crSigCdLpi 4)')
				Return .f.
			EndIf
			Select crSigCdLpi
			Go Top
			If Reccount() > 0 And (IsEmpty(crSigCdLpi.VencFs) Or BetWeen(&lcpArqCab..Datas,crSigCdLpi.VencIs, crSigCdLpi.VencFs))
				_xMoePreco = crSigCdLpi.MoeVs
				llMoePreco = .T.
			EndIf
		EndIf
	EndIf
		
	If Not llMoePreco
		_xMoePreco = ''
		If InList(crSigCdOpe.PrecoPs, 1, 9) &&crSigCdOpe.PrecoPs = 1 && Richard em 12/03/2014
			_xMoePreco = CrTmpPro.MoeVs
		Else
			If InList(crSigCdOpe.PrecoPs,2,11)
				&&_xMoePreco = CrTmpPro.MoeCs
				_xMoePreco = IIF(crSigCdOpe.PrecoPs=2, CrTmpPro.MoeCs, CrTmpPro.Moecusfs)	&& mena - 21/03/2014 pega moeda correspondente (flag = 11 -> total custo)
				If InList(crSigOpCdd.chkIfors,1,2)
					_Ct = Iif(crSigOpCdd.ChkIfors=1,TprMvCab.ContaOs,TprMvCab.Contads)
					lcQuery = [Select Moevs,Pvens From SigPrFnc Where Cpros = ']+CrTmpPro.Cpros+[' And Ifors = ']+_Ct+[']
					=ThisForm.ParentForm.poDataMgr.Sqlexecute(lcQuery,'CrTmpFpro')
					If Not Eof('CrTmpfPro') And CrTmpfPro.pvens > 0 and Not Empty(CrTmpfPro.Moevs)
						_xMoePreco = crtmpfPro.MoeVs
					EndIf
				EndIf
				lcMoeda = _xMoePreco
				=ThisForm.ParentForm.poDataMgr.Cursorquery('SigCdMoe','crSigCdMoe','Cmoes',_xMoePreco)
				If Not Empty(crSigCdMoe.MoeQs)
					_xMoePreco = crSigCdMoe.MoeQs
				EndIf
			Else
				If InList(crSigCdOpe.PrecoPs, 5, 10)  && Richard em 13/03/2014 - Incluido o tipo 10
					_xMoePreco = CrTmpPro.Moedas
				Else
					If crSigCdOpe.PrecoPs = 7
						_xMoePreco = CrTmpPro.mUltComps
					Else
						If InList(crSigCdOpe.PrecoPs,8,9)
							=fCustoMedio(&lcpArqCab..Emps,CrTmpPro.Cpros,&lcpArqCab..Datas,ThisForm.ParentForm.poDataMgr,(crSigCdOpe.PrecoPs=8))
							_xMoePreco = TmpCusMed.Moedas
						EndIf
					EndIf
				EndIf
			EndIf
		EndIf
	EndIf

	_xCotPreco = ThisForm.parentform.Cotacao( _xMoePreco )
	_xCotOper  = ThisForm.parentform.Cotacao( _xMoeOper )

	lncasaspeso = crSigCdOpe.NCPesos
	_xTotalPeso = &lcpArqItem..Pesos * &lcpArqItem..Fators
	_xTotalPeso = Round( _xTotalPeso , lncasaspeso )

	Replace TPesos With _xTotalPeso, Units With UniVals in &lcpArqItem.	

	_TotalUnit = Val(Str(Iif( InList(crSigCdOpe.QtdPrecos, 1, 5) , &lcpArqItem..Qtds, 1 ) * _xTotalPeso * &lcpArqItem..UniVals * Iif( IsEmpty( _xMoeOper ), 1 , _xCotPreco ),12,6))
	_xTotalPreco = _TotalUnit / Iif(_xCotOper = 0,1,_xCotOper)
	
	Select &lcpArqItem.				
	Replace &lcpArqItem..Totas With _xTotalPreco in &lcpArqItem.								

	If !IsEmpty( _xMoeOper )
		Replace &lcpArqItem..Moedas With _xMoeOper in &lcpArqItem.
	Else
		Replace &lcpArqItem..Moedas With _xMoePreco in &lcpArqItem.
	EndIf
Else
	If ( &lcpArqItem..Utilizas > 9 ) And Not IsEmpty(&lcpArqItem..CPros)
		_xTotalPeso = &lcpArqItem..Pesos * &lcpArqItem..Fators	
		_TotalUnit = Val(Str(Iif( InList(crSigCdOpe.QtdPrecos, 1, 5) , &lcpArqItem..Qtds, 1 ) * _xTotalPeso * &lcpArqItem..Units,12,6))

		Replace &lcpArqItem..Totas With _TotalUnit in &lcpArqItem.
	Endif 
Endif

thisform.TotalizarItn()

Return .t.

ENDPROC

* PROCEDURE totalizaritn
------------------------------------------------------------
PROCEDURE totalizaritn
Local lnReg
*** Totaliza o bloco
thisform.LockScreen = .t.
	Select crSigMvOrc
	lnReg = Recno([crSigMvOrc])	
	lnItens = crSigMvOrc.Citens
	lnItem2 = crSigMvOrc.Citem2
	lnItem3 = crSigMvOrc.Citem3
	lnItAgr = crSigMvOrc.CitAgrs

	If LocalEmp.DIVNOTAS = 2 && P&G
		Replace Tota1s With Iif(Perc1s = 0,Tota1s,(Totas * Perc1s) / 100), ;
				Tota2s With Iif(Perc2s = 0,Tota2s,((Totas + Tota1s) * Perc2s) / 100), ;
				Tota3s With Iif(Perc3s = 0,Tota3s,((Totas + Tota1s + Tota2s) * Perc3s) / 100) in CrSigMvOrc
*!*			Replace tota4s With Iif(Perc4s = 0,0,;
*!*					((totas + tota1s + tota2s + tota3s) / Perc4s) - (totas + tota1s + tota2s + tota3s)) in crSigMvOrc && (Total + Desp. Operacional + Lucro liquido + BV liquido) / percentual de custo nota - (Total + Desp. Operacional + Lucro liquido + BV liquido)		
		If crSigMvOrc.Debitos
			Replace tota4s With Iif(Perc4s = 0,0,;
					((totas + tota1s + tota2s + tota3s) / (1 - Perc4s / 100)) - (totas + tota1s + tota2s + tota3s)) in crSigMvOrc && (Total + Desp. Operacional + Lucro liquido + BV liquido) / percentual de custo nota - (Total + Desp. Operacional + Lucro liquido + BV liquido)				
		Else
			Replace tota4s With Iif(Perc4s = 0,0,;
					((totas + tota1s + tota2s + tota3s) / (1 - Perc4s / 100)) - (totas + tota1s + tota2s + tota3s)) in crSigMvOrc && (Total + Desp. Operacional + Lucro liquido + BV liquido) / percentual de custo nota - (Total + Desp. Operacional + Lucro liquido + BV liquido)		
		Endif
		Replace Tota5s With Totas + Tota1s + Tota2s + Tota3s + Tota4s in crSigMvOrc		
*!*			Replace Tota6s With Iif(Qtds * Pesos = 0,0,Tota5s / (Qtds * Pesos)) in crSigMvOrc
	Else && SRCOM
		*** Calcula total dos valores da  Nf SRCOM e fornecedor
		If Empty(crSigMvOrc.Perc2s) And Empty(crSigMvOrc.Perc3s)
			Replace Tota1s With Totas, Tota2s With 0, Tota3s With 0 in CrSigMvOrc
		Else
			Replace Tota1s With 0, Tota2s With (Perc2s * Totas) / 100, Tota3s With (Perc3s * Totas) / 100 in CrSigMvOrc
		Endif

		*** Calcular os Icentivos conforme o percentual informado
		Replace Tota4s With (Totas * Perc4s) / 100, Tota5s With (Totas * Perc5s) / 100, Tota6s With (Totas * Perc6s) / 100 in crSigMvOrc
		*** Saldo real SRCOM custos internos = Total SRCOM custos internos - Real SRCOM custos internos; Saldo real fornecedor = Total fornecedor - real fornecedor
		Replace Tota9s With Iif(!Empty(Tota1s),Tota1s-Tota8s,Tota2s) - Tota7s, Tota10s With Iif(!Empty(Tota1s),0,Tota3s  - Tota8s) in crSigMvOrc
		
		If Int(Val(Substr(crSigOpCdd.ConfigGers, 192, 1))) = 1 && Decupagem
			Select crSigMvOrc
			Go Top		

			Select Sum(Totas) as Totas, Sum(Tota1s) as Tota1s, Sum(Tota2s) as Tota2s, Sum(Tota3s) as Tota3s, Sum(tota7s) as Tota7s, Sum(tota8s) as Tota8s, ;
					Sum(tota9s) as Tota9s, Sum(tota10s) as Tota10s ;
				From crSigMvOrc Where cItAgrs = Iif(!Empty(lnItAgr),lnItAgr,lnItens) Into Cursor TmpTotBlo	
			If !Eof([TmpTotBlo]) And !IsEmpty(TmpTotBlo.Totas)
				=Seek(Iif(!Empty(lnItAgr),lnItAgr,lnItens),[crSigMvOrc],[Citens])

				*** Saldo real SRCOM custos internos = Total SRCOM custos internos - Real SRCOM custos internos; Saldo real fornecedor = Total fornecedor - real fornecedor
				Replace Tota9s With Iif(!Empty(Tota1s),Tota1s-Tota8s,Tota2s) - Tota7s, Tota10s With Iif(!Empty(Tota1s),0,Tota3s  - Tota8s) in crSigMvOrc
				
				Replace Tota9s With Tota9s - TmpTotBlo.Tota7s, Tota10s With Tota10s - TmpTotBlo.Tota8s in crSigMvOrc			
			Endif 
		Endif
	Endif	
	
	Select crSigMvOrc
	Go Top
	
	Select Sum(Totas) as Totas, Sum(tota1s) as Tota1s, Sum(tota2s) as Tota2s, Sum(tota3s) as Tota3s, ;
		Sum(tota4s) as Tota4s, Sum(tota5s) as Tota5s, Sum(tota6s) as Tota6s, Sum(tota7s) as Tota7s, ;
		Sum(tota8s) as Tota8s, Sum(tota9s) as Tota9s, Sum(tota10s) as Tota10s ;
		from crSigMvOrc ;
		where cItem2 = lnItem2 And !Inlist(cBlocos,[SUBTOTAL],[BRANCO],[SUBTOTAIS],[TAXA],[TOTAL],[IMPOSTO],[TOTALGERAL]) Group By cItem2 into cursor TmpTotBlo
		
	Select crSigMvOrc
	Go Top
	Locate For cItem2 = lnItem2 And cBlocos = [SUBTOTAL]
	If !Eof() &&And !IsEmpty(TmpTotBlo.Totas)
		Replace Totas With TmpTotBlo.Totas, Tota1s With TmpTotBlo.Tota1s, Tota2s With TmpTotBlo.Tota2s, Tota3s With TmpTotBlo.Tota3s ,;
			Tota4s With TmpTotBlo.Tota4s, Tota5s With TmpTotBlo.Tota5s, Tota6s With TmpTotBlo.Tota6s, Tota7s With TmpTotBlo.Tota7s ,;
			Tota8s With TmpTotBlo.Tota8s, Tota9s With TmpTotBlo.Tota9s, Tota10s With TmpTotBlo.Tota10s
	Endif

	Select crSigMvOrc	
	Go top

	Select Sum(Totas) as Totas, Sum(tota1s) as Tota1s, Sum(tota2s) as Tota2s, Sum(tota3s) as Tota3s, ;
		Sum(tota4s) as Tota4s, Sum(tota5s) as Tota5s, Sum(tota6s) as Tota6s, Sum(tota7s) as Tota7s, ;
		Sum(tota8s) as Tota8s, Sum(tota9s) as Tota9s, Sum(tota10s) as Tota10s ;
		from crSigMvOrc ;
		where cItem3 = lnItem3 And !Inlist(cBlocos,[SUBTOTAL],[BRANCO],[SUBTOTAIS],[TAXA],[TOTAL],[IMPOSTO],[TOTALGERAL]) Group By cItem3 into cursor TmpTotBlo	
	
	Select crSigMvOrc
	Go Top
	Locate For cItem3 = lnItem3 And cBlocos = [SUBTOTAL] And Citens = Citem2
	If !Eof() &&And !IsEmpty(TmpTotBlo.Totas)
		Replace Totas With TmpTotBlo.Totas, Tota1s With TmpTotBlo.Tota1s, Tota2s With TmpTotBlo.Tota2s, Tota3s With TmpTotBlo.Tota3s ,;
			Tota4s With TmpTotBlo.Tota4s, Tota5s With TmpTotBlo.Tota5s, Tota6s With TmpTotBlo.Tota6s, Tota7s With TmpTotBlo.Tota7s ,;
			Tota8s With TmpTotBlo.Tota8s, Tota9s With TmpTotBlo.Tota9s, Tota10s With TmpTotBlo.Tota10s
	Endif

	Select crSigMvOrc
	Go Top

	Select Sum(Totas) as Totas, Sum(tota1s) as Tota1s, Sum(tota2s) as Tota2s, Sum(tota3s) as Tota3s, ;
		Sum(Iif(Taxas,tota1s,0)) as Tota1Ts, Sum(Iif(Taxas,tota2s,0)) as Tota2Ts, Sum(Iif(Taxas,tota3s,0)) as Tota3Ts, ;	
		Sum(tota4s) as Tota4s, Sum(tota5s) as Tota5s, Sum(tota6s) as Tota6s, Sum(tota7s) as Tota7s, ;
		Sum(tota8s) as Tota8s, Sum(tota9s) as Tota9s, Sum(tota10s) as Tota10s ;
		from crSigMvOrc ;
		where !Inlist(cBlocos,[SUBTOTAL],[BRANCO],[SUBTOTAIS],[TAXA],[TOTAL],[IMPOSTO],[TOTALGERAL]) into cursor TmpTotBlo

	If LocalEmp.DIVNOTAS = 2 && P&G
		Select crSigMvOrc
		Go top
		Locate For cBlocos = [TOTALGERAL]
		If !Eof() And !IsEmpty(TmpTotBlo.Totas)
			Replace Totas With TmpTotBlo.Totas, Tota1s With TmpTotBlo.Tota1s, Tota2s With TmpTotBlo.Tota2s, Tota3s With TmpTotBlo.Tota3s, ; 
				Tota4s With TmpTotBlo.Tota4s, Tota5s With TmpTotBlo.Tota5s, Tota6s With TmpTotBlo.Tota6s, Tota7s With TmpTotBlo.Tota7s ,;
				Tota8s With TmpTotBlo.Tota8s, Tota9s With TmpTotBlo.Tota9s, Tota10s With TmpTotBlo.Tota10s in crSigMvOrc				

			thisform.getpmargem.Value = ((crSigMvOrc.Tota1s + crSigMvOrc.Tota2s) / crSigMvOrc.Tota5s) * 100
		Endif		

	Else && SRCOM
		Select crSigMvOrc
		Go top
		Locate For cBlocos = [SUBTOTAIS]
		If !Eof() &&And !IsEmpty(TmpTotBlo.Totas)
			Replace Totas With TmpTotBlo.Totas, Tota1s With TmpTotBlo.Tota1s, Tota2s With TmpTotBlo.Tota2s, Tota3s With TmpTotBlo.Tota3s, ;

		Endif
	Endif
	
	Select crSigMvOrc
*!*		Go lnReg
	=Seek(lnItens,[crSigMvOrc],[Citens])

	Select crSigMvOrp
	Scan
		If !Empty(crSigMvOrp.Percs)
			*** (Real SRCOM custos internos) * (percentual + real fornecedor * percentual)
			Replace Valos With ((crSigMvOrc.Tota7s * crSigMvOrp.Percs)/100) + ((crSigMvOrc.Tota8s * crSigMvOrp.Percs)/100) in crSigMvOrp
		Endif
	Endscan

	Select crSigMvOrc
thisform.LockScreen = .f.
thisform.grdItem.Refresh()

ENDPROC

				Group By a.Citens, a.Dpros, 3 Having TotReal > 0 And (IsEmpty(TotPrev) Or TotPrev <> TotReal) Into Cursor TmpTotPrev

PROCEDURE Valid
Local loLista

If this.Tag == this.Value
	Return .t.
Endif

If Not Empty(This.Value)
	lcQuery = [Select Cods, Descrs From SigCdAca Where Descrs like '%] + Alltrim(this.Value) + [%']
	If (ThisForm.ParentForm.poDataMgr.SqlExecute(lcQuery, 'TmpCdAca') < 1)
		=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (TmpCdAca)')
		Return .f.
	EndIf
	Go Top In TmpCdAca

	If !Eof([TmpCdAca])
		loLista = CreateObject([fwBuscaInt], [TmpCdAca], [Descrs], Alltrim(this.Value), [Acabamento], .t.)
		loLista.mAddColuna('Descrs', '', 'Descrição', .t.)
		loLista.Show()

		If Lastkey() = 27
			this.Value = []
			Replace DAcabs With [], CodAcabs With [] in crSigMvOrc
		Else
			Replace DAcabs With TmpCdAca.Descrs,;
					CodAcabs With TmpCdAca.Cods in crSigMvOrc
		Endif
	Else
		=Messagebox([Acabamento não encontrado.],48,[Aviso])
		this.Value = []
		Replace CodAcabs With [] in crSigMvOrc		
	Endif
	Use In TmpCdAca
Else
	Replace CodAcabs With [] in CrSigMvOrc
EndIf
This.Refresh

ENDPROC
