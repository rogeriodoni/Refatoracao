* PROCEDURE montagrades

PROCEDURE montagrades
With ThisForm
	With .grdItens
		.ColumnCount 	   = 7
		.RecordSource 	   = 'CrEmpenhado'
		.DeleteMark 	   = .f.
		.RecordMark 	   = .f.
		.ReadOnly 		   = .t.
		.Scrollbars 	   = 2
		.AllowHeaderSizing = .f.
		.AllowRowSizing    = .f.

		.Column1.ControlSource  = 'CrEmpenhado.Emps'
		.Column2.ControlSource  = 'CrEmpenhado.Dopes'
		.Column3.ControlSource  = 'CrEmpenhado.Numes'
		.Column3.InputMask 	    = '999999'
		.Column4.ControlSource  = 'CrEmpenhado.Datas'
		.Column5.ControlSource  = 'CrEmpenhado.GrupoDs'
		.Column6.ControlSource  = 'CrEmpenhado.ContaDs'
		.Column7.ControlSource  = 'CrEmpenhado.Qtds'
		.Column7.InputMask 	    = '9999999.99'

		For I=1 To .ColumnCount
			lcCol = Alltrim(Str(I))
			.Column&lcCol..Movable   = .f.
			.Column&lcCol..Resizable = .f.
			.Column&lcCol..ReadOnly  = .t.

			.Column&lcCol..FontName  	  = [Courier New]
			.Column&lcCol..Text1.FontName = [Courier New]
		Next
		.Refresh
	EndWith

EndWith
* Fim **************************************************************************************
ENDPROC
