PROCEDURE Init
LParameters pConta, pData, pNoper

ThisForm.cntSombra.lblSombra.Caption = ThisForm.Caption
ThisForm.cntSombra.lblTitulo.Caption = ThisForm.Caption

Local lcPos, lcLrg, llContato
Private lcCliente, _Grupo, _Conta, pCIdChave

If Type('pConta') # 'C'
	pConta = Space(10)
EndIf

If Empty(pConta)
	Return
EndIf

lcCliente = pConta
_Grupo    = Space(10)
_Conta    = Space(10)
pCIdChave = Space(20)
ThisForm.Noper = Iif(Type('pnOper') = [N],pNoper,0)

lcQryCli  = [Select a.Grupos,a.IClis,a.RClis,a.DDDs,a.Tel1s,a.Tel2s,a.EMails From SigCdCli a Where a.IClis = ?lcCliente ]
lcQryGrv  = [Select * From SigCdClh Where CIdChaves = ?pCIdChave ]

With ThisForm
	.poDataMgr = CreateObject('fSqlConector')
	With .poDataMgr
		.AddCursor('SigCdCli', 'IClis'    ,'CrSigCdCli' ,'','',lcQryCli)
		.Addcursor('SigCdClh', 'CIdChaves','CrSigCdCliG','','',lcQryGrv)
	EndWith
	=DoDefault()
EndWith

Create Cursor CrBranco (Obs m,Datas t)
Insert Into CrBranco (Obs,Datas) Values ('',{})

Select CrBranco
Go Top

ThisForm.poDataMgr.Requery('CrSigCdCli')

Select CrSigCdCli
Go Top
ThisForm.Grupo  = CrSigCdCli.Grupos
ThisForm.Conta  = Padr(pConta,10)
_Grupo          = CrSigCdCli.Grupos
_Conta          = CrSigCdCli.IClis

lStrQuery = [Select Emps,IClis,Grupos,GruContas,Autos,Usuars,Datas,DatAgens,Hists,Obs,CIdChaves,Nopers ]+;
			[From SigCdClh ]+;
			[Where IClis = ?_Conta And Grupos = ?_Grupo ]

If ThisForm.poDataMgr.SqlExecute(lStrQuery,'TmpCliH') < 1
	MessageBox('Favor reinicializar o processo !!!',16,'Falha na Conexão (TmpCliH)')
	ThisForm.Release
EndIf

Select Emps,IClis,Grupos,GruContas,Usuars,Nvl(Datas,CrBranco.Datas) As Datas,Nvl(DatAGens,CrBranco.Datas) As DatAgens,Hists,;
	Nvl(Obs,CrBranco.Obs) As Obs,Autos,CIdChaves,Nopers ;
From TmpCliH ;
Where Iif(ThisForm.Noper = 0,0=0,Nopers = ThisForm.Noper) ;
Order By Datas ;
Into Cursor CrSigCdClh ReadWrite

lcGrupo = ThisForm.Grupo
lcConta = ThisForm.Conta

With ThisForm.CntGruCon
	.GetGrupo.Value  = ThisForm.Grupo
	.GetConta.Value  = ThisForm.Conta
	.GetDConta.Value = CrSigCdCli.RClis
	.Refresh
EndWith

lcPos = 5
lcLrg = 80

With ThisForm.CmdApagaItem
	.CmdInserir.Visible = fChecaAcesso('SIGOPFLP','INSERIR')
	.CmdAlterar.Visible = fChecaAcesso('SIGOPFLP','ALTERAR')
	.CmdApagar.Visible  = fChecaAcesso('SIGOPFLP','EXCLUIR')
	.CmdListar.Visible  = fChecaAcesso('SIGOPFLP','LISTAR')

	If .CmdInserir.Visible
		.CmdInserir.Left = lcPos
		lcPos = lcPos + lcLrg
	EndIf
	If .CmdAlterar.Visible
		.CmdAlterar.Left = lcPos
		lcPos = lcPos + lcLrg
	EndIf
	If .CmdApagar.Visible
		.CmdApagar.Left = lcPos
		lcPos = lcPos + lcLrg
	EndIf

	If .CmdListar.Visible
		.CmdListar.Left = lcPos
		lcPos = lcPos + lcLrg
	EndIf
	
	If lcPos > 5
		.Width = lcPos + 5
		.Refresh
	Else
		.Visible = .F.
	EndIf
EndWith

ThisForm.CntContato.Visible = fChecaAcesso('SIGOPFLP','CONTATO')

Select CrSigCdClh
Go Top

With ThisForm.GridFollowUp
	.RecordSource = 'CrSigCdClh'
	.Column1.ControlSource = 'CrSigCdClh.Datas'
	.Column2.ControlSource = 'CrSigCdClh.Hists'
	.Refresh
	.AfterRowColChange()
EndWith

ThisForm.plAcesso = fChecaAcesso([SIGOPFLP],[ALTERAR])
ENDPROC

PROCEDURE Click
Local lcGrupo, lcConta

lcConta = ThisForm.Conta

Do Case

	Case ThisForm.Tipo = 'INSERIR'
		Select CrSigCdClh
		ldDatas  = This.Parent.GetData.Value
		ldAgenda = This.Parent.GetAgenda.Value
		lcHists  = This.Parent.GetHist.Value
		lmObs    = This.Parent.GetObs.Value
		Insert Into CrSigCdClh (Grupos,IClis,GruContas,Datas,Usuars,DatAgens,Hists,Obs,CIdChaves,Nopers) ;
						Values (lcGrupo,lcConta,lcGrupo + lcConta,ldDatas,Usuar,ldAgenda,lcHists,lmObs,;
								fUniqueIds(),ThisForm.nOper)

	Case ThisForm.Tipo = 'ALTERAR'
		Replace Datas    With This.Parent.GetData.Value ,;
				Usuars   With Usuar ,;
				DatAgens With This.Parent.GetAgenda.Value ,;
				Hists    With This.Parent.GetHist.Value ,;
				Obs      With This.Parent.GetObs.Value

EndCase

Select CrSigCdClh
Scatter MemVar Memo
Insert Into CrSigCdCliG From MemVar

ThisForm.poDataMgr.Commit()

PROCEDURE cmdApagar.Click
Local llRet

llRet = .T.
Select CrSigCdClh
If !Eof()
	If CrSigCdClh.Usuars <> Usuar
		MessageBox('Registro Inserido/Alterado por outro usuário !!!',48,'Impossível Excluir')
		llRet = .F.
	EndIf
	
	If llRet
		If !CrSigCdClh.Autos Or Empty(CrSigCdClh.Obs) Or Empty(CrSigCdClh.Datas)
			If MessageBox('Deseja realmente excluir este registro ???',4+32+256,'Confirma Exclusão') = 6
				If ThisForm.poDataMgr.SqlExecute([Delete From SigCdClh Where CIdChaves = ']+CrSigCdClh.CIdChaves+[' ],'') < 1
					MessageBox('Favor reinicializar o processo !!!',0+48,'Falha na Conexão (Delete SigCdClh)')
				Else
					ThisForm.poDataMgr.Commit()
					Select CrSigCdClh
					Delete
					Skip
					Skip -1
				EndIf
			EndIf
		Else
			MessageBox('Histórico Automático !!!',0+48,'Impossível Excluir')
		EndIf
	EndIf
EndIf
Select CrSigCdClh
ThisForm.GridFollowUp.Refresh
ThisForm.GridFollowUp.Column1.Text1.SetFocus
ENDPROC
