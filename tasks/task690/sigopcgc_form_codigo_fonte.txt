* PROCEDURE processa

PROCEDURE processa
Local lcGrEst,lcCdConta,lcDsConta,lnAnalise,ldDtInicial,ldDtFinal,lcMoeda,lnCotaConv
Local llSemDados, _totabe, _totdev,lcArquivo1

Private lcDtInicial, lcDtFinal

lcArquivo1 = Sys(2023) + '\Temp4.jpg'

*-[ Caso a consulta já esteja executada, não processa novamente ]-*
If !ThisForm.ConsultaExecutada

	*-[ Inicializa as variáveis de campos ]-*
	With ThisForm.Fld_Principal.Pag_Consulta
		lcGrEst     = .Origem.Get_Grupo.Value
		lcCdConta   = .Origem.Get_Conta.Value
		lcDsConta   = .Origem.Get_DConta.Value
		lnAnalise   = .Opt_Analise.Value
		ldDtInicial = .Get_DataI.Value
		ldDtFinal   = .Get_DataF.Value
		lcEmpresa   = .Get_Empresa.Value
		lcMoeda     = .Get_Moeda.Value
	EndWith
	
	If Used('TmpCli')
		Use In TmpCli
	EndIf
	lStrQuery = [Select iclis, rclis, endes, bairs, cidas, ceps, estas, tel1s, tel2s, faxs, cpfs, rgs, figjpgs ]+;
				[From SigCdCli ]+;
				[Where grupos = ']+Padr(lcGrEst,10)+[' And iclis = ']+Padr(lcCdConta,10)+[' ]
	lnQueryOk = ThisForm.poDataMgr.SqlExecute(lStrQuery, 'TmpCli')
	If lnQueryOk < 1
		MessageBox('Favor reinicializar o processo.',16,'Falha na Conexão')
		Return(.F.)
	EndIf
	With ThisForm.Fld_Principal.Pag_Resultado.Cadastro.Dados
		Select TmpCli
		Go Top
		If !Eof()
			.Get_CodFor.Value = TmpCli.iclis
			.Get_Nome.Value   = TmpCli.rclis
			.Get_End.Value    = TmpCli.endes
			.Get_Bairro.Value = TmpCli.bairs
			.Get_Cid.Value    = TmpCli.cidas
			.Get_Cep.Value    = TmpCli.ceps
			.Get_Est.Value    = TmpCli.estas
			.Get_Tel1.Value   = TmpCli.tel1s
			.Get_Tel2.Value   = TmpCli.tel2s
			.Get_Fax.Value    = TmpCli.faxs
			.Get_Cgc.Value    = TmpCli.cpfs
			.Get_Ie.Value     = TmpCli.rgs
		Else
			.Get_CodFor.Value = ''
			.Get_Nome.Value   = ''
			.Get_End.Value    = ''
			.Get_Bairro.Value = ''
			.Get_Cid.Value    = ''
			.Get_Cep.Value    = ''
			.Get_Est.Value    = ''
			.Get_Tel1.Value   = ''
			.Get_Tel2.Value   = ''
			.Get_Fax.Value    = ''
			.Get_Cgc.Value    = ''
			.Get_Ie.Value     = ''
		EndIf
	EndWith

	lcDtInicial = fDtoSQL(ldDtInicial)
	lcDtFinal   = fDtoSQL(ldDtFinal,'23:59:59')

	llSemDados = .T.
	
	If !Empty(lcMoeda)
		lnCotaConv = fCarregarCambio(lcMoeda,Date())
	Else
		lnCotaConv = 1
	EndIf

	Wait Window 'Processando Saldos...' NoWait

	Select dbSaldo
	Zap
	
	If Used('TmpSccr')
		Use In TmpSccr
	EndIf
	lStrQuery = [Select grupos,contas,moedas,saldos From SigMvSlc ]+;
				[Where grupos = ']+lcGrEst+[' And contas = ']+lcCdConta+[' ]+;
				[Order By grupos,contas,moedas,saldos ]
	lnQueryOk = ThisForm.poDataMgr.SqlExecute(lStrQuery, 'TmpSccr')
	If lnQueryOk < 1
		MessageBox('Favor reinicializar o processo.',16,'Falha na Conexão')
		Return(.F.)
	EndIf
	Select TmpSccr
	Go Top
	Scan
		Scatter MemVar
		Insert Into dbSaldo From MemVar
		llSemDados = .F.
	EndScan
	
	Select dbSaldo
	Go Top

	Wait Window 'Processando Histórico...' NoWait

	Select dbHistorico
	Zap
	
	If Used('TmpMccr1')
		Use In TmpMccr1
	EndIf
	lStrQuery = [Select emps,grupos,contas,datas,moedas,saldos,valors,opers,hists,hist2s,vencs,valpags,nfs ]+;
				[From SigMvCcr Where grupos = ']+lcGrEst+[' And contas = ']+lcCdConta+[' ]+;
				[Order By emps,grupos,contas,datas,opers,saldos,moedas,valors,hists,hist2s,vencs,valpags,nfs ]
	lnQueryOk = ThisForm.poDataMgr.SqlExecute(lStrQuery, 'TmpMccr1')
	If lnQueryOk < 1
		MessageBox('Favor reinicializar o processo.',16,'Falha na Conexão')
		Return(.F.)
	EndIf
	Select TmpMccr1
	Go Top
	Scan
		Scatter MemVar
		Insert Into dbHistorico From MemVar
		llSemDados = .F.
	EndScan

	Select dbHistorico
	Go Top

	Wait Window 'Processando Produtos...' NoWait

	Select dbProdFor
	Set Order to cpros
	Zap

	If Used('TmpPro')
		Use In TmpPro
	EndIf
	lStrQuery = [Select cpros,dpros,custofs,moecusfs,cunis,pvens,moedas,reffs,pesoms,linhas,situas ]+;
				[From SigCdPro Where ifors = ']+lcCdConta+[' ]+;
				[Order By cpros,dpros,custofs,moecusfs,cunis,pvens,moedas,reffs,pesoms,linhas ]
	lnQueryOk = ThisForm.poDataMgr.SqlExecute(lStrQuery, 'TmpPro')
	If lnQueryOk < 1
		MessageBox('Favor reinicializar o processo.',16,'Falha na Conexão')
		Return(.F.)
	EndIf
	
	m.marca = ' '

	Select TmpPro
	Go Top
	Scan
		Scatter MemVar
		If Seek(TmpPro.linhas,'CrSigCdLin','Linhas') And CrSigCdLin.tpcustos = '2'
			m.custofs = TmpPro.pesoms * TmpPro.custofs
		EndIf
		Insert Into dbProdFor From MemVar
	EndScan

	Select dbResumo
	Zap
	
	Select dbOperaProd
	Zap
	
	If lnAnalise = 1
	
		For _FazEst = 1 To 2

			If _FazEst = 1
				lStrQuery = [Select emps, dopes, numes, valos From SigMvCab ]+;
							[Where grupods = ']+lcGrEst+[' And contads = ']+lcCdConta+[' ]+;
								[And datas Between ?lcDtInicial And ?lcDtFinal ]+;
							[Order By emps, dopes, numes, valos ]
			Else
				lStrQuery = [Select emps, dopes, numes, valos From SigMvCab ]+;
							[Where grupoos = ']+lcGrEst+[' And contaos = ']+lcCdConta+[' ]+;
								[And datas Between ?lcDtInicial And ?lcDtFinal ]+;
							[Order By emps, dopes, numes, valos ]
			EndIf

			If Used('TprMvCab1')
				Use In TprMvCab1
			EndIf
			lnQueryOk = ThisForm.podataMgr.SqlExecute(lStrQuery, 'TprMvCab1')
			If lnQueryOk < 1
				MessageBox('Favor reinicializar o processo.',16,'Falha na Conexão')
				Return(.F.)
			EndIf
			
			lcOrigDest = Iif(_FazEst = 1,'(Destino)...','(Origem)...')
			loBarEest = CreateObject('fwprogressbar','Processando Movimentações - ' + lcOrigDest,Reccount('TprMvCab1'))
			loBarEest.Show
			Select TprMvCab1
			Go Top		
			Scan
				loBarEest.Update(.T.,'Operação: ' + TprMvCab1.dopes)
				=Seek(TprMvCab1.dopes,'CrSigCdOpe','Dopes')
				If !Empty(CrSigCdOpe.cmoes) And Seek(CrSigCdOpe.cmoes,'CrSigCdCot','cmoes')
					lnCotaEst = fCarregarCambio(CrSigCdOpe.cmoes,Date())
				Else
					lnCotaEst = 1
				EndIf
*				If CrSigCdOpe.caixas <> 1 Or !InList(CrSigCdOpe.tipoops,4,7,8)
				If CrSigCdOpe.caixas <> 1 Or Not InList(CrSigCdOpe.GerGdmis, 1, 2)
					Loop
				EndIf
				If !Empty(lcEmpresa) And lcEmpresa <> TprMvCab1.emps
					Loop
				EndIf
				If !Seek(TprMvCab1.dopes,'dbResumo','Dopes')
					Insert Into dbResumo (dopes) Values (TprMvCab1.dopes)
				EndIf
				
				Replace dbResumo.valos With dbResumo.valos + (TprMvCab1.valos * lnCotaEst / lnCotaConv)
				
				lcEDNs = TprMvCab1.emps + TprMvCab1.dopes + Str(TprMvCab1.numes,6)
				If Used('TprMvCabi1')
					Use In TprMvCabi1
				EndIf
				lStrQuery = [Select cpros, qtds, moefats, moedas, tpesos, totas From SigMvItn ]+;
							[Where empdopnums = "]+lcEDNs+[" ]+;
							[Order By cpros, qtds, moefats, moedas, tpesos, totas ]
				lnQueryOk = ThisForm.poDataMgr.SqlExecute(lStrQuery, 'TprMvCabi1')
				If lnQueryOk < 1
					MessageBox('Favor reinicializar o processo.',16,'Falha na Conexão')
					Return(.F.)
				EndIf
				
				Select TprMvCabi1
				Go Top
				Scan
					If Used('TmpProFor')
						Use In TmpProFor
					EndIf
					lStrQuery = [Select cpros, ifors, custofs, moecusfs, pvens, reffs, dpros, cunis, moedas, situas ]+;
								[From SigCdPro ]+;
								[Where cpros = ']+TprMvCabi1.cpros+[' ]+;
								[Order By cpros, ifors, custofs, moecusfs, pvens, reffs, dpros, cunis, moedas ]
					lnQueryOk = ThisForm.poDataMgr.SqlExecute(lStrQuery, 'TmpProFor')
					If lnQueryOk < 1
						MessageBox('Favor reinicializar o processo.',16,'Falha na Conexão')
						Return(.F.)
					EndIf
					If !Seek(TprMvCabi1.cpros + TprMvCab1.dopes,'dbOperaProd')
						Insert Into dbOperaProd (cpros,dopes) Values (TprMvCabi1.cpros,TprMvCab1.dopes)
					EndIf
					lnCotItem1 = fCarregarCambio(TprMvCabi1.moefats,Date())
					lnCotItem2 = fCarregarCambio(TprMvCabi1.moedas,Date())

					If Empty(CrSigCdOpe.cmoes)
						lnCotOper = fCarregarCambio(CrSigCdPam.moedaps,Date())
						_Converte = .T.
					Else
						lnCotOper = fCarregarCambio(CrSigCdOpe.cmoes,Date())
						_Converte = .F.
					EndIf

					lnValItem = 0
					If CrSigCdOpe.pesovts = 1
						lnValItem = lnValItem + (TprMvCabi1.tpesos * Iif(_Converte,lnCotItem1 / lnCotOper,1))
					EndIf
					If CrSigCdOpe.precovts = 1
						lnValItem = lnValItem + (TprMvCabi1.totas * Iif(_Converte,lnCotItem2 / lnCotOper,1))
					EndIf

					lnValItem = Round(lnValItem * lnCotOper / lnCotaConv,2)

					Replace dbOperaProd.valos With dbOperaProd.valos + lnValItem, ;
							dbOperaProd.qtds  With dbOperaProd.qtds + TprMvCabi1.qtds

					If !Seek(TprMvCabi1.cpros,'dbProdFor')
						Select TmpProFor
						Go Top
						If !Eof() And TmpProFor.ifors <> lcCdConta
							m.marca = '*'
						Else
							m.marca = ' '
						EndIf
						Scatter Fields custofs, moecusfs, pvens, moedas, reffs, dpros, cunis, situas MemVar
						Insert Into dbProdFor From MemVar
					EndIf
					
					If TmpProFor.moecusfs <> lcMoeda
						lnCota = fCarregarCambio(TmpProFor.moecusfs,Date())
						lnCota = Iif(lnCota = 0,1,lnCota)
						lnCusto = Round(TmpProFor.custofs * lnCota / lnCotaConv,2)
					Else
						lnCusto = TmpProFor.custofs
					EndIf
					Replace dbResumo.custofs With dbResumo.custofs + (lnCusto * TprMvCabi1.qtds)
					Replace dbResumo.Qtds    With dbResumo.Qtds + TprMvCabi1.qtds
					llSemDados = .F.
					Select TprMvCabi1
				EndScan
				Select TprMvCab1
			EndScan
			loBarEest.Complete(.T.)
		Next
	EndIf

	Select dbResumo
	Go Top

	Select dbProdFor
	Go Top

	Select dbOperaProd 
	Go Top

	If lnAnalise = 1

		Select CrSigCdPam
		Go Top

		Wait Window 'Verificando Movimentações...' NoWait
		If Used('TprMvCabi2')
			Use In TprMvCabi2
		EndIf
		If Empty(lcEmpresa)
*!*				lStrQuery = [Select b.emps,b.dopes,b.numes,b.cpros,b.qtds,b.totas,b.moedas,b.moefats,b.tpesos,]+;
*!*								[c.tipoops,c.cmoes,c.pesovts,c.precovts ]+;
*!*							[From SigMvCab a, SigMvItn b, SigCdOpe c ]+;
*!*							[Where a.datas Between ?lcDtInicial And ?lcDtFinal And b.empdopnums = a.empdopnums And ]+;
*!*								[c.dopes = b.dopes And (c.tipoops = 4 Or c.tipoops = 5 Or c.tipoops = 7 Or c.tipoops = 8) ]+;
*!*									[And c.caixas = 1 ]+;
*!*							[Order By b.emps,b.dopes,b.numes,b.cpros,b.qtds,b.totas,b.moedas,b.moefats,b.tpesos,]+;
*!*								[c.tipoops,c.cmoes,c.pesovts,c.precovts ]

			lStrQuery = [Select b.Opers, b.emps, b.dopes, b.numes, b.cpros, b.qtds, b.totas, b.moedas, ] + ;
							   [b.moefats, b.tpesos, c.tipoops, c.cmoes, c.pesovts, c.precovts, t.GerGdmis ] + ;
						  [From SigMvCab a, SigMvItn b, SigCdOpe c, SigCdTom t ] + ;
						 [Where a.datas Between ?lcDtInicial And ?lcDtFinal And ] + ;
						 	   [b.empdopnums = a.empdopnums And ] + ;
						 	   [c.dopes = b.dopes And ] + ;
						 	   [c.TipoOps = t.Codigos And ] + ;
						 	   [t.GerGdmis In (1, 2) And ] + ;
						 	   [c.caixas = 1 ] + ;
						 [Order By b.emps, b.dopes, b.numes, b.cpros, b.qtds, b.totas, b.moedas, ] + ;
						 	   [b.moefats, b.tpesos, c.tipoops, c.cmoes, c.pesovts, c.precovts]
		Else
*!*				lStrQuery = [Select b.emps,b.dopes,b.numes,b.cpros,b.qtds,b.totas,b.moedas,b.moefats,b.tpesos,]+;
*!*								[c.tipoops, c.cmoes, c.pesovts, c.precovts ]+;
*!*							[From SigMvCab a, SigMvItn b, SigCdOpe c ]+;
*!*							[Where a.datas Between ?lcDtInicial And ?lcDtFinal And a.emps = ']+lcEmpresa+[' ]+;
*!*								[And b.empdopnums = a.empdopnums And c.dopes = b.dopes ]+;
*!*								[And (c.tipoops = 4 Or c.tipoops = 5 Or c.tipoops = 7 Or c.tipoops = 8) And c.caixas = 1 ]+;
*!*							[Order By b.emps,b.dopes,b.numes,b.cpros,b.qtds,b.totas,b.moedas,b.moefats,b.tpesos,]+;
*!*								[c.tipoops, c.cmoes, c.pesovts, c.precovts ]

			lStrQuery = [Select b.Opers, b.emps, b.dopes, b.numes, b.cpros, b.qtds, b.totas, b.moedas, ] + ;
							   [b.moefats, b.tpesos, c.tipoops, c.cmoes, c.pesovts, c.precovts, t.GerGdmis ] + ;
						  [From SigMvCab a, SigMvItn b, SigCdOpe c, SigCdTom t ] + ;
						 [Where a.datas Between ?lcDtInicial And ?lcDtFinal And ] + ;
						 	   [a.emps = '] + lcEmpresa + [' And ] + ;
						 	   [b.empdopnums = a.empdopnums And ] + ;
						 	   [c.dopes = b.dopes And ] + ;
						 	   [c.TipoOps = t.Codigos And ] + ;
						 	   [t.GerGdmis In (1, 2) And ] + ;
						 	   [c.caixas = 1 ] + ;
						[Order By b.emps, b.dopes, b.numes, b.cpros, b.qtds, b.totas, b.moedas, ] + ;
							  [b.moefats, b.tpesos, c.tipoops, c.cmoes, c.pesovts, c.precovts]
		EndIf

		lnQueryOk = ThisForm.poDataMgr.SqlExecute(lStrQuery, 'TprMvCabi2')
		If lnQueryOk < 1
			MessageBox('Favor reinicializar o processo.',16,'Falha na Conexão')
			Return(.F.)
		EndIf

		loBarraMov = CreateObject('fwprogressbar','Verificando Movimentações...',Reccount('TprMvCabi2'))
		loBarraMov.Show

		Select TprMvCabi2
		Go Top
		Scan
			loBarraMov.Update(.T.,TprMvCabi2.emps + ' - ' + TprMvCabi2.dopes + ' - ' + Str(TprMvCabi2.numes,6))
			lnCotItem1 = fCarregarCambio(TprMvCabi2.moefats,Date())
			lnCotItem2 = fCarregarCambio(TprMvCabi2.moedas,Date())

			If Empty(TprMvCabi2.cmoes)
				lnCotOper = fCarregarCambio(CrSigCdPam.moedaps,Date())
				_Converte = .T.
			Else
				lnCotOper = fCarregarCambio(TprMvCabi2.cmoes,Date())
				_Converte = .F.
			EndIf

			lnValItem = 0
			If TprMvCabi2.pesovts = 1
				lnValItem = lnValItem + (TprMvCabi2.tpesos * Iif(_Converte,lnCotItem1 / lnCotOper,1))
			EndIf
			If TprMvCabi2.precovts = 1
				lnValItem = lnValItem + (TprMvCabi2.totas * Iif(_Converte,lnCotItem2 / lnCotOper,1))
			EndIf

			lnValItem = Round(lnValItem * lnCotOper / lnCotaConv,2)
			
			Select dbProdFor
			If Seek(TprMvCabi2.cpros)
*				If TprMvCabi2.tipoops = 4
				If (TprMvCabI2.Opers = [S] And TprMvCabI2.GerGdmis = 1)
					Replace dbProdFor.qtvend With dbProdFor.qtvend + TprMvCabi2.qtds,;
							dbProdFor.vlvend With dbProdFor.vlvend + lnValItem
				EndIf
				If (TprMvCabI2.Opers = [E] And TprMvCabI2.GerGdmis = 1)
					Replace dbProdFor.qtvend With dbProdFor.qtvend - TprMvCabi2.qtds,;
							dbProdFor.vlvend With dbProdFor.vlvend - lnValItem
				EndIf
				
				Select dbOperaProd
				If !Seek(TprMvCabi2.cpros + TprMvCabi2.dopes)
					Insert Into dbOperaProd (cpros,dopes) Values (TprMvCabi2.cpros,TprMvCabi2.dopes)
				EndIf
				Replace dbOperaProd.valos With dbOperaProd.valos + lnValItem, ;
						dbOperaProd.qtds  With dbOperaProd.qtds + TprMvCabi2.qtds

				If !Seek(TprMvCabi2.dopes,'dbResumo','Dopes')
					Insert Into dbResumo (dopes) Values (TprMvCabi2.dopes)
				EndIf
				Replace dbResumo.valos With dbResumo.valos + lnValItem
				If Used('TmpPro2')
					Use In TmpPro2
				EndIf
				lStrQuery = [Select cpros, ifors, custofs, moecusfs From SigCdPro Where cpros = ']+TprMvCabi2.cpros+[' ]

PROCEDURE procimp
Local lnFiltro

With ThisForm.Fld_Principal.Pag_Consulta
	_Fornece = 'Conta : ' + Alltrim(.Origem.Get_Grupo.Value) + ' / ' +;
				Alltrim(.Origem.Get_Conta.Value ) + ' [ ' +;
				Alltrim( .Origem.Get_DConta.Value ) + ' ] '
EndWith

lnFiltro = ThisForm.Fld_principal.Pag_resultado.Result.Movimenta.OptSituas.Value

_Moeda   = ThisForm.Fld_Principal.Pag_Consulta.Get_Moeda.Value
_Produto = dbProdFor.cpros

Select Sum(custofs * estoque) As custofs, moecusfs From dbProdFor Group by moecusfs Into Cursor TmpTot1

Select Sum(pvens * estoque) As pvens, moedas From dbProdFor Group by moedas Into Cursor TmpTot2

Select 4 As coluna, Padr(Replicate(Chr(255),2),14) As cpros, c.custofs, c.moecusfs, Space(1) As marca, 0 As qtvend,;
	0 As vlvend, 0 As estoque, Space(3) As cunis, v.pvens, v.moedas, Space(40) As reffs, Space(40) As dpros,;
	Space(20) As dopes, 0 As qtds, 0 As valos, _Fornece As fornece ;
From TmpTot1 c ;
Full Join TmpTot2 v On c.moecusfs = v.moedas ;
Into Cursor TmpTot3

Do Case
	Case ThisForm.Fld_Principal.Pag_Resultado.ImprimeOpções.Opção.Value = 1
		Select 1 As coluna, a.*, b.dopes, b.qtds, b.valos, _Fornece As fornece ;
		From dbProdFor a ;
		Left Join dbOperaProd b On a.cpros = b.cpros ;
		Where a.cpros = _Produto ;
		and Iif(lnFiltro <> 3, a.Situas = lnFiltro, .T. ) ;
		Into Cursor TmpRelat

	Case ThisForm.Fld_Principal.Pag_Resultado.ImprimeOpções.Opção.Value = 2
		Select 1 As coluna, a.*, b.dopes, b.qtds, b.valos, _Fornece As fornece ;
		From dbProdFor a ;
		Left Join dbOperaProd b On a.cpros = b.cpros ;
		Where Iif(lnFiltro <> 3, a.Situas = lnFiltro, .T. ) ;
		Union All ;
		Select 2 As coluna, Space(14) As cpros, c.custofs, Space(3) As moecusfs, Space(1) As marca, 0 As qtvend,;
			0 As vlvend, 0 As estoque, Space(3) As cunis, 0 As pvens, Space(3) As moedas, Space(40) As reffs,;
			Space(40) As dpros, 1 as Situas, c.dopes, 0 As qtds, c.valos, _Fornece As fornece ;
		From dbResumo c ;
		Union All ;
		Select 3 As coluna, Padr(Chr(255),14) As cpros, 0 As custofs, Space(3) As moecusfs, Space(1) As marca,;
			0 As qtvend, 0 As vlvend, Sum(d.estoque) As estoque, d.cunis, 0 As pvens, Space(3) As moedas, Space(40) As reffs,;
			Space(40) As dpros, d.Situas, Space(20) As dopes, 0 As qtds, 0 As valos, _Fornece As fornece ;
		From dbProdFor d ;
		Where Iif(lnFiltro <> 3, d.Situas = lnFiltro, .T. ) ;
		Group by d.cunis, d.Situas ;
		Union All ;
		Select e.coluna, e.cpros, Iif(IsNull(e.custofs),0,e.custofs) As custofs, Iif(IsNull(e.moecusfs),Space(3),e.moecusfs) As moecusfs,;
			e.marca, e.qtvend, e.vlvend, e.estoque, e.cunis, Iif(IsNull(e.pvens),0,e.pvens) As pvens,;
			Iif(IsNull(e.moedas),Space(3),e.moedas) As moedas, e.reffs, e.dpros, 1 as Situas, e.dopes, e.qtds, e.valos, e.fornece ;
		From TmpTot3 e ;
		Order by 1,2,14 ;
		Into Cursor TmpRelat

	Case ThisForm.Fld_Principal.Pag_Resultado.ImprimeOpções.Opção.Value = 3
		Select 1 As coluna, a.*, b.dopes, b.qtds , b.valos, _Fornece As fornece ;
		From dbProdFor a, dbOperaProd b ;
		Where a.cpros = b.cpros ;
		and Iif(lnFiltro <> 3, a.Situas = lnFiltro, .T. ) ;
		Union All ;
		Select 2 As coluna, Space(14) As cpros, c.custofs, Space(3) As moecusfs, Space(1) As marca, 0 As qtvend,;
			0 As vlvend, 0 As estoque, Space(3) As cunis, 0 As pvens, Space(3) As moedas, Space(40) As reffs,;
			Space(40) As dpros, 1 as Situas, c.dopes, 0 As qtds, c.valos, _Fornece As fornece ;
		From dbResumo c ;
		Union All ;
		Select 3 As coluna, Padr(Chr(255),14) As cpros, 0 As custofs, Space(3) As moecusfs, Space(1) As marca,;
			0 As qtvend, 0 As vlvend, Sum(d.estoque) As estoque, d.cunis, 0 As pvens, Space(3) As moedas, Space(40) As reffs,;
			Space(40) As dpros, d.Situas, Space(20) As dopes, 0 As qtds, 0 As valos, _Fornece As Fornece ;
		From dbProdFor d ;
		Where Iif(lnFiltro <> 3, d.Situas = lnFiltro, .T. ) ;
		Group by d.cunis, d.Situas ;
		Union All ;
		Select e.coluna, e.cpros, Iif(IsNull(e.custofs),0,e.custofs) As custofs,Iif(IsNull(e.moecusfs),Space(3),e.moecusfs) As moecusfs,;
			e.marca,e.qtvend,e.vlvend,e.estoque,e.cunis,Iif(IsNull(e.pvens),0,e.pvens) As pvens,;
			Iif(IsNull(e.moedas),Space(3),e.moedas) As moedas, e.reffs, e.dpros, 1 as Situas, e.dopes, e.qtds, e.valos, e.fornece ;
		From TmpTot3 e ;
		Order by 1,2,14 ;
		Into Cursor TmpRelat
EndCase

If ThisForm.Fld_Principal.Pag_Resultado.ImprimeOpções.Opção.Value = 1
	Create Cursor TmpFinal (TEstoque n(1), TCusto n(1), TVenda n(1))
Else
	Select [ ] as Agrupar,Sum(estoque) As TEstoque, Sum(estoque * (custofs * fCarregarCambio(moecusfs,Date()) / fCarregarCambio(_Moeda,Date()))) As TCusto,;
		Sum(estoque * (pvens * fCarregarCambio(moedas,Date()) / fCarregarCambio(_Moeda,Date()))) As TVenda ;
	From dbProdFor ;
	Where Iif(lnFiltro <> 3, Situas = lnFiltro, .T. ) ;
	Into Cursor TmpFinal group by 1
EndIf
ENDPROC
