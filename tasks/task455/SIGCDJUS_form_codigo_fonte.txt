------------------------------------------------------------
* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
LPARAMETERS pForm, pRetrab

ThisForm.oForm = pForm
ThisForm.pRetrab = pRetrab
ThisForm.oForm.Enabled = .f.

ThisForm.Caption = 'Cadastro de Justificativas do Retrabalho ' + Proper(ThisForm.pRetrab)

With ThisForm
	ThisForm.Refresh
	ThisForm.Draw
	ThisForm.Show
	.poDatamgr = CreateObject('fSqlConector')
	With .poDatamgr
		lcSql = [Select * From SigPrJst Where cRetrabs = ']+ThisForm.pRetrab+[']
		.AddCursor('SigPrJst','CidChaves','CrSigPrJst',' ',ThisForm.pagina.lista.grade,lcSql)
		.pfSqlTabela(1).pColuna('Cods','','','Código',80,.t.)
		.pfSqlTabela(1).pColuna('Descrs','','','Descrição',300,.f.)
	EndWith
	= DoDefault()
	.LockScreen = .f.
EndWith
ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Lista.Grupo_op.Click
------------------------------------------------------------
PROCEDURE Pagina.Lista.Grupo_op.Click
LPARAMETERS opcao

If !dodefault( opcao )
	ThisForm.mAtivapagina1
	Return
Endif

If Inlist(thisform.pcEscolha,'INSERIR','PROCURAR')
	ThisForm.Pagina.Dados.Get_codigo.SetFocus
Else
	If Inlist(thisform.pcEscolha,'ALTERAR')
		ThisForm.Pagina.Dados.Get_Descr.SetFocus
	Else
		Thisform.Pagina.Dados.Grupo_salva.Salva.setfocus
	EndIf
EndIf

ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
If Inlist( Thisform.pcEscolha,'INSERIR')
	If IsEmpty( Thisform.Pagina.Dados.Get_codigo.Value)
		=messagebox('Código Invalido!!!',0+64,'Atenção!')
		Thisform.Pagina.Dados.Get_codigo.setfocus
		Return
	EndIf
EndIf
If Inlist( Thisform.pcEscolha,'INSERIR','ALTERAR')
	lcCod = ThisForm.Pagina.Dados.Get_codigo.value
	lcSql = [Select * From SigPrJst Where Cods = ']+lcCod+[' ]
	If ThisForm.Podatamgr.Sqlexecute(lcsql,'TmpJust') > 0 And ;
		Reccount('TmpJust') > 0 And Inlist( Thisform.pcEscolha,'INSERIR')
		Messagebox( 'Justificativa já cadastrada para o Retrabalho '+TmpJust.cRetrabs, 48, '' )
		ThisForm.Pagina.Dados.Get_codigo.SetFocus
		Return
	Endif
	Replace cRetrabs  With ThisForm.pRetrab,;
			CidChaves With Iif(Empty(CidChaves),fUniqueIds(),CidChaves) in CrSigPrJst
Endif
=DoDefault()
ThisForm.mAtivapagina1

ENDPROC
