------------------------------------------------------------
* PROCEDURE apuradifinventario
------------------------------------------------------------
PROCEDURE apuradifinventario
Parameters pnCod, pcEmp

Private pCod, pEmp, pDt

pCod = Iif(Type([pnCod])<>'N',0,pnCod)
pEmp = Iif(Type([pcEmp])<>'N',_Empr,pcEmp)

lcQuery = [select a.cPros,a.DPros,a.CUnis,a.Moevs,a.PVens From SigCdPro a Order by 1]
If Thisform.poDataMgr.SqlExecute(lcQuery,'crSigCdPro') < 1
	Messagebox('Favor reinicializar o processo !!!',16,'Falha na Conexão (crSigCdPro)')
	Return
Endif

lcQuery = [ Select CodCors, CodTams, Cpros, Lidos as Qtds, 0 as Cbars, Contas, 'N' as Deletar ]+;
	[From SigIvTrB Where Emps = ?pEmp And Codigos = ?pCod Order by cpros]
If Thisform.poDataMgr.SqlExecute(lcQuery,'Inventario') < 1
	Messagebox('Favor reinicializar o processo !!!',16,'Falha na Conexão (Inventario)')
	Return
Endif

Select Inventario
Index On CPros + CodCors + CodTams + Contas Tag Cpros
Go Top

lcQuery = [select a.Contas, a.Grupos, A.CPros, a.Cbars ]+;
	[From SigOpEtq a ]+;
	[Join SigIvTrE b on a.cbars = b.cbars ]+;
	[And a.contas = b.contas ]+;
	[And a.grupos = b.Grupos ]+;
	[And a.cpros = b.cpros ]+;
	[Order by 1]
If Thisform.poDataMgr.SqlExecute(lcQuery,'crOpEtq') < 1
	Messagebox('Favor reinicializar o processo !!!',16,'Falha na Conexão (crOpEtq)')
	Return
Endif

Select crOpEtq
Index On CPros + Str(Cbars,14) Tag BarraPro
Go Top

lStrQuery = [Select a.*, 'N' as Deletar ] + ;
	[From SigIvTrE a, SigCdPro b ]+;
	[Where a.Emps = ?pEmp And a.Codigos = ?pCod And b.CPros = a.CPros ]+;
	[Order By a.CPros,a.CIdChaves ]
If Thisform.poDataMgr.SqlExecute(lStrQuery,'Etiqueta') < 1
	Messagebox('Favor reinicializar o processo !!!',16,'Falha na Conexão (Etiqueta)')
	Return(.F.)
Endif

Select Etiqueta
Index On CPros + CodCors + CodTams + Contas Tag CPros
Index On CPros + CodCors + CodTams + Str(Cbars,14) Tag Cbars
Go Top

lStrQuery = [Select CPros, CodCors, CodTams, Contas, CBars, Tipos, Qtds, Localis, Pesos, PesReals, Emps, Codigos, 'N' as Deletar ] + ;
	[From SigIvTrH ] + ;
	[Where Codigos = ?pCod And Emps = ?pEmp ]+;
	[Order By CIdChaves ]
If (Thisform.poDataMgr.SqlExecute(lStrQuery,'Historico') < 1)
	=Messagebox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (Historico)')
	Return 0
Endif

Select Historico
Index On CPros + CodCors + CodTams Tag CorTam
Index On CPros + CodCors + CodTams + Contas Tag CPros
Index On CPros + CodCors + CodTams + Str(Cbars,14) Tag Cbars
Go Top

Select Etiqueta
Select CPros,CodCors,CodTams,Contas,Emps,Codigos,Cbars,Qtds,Pesos, ' ' As Opers From Etiqueta Where 1=0 Into Cursor Apuracao Readwrite

Select Historico
Go Top
Scan
	If Seek(Historico.CPros + Historico.CodCors + Historico.CodTams + Str(Historico.Cbars,14),'Etiqueta','cbars')
		Replace Deletar With 'S' In Historico
	Endif
Endscan

Select Inventario
Go Top
Scan
	If Seek(Inventario.CPros + Inventario.CodCors + Inventario.CodTams,'Historico','CorTam')
		Replace Deletar With 'S' In Inventario
	EndIf 	
EndScan 

Select Etiqueta
Go Top
Scan
	If Seek(Etiqueta.CPros + Etiqueta.CodCors + Etiqueta.CodTams + Str(Etiqueta.Cbars,14),'Historico','cbars')
		Replace Deletar With 'S' In Etiqueta
	Endif
Endscan

Select Historico
Go Top
Scan For Deletar <> 'S'
	Wait Window 'Aguarde! Processando a atualização do estoque !!!' Nowait
	Scatter Memvar

	Select Apuracao
	Append Blank
	m.Opers = 'E'
	Gather Memvar
Endscan

Select Etiqueta
Go Top
Scan For Deletar <> 'S'
	Wait Window 'Aguarde! Processando a atualização do estoque !!!' Nowait
	Scatter Memvar

	Select Apuracao
	Append Blank
	m.Opers = 'S'
	Gather Memvar
Endscan

Select Inventario
Go Top
Scan For Deletar <> 'S'
	Wait Window 'Aguarde! Processando a atualização do estoque !!!' Nowait
	Scatter Memvar

	Select Apuracao
	Append Blank
	m.Opers = 'S'
	Gather Memvar
Endscan

Select Apuracao
Go Top

Select CPros, CodCors, CodTams, Contas, Emps, Codigos, Cbars, Sum(Qtds) As Qtds, Sum(Pesos) As Pesos, Opers ;
	From Apuracao ;
	group By CPros, CodCors, CodTams, Contas, Emps, Codigos, Cbars, Opers ;
	Into Cursor Diferencas Readwrite

Select Diferencas
Go Top

Select a.Emps,a.CPros,a.Qtds,a.CodTams,a.Pesos,a.CodCors,b.DPros,b.CUnis,b.Moevs,b.PVens,a.Opers,a.Cbars ;
	From Diferencas a ;
	Inner Join crSigCdPro b On b.CPros=a.CPros ;
	Order By 1 ;
	Into Cursor TmpLiRe Readwrite

lnVezes = 1
lcXOper = [X]
&&SET STEP ON 
Select TmpLiRe
Go Top
Index On Opers Tag Opers
Scan
	
	Wait Window 'Aguarde! Processando a atualização do estoque !!!' Nowait
	
	lcDopes  = crOpera.Dopes
	lcEmps   = TmpLiRe.Emps
	lcOpers  = TmpLiRe.Opers
	lnPecas  = TmpLiRe.Qtds
	lnPecHst = TmpLiRe.Qtds
	lnPesos  = TmpLiRe.Pesos
	pDt 	 = CrSigCdBal.DataInis
	
	If Not Empty(TmpLiRe.Cbars)
	
		If Not Seek(TmpLiRe.CPros + Str(TmpLiRe.Cbars,14),'crOpEtq','BarraPro')
			Loop
		EndIf
		
		*-- Verifica se o Barra possui historico
		lcQuery = [select top 1 sqtds from SigMvHst ] + ;
				[where CodBarras = ]+Str(TmpLiRe.Cbars,14)+[] + ;
				[and datas <= ?pDt ] + ;
				[order by cidchaves desc ]
				
		If (Thisform.poDataMgr.SqlExecute(lcQuery,'crHists') < 1)
			=Messagebox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (crHists)')
			Return 0
		EndIf
		
		If Reccount([crHists])=0
			Loop
		EndIf 
	Else
		
		lcQuery = [select top 1 sqtds from SigMvHst ] + ;
				[where cpros = '] + TmpLiRe.CPros 	  + [' ] + ;
				[and emps =']     + lcEmps 			  + [' ] + ;
				[and estos = ']   + CrSigCdBal.Contas + [' ] + ;
				[and grupos = ']  + CrSigCdBal.Grupos + [' ] + ;
				[and codtams = '] + TmpLiRe.CodTams   + [' ] + ;
				[and codcors = '] + TmpLiRe.CodCors   + [' ] + ;
				[and datas <= ?pDt ] + ;
				[order by cidchaves desc ]
				
		If (Thisform.poDataMgr.SqlExecute(lcQuery,'crSaldo') < 1)
			=Messagebox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (crSaldo)')
			Return 0
		Endif
		
		Select crSaldo
		If Reccount([crSaldo])>0
			lnPecas = TmpLiRe.Qtds - crSaldo.sqtds
			lcOpers = Iif(lnPecas<0,'S','E')
			
			lnPecas = lnPecas * Iif(lnPecas<0,-1,1)
			
			*--Peças do historico
			lnPecHst = lnPecas 
			lnPecas = Iif((TmpLiRe.Qtds - crSaldo.sqtds)<0,crSaldo.sqtds,lnPecas)
		Else
			Loop	
		EndIf 
	Endif
	
	If lnPecas = 0 
		Loop 
	EndIf 
	
	If TmpLiRe.Opers <> lcXOper
		lcXOper = TmpLiRe.Opers
		lnVezes = 1000
	Endif

	If lnVezes > 999
		lnNumes = fGerUniqueKey(Alltrim(lcDopes) + lcEmps)
		Insert Into CrSigMvCab (Datars,DataS,Dopes,Emps,Numes,MascNum,GrupoOs,ContaOs,CIdChaves,DtAlts,EmpDopNums, usuals, Usuars) ;
			Values (CrSigCdBal.DataInis,CrSigCdBal.DataInis,lcDopes,lcEmps,lnNumes,fGerMascara(lnNumes),;
			CrSigCdBal.Grupos,CrSigCdBal.Contas,fUniqueIds(),Datetime(),lcEmps + lcDopes + Str(lnNumes,6),Usuar, Usuar)
		lnVezes = 1
	Endif

	lnBarra = 0
	If Not Empty(TmpLiRe.Cbars) And (Not Empty(TmpLiRe.CodCors) Or Not Empty(TmpLiRe.CodTams))
		lnBarra = 0
	Else
		lnBarra = TmpLiRe.Cbars
	Endif

	Insert Into CrSigMvItn (CPros,Dopes,Emps,Numes,Opers,Qtds,CItens,Units,Totas,Moedas,DPros,AQtds,CUnis,CIdChaves,DtAlts,EmpDopNums,Pesos,codbarras) ;
		Values (TmpLiRe.CPros,lcDopes,lcEmps,lnNumes,lcOpers,lnPecas,lnVezes,TmpLiRe.PVens,;
		TmpLiRe.PVens * TmpLiRe.Qtds,TmpLiRe.Moevs,TmpLiRe.DPros,TmpLiRe.Qtds,TmpLiRe.CUnis,fUniqueIds(),Date(),lcEmps + lcDopes + Str(lnNumes,6),lnPesos,lnBarra)

	*--Lanca o Barra no SIGMVITS
	If (Not Empty(TmpLiRe.CodCors) Or Not Empty(TmpLiRe.CodTams))
		Insert Into CrSigMvIts (Emps,Dopes,Numes,CPros,CodTams,CodCors,Qtds,AQtds,;
			codbarras,Pesos,CIdChaves,EmpDopNums,chksubn,CItens) ;
			Values (lcEmps,lcDopes,lnNumes,TmpLiRe.CPros,TmpLiRe.CodTams,TmpLiRe.CodCors,lnPecas,lnPecas,;
			TmpLiRe.Cbars,TmpLiRe.Pesos,fUniqueIds(),lcEmps + lcDopes + Str(lnNumes,6),.F.,lnVezes)
	Endif

	If crOpera.Estoqs = 1

		pCidC = Dtos(CrSigCdBal.DataInis)+lcOpers+Transform(fGerUniqueKey(Dtos(CrSigCdBal.DataInis)),"@L 999999")+Thisform.Sigkey
		pSeq  = fGerUniqueKey('HISTBAR')

		Insert Into CrSigMvHst (CPros,Datars,DataS,DtAudits,Dopes,EmpOs,Emps,Opers,Numes,Qtds,Units,Totas,Grupos,;
			Estos,CIdChaves,EmpDopNums,EmpGruEsts,Oridopnums,Seqs,Pesos, Usuars, codbarras, CodTams, CodCors) ;
			Values (TmpLiRe.CPros,CrSigCdBal.DataInis,CrSigCdBal.DataInis,Null,lcDopes,lcEmps,lcEmps,;
			lcOpers,lnNumes,lnPecHst,TmpLiRe.PVens,TmpLiRe.PVens * lnPecHst,CrSigCdBal.Grupos,;
			CrSigCdBal.Contas,pCidC,lcEmps + lcDopes + Str(lnNumes,6),;
			lcEmps + CrSigCdBal.Grupos + CrSigCdBal.Contas,lcEmps + lcDopes + Str(lnNumes,6),pSeq,lnPesos, Usuar, ;
			TmpLiRe.Cbars, TmpLiRe.CodTams, TmpLiRe.CodCors)

		=SqlCalcP2( CrSigMvHst.Emps, CrSigMvHst.Grupos, CrSigMvHst.Estos, CrSigMvHst.CPros, CrSigMvHst.DataS, CrSigMvHst.CodCors, CrSigMvHst.CodTams, Thisform.poDataMgr )
		=fRecalculaC( CrSigMvHst.Emps, CrSigMvHst.CPros, CrSigMvHst.DataS, Thisform.poDataMgr )

	Endif

	If Not Empty(TmpLiRe.Cbars) And TmpLiRe.Opers = 'S'
		lcUpdate = [Update SigOpEtq Set Contas = Space(10), Grupos = Space(10) Where Cbars = ]+Str(TmpLiRe.Cbars,14)
		If Thisform.poDataMgr.SqlExecute(lcUpdate) < 1
			Messagebox('Favor reinicializar o processo !!!',16,'Falha na Conexão (lcUpdate - Inventario)')
			Return
		Endif
	Endif

	lnVezes = lnVezes + 1

	Select TmpLiRe
Endscan

ENDPROC

* PROCEDURE atualizatotal
------------------------------------------------------------
PROCEDURE atualizatotal
Local lnOldSel, lnQtde, lnLidos
lnOldSel = Select()

If Used([Historico])
	
	If Used([crQtdLidos])
		Use in crQtdLidos
	EndIf 
	
	If Used([crQtdLidos2])
		Use in crQtdLidos2
	EndIf 
	
	Select Historico
	Locate For .F.
	Select * From Historico Where 0=0 Into Cursor crQtdLidos ReadWrite

	Select crQtdLidos
	Select Count(*) As Lidos, Sum(Qtds) As Qtds ;
		From crQtdLidos ;
		Where 0=0 ;
	Into Cursor crQtdLidos2 ReadWrite
	
	Select crQtdLidos2
	lnQtde  = crQtdLidos2.Qtds
	lnLidos = crQtdLidos2.Lidos
	
	ThisForm.Pagina.Leitura.GetLidos.Value = Nvl(lnLidos,0)
	ThisForm.Pagina.Leitura.getQtde.Value  = Nvl(lnQtde,0)
	
	ThisForm.Pagina.Leitura.GetLidos.Refresh	
	ThisForm.Pagina.Leitura.getQtde.Refresh
EndIf 

Select(lnOldSel)
Return .T.
ENDPROC

------------------------------------------------------------
PROCEDURE inserial
ThisForm.ReadComm.Leitura()
Wait Window ThisForm.LeituraSerial NoWait

TotCaracter = Len(Alltrim(ThisForm.ReadComm.Campo))
For lnCnt = 1 To 8   && TotCaracter
	lnAsc = Asc(Substr(ThisForm.ReadComm.Campo,lnCnt,1))
	If Between(lnAsc,48,57)
		ThisForm.LeituraSerial = ThisForm.LeituraSerial + Substr(ThisForm.ReadComm.Campo,lnCnt,1)
	EndIf
Next
If !Empty(ThisForm.LeituraSerial)
	Keyboard CHR(13)
Else
	Clear Typeahead
EndIf

ENDPROC

PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
Local lcXML, mTpGGrupo, mTpGrupo

If ThisForm.pcEscolha = 'INSERIR'
	If Empty(CrSigCdBal.Grupos) Or !Empty(CrSigCdBal.Grupos) And ;
		!ThisForm.poDataMgr.CursorQuery('SigCdGcr','','Codigos',CrSigCdBal.Grupos)
		Messagebox('Grupo de Estoque Inválido !!!',16,'')
		This.Parent.Parent.Get_Grupo.SetFocus
		Return 0
	EndIf
	If !Empty(CrSigCdBal.Contas) And !ThisForm.poDataMgr.CursorQuery('SigCdCli','','IClis',CrSigCdBal.Contas)
		Messagebox('Conta de Estoque Inválida !!!',16,'')
		This.Parent.Parent.Get_Conta.SetFocus
		Return(.F.)
	EndIf
EndIf

If InList(ThisForm.pcEscolha,'INSERIR','ALTERAR')
	If Empty(CrSigCdBal.DtBases)
		Messagebox('Data Base Inválida !!!',16,'Campo Obrigatório')
		This.Parent.Parent.Get_DBase.SetFocus
		Return(.F.)
	EndIf
	Replace CrSigCdBal.Usuars With Usuar
	If ThisForm.pcEscolha = 'INSERIR'
		Replace CrSigCdBal.CIdChaves With fUniqueIds()
	EndIf
	
	*-- Grade de Grande Grupo
	Select CrGGrupo
	Set Order To 
	Store [] TO lcXML, mTpGGrupo, mTpGrupo
	= fCursorToXml('CrGGrupo', @lcXML, .f., .f., 0)
	mTpGGrupo = Iif(Not IsEmpty(lcXML),lcXML,[])
	
	*-- Grade de Grupo
	Select CrGrupo
	Set Order To 
	Store [] TO lcXML, mTpGrupo
	= fCursorToXml('CrGrupo', @lcXML, .f., .f., 0)
	mTpGrupo = Iif(Not IsEmpty(lcXML),lcXML,[])

	Select CrSigCdBal
	Replace mFilGGrp With mTpGGrupo ,;
			mFilGrupo With mTpGrupo in CrSigCdBal
	*--
		
	If (Reccount([crProdutos]) > 0)
		Select crProdutos
		Scan
			If Empty(crProdutos.CPros)
				Delete In crProdutos
			EndIf
			If Empty(crProdutos.Emps)
				Replace Emps With _Empr In crProdutos
			EndIf
			If Empty(crProdutos.Codigos)
				Replace Codigos With crSigCdBal.Codigos In crProdutos
			EndIf
			If Empty(crProdutos.cIdChaves)
				Replace cIdChaves With fUniqueIds() In crProdutos
			EndIf
		EndScan
	
		If Not ThisForm.poDataMgr.Update([crProdutos])
			=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (Update - crProdutos)])
			Return .f.
		EndIf
	EndIf
EndIf

=DoDefault()

ThisForm.mAtivaPagina1

	Messagebox('Inventário com Data Inicial não preenchida!!!',48,'')
	Return
Endif

lnCodBal = CrSigCdBal.codigos

If Messagebox('Este inventário será finalizado,não podendo ser efetuada mais nenhuma leitura...' + Chr(13) +;
		'Deseja encerrar o inventário ???',4+32+256,'ATENÇÃO !!!') = 6

	Wait Clear

	=Thisform.poDataMgr.Requery('Historico')

	=Thisform.poDataMgr.Requery('Balanco')
	Select Balanco
	Index On CPros + CodCors + CodTams + Contas Tag CPros
	Update Balanco Set Lidos = 0, Relidos = 0, LidoPs = 0, RelidoPs = 0
	Go Top

	Thisform.poDataMgr.Requery('Etiqueta')
	Select Etiqueta
	Index On CBars Tag CBars
	Go Top

	loBarraHis = Createobject('fwprogressbar','Aguarde! Processando...',Reccount('Historico'))
	loBarraHis.Show

	Select Historico
	Set Order To
	Go Top
	
	Select CPros, CodCors, CodTams, Contas, Tipos, Sum(Qtds) as Qtds, Sum(Pesos) as Pesos ;
		From Historico ;
	group by CPros, CodCors, CodTams, Contas, Tipos ;
	Into Cursor Historico2 ReadWrite 
	
	Select Historico2
	Go Top 
	Scan
		loBarraHis.Update(.T.)
		If !Seek(Historico2.CPros + Historico2.CodCors + Historico2.CodTams + Historico2.Contas,'Balanco','CPros')
			Insert Into Balanco (CPros,CodCors,CodTams,Contas,Emps,codigos,CIdChaves) ;

		This.Value = 1

			This.Parent.Get_Qtd2.Value = 1
		Else
			This.Parent.Get_Qtd2.Value = 0
		EndIf 		
	Endif

=ThisForm.Atualizatotal()

	This.Parent.Get_Prod.Value = ''
	This.Parent.Get_Cor.Value  = ''
	This.Parent.Get_Tam.Value  = ''
	This.Parent.Get_qtd.Value  = 1
	This.Parent.Get_Prod.SetFocus

PROCEDURE LostFocus
Public plBarra
Local lcArquivo

If Thisform.TipoLeitora = 'S'
	If Empty(Thisform.LeituraSerial)
		If !Thisform.ReadComm.Ole_Comm.PortOpen And !Thisform.ReadComm.PortaError
			Thisform.ReadComm.Ole_Comm.PortOpen = .T.
		Endif
	Else
		This.Value = Thisform.LeituraSerial
		This.Refresh
		Thisform.LeituraSerial = ''
	Endif
Endif

If CrSigCdPam.BalcTrats And Empty(This.Parent.Get_Local.Value) And !Empty(This.Value)
	Messagebox('Local não informado !!!', 0+48,'Campo Obrigatório')
	This.Value = ''
	Keyboard Chr(13)
	Return 0
Endif

If CrSigCdBal.SepEstqs = 1 And Empty(This.Parent.Get_Estoque.Value) And !Empty(This.Value)
	Messagebox('Estoque não informado !!!',0+48,'Campo Obrigatório')
	This.Value = ''
	Keyboard Chr(13)
	Return 0
Endif

plBarra = .F.
lcProd = This.Value

If Used('CsProdutos')
	Use In CsProdutos
Endif

If !Empty(lcProd)

	llBarra  = fVerificarBarras(lcProd) &&SQLfVerificarBarras(lcProd)

	If llBarra and This.parent.parent.Dados.optFilBarra.Value = 2
		Messagebox('Este Produto Controla Barra Individual e o Inventario aceita somente produto por referencia. Favor verificar !!!',48,'')
		This.Parent.Get_Prod.Value  = ''
		This.Parent.Get_Cor.Value   = ''
		This.Parent.Get_Tam.Value   = ''
		This.Parent.Get_Descr.Value = ''
		This.Parent.Get_Qtd.Value   = 1
		Keyboard Chr(13)
		Return 0
	EndIf
	
	plBarra  = .F.
	lnOkBtn = 0
	If llBarra
	
		lStrQuery = [Select a.CPros,a.CBars, b.cbars_old ] + ;
				[From SigOpEtq a ] + ;
				[left join sigopeta b on a.cpros = b.cpros and a.cbars = b.cbars_new ] + ;
				[Where (a.CBars = ]+Alltrim(lcProd)+[ or b.cbars_old = ]+Alltrim(lcProd)+[) ]
		If Thisform.poDataMgr.SqlExecute(lStrQuery,'CrEtiqs') < 1
			Messagebox('Favor reinicializar o processo !!!',16,'Falha na Conexão (CrEtiqs)')
			Return 0
		Endif
		Select CrEtiqs
		Go Top
		If Reccount([CrEtiqs])>0
			lcProd = Alltrim(Str(CrEtiqs.CBars,14))
			m.Cbars_old = CrEtiqs.cbars_old
		EndIf
			
	
		lStrQuery = [Select a.*, isnull(b.cbars_old,0) as cbars_old From SigIvTrH a left join SigOpEta b on a.cbars = b.cbars_new  ]+;
			[Where a.Emps = ']+CrSigCdBal.Emps+[' And a.Codigos = ]+Alltrim(Str(CrSigCdBal.Codigos,4))+[ ]+;
			[And a.CBars = ]+Alltrim(lcProd)+[ ]
		If Thisform.poDataMgr.SqlExecute(lStrQuery,'CrInvH') < 1
			Messagebox('Favor reinicializar o processo !!!',16,'Falha na Conexão (CrInvH)')
			Return(.F.)
		Endif

		*-- Rafael - 01/09/2016 - Nao aceita barra lido
		If Reccount([CrInvH])>0
			If !CrSigCdPam.BalcTrats
				Messagebox('Código de Barra já lido !!!',0+48,'')
				This.Value = ''
				plBarra = .F.
				Keyboard Chr(13)
				Return 0
			Endif
		Endif

		Select CrInvH
		Go Top

		Scan While Codigos = CrSigCdBal.Codigos And Emps = CrSigCdBal.Emps
			Scatter Memvar Memo
			Select Historico
			If !Seek(CrInvH.CIdChaves,'Historico','CIdChaves')
				Insert Into Historico From Memvar
			Endif
		Endscan

		Select Historico
		Set Order To
		If Seek(Str(This.Parent.TipoLeitura.Value,1) + Padl(Alltrim(lcProd),14),'Historico','TpCBars');
				Or Seek(Str(This.Parent.TipoLeitura.Value + 2,1) + Padl(Alltrim(lcProd),14),'Historico','TpCBars')
			lnOk = 7
			If !CrSigCdPam.BalcTrats
				Messagebox('Código de Barra já lido ' + Iif(Empty(Historico.Localis),'!!!','no local ' + Alltrim(Historico.Localis)),0+48,'')
			Else
				If Historico.Localis = This.Parent.Get_Local.Value
					lnOk = Messagebox('Código de Barra já lido ' + Iif(Empty(Historico.Localis),'!!!','no local ' + Alltrim(Historico.Localis)) + Chr(13) + 'Deseja Inserir',4+32+256,'')
				Else
					lnOk    = 6
					lnOkBtn = lnOk

				Endif
			Endif

			If lnOk = 7
				Select Historico
				Set Order To
				This.Value = ''
				plBarra = .F.
				Keyboard Chr(13)
				Return 0
			Endif
		Else
			If CrSigCdPam.BalcTrats And !Seek(Padl(Alltrim(lcProd),14),'Etiqueta','CBars') And Alltrim(lcProd) # '99999999999999'
				Messagebox('Código de Barra de outro Estoque !!!' + Chr(13) + 'Informe a Ocorrência.',0+48,'')
				lnOkBtn = 8
			Endif
		EndIf
		
		Select Historico
		Set Order To

		If Val(lcProd) > 0
			If Used('TmpEti')
				Use In TmpEti
			Endif

			lStrQuery = [Select a.CPros,a.CBars, IsNull(b.cbars_old,0) as cbars_old, a.Qtds,a.Pesos,a.CodCors,a.CodTams,a.Grupos,a.Contas, a.empos, a.Localizas ] + ;
				[From SigOpEtq a ] + ;
				[left join sigopeta b on a.cpros = b.cpros and a.cbars = b.cbars_new ] + ;
				[Where (a.CBars = ]+Alltrim(lcProd)+[ or b.cbars_old = ]+Alltrim(lcProd)+[) ]
				
			If Thisform.poDataMgr.SqlExecute(lStrQuery,'TmpEti') < 1
				Messagebox('Favor reinicializar o processo !!!',16,'Falha na Conexão (TmpEti)')
				Return 0
			EndIf
			
*!*				Select CrEtiqs
*!*				Go Top
*!*				If Reccount([CrEtiqs])>0
*!*					lcProd = Alltrim(Str(CrEtiqs.CBars,14))
*!*				Endif


*!*				lStrQuery = [Select CBars,CPros,Qtds,Pesos,CodCors,CodTams,Grupos,Contas ]+;
*!*					[From SigOpEtq ]+;
*!*					[Where CBars = ]+Alltrim(lcProd)+[ ]+;
*!*					[Order By CBars,CPros,Qtds,Pesos,CodCors,CodTams ]
*!*				If Thisform.poDataMgr.SqlExecute(lStrQuery,'TmpEti') < 1
*!*					Messagebox('Favor reinicializar o processo !!!',16,'Falha na Conexão (TmpEti)')
*!*					Return(.F.)
*!*				Endif

			Select TmpEti
			Go Top
			If !Eof()
				plBarra = .T.
				lcProd  = TmpEti.CPros

				*-- Rafael - 26/08/2016 - Barra ja lido
				If Seek(TmpEti.CBars,'Historico','CBars')
					Messagebox('Código de Barra já lido !!!',0+48,'Aviso')
					lnOkBtn = 7
					Select Historico
					Set Order To
					This.Value = ''
					Keyboard Chr(13)
					Return 0
				Endif
			Endif
		Endif

		If !plBarra
			If Val(lcProd) > 0
				If Used('TmpPro')
					Use In TmpPro
				Endif
				lStrQuery = [Select CBars,CPros From SigCdPro Where CBars = ]+Alltrim(lcProd)+[ ]
				If Thisform.poDataMgr.SqlExecute(lStrQuery,'TmpPro') < 1
					Messagebox('Favor reinicializar o processo !!!',16,'Falha na Conexão (TmpPro)')
					Return(.F.)
				Endif
				Select TmpPro
				Go Top
				If !Eof()
					lcProd = TmpPro.CPros
				Else
					If !CrSigCdPam.BalcTrats
						Messagebox('Produto não cadastrado !!!',0+48,'')
						This.Value = ''
						Keyboard Chr(13)
						Return 0
					Else
						If Alltrim(lcProd) # '99999999' And lnOkBtn = 0
							Messagebox('Produto não cadastrado !!!' + Chr(13) + 'Informe a Ocorrência.',0+48,'')
						Endif
						lnOkBtn = 8
					Endif
				Endif
			Endif
		Endif

		If CrSigCdBal.leitgructa = 1 And Used([TmpEti])
			If TmpEti.Grupos <> CrSigCdBal.Grupos Or TmpEti.Contas <> CrSigCdBal.Contas
				If IsEmpty(TmpEti.Grupos) And IsEmpty(TmpEti.Contas)
					lcMsg = 'Código de Barra já vendido !!!' + Chr(13) 
				Else
					lcMsg = 'Código de Barra de Outro Estoque !!!' + Chr(13)

				Messagebox(lcMsg,0+48,'Verifique.')
				This.Value = ''
				Keyboard Chr(13)
				Return 0

		Endif

		lStrQuery = [Select a.CPros,a.DPros,a.CGrus,a.IFors,a.Reffs,a.Linhas,a.FigJpgs,b.TipoEstos,c.Descs,d.Etiqs,]+;

			[From SigCdPro a ]+;
			[Left Join SigCdGrp b On b.CGrus  = a.CGrus ]+;
			[Left Join SigCdLin c On c.Linhas = a.Linhas ]+;
			[Left Join SigCdUni d On d.CUnis  = a.CUnis ]+;
			[Where a.CPros = ']+lcProd+[' ]+;
			[Order By a.CPros,a.DPros,a.CGrus,a.IFors,a.Reffs,a.Linhas,b.TipoEstos,c.Descs,d.Etiqs,d.CEstos ]

			Return(.F.)
		Endif
		Select CsProdutos
		Go Top
	Else

		If fVerificarBarras(lcProd,9) And crSigCdPac.barindivs <> 3
		
			lcSql = [Select Cpros,Dpros,Cunis,Situas, DtSituas,fVendas,fCustos,Pesoms,Moepcs,Moepvs,pvens,pvideals,Pcuss,cftios,sgrus,]+;
				[Valors,FigJpgs,Cbars,Moevs,Moecs,Moedas,cGrus,Linhas,Colecoes,Ean13,mUltComps,vUltComps,mercs,]+;
				[Consigs,CodFinP,LtMinsv, Status, OrigMercs, SitTrIcms, custofs, cproeqs, Encoms, PesoMetal, Conjunts ]+;
				[From SigCdPro Where Cbars = ]+lcProd

				Messagebox('Favor reinicializar o processo.',16,'Falha na Conexão')
				Return 0
			Endif

			If Reccount('CrTmpPro') = 0
			
				
						
				lcSql = [Select Cpros, Ean13, cproeqs From SigCdPro Where Ean13 = ]+lcProd
				If Thisform.poDataMgr.SqlExecute(lcSql,'CrTmpPro') < 1
					Messagebox('Favor reinicializar o processo.',16,'Falha na Conexão')
					Return 0
				Endif
			Endif
			Select CrTmpPro
			Go Top
			If Reccount() > 0 
				lcProd= CrTmpPro.CPros
			Else
				lcSql = [Select Cpros, cproeqs From SigCdPro Where Cpros = ']+lcProd+[']
				If Thisform.poDataMgr.SqlExecute(lcSql,'CrTmpPro') < 1
					Messagebox('Favor reinicializar o processo.',16,'Falha na Conexão')
					Return 0
				Endif
				Select CrTmpPro
				If Reccount() > 0
					lcProd= CrTmpPro.CPros
				Endif
			Endif
		EndIf 
	
		If This.parent.parent.Dados.optFilBarra.Value = 1
			lcSql = [select cpros, dpros, cbars from sigcdpro where cunis in(select cunis from sigcduni where etiqs <> 'S') And cpros = ']+lcProd+[']
			If Thisform.poDataMgr.SqlExecute(lcSql,'CrTmpProUn') < 1
				Messagebox('Favor reinicializar o processo.',16,'Falha na Conexão')
				Return 0
			Endif
			Select CrTmpProUn
			If Reccount([CrTmpProUn]) > 0
				Messagebox('Este Produto Não Controla Barra Individual e o Inventario Aceita Somente Produto com Barra Individual. Favor verificar !!!',48,'')
				This.Parent.Get_Prod.Value  = ''
				This.Parent.Get_Cor.Value   = ''
				This.Parent.Get_Tam.Value   = ''
				This.Parent.Get_Descr.Value = ''
				This.Parent.Get_Qtd.Value   = 1
				Keyboard Chr(13)
				Return 0
			Endif
		EndIf
	
		lStrQuery = [Select a.CPros,a.DPros,a.CGrus,a.IFors,a.Reffs,a.Linhas,a.FigJpgs,b.TipoEstos,c.Descs,d.Etiqs,]+;
			[d.CEstos,a.CodCors,a.Cunis,a.cUniPs,a.Mercs,a.Sgrus ]+;
			[From SigCdPro a ]+;
			[Left Join SigCdGrp b On b.CGrus  = a.CGrus ]+;
			[Left Join SigCdLin c On c.Linhas = a.Linhas ]+;
			[Left Join SigCdUni d On d.CUnis  = a.CUnis ]+;
			[Where a.CPros = ']+lcProd+[' ]+;
			[Order By a.CPros,a.DPros,a.CGrus,a.IFors,a.Reffs,a.Linhas,b.TipoEstos,c.Descs,d.Etiqs,d.CEstos ]
		If Thisform.poDataMgr.SqlExecute(lStrQuery,'CsProdutos') < 1
			Messagebox('Favor reinicializar o processo !!!',16,'Falha na Conexão (CsProdutos)')
			Return(.F.)
		Endif
		Select CsProdutos
		Go Top
		If Eof()
			Messagebox('Produto não cadastrado !!!',0+48,'')
			This.Value = ''
			Keyboard Chr(13)
			Return 0
		Else
			If Nvl(CsProdutos.Etiqs,'') = 'S'
				Messagebox('Código Inválido!!! Etiqueta Individual !!!',48,'')
				This.Value = ''
				Keyboard Chr(13)
				Return 0
			Endif
		Endif
	Endif

	Select CsProdutos
	Replace TipoEstos With Nvl(TipoEstos,0),;
		Descs     With Nvl(Descs,'')   ,;
		Etiqs     With Nvl(Etiqs,'')   ,;
		CEstos    With Nvl(CEstos,'')  In CsProdutos

	*** #Programador - 07/08/2013 - comentado para não chegar o local
	*!*		lStrQuery = [Select Emps,CPros,Locals ]+;
