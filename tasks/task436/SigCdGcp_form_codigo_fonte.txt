PROCEDURE mctrlbotoes
LParameters plHabilita

Local lnI

If Type('plHabilita') # 'L'
	plHabilita = .F.
Endif

With Thisform.ParentForm.Pagina.Lista
	For lnI=1 to .Grupo_Op.ButtonCount
		If .Grupo_Op.Buttons(lnI).BaseClass = 'Commandbutton' And .Grupo_Op.Buttons(lnI).Visible
			.Grupo_Op.Buttons(lnI).Enabled = plHabilita
		EndIf
	EndFor
	.cmdFPOper.Enabled = plHabilita
	.cmdMultiSubN.Enabled = plHabilita
	.Grupo_Saida.Sair.Enabled = plHabilita
EndWith



ENDPROC

PROCEDURE Init
LParameters poForm, pnDataSes, pcEscolha

Local lcOperacao

Set DataSession to pnDataSes

Thisform.ParentForm = poForm
Thisform.PcEscolha  = pcEscolha
Thisform.ParentForm.Enabled  = .F.
lcOperacao = crSigCdOpe.Dopes

With Thisform
	.Caption = 'Grupos / Contas Permitidas ' + AllTrim( Proper( crSigCdOpe.Dopes ) )
	.Top     = Thisform.ParentForm.Top + Thisform.ParentForm.Pagina.Lista.Grade.Top + 50
	.Left    = Thisform.ParentForm.Left + ;
			   Thisform.ParentForm.Pagina.Lista.Grade.Left + ;
			   Thisform.ParentForm.Pagina.Lista.Grade.Columns(1).Width + ;
			   Thisform.ParentForm.Pagina.Lista.Grade.Columns(3).Width + 16

	.cmdInserir.Visible = fChecaAcesso('SigCdGcp','INSERIR') And ;
						  Iif( Type( 'Thisform.PcEscolha' ) # 'C', .T., ;
						  InList( Thisform.PcEscolha, 'INSERIR', 'ALTERAR' ) )
	.cmdInserir.Enabled = fChecaAcesso('SigCdGcp','INSERIR')	And ;
						  Iif( Type( 'Thisform.PcEscolha' ) # 'C', .T., ;
						  InList( Thisform.PcEscolha, 'INSERIR', 'ALTERAR' ) )
	.cmdExcluir.Visible = fChecaAcesso('SigCdGcp','EXCLUIR') And ;
						  Iif( Type( 'Thisform.PcEscolha' ) # 'C', .T., ;
						  InList( Thisform.PcEscolha, 'INSERIR', 'ALTERAR' ) )
	.cmdExcluir.Enabled = fChecaAcesso('SigCdGcp','EXCLUIR')	And ;
						  Iif( Type( 'Thisform.PcEscolha' ) # 'C', .T., ;
						  InList( Thisform.PcEscolha, 'INSERIR', 'ALTERAR' ) )

	.shape1.Left = IIF(.cmdInserir.visible, .cmdInserir.Left - 5, IIF(.cmdExcluir.visible , .cmdExcluir.left - 5, .cmdSair.Left - 5))
	.shape1.width = IIF(.cmdInserir.visible, 3, IIF(.cmdExcluir.visible, 2, 1)) * .cmdsair.width + 16

ThisForm.cntSombra.lblSombra.Caption = ThisForm.Caption
ThisForm.cntSombra.lblTitulo.Caption = ThisForm.Caption

Thisform.mCtrlBotoes(.F.)

lcSql = [Select a.*, b.Descrs as Dgrus, c.Rclis From SigCdGcp a, SigCdGcr b, SigCdCli c ]+;
		[Where a.Dopes=']+CrSigCdOpe.Dopes+[' And a.Grupos = b.Codigos And a.Contas = c.Iclis ]
SigCdOpe.PodataMgr.Sqlexecute(lcSql,'xGcOpe')

	.Column1.ControlSource = 'TmpGcOpe.Grupos'
	.Column2.ControlSource = 'TmpGcOpe.Contas'

------------------------------------------------------------
* PROCEDURE Click
------------------------------------------------------------
PROCEDURE Click
If Thisform.cmdInserir.Visible Or Thisform.cmdExcluir.Visible
	If ( Thisform.lHouveExcl Or Thisform.lHouveIns )
	
		Select TmpGcOpe
		Delete For Grupos = '  ' or Contas = '  '
		Replace all dopes     With CrSigCdOpe.Dopes,;
				    CidChaves With fUniqueIds()

		Select Grupos, Contas, sum(1) From TmpGcOpe Group by 1,2 having sum(1) > 1 Into Cursor SecondKey
		Select SecondKey
		If Reccount() > 0
			MessageBox('Existem Grupos/Contas em Duplicidade. Favor Acertar!!!',0+16,'Atenção')
			Return .f.
		EndIf
		
		Select CrSigCdGcp
		Zap
		Append From Dbf('TmpGcOpe')
		
		lcSql = [Delete From SigCdGcp Where Dopes = ']+CrSigCdOpe.dopes+[']

		m.lnErro1 = SigCdOpe.poDatamgr.pnIdConn

		With SigCdOpe
			If m.lnErro1 > 0
				m.lnErro1 = SigCdOpe.PodataMgr.Sqlexecute(lcSql,'')
			Endif
			If m.lnErro1>0 
				m.lnErro1 = Iif(.poDataMgr.UpDate('crSigCdGcp'),1,-1)
			EndIf 
			If m.lnErro1>0 
				m.lnErro1 = .poDataMgr.Commit()
			EndIf 
			If Not m.lnErro1>0 
				.poDataMgr.RollBack()
			EndIf 
		EndWith 
		If Not m.lnErro1>0 
			= MessageBox('Erro na Gravação dos Dados do Cadastro de Grupo/Conta.',48)
		EndIf 
	EndIf
ENDIF

Thisform.ParentForm.Enabled = .T.
Thisform.mCtrlBotoes(.T.)
ThisForm.Release
ENDPROC
