PROCEDURE desmarca
Private pNot, pDay, pBai

pNot = .f.
pDay = Datetime()
pBai = crSigMvCab.Datas

ThisForm.poDataMgr.CursorQuery('SigCdOpe', 'TempOpe', 'Dopes', TmpPends.Dopes, 'NDopes')
lcNivel    = TmpPends.Emps + TmpPends.Dopes + Str(TmpPends.Numes, 6)
lnNumNivel = Val(Str(TempOpe.NDopes, 4) + Transform(TmpPends.Numes, '@L 999999'))

Select crLclMvItn
Delete All For Dopes + Str(Numes, 6) = TmpPends.Dopes + Str(TmpPends.Numes, 6)

Select TmpOperacao
If Seek(lnNumNivel, 'TmpOperacao', 'Codigos')
	Delete
EndIf

lcQuery = [Update SigMvCab ] + ;
			 [Set ChkSubn = ?pNot, ] + ;
			 	 [DtAlts = ?pDay, ] + ;
			 	 [DtBaixas = ?pBai ] + ;
		   [Where EmpDopNums = '] + lcNivel + [']

If (ThisForm.poDataMgr.SqlExecute(lcQuery, '') < 1)
	=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (Desmarca - Update - SigMvCab)')
	Return 0
EndIf
* Fim *********************************************************************************

ENDPROC

------------------------------------------------------------
* PROCEDURE marca
------------------------------------------------------------
PROCEDURE marca
Parameters llAutomatico

Private pTru, pDay, pBai

pTru = .t.
pDay = Datetime()
pBai = crSigMvCab.Datas

ThisForm.poDataMgr.CursorQuery('SigCdOpe', 'TempOpe', 'Dopes', TmpPends.Dopes, 'NDopes')
lcNivel    = TmpPends.Emps + TmpPends.Dopes + Str(TmpPends.Numes, 6)
lnNumNivel = Val(Str(TempOpe.NDopes, 4) + Transform(TmpPends.Numes, '@L 999999'))

If Seek(lnNumNivel, 'TmpOperacao', 'Codigos' )
	If Not llAutomatico
		=Messagebox('Movimentação Já Digitada!!!', 0+48, '')
	EndIf
	Select TmpPends
	Return .f.
EndIf

lcCpo = 'FixGDests, FixGOrigs, ConOSubs, ConDSubs'
ThisForm.poDataMgr.CursorQuery('SigCdOpe', 'TempOpe', 'Dopes', crSigMvCab.Dopes, lcCpo)
lcPosicao = crSigMvCab.Emps + crSigMvCab.Dopes + Str( crSigMvCab.Numes, 6 )
lnPuxaOrigem = TempOpe.ConOSubs
lnPuxaDestino = TempOpe.ConDSubs

lcCpo = 'ChkSubn, Obses, GrupoOs, ContaOs, GrupoDs, ContaDs, GrVends, Vends'
ThisForm.poDataMgr.CursorQuery('SigMvCab', 'PendEest', 'EmpDopNums', lcNivel)
_Retorna = .t.
Do Case
	Case PendEest.ChkSubn
		If Not llAutomatico
			=Messagebox('Movimentação Já Utilizada Em Outra Operação!!!', 0+48, '')
		EndIf
		_Retorna = .f.

	Case Not Empty(crSigMvCab.ContaDs) And (PendEest.ContaDs <> crSigMvCab.ContaDs)
		If Not llAutomatico
			=Messagebox('Movimentação de Outra Conta!!!', 0+48, '')
		EndIf
		_Retorna = .f.
		
	Case Not Empty(crSigMvCab.ContaOs) And (PendEest.ContaOs <> crSigMvCab.ContaOs)
		If Not llAutomatico
			=Messagebox('Movimentação de Outra Conta!!!', 0+48, '')
		EndIf
		_Retorna = .f.
EndCase

If Not _Retorna
	Select TmpPends
	Return .f.
EndIf

lcQuery = [Update SigMvCab ] + ;
			 [Set ChkSubn = ?pTru, ] + ;
			 	 [DtAlts = ?pDay, ] + ;
			 	 [DtBaixas = ?pBai ] + ;
		   [Where EmpDopNums = '] + lcNivel + [']

If (ThisForm.poDataMgr.SqlExecute(lcQuery, '') < 1)
	=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (Marca - Update - SigMvCab)')
	Return 0
EndIf

Select crLclMvItn
Delete All For Empty(Cpros)
Go Bottom

Select TmpOperacao
Insert Into TmpOperacao (Codigos) Values (lnNumNivel)

If IsEmpty(crSigMvCab.Obses)
	Replace Obses With PendEest.Obses In crSigMvCab
Endif

With ThisForm.ParentForm.Pagina.Dados
	If InList(lnPuxaOrigem,1,2)
		If Empty(crSigMvCab.GrupoOs) Or (Empty(crSigMvCab.ContaOs) And Not TempOpe.FixGOrigs)
			Replace GrupoOs With Iif(lnPuxaOrigem=1, PendEest.GrupoOs, PendEest.GrupoDs) In crSigMvCab
		EndIf
		If Empty(crSigMvCab.ContaOs) 
			Replace ContaOs With Iif(lnPuxaOrigem=1, PendEest.ContaOs, PendEest.ContaDs) In crSigMvCab
		EndIf
	EndIf
	
	If InList(lnPuxaDestino, 1, 2)
		If Empty(crSigMvCab.GrupoDs) Or (Empty(crSigMvCab.ContaDs) And Not TempOpe.FixGDests)
			Replace GrupoDs With Iif(lnPuxaDestino=1, PendEest.GrupoOs, PendEest.GrupoDs) In crSigMvCab
		EndIf
		If Empty(crSigMvCab.ContaDs) 
			Replace ContaDs With Iif(lnPuxaDestino=1, PendEest.ContaOs, PendEest.ContaDs) In crSigMvCab
		EndIf
	EndIf

	If Empty(crSigMvCab.GrVends)
		Replace GrVends With PendEest.GrVends In crSigMvCab
	EndIf
	If Empty(crSigMvCab.Vends) And (crSigMvCab.GrVends = PendEest.GrVends)
		Replace Vends With PendEest.Vends In crSigMvCab
	EndIf
EndWith

ThisForm.poDataMgr.CursorQuery('SigMvItn', 'crItens', 'EmpDopNums', lcNivel)
Select crItens
Scan
	Scatter Memvar Memo
 	Insert Into crLclMvItn From Memvar
EndScan

lnItens = 0
Select crLclMvItn
Set Order to 
Scan
	If Empty(Cpros) 
		Loop
	EndIf
	lnItens = lnItens + 1
	Replace cItens With lnItens
Endscan

Go Top
If Eof()
 	Insert Into crLclMvItn (Emps, Dopes, Numes, cItens) ;
		 		  Values (crSigMvCab.Emps, crSigMvCab.Dopes, crSigMvCab.Numes, lnItens + 1)
EndIf

Select crSigMvCab
With ThisForm.ParentForm.Pagina.Dados
	ThisForm.poDataMgr.CursorQuery('SigCdCli', 'crCli', 'IClis', crSigMvCab.ContaOs, 'RClis')
	If Not Eof('crCli')
		.Origem.get_dconta.Value = crCli.RClis
	Else
		.Origem.get_dconta.Value = ''
	EndIf

	ThisForm.poDataMgr.CursorQuery('SigCdCli', 'crCli', 'IClis', crSigMvCab.ContaDs, 'RClis')
	If Not Eof('crCli')
		.Destino.get_dconta.Value = crCli.RClis
	Else
		.Destino.get_dconta.Value = ''
	EndIf

	ThisForm.poDataMgr.CursorQuery('SigCdCli', 'crCli', 'IClis', crSigMvCab.Vends, 'RClis')
	If Not Eof('crCli')
		.Responsavel.get_dconta.Value = crCli.RClis
	Else
		.Responsavel.get_dconta.Value = ''
	EndIf
	
	.Refresh
Endwith

Select crLclMvItn
Set Order To
* Fim *********************************************************************************

ENDPROC

------------------------------------------------------------
* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
Lparameters pForm, pData, pCnx

ThisForm.cntSombra.lblSombra.Caption = ThisForm.Caption
ThisForm.cntSombra.lblTitulo.Caption = ThisForm.Caption

This.LockScreen = .t.

ThisForm.ParentForm = pForm
Set DataSession To pData
ThisForm.poDataMgr = pCnx

Create Cursor TmpPends(Flag l, Emps c(3), Dopes c(20), Numes n(6), MascNum c(6), Datas t Null, ;
					   TpFats c(3), Origem c(21), Destino c(21), Valos n(12,2))
Index On Emps + Dopes + Str(Numes, 6) Tag EmpDopNum
Index On Numes Tag Numes

Select TmpOperacao
Delete All For Empty(Codigos)

ThisForm.poDataMgr.CursorQuery('SigCdOpe', 'TempOpe', 'Dopes', crSigMvCab.Dopes, 'Nivels, ConOSubs, ConDSubs')
lcDopesNivel  = TempOpe.Nivels
lnPuxaOrigem  = TempOpe.ConOSubs
lnPuxaDestino = TempOpe.ConDSubs

ThisForm.poDataMgr.CursorQuery('SigCdOpe', 'TempOpe', 'Dopes', lcDopesNivel, 'NDopes')
lcNumeroNivel = Str(TempOpe.NDopes, 4 )

lcEd1 = _Empr + Padr(lcDopesNivel, 20) + Str(0, 6)
lcEd2 = _Empr + Padr(lcDopesNivel, 20) + Str(999999, 6)

lcQuery = [Select Emps, Dopes, Numes, MascNum, Datas, Valos, ChkSubn, GrupoOs, GrupoDs, ContaOs, ContaDs ] + ;
		    [From SigMvCab ] + ;
		   [Where EmpDopNums BetWeen '] + lcEd1 + [' And '] + lcEd2 + [']

If (ThisForm.poDataMgr.SqlExecute(lcQuery, 'PendEest') < 1)
	=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (PendEest)')
	This.LockScreen = .f.
	Return 0
EndIf

Select PendEest
Scan
	If PendEest.ChkSubn And Not Seek(Val(lcNumeroNivel + Transf(PendEest.Numes, '@L 999999')), 'TmpOperacao', 'Codigos')
		Loop
	EndIf

	Store '' To lcOrigem, lcDestino
	If Inlist(lnPuxaOrigem, 1, 2)
		If (lnPuxaOrigem = 1)
			If Not Empty(crSigMvCab.ContaOs) And (crSigMvCab.Contaos <> PendEest.ContaOs)
				Loop
			EndIf
			lcOrigem = PendEest.GrupoOs + '/' + PendEest.ContaOs
		Else
			If Not Empty(crSigMvCab.ContaOs) And (crSigMvCab.ContaOs <> PendEest.ContaDs)
				Loop
			EndIf
			lcOrigem = PendEest.GrupoDs + '/' + PendEest.ContaDs
		EndIf
	Else
		lcOrigem = PendEest.GrupoOs + '/' + PendEest.ContaOs
	EndIf

	If Inlist(lnPuxaDestino, 1, 2)
		If (lnPuxaDestino = 1)
			If Not Empty(crSigMvCab.ContaDs) And (crSigMvCab.Contads <> PendEest.ContaOs)
				Loop
			EndIf
			lcDestino = PendEest.GrupoOs + '/' + PendEest.ContaOs
		Else
			If Not Empty(crSigMvCab.ContaDs) And (crSigMvCab.ContaDs <> PendEest.ContaDs)
				Loop
			EndIf
			lcDestino = PendEest.GrupoDs + '/' + PendEest.ContaDs
		EndIf
	Else
		lcDestino = PendEest.GrupoDs + '/' + PendEest.ContaDs
	EndIf

	_Flag = Seek(Val(lcNumeroNivel + Transform(Numes, '@L 999999')), 'TmpOperacao', 'Codigos')

	Insert Into TmpPends (Flag, Emps, Dopes, Numes, Mascnum, Datas, Origem, Destino, Valos) ;
				  Values (_Flag, PendEest.Emps, PendEest.Dopes, PendEest.Numes, PendEest.MascNum,;
						  PendEest.Datas, lcOrigem, lcDestino, PendEest.Valos)
EndScan

With ThisForm.Grade
	.RecordSource 			 = 'TmpPends'
	.Seleciona.ControlSource = 'TmpPends.Flag'
	.Operacao.ControlSource  = 'TmpPends.Dopes'
	.Codigo.ControlSource 	 = 'TmpPends.MascNum'
	.Data.ControlSource 	 = 'TmpPends.Datas'
	.Grupo.ControlSource 	 = 'TmpPends.Origem'
	.Conta.ControlSource 	 = 'TmpPends.Destino'
	.Column8.ControlSource 	 = 'TmpPends.Valos'
EndWith

ThisForm.Op_Escolha = ThisForm.Parentform.pcEscolha

ThisForm.SelTudo.Visible 	 = (ThisForm.Op_Escolha = 'INSERIR')
ThisForm.Apaga.Visible 		 = (ThisForm.Op_Escolha = 'INSERIR')
ThisForm.Say_Procura.Visible = (ThisForm.Op_Escolha = 'INSERIR')
ThisForm.Get_Procura.Visible = (ThisForm.Op_Escolha = 'INSERIR')

If (ThisForm.Op_Escolha = 'INSERIR')
	ThisForm.Get_Procura.SetFocus
Else
	ThisForm.Grade.Codigo.Ingrid1.SetFocus
EndIf
ThisForm.Refresh
This.LockScreen = .f.
* Fim *********************************************************************************
ENDPROC
