------------------------------------------------------------
* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
Lparameters pCli

With ThisForm
	.poDatamgr = CreateObject('fSqlConector',ThisForm.Name)
	With .poDatamgr
		ThisForm.Grupos = []
		ThisForm.poDataMgr.CursorQuery([SigCdPam], [crSigCdPam], .f., .f., [GrPadClis])
		If Not Used([crSigCdPam])
			=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (CursorQuery - crSigCdPam)])
			Return 0
		EndIf
		ThisForm.Grupos   = Padr(crSigCdPam.GrPadClis, 10)
		ThisForm.Clientes = pCli

		If Empty(ThisForm.Grupos)
			lcQuery = [Select * From SigCdClc]
		Else
			If ThisForm.Clientes
				lcQuery = [Select * From SigCdClc] + Iif(Empty(ThisForm.Grupos), [], [ Where Grupos = '] + ThisForm.Grupos + ['])
			Else
				lcQuery = [Select * From SigCdClc] + Iif(Empty(ThisForm.Grupos), [], [ Where Not Grupos = '] + ThisForm.Grupos + ['])
			EndIf
		EndIf

		.AddCursor('SigCdClc','cidchaves','crSigCdClc','',ThisForm.pagina.lista.grade, lcQuery)
		.pfSqlTabela(1).pColuna('Emps','','','Emp',30,.t.)
		.pfSqlTabela(1).pColuna('grupos','','','Grupo',80,.t.)
		.pfSqlTabela(1).pColuna('codigos','','','Código',80,.t.)
		.pfSqlTabela(1).pColuna('descs','','','Descriçãos',220,.t.)		
		.pfSqlTabela(1).pColuna('Prioridades','','','P',27,.t.)
		.pfSqlTabela(1).pColuna('faixais','9999999999.99','','Faixa Inicial',101,.t.)
		.pfSqlTabela(1).pColuna('faixafs','9999999999.99','','Faixa Final',101,.t.)		

		.AddCursor([SigCdCmt], [cIdChaves], [crSigCdCmt])
	EndWith
	=DoDefault()

	lcQuery = [Select Codigos, Descrs From SigCdGcr]
	If (ThisForm.poDataMgr.SqlExecute(lcQuery, [crSigCdGcr]) < 1)
		=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (crSigCdGcr)])
		Return .f.
	EndIf
	Select crSigCdGcr
	Index On Codigos Tag Codigos
	Index On Descrs Tag Descrs
	
	.LockScreen = .f.
EndWith
ENDPROC

If Not Empty(ThisForm.Grupos) And ThisForm.Clientes
	If (ThisForm.pcEscolha = [INSERIR]) Or (Empty(crSigCdClc.Grupos))
		Replace Grupos With ThisForm.Grupos In crSigCdClc
	EndIf
	ThisForm.Pagina.Dados.get_grupo.Enabled = .f.
EndIf
ThisForm.Pagina.Dados.get_grupo.Refresh 

Zap In crMotivos

If InList(ThisForm.pcEscolha, [ALTERAR], [CONSULTA], [EXCLUIR])
	lcQuery = [Select a.*, b.Descrs ] + ;
			    [From SigCdCmt a, SigCdFs b ] + ;
			   [Where a.fkChaves = '] + crSigCdClc.cIdChaves + [' And ] + ;
			   		 [a.Codigos = b.Codigos]
		   
	If (ThisForm.poDataMgr.SqlExecute(lcQuery, [LocalClaMt]) < 1)
		=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (LocalClaMt)])
		Return 0
	EndIf
	Select LocalClaMt
	Scan
		Insert Into crMotivos (Codigos, Descrs, Tipos) ;
					   Values (LocalClaMt.Codigos, LocalClaMt.Descrs, LocalClaMt.Tipos)
	EndScan
EndIf

If InList(ThisForm.pcEscolha, [INSERIR], [ALTERAR])
	lcQuery = [Select * From SigCdFs]
	If (ThisForm.poDataMgr.SqlExecute(lcQuery, [LocalMDsc]) < 1)
		=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (LocalMDsc)])
		Return 0
	EndIf

	Select LocalMDsc
	Scan
		If Not Seek(LocalMDsc.Codigos, [crMotivos], [Codigos])
			Insert Into crMotivos (Codigos, Descrs, Tipos) Values (LocalMDsc.Codigos, LocalMDsc.Descrs, 3)
		EndIf
	EndScan
EndIf

Go Top In crMotivos
ThisForm.Pagina.Dados.grdMotivos.Refresh 

Do Case
	Case Inlist( ThisForm.pcEscolha, 'INSERIR', 'PROCURAR' )
		ThisForm.Pagina.Dados.get_codigo.SetFocus
	Case ThisForm.pcEscolha = 'ALTERAR'
		ThisForm.Pagina.Dados.Get_Descricao.SetFocus
	Otherwise 
		ThisForm.Pagina.Dados.Grupo_Salva.Salva.SetFocus
EndCase
ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
Local laCampos(3,2) , lcCampos(2,2)

If ThisForm.pcEscolha = 'INSERIR'
	If IsEmpty( crSigCdClc.grupos )
		Messagebox( 'Grupo não pode ficar em branco!', 48, '' )
		ThisForm.Pagina.Dados.Get_grupo.SetFocus
		Return
	Endif

	If IsEmpty( CrSigCdClc.codigos)
		Messagebox( 'Código não pode ficar em branco!', 48, '' )
		ThisForm.Pagina.Dados.get_codigo.SetFocus
		Return
	Endif

	m.laCampos(1,1) = 'Emps'
	m.laCampos(1,2) = thisform.pagina.dados.GetEmps.value
	m.laCampos(2,1) = 'grupos'
	m.laCampos(2,2) = thisform.pagina.dados.get_grupo.value
	m.laCampos(3,1) = 'codigos'
	m.laCampos(3,2) = thisform.pagina.dados.get_codigo.value

	If ThisForm.poDataMgr.ChkRegister('SigCdClc',@m.laCampos)
		Messagebox( 'Grupo e Codigo já Cadastrado!', 48, '' )
		ThisForm.Pagina.Dados.Get_Grupo.SetFocus
		Return
	Endif

	Select CrSigCdClc
	Replace cidchaves with fUniqueIds()

Else

	If ThisForm.pcEscolha = 'EXCLUIR'
		m.lcCampos(1,1) = 'grupos'
		m.lcCampos(1,2) = thisform.pagina.dados.get_grupo.value
		m.lcCampos(2,1) = 'codigos'
		m.lcCampos(2,2) = thisform.pagina.dados.get_codigo.value

		If ThisForm.poDataMgr.ChkRegister('SigCdCli',@m.lcCampos)
			Messagebox( 'Existe Conta Cadastrada com essa Classificação !', 48, 'Impossível Exclusão' )
			ThisForm.Pagina.Dados.Grupo_Salva.Cancelar.SetFocus
			Return
		Endif

	EndIf
Endif

If ThisForm.pcEscolha = 'ALTERAR' OR ThisForm.pcEscolha = 'INSERIR'
	If Thisform.pagina.dados.get_faixai.value > Thisform.pagina.dados.get_faixaf.value
		Messagebox( 'Valor da Faixa Inicial > que o valor da Faixa Final !', 48, '' )
		ThisForm.Pagina.Dados.Get_faixaf.SetFocus
		Return
	EndIf
	lnValIni = thisform.pagina.dados.Get_faixai.Value
	lnValFim = thisform.pagina.dados.Get_faixaf.Value
	If (lnValIni + lnValFim <> 0)
		SqlClaCo = [Select Descs, FaixaIs, FaixaFs ] + ;
					 [From SigCdClc ]+;
					[Where Emps = ']+thisform.pagina.dados.GetEmps.value+[' ]+;
						   [And grupos = ']+thisform.pagina.dados.get_grupo.value+[' ]+;
		                   [And Not cIdChaves = ']+crSigCdClc.cIdChaves+[' ]+;
		                   [And ( (?lnValIni Between Faixais And Faixafs) ]+;
		                   [Or    (?lnValFim Between Faixais And Faixafs) ]+;
		                   [Or    (Faixais Between ?lnValIni And ?lnValFim) ]+;
		                   [Or    (Faixafs Between ?lnValIni And ?lnValFim) ) ]

		ThisForm.poDatamgr.SqlExecute(SqlClaCo,'csSigCdClc')
		Select csSigCdClc
		If Not Eof()
			=MessageBox([Faixa Cadastrada no Codigo ] + Alltrim(Descs) + Chr(13) + ;
						[Valor : ] + Alltrim(Str(FaixaIs, 12, 2)) + [ à ] + Alltrim(Str(FaixaFs, 12, 2)), 16, [Atenção!!!])
			Return 0
		Endif
	EndIf
EndIf

* Gravando Dados da Grade de Motivos
If InList(ThisForm.pcEscolha, [INSERIR], [ALTERAR], [EXCLUIR])
	lcQuery = [Delete From SigCdCmt Where fkChaves = '] + crSigCdClc.cIdChaves + [']
	If (ThisForm.poDataMgr.SqlExecute(lcQuery, []) < 1)
		=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (Delete - crSigCdCmt)])
		Return 0
	EndIf

	Zap In crSigCdCmt
	
	Select crMotivos
	Scan
		If InList(crMotivos.Tipos, 1, 2)
			Insert Into crSigCdCmt (fkChaves, cIdChaves, Codigos, Tipos) ;
							Values (crSigCdClc.cIdChaves, fUniqueIds(), crMotivos.Codigos, crMotivos.Tipos)
		EndIf
	EndScan
EndIf

=DoDefault()

ThisForm.mAtivapagina1

ENDPROC

	ThisForm.mAtivapagina1
Endif
