		    [From SigCdCli] + ;
		    [ Where Grupos=']+crSigCdMpd.GruOrigs+[']+;
		    [ and   iclis=']+crSigCdMpd.ConOrigs +[']+;
		    [ Order By grupos,iclis,rclis ]

	=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (crPesqSigCdCli)')

	ThisForm.Pagina.Dados.PagDados.PgEstoque.Get_DContao.Value = crPesqSigCdCli.rclis

* PROCEDURE msv_excluir
------------------------------------------------------------
PROCEDURE msv_excluir
m.lcProg = 'P' + Padl( crSigCdMpd.Ndopps, 8, '0' )
m.lnErro1 = Thisform.podatamgr.pnIdConn
With ThisForm
	If m.lnErro1 > 0
		m.lnErro1 = .Podatamgr.Requery("crSigCdPrg")
	EndIf 
	Select crSigCdPrg
	Delete All

	If m.lnErro1 > 0
		m.lnErro1 = ThisForm.Podatamgr.SQLExecute([Select pkChaves From SigCdAcb Where Programas='] + m.lcProg + ['], 'crTmpSigCdAcb')
	EndIf 
	Select crTmpSigCdAcb
	Scan While m.lnErro1 > 0
		m.lnErro1 = ThisForm.Podatamgr.SQLExecute([Delete From SigCdAcb Where pkChaves='] + crTmpSigCdAcb.pkChaves + ['])
	EndScan
	Use In crTmpSigCdAcb 

	If m.lnErro1 > 0
		m.lnErro1 = .Podatamgr.Requery("crSigOpOeo")
	EndIf 
	Select crSigOpOeo
	Delete All

	Select crSigCdMpd
	Delete 
	
	If m.lnErro1>0 
		m.lnErro1 = Iif(.poDataMgr.UpDate('crSigCdMpd'),1,-1)
	EndIf 
	If m.lnErro1>0 
		m.lnErro1 = Iif(.poDataMgr.UpDate('crSigOpOeo'),1,-1)
	EndIf 
	If m.lnErro1>0 
		m.lnErro1 = Iif(.poDataMgr.UpDate('crSigCdPrg'),1,-1)
	EndIf 
	
	If m.lnErro1>0 
		m.lnErro1 = .poDataMgr.Commit()
	EndIf 

	If Not m.lnErro1>0 
		.poDataMgr.RollBack()
	EndIf 
EndWith 

If Not m.lnErro1>0 
	= MessageBox('Erro na Exclusão dos Dados do Cadastro de Operações.',48,'Erro')
EndIf 

ENDPROC
