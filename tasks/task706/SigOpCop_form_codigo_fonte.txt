LOCAL xBrox

If !IsEmpty( This.Value )
	xBrox = CreateObject('fwBuscaExt',Thisform.poDatamgr.pnIdconn,'SigCdOpe','csSigCdOpe','Dopes',This.Value,'Operações',.t.,'Dopes')

	If  Not xBrox.plAchouRegistro

		xBrox.Show()
	EndIf 
	This.Value = Iif( LastKey() # 27, Evaluate('csSigCdOpe.Dopes'), '' )
Else
	If Lastkey() <> 27
		MessageBox("Operação Deve ser informada !!!",16,"")
		Return .f.
	Endif
Endif
