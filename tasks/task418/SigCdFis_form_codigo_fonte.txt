PROCEDURE Init
ThisForm.cntSombra.lblSombra.Caption = ThisForm.Caption
ThisForm.cntSombra.lblTitulo.Caption = ThisForm.Caption

With ThisForm
	=fLerFiscal()

	Select crSigFiFis
	Go Top

	If Eof()
		Append Blank
	EndIf

	If IsEmpty(crSigFiFis.Settings)
		Replace crSigFiFis.Settings with '9600,n,8,1'
	Endif

	With This.Cmb_Hand
		.Clear
		.AddItem( '0-None    ' )
		.AddItem( '1-Xon Xoff' )
		.AddItem( '2-RTS     ' )
		.AddItem( 'Em Branco ' )
	Endwith
	With This.Cmb_Imp
		.Clear
		.AddItem( 'DATAREGIS   ' )
		.AddItem( 'SWEDA       ' )
		.AddItem( 'BEMATECH    ' )
		.AddItem( 'BEMA8501    ' )
		.AddItem( 'NDATAREGIS  ' )
		.AddItem( 'DARUMA      ' )
		.AddItem( 'S/IMPRESSORA' )
		.AddItem( 'NSWEDA      ' )
		.AddItem( 'EPSON       ' )
		.AddItem( 'Em Branco   ' )
	Endwith

	If (crSigFiFis.Imps = Space(20))
		Replace Imps With 'Em Branco' In crSigFiFis
	EndIf

	If (crSigFiFis.Hands = Space(20))
		Replace Hands With 'Em Branco' In crSigFiFis
	EndIf

	=DoDefault()

EndWith
ThisForm.Spinner1.SetFocus
ThisForm.Refresh

ENDPROC

------------------------------------------------------------
* PROCEDURE Click
------------------------------------------------------------
PROCEDURE Click
DO Case

	Case ATC('SIGFIOP5', upper(SET('PROCEDURE'))) <> 0
		Release Procedure SIGFIOP5.Prg 

	Case ATC('SIGFIOP4', upper(SET('PROCEDURE'))) <> 0
		Release Procedure SIGFIOP4.Prg 
		
	Case ATC('SIGFIOP3', upper(SET('PROCEDURE'))) <> 0
		Release Procedure SIGFIOP3.Prg 

	Case ATC('SIGFIOP0', upper(SET('PROCEDURE'))) <> 0
		Release Procedure SIGFIOP0.Prg 

	Case ATC('SIGFIOP2', upper(SET('PROCEDURE'))) <> 0
		Release Procedure SIGFIOP2.Prg 

	Case ATC('SIGFIOP1', upper(SET('PROCEDURE'))) <> 0
		Release Procedure SIGFIOP1.Prg 

	Case ATC('SIGFIOP8', upper(SET('PROCEDURE'))) <> 0
		Release Procedure SIGFIOP8.Prg 

EndCase

Do Case
	Case crSigFiFis.Imps=='SWEDA'
		Set Procedure to SIGFIOP0.Prg  Additive

	Case crSigFiFis.Imps=='BEMATECH'
		Set Procedure to SIGFIOP1.Prg  Additive

	Case crSigFiFis.Imps=='BEMA8501'
		Set Procedure to SIGFIOP2.Prg  Additive

	Case crSigFiFis.Imps=='DATAREGIS'
		Set Procedure to SIGFIOP3.Prg  Additive

	Case crSigFiFis.Imps=='NDATAREGIS'
		Set Procedure to SIGFIOP4.Prg  Additive

	Case crSigFiFis.Imps=='DARUMA'
		Set Procedure to SIGFIOP5.Prg  Additive

	Case crSigFiFis.Imps=='S/IMPRESSORA'
		Set Procedure to SIGFIOP9.Prg  Additive

	Case crSigFiFis.Imps=='NSWEDA'
		Set Procedure to SIGFIOP8.Prg  Additive

	Case crSigFiFis.Imps=='EPSON'
		Set Procedure to SIGFIOP7.Prg  Additive

EndCase

If (Alltrim(crSigFiFis.Imps) = 'Em Branco')
	Replace Imps With '' In crSigFiFis
EndIf

If (Alltrim(crSigFiFis.Hands) = 'Em Branco')
	Replace Hands With '' In crSigFiFis
EndIf

lcArq = Sys(5)+Sys(2003)+'\SetFiscal.Ini'
Select CrSigFiFis
Go Top
Copy To &lcArq. type sdf
Go Top

lcFile = GetEnv("SystemRoot") + '\system32\BEMAFI32.INI'
If File(lcFile)
	lcConfig = FileToStr(lcFile)
	lcComs   = StrExtract(lcConfig, 'Porta', Chr(13) + Chr(10), 1, 1)
	*** mena - 27/02/2015 - se porta = 9 considera USB ***
	lcPor  = Alltrim(Str(ThisForm.Spinner1.Value))
	lcNewCon = Strtran(lcConfig, 'Porta' + lcComs, IIF(lcPor='9', 'Porta=USB', 'Porta=COM' + lcPor), -1, 1, 1)
	StrToFile(lcNewCon, lcFile)
EndIf  

*** Tiago - 06/05/2015 - Atualiza DarumaFramework.xml
lcFile = Sys(5)+Sys(2003) + '\DarumaFramework.xml'
If File(lcFile)
	lcConfig = FileToStr(lcFile)
	lcComs   = StrExtract(lcConfig, '<PortaSerial>', Chr(13) + Chr(10),1,1)
	lcNewCon = Strtran(lcConfig, '<PortaSerial>' + lcComs, '<PortaSerial>COM' + Alltrim(Str(ThisForm.Spinner1.Value)) + '</PortaSerial>', -1, 1, 1)
	
	lcComs   = StrExtract(lcNewCon, '<Velocidade>', Chr(13) + Chr(10),2,1)
	lcVel = Substr(Alltrim(crSigFiFis.settings),1,At([,],Alltrim(crSigFiFis.settings))-1)
	lcNewCon = Strtran(lcNewCon, '<Velocidade>' + lcComs, '<Velocidade>' + lcVel + '</Velocidade>', -1, 1, 1)
	StrToFile(lcNewCon, lcFile)	
EndIf 

Thisform.Release()
ENDPROC
