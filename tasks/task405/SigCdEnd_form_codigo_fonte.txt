		.cmdEscolha.Visible 		   = .f.
		.cmdSalva.btnConfirmar.Visible = .f.
		.cmdSalva.btnCancelar.Visible  = .f.
	Else
		llIns = fChecaAcesso('SIGCDEND', 'INSERIR')
		llAlt = fChecaAcesso('SIGCDEND', 'ALTERAR')
		llExc = fChecaAcesso('SIGCDEND', 'EXCLUIR')
		llCop = fChecaAcesso('SIGCDEND', 'COPIARP')

		.cmdEscolha.Visible = llIns Or llAlt Or llExc Or llCop
		
		If llIns Or llAlt Or llExc Or llCop
			lnRet = (8 + Iif(llIns, 34, 0) + Iif(llAlt, 34, 0) + Iif(llExc, 34, 0) + Iif(llCop, 34, 0))
		EndIf

		.cmdEscolha.btnInserir.Visible = llIns
		.cmdEscolha.btnAlterar.Visible = llAlt
		.cmdEscolha.btnExcluir.Visible = llExc
		.cmdEscolha.btnCopiarP.Visible = llCop

		If llIns Or llAlt Or llExc Or llCop
			lnRet = .cmdEscolha.Left
			If Not llIns
				lnRet = lnRet + 34
				.cmdEscolha.btnAlterar.Left = .cmdEscolha.btnAlterar.Left - 34
				.cmdEscolha.btnExcluir.Left = .cmdEscolha.btnExcluir.Left - 34

				.cmdEscolha.btnCopiarP.Left = .cmdEscolha.btnCopiarP.Left - 34

PROCEDURE atualizatela
Lparameters pChk
Local lcDes, llMar

lcDes = crGrade.Descrs
lnCod = crGrade.Codigos

With ThisForm
	If Not Used('crSigCdCeE')
		If (.poDataMgr.Requery('crCursorCliEE') < 1)
			=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (AtualizaTela - Requery - crCursorCliEE)')
			Return 0
		EndIf

	EndIf 
	
	Update crCursorCliEE Set Refers = [] Where IsNull(Refers)

	With .grdLocais
		.RecordSource 	   = ''

		Select crGrade
		Zap In crGrade
		Locate For .f.

		.ColumnCount  	   = 2
		.RecordSource 	   = 'crGrade'
		.HeaderHeight 	   = 19
		.DeleteMark   	   = .f.
		.ReadOnly     	   = .f.
		.Scrollbars   	   =  2
		.AllowHeaderSizing = .f.
		.AllowRowSizing    = .f.

		.Column1.ControlSource = 'crGrade.Atuals'
		.Column1.Movable   = .f.
		.Column1.Resizable = .f.
		.Column1.ReadOnly  = .f.
		.Column1.Sparse    = .f.
		.Column1.Check1.AutoSize = .t.

		.Column2.ControlSource  = 'crGrade.Descrs'
		.Column2.Movable   = .f.
		.Column2.Resizable = .f.
		.Column2.ReadOnly  = .f.
	EndWith

	If InList(ThisForm.pcEscolha, 'INSERIR', 'ALTERAR') And Not Empty(ThisForm.Codigos)
		Update crCursorCliEE Set Atuals = .f.
		Update crCursorCliEE Set Atuals = .t. Where Codigos = ThisForm.Codigos

		Update crGrade Set Atuals = .f.
		Update crGrade Set Atuals = .t. Where Descrs = lcDes

		ThisForm.Codigos = 0
	EndIf
	
	llMar = .f.
	Select crCursorCliEE
	Scan
		Insert Into crGrade (Atuals, Descrs, Codigos) Values (crCursorCliEE.Atuals, crCursorCliEE.Descrs, crCursorCliEE.Codigos)
		If Not llMar And crCursorCliEE.Atuals
			llMar = .t.
		EndIf
	EndScan
	
	Select crGrade
	Go Top
	If Eof('crGrade')
 		If (ThisForm.pcEscolha = 'CONSULTAR')
			.EnabledObjects(.f.)
		Else
	 		Insert Into crCursorCliEE (Iclis, Codigos, Atuals) Values (ThisForm.IClis, fGerUniqueKey('SigCdCeE'), .t.)
			Insert Into crGrade (Atuals, Descrs) Values (.t., '')
			Go Top In crCursorCliEE
			Go Top In crGrade

			.pcEscolha = 'INSERIR'
			.EntraInserindo = .t.
			.EnabledObjects(.t.)
		EndIf
	Else
		.pcEscolha = 'CONSULTAR'
		.EnabledObjects(.f.)
	EndIf

	If Not llMar And Not Eof('crGrade') And (ThisForm.pcEscolha <> 'CONSULTAR')
		Replace Atuals With .t. In crGrade
		Update crCursorCliEE Set Atuals = .t. Where Descrs = crGrade.Descrs
	EndIf

	Select crGrade
	Go Top
	If pChk
		Locate For Atuals
	Else
		Locate For Descrs = lcDes
	EndIf

	.Refresh
	.grdLocais.Refresh
	.grdLocais.Column2.SetFocus
EndWith
* Fim **************************************************************************************
ENDPROC

* PROCEDURE m_confirmar
------------------------------------------------------------
PROCEDURE m_confirmar

Local lcDes
private lcCep, lcEnd, lcBai, lcCod
 
lcDes = crGrade.Descrs
lcCep = thisform.getCeps.Value
lcEnd = thisform.getEndes.value  
lcBai = thisform.getBairs.value
lcCod = crCursorCliEE.codigos

If (ThisForm.pcEscolha = 'INSERIR')
	ThisForm.EntraInserindo = .f.
	
	lcQuery = [Select Descrs ] + ;
			    [From SigCdCeE ] + ;
			   [Where IClis = '] + ThisForm.IClis + [' And ] + ;
			   		 [Descrs = '] + lcDes + [']

	If (ThisForm.poDataMgr.SqlExecute(lcQuery, 'crDuplicados') < 1)
		=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (m_Confirmar - crDuplicados)')
		Return 0
	EndIf

	Select crDuplicados
	Go Top

	If Not Eof('crDuplicados')
		Replace Descrs With Space(50) In crCursorCliEE
		Replace Descrs With Space(50) In crGrade

		=MessageBox('Atenção!!! A Descrição [' + Alltrim(crDuplicados.Descrs) + '] Já Existe!!!', ;
					16, 'Descrição do Local Duplicada')
		Return 0
	EndIf
EndIf

If InList(ThisForm.pcEscolha, 'INSERIR', 'ALTERAR')

	* Verifica se endereço duplicado - mena - 23/03/2012 * 
	If thisform.chkendds=1					&&Verifica se endereço duplicado
		ThisForm.poDataMgr.SqlExecute('Select iclis From SigCdCee Where ceps = ?lcCep and endes = ?lcEnd and bairs = ?lcBai and codigos <> ?lcCod', 'crTmp')
		If (Reccount('crTmp') > 0)
			lcCta = ''
			SELECT crTmp
			SCAN
				lcCta = lcCta + iclis + ' '
			ENDSCAN
			=MessageBox('Endereço Duplicado. Contas: ' + lcCta, 0+64, 'Atenção!!!')
			Do Form SigOpSen With 'LIBDUPEND', 'Libera Endereço Duplicado', '' To _Retorno
			llOk = ( Left( _Retorno, 1 ) = "*" )
			If !llOk
				This.getCeps.SetFocus 
				Return (0)
			endif
		endif
	EndIf
	********************************************************************

	Replace Descrs With lcDes In crCursorCliEE

	Select crCursorCliEE
	Scan
		If Empty(crCursorCliEE.Descrs)
			Delete In crCursorCliEE
		Else
			If (ThisForm.pcEscolha = 'INSERIR')
				Replace Atuals with .f. In crCursorCliEE
			EndIf
		EndIf
	EndScan

	If (ThisForm.pcEscolha = 'INSERIR')
		Update crCursorCliEE Set Atuals = .t. Where Descrs = lcDes
	EndIf
EndIf

Select crCursorCliEE
Scan
	lcQuery = [Delete From SigCdClt Where IClis = '] + crCursorCliEE.IClis + [']

	If (ThisForm.poDataMgr.SqlExecute(lcQuery, '') < 1)
		=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (Delete - SigCdClt)')
		Return 0
	EndIf
EndScan

&& se o cursor CRSigCdCeE estiver aberto, indica que foi chamado 
&& pela tela de cadastro de clientes, onde sera gravado...

If Not Used('crSigCdCeE') 
	If Not ThisForm.poDataMgr.Update('crCursorCliEE')
		=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (Update - crCursorCliEE)')
		Return 0

	EndIf
EndIf 

ThisForm.AtualizaTela

Select crGrade
Go Top
Locate For Descrs = lcDes

ThisForm.Refresh
* Fim ********************************************************************************************
ENDPROC

PROCEDURE m_copiarp
Select crGrade
Locate For Descrs = Padr('PRINCIPAL', 50)
If Eof('crGrade')
	lcQuery = [Select Ceps, Paises, Endes, Nums, Compls, Bairs, Cidas, Estas, Ddds, Tel1s, Tel2s, Contato ] + ;
			    [From SigCdCli ] + ;
			   [Where IClis = '] + ThisForm.IClis + [']

	If (ThisForm.poDataMgr.SqlExecute(lcQuery, 'crSigCdCeECli') < 1)
		=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (m_CopiarP - crSigCdCeECli)')
		Return 0
	EndIf
	Select crSigCdCeECli
	Go Top
	If Eof('crSigCdCeECli') And Used('crSigCdCli')
		lnRec = Recno('crSigCdCli')
		Go Top In crSigCdCli
		Go Bottom In crSigCdCli
		Go lnRec In crSigCdCli

		Select Ceps, Paises, Endes, Nums, Compls, Bairs, Cidas, Estas, Ddds, Tel1s, Tel2s, Contato ;
		  From crSigCdCli ;
		 Where IClis = Padr(ThisForm.IClis, 10) ;
		  Into Cursor crSigCdCeECli ReadWrite
	EndIf
	Select crSigCdCeECli
	Go Top
	
	If Not Eof('crSigCdCeECli') And Not Empty(crSigCdCeECli.Endes)
		Update crCursorCliEE Set Atuals = .f.
		Update crGrade Set Atuals = .f.

		Select crGrade
		Locate For Empty(Descrs)
		If Eof('crGrade')
			Insert Into crGrade (Atuals, Descrs) Values (.t., 'PRINCIPAL')
		Else
			Replace Atuals With .t., ;
					Descrs With 'PRINCIPAL' In crGrade
		EndIf

		Insert Into crCursorCliEE (Iclis, Codigos, Atuals, Descrs, Ceps, Paises, Endes, Nums, ;
								   Compls, Bairs, Cidas, Estas, Ddds, Tel1s, Tel2s, Contatos) ;
						   Values (ThisForm.IClis, fGerUniqueKey('SigCdCeE'), .t., 'PRINCIPAL', ;
						   		   crSigCdCeECli.Ceps, crSigCdCeECli.Paises, crSigCdCeECli.Endes, ;
						   		   crSigCdCeECli.Nums, crSigCdCeECli.Compls, crSigCdCeECli.Bairs, ;
						   		   crSigCdCeECli.Cidas, crSigCdCeECli.Estas, crSigCdCeECli.Ddds, ;
						   		   crSigCdCeECli.Tel1s, crSigCdCeECli.Tel2s, crSigCdCeECli.Contato)
		ThisForm.grdLocais.Refresh

		ThisForm.pcEscolha = 'INSERIR'

		ThisForm.m_Confirmar
	EndIf
EndIf

Select crGrade
ThisForm.Refresh

ThisForm.grdLocais.Column2.SetFocus
ThisForm.grdLocais.Refresh

* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
Lparameters pGru, pCon, pDes, pEsc, pCnx, pCod

ThisForm.cntSombra.lblSombra.Caption = ThisForm.Caption
ThisForm.cntSombra.lblTitulo.Caption = ThisForm.Caption


* Parâmetros : pGru - Grupo de Conta (Obrigatório)
*              pCon - Código da Conta (Obrigatório)
*              pDes - Nome da Conta (Default = Brancos)
*              pEsc - Tipo de Opção : 'CONSULTAR', 'INSERIR', 'ALTERAR' (Default = 'CONSULTAR') 
*              pCnx - Conexão a Ser Utilizada (Obrigatório)
*			   pCod - Código do Local a Procurar (Default = 0)
*              pVal - Indica que o Cadastro Foi Chamado a Partir de um "Valid" (.t. ou .f.)

Local iclis,codigos,atuals,descrs,ceps,paises,endes,nums,compls, ;
		bairs,cidas,estas,ddds,tel1s,tel2s,contatos,refers

If (Type('pGru') <> 'C') Or Empty(pGru)
	=Messagebox('O Parâmetro de Grupo Passado Não É Valido...', 0+48, 'Parâmetros Inválidos!')
	Return .f.
EndIf

If (Type('pCon') <> 'C') Or Empty(pCon)
	=Messagebox('O Parâmetro de Conta Passado Não É Valido...', 0+48, 'Parâmetros Inválidos!')
	Return .f.
Else
	ThisForm.IClis = Padr(pCon, 10)
EndIf

If (Type('pDes') <> 'C')
	pDes = Space(50)
EndIf

ThisForm.pcEscolha = 'CONSULTAR'
If (Type('pEsc') = 'C') And InList(pEsc, 'INSERIR', 'ALTERAR')
	ThisForm.pcEscolha = pEsc
EndIf

ThisForm.Codigos = Iif(Type('pCod')='N', pCod, 0)

If (Type('pCnx') <> 'O')
	=Messagebox('O Parâmetro de Conexão Passado Não É Valido...', 0+48, 'Parâmetros Inválidos!')
	Return .f.
EndIf

If Used('crCursorCliEE')
	=Messagebox('O Cursor crCursorCliEE Já Existe!!! Impossível Prosseguir!!!', 0+48, 'Erro de Utilização!')
	Return .f.
EndIf

Do Case 
	Case (ThisForm.pcEscolha <> 'CONSULTAR') Or ((ThisForm.pcEscolha = 'CONSULTAR') And Empty(ThisForm.Codigos))
		lcQryCliEE = [Select * From SigCdCeE Where IClis = '] + ThisForm.IClis + [']

	Case (ThisForm.pcEscolha = 'CONSULTAR') And Not Empty(ThisForm.Codigos)
		lcQryCliEE = [Select * From SigCdCeE ] + ;
					  [Where IClis = '] + ThisForm.IClis + [' And ] + ;
					  		[Codigos = ] + Alltrim(Str(ThisForm.Codigos, 10))
EndCase

With ThisForm
	.poDataMgr = pCnx

	If (.poDataMgr.pnIdconn > 0)
		lnTam = Alen(.poDataMgr.pfSqlTabela)
		lnCnx = 0

		For I = 1 To lnTam
			If (Upper(.poDataMgr.pfSqlTabela(I).pcNomeCursor) = Upper('crCursorCliEE'))
				lnCnx = I
			EndIf
		Next

		If (lnCnx = 0)
			With .poDataMgr
				.AddCursor('SigCdCeE', 'Codigos', 'crCursorCliEE', '', '', lcQryCliEE)
			EndWith

			lnTam = Alen(.poDataMgr.pfSqlTabela)
			lnCnx = 0

			For I = 1 To lnTam
				If (Upper(.poDataMgr.pfSqlTabela(I).pcNomeCursor) = Upper('crCursorCliEE'))
					lnCnx = I
				EndIf
			Next
		EndIf

		.poDataMgr.pfSqlTabela(lnCnx).pcStrquery = lcQryCliEE

		If (.poDataMgr.Requery('crCursorCliEE') < 1)
			=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (Init - Requery - crCursorCliEE)')
			Return .f.

		EndIf

		Select crCursorCliEE
		Zap In crCursorCliEE

		If Used('crSigCdCeE')
			Select crSigCdCeE
			Locate 

			Scan 
				Scatter Memo Memvar
				
				If Not Empty(m.codigos) And Not Empty(m.iclis)
					Insert Into crCursorCliEE From Memvar

				EndIf 

			EndScan 

		EndIf 

		** Incluido para pegar flags de configuração - mena - 23/03/2012 **
		.poDataMgr.SqlExecute([Select CepObris, ChkEndds, ChkEndrs From SigCdGcr Where codigos = '] + pGru + ['], 'crCursorGccr')
		.CepObris = Iif(Eof('crCursorGccr'), 2, crCursorGccr.CepObris)
		.ChkEndDs = Iif(Eof('crCursorGccr'), 2, crCursorGccr.ChkEndds)
		.ChkEndRs = Iif(Eof('crCursorGccr'), 2, crCursorGccr.ChkEndrs)

		.TipoCep  = [1]
		.TipoPais = .f.
		.EntraInserindo = .f.
		
		.getCodigos.Value = pCon
		.getNomes.Value   = pDes
		.getCodigos.Refresh
		.getNomes.Refresh
		
		.getCeps.ControlSource     = 'crCursorCliEE.Ceps'
		.getPaises.ControlSource   = 'crCursorCliEE.Paises'
		.getEndes.ControlSource    = 'crCursorCliEE.Endes'
		.getNums.ControlSource     = 'crCursorCliEE.Nums'
		.getCompls.ControlSource   = 'crCursorCliEE.Compls'
		.getBairs.ControlSource    = 'crCursorCliEE.Bairs'
		.getCidas.ControlSource    = 'crCursorCliEE.Cidas'
		.getEstas.ControlSource    = 'crCursorCliEE.Estas'
		.getDdds.ControlSource     = 'crCursorCliEE.Ddds'
		.getTel1s.ControlSource    = 'crCursorCliEE.Tel1s'
		.getTel2s.ControlSource    = 'crCursorCliEE.Tel2s'
		.getContatos.ControlSource = 'crCursorCliEE.Contatos'
		.getRefers.ControlSource   = 'crCursorCliEE.Refers'

		.getCeps.Refresh
		.getPaises.Refresh
		.getEndes.Refresh
		.getNums.Refresh
		.getCompls.Refresh
		.getBairs.Refresh
		.getCidas.Refresh
		.getEstas.Refresh
		.getDdds.Refresh
		.getTel1s.Refresh
		.getTel2s.Refresh
		.getContatos.Refresh
		.getRefers.Refresh

		.AjustarBotoes

		.AtualizaTela(.t.)

		=DoDefault()

		ThisForm.grdLocais.Column2.SetFocus
	Else
		=Messagebox('Impossível Efetuar Conexão Com o Servidor de Banco de Dados...', 0+48, 'CreateObject')
		Return .f.
	EndIf
EndWith

* Fim **************************************************************************************
ENDPROC

------------------------------------------------------------
* PROCEDURE Release
------------------------------------------------------------
PROCEDURE Release
If Used('crCursorCliEE')
	If Used('crSigCdCeE')
		Local iclis,codigos,atuals,descrs,ceps,paises,endes,nums,compls, ;
				bairs,cidas,estas,ddds,tel1s,tel2s,contatos,refers
	
		Select crCursorCliEE
		Locate 

		Scan 
			Scatter Memo Memvar
			
			If Not Empty(m.codigos) And Not Empty(m.iclis)
				Select crSigCdCeE
				Locate For codigos==m.codigos
				
				If Eof('crSigCdCeE')
					Insert Into crSigCdCeE From Memvar

				Else
