PROCEDURE Init
LPARAMETERS _acesso

ThisForm.Busca=1

WITH Thisform
	.poDatamgr = CreateObject('fSqlConector',ThisForm.Name)

	If Not .poDatamgr.pnIdconn>0
		Messagebox('Impossivel Efetuar Conexão com Servidor de Banco de Dados.',(0+48),goSistema.TituloSistema)
		Return(.f.)
	Endif

	With .poDatamgr
		.AddCursor('SigBaOco','Codigos','crSigBaOco','Codigos',ThisForm.pagina.lista.grade)
		.pfSqlTabela(1).pColuna('Tipos','','','Tipo',30,.t.)
		.pfSqlTabela(1).pColuna('Codigos','','','Código',80,.t.)
		.pfSqlTabela(1).pColuna('Descrs','','','Descrição',290,.t.)
		.pfSqlTabela(1).pColuna('Apuras','','','Apuras',45,.t.)
		.pfSqlTabela(1).pColuna('Operacaos','','','Operação',60,.t.)
	EndWith
Endwith	

=DoDefault()


ENDPROC

=DoDefault()

ENDPROC

------------------------------------------------------------
* PROCEDURE msv_procurar
------------------------------------------------------------
PROCEDURE msv_procurar
WITH ThisForm
	WITH .pagina.dados
		lcCodigo=.GetCodigos.Value
		lcTipo=.getTipos.Value
		IF ! EMPTY(.getTipos.Value)
			SELECT crSigBaOco
			LOCATE FOR Tipos = lcTipo .AND. codigos = ALLTRIM(lcCodigo)
			ThisForm.Busca=1
		ELSE
			SELECT crSigBaOco
			SET ORDER TO Codigos
			SET NEAR on
			SEEK lcCodigo
			SET NEAR off
			Thisform.pagina.lista.grade.columns(2).setfocus
			ThisForm.Busca=2
		Endif
	ENDWITH
EndWith

ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Lista.Grupo_op.Click
------------------------------------------------------------
PROCEDURE Pagina.Lista.Grupo_op.Click
LPARAMETERS opcao

If !DoDefault( Opcao )
	ThisForm.mAtivaPagina1
	Return
Endif

Select crSigBaOco

Do Case
	Case Inlist( ThisForm.pcEscolha, 'PROCURAR', 'INSERIR' )
		ThisForm.Pagina.Dados.getTipos.Setfocus	

	Case ThisForm.pcEscolha = 'ALTERAR'
		ThisForm.Pagina.Dados.getDescrs.Setfocus	

	Case Inlist( ThisForm.pcEscolha, 'CONSULTAR', 'EXCLUIR' )
		ThisForm.Pagina.Dados.Grupo_Salva.Salva.Setfocus
Endcase

Select crSigBaOco
ThisForm.Pagina.Dados.Refresh

ENDPROC

	  ThisForm.Pagina.Dados.getCodigos.SetFocus
	  Return
   Endif
ENDIF

=DoDefault()
ThisForm.mAtivaPagina1
