PROCEDURE msv_procurar
Local lcBusca

If !IsEmpty(ThisForm.Pagina.Dados.getcods.Value)
	ThisForm.poDataMgr.LocateCursor('crSigCdEmb','cods',ThisForm.Pagina.Dados.Getcods.Value)
EndIf

ENDPROC

------------------------------------------------------------
* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
LPARAMETERS lcacesso

With ThisForm
	.poDatamgr = CreateObject('fSqlConector')
	With .poDatamgr
		.AddCursor('SigCdEmb','Cods','crSigCdEmb','',ThisForm.pagina.lista.grade)
		.pfSqlTabela(1).pColuna('Cods','','','Código',75,.t.)
		.pfSqlTabela(1).pColuna('descs','','','Descrição',297,.t.)
		.pfSqlTabela(1).pColuna('Pesos','','','Peso',107,.t.)
	EndWith
	= DoDefault()
	.LockScreen = .f.
EndWith

ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Lista.Grupo_op.Click
------------------------------------------------------------
PROCEDURE Pagina.Lista.Grupo_op.Click
LParameters opcao

If !dodefault( Opcao )
	ThisForm.mAtivapagina1
	Return
Endif

Do Case
	Case Inlist( ThisForm.pcEscolha, 'INSERIR', 'PROCURAR' )
		If ThisForm.pcEscolha = "INSERIR"
			Replace Multis with 1
		Endif
		ThisForm.Pagina.Dados.Refresh
		ThisForm.Pagina.Dados.GetCods.SetFocus
	Case ThisForm.pcEscolha = 'ALTERAR'
		ThisForm.Pagina.Dados.GetDescs.SetFocus
	Case Inlist( ThisForm.pcEscolha, 'EXCLUIR', 'CONSULTAR' )
		ThisForm.Pagina.Dados.Grupo_Salva.Salva.SetFocus
Endcase

ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
If ThisForm.pcEscolha = 'INSERIR'
	If IsEmpty( crSigCdEmb.Cods )
		Messagebox( 'Código não pode ficar em branco!', 48, '' ) 
		ThisForm.Pagina.Dados.GetCods.SetFocus
		Return
	Endif
	If ThisForm.poDataMgr.ChkRegister('SigCdEmb','Cods',ThisForm.Pagina.Dados.GetCods.Value)
		Messagebox( 'Código já Cadastrado!', 48, '' ) 
		ThisForm.Pagina.Dados.GetCods.SetFocus
		Return
	Endif
Endif

=DoDefault()
ThisForm.mAtivapagina1

ENDPROC

If ThisForm.plCancelar
	ThisForm.mAtivapagina1
Endif

ENDPROC
