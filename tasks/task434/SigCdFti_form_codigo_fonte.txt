* PROCEDURE msv_procurar
------------------------------------------------------------
PROCEDURE msv_procurar
Local lcGLinhas, lcGDescrs

lcGCodigo = Alltrim(ThisForm.Pagina.Dados.Get_Código.Value)
lcGDescrs = Alltrim(ThisForm.Pagina.Dados.Get_Descrição.Value)

Select CrSigPrFti
Set Near On

Do Case

	Case !Empty(lcGCodigo)
		Set Order To Cods
		Seek lcGCodigo
		ThisForm.Busca = 1

	Case !Empty(lcGDescrs)
		Set Order To Descs
		Seek lcGDescrs
		ThisForm.Busca = 2

EndCase

Set Near Off
ENDPROC

PROCEDURE Init
LParameters _Acesso

ThisForm.Busca = 1

With ThisForm
	.poDataMgr = CreateObject('fSqlConector',.name)
	With .poDataMgr
		.AddCursor('SigPrFti', 'cods', 'CrSigPrFti', '', ThisForm.Pagina.Lista.Grade)
		.pfSqlTabela(1).pColuna('cods'   , '', ''              , 'Código'   ,  50, .T.)
		.pfSqlTabela(1).pColuna('descs'  , '', ''              , 'Descrição', 270, .T.)
		.pfSqlTabela(1).pColuna('valors' , '', '999,999,999.99', 'Valor'    , 120, .F.)
		.pfSqlTabela(1).pColuna('moedas' , '', ''              , 'Moeda'    ,  45, .F.)
		.pfSqlTabela(1).pColuna('acrescs', '', '99999.999999'       , 'Acréscimo',  75, .F.)

		.AddCursor('SigPrFto','Cidchaves','CrSigPrFto','','')
	EndWith
	=DoDefault(_Acesso)
EndWith

If ThisForm.poDataMgr.SqlExecute([Select cmoes,dmoes From SigCdMoe ],'CrSigCdMoe') < 1
	MessageBox('Favor reinicializar o processo.',16,'Falha na Conexão')
	Return(.F.)
EndIf
Select CrSigCdMoe
Index On cmoes Tag CMoes

lcSql = [Select CgruCfxs From SigCdPac ]
If ThisForm.poDataMgr.SqlExecute(lcSql,'TmpParac') < 1
	MessageBox('Favor reinicializar o processo.',16,'Falha na Conexão (CrSigCdPac)')
	Return(.F.)
EndIf

lcSql = [Select Cgrus, Dgrus, Coefs From SigCdGrp ]
If ThisForm.poDataMgr.SqlExecute(lcSql,'CrSigCdGrp') < 1
	MessageBox('Favor reinicializar o processo.',16,'Falha na Conexão (CrSigCdGrp)')
	Return(.F.)
EndIf
Select CrSigCdGrp
Index On Cgrus Tag Cgrus

lcSql = [Select Cpros,dpros,pVens From SigCdPro Where Cgrus = ']+TmpParac.CgruCfxs+[']
If ThisForm.poDataMgr.SqlExecute(lcSql,'CrSigCdPro') < 1
	MessageBox('Favor reinicializar o processo.',16,'Falha na Conexão (CrSigCdPro)')
	Return(.F.)
EndIf
Select CrSigCdPro
Index On cpros Tag Cpros

Select CrSigPrFti
Set Order To Cods
Go Top

ThisForm.Pagina.Lista.Grade.Refresh
ThisForm.Pagina.Lista.Grade.Column1.SetFocus

ThisForm.LockScreen = .T.
=fCarregaGrade(ThisForm.Pagina.Lista.Grade)
ThisForm.LockScreen = .F.
ENDPROC

	If ThisForm.Podatamgr.Sqlexecute(lcSql,'TmpFtioc')  <1
		MessageBox('Favor reinicializar o processo !!!',16,'Falha na Conexão (SigPrFto)')

	EndIf
	
	Select TmpFtioC
	Scan
		Scatter Memvar
		If Not Empty(m.Cgrus)
			=Seek(m.Cgrus,'CrSigCdGrp','Cgrus')
			m.Dgrus = CrSigCdGrp.Dgrus
			Insert Into CsGrade (Cgrus, CPros, Dgrus, Coefs, Ordem ) Values ;
								(m.Cgrus, m.CPros, m.Dgrus, m.Coefs, m.Ordem)

									(m.Cgrus, m.CPros, m.Dgrus, m.Coefs, m.Ordem)
			Else
				=Seek(m.cpros,'CrSigCdPro','Cpros')
				m.Dpros = CrSigCdPro.Dpros
				Insert Into CsCusto (cpros, Dpros, Seqs, Coefs ) Values ;
									(m.Cpros, m.Dpros, m.Seqs, m.Coefs)
			EndIf 					
		EndIf
	EndScan
EndIf

	Append Blank

	Append Blank
EndIf

Go top

	.RecordSource = [CsGrade]
	.column1.ControlSource = [CsGrade.Cgrus]
	.column2.ControlSource = [CsGrade.Dgrus]
	.column3.ControlSource = [CsGrade.Coefs]
	.column4.ControlSource = [CsGrade.CPros]
	.column5.ControlSource = [CsGrade.Ordem]
	.Refresh()
EndWith 

With This.Parent.Parent.Dados
	If ThisForm.pcEscolha = 'PROCURAR'
		.Get_Ds_Moeda.Value = ''
	Else
		.Get_Ds_Moeda.Value = Iif(Seek(CrSigPrFti.moedas ,'CrSigCdMoe','CMoes'),CrSigCdMoe.dmoes,'')
	EndIf
	
	Do Case

		Case InList(ThisForm.pcEscolha,'INSERIR','PROCURAR')
			.Get_Código.SetFocus

		Case InList(ThisForm.pcEscolha,'CONSULTAR','EXCLUIR')
			.Grupo_Salva.Salva.SetFocus

		Case ThisForm.pcEscolha = 'ALTERAR'
			.Get_Descrição.SetFocus

------------------------------------------------------------
* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
Do Case
	Case Empty(This.Parent.Parent.Get_Código.Value)
		MessageBox('Código Inválido !!!',48,'')
		This.Parent.Parent.Get_Código.SetFocus
		Return

	Case Empty(This.Parent.Parent.Get_Descrição.Value)
		MessageBox('Descrição Inválida !!!',48,'')
		This.Parent.Parent.Get_Descrição.SetFocus
		Return

	Case Empty(This.Parent.Parent.Get_cd_moeda.Value) And Not Empty(This.Parent.Parent.Get_Valor.Value)
		MessageBox('Obrigatório Informar Moeda!!!',48,'')
		This.Parent.Parent.Get_Cd_Moeda.SetFocus
		Return

EndCase

If ThisForm.pcEscolha = 'INSERIR'
	If ThisForm.poDataMgr.CursorQuery('SigPrFti',,'cods',CrSigPrFti.cods)
		MessageBox('Código de Feitio já cadastrado !!!',48,'')
		ThisForm.Pagina.Dados.Get_Código.SetFocus
		Return
	EndIf
EndIf
If InList(ThisForm.PcEscolha,'INSERIR','ALTERAR')
	Select Cgrus, CPros, sum(1) From CsGrade Where Cgrus <> ' ' Group by Cgrus, CPros Into Cursor CsDuplica having sum(1) > 1
	Select CsDuplica
	If Reccount() > 0
		MessageBox('Existem Grupos/Produtos Configurados em duplicidade!!!',0+16,'')
		Return .f.
	EndIf
	Select Cpros,sum(1) From CsCusto Where Cpros <> ' ' Group by Cpros Into Cursor CsDuplica having sum(1) > 1
	Select CsDuplica
	If Reccount() > 0
		MessageBox('Existem Custos Fixo Configurados em duplicidade!!!',0+16,'')
		Return .f.
	EndIf
	If CrSigPrFti.FaixaFs > 0 And Not Between(CrSigPrFti.Acrescs,CrSigPrFti.FaixaIs,CrSigPrFti.FaixaFs)
		MessageBox('O multiplicador tem que estar dentro da Faixa!!!!',0+64,'')
		Return .f.
	Endif

	If CrSigPrFti.Acrescs <> 0 Or CrSigPrFti.Valors <> 0
		Select [ ] as Agrupar,Sum(1) as Tot From Csgrade Where Cgrus <> ' ' And Coefs <> 0 ;
			Into Cursor ChkDados group by 1
		Select ChkDados
		Go top
		If Tot > 0
			MessageBox('o Campo Multiplicador Ou Valor não podem ser informados '+Chr(13)+;
						'em conjunto com grade de custos',0+16,'')
			Return .f.
		Else
			Select [ ] as Agrupar,sum(1) as Tot From CsCusto Where Cpros <> ' ' Into Cursor ChkDados group by 1
			Select ChkDados
			Go top
			If Tot > 0
				MessageBox('o Campo Multiplicador Ou Valor não podem ser informados '+Chr(13)+;
							'em conjunto com grade de custos',0+16,'')
				Return .f.
			Endif
		Endif
	Endif
EndIf

If InList(ThisForm.pcEscolha, 'INSERIR', 'ALTERAR', 'EXCLUIR')
	llTaOk = .t.

	If InList(ThisForm.pcEscolha, 'ALTERAR', 'EXCLUIR')
		lcCod = crSigPrFti.Cods

		If (ThisForm.pcEscolha = 'EXCLUIR')
			lcQuery = [Delete From SigPrFti Where Cods = '] + lcCod + [']
			If (ThisForm.poDataMgr.SqlExecute(lcQuery, '') < 1)
				=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (Delete - SigPrFti)')
				llTaOk = .f.
			EndIf
		EndIf

		If llTaOk
			lcQuery = [Delete From SigPrFto Where Cods = '] + lcCod + [']
			If (ThisForm.poDataMgr.SqlExecute(lcQuery, '') < 1)
				=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (Delete - SigPrFto)')
				llTaOk = .f.
			EndIf
		EndIf
	EndIf

	If (ThisForm.pcEscolha <> 'EXCLUIR')
		Select CrSigPrFti
		Replace usuars With Usuar,;
				datars With Date() in CrSigPrFti

		Select CsGrade
		Go Top
		Scan
			Scatter Memvar
			If (Not Empty(Cgrus) And Coefs <> 0) or (Ordem <> 0 And Coefs <> 0)
				Insert Into CrSigPrFto (Cods, Cgrus, CPros, Coefs, cidChaves, Ordem ) Values ;
					(CrSigPrFti.Cods, m.Cgrus, m.CPros, m.Coefs, fUniqueIds(), m.Ordem)
			EndIf
		EndScan
		Select CsCusto
		Go Top
		Scan
			Scatter Memvar
			If Not Empty(Cpros) And Coefs <> 0
				Insert Into CrSigPrFto (Cods, Cpros, Seqs, Coefs, cidChaves ) Values ;
					(CrSigPrFti.Cods, m.Cpros, m.Seqs, m.Coefs, fUniqueIds())
			EndIf
		EndScan
	
		If llTaOk
			If Not ThisForm.poDataMgr.Update('crSigPrFti')
				=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (Update - crSigPrFti)')
				llTaOk = .f.
			EndIf
		EndIf

		If llTaOk
			If Not ThisForm.poDataMgr.Update('CrSigPrFto')
				=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (Update - CrSigPrFto)')
				llTaOk = .f.
			EndIf
		EndIf
	EndIf

	If llTaOk
		If (ThisForm.poDataMgr.Commit() < 1)
			=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (Commit)')
			llTaOk = .f.
		EndIf
	EndIf

	If Not llTaOK
		ThisForm.poDataMgr.RollBack()
		Return 0
	EndIf
Else
	=DoDefault()
EndIf

ThisForm.mAtivaPagina1

ENDPROC
