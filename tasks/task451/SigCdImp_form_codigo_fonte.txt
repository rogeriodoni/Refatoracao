PROCEDURE Init
lcQryCImp = [Select * From SigCdmp]

With ThisForm
	.poDataMgr = CreateObject('fSqlConector', ThisForm.Name)

	If (.poDataMgr.pnIdconn > 0)
		With .poDataMgr
			.AddCursor('SigCdmp', 'cIdChaves', 'crSigCdmp', '', ThisForm.Pagina.Lista.Grade, lcQryCImp)
			.pfSqlTabela(1).pColuna('Impres', '', '', 'Impressoras', 280, .t.)
			.pfSqlTabela(1).pColuna('MaxCols', '', '', 'Máximo de Colunas', 130, .t.)
		EndWith
		
		Select crSigCdmp
		Go Top

		ThisForm.Pagina.Lista.Grade.Column1.SetFocus

		=DoDefault()

		ThisForm.Pagina.Lista.Grade.Column1.SetFocus
	Else
		=Messagebox('Impossivel Efetuar Conexão com Servidor de Banco de Dados...', 0+48, 'CreateObject')
		Return .f.
	EndIf
EndWith

lcQryTeti = [Select * ] + ;
		    [From SigCdTpe ] +;
		    [Where nSituas = 1 ]

If (ThisForm.poDataMgr.SqlExecute(lcQryTeti, [crSigCdTpe]) < 1)
	=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (crSigCdTpe)])
	Return 0
EndIf

Select crSigCdTpe
Append Blank 

* Fim **************************************************************************************


ENDPROC
