PROCEDURE Init
lcQryOpt = [Select * From SigCdRpo ]

With ThisForm
	.poDataMgr = CreateObject('fSqlConector', ThisForm.Name)

	If (.poDataMgr.pnIdconn > 0)
		With .poDataMgr
			.AddCursor('SigCdRpo', 'cIdChaves', 'crSigCdRpo', '', ThisForm.Pagina.Lista.Grade, lcQryOpt)
			.pfSqlTabela(1).pColuna('Dopps', '', '', 'Operação', 150, .t.)
			.pfSqlTabela(1).pColuna('TpOps', '', '', 'Tipo de Material', 120, .t.)
			.pfSqlTabela(1).pColuna('Cgrus', '', '', 'Grp.Produto', 80, .t.)
			.pfSqlTabela(1).pColuna('opers', '', '', 'E/S', 30, .f.)
			.pfSqlTabela(1).pColuna('mercs', '', '', 'Gde.Grupo', 70, .f.)
			.pfSqlTabela(1).pColuna('Grupos', '', '', 'Grupo', 80, .f.)
			.pfSqlTabela(1).pColuna('Contas', '', '', 'Conta Destino', 80, .f.)
			.pfSqlTabela(1).pColuna('ContaOrig', '', '', 'Conta Origem', 80, .f.)
			.pfSqlTabela(1).pColuna('CodCors', '', '', 'Cor', 44, .f.)
		EndWith

		=DoDefault()
		
		CREATE Cursor TmpOpe (Dopps c(20))
		INDEX ON Dopps TAG Dopps
		
		If ThisForm.poDataMgr.SqlExecute([Select Dopes From SigCdOpe ],'CrOpe') < 1
			MessageBox('Favor reinicializar o processo !!!',16,'Falha na Conexão (CrOpe)')
			Return(.F.)
		EndIf
		
		Select CrOpe
		Scan
			Insert Into TmpOpe (Dopps ) Values (CrOpe.Dopes )
		EndScan

		If ThisForm.poDataMgr.SqlExecute([Select Dopps From SigCdOpd ],'Cropp') < 1
			MessageBox('Favor reinicializar o processo !!!',16,'Falha na Conexão (Cropp)')
			Return(.F.)
		EndIf
		
		Select CrOpp
		Scan
			Insert Into TmpOpe (Dopps ) Values (Cropp.dopps )
		EndScan

		ThisForm.Pagina.Lista.Grade.Column1.SetFocus
	Else
		=Messagebox('Impossivel Efetuar Conexão com Servidor de Banco de Dados...', 0+48, 'CreateObject')
		Return .f.
	EndIf
EndWith
* Fim **************************************************************************************


ENDPROC

PROCEDURE Pagina.Lista.Grupo_op.Click
LPARAMETERS opcao
If !dodefault( Opcao )
	ThisForm.mAtivapagina1
	Return
Endif

lcSql = [Select Dgrus From SigCdGrp Where Cgrus = ']+CrSigCdRpo.Cgrus+[']
If ThisForm.Podatamgr.Sqlexecute(lcSql,'LocalGru') < 0
	MessageBox('Favor reinicializar o processo !!!',16,'Falha na Conexão (LocalGru)')
	Return(.F.)
Endif
Select crSigCdRpo

ThisForm.Pagina.Dados.Get_ds_grupo.Value = LocalGru.Dgrus

* posiciona o cursor de acordo com a opção
Do Case
	Case Inlist( ThisForm.pcEscolha, 'INSERIR', 'PROCURAR' )
		thisform.pagina.dados.get_Dopp.setfocus	
	Case ThisForm.pcEscolha = 'ALTERAR'
		thisform.pagina.dados.get_Dopp.setfocus
	Case Inlist( ThisForm.pcEscolha, 'CONSULTAR', 'EXCLUIR' )
		thisform.pagina.dados.grupo_salva.salva.setfocus
Endcase

ThisForm.Pagina.Dados.Refresh

ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
With ThisForm.Pagina.Dados
	If InList(ThisForm.pcEscolha,'INSERIR','ALTERAR')
		If IsEmpty(.Get_Dopp.Value)
			MessageBox( 'Obrigatório Informar o Operação de Produção.', 48, '' )
			.Get_Dopp.SetFocus
			Return
		EndIf
		If IsEmpty(.Get_TpOp.Value) And IsEmpty(.Get_Cgru.Value) AND IsEmpty(.Get_gde.Value) And IsEmpty(.Get_Grupos.Value)
			MessageBox( 'Obrigatório Informar o Tipo de Material Ou Grupo de Produto Ou Grande Grupo Ou Grupo Conta', 48, '' )
			.Get_Gde.SetFocus
			Return
		EndIf
		
		If ThisForm.pcEscolha = 'INSERIR'
			Replace cIdChaves With fUniqueIds() In crSigCdRpo
		Endif

		lcQuery = [Select cIdChaves ] + ;
				    [From SigCdRpo ] + ;
				   [Where Dopps = '] + crSigCdRpo.Dopps + [' And ] + ;
				   		 [TpOps = '] + crSigCdRpo.TpOps + [' And ] + ;
				   		 [CGrus = '] + crSigCdRpo.CGrus + [' And ] + ;
				   		 [Mercs = '] + crSigCdRpo.Mercs + [' And ] + ;
				   		 [Grupos = ']+ crSigCdRpo.Grupos+ [' And ] + ;
				   		 [Contas = ']+ crSigCdRpo.Contas+ [' And ] + ;
				   		 [ContaOrig = ']+ crSigCdRpo.ContaOrig+ [' And ] + ;
				   		 [CodCors= ']+ crSigCdRpo.CodCors+[' And ] + ;
				   		 [Not cIdChaves = '] + crSigCdRpo.cIdChaves + [']

		If (ThisForm.poDataMgr.SqlExecute(lcQuery, [LocalBusca]) < 1)
			=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (LocalBusca)])
			Return 0
		EndIf
