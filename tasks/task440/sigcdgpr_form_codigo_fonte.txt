------------------------------------------------------------
* PROCEDURE Pagina.Lista.Grupo_op.Click
------------------------------------------------------------
PROCEDURE Pagina.Lista.Grupo_op.Click
Lparameters Opcao

If DoDefault(m.Opcao)
	Select CrSigCdGpr

	If Not Inlist(ThisForm.pcEscolha, 'INSERIR', 'PROCURAR') And Not Empty(crSigCdGpr.CodConPads)
		ThisForm.poDataMgr.CursorQuery('SigCdPro', 'crSigCdPro', 'CPros', crSigCdGpr.CodConPads, 'DPros')
		ThisForm.Pagina.Dados.getDCodConPads.Value = crSigCdPro.DPros
	EndIf

	ThisForm.Pagina.Dados.getDLin.Value = []
	If Not Inlist(ThisForm.pcEscolha, 'INSERIR', 'PROCURAR') And Not Empty(crSigCdGpr.Linhas)
		ThisForm.poDataMgr.CursorQuery('SigCdLin', 'crSigCdLin', 'Linhas', crSigCdGpr.Linhas, 'Descs')
		ThisForm.Pagina.Dados.getDLin.Value = crSigCdLin.Descs
	EndIf

	Do Case
		Case Inlist(ThisForm.pcEscolha,'INSERIR','PROCURAR')
			  ThisForm.Pagina.Dados.getCodigos.SetFocus	
		Case ThisForm.pcEscolha=='ALTERAR'
			  ThisForm.Pagina.Dados.getDescs.SetFocus	
		Case Inlist(ThisForm.pcEscolha,'CONSULTAR','EXCLUIR')
			  ThisForm.Pagina.Dados.Grupo_Salva.Salva.SetFocus
	Endcase
	ThisForm.Pagina.Dados.Refresh
Else
	ThisForm.mAtivapagina1
EndIf
ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
Local llRetorno,lcChave

m.llRetorno = .t.
m.lcChave = ThisForm.Pagina.Dados.getCodigos.Value

If Inlist(ThisForm.pcEscolha,'INSERIR','ALTERAR')
	With ThisForm.Pagina.Dados
		Do Case
			Case IsEmpty(.getCodigos.Value)
				  MessageBox('Código Inválido.',48,goSistema.NomeSistema)
				  .getCodigos.SetFocus
				  m.llRetorno = .f.

			Case IsEmpty(.getDescs.Value)
			
				  MessageBox('Descrição de Código Inválido.',48,goSistema.NomeSistema)
				  .getDescs.SetFocus
				  m.llRetorno = .f.
				  
			Case ThisForm.poDataMgr.ChkRegister('SigCdGpr','Codigos',ThisForm.Pagina.Dados.GetCodigos.Value) and thisform.pcEscolha = 'INSERIR'

	               Messagebox('Código de Produto Já Cadastrado.',0+64,goSistema.NomeSistema)
				   .getCodigos.SetFocus
				   m.llRetorno = .f.
				  
		EndCase	
	EndWith
Endif

If (ThisForm.pcEscolha=='EXCLUIR'.And.;
	ThisForm.poDataMgr.ChkRegister('SigCdGrp','Mercs',ThisForm.Pagina.Dados.GetCodigos.Value))

   Messagebox('Existem Grupos Cadastrados c/ Esse Grande Grupo,'+Chr(13)+'Impossivel Excluir.',0+64,goSistema.NomeSistema)
   m.llRetorno = .f.
   ThisForm.mAtivapagina1
	
EndIf

If m.llRetorno
	=dodefault()
	ThisForm.mAtivapagina1
EndIf
ENDPROC
