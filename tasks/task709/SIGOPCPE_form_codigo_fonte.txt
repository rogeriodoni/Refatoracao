PROCEDURE Init
ThisForm.cntSombra.lblSombra.Caption = ThisForm.Caption
ThisForm.cntSombra.lblTitulo.Caption = ThisForm.Caption

With ThisForm
	.poDataMgr = CreateObject('fSqlConector', ThisForm.Name)

	If (.poDataMgr.pnIdconn > 0)
		lcQryCFis = [Select * From SigSyCom]

		With .poDataMgr
			.AddCursor('SigSyCom', 'Tipos', 'crSigSyCom', '', '', lcQryCFis)
		EndWith

		.poDataMgr.ReQuery('crSigSyCom')

		Select crSigSyCom
		Index On Tipos Tag PesqTipo
		Go Top

		If Not Seek(Padr('Balança',20))
			Insert Into crSigSyCom (Tipos, CidChaves) Values ('Balança',fUniqueIds())
		EndIf
		If IsEmpty(crSigSyCom.Settings)
			Replace crSigSyCom.Settings with '9600,n,8,1'
		Endif
		If Not Seek(Padr('Cupom Fiscal',20))
			Insert Into crSigSyCom (Tipos, CidChaves) Values ('Cupom Fiscal',fUniqueIds())
		EndIf
		If IsEmpty(crSigSyCom.Settings)
			Replace crSigSyCom.Settings with '9600,n,8,1'
		Endif
		If Not Seek(Padr('Leitora Cheques',20))
			Insert Into crSigSyCom (Tipos, CidChaves) Values ('Leitora Cheques',fUniqueIds())
		EndIf
		If IsEmpty(crSigSyCom.Settings)
			Replace crSigSyCom.Settings with '9600,n,8,1'
		Endif
		If Not Seek(Padr('Leitora Codigo Barra',20))
			Insert Into crSigSyCom (Tipos, CidChaves) Values ('Leitora Codigo Barra',fUniqueIds())
		EndIf
		If IsEmpty(crSigSyCom.Settings)
			Replace crSigSyCom.Settings with '9600,n,8,1'
		Endif
		If Not Seek(Padr('Impressora de Cheque',20))
			Insert Into crSigSyCom (Tipos, CidChaves) Values ('Impressora de Cheque',fUniqueIds())
		EndIf
		If IsEmpty(crSigSyCom.Settings)
			Replace crSigSyCom.Settings with '9600,n,8,1'
		Endif
		If Not Seek(Padr('Gaveta Serial',20))
			Insert Into crSigSyCom (Tipos, CidChaves) Values ('Gaveta Serial',fUniqueIds())
		EndIf
		If IsEmpty(crSigSyCom.Settings)
			Replace crSigSyCom.Settings with '9600,n,8,1'
		Endif

		With This.Cmb_Hand
			.Clear
			.AddItem( '0-None' )
			.AddItem( '1-Xon Xoff' )
			.AddItem( '2-RTS' )
			.AddItem( ' ' )
		Endwith
		With This.Cmb_Tipo
			.ListIndex = 1
			.Clear
			.AddItem( 'Balança' )
			.AddItem( 'Cupom Fiscal' )
			.AddItem( 'Leitora Cheques' )
			.AddItem( 'Leitora Codigo Barra' )
			.AddItem( 'Impressora de Cheque' )
			.AddItem( 'Gaveta Serial' )
		Endwith

		=DoDefault()
	Else
		=Messagebox('Impossivel Efetuar Conexão com Servidor de Banco de Dados...', 0+48, 'CreateObject')
		Return .f.
	EndIf
EndWith
If Seek(ThisForm.Cmb_Tipo.value,'crSigSyCom','PesqTipo')
  Thisform.Refresh
EndIf

ThisForm.Say12.Visible = .f.
ThisForm.Say14.Visible = .f.
ThisForm.Say15.Visible = .f.
ThisForm.Say16.Visible = .f.
ThisForm.Get_TpLeChq.Visible = .f.
ThisForm.Say13.visible = .f.
ThisForm.Opt_leitor.Visible = .f.
ThisForm.Chk_BomPara.Visible = .f.
ThisForm.Chk_Nominal.Visible = .f.
ThisForm.Chk_Cruzado.Visible = .f.

If CrSigSyCom.Tipos = 'Leitora Cheques' or CrSigSyCom.Tipos = 'Impressora de Cheque'
	ThisForm.Say12.Visible = .t.
	ThisForm.Say14.Visible = .t.
	ThisForm.Say15.Visible = .t.
	ThisForm.Say16.Visible = .t.
	ThisForm.Get_TpLeChq.Visible = .t.
	ThisForm.Chk_BomPara.Visible = .t.
	ThisForm.Chk_Nominal.Visible = .t.
