PROCEDURE Init
LParameters loPForm, pEmp, pDop, pNum, pCursor

ThisForm.ParentForm = loPForm
ThisForm.DataSessionId = loPForm.DataSessionId

ThisForm.lpCursor = Iif(Type('pCursor')='C', pCursor, [crSigMvCab])
lcCursor = ThisForm.lpCursor

ThisForm.lpEmps   = Iif(Type('pEmp')='C', pEmp, &lcCursor..Emps)
ThisForm.lpDopes  = Iif(Type('pDop')='C', pDop, &lcCursor..Dopes)
ThisForm.lpNumes  = Iif(Type('pNum')='N', pNum, &lcCursor..Numes)

ThisForm.Caption = 'SubNiveis - ' + Proper( Alltrim( ThisForm.lpDopes ) ) +;
					' ( ' + Alltrim( Str( ThisForm.lpNumes ) ) + ' ) '

ThisForm.cntSombra.lblSombra.Caption = ThisForm.Caption
ThisForm.cntSombra.lblTitulo.Caption = ThisForm.Caption

Create Cursor TmpSubN (Ordes n(4), Emps c(3), Dopes c(20), Numes n(6), Datas T, Notas c(6), ;
					   Valos n(12,2), ValInis n(12,2), Usuars c(10), Nops n(6), PrazoEnts t Null, pstatus c(1), Descs c(20))
Index On Dtos(Datas) + Str(Ordes, 4) + Emps + Dopes + Str(Numes,6) Tag EmpDopNum

lcSql = [Select nDopes From SigCdOpe Where Dopes = ']+ThisForm.lpDopes+[']
If Thisform.parentform.PoDataMgr.SqlExecute(lcSql,'CrxOpe') < 1
	MessageBox('Favor reinicializar o processo.',16,'Falha na Conexão (CrXOpe)')
	RETURN .F.
EndIf

ThisForm.lAcOficina = fChecaAcesso( padl(crXOpe.nDopes,8,'0'), "OPPEND" )

ThisForm.fwGrade.Visible = .t.
If (Usuar <> [4CONTROL]) And fChecaAcesso(Padl(crXOpe.nDopes,8,[0]), [OCULTAPRS]) 
	ThisForm.fwGrade.Visible = .f.
EndIf

_lnCodigo = (( CrxOpe.nDopes * 1000000 ) + ThisForm.lpNumes )

lcSql = [Select a.Emps, a.Dopes, a.Numes, b.Datas, b.Notas, b.Valos, b.ValInis, b.Usuars, ] + ;
		[b.Nops, b.Datars, c.Ordes, b.PrazoEnts, b.pstatus, IsNull(d.descs,Space(20)) as Descs ] + ;
		  [From SigMvPec a ] + ;
		  	[Join SigMvCab b on a.EmpDopnums = b.EmpDopNums ] + ;
		  	[Join SigCdOpe c on b.Dopes = c.Dopes ] + ;
		  	[Left Join SigStats d on b.pstatus = d.cods ] + ;
		 [Where a.Codigos = ] + Str(_lncodigo,10) + [ And ] + ;
			   [a.EmpSubns = '] + ThisForm.lpEmps + [' ] + ;
		 [Union All ]+;
		[Select a.Emps, a.Dopes, a.Numes, b.Datas, b.Notas, b.Valos, b.ValInis, b.Usuars, ] + ;
		[b.Nops, b.Datars, c.Ordes, b.PrazoEnts, b.pstatus, IsNull(d.descs,Space(20)) as Descs ] + ;
		  [From SigOpDev a ] + ;
		  	[Join SigMvCab b on a.EmpDopnums = b.EmpDopNums ] + ;
		  	[Join SigCdOpe c on b.Dopes = c.Dopes ] + ;
		  	[Left Join SigStats d on b.pstatus = d.cods ] + ;
		 [Where a.Codigos = ] + Str(_lncodigo,10) + [ ] + ;
		 [Order By b.Datars]
		
If Thisform.Parentform.PoDataMgr.SqlExecute(lcSql,'CrxEstPe') < 1
	MessageBox('Favor reinicializar o processo.',16,'Falha na Conexão (CrxEstPe)')
	RETURN .f.
EndIf

Select CrxEstPe
If (Reccount([CrxEstPe]) = 0)
	MessageBox('Nenhum registro encontrado!!!',0+48,'')
	Return .f.
EndIf

ThisForm.ParentForm.Enabled = .f.

SELECT TmpSubN
APPEND FROM DBF('CrxEstPe')

Update TmpSubN Set PrazoEnts = Nvl(PrazoEnts, Ctot([]))

SELECT TmpSubN
GO Top

With ThisForm.Grade
	.RecordSource      	   = 'TmpSubn'
	.Column1.ControlSource = 'TmpSubN.Emps'
	.Column2.Controlsource = 'TmpSubN.Dopes'
	.Column3.Controlsource = 'TmpSubN.Numes'
	.Column4.Controlsource = 'TmpSubN.Datas'
	.Column5.Controlsource = 'TmpSubN.Notas'
	.Column6.Controlsource = 'TmpSubN.Usuars'
	.Column7.Controlsource = 'TmpSubN.PrazoEnts'
	.Column8.Controlsource = 'TmpSubN.Pstatus + [-] + TmpSubN.Descs'
	.SetFocus
EndWith


ENDPROC
