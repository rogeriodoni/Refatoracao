	ThisForm.get_dcod.Value = ''
	ThisForm.get_dcod.Refresh
	Return(.T.)
EndIf
m.loLista = CreateObject('fwBuscaExt',ThisForm.poDataMgr.pnIdConn,'SigCdPro','CrProds','cpros',This.Value,'Seleção',1000)
If .Not.m.loLista.plAchouRegistro
	m.loLista.mAddColuna('cpros','','Código')
	m.loLista.mAddColuna('dpros','','Descrição')
	m.loLista.Show()
EndIf

If LastKey() <> 27 And Used('CrProds')
	ThisForm.get_cod.Value  = Evaluate('CrProds.cpros')
	ThisForm.get_dcod.Value = Evaluate('CrProds.dpros')
Else
	ThisForm.get_cod.Value  = ''
	ThisForm.get_dcod.Value = ''
EndIf
ThisForm.get_cod.Refresh
ThisForm.get_dcod.Refresh
Return(.T.)
ENDPROC

PROCEDURE consulta.Click
* Declara variáveis de campos
Private pnOpcao,pcCdEstoque,pcDsEstoque,pcCdProduto,pcDsProduto,pdDataIni,pdDataFin

* Inicializa variáveis de campos
pcCdGrupo	= ThisForm.get_grupo.Value
pcCdConta	= ThisForm.get_est.Value
pcDsConta	= ThisForm.get_dest.Value
pcCdProduto = ThisForm.get_cod.Value
pcDsProduto = ThisForm.get_dcod.Value
pdDataIni   = ThisForm.get_dataini.Value
pdDataFin   = ThisForm.get_datafin.Value

* Consiste o Campo Grupo
If IsEmpty(pcCdGrupo)
	MessageBox('Grupo Inválido !',16,'')
	ThisForm.get_grupo.SetFocus
	Return(.F.)
EndIf

* Consiste o campo Conta
If IsEmpty(pcCdConta) Or !(ThisForm.poDataMgr.ChkRegister('SigCdCli','iclis',pcCdConta))
	MessageBox('Conta Inválida !',16,'')
	ThisForm.get_est.SetFocus
	Return(.F.)
EndIf

* Consiste o campo produto
If IsEmpty(pcCdProduto) Or !(ThisForm.poDataMgr.ChkRegister('SigCdPro','cpros',pcCdProduto))
	MessageBox('Produto Inválido !',16,'')
	ThisForm.get_cod.SetFocus
	Return(.F.)
EndIf

* Consiste o campo data inicial
If IsEmpty(pdDataIni)
	MessageBox('Data Inicial Inválida !',16,'')
	ThisForm.get_dataini.SetFocus
	Return(.F.)
EndIf

* Consiste o campo data final
If IsEmpty(pdDataFin)
	MessageBox('Data Final Inválida !',16,'')
	ThisForm.get_datafin.SetFocus
	Return(.F.)
EndIf

* Consiste o período
If pdDataIni > pdDataFin
	MessageBox('Data Inicial maior que Data Final !',16,'')
	ThisForm.get_dataini.SetFocus
	Return(.F.)
EndIf

ThisForm.Enabled = .F.
Do Form SigPrHpr With ThisForm
ENDPROC
