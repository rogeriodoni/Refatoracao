* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
Private pPro, pKey

pKey = Replicate(Chr(254), 20)

lcQryCmCab = [Select * From SigCmCab]
lcQryCmItn = [Select * From SigCmItn Where fkChaves = ?pKey]

With ThisForm
	.poDataMgr = CreateObject([fSqlConector], .Name)
	
	If (.poDataMgr.pnIdconn > 0)
		With .poDataMgr
			.AddCursor([SigCmCab], [pkChaves], [crSigCmCab], [], ThisForm.Pagina.Lista.Grade, lcQryCmCab)
			.pfSqlTabela(1).pColuna([Cods],     [], [], [Descrição],     290, .t.)
			.pfSqlTabela(1).pColuna([Emps],     [], [], [Emp],           031, .t.)
			.pfSqlTabela(1).pColuna([DtInis],   [], [], [Dt.Inicial],    073, .f.)
			.pfSqlTabela(1).pColuna([DtFims],   [], [], [Dt.Final],      073, .f.)
			.pfSqlTabela(1).pColuna([MetaEmps], [], [], [Meta Empresa],  100, .f.)

			.AddCursor([SigCmItn], [pkChaves], [crSigCmItn], [], [], lcQryCmItn)
		EndWith

		lcQuery = [Select GrPadVens, GrPadResps From SigCdPam]

		If (.poDataMgr.SqlExecute(lcQuery, [crSigCdPam]) < 1)
			=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (crSigCdPam)])
			Return .f.
		EndIf
		Go Top In crSigCdPam
		If Eof([crSigCdPam])
			=MessageBox([Arquivo de Configuração de Parâmetros Não Encontrado!!!], 16, [Erro!!!])
			Return .f.
		EndIf

		.loObj = CreateObject([Relation])
		.loObj.AddProperty([GrPadVens], crSigCdPam.GrPadVens)
		.loObj.AddProperty([GrPadRess], crSigCdPam.GrPadResps)
	Else
		=Messagebox([Impossível Efetuar Conexão Com o Servidor de Banco de Dados...], 0+48, [CreateObject])
		Return .f.
	EndIf
EndWith

=DoDefault()

If Empty(ThisForm.loObj.GrPadVens) And Empty(ThisForm.loObj.GrPadRess)
	=MessageBox([O Grupo de Vendedores ou Responsáveis Não Está Configurado] + Chr(13) + ;
				[em Parâmetros. Para Incluir Comissões é Necessário Ajustar], 0+64, [Atenção!!!])
	ThisForm.Pagina.Lista.Grupo_Op.Inserir.Enabled = .f.
	ThisForm.Pagina.Lista.Grupo_Op.Alterar.Enabled = .f.
	ThisForm.Pagina.Lista.Grupo_Op.Excluir.Enabled = .f.
EndIf

ThisForm.Pagina.Lista.Grade.Column1.SetFocus
* Fim **************************************************************************************
ENDPROC

PROCEDURE Pagina.Lista.Grupo_op.Click
LParameters lnOpc

Private pKey, loDad

	This.Value = lnOpc
EndIf

loDad = ThisForm.Pagina.Dados

Select crSigCmCab

If Not DoDefault(This.Value)
	ThisForm.mAtivaPagina1
	Return .f.
EndIf

If (ThisForm.pcEscolha = [INSERIR])
	Replace pkChaves With fUniqueIds(), ;
			PctVd1   With 100 In crSigCmCab
	Zap In crGrdItP
	Zap In crGrdItV
	Zap In crGrdItM
EndIf

loDad.getDEmpresa.Value = []

If InList(Thisform.pcEscolha, [CONSULTAR], [ALTERAR], [EXCLUIR])
	ThisForm.poDataMgr.Requery([crSigCmItn])
	
	Zap In crGrdItP
	Select * From crSigCmItn Where Tipos = [PRO] Order By Mercs, CGrus, SGrus Into Cursor LocalItn
	Select LocalItn
	Scan
		Scatter Memvar Memo
		Insert Into crGrdItP From Memvar
	EndScan

		=Messagebox([A Empresa Necessita Ser Preenchida!!!], 0+48, [Empresa Inválida!!!])

		Return .f.

		=Messagebox([A Data Inicial Não Pode Ser Maior Que a Final!!!], 0+48, [Data Inválida!!!])
		ThisForm.Pagina.Dados.getDtFinal.SetFocus
		Return .f.
	EndIf

		=Messagebox([O Percentual de Comissão Dividida do Vendedor 1] + Chr(13) + ;
					[Necessita Ser Preenchido. Caso Não Trabalhe Com] + Chr(13) + ;
					[Divisão Coloque 100% da Comissão no Vendedor 1.], 0+48, [Comissão Dividida Inválida!!!])
		ThisForm.Pagina.Dados.getPctVd1.SetFocus
		Return .f.

			=Messagebox([A Descrição Está Em Uso Nessa Empresa!!!], 0+48, [Descrição Inválida!!!])

		EndIf

					[Não É possível Gerar Comissões Sem Essa Informação] + Chr(13) + ;
					[Complete Esses Dados e Tente Salvar Novamente!!!], 0+48, [Atenção!!!])
		Go Top In crGrdItP
		Locate For Empty(crGrdItP.Mercs)
		If Eof([crGrdItP])
			Insert Into crGrdItP (Tipos, pkChaves, fkChaves) Values ([PRO], fUniqueIds(), crSigCmCab.pkChaves)
		EndIf
		ThisForm.Pagina.Dados.grdProdutos.Column1.SetFocus
		Return .f.
	EndIf
Set Step On 
	* Validando Grade de Vendedores
	Select crGrdItV
	Scan
		If Empty(crGrdItV.fkChaves)
			Replace fkChaves With crSigCmCab.pkChaves In crGrdItV
		EndIf
		If Empty(crGrdItV.pkChaves)
			Replace pkChaves With fUniqueIds() In crGrdItV
		EndIf
		If Empty(crGrdItV.Tipos)
			Replace Tipos With [VEN] In crGrdItV
		EndIf
		If Empty(crGrdItV.CodVens)
			Delete In crGrdItV
		Else
			lnRec = Recno([crGrdItV])
			lcChv = crGrdItV.CodVens

			Select * From crGrdItV ;
			 Where Recno() <> lnRec And ;
			 	   CodVens = lcChv ;
			  Into Cursor LocalDups

			lnDup = Reccount([LocalDups])
			Use In LocalDups
			If (lnDup > 0)
				lcDup = [Vendedor : ] + Alltrim(crGrdItV.CodVens)
				
				=MessageBox([Existem Registros Informados Na Grade de Vendedores] + Chr(13) + ;
							[Em Duplicidade. Corrija e Tente Salvar Novamente.] + Chr(13) + ;
							lcDup, 0+48, [Atenção!!!])
				ThisForm.Pagina.Dados.grdVendedoras.Column1.SetFocus
				Return .f.
			EndIf
		EndIf
	EndScan
	Go Top In crGrdItV
	If Eof([crGrdItV])
		=MessageBox([Nenhum Registro Foi Informado Na Grade de Vendedores] + Chr(13) + ;
					[Não É possível Gerar Comissões Sem Essa Informação] + Chr(13) + ;
					[Complete Esses Dados e Tente Salvar Novamente!!!], 0+48, [Atenção!!!])
		Go Top In crGrdItV
		Locate For Empty(crGrdItV.CodVens)
		If Eof([crGrdItV])
			Insert Into crGrdItV (Tipos, pkChaves, fkChaves) Values ([VEN], fUniqueIds(), crSigCmCab.pkChaves)
		EndIf
		ThisForm.Pagina.Dados.grdVendedoras.Column1.SetFocus
		Return .f.
	EndIf

	Select Sum(1) as Soma From crGrdItV Into Cursor crVenTots
	Go Top In crVenTots
	If (crVenTots.Soma > 12)
		=MessageBox([Apenas 12 Vendedores Podem Ser Informados Por Cadastro.] + Chr(13) + ;
					[Insira os Demais Em Outro Cadastro Se Possui Mais de 12], 0+48, [Atenção!!!])
		ThisForm.Pagina.Dados.grdVendedoras.Column1.SetFocus
		Return .f.
	EndIf
	
	* Validando Grade de Metas e Remuneração
	Select crGrdItM
	Scan
		If Empty(crGrdItM.fkChaves)
			Replace fkChaves With crSigCmCab.pkChaves In crGrdItM
		EndIf
		If Empty(crGrdItM.pkChaves)
			Replace pkChaves With fUniqueIds() In crGrdItM
		EndIf
		If Empty(crGrdItM.Tipos)
			Replace Tipos With [MET] In crGrdItM
		EndIf
		If (crGrdItM.FxaInis + crGrdItM.FxaFims <= 0) Or (Not Empty(crGrdItM.FxaInis) And Empty(crGrdItM.FxaFims))
			Delete In crGrdItM
		EndIf
	EndScan

	Go Top In crGrdItM
	Scan
		lnCod = crGrdItM.CodFaixas
		lnIni = crGrdItM.FxaInis
		lnFim = crGrdItM.FxaFims
		
		If (crGrdItM.FxaFims <= crGrdItM.FxaInis)
			lcDup = [Faixa : ] + Padl(crGrdItM.CodFaixas, 2, [0])

						lcDup, 0+48, [Atenção!!!])
			ThisForm.Pagina.Dados.grdMetas.Column1.SetFocus
			Return .f.
		EndIf

		Select * From crGrdItM ;
		 Where CodFaixas <> lnCod And ;
		 	   ((lnIni > FxaInis And lnIni < FxaFims) Or (lnFim > FxaInis And lnFim < FxaFims)) ;
		  Into Cursor LocalDups

		Go Top In LocalDups
		If Not Eof([LocalDups])
			=MessageBox([A Faixa "] + Padl(lnCod, 2, [0]) + [" de ] + Alltrim(Str(lnIni, 16, 2)) + [ a ] + Alltrim(Str(lnFim, 16, 2)) + Chr(13) + ;
						[Está Contida Na "] + Padl(LocalDups.CodFaixas, 2, [0]) + [" de ] + Alltrim(Str(LocalDups.FxaInis, 16, 2)) + ;
						[ a ] + Alltrim(Str(LocalDups.FxaFims, 16, 2)), 0+16, [Faixas Em Intersecção!!!])
			ThisForm.Pagina.Dados.grdMetas.Column1.SetFocus
			Return .f.
		EndIf
	EndScan

	Go Top In crGrdItM
	If Eof([crGrdItM])
		=MessageBox([Nenhum Registro Foi Informado Na Grade de Vendedores] + Chr(13) + ;
					[Não É possível Gerar Comissões Sem Essa Informação] + Chr(13) + ;
					[Complete Esses Dados e Tente Salvar Novamente!!!], 0+48, [Atenção!!!])
		Go Top In crGrdItM
		Locate For Empty(crGrdItM.CodVens)
		If Eof([crGrdItM])
			Insert Into crGrdItM (Tipos, pkChaves, fkChaves) Values ([VEN], fUniqueIds(), crSigCmCab.pkChaves)
		EndIf
		ThisForm.Pagina.Dados.grdVendedoras.Column1.SetFocus
		Return .f.
	EndIf

	* Salvando Dados do Cursor Temporário das Grades Para o Cursor de Gravação
	Zap In crSigCmItn

	Select crGrdItP
	Scan
		Insert Into crSigCmItn (Mercs, CGrus, SGrus, PPctFixs, fkChaves, pkChaves, Tipos) ;
						Values (crGrdItP.Mercs, crGrdItP.CGrus, crGrdItP.SGrus, crGrdItP.PPctFixs, ;
								crGrdItP.fkChaves, crGrdItP.pkChaves, crGrdItP.Tipos)

	EndScan

	Select crGrdItV
	Scan
		Insert Into crSigCmItn (CodVens, VMetas, SalBases, CodResps, RPctFixs, fkChaves, pkChaves, Tipos, VMetItns) ;
						Values (crGrdItV.CodVens, crGrdItV.VMetas, crGrdItV.SalBases, crGrdItV.CodResps, ;
								crGrdItV.RPctFixs, crGrdItV.fkChaves, crGrdItV.pkChaves, crGrdItV.Tipos, crGrdItV.VMetItns)
	EndScan

	Select crGrdItM
	Scan
		Insert Into crSigCmItn (CodFaixas, FxaInis, FxaFims, PctFaixas, Bonus, fkChaves, pkChaves, Tipos) ;
						Values (crGrdItM.CodFaixas, crGrdItM.FxaInis, crGrdItM.FxaFims, crGrdItM.PctFaixas, ;
								crGrdItM.Bonus, crGrdItM.fkChaves, crGrdItM.pkChaves, crGrdItM.Tipos)
	EndScan
EndIf

=DoDefault()

ThisForm.mAtivaPagina1
