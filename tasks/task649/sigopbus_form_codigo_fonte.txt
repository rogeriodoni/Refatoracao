PROCEDURE Valid
Local loLista

If Not Empty(This.Value)

	ThisForm.Label4.Visible = .t.
	
	lcFiltro = [ Dpros Like '%]+Alltrim(This.Value)+[%' ]
	If Thisform.Situas <> 0 And ThisForm.FiltraSituas <> 0
		Do case
		Case Thisform.Situas = 1
			lcFiltro = lcFiltro + [ Situas <> 2 ]
		Case Thisform.Situas = 2
			lcFiltro = lcFiltro + [ Situas = 2 ]
		Case Thisform.Situas = 3
			lcFiltro = lcFiltro + [ Situas In(0,1,2) ]
		Case Thisform.Situas = 4
			lcFiltro = lcFiltro + [ Situas <> 2 And Encoms = 1 ]
		Case Thisform.Situas = 5
			lcFiltro = lcFiltro + [ Encoms = 1 ]
		EndCase
	EndIf 

	loLista = CreateObject('fwBuscaExt', ThisForm.poDataMgr.pnIdConn, 'SigCdPro', ;
						   'crListaRemota', 'dpros', This.Value, 'Seleção', 1000 , .F. , lcFiltro, .t., .t. )
	loLista.mAddColuna('Dpros', '', 'Descrição',.t.,'Dpros')
	loLista.mAddColuna('Cpros', '', 'Código',.t.,'Cpros')
	loLista.mAddColuna('cgrus', '', 'Grupo',.t.,'Cgrus')
	loLista.mAddColuna('Dpro2s', '', 'Descrição',.t.,'Dpro2s')
	loLista.mAddColuna('PesoMs', '99,999.999', 'Peso Médio',.t.,'PesoMs')
	If ThisForm.Conjunto 
		loLista.mAddColuna('Conjunts', '', 'Cod. Pai',.t.,'Conjunts')
	EndIf 
	If ThisForm.CodMacro
		loLista.mAddColuna('CodMacro', '', 'Cod. Macro',.t.,'CodMacro')
	EndIf 
	If ThisForm.CnjLacto
		loLista.mAddColuna('CnjLacto', '', 'Conjunto',.t.,'CnjLacto')
	EndIf 
    loLista.mAddColuna('figjpgs', .f., .f., .f., .f., .t.)
	loLista.Show()	
	
	ThisForm.Produto = Space(14)
	If Lastkey() <> 27
		This.Value = CrListaRemota.Dpros
		ThisForm.Produto = Iif(ThisForm.Conjunto,CrListaRemota.Conjunts,Iif(ThisForm.CodMacro,CrListaRemota.CodMacro,Iif(ThisForm.Tiporetorno,Str(CrListaRemota.Cbars),CrListaRemota.Cpros)))
	EndIf
	ThisForm.Release
