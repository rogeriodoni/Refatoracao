------------------------------------------------------------
* PROCEDURE mctrlbotoes
------------------------------------------------------------
PROCEDURE mctrlbotoes
LParameters plHabilita

Local lnI

If Type('plHabilita') # 'L'
	plHabilita = .F.
Endif

With Thisform.ParentForm.Pagina.Lista
	For lnI=1 to .Grupo_Op.ButtonCount
		If .Grupo_Op.Buttons(lnI).BaseClass = 'Commandbutton' And .Grupo_Op.Buttons(lnI).Visible
			.Grupo_Op.Buttons(lnI).Enabled = plHabilita
		EndIf
	EndFor
	.cmdFPOper.Enabled = plHabilita
	.cmdMultiSubN.Enabled = plHabilita
	.Grupo_Saida.Sair.Enabled = plHabilita
EndWith



ENDPROC

------------------------------------------------------------
* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
LParameters poForm, pnDataSes, pcEscolha, plInverte

Local lcOperacao

Set DataSession to pnDataSes

Thisform.ParentForm = poForm
Thisform.PcEscolha  = pcEscolha
Thisform.ParentForm.Enabled  = .F.
lcOperacao = crSigCdOpe.Dopes

If plInverte
	With Thisform
		.Caption = 'Operações do Sub-Nível ' + AllTrim( Proper( crSigCdOpe.Dopes ) )
		.Top     = Thisform.ParentForm.Top + Thisform.ParentForm.Pagina.Lista.Grade.Top + 50
		.Left    = Thisform.ParentForm.Left + ;
				   Thisform.ParentForm.Pagina.Lista.Grade.Left + ;
				   Thisform.ParentForm.Pagina.Lista.Grade.Columns(1).Width + ;
				   Thisform.ParentForm.Pagina.Lista.Grade.Columns(3).Width + 14

		.cmdInserir.Visible = .f.
		.cmdInserir.Enabled = .f.
		.cmdExcluir.Visible = .f.
		.cmdExcluir.Enabled = .f.
		.shape1.Left = .cmdSair.Left - 5
		.shape1.width = .cmdsair.width + 10
	EndWith

	Thisform.mCtrlBotoes(.F.)
	
	ThisForm.cntSombra.lblSombra.Caption = ThisForm.Caption
	ThisForm.cntSombra.lblTitulo.Caption = ThisForm.Caption

	lcSql = [Select a.*,b.nDopes From SigOpCsn a Inner Join SigCdOpe b On a.cDopes = b.Dopes Where a.cNivels = ']+lcOperacao+[' Order by a.cNivels]
	SigCdOpe.PodataMgr.SqlExecute(lcSql,'CsSigOpCsn')

	Select csSigOpCsn
	Goto Top	
	With Thisform.GradeSubN
		.RecordSource          = 'cSSigOpCsn'
		.Column1.ControlSource = 'cSSigOpCsn.cDopes'
		.Column2.ControlSource = 'cSSigOpCsn.nDopes'
		.Column3.ControlSource = 'cSSigOpCsn.Obrigs'
		.Column4.ControlSource = 'cSSigOpCsn.PgSubNNs'
		.Column5.ControlSource = 'cSSigOpCsn.cBxPagos'
		.Column6.ControlSource = 'cSSigOpCsn.tStatus'
		.Column7.ControlSource = 'cSSigOpCsn.CodObs'
		.Column8.ControlSource = 'cSSigOpCsn.TpFats'
		.Column9.ControlSource = 'cSSigOpCsn.ckAtCta'
		.Refresh
	EndWith

	lcSql = [Select * From SigOpOss Where cdopes = ']+lcOperacao+[' ]
	SigCdOpe.PodataMgr.SqlExecute(lcSql,'LocalSsOpe')
	
	Select CrSigOpOss
	Zap
	Append From Dbf('LocalSsOpe')
	Index on cNivels Tag cNivels
		
	Select cRSigOpOss
	Goto Top	
	With Thisform.GradeSubS 
		.RecordSource          = 'CrSigOpOss'
		.Column1.ControlSource = 'crSigOpOss.Dopess'
		.Refresh
	EndWith

Else

	With Thisform
		.Caption = 'Subníveis da Operação ' + AllTrim( Proper( crSigCdOpe.Dopes ) )
		.Top     = Thisform.ParentForm.Top + Thisform.ParentForm.Pagina.Lista.Grade.Top + 50
		.Left    = Thisform.ParentForm.Left + ;
				   Thisform.ParentForm.Pagina.Lista.Grade.Left + ;
				   Thisform.ParentForm.Pagina.Lista.Grade.Columns(1).Width + ;
				   Thisform.ParentForm.Pagina.Lista.Grade.Columns(3).Width + 16

		.cmdInserir.Visible = fChecaAcesso('SigOpSno','INSERIR') And ;
							  Iif( Type( 'Thisform.PcEscolha' ) # 'C', .T., ;
							  InList( Thisform.PcEscolha, 'INSERIR', 'ALTERAR' ) )
		.cmdInserir.Enabled = fChecaAcesso('SigOpSno','INSERIR')	And ;
							  Iif( Type( 'Thisform.PcEscolha' ) # 'C', .T., ;
							  InList( Thisform.PcEscolha, 'INSERIR', 'ALTERAR' ) )
		.cmdExcluir.Visible = fChecaAcesso('SigOpSno','EXCLUIR') And ;
							  Iif( Type( 'Thisform.PcEscolha' ) # 'C', .T., ;
							  InList( Thisform.PcEscolha, 'INSERIR', 'ALTERAR' ) )
		.cmdExcluir.Enabled = fChecaAcesso('SigOpSno','EXCLUIR')	And ;
							  Iif( Type( 'Thisform.PcEscolha' ) # 'C', .T., ;
