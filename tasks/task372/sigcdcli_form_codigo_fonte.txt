PROCEDURE checaibge
Lparameters pMun, pUFs, pObC, pObU

lnRtC = 9999999

lcQuery = [Select a.Codigos ] + ;
		    [From SigCdMun a, SigCdUfs b ] + ;
		   [Where a.Descs = '] + Padr(fLimpaTexto(Upper(pMun)), 40) + [' And ] + ;
		   		 [a.UFIBGEs = b.UFIBGEs And ] + ;
		   		 [b.Estados = '] + pUFs + [']

If (ThisForm.poDataMgr.SqlExecute(lcQuery, [LocalMunic]) < 1)
	=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (LocalMunic)])
	Return 0
EndIf

Select LocalMunic
Go Top In LocalMunic
If Not Eof([LocalMunic])
	lnRtC = Int(Val(LocalMunic.Codigos))
EndIf

pObC.Value = lnRtC
pObC.Refresh
pObC.Forecolor = Rgb(0,0,0)
If (lnRtC = 9999999)
	pObC.Forecolor = Rgb(255,0,0)
EndIf

lnRtU = 99

lcQuery = [Select UFIBGEs ] + ;
		    [From SigCdUfs ] + ;
		   [Where Estados = '] + Padr(pUFs, 2) + [']

If (ThisForm.poDataMgr.SqlExecute(lcQuery, [LocalUF]) < 1)
	=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (LocalUF)])
	Return 0
EndIf

Select LocalUF
Go Top In LocalUF
If Not Eof([LocalUF])
	lnRtU = LocalUF.UFIBGEs
EndIf

pObU.Value = lnRtU
pObU.Refresh
pObU.Forecolor = Rgb(0,0,0)
If (lnRtU = 99)
	pObU.Forecolor = Rgb(255,0,0)
EndIf
* Fim ********************************************************************************************
ENDPROC

* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init

Lparameters pCpf, pGrupo, pCep, pVal, pCli, pcTpBloqCar, pcMudaCpfCgc
Local lnErro1,lcOnError, lcCpf, lcCpfs, llVC

pCpf = Padr(Iif(Type('pCpf')=='C', pCpf, ''),20)
pGrupo = Padr(Iif(Type('pGrupo')=='C', pGrupo, ''),10)
pCep = Iif(Type('pCep')=='N', pCep, 0)
pVal = Iif(Type('pVal')=='L', pVal, .f.)
pCli = Padr(Iif(Type('pCli')=='C', pCli, ''),10)
pcTpBloqCar = Iif(Type('pcTpBloqCar')=='C', pcTpBloqCar, '0') && se for 1 bloqueia inserir/excl. caracteristica. ...
pcMudaCpfCgc = Iif(Type('pcMudaCpfCgc')=='C', pcMudaCpfCgc, '0') && se for 1 permite a mudanca de CPF p/ CNPG e vice/versa...
lcCpfs = pCpf
lcCpfs = StrTran( lcCpfs, '.', '' )
lcCpfs = StrTran( lcCpfs, '-', '' )
lcCpfs = StrTran( lcCpfs, '/', '' )
lcCpfs = AllTrim( lcCpfs )

With Thisform
	.LockScreen = .t.
	.RetCodCliente = ' '
	.MudaVend = fChecaAcesso([SIGCDCTA], [VENDEDOR])
	
	If Not Empty(lcCpfs) Or Not Empty(pCli)
		.poDataMgr = CreateObject('fSqlConector','sige')

		If .poDataMgr.pnIdConn>0
			* Acessando a Dll de Validação de Inscrição Estadual (Fornecida Pelo Sintegra)
			.ValidaIE = .f.
			lcOnError = On('Error')
			On Error lnErro1 = Error()
			lnErro1 = 0

			Declare Integer ConsisteInscricaoEstadual In 'DllInscE32.Dll' String @Insc, String @Uf

			.ValidaIE = (lnErro1==0)
			On Error &lcOnError.

			.cntConta.mIniConta() && checar / crias propriedades necessarios e os cursores...
			ThisForm.cntConta.cmdgPessoal.cmdPessoal.ToolTipText = 'F5 - Dados Pessoais/Comerciais'     
			
			lnErro1 = .poDataMgr.pnIdConn
		
			If Used('crSigCdPam') And Reccount('crSigCdPam')>0
				If Used('crSigCdGcr') And Reccount('crSigCdGcr')>0
					Select crSigCdPam
					Locate 

					If Empty(pGrupo)
						pGrupo = Padr(crSigCdPam.GrPadClis,10)
					EndIf 

					Select crSigCdGcr
					Locate For Codigos==pGrupo
Set Step On 
					&& habilita a tela se for cadastro de conta,se for de cliente habilita apenas se existe o grupo do crSigCdPam.GrPadClis...
					If Not Empty(pGrupo) And Not Eof('crSigCdGcr')
						lcQuery = [Select a.Iclis From SigCdCli a] + ;
								   [ Where ] + Iif(Not Empty(lcCpfs), [a.Cpfs = '] + pCpf + ['], [a.IClis = '] + pCli + [']) + [ Order By 1]
						If lnErro1>0
							lnErro1 = ThisForm.poDataMgr.SQLExecute(lcQuery,'crTmpCli')
						EndIf
						If Used('crTmpCli')
							Select crTmpCli
							Locate 
							
							If Not Eof('crTmpCli')
								pCli = crTmpCli.Iclis
							EndIf
						EndIf 
						llvc = Iif(Not IsEmpty(CrSigCdPac.CtVendaCs) And CrSigCdPac.CtVendaCs = pCli,.t.,.f. )
						.plAltCd = (Not Eof('crTmpCli') And Not Empty(pCli))
						.pcEscolha = Iif(fChecaAcesso('SIGCDCTA', 'ALTERAR') And Not llVC, Iif(Not .plAltCd,'INSERIR','ALTERAR'), 'CONSULTAR')

						If .pcEscolha=='INSERIR' Or Not Eof('crTmpCli')
							If .cntConta.mLeDados(pGrupo, pCli, '1', pcTpBloqCar, pcMudaCpfCgc)
								ThisForm.cntConta.cmdgftec.Visible = .F.  
								If .pcEscolha=='INSERIR'

									lcCpf = pCpf
									lcCpf = Alltrim(Strt(Strt(Strt(lcCpf, '.', ''), '-', ''), '/', ''))
									
									With This.cntConta.pgframeDados.pgframeDados1
										.OpcaoCPFCGC.Value = Iif(Not Len(lcCpf)==14, 1, 2)
										.GetCPFCGC.Value = pCpf
									EndWith 

									This.cntConta.mAtuGetCpf()
								EndIf 
								
								.cntConta.Visible = .t.
								.cntConta.Refresh
							Else 
								= Messagebox('Erro Na Leitura dos Dados',(0+48),'Erro')
								Return(.f.)
							EndIf 
						Else 
							= Messagebox('Usuário Não Possui Acesso p/ Incluir / Alterar Dados de Clientes.',(0+48),'Erro')
							Return(.f.)
						EndIf 
					Else 
						= Messagebox('Grupo Padrão Não Configurado.',(0+48),'Erro')
						Return(.f.)
					EndIf 
				Else 
					= Messagebox('Nenhum Grupo de Conta Cadastrado.',(0+48),'Erro')
					Return(.f.)
				EndIf 
			Else 
				= Messagebox('Configuração de Parametros do Sistema Não Encontrado.',(0+48),'Erro')
				Return(.f.)
			EndIf 
		Else 
			= Messagebox('Impossivel Efetuar Conexão com Servidor de Banco de Dados.',(0+48),'Erro')
			Return(.f.)
		EndIf 
	Else
		= Messagebox('Referencia do Cliente Inválido (Cpf / Conta).',(0+48),'Erro')
		Return(.f.)
	EndIf 
	.LockScreen = .f.
EndWith

ENDPROC
