------------------------------------------------------------
* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
LPARAMETERS pForm, pOperacao

ThisForm.oForm = pForm
ThisForm.pDopes = Iif(Type('pOperacao')=[C],pOperacao,[ ])
pDopFrt = ThisForm.pDopes
If Type('ThisForm.oForm') = [O]
	ThisForm.oForm.Enabled = .f.
EndIf

ThisForm.Caption = 'Frete por Cidade '+ThisForm.pDopes

With ThisForm
	ThisForm.Refresh
	ThisForm.Draw
	ThisForm.Show
	.poDatamgr = CreateObject('fSqlConector')
	With .poDatamgr
		If Empty(ThisForm.pDopes)
			.AddCursor('SigCdFrt','Cidas','CrSigCdFrt',' ',ThisForm.pagina.lista.grade)
		Else
			lcSql = [Select * From SigOpFrt Where Dopes = ']+ThisForm.pDopes+[']
			.AddCursor('SigOpFrt','CidChaves','CrSigCdFrt',' ',ThisForm.pagina.lista.grade,lcSql)
		Endif
		.pfSqlTabela(1).pColuna('Cidas','','','Cidade',250,.t.)
		.pfSqlTabela(1).pColuna('Valors','','','Valor do Frete',100,.f.)
	EndWith
	.poDataMg2 = CreateObject('fSqlConector', 'cep',.t. )
	If (.poDataMg2.pnIdconn > 0)
		ThisForm.pagina.lista.BtnAguarde.Visible = .t.	
		ThisForm.Draw
		Thisform.Pagina.Lista.BtnAguarde.Refresh
		lcSql = [Select distinct Cidades From Cep ]
		ThisForm.Podatamg2.Sqlexecute(lcsql,'TmpCep')
	ELSE
		Create Cursor TmpCep (Cidades c(30), Estados c(2))
	EndIf
	Index On Cidades Tag Cidades
	ThisForm.pagina.lista.BtnAguarde.Visible = .f.
	= DoDefault()
	.LockScreen = .f.
EndWith
ENDPROC

PROCEDURE Pagina.Lista.Grupo_op.Click
LPARAMETERS opcao

If !dodefault( opcao )
	ThisForm.mAtivapagina1
	Return
Endif

If Inlist(thisform.pcEscolha,'INSERIR','PROCURAR')
	ThisForm.Pagina.Dados.GetCidade.SetFocus
Else
	If Inlist(thisform.pcEscolha,'ALTERAR')
		ThisForm.Pagina.Dados.Get_Frete.SetFocus
	Else
		Thisform.Pagina.Dados.Grupo_salva.Salva.setfocus
	EndIf
EndIf

ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
If Inlist( Thisform.pcEscolha,'INSERIR')
	If IsEmpty( Thisform.Pagina.Dados.getCidade.value )
		=messagebox('Cidade Inválida!!!',0+64,'Atenção!')
		Thisform.Pagina.Dados.GetCidade.setfocus
		Return
	EndIf
EndIf
If Inlist( Thisform.pcEscolha,'INSERIR','ALTERAR')
	If Thisform.Pagina.Dados.get_Frete.Value <= 0 And Empty(ThisForm.pDopes)
		=messagebox('Valor do Frete Inválido.',0+64,'Atenção!')
		Thisform.Pagina.Dados.Get_Frete.setfocus
		Return
	EndIf
	
	If Empty(ThisForm.pDopes)
		lcSql = [Select * From SigCdFrt Where Cidas = ']+ThisForm.Pagina.Dados.GetCidade.Value+[']
	Else
		lcSql = [Select * From SigOpFrt Where Cidas = ']+ThisForm.Pagina.Dados.GetCidade.Value+[' And Dopes = ']+ThisForm.pDopes+[' ]
	Endif
	If ThisForm.Podatamgr.Sqlexecute(lcsql,'TmpFrete') > 0 And ;
		Reccount('TmpFrete') > 0 And Inlist( Thisform.pcEscolha,'INSERIR')
		Messagebox( 'Cidade já Cadastrada.', 48, '' )
		ThisForm.Pagina.Dados.GetCidade.SetFocus
		Return
	Endif

	If Not Empty(ThisForm.pDopes)
		Replace Dopes     With ThisForm.pDopes,;
				CidChaves With Iif(Empty(CidChaves),fUniqueIds(),CidChaves) in CrSigCdFrt
	EndIf

Endif

=DoDefault()

ThisForm.mAtivapagina1
ENDPROC
