* PROCEDURE salvardados
------------------------------------------------------------
PROCEDURE salvardados
Local llOks

llOks = .t.
If Not ThisForm.poDataMgr.Update([crSigOpArq])
	=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (Update - crSigOpArq)])
	llOks = .f.
EndIf

If llOks And (ThisForm.poDataMgr.Commit() < 1)
	=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (Commit)])
	llOks = .f.
EndIf

If Not llOks
	ThisForm.poDataMgr.RollBack()
EndIf

Return llOks
* Fim ********************************************************************************* 
ENDPROC

------------------------------------------------------------
* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
Lparameters pKey, pEsc
* Função: Mostrar Arquivos (Endereços de Rede) Relacionados a Chave Passada
* pKey = Chave de Amarração 30 Posições (EmpDopNums, CPros, IClis, etc...) 
* pEsc = Equivalente ao .pcEscolha

ThisForm.cntSombra.lblSombra.Caption = ThisForm.Caption
ThisForm.cntSombra.lblTitulo.Caption = ThisForm.Caption

If Empty(pKey) Or Empty(pEsc)
	=Messagebox([Parâmetros Inválidos!!!], 0+16, [Erro!!!])
	Return .f.
EndIf

lcQryOpArq = [Select * From SigOpArq Where fkChaves = '] + Padr(pKey, 30) + [']

With ThisForm
	.poDataMgr = CreateObject([fSqlConector], .Name)
	.ArqChaves = pKey
	.pcEscolha = pEsc

	If (.poDataMgr.pnIdconn > 0)
		.poDataMgr.AddCursor([SigOpArq], [pkChaves], [crSigOpArq], [], [], lcQryOpArq)

		.poDataMgr.Requery([crSigOpArq])
		Select crSigOpArq
		Update crSigOpArq Set ObsArqs = Nvl(ObsArqs, [])
		Go Top In crSigOpArq

		If Not InList(.pcEscolha, [INSERIR], [ALTERAR])
			.shpAlterar.Visible = .f.
			.btnInsArqs.Visible = .f.
			.btnExcArqs.Visible = .f.
			.cmdBtnForm.btnConfirmar.Enabled = .f.
		EndIf

		* Acertando a Grade de Arquivos
		With ThisForm.grdArquivos
			.AllowHeaderSizing      = .f.
			.AllowRowSizing         = .f.
			.ColumnCount            =  1
			.DeleteMark             = .f.
			.RecordSource           = [crSigOpArq]

			.Column1.ControlSource  = [crSigOpArq.Arquivos]
			.Column1.Movable   		= .f.
			.Column1.Resizable 		= .f.
		EndWith
		
		.grdArquivos.Column1.SetFocus
		.grdArquivos.AfterRowColChange()
	Else
		=Messagebox([Impossível Efetuar Conexão Com o Servidor de Banco de Dados...], 0+48, [Conexão Inválida])
		Return .f.
	EndIf
EndWith
* Fim **************************************************************************************
ENDPROC
