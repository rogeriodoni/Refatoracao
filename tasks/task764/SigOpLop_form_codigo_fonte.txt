PROCEDURE Init
LParameters pFrm

ThisForm.cntSombra.lblSombra.Caption = ThisForm.Caption
ThisForm.cntSombra.lblTitulo.Caption = ThisForm.Caption

ThisForm.ParentForm = pFrm

ThisForm.ParentForm.Enabled = .f.

With ThisForm.grdOperacoes
	.ColumnCount  	   = 2
	.RecordSource      = [LocalSacOp]
	.HeaderHeight 	   = 19
	.DeleteMark   	   = .f.
	.RecordMark   	   = .f.
	.ReadOnly     	   = .f.
	.Scrollbars   	   =  2
	.AllowHeaderSizing = .f.
	.AllowRowSizing    = .f.

	.Column1.ControlSource 	 = [LocalSacOp.Selecionada]
	.Column1.Movable   		 = .f.
	.Column1.Resizable 		 = .f.
	.Column1.ReadOnly  		 = .f.
	.Column1.Sparse    		 = .f.
	.Column1.Check1.AutoSize = .t.
	.Column2.ControlSource   = [LocalSacOp.Dopes]

	For I=2 To .ColumnCount
		lcCol = Alltrim(Str(I))
		.Column&lcCol..FontName  = [Courier New]
		.Column&lcCol..Movable   = .f.
		.Column&lcCol..Resizable = .f.
		.Column&lcCol..ReadOnly  = .t.
	Next
EndWith
* Fim **************************************************************************************
ENDPROC

PROCEDURE KeyPress
LParameters nKeyCode, nShiftAltCtrl

Local Selecionada

If InList(nKeyCode, 13, 32)
	m.Selecionada = Iif(Selecionada = 0, 1, 0)

	Select LocalSacOp
	Gather Memvar Fields Selecionada
	ThisForm.grdOperacoes.Refresh

	NoDefault
EndIf
* Fim ********************************************************************************************
ENDPROC
