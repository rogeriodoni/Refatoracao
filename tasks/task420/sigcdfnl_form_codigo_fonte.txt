* PROCEDURE Init

PROCEDURE Init
LPARAMETERS lcacesso

lcQryFinal = [Select a.*, b.Descrs As Descs from SigFiNal a ] + ;
             [Left Join SigFiRlg b on a.nRelGers = b.Ordems And a.Imps = b.Imps]

With ThisForm
	.poDatamgr = CreateObject('fSqlConector')
	With .poDatamgr
		.AddCursor('SigFiNal','cidchaves','crSigFiNal','',ThisForm.pagina.lista.grade, lcQryFinal)
		.pfSqlTabela(1).pColuna('Imps','','','Impressora',160,.t.)
		.pfSqlTabela(1).pColuna('ordems','','','Ordem',80,.t.)
		.pfSqlTabela(1).pColuna('descrs','','','Descrição',200,.t.)
		.pfSqlTabela(1).pColuna("Iif(CrSigFiNal.nfixos=1, 'SIM', 'NÃO')",'','','Fixo',50,.t.)
		.pfSqlTabela(1).pColuna('ncodfinzs','','','Código',80,.t.)
	EndWith
	= DoDefault()
	.LockScreen = .f.

	With ThisForm.Pagina.Dados.Cmb_Imp
		.Clear
		.AddItem( 'DATAREGIS   ' )
		.AddItem( 'SWEDA       ' )
		.AddItem( 'BEMATECH    ' )
		.AddItem( 'BEMA8501    ' )
		.AddItem( 'NDATAREGIS  ' )
		.AddItem( 'EPSON       ' )
		.AddItem( 'DARUMA      ' )
		.AddItem( 'S/IMPRESSORA' )
		.AddItem( 'NSWEDA      ' )
		.AddItem( 'Em Branco   ' )
	Endwith

EndWith
ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Lista.Grupo_op.Click
------------------------------------------------------------
PROCEDURE Pagina.Lista.Grupo_op.Click
lparameters opcao

select CrSigFiNal
If !dodefault( Opcao )
	ThisForm.mAtivapagina1
	Return
Endif
If thisform.pcEscolha = 'INSERIR'
	Replace CrSigFiNal.Cidchaves With fUniqueIds() in CrSigFiNal
	lncod = fGerUniqueKey('SigFiNal')
	Thisform.Pagina.Dados.get_finalz.value = lncod
Endif
Do Case
	Case Inlist( thisform.pcEscolha,'INSERIR','PROCURAR')
		ThisForm.Pagina.Dados.Cmb_Imp.SetFocus
	Case Inlist( thisform.pcEscolha,'ALTERAR')
		ThisForm.Pagina.Dados.Get_ordem.SetFocus
	Case Inlist( thisform.pcEscolha, 'CONSULTAR', 'EXCLUIR' )
		ThisForm.Pagina.Dados.Grupo_Salva.Salva.SetFocus
Endcase
Select CrSigFiNal

ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
With ThisForm.Pagina.Dados
	if Inlist( thisform.pcEscolha, 'INSERIR')
		lcSql = [Select * From SigFiNal Where Imps = ']+CrSigFiNal.Imps+[' And Ordems = ]+Str(.Get_ordem.Value)
		If ThisForm.Podatamgr.SqlExecute(lcSql,'CrTmpFinal') > 0 And Reccount('CrTmpFinal') > 0
			messagebox('Ordem já cadastrada ! ',0+48,'')
			.get_ordem.setfocus
			Return
		Endif
	EndIf
	if Inlist( thisform.pcEscolha, 'INSERIR')
		lcSql = [Select * From SigFiNal Where Imps = ']+CrSigFiNal.Imps+[' And ncodfinzs = ]+Str(.Get_Finalz.Value)
		If ThisForm.Podatamgr.SqlExecute(lcSql,'CrTmpFinal') > 0 And Reccount('CrTmpFinal') > 0
			messagebox('Codigo já cadastrado ! ',0+48,'')
			.get_Finalz.setfocus
			Return
		Endif
	EndIf
EndWith

=DoDefault()

ThisForm.Refresh
Thisform.mAtivapagina1

ENDPROC

PROCEDURE Valid
LOCAL xBrox

If !IsEmpty( This.Value )

	ThisForm.poDatamgr.SqlExecute([Select * From SigFiRlg Where Imps = '] + Alltrim(This.Parent.Cmb_Imp.Value) + ['], [csTmpRelg])

	SELECT csTmpRelg
	LOCATE FOR Ordems = This.Value

	If EOF()
		xBrox = CreateObject('fwBuscaInt','csTmpRelg','Ordems',This.Value,'Rel. Gerencial',.t.)
		xBrox.mAddColuna('Ordems', '', 'Código', .t.)
		xBrox.mAddColuna('Descrs', '', 'Descrição', .t.)
		xBrox.Show()
	EndIf 

	This.Parent.GetCodRelGer.Value = Iif( LastKey() # 27, Evaluate('csTmpRelg.Ordems'), '' )
	This.Parent.GetDesRelGer.Value = Iif( LastKey() # 27, Evaluate('csTmpRelg.Descrs'), '' )

Else

	This.Parent.GetCodRelGer.Value = []
	This.Parent.GetDesRelGer.Value = []

Endif
ENDPROC
