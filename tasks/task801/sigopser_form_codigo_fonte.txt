* PROCEDURE apaite

PROCEDURE apaite
Select crTpmMvItn3

Replace DPros     With [], ;
		Obs       With [], ;
		CodBarras With 00, ;
		Utilizas  With 00, ;
		TamDe	  With [], ;
		TamPara	  With [], ;
		iConfs	  With .F., ;
		Notas     With [] In crTpmMvItn3

With Thisform
	With .fwGrade
		.Produto.Text1.Value  = []
		.Qtd.Text1.Value      = 0
		.Valor.Text1.Value    = 0
		.Moeda.Text1.Value    = ''
		.Total.Text1.Value    = 0
		.Produto.Text1.Refresh
		.Qtd.Text1.Refresh
		.Valor.Text1.Refresh
		.Moeda.Text1.Refresh
		.Total.Text1.Refresh
		.Refresh
	EndWith
EndWith
* Fim **************************************************************************************
ENDPROC

ENDPROC

* PROCEDURE minserelinha
------------------------------------------------------------
PROCEDURE minserelinha
Select crTpmMvItn3
Locate For cpros = Space(14)
If Eof()
	Insert Into crTpmMvItn3(Emps, Dopes, Numes, Citens) ;
				Values (TprMvCab.Emps, TprMvCab.Dopes, TprMvCab.Numes, ThisForm.Citem)

	Insert Into csTmpDesc (nControle, nLInhas) Values (RecNo([crTpmMvItn3]), ThisForm.nCItem2)

	ThisForm.nCItem2 = ThisForm.nCItem2 + 1
EndIf

Select crTpmMvItn3 
Set Order To CItens
Set Key To ThisForm.Citem

Select crTpmMvItn3
Set Relation To Str(Recno([crTpmMvItn3]),8) Into csTmpDesc

Select crTpmMvItn3
If Not ThisForm.lAutPreco
	Keyboard [{DNARROW}]	
EndIf

Keyboard [{DNARROW}]

ThisForm.OldCodBarras = []
ThisForm.AntCPro = Space(10)
ThisForm.AntValu = Space(3)
* Fim **************************************************************************************
ENDPROC
