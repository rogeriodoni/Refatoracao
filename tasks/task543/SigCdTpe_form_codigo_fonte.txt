------------------------------------------------------------
PROCEDURE mcontaetiqs
lcQueryParam = [Select Count(1) As Total From SigCdTpe Where nSituas = 1]

If (ThisForm.podatamgr.SqlExecute(lcQueryParam, 'csTotal') < 1)
	=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (SigCdTpe)')
	Return 0
EndIf

ThisForm.pagina.lista.Get_EtqAtivas.Value = csTotal.Total
ENDPROC

------------------------------------------------------------
* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
lcQryTPeti = [Select * From SigCdTpe]

With ThisForm
	.poDataMgr = CreateObject('fSqlConector', ThisForm.Name)

	If (.poDataMgr.pnIdconn > 0)
		With .poDataMgr
			.AddCursor('SigCdTpe', 'ntipos', 'crSigCdTpe', '', ThisForm.Pagina.Lista.Grade, lcQryTPeti)
			.pfSqlTabela(1).pColuna('ntipos', '', '', 'Tipo', 50, .t.)
			.pfSqlTabela(1).pColuna('cetiquetas', '', '', 'Descrição', 280, .t.)
			.pfSqlTabela(1).pColuna('nsituas', '', '', 'Situação', 50, .t.)
			.pfSqlTabela(1).pColuna('cordems', '', '', 'Ordem', 50, .t.)

			lcQueryParam = [Select nMaxTpEtis From SigCdPam]

			If (.SqlExecute(lcQueryParam, 'csSigCdPam') < 1)
				=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (SigCdPam)')
				Return 0
			EndIf
			
		EndWith

		If IsEmpty( csSigCdPam.nMaxTpEtis )
			lcQueryParam = [Update SigCdPam Set nMaxTpEtis = 1]
			If (.SqlExecute(lcQueryParam, 'csSigCdPam') < 1)
				=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (SigCdPam)')
				Return 0
			EndIf		
            ThisForm.pagina.lista.Get_EtqAtivas.Value = 1
		Else
          ThisForm.pagina.lista.Get_EtqAtivas.Value = csSigCdPam.nMaxTpEtis
        EndIf

		Thisform.mContaEtiqs()

		With Thisform.Pagina.Lista.Grade
			.SetAll('DynamicForeColor','RGB(Iif(crSigCdTpe.nSituas # 1, 255, 0),0,0)','Column')
		EndWith

		Select crSigCdTpe
		Goto Top

		=DoDefault()

		ThisForm.Pagina.Lista.Grade.Column1.SetFocus
	Else
		=Messagebox('Impossivel Efetuar Conexão com Servidor de Banco de Dados...', 0+48, 'CreateObject')
		Return .f.
	EndIf
EndWith
* Fim **************************************************************************************

ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Lista.Grupo_op.Click
------------------------------------------------------------
PROCEDURE Pagina.Lista.Grupo_op.Click
LPARAMETERS opcao

If Not DoDefault( Opcao )
	ThisForm.mAtivapagina1
	Return
Endif

IF EMPTY(crSigCdTpe.tpespes)
	thisform.pagina.dados.cmbetqesp.ListItemId = 1
ENDIF

thisform.pagina.dados.dobs.Value = ''
IF !EMPTY(crSigCdTpe.codobs)
	thisform.podatamgr.sqlexecute([select descrs from SigCdObs where codigos = ?crSigCdTpe.codobs],[crSigCdObs])
	thisform.pagina.dados.dobs.Value = crSigCdObs.descrs
ENDIF

Do Case
	Case InList( Thisform.pcEscolha, 'CONSULTAR', 'EXCLUIR' )
		Thisform.Pagina.Dados.Grupo_Salva.Salva.SetFocus
		
	Case Thisform.pcEscolha = 'ALTERAR'
		Thisform.Pagina.Dados.Get_cEtiquetas.SetFocus
		
	Case InList( Thisform.pcEscolha, 'INSERIR', 'PROCURAR' )
		Thisform.Pagina.Dados.Get_nTipos.SetFocus
		
EndCase
ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
Local lnTipos, lcEtiquetas

If InList( Thisform.pcEscolha, 'INSERIR', 'ALTERAR' )
	Select crSigCdTpe
	Do Case
		Case IsEmpty( Thisform.Pagina.Dados.Get_nTipos.Value )
			Delete
			
		Case IsEmpty( Thisform.Pagina.Dados.Get_cEtiquetas.Value )
			MessageBox('Descrição da Etiqueta é informação obrigatória.', 48, '')
			Thisform.Pagina.Dados.Get_cEtiquetas.SetFocus
			Return
			
		OtherWise			
			With Thisform.Pagina.Dados
				lnTipos = .Get_nTipos.Value
				lcEtiquetas = .Get_cEtiquetas.Value
			EndWith
            If InList( Thisform.pcEscolha, 'INSERIR')
				If ThisForm.poDataMgr.ChkRegister('SigCdTpe','ntipos',lnTipos)
					MessageBox('Tipo de Etiqueta já cadastrado.', 48, '')
	 				Thisform.Pagina.Dados.Get_nTipos.SetFocus
	                Return
				EndIf
				If ThisForm.poDataMgr.ChkRegister('SigCdTpe','cetiquetas',lcEtiquetas)
					MessageBox('Não é Permitido Etiquetas com a mesma descrição.', 48, '' )
					Thisform.Pagina.Dados.Get_cEtiquetas.SetFocus
	                Return
				Endif
			EndIF
	EndCase
	
	If Not IsEmpty( crSigCdTpe.cOrdems )
		Replace crSigCdTpe.cOrdems With PadL(AllTrim(Str(Val(crSigCdTpe.cOrdems))),3)
	EndIf
EndIf

=DoDefault()
Thisform.mContaEtiqs()
Thisform.mAtivapagina1
ENDPROC
