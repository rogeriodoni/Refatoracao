Local loLista

	loLista = CreateObject('fwBuscaExt', ThisForm.poDataMgr.pnIdConn, 'SigCdMoe', ;
						   'crListaRemota', 'DMoes', This.Value, 'Seleção', .t.)

	EndIf

	ThisForm.get_moeda_desc.Value = crListaRemota.Dmoes
	Use In crListaRemota

	ThisForm.get_moeda.Value = ''
	ThisForm.get_moeda_desc.Value = ''
EndIf

* Fim **************************************************************************************
ENDPROC

* PROCEDURE cmd_ok.Click

PROCEDURE cmd_ok.Click
local lnCopias, lcMoedaRef, lnVlCotRef, lcMoeda, lnVlCotacao, lcTabela, lnVlDesconto
local lnConfirma, lnContador, llExisteImpressao, lnVlRef, lnImpress, llImpBol

if empty(thisform.get_moeda.value)
	messagebox('Moeda inválida !',16,'')
	thisform.get_moeda.setfocus
	return(.f.)
endif

if thisform.get_copias.value < 1
	messagebox('Quantidade de cópias inválida !',16,'')
	thisform.get_copias.setfocus
	return(.f.)
endif 

lcQuery = [Select * ] + ;
		    [From SigCdMoe ] + ;
		   [Order By Cmoes]

If (ThisForm.poDataMgr.SqlExecute(lcQuery, 'crSigCdMoe') < 1)
	=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (crSigCdMoe)')
	Return 0
EndIf
Select crSigCdMoe
Index On cmoes tag cmoes
Index On dmoes tag dmoes

lcQuery = [Select cmoes,datas,horas,valos,cidchaves,usuars,dtalts ] + ;
			[From SigCdCot ] + ;
		   [Order By cmoes,datas,horas,valos,cidchaves,usuars,dtalts ]

If (ThisForm.poDataMgr.SqlExecute(lcQuery, 'crSigCdCot') < 1)
	=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (crSigCdCot)')
	Return 0
EndIf
Select crSigCdCot
Index On cmoes+DTOS(datas) tag CMOEDATA

ThisForm.poDataMgr.CursorQuery('SigOpTdz', 'crSigOpTdz')

lnImpress = aprinters(gaprinters)

llFechaCimp = .f.
llFechaAcImp = .f.
llFechaGrupUsua = .f.
lnEtqImpres = 0

If lnImpress > 0

	* Apanhando Dados de Configuração de Impressoras (SigCdmp)
	pTip = 1
	lcSQLCmd = [Select Impres, nTpImpres From SigCdmp ] + ;
				[Where nTpImpres = ?pTip]
				
	lnSQLErr = ThisForm.Podatamgr.SQLExecute(lcSQLCmd, 'Tmp1')
	If (lnSQLErr < 1)				&& Caso Falhe ao Executar a Instrução SQL Acima
		=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (SigCdmp)')
		Return .F.
	EndIf

	llFlags = .f.
	Select Impres, nTpImpres, llFlags as lMarcas, llFlags as lSituas ;
	  From Tmp1 ;
	  Into Cursor csSigCdmp ReadWrite

	Select csSigCdmp
	Index On Impres Tag Imprime For ( lSituas = .T.)
	Index On Impres Tag Impres
	Goto Top In csSigCdmp

	lcSql = [Select b.Impres, b.nTpImpres  From SigSyImp a, SigCdmp b Where a.UsuAcess = ']+Usuar+[' And a.cImps = b.Impres And b.nTpImpres = ?pTip ]+;
			[Union all ]+;
			[Select c.Impres, c.nTpImpres  From SigCdAcG a, SigSyImp b, SigCdmp c Where a.Usuarios=']+Usuar+[' ]+;
			[And a.Grupos = b.GrAcess And b.Cimps = c.Impres And c.nTpImpres = ?pTip]

	If ThisForm.Podatamgr.sqlexecute(lcSql,'CrTmpCimp') <= 0
		Messagebox('Impossivel Efetuar Conexão com Servidor de Banco de Dados.',(0+48),goSistema.TituloSistema)
		Return .f.
	EndIf

	Select CrTmpCimp
	If Reccount() > 0
		Select Impres, nTpImpres  From CrTmpCimp Into Cursor LocalcImp Group by Impres, nTpImpres ReadWrite
		llFlags = .f.
		Select *, llFlags as lMarcas, llFlags as lSituas ;
		  From LocalcImp ;
		  Into Cursor csSigCdmp ReadWrite

		Select csSigCdmp
		Index On Impres Tag Imprime For ( lSituas = .T.)
		Index On Impres Tag Impres
		Goto Top In csSigCdmp
	Endif
 	For I=1 to APrinters(gaPrinters)
 		lcImpres = Upper(gaPrinters(I, 1))
 		If Seek(lcImpres, 'csSigCdmp', 'Impres')
	 		lnEtqImpres = lnEtqImpres + 1
	 		Select csSigCdmp
	 		Replace lMarcas With .T., ;
	 				lSituas With .T.
	 	Endif
	Next
	
	If (lnEtqImpres = 0)
		Messagebox( 'Nenhuma Impressora de Etiqueta Está Configurada no' + Chr(13) + ;
					'Painel de Controle.  Contate o Suporte Técnico.', 0+48, ;
					'Sem Impressora Configurada no Painel de Controle' )
		Return ''
	Else
		If (lnEtqImpres # 1)	&& Se Tiver Mais Que Uma Impressora, Habilita Formulário
			Do Form SigOpIpb	&& Seleciona Impressora Para Etiqueta / Boletos	
		EndIf
	EndIf
	
	Select csSigCdmp
	Locate for lSituas

	Set Printer To Name (AllTrim(CsSigCdmp.impres))
	set device to printer
else
	messagebox('Não Existe Impressora Instalada !',0+48,'')					
endif

lnConfirma = messagebox('Confirma impressão das tabelas de referência ?',32+4,'')
if lnConfirma <> 6
	Set Printer to Default
	Set Device to Screen
	return(.f.)
endif

* criação do arquivo de cabeçalho
create cursor dbcabecalho (moeda_referencia c(45),cabecalho c(45),linha c(45))

* criação do arquivo de impressão
create cursor dbrelatorio (detalhe c(45))

* criação do arquivo temporário
create cursor dbtemp1 (cd_moeda c(4),nm_tabela c(30),vl_cotacao n(9,2))

* inicializa variáveis de campos
lnCopias   = thisform.get_copias.value
lcMoedaRef = thisform.get_moeda.value
lnVlCotRef = fBuscarCambio(ctod(''),date(),space(03),lcMoedaRef)

* loop principal
llExisteImpressao = .f.

Select crSigCdMoe
scan for crSigCdMoe.cmoes <> lcMoedaRef and crSigCdMoe.TabRefs 
	lcMoeda     = crSigCdMoe.cmoes
	lcMoedaI    = ' ' + crSigCdMoe.cmoes
	lnVlCotacao = fBuscarCambio(ctod(''),date(),space(03),lcMoeda)

	select crSigOpTdz
	scan for crSigOpTdz.rel_descs = 1
		llExisteImpressao = .t.
		lcTabela     = crSigOpTdz.codigos
		lnVlDesconto = 1-(crSigOpTdz.descos/100)
		lnVlRef      = iif(crSigCdMoe.FMults = 'S',round(lnVlCotRef*lnVlCotacao*lnVlDesconto,2),round(lnVlCotacao*lnVlDesconto,2))
		insert into dbtemp1 (cd_moeda,nm_tabela,vl_cotacao) ;
				values (lcMoedaI,lcTabela,lnVlRef)
	endscan
endscan

if not llExisteImpressao
	messagebox('Não há tabelas de desconto selecionadas para impressão !',16,'')
	Thisform.Release
	Return(.t.)
EndIf

select a.cd_moeda,a.nm_tabela,str(a.vl_cotacao,9,2) as vl_cotacao;
	from dbtemp1 a into cursor dbtemp2 Order by a.Cd_Moeda

use dbf('dbtemp2') again alias dbtemp3 in 0 &&order dbtemp2
select dbtemp3

goto top
do VFPxtab with 'dbtemporario',.T.
select dbtemporario
lnTotalRegistros = afields(lvRelatorio)

scan
	lcReferencia = padr(&lvRelatorio(1,1).,5)
	if empty(field(2))
		lcCampo1 = space(07)
	else
		lcCampo1 = padl(alltrim(right(&lvRelatorio(2,1).,7)),7)
	endif
	if empty(field(3))
		lcCampo2 = space(07)
	else
		lcCampo2 = padl(alltrim(right(&lvRelatorio(3,1).,7)),7)
	endif
	if empty(field(4))
		lcCampo3 = space(07)
	else
		lcCampo3 = padl(alltrim(right(&lvRelatorio(4,1).,7)),7)
	endif
	if empty(field(5))
		lcCampo4 = space(07)
	else
		lcCampo4 = padl(alltrim(right(&lvRelatorio(5,1).,7)),7)
	endif
	
	lcDetalhe = lcReferencia+space(2)+lcCampo1+space(2)+lcCampo2+space(2)+lcCampo3+space(2)+lcCampo4
	insert into dbrelatorio (detalhe) ;
				values (lcDetalhe)
endscan


* atualiza cursor de cabeçalho
lcMoeda_Referencia = 'MOEDA: '+alltrim(lcMoedaRef)+' ('+alltrim(str(lnVlCotRef,9,2))+')'
lcMoeda            = 'REF. '
dimension lvCabecalho(4)
lvCabecalho(1) = ''
lvCabecalho(2) = ''
lvCabecalho(3) = ''
lvCabecalho(4) = ''
lnContador = 1
select crSigOpTdz
scan for lnContador < 5 and crSigOpTdz.rel_descs = 1
	lvCabecalho(lnContador) = padl(alltrim(left(crSigOpTdz.descrs,7)),7)
	lnContador = lnContador + 1
endscan

select dbtemporario
lcCabecalho = lcMoeda+space(2)+lvCabecalho(1)+space(2)+lvCabecalho(2)+space(2)+lvCabecalho(3)+space(2)+lvCabecalho(4)
lcLinha = replicate('=',5)+space(2)+replicate(iif(empty(field(2)),' ','='),7)+space(2)+replicate(iif(empty(field(3)),' ','='),7)+space(2)+replicate(iif(empty(field(4)),' ','='),7)+space(2)+replicate(iif(empty(field(5)),' ','='),7)
insert into dbcabecalho (moeda_referencia,cabecalho,linha) ;
		 values (lcMoeda_Referencia,lcCabecalho,lcLinha)

select dbrelatorio

for lnContador = 1 to lnCopias
	
	* impressão da área de cabeçalho
	??? '' + chr(13) + chr(10)
	??? replicate('=',40) + chr(13) + chr(10)
	??? 'TABELAS DE REFERENCIA EM ' + DToC(Date()) + chr(13) + chr(10)
	??? replicate('=',40) + chr(13) + chr(10)
	??? dbcabecalho.moeda_referencia + chr(13) + chr(10)
	??? '' + chr(13) + chr(10)
	??? dbcabecalho.cabecalho + chr(13) + chr(10)
	??? dbcabecalho.linha + chr(13) + chr(10)
	scan
		??? dbrelatorio.detalhe + chr(13) + chr(10)
	endscan
	??? '' + chr(13) + chr(10)
	??? '' + chr(13) + chr(10)
	??? replicate('-',40) + chr(13) + chr(10)
	??? '' + chr(13) + chr(10)
	??? '' + chr(13) + chr(10)
	??? '' + chr(13) + chr(10)
	??? '' + chr(13) + chr(10)
	??? '' + chr(13) + chr(10)
	??? '' + chr(13) + chr(10)	
	??? '' + chr(13) + chr(10)
	??? '' + chr(13) + chr(10)	
	
next lnContador

If !llImpBol
	??? Chr(12)
EndIf

set printer to default
set device to screen

ThisForm.Release

ENDPROC
