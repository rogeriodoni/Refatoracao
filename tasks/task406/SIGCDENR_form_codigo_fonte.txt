* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
lcQryEnr = [Select * From SigCdEnr]

With ThisForm
	.poDataMgr = CreateObject('fSqlConector', ThisForm.Name)

	If (.poDataMgr.pnIdconn > 0)
		With .poDataMgr
			.AddCursor('SigCdEnr', 'pkChaves', 'crSigCdEnr', '', ThisForm.Pagina.Lista.Grade, lcQryEnr)
			.pfSqlTabela(1).pColuna('ceps', '', '', 'Cep', 100, .t.)
			.pfSqlTabela(1).pColuna('endes', '', '', 'Endereço', 200, .t.)
			.pfSqlTabela(1).pColuna('bairs', '', '', 'Bairro', 100, .t.)
		EndWith

		Select crSigCdEnr
		Index On ceps tag ceps
		Set Order To ceps
		ThisForm.Pagina.Lista.Grade.Column1.SetFocus
		=DoDefault()
	Else
		=Messagebox('Impossivel Efetuar Conexão com Servidor de Banco de Dados...', 0+48, 'CreateObject')
		Return .f.
	EndIf
EndWith
* Fim **************************************************************************************



ENDPROC

* PROCEDURE Pagina.Lista.Grupo_op.Click
------------------------------------------------------------
PROCEDURE Pagina.Lista.Grupo_op.Click
LPARAMETERS opcao
If !dodefault( Opcao )
	ThisForm.mAtivapagina1
	Return
Endif

Select crSigCdEnr

* posiciona o cursor de acordo com a opção
Do Case
	Case Inlist( ThisForm.pcEscolha, 'INSERIR', 'PROCURAR', 'ALTERAR' )
		thisform.pagina.dados.getCep.setfocus	
	Case Inlist( ThisForm.pcEscolha, 'CONSULTAR', 'EXCLUIR' )
		thisform.pagina.dados.grupo_salva.salva.setfocus
Endcase

Select crSigCdEnr
=DoDefault()
ThisForm.Pagina.Dados.Refresh

ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
LOCAL lcQuery
private lcQuery, lcCep, lcEnd, lcBai

If ThisForm.pcEscolha = 'INSERIR'
	*** Verifica se registro ja cdastrado ***
	lcCep = ThisForm.Pagina.Dados.GetCep.Value
	lcEnd = ThisForm.Pagina.Dados.GetEndereço.Value
	lcBai = ThisForm.Pagina.Dados.GetBairro.Value

	lcQuery = 'Select * From SigCdEnr Where ceps=?lcCep And endes=?lcEnd And bairs=?lcBai'
	ThisForm.poDataMgr.SqlExecute(lcQuery, [crLocal])

	IF RECCOUNT('crLocal') > 0
		MessageBox( 'Endereço já cadastrado...', 48, '' )
		ThisForm.Pagina.Dados.GetCep.SetFocus
		SELECT crSigCdEnr
		Return
	EndIf

	If IsEmpty(ThisForm.Pagina.Dados.getCEP.Value)
		MessageBox( 'Obrigatório Informar o Cep.', 48, '' )
		ThisForm.Pagina.Dados.GetCep.SetFocus
		Return
	ENDIF  

	Replace pkChaves With fUniqueIds() in crSigCdEnr
ENDIF
SELECT crSigCdEnr
=DoDefault()
ThisForm.mAtivapagina1

ENDPROC

* PROCEDURE LostFocus
------------------------------------------------------------
PROCEDURE LostFocus
If Not Empty(Strtran(This.Value, '-', ''))
	loCep = CreateObject('fwCep', This.Value, 'CEPS', .f.)
	loCep.Show

	If Not loCep.plAchouRegistro
		loCep.Release
		Return .f.
	EndIf

	lcCep = loCep.Ceps
	lcEnd = loCep.Nomes
	lcTpC = loCep.TipoCeps
	lcTpN = loCep.TipoNomes
	lcBai = loCep.Bairros

	loCep.Release

	If Not Empty(At([,], lcEnd)) And (lcTpC <> [9])
		lcEnd = Alltrim(Substr(lcEnd, At([,], lcEnd) + 1)) + [ ] + Substr(lcEnd, 1, At([,], lcEnd) - 1)
	EndIf

	Replace Endes With Alltrim(lcTpN) + [ ] + Alltrim(lcEnd), ;
			Bairs With Alltrim(lcBai), ;
			Ceps  With lcCep In crSigcdenr

	ThisForm.Refresh
ENDIF

ENDPROC
