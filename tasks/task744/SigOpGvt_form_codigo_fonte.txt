------------------------------------------------------------
* PROCEDURE Click

PROCEDURE Click
Local llOk
ThisForm.Mensagem.Caption = 'Verificando Status'
llOk = .t.
DO Case
	Case  InList(ThisForm.Impressora,'DATAREGIS','NDATAREGIS')
		VerStatus = RotStatus(.t.,ThisForm.oFormulario)
		If Type('VerStatus')<>'L' And Substr(VerStatus,4,1) = 'S'
			llOk = .f.
		Endif
	Case InList(ThisForm.Impressora,'SWEDA','NSWEDA')
		VerStatus = StatGaveta(ThisForm.oFormulario)
		If VerStatus = '0'
			llOk = .f.
		Endif
	Case InList(ThisForm.Impressora,'BEMATECH','BEMA8501','DARUMA','EPSON')
		VerStatus = StatGaveta(Thisform.oFormulario)
		If VerStatus = '00'
			llOk = .f.
		Endif
	Case ThisForm.Impressora = 'ELGIN'
		VerStatus = StatGaveta(ThisForm.oFormulario)
		If VerStatus = '0'
			llOk = .f.
		Endif

	Case ThisForm.Impressora = 'Gaveta Serial'
		If Not fChecarSerial(Thisform.Portaserial)
			llOk = .f.
		Endif
EndCase
If Not llOk
	ThisForm.Mensagem.Caption = 'Gaveta Aberta... Feche'
	Return .f.
Endif

ThisForm.Release
ENDPROC
