PROCEDURE Init
LParameters ParentForm, pdDatas

ThisForm.cntSombra.lblSombra.Caption = ThisForm.Caption
ThisForm.cntSombra.lblTitulo.Caption = ThisForm.Caption

ThisForm.ParentForm = ParentForm

If (Type([pdDatas]) = [D])
	ThisForm.dDatas = pdDatas
Else	
	ThisForm.dDatas = Date()
EndIf

ThisForm.ParentForm.poDataMgr.CursorQuery([SigCdUsu], [crSigCdUsu], [Usuarios], Padr(Alltrim(Usuar),10))

ThisForm.btnCotacao.Visible = Not (Seek(crSigCdUsu.Ccargs, [crSigCdCrg], [Ccargs]) And ;
							  (crSigCdCrg.AltCots = [S]) And InList(ThisForm.ParentForm.pcEscolha, [INSERIR], [ALTERAR]))
ThisForm.shpBtnGs.Visible = ThisForm.btnCotacao.Visible
* Fim **************************************************************************************
ENDPROC

PROCEDURE Click
Do Form SigOpAlC With TmpCot.CMoes, TmpCot.Valos, ThisForm.dDatas To lcRetorno

If (MLine(lcRetorno, 1) = [OK])
	Replace Valos With Val(MLine(lcRetorno, 2)) In TmpCot

	If Used([TprMvCab])
		Replace CotUsus With MLine(lcRetorno, 3) In TprMvCab
	EndIf
EndIf
