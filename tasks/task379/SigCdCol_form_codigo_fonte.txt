PROCEDURE msv_procurar
Local lcBusca

If .Not.IsEmpty(ThisForm.Pagina.Dados.Getcolecoes.Value)
	ThisForm.poDataMgr.LocateCursor('crSigCdCol','colecoes',ThisForm.Pagina.Dados.Getcolecoes.Value)
Else
	If .Not.IsEmpty(ThisForm.Pagina.Dados.Getdescs.Value)
		ThisForm.poDataMgr.LocateCursor('crSigCdCol','descs',ThisForm.Pagina.Dados.Getdescs.Value)
	EndIf
EndIf

ENDPROC

------------------------------------------------------------
* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
LParameters lcacesso

With ThisForm
	.poDatamgr = CreateObject('fSqlConector')
	With .poDatamgr
		.AddCursor('SigCdCol','colecoes','crSigCdCol','',ThisForm.pagina.lista.grade)
		.pfSqlTabela(1).pColuna('colecoes','','','Grupo',79,.t.)
		.pfSqlTabela(1).pColuna('descs','','','Descrição',290,.t.)
		.pfSqlTabela(1).pColuna('codcols','','','Cod. Coletor',75,.t.)
	EndWith
	= DoDefault()
	.LockScreen = .f.
EndWith
ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Lista.Grupo_op.Click
------------------------------------------------------------
PROCEDURE Pagina.Lista.Grupo_op.Click

Lparameters opcao

ThisForm.LockScreen = .t.

If Type( 'Opcao' ) = 'N'
	This.Value = Opcao

EndIf 

If This.Value == 4 And ThisForm.poDataMgr.ChkRegister('SigCdPro', 'Colecoes', crSigCdCol.Colecoes)
	= MessageBox('Existem Produtos Utilizando Esse Grupo de Venda.', 0+48, 'Impossível Exclusão...')

	Thisform.mAtivapagina1
	ThisForm.LockScreen = .f.
	Return
EndIf

If DoDefault(m.Opcao)

	Select CrSigCdCol

	If (ThisForm.pcEscolha = [INSERIR])
		Replace Usuars With Usuar, ;
				DtIncs With Datetime() In crSigCdCol
	EndIf
	
	Do Case
		Case Inlist(ThisForm.pcEscolha,'INSERIR','PROCURAR')
			  ThisForm.Pagina.Dados.getColecoes.SetFocus	
		Case ThisForm.pcEscolha=='ALTERAR'
			  ThisForm.Pagina.Dados.getDescs.SetFocus	
		Case Inlist(ThisForm.pcEscolha,'CONSULTAR','EXCLUIR')
			  ThisForm.Pagina.Dados.Grupo_Salva.Salva.SetFocus
	Endcase
	
	Thisform.Pagina.Dados.Say3.Visible=.T.
	Thisform.Pagina.Dados.Get1.Visible=.T.
	
	ThisForm.Pagina.Dados.Refresh

Else
	ThisForm.mAtivapagina1
Endif

ThisForm.LockScreen = .f.
ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
Local llRetorno

m.llRetorno = .t.

If Inlist(ThisForm.pcEscolha,'INSERIR','ALTERAR')

	With ThisForm.Pagina.Dados

		Do Case
			Case IsEmpty(.getColecoes.Value)
			
				  MessageBox('Grupo de Venda Inválido.',48,'')
				  .getColecoes.SetFocus
				  m.llRetorno = .f.

			Case IsEmpty(.getDescs.Value)
			
				  MessageBox('Descrição Inválida.',48,'')
				  .getDescs.SetFocus
				  m.llRetorno = .f.

			Case !IsEmpty(.get1.Value)
				 If ThisForm.poDataMgr.ChkRegister('SigCdCol','codcols',ThisForm.Pagina.Dados.Get1.Value)
					=MESSAGEBOX("Código do Coletor já Existe", 16, "Atenção")
			 		.get1.SetFocus
					m.llRetorno = .f.
				 Endif   
				  
			Case ThisForm.poDataMgr.ChkRegister('SigCdCol','colecoes',ThisForm.Pagina.Dados.Getcolecoes.Value) ;
					And ThisForm.pcEscolha='INSERIR'

	               Messagebox('Grupo de Venda Já Cadastrada.',0+64,goSistema.NomeSistema)
				   .getColecoes.SetFocus
				   m.llRetorno = .f.
				  
		EndCase

	EndWith

Endif

If m.llRetorno
	= DoDefault()
	ThisForm.mAtivapagina1
EndIf
ENDPROC

ENDPROC

Local loLista

						   'crListaRemota', 'Usuarios', This.Value, 'Seleção', .t., .f., [Not cAtivos='N'])

		loLista.mAddColuna('Usuarios', '', 'Usuário')
		loLista.Show()
	EndIf
	
	This.Value = Iif(Lastkey()=27, '', crListaRemota.Usuarios)
	Use In crListaRemota
EndIf
This.Refresh

If Not Empty(This.Value) And IsEmpty(crSigCdCol.DtIncs)
	Replace DtIncs With Datetime() In crSigCdCol
EndIf
