PROCEDURE calctotais
lnTotQtde  = 0
lnTotPeso  = 0
lnTotPesoC = 0
lnTotPes   = 0

select xvalite
zap
set order to moeda
select crTpmMvItn
scan
	If IsEmpty(Cpros)
		Loop
	EndIf
	lnMulti = Iif(crSigCdOpe.opers=3,Iif(crTpmMvItn.Opers=[E],-1,1),Iif(crTpmMvItn.Opers=[E],1,-1))

	lnTotPes = lnTotPes + crTpmMvItn.TPesos

	If crSigCdOpe.PesoVts = 1
		_MoePeso = Iif( IsEmpty( crSigCdOpe.Cmoes ), crTpmMvItn.MoeFats, crSigCdOpe.Cmoes )
		Select xValIte
		If !Seek( _MoePeso )
			Insert into xValite ( Moeda ) Values ( _MoePeso )
		Endif
		Replace Total With Total + ( lnMulti * crTpmMvItn.TPesos )
	Endif

	If crSigCdOpe.PrecoVts = 1
		*_MoePreco = Iif( IsEmpty( crSigCdOpe.Cmoes ), crTpmMvItn.Moedas, crSigCdOpe.Cmoes )
		_MoePreco = Iif( Not IsEmpty( crTpmMvItn.Moedas ), crTpmMvItn.Moedas, crSigCdOpe.Cmoes )
		Select xValIte
		If !Seek( _MoePreco )
			Insert into xValite ( Moeda ) Values ( _MoePreco )
		Endif
		Replace Total With Total + ( lnMulti * crTpmMvItn.Totas )
	EndIf
	
	=ThisForm.podatamgr.cursorquery('SigCdUni','CrTmpUni','Cunis',crTpmMvItn.Cunis)

	If InList(CrSigCdOpe.ChkQtds,6,7)
		_ChkQtds  = CrSigCdOpe.ChkQtds
		lnTotQtde = lnTotQtde + ( Iif( (_ChkQtds=6 And crTpmMvItn.Opers='S') Or (_ChkQtds=7 And crTpmMvItn.Opers='E') , 1, 0) * crTpmMvItn.Qtds )
	Else
		lnTotQtde = lnTotQtde + ( lnMulti * crTpmMvItn.Qtds )
	Endif
	lnQtde 	 = crTpmMvItn.Qtds

	Select crTpmMvIts
	Set Order To
	Set Order to CItens
	If Seek(crTpmMvItn.CItens)
		Scan While CItens = crTpmMvItn.CItens
			If InList(CrSigCdOpe.ChkQtds,6,7)
				_ChkQtds  = CrSigCdOpe.ChkQtds
				lnTotPeso = lnTotpeso + ( Iif( (_ChkQtds=6 And crTpmMvItn.Opers='E') Or (_ChkQtds=7 And crTpmMvItn.Opers='S') , 1, 0) * crTpmMvIts.Pesos * crTpmMvIts.Qtds  )
			Else
				lnTotPeso = lnTotPeso + ( lnMulti * crTpmMvIts.Pesos * crTpmMvIts.Qtds )
			Endif
		EndScan
	Else
		If InList(CrSigCdOpe.ChkQtds,6,7)
			_ChkQtds  = CrSigCdOpe.ChkQtds
			lnTotPeso = lnTotpeso + ( Iif( (_ChkQtds=6 And crTpmMvItn.Opers='E') Or (_ChkQtds=7 And crTpmMvItn.Opers='S') , 1, 0) * ;
						Round(Iif(Empty(crTpmMvItn.Codbarras) And CrtmpUni.Cestos='2',lnQtde,1) * crTpmMvItn.Pesos,2) )
		Else
			lnTotPeso = lnTotPeso +  ( lnMulti * ;
						Round(Iif(Empty(crTpmMvItn.Codbarras) And CrtmpUni.Cestos='2',lnQtde,1) * crTpmMvItn.Pesos,2) )
		Endif
	Endif

	If InList(Thisform.ParentForm.Op_Escolha,'INSERIR','ALTERAR')
		lnPeso = crTpmMvItn.Pesos + TprMvCab.VCompensas - iif(Thisform.ParentForm.Op_Escolha='ALTERAR',TprMvCab.VCompensa2,0)
		lnTotPesoC = lnTotPesoC + lnMulti * Round(iif(Empty(crTpmMvItn.Codbarras) And crTmpUni.CEstos='2',crTpmMvItn.Qtds,1) * lnPeso,2)
	Else
		lnTotPesoC = lnTotPeso
	EndIf
EndScan

With ThisForm
	.Get_QOper.Value    = Iif(lnTotQtde = 0,' ',Iif(lnTotQtde > 0,'E','S'))
	.get_TotQtde.Value  = Abs(lnTotQtde)
	.Get_POper.Value    = Iif(lnTotPeso = 0,' ',Iif(lnTotQtde > 0,'E','S'))
	.Get_TotPeso.Value  = Abs(lnTotPeso)
	.Get_TotPesoC.Value = Abs(lnTotPesoC)
	.getTotPes.Value	= Abs(lnTotPes)
EndWith

lnvn  = 0
lnEtq = 0
lnCol = 0

select xvalite
Delete For total = 0

If CrSigOpCdd.MtCubs <> 0
	ThisForm.Get_mtCub.Value = ThisForm.ParentForm.ParentForm.VerifMtCub(.t.)
	ThisForm.Label5.Visible = .t.
	ThisForm.Get_mtCub.Visible = .t.
Endif

Select xValIte
Go Top
* Fim **************************************************************************************
ENDPROC

PROCEDURE Init
LParameters pform,datses

ThisForm.cntSombra.lblSombra.Caption = ThisForm.Caption
ThisForm.cntSombra.lblTitulo.Caption = ThisForm.Caption

Local lnTotQtde, lnTotPeso, lnQtde

lcCidQuery3 = fUniqueIds()

Set Datasession To DatSes
Thisform.parentform = pform

lcQueryTmpd  = [Select * From SigTempD Where CidQuerys = ?CidQuery3 ]

With ThisForm
	.poDataMgr = CreateObject('fSqlConector', ThisForm.Name)
	With .poDataMgr
		.Addcursor('SigTempD','CidQuerys','CrSigTempD','','',lcQuerytmpd)
	Endwith
EndWith


ThisForm.Get_Compensacao.ControlSource = 'TprMvCab.VCompensas'
ThisForm.Get_Compensacao.Refresh
ThisForm.lblTotPes.Caption = Iif(Not Empty(crSigOpCdc.DocQtd2s),'Total '+alltrim(crSigOpCdc.DocQtd2s),'Total Peso') + [ :]

llTemCompo = .f.
llTemServ  = .f.

If Not Used([crSigCdOpe])
	lcQuery = [Select * ] + ;
			    [From SigCdOpe ] + ;
			    [Where Dopes = '] + TprMvCab.Dopes + [']

	If (ThisForm.poDataMgr.SqlExecute(lcQuery, [crSigCdOpe]) < 1)
		=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (crSigCdOpe)])
		Return 0
	EndIf
EndIf

If crSigCdOpe.TotCompos = 1

	Select CrSigTempD
	Zap
	Select Cpros,Sum(Qtds) as Qtds From crTpmMvItn Group by Cpros Into cursor wEesti
	
	Select wEesti
	Scan
		Select CrSigTempD
		Append Blank
		Replace Cpros      With wEesti.Cpros,;
				CidQuerys  With lcCidQuery3,;
				Qtds       With wEesti.Qtds,;
				CidChaves  With fUniqueIds()
	EndScan
	If !ThisForm.Podatamgr.UpDate('CrSigTempD')
		=ThisForm.Podatamgr.RollBack()
		MessageBox('Favor reinicializar o processo.',16,'Falha na Conexão')
		Thisform.release
		Return .f.
	EndIf
	If ThisForm.PoDataMgr.commit() < 1
		MessageBox('Favor reinicializar o processo.',16,'Falha na Conexão')
		Thisform.release
		Return .f.
	Endif
		
	lcQuery = [Select b.Mats as Codigos, sum(a.Qtds * b.Qtds) as Qtds, b.UniCompos, Sum(a.Qtds * b.QtsCons) as QtsCons From SigTempD a, SigPrCpo b ]+;
			  [Where a.CidQuerys = ?lcCidQuery3 And a.cpros = b.cpros Group by b.mats, b.unicompos]

	If ThisForm.PoDatamgr.SqlExecute(lcQuery,'CsCompo') < 1
		MessageBox('Favor reinicializar o processo.',16,'Falha na Conexão')
		Thisform.release
		Return .f.
	EndIf

	lcDelete = [Delete From SigTempD Where CidQuerys = ?lcCidQuery3 ]
	If ThisForm.Podatamgr.SqlExecute(lcDelete,'') < 1
		MessageBox('Favor reinicializar o processo.',16,'Falha na Conexão')
		ThisForm.release
		Return .f.
	EndIf
	
	Select CsCompo
	Go Top
	If Not Eof('csCompo')
		With Thisform.GrdTCompo
			.RecordSource = 'csCompo'
			.Visible = .T.
			.Enabled = .T.
			.Column1.ControlSource = 'csCompo.Codigos'
			.Column2.ControlSource = 'csCompo.Qtds'
			.Column3.ControlSource = 'csCompo.UniCompos'
			.Column4.ControlSource = 'csCompo.QtsCons'
		EndWith
		llTemCompo = .t.
	EndIf
EndIf

Thisform.CalcTotais()

With Thisform.Grade_Total
	.Visible = .t.
	.Recordsource = 'xValIte'
	.ChildOrder = 'Moeda'
	.Moeda.ControlSource = 'xValIte.Moeda'
	.Total.ControlSource = 'xValIte.Total'
	.Refresh
EndWith

Select crTpmMvItn
Select Cpros, Sum(Qtds) as Qtds From crTpmMvItn Where Not Empty(Cpros) Group by Cpros Into cursor ProEesti ReadWrite 

Select ProEesti
Go Top 

With ThisForm.Grade_Produto
	.Visible = .t.
	.Recordsource = 'ProEesti'
	.Produto.ControlSource = 'ProEesti.Cpros'
	.Total.ControlSource = 'ProEesti.Qtds'
	.Refresh
EndWith

*** Mena - 20/10/2017 - mostra total de produtos por tamanho
Select Cpros+'-'+ALLTRIM(codtams) as cpros, Sum(Qtds) as Qtds From crTpmMvIts Where Not Empty(Cpros) Group by Cpros,Codtams Into cursor ProTam ReadWrite 
Select ProTam
Go Top 
With ThisForm.grade_ProdutoTam
	.Visible = .t.
	.Recordsource = 'ProTam'
	.Produto.ControlSource = 'ProTam.Cpros'
	.Total.ControlSource = 'ProTam.Qtds'
	.Refresh
EndWith
***************************************************************

If CrSigCdOpe.HabServs = 1
	Select xvalite
	=AFields(v01)
	Create Cursor xValServ From Array v01
	Select xValServ
	Index On Moeda Tag Moeda

	Select crTpmMvItn3
	Set Order To
	Scan
		If IsEmpty(Cpros)
			Loop
		Endif
		_MoePreco = Iif( IsEmpty( crSigCdOpe.Cmoes ), crTpmMvItn3.Moedas, crSigCdOpe.Cmoes )
		Select xValServ
		If !Seek( _MoePreco )
			Insert into xValServ ( Moeda ) Values ( _MoePreco )
		Endif
		Replace Total With Total + crTpmMvItn3.Totas
	Endscan
	select xvalServ
	Delete For total=0
	Goto top

	With Thisform.Grade_Serv
		.Visible = .t.
		.Recordsource = 'xValServ'
		.ChildOrder = 'Moeda'
		.Moeda.ControlSource = 'xValServ.Moeda'
		.Total.ControlSource = 'xValServ.Total'
		.Refresh
	EndWith
	llTemServ = .t.
EndIf

*** Mena - 20/10/2017 - a grade de produto+tamanho estã na mesma posicao do servico ***
If Not llTemServ
	ThisForm.Grade_Serv.Visible = .f.
	ThisForm.Grade_ProdutoTam.Visible = .T.
ELSE
	ThisForm.Grade_Serv.Visible = .T.
	ThisForm.Grade_ProdutoTam.Visible = .F.
EndIf

If Not llTemCompo
	ThisForm.GrdTCompo.Visible = .f.
EndIf
* Fim **************************************************************************************
ENDPROC
