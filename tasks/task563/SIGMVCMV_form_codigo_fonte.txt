------------------------------------------------------------
* PROCEDURE processamento
------------------------------------------------------------
PROCEDURE processamento

Local ldDtIni, ldDtFim, lcEmp, lcMoeda, lcTransacao

Private pDtI, pDtF, lnNopers

ldDtIni = Thisform.getDtInicial.Value
ldDtFim = Thisform.getDtFinal.Value
lcEmp	= Thisform.getEmpresa.Value
lcMoeCV = Thisform.get_cd_moeda.Value

Select CrAltCc
Zap

If Used([csMovimento])
	Use In csMovimento
Endif

Create Cursor csMovimento(Emps c(3), Dopes c(20), DataS T, custofs N(12,3), pcuss N(12,5), moecusfs c(3), moedas c(3), qtds N(10,3),;
	GrupoEDs c(11), ContaEDs c(11), GrupoECs c(11), ContaECs c(11), MoeOpFin c(3), Cotacaos N(16,7), Numes n(6), EmpDopNums c(29))

Select csMovimento
Index On Emps + Dopes + Ttoc(DataS) Tag EDD


pDtI = fDtoSQL(ldDtIni)
pDtF = fDtoSQL(ldDtFim,[23:59:59])

lcQuery = [select sigmvcab.valos, sigmvcab.Datas, sigmvcab.Emps, sigmvcab.dopes, sigmvcab.Numes, sigmvcab.empdopnums ] + ;
	[from sigmvcab ] + ;
	[join sigcdope on sigcdope.dopes = sigmvcab.Dopes ] + ;
	[join SigOpCdc on SigOpCdc.dopes = sigcdope.dopes ] + ;
	[And SigOpCdc.opeinatvs = 0 ] + ;
	[where sigcdope.vendas=1 and sigcdope.copers in(1,2) and sigcdope.caixas=1 ] + ;
	[And empdopnums not in ( Select distinct a.empdopnums from sigmvccr a ] + ;
	[join sigcdope on sigcdope.dopes = a.Dopes ] + ;
	[and sigcdope.vendas=1 and sigcdope.copers=1 and sigcdope.caixas=1 ] + ;
	[join SigOpCdc on SigOpCdc.dopes = sigcdope.dopes ] + ;
	[And SigOpCdc.opeinatvs = 0 ] + ;
	[join (select GrupoEDs, ContaEDs, GrupoECs, ContaECs, SigOpFin.Dopes, SigOpFin.Moedas ] + ;
	[from SigOpFin ] + ;
	[join SigOpCdc on SigOpCdc.dopes = SigOpFin.dopes ] + ;
	[And SigOpCdc.opeinatvs = 0 ] + ;
	[where contaeds <> space(11)) b on a.dopes = b.dopes ] + ;
	[and a.contas = b.contaecs and a.scontas = b.ContaEDs ] + ;
	[and a.grupos = b.GrupoECs and a.sgrupos = b.GrupoEDs)] + ;
	[And sigmvcab.Datas BetWeen ?pDtI and ?pDtF ] + ;
	Iif(Not Empty(lcEmp),[And sigmvcab.Emps = ']+lcEmp+[' ],[])

If Thisform.Podatamgr.SqlExecute(lcQuery,[crTmpMov])<1
	Messagebox([Favor Reiniciar o Processo!!!],0+16,[])
	Return .F.
Endif

Select crTmpMov

Select crTmpMov.* ;
	From crTmpMov, csOperacaos ;
	where csOperacaos.Operacaos = crTmpMov.Dopes ;
	And csOperacaos.Marcas = .T. ;
	Into Cursor crMov Readwrite

Select crMov
Index On Dopes Tag Dopes
Go Top

If Reccount()=0
	Messagebox([Nenhum registro foi selecionado.],0+64,[Aviso])
	Return .F.
Endif

Scan

	lnCustofs = 0
	lnPcuss   = 0

	lcQuery = [Select distinct a.Emps, a.dopes, a.numes, a.EmpDopnums, a.datas, sum(c.custofs) as custofs, ] + ;
		[Sum(c.pcuss) as pcuss, c.moecusfs, b.moedas, SUM(b.qtds) as qtds, ] + ;
		[d.GrupoEDs, d.ContaEDs, d.GrupoECs, d.ContaECs, d.Moedas as MoeOpFin ] + ;
		[From sigmvcab a ] + ;
		[join SigMvItn b on b.empdopnums = a.empdopnums ] + ;
		[join SigCdPro c on c.Cpros = b.Cpros ] + ;
		[join (select GrupoEDs, ContaEDs, GrupoECs, ContaECs, SigOpFin.Dopes, SigOpFin.Moedas ] + ;
		[from SigOpFin ] + ;
		[join SigOpCdc on SigOpCdc.dopes = SigOpFin.dopes ] + ;
		[And SigOpCdc.opeinatvs = 0 ] + ;
		[where contaeds <> space(11)) d on d.dopes = a.dopes ] + ;
		[Where c.pcuss >0 ] + ;
		[And a.empdopnums = ']+crMov.empdopnums+[' ] + ;
		[group by a.Emps, a.dopes, c.moecusfs, b.moedas, a.datas, ] + ;
		[d.GrupoEDs, d.ContaEDs, d.GrupoECs, d.ContaECs, d.Moedas, a.numes, a.EmpDopnums ]

	If Thisform.Podatamgr.SqlExecute(lcQuery,[crMovIt])<1
		Messagebox([Favor Reiniciar o Processo!!!],0+16,[])
		Return .F.
	Endif

	Select crMovIt
	Scan

		lnCustofs = crMovIt.custofs
		lnPcuss   = crMovIt.pcuss
		lcMoeCf	  = Iif(Empty(crMovIt.moecusfs),crMovIt.moedas,crMovIt.moecusfs)
		lcMoeda	  = Iif(Not Empty(lcMoeCV),lcMoeCV,Iif(Empty(MoeOpFin),lcMoeCf,crMovIt.MoeOpFin))

		lnCotacao = fCarregarCambio(lcMoeCf,crMovIt.DataS)
		lnCotaMoe = fCarregarCambio(lcMoeda,crMovIt.DataS)

		lnCustofs = Round(lnCustofs * lnCotacao / lnCotaMoe,2)
		lnPcuss   = Round(lnPcuss 	* lnCotacao / lnCotaMoe,2)

		Select csMovimento
		Append Blank

		Replace Emps 	With crMovIt.Emps  ,;
			Dopes 	 	With crMovIt.Dopes ,;
			Numes	 	With crMovIt.Numes ,;
			EmpDopNums	With crMovIt.EmpDopNums ,;
			Datas 	 	With crMovIt.DataS ,;
			moecusfs 	With crMovIt.moecusfs ,;
			moedas 	 	With crMovIt.moedas ,;
			GrupoEDs 	With crMovIt.GrupoEDs ,;
			ContaEDs 	With crMovIt.ContaEDs ,;
