LParameters lcAcesso

Private pPro, pKey

pPro = Replicate(Chr(254), 20)
pKey = Replicate(Chr(254), 20)

lcQryCopOp = 'Select * From SigPcOop'
lcQryCopOI = 'Select * From SigPcOoI Where Processos = ?pPro'
lcQryCopOG = 'Select * From SigPcOoG Where Processos = ?pPro And fkChaves = ?pKey'

With ThisForm
	.poDataMgr = CreateObject('fSqlConector', ThisForm.Name)
	
	If (.poDataMgr.pnIdconn > 0)
		With .poDataMgr
			* Cursor Principal Para Montagem da Lista (Tabela Grande)
			.AddCursor('SigPcOop', 'Processos', 'crSigPcOop', '', ThisForm.Pagina.Lista.Grade, lcQryCopOp)
			.pfSqlTabela(1).pColuna('Processos', '', '', 'Processos', 150, .t.)
			.pfSqlTabela(1).pColuna('Emps',      '', '', 'Emp',       031, .f.)
