------------------------------------------------------------
* PROCEDURE msv_procurar
------------------------------------------------------------
PROCEDURE msv_procurar
SELECT crSigCdPai
If not Empty(ThisForm.Pagina.Dados.get_Codigos.Value) 
	LOCATE FOR Codigos = ThisForm.Pagina.Dados.get_Codigos.Value
Else
	LOCATE FOR Descs = ThisForm.Pagina.Dados.get_Descs.Value 
Endif	
* Fim **************************************************************************************

ENDPROC

------------------------------------------------------------
* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
Lparameters lcAcesso

With ThisForm
	.poDatamgr = CreateObject('fSqlConector', 'SigCdPai')
	With .poDataMgr
		lcQry = [Select * from SigCdPai]
		.AddCursor('SigCdPai', 'cIdChaves', 'crSigCdPai', '', ThisForm.Pagina.Lista.Grade,lcQry)
		.pfSqlTabela(1).pColuna('Codigos', '', '', 'Código',    050, .t.)
		.pfSqlTabela(1).pColuna('Descs',    '', '', 'País', 290, .t.)
	EndWith	
EndWith

=DoDefault()

ThisForm.Pagina.Lista.Grade.Column1.SetFocus
* Fim **************************************************************************************

ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Lista.Grupo_op.Click
------------------------------------------------------------
PROCEDURE Pagina.Lista.Grupo_op.Click
Lparameters Opcao

If Not DoDefault(Opcao)
	ThisForm.mAtivapagina1
	Return 0
EndIf

Select crSigCdPai
Do Case
	Case Inlist(ThisForm.pcEscolha, 'CONSULTAR', 'EXCLUIR')
		ThisForm.Pagina.Dados.Grupo_Salva.Salva.Setfocus
	Otherwise 
		This.Parent.Parent.Dados.get_Codigos.Setfocus
EndCase 

Select crSigCdPai

ThisForm.Pagina.Dados.Activate
* Fim **************************************************************************************
ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
Local lnRecno1,lnNreg1,lcGrupo,lcDope,laCampos(2,2)

If InList(ThisForm.pcEscolha,'INSERIR')
	If Empty(ThisForm.Pagina.Dados.get_Codigos.Value)
		=MessageBox('Obrigatório Informar o Código do País!!!', 48, '')
		This.Parent.Parent.Get_Codigos.SetFocus
		Return 0
	EndIf
	
	If Empty(This.Parent.Parent.Get_Descs.Value)
		=MessageBox('Obrigatório Informar o Nome do País!!!', 48, '')
		This.Parent.Parent.Get_Descs.SetFocus
		Return 0
	EndIf
	
	If ThisForm.poDataMgr.ChkRegister('SigCdPai','Codigos',ThisForm.Pagina.Dados.Get_Codigos.Value )
		=MessageBox('País já cadastrado!!!', 48, '')
		This.Parent.Parent.Get_Codigos.SetFocus
		Return 0
	EndIf 

	Select crSigCdPai
	If Empty(crSigCdPai.cIdChaves)
		Replace cIdChaves With fUniqueIds() In crSigCdPai
	EndIf
EndIf

=DoDefault()

ThisForm.mAtivapagina1
* Fim **************************************************************************************

ENDPROC

ENDPROC
