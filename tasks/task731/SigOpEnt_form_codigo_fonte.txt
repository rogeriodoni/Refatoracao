PROCEDURE Init
Lparameters pEmp, pDop, pNum, pPrz, pFor, pMin, pDat, pBlqAlt, pMax

ThisForm.cntSombra.lblSombra.Caption = ThisForm.Caption
ThisForm.cntSombra.lblTitulo.Caption = ThisForm.Caption

lcQrySigOpMvv = [Select * ] + ;
				 [From SigOpMvv ] + ;
				[Where EmpDopNcs = '] + pEmp + pDop + Str(pNum, 6) + [' ] + ;
				[Order By Vencs]

With ThisForm
	.poDataMgr = CreateObject([fSqlConector], .Name)

	If (.poDataMgr.pnIdconn > 0)
		With .poDataMgr
			.AddCursor('SigOpMvv', 'cIdChaves', 'crSigOpMvv', '', '', lcQrySigOpMvv)
		EndWith

		.pEmp = pEmp
		.pDop = pDop
		.pNum = pNum
		.pPrz = pPrz
		.pMin = pMin
		.pMax = pMax
		.pDat = pDat
		.Bloquear = pBlqAlt

		.ParentForm = pFor
		
		.getAtual.Value = .pPrz		
		.getNova.Value  = Ctod('')

		.poDataMgr.Requery('crSigOpMvv')
		.GradeVenc.RecordSource = 'crSigOpMvv'
		.GradeVenc.Column1.ControlSource = 'ttod(crSigOpMvv.Vencs)'
		.GradeVenc.Column2.ControlSource = 'crSigOpMvv.Usuars'
		.GradeVenc.Refresh

		.poDataMgr.CursorQuery([SigCdOpe], [crSigCdOpe], [Dopes], .pDop, [DtEntrs])
		If Not Used([crSigCdOpe])
			=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (CursorQuery - crSigCdOpe)])
			Return .f.
		EndIf

		.poDataMgr.CursorQuery([SigOpCdd], [crSigOpCdd], [Dopes], .pDop, [chkUteis])
		If Not Used([crSigOpCdd])
			=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (CursorQuery - crSigOpCdd)])
			Return .f.
		EndIf

		=DoDefault()

		If ThisForm.Bloquear
			.GetNova.Enabled = .f.
			.altera.Enabled = .f.
			.Gradevenc.SetFocus
		Else
			.getNova.SetFocus
		Endif
	Else
		=Messagebox('Impossível Efetuar Conexão Com o Servidor de Banco de Dados...', 0+48, 'CreateObject')
		Return .f.
	EndIf
EndWith
* Fim **************************************************************************************
ENDPROC

PROCEDURE Valid
Local ldEnt

ldEnt  = ThisForm.pDat + ThisForm.pMin
ldEntm = ThisForm.pDat + ThisForm.pMax

If Not IsEmpty(This.Value)
	ldDat = Ttod(fDtoSQL(This.Value))
	This.Value = ldDat
	This.Refresh
EndIf

If InList(crSigCdOpe.DtEntrs, 1, 4, 6)
	If IsEmpty(This.Value)
		=Messagebox([Prazo de Entrega Deve Ser Informado...], 32, [])
		Return .f.
	EndIf
	
	If (Ttod(fDtoSQL(This.Value)) < ThisForm.pDat)
		=Messagebox([Prazo de Entrega Deve Ser Maior Que a Data da Movimentação...], 32, [])
		Return .f.
	EndIf

	If (ThisForm.pMin > 0)
		ldEnt = (Ttod(fDtoSQL(ThisForm.pDat)) + ThisForm.pMin)
		If (Ttod(fDtoSQL(This.Value)) < ldEnt)
			Do Form SigOpSen With 'LIBPRZENT', 'Libera Prazo de Entrega' To _Retorno
			llOk = ( Left( _Retorno, 1 ) = "*" )
			lcUsuLib = Substr( _Retorno, 2 )
			If !llOk
				Messagebox('Prazo de Entrega Mínimo é ' + Dtoc(ldEnt) + '...', 32, '' )
				Return .f.
			Else
				lcTransacao = 'Emp: '+_Empr + ' Nº Oper: ' + Str(thisform.pnum,6)+ ' - ' + 'Prazo Mínimo: ' + Dtoc(ldEnt) +  ;
								' Prazo Liberado ' + Dtoc(This.Value)				
				=fGravarLog('T', thisform.pdop, 'LIBPRZENT', lcTransacao, lcUsuLib )
			EndIf							
		EndIf
	EndIf

	If (ThisForm.pMax > 0)
		ldEnt = (Ttod(fDtoSQL(ThisForm.pDat)) + ThisForm.pMax)
		If (Ttod(fDtoSQL(This.Value)) > ldEnt)
			*** Tiago - 24/06/2014 - Mondoro - Checa liberação para alterar prazo de entrega fora do mínimo e máximo permitido
			Do Form SigOpSen With 'LIBPRZENT', 'Libera Prazo de Entrega' To _Retorno
			llOk = ( Left( _Retorno, 1 ) = "*" )
			lcUsuLib = Substr( _Retorno, 2 )
			If !llOk
				Messagebox([Prazo de Entrega Máximo é ] + Dtoc(ldEnt) + [...], 32, [])
				Return .f.
			Else
				lcTransacao = 'Emp: '+_Empr + ' Nº Oper: ' + Str(thisform.pnum,6)+ ' - ' + 'Prazo Máximo: ' + Dtoc(ldEnt) +  ;
								' Prazo Liberado ' + Dtoc(This.Value)				
				=fGravarLog('T', thisform.pdop, 'LIBPRZENT', lcTransacao, lcUsuLib )
			EndIf			
		EndIf
	EndIf
EndIf
* Fim **************************************************************************************
ENDPROC

PROCEDURE Valid
Private pDtp
Local ldAnt, ldNew, llOks

ldAnt = ThisForm.getAtual.Value
ldNew = ThisForm.getNova.Value

If Not Empty(ldNew)
	Insert Into crSigOpMvv (Emps, Dopcs, NoTitulos, Vencs, EmpDopNcs, cIdChaves, Usuars, Datas) ;
				   Values (ThisForm.pEmp, ThisForm.pDop, ThisForm.pNum, fDtoSQL(ldAnt), ;
				   		   ThisForm.pEmp + ThisForm.pDop + Str(ThisForm.pNum, 6), ;
				   		   fUniqueIds(), Usuar, Datetime())
	
	llOks = .t.

	pDtp = fDtoSQL(ldNew)
	
	If llOks And (ThisForm.ParentForm.pcEscolha <> 'ALTERAR')
		lcQuery = [Update SigMvCab ] + ;
					 [Set PrazoEnts = ?pDtp ] + ;
				   [Where EmpDopNums = '] + ThisForm.pEmp + ThisForm.pDop + Str(ThisForm.pNum, 6) + [']

		If (ThisForm.poDataMgr.SqlExecute(lcQuery, '') < 1)
			=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (Update - )')
			llOks = .f.
		EndIf
	EndIf

	If llOks
		If Not ThisForm.poDataMgr.Update('crSigOpMvv')
			=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (Update - crSigOpMvv)')
			llOks = .f.
		EndIf
	EndIf

	If llOks
		If (ThisForm.poDataMgr.Commit() < 1)
			=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (Commit)')
			llOks = .f.
		EndIf
	EndIf

	If Not llOks
		ThisForm.poDataMgr.RollBack()
	Else
		ThisForm.pPrz = ldNew
		thisform.parentform.validOk = .t.

		ThisForm.ParentForm.cntEntregas.getEntregas.Value = ThisForm.pPrz
		ThisForm.getAtual.Value = ThisForm.pPrz
		ThisForm.getNova.Value  = Ctod('')
	EndIf

	ThisForm.getAtual.Refresh
	ThisForm.getNova.Refresh
	ThisForm.GradeVenc.Refresh
EndIf
* Fim **************************************************************************************
ENDPROC
