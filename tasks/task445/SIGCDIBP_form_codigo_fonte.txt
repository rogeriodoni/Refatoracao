PROCEDURE importaplan
PRIVATE ldDatIni, ldDatFin
Local alUF, lnQtdArq, laArqs, lcArq, lcDir

SELECT crSigcdIbp
ZAP

DIMENSION laArqs[1,5]

Create Cursor localIbpt (c1 c(10), c2 c(50), c3 c(50), c4 c(100), c5 c(50), c6 c(50), c7 c(50), c8 c(50), c9 c(50), c10 c(50), c11 c(50), c12 c(50), c13 c(50))

lcDir = GetDir([], [Selecione a Pasta Ou Caminho de Rede], [Seleção de Diretórios], 2)
lnQtdArq = ADIR(laArqs, lcDir+'*.csv')  && Create array

FOR i=1 to lnQtdArq
	lcArq = lcDir+laArqs[i,1]
	If Empty(lcArq)
		Return .f.
	EndIf

	SELECT localIbpt
	ZAP
	APPEND FROM (lcArq) DELIMITED  WITH  character ';'
	GO top
	DELETE 
	lcUf = SUBSTR(ALLTRIM(laArqs[i,1]),13,2)
	SELECT localIbpt
	DELETE all FOR !EMPTY(c2)
	GO top
	Scan
		lcCodigo   = ALLTRIM(c1)
		lcDesc   = ALLTRIM(c4)
		lnAliq5  = VAL(ALLTRIM(c5))
		lnAliq6  = VAL(ALLTRIM(c6))
		lnAliq7  = VAL(ALLTRIM(c7))
		lnAliq8  = VAL(ALLTRIM(c8))
		ldDatIni = DTOT(CTOD(ALLTRIM(c9)))
		ldDatFin = DTOT(CTOD(ALLTRIM(c10)))
		lcChave = ALLTRIM(c11)
		lcVersao= ALLTRIM(c12)
		lcFonte = ALLTRIM(c13)

		lcQuery = 	[Select cidchaves From SigcdIbp where ufs = ']+lcUf+[' and codigo =']+lcCodigo+[' and chave = ']+lcChave+[' and versao =']+lcVersao+[']
		If ThisForm.Podatamgr.Sqlexecute(lcQuery,'crAux') < 1
			Messagebox('Impossivel Efetuar Conexão com Servidor de Banco de Dados.',(0+48),'Erro')
			Return
		Endif
		IF RECCOUNT('crAux') = 0
			lcCidChaves = fUniqueIds()
		ELSE
			lcCidChaves = crAux.cidchaves
		ENDIF
					
		Insert Into crSigcdIbp (cidchaves, ufs, codigo, ex, tipo, descricao, datini, datfin,nacfederal, impfederal, estadual, municipal, chave, versao, fonte);
			             Value (lcCidChaves, lcUf, lcCodigo, '', 0, lcDesc, ldDatIni, ldDatFin,lnAliq5, lnAliq6,lnAliq7,lnAliq8,lcChave, lcVersao, lcFonte)
		Select LocalIbpt
	Endscan
ENDFOR

&&datini, datfin, 
&&ldDatIni, ldDatFin, 


*!*		[cidchaves] [char](20) NOT NULL,
*!*		[ufs] [char](2) NOT NULL,
*!*		[codigo] [char](10) NOT NULL,
*!*		[ex] [char](3) NOT NULL,
*!*		[tipo] [numeric](3,0) NOT NULL,
*!*		[descricao] [text] NULL,
*!*		[datini] [datetime] NULL,
*!*		[datfin] [datetime] NULL,
*!*		[nacfederal] [numeric](5,2) not null,
*!*		[impfederal] [numeric](5,2) not null,
*!*		[estadual] [numeric](5,2) not null,
*!*		[municipal] [numeric](5,2) not null,
*!*		[chave] [char](10) NOT NULL,
*!*		[versao] [char](10) NOT NULL,
*!*		[fonte] [char](10) NOT NULL)





llOks = .T.

If Not ThisForm.poDataMgr.Update([crSigCdIbp])
	=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (Update - crSigCdIbp)])
	llOks = .f.
EndIf
If llOks
	If (ThisForm.poDataMgr.Commit() < 1)
		=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (Commit)])
		llOks = .f.
	EndIf
EndIf

ThisForm.Refresh 
RETURN llOks

ENDPROC
