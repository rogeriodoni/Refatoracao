PROCEDURE mnovalinha

Local lnI, lcColuna, lnPgDown

If Thisform.lNovaLinha
	Thisform.lNovaLinha = .F.
	lnPgDown = Thisform.nPgDown
	Thisform.nPgDown = 1
	Select crTpmMvIts
	With Thisform.fwgrade
		.Column3.SetFocus
	EndWith
	Skip -1
	Skip
	Keyboard "{DNARROW}"
	Keyboard "{PGDN}"
EndIf
ENDPROC

------------------------------------------------------------
* PROCEDURE mvalidacolunas
------------------------------------------------------------
PROCEDURE mvalidacolunas
Lparameters poColumn

Local lnQtds, lnRetorno, lnColunaAtiva, lnI, lnNoCols, lcColuna, lcDescs, lnRecNo, loLista

Select CrTmpGru

Thisform.cFoco = ''
Thisform.nPgDown = 1
lnRetorno     = .T.

With Thisform.fwgrade
	lnColunaAtiva = .ActiveColumn
	lnNoCols      = .ColumnCount
Endwith

Do Case
Case poColumn.ControlSource = 'crTpmMvIts.CodTams'
	If !Inlist( Lastkey(), 5, 24 ) And !IsEmpty( poColumn.Value )
		loLista = Createobject('fwBuscaExt', Thisform.Parentform.Parentform.poDataMgr.pnIdConn, 'SigCdTam', ;
			'crListaRemota', 'Cods', Padr(poColumn.Value,4), 'Seleção', .T.)
		If Not loLista.plAchouRegistro
			loLista.mAddColuna('Cods', '', 'Código',.T.,'Cods')
			loLista.mAddColuna('Descs', '', 'Descrição',.T.,'Descs')
			loLista.Show()

			If Lastkey() == 27
				poColumn.Value = Iif( CrSigOpCdc.AltEtiqs = 1, Thisform.OldValue, '' )
				lcDescs = ''
			Else
				poColumn.Value = CrListaRemota.Cods
				lcDescs = CrListaRemota.Descs
			Endif
		Else
			lcDescs = CrListaRemota.Descs
		Endif
		If Type('poColumn.Value') <> Type('ThisForm.OldValue') Or poColumn.Value <> Thisform.OldValue
			Replace aQtds With 0 In crTpmMvIts

			lcQuery = [Select PesoMs, Percs, Valor from SigPrTam where Cpros = '] + crTpmMvIts.Cpros + [' And CodTams = '] + CrListaRemota.Cods + [']
			If Thisform.Parentform.Parentform.poDataMgr.SqlExecute(lcQuery,[LocalPrTam]) < 1
				Return .F.
			Endif
			If !Empty(LocalPrTam.PesoMs)
				Replace Pesos With LocalPrTam.PesoMs In crTpmMvIts
			Endif

			If crSigCdOpe.Precops = 1
				If !Empty(LocalPrTam.Valor)
					Replace Units With LocalPrTam.Valor In crTpmMvItn
				Endif
			Endif
		Endif

		Select csTmpDesc
		Seek Str(Recno('crTpmMvIts'),8)
		If Not Eof()
			Replace csTmpDesc.cDescTams With lcDescs
		Endif
	Else
		If IsEmpty( poColumn.Value ) And ( CrSigOpCdc.AltEtiqs = 1 )
			poColumn.Value = Thisform.OldValue
			poColumn.Refresh
		Endif
	Endif

Case poColumn.ControlSource = 'crTpmMvIts.CodCors'
	If !Inlist( Lastkey(), 5, 24 ) And !IsEmpty( poColumn.Value )
		loLista = Createobject('fwBuscaExt', Thisform.Parentform.Parentform.poDataMgr.pnIdConn, 'SigCdCor', ;
			'crListaRemota', 'Cods', poColumn.Value, 'Seleção', .T.)
		If Not loLista.plAchouRegistro
			loLista.mAddColuna('Cods', '', 'Código',.T.,'Cods')
			loLista.mAddColuna('Descs', '', 'Descrição',.T.,'Descs')
			loLista.Show()

			If Lastkey() == 27
				poColumn.Value = Iif( CrSigOpCdc.AltEtiqs = 1, Thisform.OldValue, '' )
				lcDescs = ''
			Else
				poColumn.Value = CrListaRemota.Cods
				lcDescs = CrListaRemota.Descs
			Endif
		Else
			lcDescs = CrListaRemota.Descs
		Endif

		*** Tiago - 06/06/2011 - Restrição de operação - Checa se a operação tem alguma cor restringida - Carat
		lcQuery = [Select Cgrus, Opers, Mercs, Contas, CodCors, ContaOrig From SigCdRpo Where Dopps = ']+TprMvCab.Dopes+[']
		If (Thisform.Parentform.Parentform.poDataMgr.SqlExecute(lcQuery, 'LocalRopt') < 1)
			=Messagebox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (LocalRopt)')
			Return 0
		Endif

		Select LocalRopt
		If Reccount('LocalRopt') > 0
			Select Cgrus, Mercs, CodCors From LocalRopt Where Contas = ' ' And ContaOrig = ' ' Into Cursor Selecao
			Select Selecao
			If Reccount('Selecao') > 0
				Go Top
				Locate For CodCors = poColumn.Value
				If Eof()
					=Messagebox('Produto Bloqueado para esta Movimentação!', 16, 'Restrição')
					poColumn.Value = []
					lnRetorno = .F.
				Endif
			Else
				Select Cgrus, Mercs, CodCors From LocalRopt Where Contas = TprMvCab.ContaDs Into Cursor Selecao2
				Select Selecao2
				If Reccount('Selecao2') > 0
					Go Top
					Locate For CodCors = poColumn.Value
					If Not Eof()
						=Messagebox('Produto Bloqueado para Conta Destino desta Movimentação!', 16, 'Restrição')
						poColumn.Value = []
						lnRetorno = .F.
					Endif
				Endif

				Select Cgrus, Mercs, CodCors From LocalRopt Where ContaOrig = TprMvCab.ContaOs Into Cursor Selecao3
				Select Selecao3
				If Reccount('Selecao3') > 0
					Go Top
					Locate For CodCors = poColumn.Value
					If Not Eof()
						=Messagebox('Produto Bloqueado para Conta Origem desta Movimentação!', 16, 'Restrição')
						poColumn.Value = []
						lnRetorno = .F.
					Endif
				Endif

			Endif
		Else
			lcQuery = [Select distinct dopps From SigCdRpo Where Exclus = 1 And (Cgrus = ']+CrTmpGru.Cgrus+[' Or ]+;
				[Mercs = ']+CrTmpGru.Mercs+[' Or CodCors = '] + crTpmMvIts.CodCors + [')]
			If (Thisform.Parentform.Parentform.poDataMgr.SqlExecute(lcQuery, 'LocalRoptG') < 1)
				=Messagebox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (LocalRoptG)')
				Return .F.
			Endif
			Select LocalRoptG
			If Reccount('LocalRoptG') > 0
				=Messagebox('Produto Bloqueado para esta Movimentação!!!', 16, 'Restrição de Exclusivo')
				poColumn.Value = []
				lnRetorno = .F.
			Endif
		Endif
		***

		If Type('poColumn.Value') <> Type('ThisForm.OldValue') Or poColumn.Value <> Thisform.OldValue
			Replace aQtds With 0 In crTpmMvIts
		Endif
		Select csTmpDesc
		Seek Str(Recno('crTpmMvIts'),8)
		If Not Eof()
			Replace csTmpDesc.cDescCors With lcDescs
		Endif
	Else
		If IsEmpty( poColumn.Value ) And ( CrSigOpCdc.AltEtiqs = 1 )
			poColumn.Value = Thisform.OldValue
			poColumn.Refresh
		Endif
	Endif

Case poColumn.ControlSource = 'crTpmMvIts.CodEmbs'
	If !Inlist( Lastkey(), 5, 24 ) And !IsEmpty( poColumn.Value )
		loLista = Createobject('fwBuscaExt', Thisform.Parentform.Parentform.poDataMgr.pnIdConn, 'SigCdEmb', ;
			'crListaRemota', 'Cods', poColumn.Value, 'Seleção', .T.)
		If Not loLista.plAchouRegistro
			loLista.mAddColuna('Cods', '', 'Código',.T.,'Cods')
			loLista.mAddColuna('Descs', '', 'Descrição',.T.,'Descs')
			loLista.Show()

			If Lastkey() == 27
				poColumn.Value = Iif( CrSigOpCdc.AltEtiqs = 1, Thisform.OldValue, '' )
				lcDescs = ''
			Else
				poColumn.Value = CrListaRemota.Cods
				lcDescs = CrListaRemota.Descs
			Endif
		Else
			lcDescs = CrListaRemota.Descs
		Endif

		Select csTmpDesc
		Seek Str(Recno('crTpmMvIts'),8)
		If Not Eof()
			Replace csTmpDesc.cDescEmbs With lcDescs
		Endif

		Select crTpmMvIts
		If !Eof() And !Bof()
			Replace QtdEmbs With CrListaRemota.MultiS, PrEmbs With CrListaRemota.Precos
			_RecEmb   = Recno()
			_PrecoEmb = CrListaRemota.Precos
			Scan
				If PrEmbs # _PrecoEmb And !IsEmpty( CodEmbs )
					=Messagebox( "Todas as embalagens devem ter o mesmo preço!", 48, '' )
					Select crTpmMvIts
					Go _RecEmb
					Replace QtdEmbs With 0, PrEmbs With 0
					_PrecoEmb = 0
					poColumn.Value = ''
					Exit
				Endif
			Endscan
			Select crTpmMvIts
			Go _RecEmb
			If _PrecoEmb # 0
				Select crTpmMvItn
				Replace unitembs With _PrecoEmb
				Select crTpmMvIts
			Endif
			If IsEmpty( crTpmMvIts.CodEmbEnts ) And !IsEmpty( poColumn.Value )
				Select crTpmMvIts
				Replace crTpmMvIts.CodEmbEnts With poColumn.Value
			Endif
		Endif
	Endif

Case poColumn.ControlSource = 'crTpmMvIts.Qtds'
	lnQtds = poColumn.Value
	If Lastkey() = 27
		poColumn.Value = 0
	Else
		If poColumn.Value < 0
			Messagebox( "Quantidade Inválida...", 48, 'Quantidade Negativa' )
			lnRetorno = 0
		Else
			If (crTpmMvItn.SubNObrigs = 1)	&& Sub-Nível Obrigatório
				Select crTpmMvIts
				If lnQtds > crTpmMvIts.QtBaixas
					Messagebox( "Quantidade não pode ser maior que quantidade a ser baixada...", 48, 'Erro na Quantidade' )
					poColumn.Value = Thisform.OldValue
					lnRetorno = 0
				Endif
			Else
				If Thisform.lSubNivel And crSigCdOpe.ChkQtdss = 1
					If Inlist( crSigCdOpe.ItAutos,1,3,4,5 )
						If lnQtds > crTpmMvIts.QtBaixas
							Messagebox('Impossível Baixar Uma Quantidade (' + ;
								AllTrim(Str(lnQtds,9,2)) + ')' + Chr(13) + ;
								'Maior Que a Pendente no Sub-Nível (' + ;
								AllTrim(Str(crTpmMvIts.QtBaixas,9,2)) + ').', 48, '')
							If Not IsEmpty( crTpmMvIts.QtBaixas )
								poColumn.Value = Thisform.OldValue
							Else
								poColumn.Value = 0
							Endif
							lnRetorno = 0
						Endif
					Else
						If Not Seek( crTpmMvIts.Cpros + crTpmMvIts.CodCors + crTpmMvIts.CodTams, 'xBaixa2', 'CPros' )
							Messagebox( 'Produto ' + Alltrim(crTpmMvIts.Cpros) + ' ' + ;
								AllTrim(Alltrim(crTpmMvIts.CodCors) + ' ' + ;
								AllTrim(crTpmMvIts.CodTams ) ) + ' Não Encontrado Para Baixa.', 48, '' )
							poColumn.Value = Iif( Not IsEmpty( crTpmMvIts.QtBaixas ), ;
								ThisForm.OldValue, 0 )
							poColumn.Refresh
							lnRetorno = 0
						Else
							If lnQtds > xBaixa2.Qtds - xBaixa2.QtBaixas + Thisform.OldValue
								Messagebox('Impossível Baixar Uma Quantidade (' + ;
									AllTrim(Str(lnQtds,9,2)) + ')' + Chr(13) + ;
									'Maior Que a Pendente no Sub-Nível (' + ;
									AllTrim(Str(crTpmMvIts.QtBaixas,9,2)) + ').', 48, '')
								poColumn.Value = Iif( Not IsEmpty( crTpmMvIts.QtBaixas ), ;
									ThisForm.OldValue, 0 )
								poColumn.Refresh
								lnRetorno = 0
							Else
								Replace xBaixa2.QtBaixas With xBaixa2.QtBaixas + lnQtds - Thisform.OldValue
							Endif
						Endif
					Endif
				Endif
			Endif

			If (lnQtds-Int(lnQtds))<>0
				lcQuery = [select b.blqqtdfrac, a.cunis from SigCdPro a join sigcduni b on a.cunis = b.cunis where cpros = ']+crTpmMvIts.Cpros+[' ]

						Messagebox('Impossível Informar Quantidade Fracionada para a Unidade (' + Alltrim(crQtdFrac.cunis) + ').', 48, '')
						poColumn.Value = 0
						lnRetorno = 0
					Endif
				Endif
			Endif

			If Type('lnRetorno') = 'L' And lnRetorno And Inlist( Thisform.op_escolha, 'INSERIR', 'ALTERAR' ) And poColumn.Value # 0
				Select crTpmMvIts
				If !fVerEst(Thisform.Parentform.Parentform.poDataMgr,(Thisform.OldValue = poColumn.Value))
					Replace crTpmMvItn.Qtds With 0
					poColumn.Value = 0
					poColumn.Refresh
					lnRetorno = 0
				Endif
			Endif

			If Type('lnRetorno') = 'L' And lnRetorno And !IsEmpty( crTpmMvIts.CodEmbEnts )
				If Thisform.Parentform.Parentform.poDataMgr.Cursorquery('SigCdEmb','CrTmpEmb','Cods',crTpmMvIts.CodEmbEnts)
					If !IsEmpty(CrTmpEmb.MultiS)
						Select crTpmMvIts
						Replace crTpmMvIts.QtdEnts With crTpmMvIts.QtdEmbs * crTpmMvIts.Qtds / CrTmpEmb.MultiS
					Endif
				Endif
			Endif

		Endif
	Endif

Case poColumn.ControlSource = 'crTpmMvIts.CodEmbEnts'
	If !Inlist( Lastkey(), 5, 24 ) And !IsEmpty( poColumn.Value )
		loLista = Createobject('fwBuscaExt', Thisform.Parentform.Parentform.poDataMgr.pnIdConn, 'SigCdEmb', ;
			'crListaRemota', 'Cods', poColumn.Value, 'Seleção', .T.)
		If Not loLista.plAchouRegistro
			loLista.mAddColuna('Cods', '', 'Código',.T.,'Cods')
			loLista.mAddColuna('Descs', '', 'Descrição',.T.,'Descs')
			loLista.Show()
			poColumn.Value = Iif( Lastkey() == 27, '', CrListaRemota.Cods )
		Endif
	Endif

	If !IsEmpty( crTpmMvIts.CodEmbEnts )
		If Thisform.Parentform.Parentform.poDataMgr.Cursorquery('SigCdEmb','CrTmpEmb','Cods',crTpmMvIts.CodEmbEnts)
			If !IsEmpty(CrTmpEmb.MultiS)
				Select crTpmMvIts
				Replace crTpmMvIts.QtdEnts With crTpmMvIts.QtdEmbs * crTpmMvIts.Qtds / CrTmpEmb.MultiS
			Endif
		Endif
	Endif

Case poColumn.ControlSource = 'crTpmMvIts.QtdEnts'
	If Inlist( Thisform.op_escolha, 'INSERIR', 'ALTERAR' ) And !IsEmpty( poColumn.Value )
		Select crTpmMvIts
		Skip
		If Eof() And (crSigCdOpe.SubNObrigs # 1)
			Select CrTmpGru
			Insert Into crTpmMvIts( emps, Dopes, numes, Cpros, citens, QtdEmbs, Pesos, CodCors, CodTams, CodEmbs ) ;
				Value ( TprMvCab.emps, TprMvCab.Dopes, TprMvCab.numes, crTpmMvItn.Cpros, ;
				crTpmMvItn.citens, Iif( CrTmpGru.Embs = 1, 0, 1 ), ;
				Iif( CrTmpGru.Pesos = 1, crTpmMvItn.Pesos, 0 ) ,;
				Iif( CrTmpGru.Cores=1,CrTmpPro.CodCors,''), ;
				Iif( CrTmpGru.Tams=1,CrTmpPro.CodTams,' ') ,;
				Iif( CrTmpGru.Embs = 1, CrTmpPro.CuniFors,' '))
			Insert Into csTmpDesc ( nControle, nLinhas ) Values ( Recno('crTpmMvIts'), ;
				Thisform.nCItem2 )
			Select crTpmMvIts
		Endif
		Keyboard "{DNARROW}"
	Endif

Case poColumn.ControlSource = 'crTpmMvIts.Pesos'
	If Inlist( Thisform.op_escolha, 'INSERIR', 'ALTERAR' ) And IsEmpty( poColumn.Value )
		poColumn.Value = crTpmMvItn.Pesos
		poColumn.Refresh
	Endif

Otherwise
	poColumn.Refresh

Endcase

Select crTpmMvIts
For lnI = 1 To lnNoCols
	lcColuna = Alltrim(Str(lnI))
	If Not Thisform.SLeitura&lcColuna.
		Thisform.fwgrade.Column&lcColuna..Refresh
	Endif
Endfor

If Inlist( Thisform.op_escolha, 'INSERIR', 'ALTERAR' ) And Type('lnRetorno') = 'L' And ;
		lnRetorno And lnNoCols = lnColunaAtiva And Type('poColumn.Value') = 'N' And ;
		poColumn.Value > 0 And Type('lnQtds') = 'N'

	Select crTpmMvIts
	lnRecNo = Iif( Not Eof() And Not Bof(), Recno(), -1 )
	If lnRecNo > 0
		If ( lnQtds > 1 ) And ( Mod(lnQtds,1) = 0 ) And ;
				( CrTmpUni.Desmembs = 'S' ) And ( crSigCdOpe.Desmembs = 1 )
			lnQtds = lnQtds - 1
			Thisform.nPgDown = Int(lnQtds / 8 ) + 1
			Select csTmpDesc
			Scatter Memvar
			Select crTpmMvIts
			Replace crTpmMvIts.Qtds With 1
			This.Refresh
			Select crTpmMvIts
			Scatter Memvar
			m.Qtds = 1
			Do While lnQtds > 0
				Insert Into crTpmMvIts From Memvar
				Insert Into csTmpDesc ( nControle, nLinhas, cDescTams, cDescCors, cDescEmbs ) ;
					Values ( Recno('crTpmMvIts'), Thisform.nCItem2, m.cDescTams, m.cDescCors, ;
					m.cDescEmbs )
				Thisform.nCItem2 = Thisform.nCItem2 + 1
				lnQtds = lnQtds - 1
			Enddo
			Select crTpmMvIts
		Endif
	Endif

	Thisform.nCItem2 = 1
	Select crTpmMvIts
	Set Order To
	Set Order To citens
	Set Key To crTpmMvItn.citens
	Scan
		lcWhere = Thisform.cWhere
		If &lcWhere. Or IsEmpty(crTpmMvIts.Qtds)
			Delete
		Else
			Select csTmpDesc
			Replace nLinhas With Thisform.nCItem2
			Thisform.nCItem2 = Thisform.nCItem2 + 1
		Endif
	Endscan

	Insert Into crTpmMvIts ( citens, Cpros, emps, Dopes, numes, QtdEmbs, Pesos, CodCors, CodTams, CodEmbs ) ;
		Values ( crTpmMvItn.citens, crTpmMvItn.Cpros, TprMvCab.emps, TprMvCab.Dopes, TprMvCab.numes, ;
		Iif( CrTmpGru.Embs = 1, 0, 1 ),;
		Iif( CrTmpGru.Pesos = 1, crTpmMvItn.Pesos, 0 ),;
		Iif( CrTmpGru.Cores=1,CrTmpPro.CodCors,''), ;
		Iif( CrTmpGru.Tams=1,CrTmpPro.CodTams,' '), ;
		Iif( CrTmpGru.Embs = 1, CrTmpPro.CuniFors,' '))

	Insert Into csTmpDesc ( nControle, nLinhas ) Values ( Recno('crTpmMvIts'), Thisform.nCItem2 )
	Thisform.nCItem2 = Thisform.nCItem2 + 1
	Thisform.fwgrade.Refresh
	Thisform.lNovaLinha = .T.
Endif

Return( lnRetorno )

ENDPROC

------------------------------------------------------------
* PROCEDURE Click
------------------------------------------------------------
PROCEDURE Click
Local _Qtds, _PesoMedio, _TotalItens, _TotalPeso, lcMsg, lcFoco, lcCaption, lnCasasPeso

If Inlist( ThisForm.Op_Escolha, 'INSERIR', 'ALTERAR' )
	lnCasasPeso = CrSigCdOpe.NCPesos
	_TotalItens = 0
	_TotalPeso  = 0
	_Qtds = 0
	lcMsg = ''
	lcFoco = ''
	lcCaption = ''
	Select crTpmMvIts
	Goto Top
	Do While Not Eof('crTpmMvIts') And IsEmpty(lcMsg)
		Do Case
			Case crTpmMvIts.Qtds = 0
				Delete
					
			Case CrTmpGru.Pesos = 1 And CrSigCdOpe.EPesos = 3 And Not IsEmpty(crTpmMvIts.Qtds) And ;
					IsEmpty(crTpmMvIts.Pesos)
				lcMsg  = 'Obrigatório Informar Peso do Item ' + AllTrim(Str(csTmpDesc.nLinhas)) + '.'
				lcFoco = 'fwgrade.Column' + Thisform.cColunaPeso

			OtherWise
				_TotalItens = _TotalItens + ( crTpmMvIts.Qtds * QtdEmbs )
				_TotalPeso  = _TotalPeso + (crTpmMvIts.Pesos * ( crTpmMvIts.Qtds * QtdEmbs ) )
				_Qtds       = _Qtds + ( crTpmMvIts.Qtds * QtdEmbs )
		EndCase
		
		If IsEmpty(lcMsg)
			Select crTpmMvIts
			Skip
		Endif
	EndDo
	
	If Not IsEmpty(lcMsg)
		MessageBox( lcMsg, 48, lcCaption )
		Thisform.&lcFoco..SetFocus
		Return
	Else
		If CrSigCdOpe.ZerQtds = 2 And _Qtds > 1
			MessageBox('Quantidade do Produto não pode ser Diferente de 1',0+48,'Qtde Sem Alteração ')
			Return
		Endif
		If CrSigOpCdd.ChkQtMinv = 1 And CrTmpPro.LtMinSv > 0 And Mod(_Qtds,CrTmpPro.LtMinSv) <> 0
			MessageBox('Quantidade não é multiplo do limite de venda : '+Alltrim(Str(CrTmpPro.LtMinSv)),0+48,'Atenção')
			Return 0
		EndIf

		Select crTpmMvIts
		Set Order To
		_PesoMedio  = Iif( _TotalItens = 0, crTpmMvItn.Pesos, Round( _TotalPeso / _TotalItens, lnCasasPeso ) )
		_Peso       = Iif( CrTmpGru.Pesos = 1, _PesoMedio, crTpmMvItn.Pesos ) 

		_CarregaItens = InList( CrSigCdOpe.ItAutos,1,3,4,5 )
		_ChecaQtde    = ( CrSigCdOpe.ChkQtdss = 4 )
		_ChecaOper    = ( Empty(CrSigOpCdc.ChkOpers) Or crTpmMvItn.Opers = CrSigOpCdc.ChkOpers )

		Select TmpOperacao
		_SubNiv = .f.
		Scan for !IsEmpty( Codigos )
			_SubNiv = .t.
		EndScan
		
		_Voltar = .f.

		If _SubNiv And _ChecaQtde And _ChecaOper And Not _CarregaItens And _Qtds > 0
			Select xBaixa
			If !Seek( crTpmMvItn.CPros )
				Messagebox( 'Produto '+crTpmMvItn.cpros+' não encontrado para baixa!!', 48,'' )
				_Voltar = .t.
			Else
				If _Qtds > xBaixa.Qtds - xBaixa.QtBaixas + crTpmMvItn.Qtds
					Messagebox( 'Impossível baixar uma quantidade maior do que a pendente no subnível!', 48,'' )
					_Voltar = .t.
				Else
					Replace xbaixa.qtbaixas with xbaixa.qtbaixas + _Qtds - crTpmMvItn.Qtds in xbaixa
				Endif
			Endif
		Endif
		Select crTpmMvIts
		Set Order to
		Set Order to CItens
		Set Key to crTpmMvItn.CItens
		If _Voltar
			Return
		Endif

		Select crTpmMvItn
		With Thisform.Parentform.fwgrade
			.Qtd.Text1.Value = _Qtds
			.Peso.Text1.Value = _Peso
			.Qtd.Refresh
			.Peso.Refresh
			.Refresh
		EndWith
		KeyBoard Chr( 27 )
		Wait Window '' 
	EndIf
Endif

If Used('csTmpDesc')
	Use In csTmpDesc
EndIf

With Thisform.ParentForm
	.Enabled    = .T.
	.lValidaQtd = .T.
EndWith

Thisform.Release
ENDPROC
