PROCEDURE Init
Lparameters pOldForm
ThisForm.ParentForm = pOldForm

ThisForm.cntSombra.lblSombra.Caption = ThisForm.Caption
ThisForm.cntSombra.lblTitulo.Caption = ThisForm.Caption

*** mena - 18/08/2016 - foi incluido o nome da operacao na grade, um campo para os juros e um botao para chamar inclusao automatica na carteira ***
lcWher = [Select b.codigos, a.* From SigCcChm a, sigcqlcm b Where Nums = ']+CrSigChe.Nums+[' and a.numlotes = b.numlotes Order By DtSis ]

ThisForm.ParentForm.podatamgr.SqlExecute(lcWher,'CrSigCcChm')

Select CrSigCcChm
Go Top

With ThisForm.GrdHistCart
	.ColumnCount = 7
	.ReadOnly = .t.
	.RecordSource = 'crSigCcChm'
	.ScrollBars = 2

	.Column1.ControlSource = 'crSigCcChm.Dtsis'
	.Column1.Width = 80
	.Column1.Header1.Caption = 'Data'

	.Column2.ControlSource = 'crSigCcChm.GruCarts'
	.Column2.Width = 80
	.Column2.Header1.Caption = 'Grupo Dest'
	
	.Column3.ControlSource = 'crSigCcChm.ConCarts'
	.Column3.Width = 80
	.Column3.Header1.Caption = 'Conta Dest'

	.Column4.ControlSource = 'crSigCcChm.Grupos'
	.Column4.Width = 80
	.Column4.Header1.Caption = 'Grupo Orig'

	.Column5.ControlSource = 'crSigCcChm.Contas'
	.Column5.Width = 80
	.Column5.Header1.Caption = 'Conta Orig'

	.Column6.ControlSource = 'crSigCcChm.NumLotes'
	.Column6.Width = 66
	.Column6.Header1.Caption = 'Lote'
	
	.Column7.ControlSource = 'crSigCcChm.codigos'
	.Column7.Width = 110
	.Column7.Header1.Caption = 'Operacao'
	
	For I=1 To .ColumnCount
		lnCol = Alltrim(Str(I))

		.Column&lnCol..ReadOnly = .t.

PROCEDURE Click
*** mena - 17/08/2016 - gera baixa automatica - HDG
SELECT crSigcdOpt
LOCATE for operacaos = crSigccchm.codigos
IF FOUND()
	IF crSigcdOpt.devolvidos = 'S'
		SELECT crSigcdOpt
		LOCATE for opants = crSigccchm.codigos
		IF FOUND()
			do form sigcdche with .F.,.F.,.F.,.T.,crSigcdOpt.operacaos,crSigccchm.codigos,CrSigChe.Nums, thisform.get_taxa.value 	&& gera carteira de baixa de cheque devolvido automaticamente
			thisform.Init(thisform.parentform)
		ENDIF
	ENDIF
