PROCEDURE processamento
Local lnMax, loCnx, lcFig, lcArq, lcFor, lcRef, lcEnt, lcVen, llBar, lcBar

loCnx = ThisForm.poDataMgr

Create Cursor crCabecalho (Dopes c(20), Numes n(6), Datas d(8), RazSocs c(50), Cgcs c(20), RespFretes c(16), ;
						   Obses m(4), Ies c(15), GrVends c(10), Vends c(10), DVends c(50), DddVends c(4), ;
						   Tel1sVends c(20), Tel2sVends c(20), eMailVends c(50), GrResps c(10), Resps c(10), ;
						   DResps c(40), ForIClis c(10), ForRClis c(50), ForEndes c(60), ForNums c(10), ;
						   ForCompls c(15), ForCeps c(9), ForCidas c(30), ForBairs c(40), ForEstas c(2), ;
						   ForContato c(20), ForDdds c(4), ForTel1s c(20), ForTel2s c(20), ForFaxs c(20), ;
						   ForeMails c(50), ForCgcs c(20), ForIes c(15), ForIMuns c(20), ForTpClis n(1), ;
						   EntIClis c(10), EntRClis c(50), EntEndes c(80), EntNums c(10), EntCompls c(15), ;
						   EntCeps c(9), EntCidas c(30), EntBairs c(40), EntEstas c(2), EntContato c(20), ;
						   EntDdds c(4), EntTel1s c(20), EntTel2s c(20), EntFaxs c(20), EnteMails c(50), ;
						   EntCgcs c(20), EntIes c(15), EntTpClis n(1), PrazoEnts d(8), Icms n(16,2), ;
						   TipoVars c(20), Variacao n(16,2), CondPagto m(4), TotPag n(4), Fretes c(12), ;
						   TitFornec c(15), DckPrzE n(1), DocPrzE c(15), DckReffs n(1), DocReffs c(15), ;
						   DckVend n(1), TitVend c(15), DckTVend n(1), Barras c(14), ImpCabBar l(1), ;
						   Imagem c(64), ImgTec c(64), DckFretes n(1), DocFretes c(12), DckIcmIpi n(1), ;
						   DocPag c(12), DckEntregs n(1), DocEntregs c(12), DckObsI n(1), Telefone c(41), ;
						   VlFretes n(16,2), BarVends c(14), TitTermos m(4), TxtTermos m(4), TitTermgs m(4), ;
						   TxtTermgs m(4), TxtTermBs m(4), Pagina n(4), ObsEntrega c(254))

Create Cursor crDados (Pagina n(4), CItens n(4), CPros c(14), Reffs c(40), DPros c(40), CUnis c(3), ;
					   Qtds n(9,3), Units n(15,6), Totas n(11,2), Aliqs n(4,2), ValIpis n(11,2), Obs c(40))

If ThisForm.SelecionaDados()
	* Número Máximo de Produtos Por Pedido
	If (LocalOpe.Docus = 3) && 3 - Documento Fixo
		lnMax = Iif(LocalDcOpe.DckObsI = 1, 7, 14)
	Else					&& 4 - Fixo c/ Imagem
		lnMax = 3
	EndIf
	
	lcEmail = ''
	If Not IsEmpty(LocalEest.ContaOs)
		lcSql = [Select Emails From SigCdCli where Iclis = ']+LocalEest.contaOs+[']
		=ThisForm.Podatamgr.Sqlexecute(lcSql,'CrTmpCli')
		If Not IsEmpty(CrTmpCli.Emails)
			lcEmail = ALLTRIM(CrTmpCli.Emails)+';'
		EndIf
	EndIf
	If Not IsEmpty(LocalEest.ContaDs)
		lcSql = [Select Emails From SigCdCli where Iclis = ']+LocalEest.contaDs+[']
		=ThisForm.Podatamgr.Sqlexecute(lcSql,'CrTmpCli')
		If Not IsEmpty(CrTmpCli.Emails)
			lcEmail = lcEmail + ALLTRIM(CrTmpCli.Emails)
		EndIf
	EndIf
	ThisForm.Contaemail = lcEmail

	* Título do Label do Fornecedor: "Dados do ..."
	lcFor = [Fornecedor]
	If InList(LocalOpe.Opers, 1, 3) And Not Empty(LocalDcOpe.DocOrig)
		lcFor = Alltrim(LocalDcOpe.DocOrig)
	EndIf
	If InList(LocalOpe.Opers, 2, 3) And Not Empty(LocalDcOpe.DocDest)
		lcFor = Alltrim(LocalDcOpe.DocDest)
	EndIf

	* Título do Label da Data de Entrega
	lcEnt = [Data de Entrega]
	If Not Empty(LocalDcOpe.DocPrzE)
		lcEnt = Alltrim(LocalDcOpe.DocPrzE)
	EndIf

	* Título da Coluna de Referência do Fornecedor
	lcRef = [Ref. Fornecedor]
	If Not Empty(LocalDcOpe.DocReffs)
		lcRef = Alltrim(LocalDcOpe.DocReffs)
	EndIf

	* Título do Label do Comprador
	lcVen = [Comprador]
	If Not Empty(LocalDcOpe.DocVend)
		lcVen = Alltrim(LocalDcOpe.DocVend)
	EndIf

	* Título do Label das Condições Comerciais
	lcPag = [Comerciais]
	If Not Empty(LocalDcOpe.DocPag)
		lcPag = Alltrim(LocalDcOpe.DocPag)
	EndIf

	* Título do Label da Dados de Entrega
	lcDad = [de Entrega]
	If Not Empty(LocalDcOpe.DocEntregs)
		lcDad = Alltrim(LocalDcOpe.DocEntregs)
	EndIf

	llBar = (LocalDcOpe.DckBarra = 1)
	lcBar = fAltBar2de5(Padl(LocalOpe.NDopes, 4, '0') + Padl(ThisForm.pcNumes, 6, '0')) && Barra do Cabeçalho
	lcTel = Alltrim(LocalEmp.Tel1s) + Iif(Empty(Alltrim(LocalEmp.Tel2s)), '', '/' + Alltrim(LocalEmp.Tel2s))

	Insert Into crCabecalho (Dopes, Numes, Datas, RazSocs, Cgcs, Ies, Obses, TitFornec, DckPrzE, DocPrzE, DckReffs, ;
							 DocReffs, DckVend, TitVend, DckTVend, Barras, ImpCabBar, DckFretes, DocFretes, DckIcmIpi, ;
							 DocPag, DckEntregs, DocEntregs, DckObsI, Telefone, TitTermos, TxtTermos, TitTermgs, ;
							 TxtTermgs, TxtTermBs) ;
					 Values (LocalDcOpe.DocTit, ThisForm.pcNumes, Nvl(LocalEest.Datas, Ctod('')), LocalEmp.RazSocs, ;
					 		 LocalEmp.Cgcs, LocalEmp.Ies, Nvl(LocalEest.Obses, ''), lcFor, LocalDcOpe.DckPrzE, lcEnt, ;
					 		 LocalDcOpe.DckReffs, lcRef, LocalDcOpe.DckVend, lcVen, LocalDcOpe.DckTVend, lcBar, llBar, ;
					 		 LocalDcOpe.DckFretes, LocalDcOpe.DocFretes, LocalDcOpe.DckIcmIpi, lcPag, ;

					 		 Nvl(LocalDcOpe.TxtTermBs, []))

	lcBVd = Alltrim(Strtran(LocalEest.Vends, '/', '')) + '0'
	lcBVd = fAltBar2de5(lcBVd) && Barra do Cabeçalho

	Replace GrVends	   With Proper(LocalEest.GrVends), ;
			Vends      With LocalEest.Vends, ;
			DVends 	   With Iif(Empty(LocalCli.Razaos), LocalCli.RClis, LocalCli.Razaos), ;
			DddVends   With LocalCli.Ddds, ;
			Tel1sVends With LocalCli.Tel1s, ;
			Tel2sVends With LocalCli.Tel2s, ;
			eMailVends With LocalCli.eMails, ;
			BarVends   With lcBvd In crCabecalho

	* Inserindo Dados do Parceiro
	loCnx.CursorQuery('SigCdCli', 'LocalCli', 'IClis', LocalEest.Resps, 'Razaos, RClis')

	Replace GrResps	   With Proper(LocalEest.GrResps), ;
			Resps      With LocalEest.Resps, ;
			DResps 	   With Iif(Empty(LocalCli.Razaos), LocalCli.RClis, LocalCli.Razaos) In crCabecalho

	* Inserindo Dados do Fornecedor	
	If InList(LocalOpe.Opers, 1, 3) And Not Empty(LocalDcOpe.DocOrig)
		lcFor = LocalEest.ContaOs
	EndIf
	If InList(LocalOpe.Opers, 2, 3) And Not Empty(LocalDcOpe.DocDest)
		lcFor = LocalEest.ContaDs
	EndIf
	
	loCnx.CursorQuery('SigCdCli', 'LocalCli', 'IClis', lcFor, ;
					  'Razaos, RClis, Endes, Nums, Compls, Ceps, Cidas, Bairs, Estas, Contato, ' + ;
					  'Ddds, Tel1s, Tel2s, Faxs, eMails, Cpfs, Rgs, InscMuns, TpClis')

	Replace ForIClis   With lcFor, ;
			ForRClis   With Iif(Empty(LocalCli.Razaos), LocalCli.RClis, LocalCli.Razaos), ;
			ForEndes   With LocalCli.Endes, ;
			ForNums    With LocalCli.Nums, ;
			ForCompls  With LocalCli.Compls, ;
			ForCeps    With LocalCli.Ceps, ;
			ForCidas   With LocalCli.Cidas, ;
			ForBairs   With LocalCli.Bairs, ;
			ForEstas   With LocalCli.Estas, ;
			ForContato With LocalCli.Contato, ;
			ForDdds    With LocalCli.Ddds, ;
			ForTel1s   With LocalCli.Tel1s, ;
			ForTel2s   With LocalCli.Tel2s, ;
			ForFaxs    With LocalCli.Faxs, ;
			ForeMails  With LocalCli.eMails, ; 
			ForCgcs    With LocalCli.Cpfs, ;
			ForIes     With LocalCli.Rgs, ;
			ForIMuns   With LocalCli.InscMuns, ;
			ForTpClis  With LocalCli.TpClis In crCabecalho

	* Inserindo Dados de Entrega
	loCnx.CursorQuery('SigCdCli', 'LocalCli', 'IClis', LocalEest.ContaEs, 'Razaos, RClis, eMails, Faxs, Cpfs, Rgs, TpClis')
	If Not Eof('LocalCli')
		Replace EntIClis    With LocalEest.ContaEs, ;
				EntRClis    With Iif(Empty(LocalCli.Razaos), LocalCli.RClis, LocalCli.Razaos), ;
				EntFaxs     With LocalCli.Faxs, ;
				EnteMails   With LocalCli.eMails, ;
				EntCgcs     With LocalCli.Cpfs, ;
				EntIes      With LocalCli.Rgs, ;
				EntTpClis	With LocalCli.TpClis In crCabecalho
		
		Dimension laArray(2)
		laArray(1) = LocalEest.ContaEs
		laArray(2) = LocalEest.LocalEnts

		If fEndEntrega(@laArray, loCnx)
			lcEnd = Alltrim(laArray(06)) + ;
					Iif(Empty(laArray(07)), '', ', ' + Alltrim(laArray(07))) + ;
					Iif(Empty(laArray(08)), '', ' - ' + Alltrim(laArray(08)))
			
			Replace EntEndes    With lcEnd, ;
					EntCeps     With laArray(04), ;
					EntBairs    With laArray(09), ;
					EntCidas    With laArray(10), ;
					EntEstas    With laArray(11), ;
					EntDdds     With laArray(12), ;
					EntTel1s    With laArray(13), ;
					EntTel2s    With laArray(14), ;
					EntContato  With laArray(15) In crCabecalho
		EndIf
	Else
		loCnx.CursorQuery('SigCdCli', 'LocalCli', 'IClis', lcFor, ;
						  'Razaos, RClis, Endes, Nums, Compls, Ceps, Cidas, Bairs, Estas, Contato, ' + ;
						  'Ddds, Tel1s, Tel2s, Faxs, eMails, Cpfs, Rgs, InscMuns, TpClis')

		lcEnd = Alltrim(LocalCli.Endes) + ;
				Iif(Empty(LocalCli.Nums), '', ', ' + Alltrim(LocalCli.Nums)) + ;
				Iif(Empty(LocalCli.Compls), '', ' - ' + Alltrim(LocalCli.Compls))

		Replace EntIClis    With lcFor, ;
				EntRClis    With Iif(Empty(LocalCli.Razaos), LocalCli.RClis, LocalCli.Razaos), ;
				EntFaxs     With LocalCli.Faxs, ;
				EnteMails   With LocalCli.eMails, ;
				EntCgcs     With LocalCli.Cpfs, ;
				EntIes      With LocalCli.Rgs, ;
				EntTpClis	With LocalCli.TpClis, ;
				EntEndes    With lcEnd, ;
				EntCeps     With LocalCli.Ceps, ;
				EntBairs    With LocalCli.Bairs, ;
				EntCidas    With LocalCli.Cidas, ;
				EntEstas    With LocalCli.Estas, ;
				EntDdds     With LocalCli.Ddds, ;
				EntTel1s    With LocalCli.Tel1s, ;
				EntTel2s    With LocalCli.Tel2s, ;
				EntContato  With LocalCli.Contato In crCabecalho
	EndIf

	* Inserindo Condições Comerciais
	lcPag = []
	Select LocalPar
	Scan
		If (LocalPar.Total > 1)
			lcPag = lcPag + Iif(Empty(lcPag), [], [; ]) + Alltrim(Str(LocalPar.Total, 10)) + 'x ' + ;
					Alltrim(LocalPar.FPags) + ' - ' + Alltrim(LocalPar.MoeFpgs) + '$ ' + ;
					Alltrim(Str(LocalPar.Valos, 16, 2))
		Else
			lcPag = lcPag + Iif(Empty(lcPag), [], [; ]) + Alltrim(LocalPar.FPags) + ' - ' + ;
					Alltrim(LocalPar.MoeFpgs) + '$ ' + Alltrim(Str(LocalPar.Valos, 16, 2))
		EndIf
	EndScan

	* Título do Label de Variação
	If (LocalEest.ValVars > 0)
		lcVar = [Encargos Financeiros]
		If Not Empty(LocalDcOpe.DocAcres)
			lcVar = Alltrim(LocalDcOpe.DocAcres)
		EndIf
	Else
		lcVar = [Desconto Concedido]
		If Not Empty(LocalDcOpe.DocDesco)
			lcVar = Alltrim(LocalDcOpe.DocDesco)
		EndIf
	EndIf
	
	lcFrt = []
	lnIcm = 0
	If Used('LocalNFis')
		lnIcm = LocalNFis.ValIcms
		If Empty(LocalNFis.VlFretes)
			lcFrt = Iif(LocalNFis.Fretes <> 1, [Destinatário/FOB], [Emitente/CIF])
		Else
			lcFrt = Transform(LocalNFis.VlFretes, '9,999,999,999.99')
			Replace VlFretes With LocalNFis.VlFretes In crCabecalho
		EndIf
	EndIf

	* Apanhando o Prazo de Entrega
	ldOnt = Ttod(Nvl(LocalEest.PrazoEnts, Ctot([])))

	* Montando a Observação do Prazo de Entrega
	lcOnt = [] 
	If (LocalDcOpe.ChkPrzEnts = 1) And Not IsEmpty(LocalDcOpe.TxtPrzEnts)
		lcOnt = Strtran(Alltrim(LocalDcOpe.TxtPrzEnts), [#], Dtoc(ldOnt))
	EndIf
		
	Replace PrazoEnts  With ldOnt, ;
			Variacao   With Abs(LocalEest.ValVars), ;
			TipoVars   With lcVar, ;
			Icms	   With lnIcm, ;
			RespFretes With lcFrt, ;
			CondPagto  With lcPag, ;
			ObsEntrega With lcOnt In crCabecalho

	* Inserindo Produtos
	lnPgn = 1
	lnItn = 0
	lnSeq = 0
	llMud = .f.
	Select LocalEestI
	Scan
		llMud = .f.

		Select LocalEstI2
		If Seek(Str(LocalEestI.CItens, 6), 'LocalEstI2', 'CItens')
			Select LocalEstI2
			Scan While (CItens = LocalEestI.CItens)
				lnItn = lnItn + 1
				lnSeq = lnSeq + 1

				Insert Into crDados (Pagina, CItens, CPros, Reffs, DPros, CUnis, Qtds, Units, Totas, Aliqs, ValIpis, Obs) ;
							 Values (lnPgn, lnSeq, LocalEstI2.CPros, LocalEstI2.Reffs, LocalEstI2.DPros, ;
							 		 LocalEstI2.CUnis, LocalEstI2.Qtds, LocalEstI2.Units, LocalEstI2.Totas, ;
							 		 LocalEstI2.Aliqs, LocalEstI2.ValIpis, Nvl(LocalEstI2.Obs, ''))

				If (lnItn >= lnMax)
					lnItn = 1
					lnPgn = lnPgn + 1
					llMud = .t.
				EndIf
			EndScan
			Select LocalEestI
		Else
			lnItn = lnItn + 1
			lnSeq = lnSeq + 1

			Select LocalEestI
			Insert Into crDados (Pagina, CItens, CPros, Reffs, DPros, CUnis, Qtds, Units, Totas, Aliqs, ValIpis, Obs) ;
						 Values (lnPgn, lnSeq, LocalEestI.CPros, LocalEestI.Reffs, LocalEestI.DPros, ;
						 		 LocalEestI.CUnis, LocalEestI.Qtds, LocalEestI.Units, LocalEestI.Totas, ;
						 		 LocalEestI.Aliqs, LocalEestI.ValIpis, Nvl(LocalEestI.Obs, ''))

			If (lnItn >= lnMax)
				lnItn = 1
				lnPgn = lnPgn + 1
				llMud = .t.
			EndIf
		EndIf
	EndScan
	* Completando o Cursor Com Registros em Branco Até Preencher a Folha
	If llMud
		lnPgn = lnPgn - 1
	EndIf

	If (lnItn < lnMax) And Not llMud
		For I = lnItn To (lnMax - 1)
			Insert Into crDados (Pagina) Values (lnPgn)
		Next
	EndIf

	* Criando o Arquivo do Logotipo Para o Report
	ThisForm.ImgLogotipo = Sys(2023) + [\] + Sys(2015) + [.jpg]
	ThisForm.ImgTecnicas = Sys(2023) + [\] + Sys(2015) + [.jpg]
	ThisForm.ImgBranco   = fCriarIcone([brjpg.jpg], Sys(2023), loCnx)

	If Empty(ThisForm.ImgBranco) Or Not File(ThisForm.ImgBranco)
		=MessageBox('Não Foi Encontrado Um .JPG em Branco!!!' + Chr(13) + ;
					'Procurar Por: ' + Alltrim(ThisForm.ImgBranco), 0+48, '')
		Return 0
	EndIf

	Create Cursor csBranco (Branco m(4))
	Select csBranco
	Append Blank
	Append Memo Branco From (ThisForm.ImgBranco) OverWrite

	If Not IsEmpty(LocalParam.Logos)
		Clear Resources
		Copy Memo LocalParam.Logos To (ThisForm.ImgLogotipo)
	Else
		Clear Resources
		Copy Memo csBranco.Branco To (ThisForm.ImgLogotipo)
	EndIf

	If Not IsEmpty(LocalDcOpe.ImgTecs)
		Clear Resources
		Copy Memo LocalDcOpe.ImgTecs To (ThisForm.ImgTecnicas)
	Else
		Clear Resources
		Copy Memo csBranco.Branco To (ThisForm.ImgTecnicas)
	EndIf

	* Total de Páginas do Relatório
	Replace TotPag With lnPgn, ;
			Imagem With ThisForm.ImgLogotipo, ;
			ImgTec With ThisForm.ImgTecnicas In crCabecalho
EndIf
Select crCabecalho
Go Top

Select crDados
Go Top
* Fim **************************************************************************************
ENDPROC

* PROCEDURE selecionadados

PROCEDURE selecionadados
Local lcEdn, loCnx

lcEdn = Padr(ThisForm.pcEmps, 3) + Padr(ThisForm.pcDopes, 20) + Str(ThisForm.pcNumes, 6)
loCnx = ThisForm.poDataMgr

* Campos Usados Do SigCdOpe
* Opers		= Tipo da Operação = 1-Entrada; 2-Saida
* NDopes    = Número da Operação
loCnx.CursorQuery('SigCdOpe', 'LocalOpe', 'Dopes', Padr(ThisForm.pcDopes, 20), 'Opers, NDopes, Docus')
Go Top In LocalOpe
If Eof('LocalOpe')
	Return .f.
EndIf

* Campos Usados Do SigOpCdc
* DocTit	 = Título a Ser Impresso no Documento
* DocOrig	 = Descritivo do Label de Fornecedor Quando For Uma Operação de Entrada
* DocDest	 = Descritivo do Label de Fornecedor Quando For Uma Operação de Saída
* DckReffs   = Referência do Fornecedor = 1-Sim; 2-Não
* DocReffs   = Descritivo do Label de Referência do Fornecedor
* DckPrzE    = Prazo de Entrega = 1-Sim; 2-Não
* DocPrzE    = Descritivo do Label de Prazo de Entrega
* TxtPrzEnts = Texto de Prazo de Entrega Após Linha de Ítens
* ChkPrzEnts = Imprime ou Não o Texto de Prazo de Entrega
* DckVend	 = Responsável = 1-Sim; 2-Não
* DocVend	 = Descritivo do Label do Responsável
* DocPag	 = Descritivo do Label das Condições
* DckTVend	 = Imprime Dados de Contato do Responsável = 1-Sim; 2-Não
* DckBarra   = Imprime Código de Barra = 1-Sim; 2-Não
* DckFretes  = Imprime Frete = 1-Sim; 2-Não
* DocFretes  = Descritivo do Label de Frete
* DckIcmIpi  = ICMS/IPI = 1-Nenhum; 2-ICMS; 3-IPI; 4-Ambos
* DocDesco   = Descritivo do Label de Desconto
* DocAcres   = Descritivo do Label de Acréscimo
* DckEntregs = Imprime Dados de Entrega = 1-Sim; 2-Não
* DocEntregs = Descritivo do Label de Entrega
* DckObsI    = Imprime Observação dos Ítens = 1-Sim; 2-Não
* ImgTecs    = Imagem Técnica do Documento
* DckTermos  = Imprime Termos
* TitTermos  = Título do Termo
* TxtTermos  = Conteúdo do Termo
* ViaTermos  = Número de Vias do Termo
* TitTermGs  = Título do Termo de Garantia
* TxtTermGs  = Conteúdo do Termo de Garantia
* TxtTermBs  = Conteúdo do Termo Complementar de Garantia

loCnx.CursorQuery('SigOpCdc', 'LocalDcOpe', 'Dopes', Padr(ThisForm.pcDopes, 20), 'DocTit, DocOrig, DocDest, ' + ;
				  'DckReffs, DocReffs, DckPrzE, DocPrzE, TxtPrzEnts, ChkPrzEnts, DckVend, DocVend, DocPag, ' + ;
				  'DckTVend, DckBarra, DckFretes, DocFretes, DckIcmIpi, DocDesco, DocAcres, DckEntregs, DocEntregs, ' + ;
				  'DckObsI, ImgTecs, DckTermos, TitTermos, TxtTermos, ViaTermos, TitTermgs, TxtTermgs, TitTermGs, TxtTermBs')

Go Top In LocalDCOpe
If Eof('LocalDcOpe')
	Return .f.
EndIf

* Campos Usados Do SigCdPam
* Logos		= Logotipo da Empresa
loCnx.CursorQuery('SigCdPam', 'LocalParam', .f., .f., 'Logos')
Go Top In LocalParam
If Eof('LocalParam')
	Return .f.
EndIf

* Campos Usados Do SigCdEmp
* RazSocs	= Razão Social
* Cgcs		= CNPJ
* Ies		= Inscrição Estadual
loCnx.CursorQuery('SigCdEmp', 'LocalEmp', 'Cemps', Padr(ThisForm.pcEmps, 3), 'RazSocs, Cgcs, Ies, Tel1s, Tel2s')
Go Top In LocalEmp
If Eof('LocalEmp')
	Return .f.
EndIf

* Campos Usados Do SigMvCab
* Emps		= Empresa
* Numes		= Nº do Pedido
* Datas 	= Data
* GrVends 	= Grupo do Comprador
* Vends 	= Comprador
* GrResps 	= Grupo do Parceiro
* Resps 	= Parceiro
* ContaOs	= Fornecedor
* ContaEs	= Entrega
* PrazoEnts	= Data de Entrega
* Obses		= Observação do Rodapé
* ValVars	= Se Negativo Considera Como Desconto
loCnx.CursorQuery('SigMvCab', 'LocalEest', 'EmpDopNums', lcEdn, 'Emps, Dopes, Numes, Datas, GrVends, Vends, ' + ;
				  'GrResps, Resps, ContaOs, ContaDs, ContaEs, PrazoEnts, Obses, ValVars, LocalEnts')
Go Top In LocalEest
If Eof('LocalEest')
	Return .f.
EndIf

* Campos Usados Do SigMvItn
* CItens 	= Nº do Ítem
* CPros		= Código do Produto
* Reffs		= Referência do Fornecedor (SigCdPro)
* DPros		= Descrição do Produto
* CUnis		= Unidade
* Qtds		= Quantidade
* Units		= Valor Unitário
* Totas		= Valor Total
* Aliqs		= Alíquota do IPI
* ValIpis	= Valor do IPI
lcQuery = [Select a.CItens, a.CPros, b.Reffs, a.DPros, a.CUnis, a.Qtds, a.Units, a.Totas, a.Aliqs, a.ValIpis, a.Obs ] + ;
		    [From SigMvItn a, SigCdPro b ] + ;
		   [Where a.EmpDopNums = '] + lcEdn + [' And ] + ;
		   		 [a.CPros = b.CPros ] + ;
		   [Order By a.CItens]

If (loCnx.SqlExecute(lcQuery, 'LocalEestI') < 1)
	=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (LocalEestI)')
	Return 0
EndIf
Go Top In LocalEestI

* Campos Usados Do SigMvIts
* CItens 	= Nº do Ítem
* CPros		= Código do Produto
* Reffs		= Referência do Fornecedor (SigCdPro)
* DPros		= Descrição do Produto
* CUnis		= Unidade
* Qtds		= Quantidade
* Units		= Valor Unitário
* Totas		= Valor Total
* Aliqs		= Alíquota do IPI
* ValIpis	= Valor do IPI
lcQuery = [Select a.CItens, a.CPros, c.Reffs, a.DPros, b.CodEmbs as CUnis, b.Qtds, ] + ;
				 [a.Units, b.QtdEmbs, a.Aliqs, a.ValIpis, a.Qtds as QtdsA, a.Obs ] + ;
		    [From SigMvItn a, SigMvIts b, SigCdPro c ] + ;
		   [Where a.EmpDopNums = '] + lcEdn + [' And ] + ;
		   		 [a.EmpDopNums = b.EmpDopNums And ] + ;
		   		 [a.CItens = b.CItens And ] + ;
		   		 [b.CPros = c.CPros ] + ;
		   [Order By a.CItens]

If (loCnx.SqlExecute(lcQuery, 'crLclMvSub') < 1)
	=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (crLclMvSub)')
	Return 0
EndIf

Select CItens, CPros, Reffs, DPros, CUnis, Qtds, Units * Iif(QtdEmbs > 0, QtdEmbs, 1) as Units, ;
	  (Qtds * Units) * Iif(QtdEmbs > 0, QtdEmbs, 1) as Totas, Aliqs, ;
	  (ValIpis / QtdsA) * (Qtds * Iif(QtdEmbs > 0, QtdEmbs, 1)) as ValIpis, Obs ;
  From crLclMvSub ;
 Order By CItens ;
  Into Cursor LocalEstI2 ReadWrite

Go Top In LocalEstI2
Select LocalEstI2
Index On Str(CItens, 6) Tag CItens

* Campos Usados Do SigMvNfi
* ValIcms	= Valor do Icms da Nota
* Fretes	= Responsável Pelo Frete
* VlFretes  = Valor do Frete
loCnx.CursorQuery('SigMvNfi', 'LocalNFis', 'EmpDopNums', lcEdn, 'ValIcms, Fretes, VlFretes')
Go Top In LocalNFis
If Not Used('LocalNFis')
	Return .f.
EndIf

* Campos Usados Do SigMvPar
* FPags		= Forma de Pagamento
* MoeFpgs	= Moeda da Forma de Pagamento
* Valos		= Valor
loCnx.CursorQuery('SigMvPar', 'LocalPar1', 'EmpDopNums', lcEdn, 'FPags, MoeFpgs, Valos')
Go Top In LocalPar1
If Not Used('LocalNFis')
	Return .f.
Else
	Select Sum(1) as Total, FPags, MoeFpgs, Valos ;
	  From LocalPar1 ;
	 Group By FPags, MoeFpgs, Valos ;
	  Into Cursor LocalPar ReadWrite
EndIf
* Fim **************************************************************************************
ENDPROC
