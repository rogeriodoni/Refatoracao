PROCEDURE msv_procurar

Locate for Codigos >= ThisForm.Pagina.Dados.GetCodigos.Value
* Fim **************************************************************************************

------------------------------------------------------------
* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
With ThisForm
	.poDataMgr = CreateObject('fSqlConector')

	If (.poDataMgr.pnIdconn > 0)
		With .poDataMgr
			.AddCursor('SigCrRac', 'Codigos', 'crSigCrRac', '', ThisForm.Pagina.Lista.Grade)
			.pfSqlTabela(1).pColuna('Grupos',  '', '', 'Grupo', 	 080, .t.)
			.pfSqlTabela(1).pColuna('Codigos', '', '', 'Código',    150, .t.)
			.pfSqlTabela(1).pColuna('Descrs',  '', '', 'Descrição', 290, .t.)
		EndWith

		=DoDefault()

		ThisForm.Pagina.Lista.Grade.Column1.SetFocus
	Else
		=Messagebox('Impossível Efetuar Conexão Com o Servidor de Banco de Dados...', 0+48, 'CreateObject')
		Return .f.
	EndIf
EndWith
* Fim **************************************************************************************
ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Lista.Grupo_op.Click
------------------------------------------------------------
PROCEDURE Pagina.Lista.Grupo_op.Click
Lparameters Opcao

If Not DoDefault(Opcao)
	ThisForm.mAtivaPagina1
	Return 0
EndIf

Do Case
	Case InList(Thisform.pcEscolha, 'INSERIR', 'PROCURAR')
		Thisform.Pagina.Dados.getCodigos.Setfocus

	Case (Thisform.pcEscolha = 'ALTERAR')
		Thisform.Pagina.Dados.getDescrs.Setfocus

	Case InList(Thisform.pcEscolha, 'CONSULTAR', 'EXCLUIR')
		Thisform.Pagina.Dados.Grupo_Salva.Salva.Setfocus
EndCase

Select crSigCrRac

ThisForm.Pagina.Dados.getGrupos.Valid()
* Fim **************************************************************************************
ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
If Inlist(ThisForm.Pagina.Lista.Grupo_op.Value, 1, 3)
	If Empty(Thisform.Pagina.Dados.getCodigos.Value)
		=Messagebox('Código Inválido!!!', 0+64, '')
		Thisform.Pagina.Dados.getCodigos.Setfocus
		Return 0
	Else
		If InList(ThisForm.pcEscolha, 'INSERIR')
			ThisForm.poDataMgr.CursorQuery('SigCrRac', 'crBusca', 'Codigos', crSigCrRac.Codigos, 'Codigos')
			If Not Eof('crBusca')
				=MessageBox('Código da Característica Já Cadastrado!!!', 0+64, '')
				Thisform.Pagina.Dados.getDescrs.Setfocus
				Return 0
			EndIf
		EndIf
	EndIf

	If Empty(Thisform.Pagina.Dados.GetDescrs.Value)
		=Messagebox('Descricao Inválida!!!', 0+64, '')
		Thisform.Pagina.Dados.getDescrs.Setfocus
		Return 0
	Else
		If InList(ThisForm.pcEscolha, 'INSERIR', 'ALTERAR')
			ThisForm.poDataMgr.CursorQuery('SigCrRac', 'crBusca', 'Descrs', crSigCrRac.Descrs, 'Descrs')
			If Not Eof('crBusca')
				=MessageBox('Descrição da Característica Já Cadastrada!!!', 0+64, '')
				Thisform.Pagina.Dados.getDescrs.Setfocus
				Return 0
			EndIf
		EndIf
	EndIf
EndIf

If InList(ThisForm.pcEscolha, 'EXCLUIR')
	lcQuery = [Select * ] + ;
			    [From SigClCrc ] + ;
			   [Where Caracteris = '] + ThisForm.Pagina.Dados.GetDescrs.Value + [']

	If (ThisForm.poDataMgr.SqlExecute(lcQuery, 'crSigClCrc') < 1)
		=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (crSigClCrc)')
		Return 0
	EndIf

	If (Reccount('crSigClCrc') > 0)
		=MessageBox('Característica Sendo Utilizada!!!', 16, '')
		Return 0
	EndIf
EndIf

Select crSigCrRac

=DoDefault()

ThisForm.mAtivaPagina1
* Fim **************************************************************************************
ENDPROC
