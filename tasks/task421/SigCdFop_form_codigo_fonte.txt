PROCEDURE Init
LParameters poForm, pnDataSes, pcEscolha, pcProduto

ThisForm.cntSombra.lblSombra.Caption = ThisForm.Caption
ThisForm.cntSombra.lblTitulo.Caption = ThisForm.Caption

Private pPro

Set DataSession to pnDataSes

=DoDefault()

With ThisForm
	.ParentForm = poForm
	.pcescolha = pcEscolha
	.houveincl = .F.
	.houveexcl = .F.
	
	*-- Rafael - 10/11/2016 - Opção para repetir o mesmo registro mais de uma vez (Rosana Joias)
	.getQtde.Visible = .F.
	.shpQtde.Visible = .F.
	.lblQtde.Visible = .F.
	.ShpRepetir.Visible = fChecaAcesso([SIGCDPRO],[REPETEFORN])
	.cmdRepetir.Visible = fChecaAcesso([SIGCDPRO],[REPETEFORN])
	
	.poDataMgr = CreateObject('fSqlConector', ThisForm.Name)
	
	If Type('pcProduto') = [C]
		pcPro = pcProduto
		lcSql = [Select Dpros From SigCdPro Where Cpros = ']+pcPro+[']
		.PodataMgr.Sqlexecute(lcSql,'LocalPro')
		pdPro = LocalPro.Dpros
		
		.Grade.RecordSource = ''

		If Used('CrSigPrFnc')
			Use in CrSigPrFnc
		Endif
		
		lcSql = [Select a.*, b.rClis From SigPrFnc a Inner Join SigCdCli b on b.iClis = a.IFors Where cPros = ']+pcPro+[']
		.Podatamgr.Sqlexecute(lcSql,'CrSigPrFnc')
	Else
		pcPro = crSigCdPro.cPros
		pdPro = CrSigCdPro.Dpros
		pPro = pcPro

		If (.Parentform.ProcForn) 
			WITH ThisForm.Parentform 
				With .poDatamgr
		    	    .ReQuery('crSigPrFnc')
				EndWith
			Endwith	
		EndIf
	EndIf

	llVis = InList(.pcEscolha,'INSERIR','ALTERAR')
	.Caption = [Fornecedores do Produto : ] + Alltrim(pcPro) + [-] + pdPro
	.cmdInserir.Visible = llVis
	.cmdInserir.Enabled = llVis
	.cmdExcluir.Visible = llVis
	.cmdExcluir.Enabled = llVis
	IF !llvis
		.shape1.width = .cmdsair.width + 10
		.shape1.left = .cmdsair.left - 5
	ENDIF
EndWith

Update crSigPrFnc Set DtValid = Nvl(DtValid, Ctot([]))

Select crSigPrFnc
Go Top
With ThisForm.Grade
	.RecordSource = [crSigPrFnc]
	.Column1.ControlSource = [crSigPrFnc.iFors]
	.Column2.ControlSource = [crSigPrFnc.rClis]
	.Column3.ControlSource = [crSigPrFnc.Reffs]
	.Column4.ControlSource = [crSigPrFnc.Moevs]
	.Column5.ControlSource = [crSigPrFnc.PVens]
	.Column6.ControlSource = [crSigPrFnc.dtValid]
	.Column7.ControlSource = [crSigPrFnc.Situas]
	.Column8.ControlSource = [crSigPrFnc.Colecoes]
	.Column9.ControlSource = [crSigPrFnc.CodCors]
	.Column10.ControlSource= [crSigPrFnc.CodTams]
EndWith
ENDPROC

Else
	Replace iFors with '',;
			rClis with '' in crSigPrFnc
EndIf

Return(.T.)
ENDPROC

* OBJETO: SIGCDFOP.cmdSair
* Classe: fwbtng (Base: commandbutton)
* Registro: 32
================================================================================

------------------------------------------------------------
* PROCEDURE Click
------------------------------------------------------------
PROCEDURE Click
If Thisform.cmdInserir.Visible Or Thisform.cmdExcluir.Visible
	If( InList(ThisForm.pcescolha,'INSERIR','ALTERAR') ) 
		Select crSigPrFnc
		Go Top
		Scan
			If( IsEmpty(crSigPrFnc.iFors) ) 
				MessageBox('Fornecedor deve ser informado!',48,'Atenção')
				ThisForm.Grade.Column1.SetFocus
				Return(.F.)  
			EndIf
			If( IsEmpty(crSigPrFnc.Reffs) )
				MessageBox('A referência do fornecedor deve ser informada!',48,'Atenção')
				Thisform.Grade.Column3.SetFocus 
				Return(.F.)  
			EndIf
			If( IsEmpty(crSigPrFnc.Moevs) )
				MessageBox('A Moeda deve ser informada!',48,'Atenção')
				ThisForm.Grade.Column4.SetFocus
				Return(.F.)  
			EndIf
			
			*-- Rafael - 11/11/2016 - quando o Usuario tiver acesso a repetir nao valida se tem informação repetida
			llPermiteRep = fChecaAcesso([SIGCDPRO],[REPETEFORN])
			
			Select a.IFors ;
			  From crSigPrFnc a ;
			 Where a.IFors = crSigPrFnc.IFors And ;
				   a.Reffs = crSigPrFnc.Reffs And ;
			 	   a.Colecoes = crSigPrFnc.Colecoes And ;
			 	   a.cIdChaves <> crSigPrFnc.cIdChaves and;
			 	   a.PVens = crSigPrFnc.PVens ;
			  Into Cursor crAux
			Select crAux
			Go Top In crAux
			
			If Not Eof('crAux') And Not llPermiteRep
				MessageBox('Fornecedor já informado para este produto!',48,'Atenção')
				ThisForm.Grade.Column1.SetFocus
				Return(.F.)  				
			Endif			
		EndScan	
	EndIf
EndIf

Thisform.ParentForm.Enabled = .T.
ThisForm.Release
ENDPROC
