* PROCEDURE desmarca

PROCEDURE desmarca
Private pNot

pNot = .f.

=Seek(TmpPends.Dopps, 'crSigCdOpd', 'Dopps')
_Nivel = TmpPends.Emps + TmpPends.Dopps + Str(TmpPends.Numps, 10)

ThisForm.poDataMgr.CursorQuery('SigCdNec', 'LocalNens', 'EmpDNPs', _Nivel, 'cIdChaves')
Select LocalNens

lcQuery = [Update SigCdNec ] + ;
			 [Set ChkSubn = ?pNot ] + ;
		   [Where cIdChaves = '] + LocalNens.cIdChaves + [']

If (ThisForm.poDataMgr.SqlExecute(lcQuery, '') < 1)
	=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (Marca - Update - SigCdNec)')
	Return 0
EndIf

=Seek(TmpNens.Dopps, 'crSigCdOpd', 'Dopps')
* Fim **************************************************************************************

ENDPROC

------------------------------------------------------------
* PROCEDURE marca
------------------------------------------------------------
PROCEDURE marca
Parameters _Automatico

Private pTru
Local _PuxaOrigem, _PuxaDestino

pTru = .t.

=Seek(TmpPends.Dopps, 'crSigCdOpd', 'Dopps')
_Nivel    = TmpPends.Emps + TmpPends.Dopps + Str(TmpPends.Numps, 10)
_NumNivel = Val(Str(crSigCdOpd.Ndopps, 2) + Transform(TmpPends.Numps, '@L 9999999999'))

=Seek(TmpNens.Dopps, 'crSigCdOpd', 'Dopps')
_NumeroAnt   = (crSigCdOpd.Numeras = 3)
_BaixaSubN   = (crSigCdOpd.BxAutos = 1)
_Posicao     = TmpNens.Emps + TmpNens.Dopps + Str(TmpNens.Numps, 10)
_PuxaOrigem  = crSigCdOpd.ConOSubs
_PuxaDestino = crSigCdOpd.ConDSubs

If _NumeroAnt
	Select TmpPends
	_RecnoPend = Recno()
	_Conta = 0
	Scan
		If Flag Or Recno() = _RecnoPEnd
			_Conta = _Conta + 1
		EndIf
	EndScan

	If (_Conta > 1)
		=Messagebox('Este Tipo de Operação Só Aceita 1 Subnível!!!', 48, '')
		Go _RecnoPend
		Return .f.
	EndIf
	Go _RecnoPend
EndIf

Select crSigCdPam
Go Top

_Retorna = .t.

If TmpPends.Datas > TmpNens.Datas
	=MessageBox('Movimentação com divergência de Datas!!!', 0+48, '')
	_Retorna = .f.
Endif

ThisForm.poDataMgr.CursorQuery('SigCdNec', 'LocalNens', 'EmpDNPs', _Nivel, 'cIdChaves, ChkSubn')
Select LocalNens
If LocalNens.ChkSubn
	If Not _Automatico
		=MessageBox('Movimentação Já Utilizada em Outra Operação!!!', 0+48, '')
	EndIf
	_Retorna = .f.
EndIf

If _Retorna And Not _Automatico
	lcQuery = [Update SigCdNec ] + ;
				 [Set cIdChaves = '] + LocalNens.cIdChaves + [' ] + ;
			   [Where cIdChaves = '] + LocalNens.cIdChaves + [']

	If (ThisForm.poDataMgr.SqlExecute(lcQuery, '') < 1)
		=Messagebox('Subnivel em Uso Por Outra Movimentação!!!', 0+48, '')
		_Retorna = .f.
	EndIf
EndIf

If Not _Retorna
	Select TmpPends
	Return .f.
EndIf

Select LocalNens
If _BaixaSubN
	lcQuery = [Update SigCdNec ] + ;
				 [Set ChkSubn = ?pTru ] + ;
			   [Where cIdChaves = '] + LocalNens.cIdChaves + [']

	If (ThisForm.poDataMgr.SqlExecute(lcQuery, '') < 1)
		=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (Marca - Update - SigCdNec)')
		Return 0
	EndIf
EndIf

If _NumeroAnt
	Select TmpNens
	Go top
	Replace Numps With LocalNens.Numps, ;
			Obss  With LocalNens.Obss
EndIf

If _NumeroAnt
	This.ok.click
EndIf
* Fim **************************************************************************************

ENDPROC

------------------------------------------------------------
* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
LParameters pForm, DatSes, Operacao


ThisForm.cntSombra.lblSombra.Caption = ThisForm.Caption
ThisForm.cntSombra.lblTitulo.Caption = ThisForm.Caption

Private pEd1, pEd2

This.LockScreen = .t.

ThisForm.ParentForm = pForm
ThisForm.poDataMgr  = ThisForm.ParentForm.poDataMgr
ThisForm.pEmps		= ThisForm.ParentForm.pEmps
ThisForm.contador   = 0
Set DataSession To DatSes

ThisForm.pcEscolha = ThisForm.ParentForm.pcEscolha
ThisForm.GradeTotal.Visible = .f.

Create Cursor TmpPends(Flag l, Emps c(3), Dopps c(20), Numps n(10), Datas d Null, Obss m Null, ;
					   Grupo c(10), Conta c(10), TPesos n(12,2), Usuars c(10))
Index On NumPs Tag Numps
Index On Emps + Dopps + Str(Numps, 10) Tag EmpDopNum


Select *, 000000.000 as ps2FaseAnt, Space(3) as cUnips, .f. as OsRets, .f. as Retrabs, Space(10) as MatPrincs,;
	000000.000 as PsFaseAnt, Space(60) as Descs, ;
	0 as Rateio, .f. as Visivel, 000000.000 as QtFaseAnt ;
From CrSigCdNei Where 1=0 Into Cursor xpNensi ReadWrite
Index On Cats Tag Cats
Index On Str(NEnvs, 10) + CMats + TpOps For Visivel Tag NEnvTpops
Index On Emps + Dopps + Str(Numps, 10)  For Visivel Tag EmpDopNum

Select *, 000000.000 as ps2FaseAnt, Space(3) as cUnips, .f. as OsRets, .f. as Retrabs, Space(10) as MatPrincs,;
	000000.000 as PsFaseAnt, Space(60) as Descs, ;
	0 as Rateio, .f. as Visivel, 000000.000 as QtFaseAnt ;
From CrSigCdNei Where 1=0 Into Cursor xpNensiT ReadWrite
Index On Emps + Dopps + Str(Numps, 10) + Cmats + Str(Nops,10) For Visivel Tag EmpDopNum

Select TmpOperacao
Delete All For Empty(Codigos)

=Seek(TmpNens.Dopps, 'crSigCdOpd', 'Dopps')

_DoppsNivel  = crSigCdOpd.Nivels
_PuxaOrigem  = crSigCdOpd.ConOSubs
_PuxaDestino = crSigCdOpd.ConDSubs

ThisForm.OpSubNível.Value = _DoppsNivel

=Seek(_DoppsNivel, 'crSigCdOpd', 'Dopps')
_NumeroNivel = Str(crSigCdOpd.NDopps, 2)

pEd1 = ThisForm.pEmps + _DoppsNivel + '         0'
pEd2 = ThisForm.pEmps + _DoppsNivel + '9999999999'
llSubn = .f.
lcQuery = [Select Emps, Dopps, Numps, Datas, GrupoOs, ContaOs, ChkSubn, TotPesos, Usuars, Obss ] + ;
		    [From SigCdNec ] + ;
		   [Where EmpDNPs BetWeen ?pEd1 And ?pEd2 And ] + ;
		   		 [GrupoDs = '] + TmpNens.GrupoOs + [' And ] + ;
		   		 [ContaDs = '] + TmpNens.ContaOs + [' And ChkSubn = ?llSubn]

If (ThisForm.poDataMgr.SqlExecute(lcQuery, 'LocalNens') < 1)
	=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (LocalNens)')
	Return 0
EndIf

Select LocalNens
Scan
	_Grupo = LocalNens.GrupoOs
	_Conta = LocalNens.ContaOs

	_Flag = Seek(Val(_NumeroNivel + Transform(LocalNens.Numps, '@L 9999999999')), 'TmpOperacao', 'Codigos')
	Insert Into TmpPends (Flag, Emps, Dopps, Numps, Datas, Grupo, Conta, TPesos, Usuars, Obss) ;
				  Values (_Flag, LocalNens.Emps, LocalNens.Dopps, LocalNens.Numps, LocalNens.Datas, ;
				  		  _Grupo, _Conta, LocalNens.TotPesos, LocalNens.Usuars, LocalNens.Obss)

	_ChvNens = LocalNens.Emps + LocalNens.Dopps + Str(LocalNens.Numps, 10)

	ThisForm.poDataMgr.CursorQuery('SigCdNei', 'LocalNensi', 'EmpDNPs', _ChvNens, ;
								   'Emps, Dopps, Numps, Nenvs, CMats, Nops, Servicos, Cats, CDescs, Qtds, ' + ;
								   'Pesos, TpOps, FlagIncs, CUnis, Obss, Fators, Moedas, GrupoFins, Peso2s')
	Select LocalNensi
	Scan
		_NEnv     = LocalNensi.NEnvs
		_TotalNop = 0
		_RecNensi = Recno()

		Scan While (NEnvs = _NEnv)
			ThisForm.poDataMgr.CursorQuery('SigCdPro', 'LocalPro', 'CPros', LocalNensi.CMats, 'DPros, CUnis, CGrus, Varias')

			If (LocalNensi.Nops = 0)
				lnRateio  = 5
				llVisivel = .t.
			Else
				lnRateio  = Iif(LocalPro.Varias = 1, 2, 3)
				llVisivel = .f.
			EndIf

			_Servico = LocalNensi.Servicos

			If Not _Servico And Not Empty(LocalNensi.CMats) And Empty(LocalNensi.Cats)
				Insert Into xPNensi (Emps, Dopps, Numps, Nops, NEnvs, CMats, CDescs, QtFaseAnt, Qtds, PsFaseAnt, ;
								     Pesos, Visivel, Rateio, TpOps, Cats, FlagIncs, CUnis, CGrus, Obss, aQtds,;
								     Peso2s, Ps2FaseAnt) ;
							 Values (LocalNensi.Emps, LocalNensi.Dopps, LocalNensi.Numps, LocalNensi.Nops, ;
							 		 LocalNensi.NEnvs, LocalNensi.CMats, LocalNensi.CDescs, LocalNensi.Qtds, ;
							 		 LocalNensi.Qtds, LocalNensi.Pesos, LocalNensi.Pesos, llVisivel, lnRateio, ;
							 		 LocalNensi.TpOps, LocalNensi.Cats, LocalNensi.FlagIncs, LocalNensi.CUnis, ;
							 		 LocalPro.CGrus, LocalNensi.Obss, LocalNensi.Qtds, ;
							 		 LocalNensi.Peso2s, LocalNensi.Peso2s)

				If (LocalNensI.FlagIncs) Or CrSigCdOpd.SepPedras = 4
					_BMat   = LocalNensI.CMats
					_DMat   = LocalPro.DPros
					_CMat   = LocalPro.CUnis
					_GMat   = LocalPro.CGrus
					_CatMat = LocalNensI.Cats
				Else
					_BMat   = Space(14)
					_DMat   = ''
					_CMat   = ''
					_GMat   = ''
					_CatMat = '   '
				EndIf
				_TMat = LocalNensI.TpOps

				If (crSigCdOpd.CompAgrus = 1) And CrSigCdOpd.SepPedras <> 4
					_Nenv = 0
				EndIf

				If (lnRateio <> 5)
					If Not Seek(Str(_Nenv, 10) + _BMat + _TMat, 'xPNensi', 'NEnvTpops')
						Insert Into xPNensi (Emps, Dopps, Numps, Nops, Nenvs, CMats, CDescs, CUnis, ;
											 CGrus, Visivel, Rateio, Cats, TpOps, Servicos) ;
									 Values (LocalNensi.Emps, LocalNensi.Dopps, LocalNensi.Numps, _NEnv, ;
									 		 _NEnv, _BMat, _DMat, _CMat, _GMat, .t., 1, _CatMat, _TMat, _Servico)
					Else
						If (xPNensi.Rateio <> 1)
							Replace Rateio With 1 In xPNensi
						EndIf
					EndIf

					Select xPNensi
					Replace QtFaseAnt  With xPNensi.QtFaseAnt + LocalNensi.Qtds, ;
							PsFaseAnt  With xPNensi.PsFaseAnt + LocalNensi.Pesos, ;
							Ps2FaseAnt With xNensi.Ps2FaseAnt + LocalNensi.Peso2s, ;
							Qtds       With xPNensi.Qtds + LocalNensi.Qtds, ;
							Pesos      With xPNensi.Pesos + LocalNensi.Pesos In xPNensi
				EndIf
			Else
				If Not Seek(LocalNensi.Cats, 'xPNensi', 'Cats') Or LocalNensi.FlagIncs
					Insert Into xPNensi (Emps, Dopps, Numps, CMats, CDescs, CUnis, CGrus, Visivel, Rateio, ;
										 FlagIncs, Cats, Servicos, Fators, Moedas, GrupoFins) ;
								 Values (LocalNensi.Emps, LocalNensi.Dopps, LocalNensi.Numps, LocalNensi.CMats, ;
								 		 LocalPro.Dpros, LocalPro.CUnis, LocalPro.CGrus, .t., 0, .t., LocalNensi.Cats, ;
								 		 .t., LocalNensi.Fators, LocalNensi.Moedas, LocalNensi.GrupoFins)
				EndIf
				
				Replace Qtds with xPNensi.Qtds + LocalNensi.Qtds In xPNensi
			EndIf
		EndScan
		Skip -1
	EndScan
EndScan

Select xPnensi
Scan
	Scatter Memvar
	m.Nops = Int(m.Nops/10000)
	lcEmpDnp = Emps + Dopps + Str(Numps,10)
	Select xPnensiT
	If NOt Seek(lcEmpDnP + m.Cmats + Str(m.Nops,10))
		Insert Into xPNensit (Emps, Dopps, Numps, Nops, NEnvs, CMats, CDescs, ;
					     Visivel, CUnis, CGrus, aQtds) ;
				 Values (m.Emps, m.Dopps, m.Numps, m.Nops, ;
				 		 m.nops, m.CMats, m.CDescs, .t., m.CUnis, ;
				 		 m.CGrus, LocalNensi.Qtds)
	EndIf
	Replace Qtds  With Qtds + m.Qtds,;
			Pesos With Pesos + m.Pesos
EndScan

Select TmpPends
Go Bottom

With Thisform.Grade
	.RecordSource            = 'TmpPends'

							   		   CrSigCdNec.Datas, CrSigCdNec.Usuars, CrSigCdNec.Grupoos, CrSigCdNec.Contaos, ;
							   		   CrSigCdNec.Grupods, CrSigCdNec.Contads, LocalNensI.Nops, LocalNensI.Nenvs, ;
							   		   _Cpro, LocalNensI.Cunis, 0, _Qtde)
				EndIf

				If Not Seek((_NumNivel * 10000000000) + xMFas.NEnvs, 'TmpOperacao', 'Codigos')
					Insert Into TmpOperacao (Codigos, EmpDnPs, Autos) Values ((_NumNivel * 10000000000) + xMFas.NEnvs, _ChaveNens,.t.)
				EndIf
			EndIf

		EndScan
		Select LocalNensi
		Skip - 1
	EndScan
EndScan
ThisForm.ParentForm.Enabled = .t.
ThisForm.Release

ENDPROC
