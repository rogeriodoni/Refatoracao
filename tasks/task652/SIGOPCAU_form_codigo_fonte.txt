PROCEDURE Init
WITH Thisform

	If Not .poDatamgr.pnIdconn>0
		Messagebox('Impossivel Efetuar Conexão com Servidor de Banco de Dados.',(0+48),'Erro')
		Return(.f.)
	Endif
Endwith	
	
Local m.loTree1

m.loTree1 = ThisForm.TreeView

m.loTree1.AddImage(LoadPicture('\vbmp\Menu1.Bmp'))
m.loTree1.AddImage(LoadPicture('\vbmp\x_planilha1.bmp'))

* PROCEDURE maketree
------------------------------------------------------------
PROCEDURE maketree
Local loTree1,lnNode1

If ! Empty(ThisForm.GetcUsu.Value)
	lcQuery = [Select * From SigCdUsu Where Usuarios = ']+ThisForm.GetcUsu.Value+[']
	ThisForm.poDataMgr.SqlExecute(lcQuery,'crSigCdUsu')

	* Richard em 29/11/2016 - Eliminando SUPERVIS
	If (Upper(Alltrim(Usuar))=[4CONTROL])
*!*		If Lower(Alltrim(crSigCdUsu.supervis))=='s' Or Lower(Alltrim(crSigCdUsu.supervis))=='i'
		lcQuery = [select a.programas,a.parametros,a.descricaos from SigCdPrg a where not a.programas='          ' ]+;
		          [group by a.programas,a.parametros,a.Descricaos order by a.Descricaos,a.programas,a.parametros]
	Else 
		lcQuery = [select a.programas,a.parametros,c.descricaos from SigCdAcU a ]+;
				  [left join SigCdAcG b on a.grupos=b.grupos ]+;
				  [left Join SigCdPrg c on a.Programas = c.Programas And a.Parametros = c.Parametros ]+;
				  [where not a.programas='          ' And (a.usuarios=']+ThisForm.GetcUsu.Value+;
				  [' or b.usuarios=']+ThisForm.GetcUsu.Value+[') ]+;
				  [group by a.programas,a.parametros,c.descricaos order by c.Descricaos, a.programas,a.parametros]
	EndIf
Else
	lcQuery = [select a.programas,a.parametros,c.descricaos from SigCdAcU a ]+;
			  [left join SigCdAcG b on a.grupos=b.grupos ]+;
			  [left Join SigCdPrg c on a.Programas = c.Programas And a.Parametros = c.Parametros ]+;
			  [where not a.programas='          ' And (a.Grupos=']+ThisForm.GetcUsu.Value+;
			  [' or b.Grupos=']+ThisForm.GetcUsu.Value+[') ]+;
			  [group by a.programas,a.parametros,c.descricaos order by c.Descricaos, a.programas,a.parametros]
	
Endif

ThisForm.poDataMgr.SqlExecute(lcQuery,'crSigCdAcU')

m.loTree1 = ThisForm.TreeView

m.loTree1.ClearNodes
m.loTree1.TipsDisplay = .t.
m.loTree1.ExpandOnDClick
m.loTree1.SmallIcons = .t.
m.loTree1.PicAlign = 0
m.loTree1.SelectedStyle = 2
m.loTree1.PictureType = 1

Select crSigCdAcU

Declare laNivel[1]
laNivel[1]=''

Scan
	lcDesc = fLimpaTexto(Alltrim(crSigCdAcU.Descricaos))
	If At('-',lcDesc) = 0
		m.lnNode1 = m.loTree1.AddPictureNode(Alltrim(lcDesc), 0, 1, 1, 0, 0)
		m.loTree1.NodeHeader(m.lnNode1) = .t.
		m.loTree1.NodeExpanded(m.lnNode1) = .f.
		m.loTree1.NodeShowChecks(m.lnNode1) = .f.
		m.loTree1.NodeFontBold(m.lnNode1) = 2
		m.loTree1.NodeCargo(m.lnNode1) = crSigCdAcU.Programas+crSigCdAcU.Parametros
		laNivel[1] = lcDEsc
	Else
		If ! IsNull(lcDesc)
			lnNivel = 1
			DO While .t.
				If Alen(laNivel) < lnNivel
					Declare laNivel[lnNivel]
					laNivel[lnNivel] = ''
					Exit
				Endif
				lcDescs = Alltrim(Substr(lcDesc,Iif(lnNivel=1,1,At('-',lcDesc,lnNivel-1)+1),At('-',lcDesc,lnNivel)-Iif(lnNivel=1,1,At('-',lcDesc,lnNivel-1)+1)-1))
				If Type('laNivel[lnNivel]') = 'L' And ! laNivel[lnNivel]
					laNivel[lnNivel] = ''
					Exit
				Endif
				If Upper(laNivel[lnNivel]) == Upper(lcDescs)
					lnNivel = lnNivel + 1
				Else
					Declare laNivel[lnNivel]
					laNivel[lnNivel] = ''
					Exit
				EndIf
			EndDo
			DO While At('-',lcDesc,lnNivel) <> 0
				If laNivel[Alen(laNivel)] <> Alltrim(Left(lcDesc,At('-',lcDesc,lnNivel)-1))
					lcDescs = Alltrim(Substr(lcDesc,Iif(lnNivel=1,1,At('-',lcDesc,lnNivel-1)+1),At('-',lcDesc,lnNivel)-Iif(lnNivel=1,1,At('-',lcDesc,lnNivel-1)+1)-1))
					laNivel[Alen(laNivel)] = lcDescs
					lnPicture = 3
					m.lnNode1 = m.loTree1.AddPictureNode(Alltrim(lcDescs), 0, lnNivel, lnPicture, 0, 0)
					m.loTree1.NodeHeader(m.lnNode1) = .t.
					m.loTree1.NodeExpanded(m.lnNode1) = .f.
					m.loTree1.NodeShowChecks(m.lnNode1) = .f.
					m.loTree1.NodeFontBold(m.lnNode1) = 2
					m.loTree1.NodeCargo(m.lnNode1) = crSigCdAcU.Programas+crSigCdAcU.Parametros
				EndIf
				lnNivel = lnNivel + 1
				Declare laNivel[Alen(laNivel)+1]
				laNivel[Alen(laNivel)] = ''
			EndDo
			laNivel[Alen(laNivel)] = Alltrim(Left(lcDesc,At('-',lcDesc,lnNivel)-1))
			lcDescs = Alltrim(Substr(lcDesc,Iif(lnNivel=1,1,At('-',lcDesc,lnNivel-1)+1),Len(lcDesc)))
			lnPicture = Iif(lnNivel=1,1,2)
			m.lnNode1 = m.loTree1.AddPictureNode(Alltrim(lcDescs), 0, lnNivel, lnPicture, 0, 0)
			m.loTree1.NodeHeader(m.lnNode1) = .t.
			m.loTree1.NodeExpanded(m.lnNode1) = .f.
			m.loTree1.NodeShowChecks(m.lnNode1) = .f.
			m.loTree1.NodeFontBold(m.lnNode1) = 2
			m.loTree1.NodeCargo(m.lnNode1) = crSigCdAcU.Programas+crSigCdAcU.Parametros
			laNivel[lnNivel] = lcDescs
		EndIf 
	EndIf
EndScan

m.loTree1.ReDraw(.t.)
m.loTree1.Refresh
ThisForm.pnNodeExp = 0



ENDPROC

PROCEDURE Command1.Click
loTree1 = ThisForm.TreeView
llExpande = .f.

For lnNode = 0 To loTree1.lIstCount-1
	If ThisForm.pnNodeExp + 1 = loTree1.NodeLevel(lnNode)
		loTree1.NodeExpanded(lnNode) = .t.
		llExpande = .t.
	Endif
EndFor

If llExpande 
	ThisForm.pnNodeExp = ThisForm.pnNodeExp + 1
Endif

ENDPROC

PROCEDURE Valid
Local xBrox

If not IsEmpty(This.value)

	xBrox = CreateObject('fwBuscaExt',Thisform.poDatamgr.pnIdconn,'SigCdUsu','crSigCdUsu','Usuarios',This.Value,'Seleção',.t.,'Usuarios')

	If  Not xBrox.plAchouRegistro
		xBrox.mAddColuna('Usuarios','','Código',.t.)
		xBrox.mAddColuna('nComps','','Nome',.T.)
		xBrox.Show()
	EndIf 

	If lastkey()<>27
		This.Parent.GetCUsu.Value = crSigCdUsu.Usuarios
		This.Parent.GetDUsu.Value = crSigCdUsu.nComps
	Else
		This.Parent.GetCUsu.Value = ''
		This.Parent.GetDUsu.Value = ''
	endif
	ThisForm.Refresh
EndIf

If ! Empty(This.Value) Or ! Empty(This.Parent.GetGrupo.Value)
	ThisForm.MakeTree()
Endif

return(.t.)

ENDPROC

------------------------------------------------------------
PROCEDURE When
Return(Empty(ThisForm.GetGrupo.Value))
ENDPROC
