PROCEDURE impressão
LParameters lcSaida

Create Cursor TmpCabec (Titulo c(100), SubTitulo c(100), Empresa c(100))

ThisForm.poDataMgr.CursorQuery('SigCdEmp', 'crSigCdEmp', 'Cemps', _Empr, 'Razas')

lcTitulo    = 'Saldos Por Funcionário'
lcSubTitulo = 'Material : ' + Alltrim(This.Get_Mat.Value) + ' - ' + Alltrim(This.Get_DMat.Value)
lcEmpresa   = Iif(Eof('crSigCdEmp'), '', crSigCdEmp.Razas)

Insert Into TmpCabec (Empresa, Titulo, SubTitulo) Values (lcEmpresa, lcTitulo, lcSubTitulo)

Select TmpSaldo
Set Order to Impressao
Go Top
If Eof() 
	=Messagebox('Nenhuma Informação Selecionada!!!', 32)
Else
	If (lcSaida = 'V')
		Report Form SigOpCfe Preview NoConsole
	Else
		Report Form SigOpCfe To Print Prompt
	EndIf
EndIf
Use In TmpCabec
* Fim **************************************************************************************

ENDPROC

ENDPROC

PROCEDURE Load
=fConfigGeral()

Create Cursor TmpResumo (Flag l, Flag2 l, Grupo c(10), Conta c(10), CMats c(14), CUnis c(3), PesoEnts n(12,3), ;
						 QtdeEnts n(12,3), PesoSais n(12,3), QtdeSais n(12,3), Saldoi n(12,3), Pesagem n(12,3), ;
						 FReal n(12,3), FAdmin n(12,3), Saldof n(12,3), PesoPEnts n(12,3), PesoPSais n(12,3), ;
						 PfTrabs n(9,2), Flag3 l, Varias n(1), PesoFabre n(12,3), PesoFabrs n(12,3), ;
						 cUniPs c(3), CodCors c(4), CodTams c(4), Visivel L, Agregas n(1))
Index On Cmats + CodCors + CodTams Tag cpros
Index On Grupo + Conta + CMats + CodCors + CodTams Tag GrConMat For VisiVel

Create Cursor TmpOpera (Grupo c(10), Conta c(10), CMats c(14), TpOps c(15), PesoEnts n(12,3), QtdeEnts n(12,3), ;
						PesoSais n(12,3), QtdeSais n(12,3), PFalha n(6,2), Falha n(12,3), PFalhaE n(6,2), ;
						FalhaE n(12,3), Varias n(1), Agregas n(1), AplicFlhs n(1), ;
						PesoBSais n(12,3), PesoBEnts n(12,3), QtdeBEnts n(12,3), QtdeBSais n(12,3))
Index On Grupo + Conta + CMats + TpOps Tag GrConMatTp

Create Cursor TmpSaldo (Codigo c(10), Nome c(50), Fase c(10), Saldo n(12,3), Data d)
Index On Codigo + Fase Tag Impressao For Saldo <> 0
Index On Codigo + Fase Tag Codigo
Index On Fase + Codigo Tag Carrega

ENDPROC

* PROCEDURE Init

PROCEDURE Init
ThisForm.cntSombra.lblSombra.Caption = ThisForm.Caption
ThisForm.cntSombra.lblTitulo.Caption = ThisForm.Caption
With ThisForm
	.poDataMgr = CreateObject('fSqlConector', ThisForm.Name)

	If (.poDataMgr.pnIdconn > 0)
		.CntMensagem.visible = .t.
		.CntMensagem.Label1.Caption = 'Aguarde! Preparando arquivos.'
		.Refresh
		.Draw
		.Show

		ThisForm.poDataMgr.CursorQuery('SigCdPam', 'crSigCdPam', .f., .f., 'Ouros, BalPortas, BalVelocs, ' + ;
									   'BalTppars, BalBits, BalParids, AcertBals, DopeBals, MascNums, ' + ;
									   'GrupoRecs, ContaRecs, GrupoPags, ContaPags')

		ThisForm.poDataMgr.CursorQuery('SigCdPac', 'crSigCdPac', .f., .f., 'ndFechas')

		ThisForm.poDataMgr.CursorQuery('SigOpOpt', 'crSigOpOpt', .f., .f., 'Cods, PFalhas, TFalhas, Trabas, Prodzs')
		Select crSigOpOpt
		Index On Cods Tag Cods

		ThisForm.poDataMgr.CursorQuery('SigCdOpe', 'crSigCdOpe', .f., .f., 'Dopes, Origems, EstOrigs, Destinos, EstDests, Opers')
		Select crSigCdOpe
		Index On Dopes Tag Dopes

		ThisForm.poDataMgr.CursorQuery('SigCdOpd', 'LocalOpp', .f., .f., 'Dopps, Origems, EstOrigs, Destinos, EstDests')
		Select LocalOpp
		Index On Dopps Tag Dopps

		ThisForm.poDataMgr.CursorQuery('SigCdGcr', 'crSigCdGcr', .f., .f., 'Codigos, UnifBals, GerBals')
		Select crSigCdGcr
		Index On Codigos Tag Codigos

		If (ThisForm.poDataMgr.getDriver() = 'foxpro')
			lcDiretorio = Addbs(JustPath(ThisForm.poDataMgr.GetDatabase()))
			Use &lcDiretorio.\SigCdPro Alias TmpPro In 0
		Else
			lcQuery = [Select CPros, DPros, CUnis, CGrus, Varias, Custofs, MoeCusfs, MatPrincs, cUniPs ] + ;

		Scan While Grupo + Conta + Cmats = TmpResumo.Grupo + TmpResumo.Conta + TmpResumo.Cmats
			=Seek(TmpOpera.TpOps, 'crSigOpOpt', 'Cods')

			If (crSigOpOpt.PFalhas <> 0 And TmpPro.Varias = 1 And crSigOpOpt.TFalhas <> 'Q' And TmpOpera.AplicFlhs <> 2) And Thisform.chkFalha.Value = 0 && Tiago - 15/02/2012 - Vianna - Alterado para checar flag se inclui falha no cálculo
				_FalhaAdmitida = _FalhaAdmitida + Round(TmpOpera.QtdeBSais * crSigOpOpt.PFalhas / 100, 3)
				Replace PFalha With crSigOpOpt.PFalhas, ;
						Falha  With Round(TmpOpera.QtdeBSais * crSigOpOpt.PFalhas / 100, 3) In TmpOpera
			EndIf

			If (crSigOpOpt.Trabas = 1) Or Empty(TmpOpera.TpOps)
				_Trabalhado = _Trabalhado + TmpOpera.QtdeSais
			EndIf
		EndScan

		Select TmpResumo
		Replace Saldoi  With _Saldoi, ;
				FReal   With _Saldoi + TmpResumo.QtdeEnts - TmpResumo.QtdeSais - TmpResumo.Pesagem, ;
				FAdmin  With _FalhaAdmitida, ;
				Saldof  With _Saldoi + TmpResumo.QtdeEnts - TmpResumo.QtdeSais - TmpResumo.Pesagem - _FalhaAdmitida, ;
				PfTrabs With Iif(_Trabalhado <> 0, (Saldof / _Trabalhado * 100), 0)
		
		If (TmpResumo.Saldof <> 0) And (CrSigCdCli.PagFals = 1 Or CrSigCdCli.RecFals = 1)
			Replace Flag  With .t., ;
					Flag2 With (CrSigCdCli.PagFals = 1 Or CrSigCdCli.RecFals = 1) In TmpResumo

		EndIf
	EndScan

	Select CrSaldoI
	loBarra = CreateObject( 'fwprogressbar', 'Preparando Resumos', Reccount('CrSaldoI'))
	loBarra.Top = loBarra.Top + Int(loBarra.Height/2)
	loBarra.Show
	_cMat = Space(10)
	Scan
		loBarra.Update(.t.)
		If CrSaldoI.Pesagems = 0 
			Loop
		EndIf
		=Seek(CrSaldoI.Cpros, 'TmpPro', 'CPros')
		lcSql = [Select GruEstPs,ConEstPs,nAgMts,TipoEstos,mercs From SigCdGrp Where Cgrus = ']+TmpPro.Cgrus+[']
		=ThisForm.Podatamgr.Sqlexecute(lcsql,'LocalGru')

		_Saldoi = CrSaldoI.Pesagems

		Select TmpResumo
		Locate For Grupo + Conta + cMats = _Grupo + _Conta + CrSaldoI.Cpros
		If Eof()
			Insert Into TmpResumo (Grupo, Conta, CMats, CUnis, Varias, Agregas, Visivel) ;
						   Values (LocalNens.GrupoOs, LocalNens.ContaOs, CrSaldoI.Cpros, ;
						   		   TmpPro.CUnis, TmpPro.Varias, LocalGru.nAgMts, .t.)

			Select TmpResumo
			Replace Saldoi  With _Saldoi, ;
					FReal   With _Saldoi + TmpResumo.QtdeEnts - TmpResumo.QtdeSais - TmpResumo.Pesagem, ;
					FAdmin  With 0 , ;
					Saldof  With _Saldoi + TmpResumo.QtdeEnts - TmpResumo.QtdeSais - TmpResumo.Pesagem - 0, ;
					PfTrabs With 0
		Endif
	EndScan
	loBarra.Complete
EndScan
loBarrap.Complete

Select TmpSaldo
Scan
	Select TmpResumo
	=Seek(TmpSaldo.Fase + TmpSaldo.Codigo + lcCodMat)
	Scan while Grupo + Conta + cMats = (TmpSaldo.Fase + TmpSaldo.Codigo + lcCodMat)
		Replace Saldo With Saldo + TmpResumo.Saldof in TmpSaldo
	EndScan
EndScan

Select TmpSaldo
If Thisform.chk_Saldo.Value = 1
	Delete For Saldo = 0
Endif
Set Order To Codigo
Go Top
If Eof()
	=Messagebox('Nenhum Saldo Selecionado!!!', 32)
	Return
EndIf
ThisForm.Grade.Visible = .t.
ThisForm.Visualizar.Enabled = .t.
ThisForm.Imprimir.Enabled = .t.
ThisForm.Grade.Refresh

ENDPROC
