------------------------------------------------------------
* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
With ThisForm
	.poDataMgr = CreateObject('fSqlConector', ThisForm.Name)

	If (.poDataMgr.pnIdconn > 0)

		ThisForm.Pagina.Lista.Dt_inicial.Value = Date() - goSistema.PerFiltros
		ThisForm.Pagina.Lista.Dt_Final.Value   = Date() + goSistema.PerFiltros	
		ThisForm.Pagina.Lista.Get_cd_empresa.Value = _Empr
		ldDtIni = ThisForm.Pagina.Lista.Dt_inicial.Value
		ldDtFin = ThisForm.Pagina.Lista.Dt_final.Value
		lcEmpr  = ThisForm.Pagina.Lista.Get_cd_empresa.Value
		ldDtIni = fDtoSQL(ldDtIni)
		ldDtFin= fDtoSQL(ldDtFin,'23:59:59')
		
		lcQry = [Select * From SigCdPat where Emps = ?lcEmpr And Datas Between ?ldDtIni And ?ldDtFin]

		With .poDataMgr		
			.AddCursor('SigCdPat', 'pkChaves', 'crSigCdPat', '', ThisForm.Pagina.Lista.Grade, lcQry)
			.pfSqlTabela(1).pColuna('emps', '', '', 'Empresa', 80, .t.)
			.pfSqlTabela(1).pColuna('datas', '', '', 'Data', 200, .t.)
			.pfSqlTabela(1).pColuna('codvends', '', '', 'Vendedor', 100, .t.)
			.pfSqlTabela(1).pColuna('clientes', '', '', 'Cliente', 300, .t.)
		EndWith

		Select crSigCdPat
		Index On datas tag datas
		Index On clientes tag clientes
		Set Order To datas
		=DoDefault()
		ThisForm.Pagina.Lista.get_cd_empresa.SetFocus()		
	Else
		=Messagebox('Impossivel Efetuar Conexão com Servidor de Banco de Dados...', 0+48, 'CreateObject')
		Return .f.
	EndIf
EndWith
* Fim **************************************************************************************



ENDPROC

* PROCEDURE Pagina.Lista.Grupo_op.Click
------------------------------------------------------------
PROCEDURE Pagina.Lista.Grupo_op.Click
LPARAMETERS opcao

If !dodefault( Opcao )
	ThisForm.mAtivapagina1
	Return
Endif

Select crSigCdPat

* posiciona o cursor de acordo com a opção
Do Case
	Case Inlist( ThisForm.pcEscolha, 'INSERIR', 'PROCURAR', 'ALTERAR' )
		thisform.pagina.dados.getEmps.setfocus	
	Case Inlist( ThisForm.pcEscolha, 'CONSULTAR', 'EXCLUIR' )
		thisform.pagina.dados.grupo_salva.salva.setfocus
Endcase

Select crSigCdPat
=DoDefault()

**** mostra descricao de empresa e de vendedor ****
ThisForm.Pagina.Dados.getDEmps.Value = ''
ThisForm.Pagina.Dados.getDVends.Value = ''
If InList(ThisForm.pcEscolha, 'ALTERAR', 'CONSULTAR', 'EXCLUIR')
	ThisForm.poDataMgr.CursorQuery('SigCdEmp', 'crSigCdEmp', 'Cemps', crSigCdpat.Emps)
	ThisForm.Pagina.Dados.getDEmps.Value = crSigCdEmp.Razas
	ThisForm.poDataMgr.CursorQuery('SigCdcli', 'crSigCdCli', 'iclis', crSigCdpat.codvends)
	ThisForm.Pagina.Dados.getDVends.Value = crSigCdCli.rclis
EndIf

IF ThisForm.pcEscolha='INSERIR'
	ThisForm.Pagina.Dados.getEmps.value = _empr
	ThisForm.Pagina.Dados.getDatas.value = DATETIME()
ENDIF
ThisForm.Pagina.Dados.Refresh

ENDPROC

PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
LOCAL lcQuery, lcCon, lcCod

	If IsEmpty(ThisForm.Pagina.Dados.GetClientes.Value)
		MessageBox( 'Obrigatório Informar o Cliente.', 48, '' )
		ThisForm.Pagina.Dados.GetClientes.SetFocus
		Return
	ENDIF

		MessageBox( 'Obrigatório Informar o Vendedor.', 48, '' )
		ThisForm.Pagina.Dados.GetVends.SetFocus
		Return
	ENDIF  

	Replace datas With datetime() in crSigCdPat
	Replace pkChaves With fUniqueIds() in crSigCdPat
ENDIF
SELECT crSigCdPat
=DoDefault()
ThisForm.mAtivapagina1

ENDPROC
