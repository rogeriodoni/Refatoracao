------------------------------------------------------------

------------------------------------------------------------
PROCEDURE Init
LParameters poForm, pnDataSes, pTipo
PUBLIC pcTipo
pcTipo = pTipo

ThisForm.Cnt_Planilha.cntSombra.lblSombra.Caption = ThisForm.Caption
ThisForm.Cnt_Planilha.cntSombra.lblTitulo.Caption = ThisForm.Caption

Set DataSession to pnDataSes

Thisform.ParentForm = poForm
Thisform.ParentForm.Enabled  = .T.

ThisForm.Caption = CrSigCdPro.Cpros + ' - ' + CrSigCdPro.Dpros
ThisForm.Cnt_Planilha.label1.Caption = 'Planilha de Custo ('+Iif(ptipo=[C],'Fator','Venda')+')'

Select CsGrade
Set Key To pTipo
Sum Valos to _TTA
Goto Top
With Thisform.Cnt_Planilha.GrdGrade
	.RecordSource          = 'CsGrade'
	.Column1.ControlSource = 'CsGrade.Cpros'
	.Column2.ControlSource = 'CsGrade.Dpros'
	.Column3.ControlSource = 'CsGrade.Totas'
	.Column4.ControlSource = 'CsGrade.Coefs'
	.Column5.ControlSource = 'CsGrade.Valos'
	.Column6.ControlSource = 'CsGrade.totalG'
	.Column7.ControlSource = 'CsGrade.Acumulas'
	.Refresh
ENDWITH
ThisForm.Cnt_Planilha.grdGrade.SetAll('DynamicForeColor', 'Iif(csGrade.AltCoef=1, RGB(255,0,0), RGB(0,0,0))', 'Column')

ThisForm.Cnt_Planilha.Get_Adicional.Value = _tta
ENDPROC

PROCEDURE Valid
LOCAL nReg, lnVr

IF this.Value <> lnCoef
	replace altcoef with 1 in csGrade
	replace cidChaves with fUniqueIds() in csGrade
ENDIF

reg = RECNO()
replace valos with Round((( totas * coefs ) - totas ),3) in csGrade
replace totalg with totas + valos in csGrade

GO top
lnVr = 0
SCAN
	replace acumulas with lnVr + valos + totas in csGrade
	lnVr = acumulas
ENDSCAN

Sum Valos to lnVr
thisform.cnt_Planilha.get_Adicional.Value = lnVr
GO reg

thisform.cnt_Planilha.Refresh

ENDPROC
