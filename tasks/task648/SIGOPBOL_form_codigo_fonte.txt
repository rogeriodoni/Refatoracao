* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
LParameters poForm, pnTpBol, pCursor

Set DataSession to poForm.DataSessionId

If poForm.podatamgr.sqlexecute('select * from SigCdmp order by impres','CrSigCdmp') <= 0
	Messagebox('Impossivel Efetuar Conexão com Servidor de Banco de Dados.',(0+48),'Erro')
	Return(.f.)
EndIf

lcErro = ''
llSair = .F.
With Thisform
	.vnTpBol  = Iif(Type('pnTpBol') = 'N', pnTpBol, 0)
	.vnCursor = pCursor
	Do Case
		Case .vnTpBol = 1
			.Caption = 'Impressão de Protocolo de Cheque'
			
		Case InList( .vnTpBol, 2, 3, 4, 5, 7, 9 )
			.Caption = 'Impressão de Boleto'

		Case .vnTpBol = 6
			.Caption = 'Impressão de Protocolo de Cartão de Crédito'
			
		Case .vnTpBol = 8
			.Caption = 'Impressão de Boleto de Conserto'
			
		Case .vnTpBol = 12
			.Caption = 'Impressão de Boleto de Ordem de Serviço'

		OtherWise
			.Caption = 'Impressão de Boleto (*)'
	EndCase
				
EndWith

lnPrinters = APrinters( laPrinters )
If lnPrinters # 0
	For lnI = 1 to lnPrinters
		laPrinters( lnI, 2 ) = laPrinters( lnI, 1 )
		laPrinters( lnI, 1 ) = Padr(Upper( laPrinters( lnI, 1 ) ),30)
	EndFor
	=ASort( laPrinters )
EndIf

lcSql = [Select b.* From SigSyImp a, SigCdmp b Where a.UsuAcess = ']+Usuar+[' And a.cImps = b.impres ]+;
		[Union all ]+;
		[Select c.* From SigCdAcG a, SigSyImp b, SigCdmp c Where a.Usuarios=']+Usuar+[' ]+;
		[And a.Grupos = b.GrAcess And b.Cimps = c.Impres ]

If poForm.podatamgr.sqlexecute(lcSql,'CrTmpCimp') <= 0
	Messagebox('Impossivel Efetuar Conexão com Servidor de Banco de Dados.',(0+48),'Erro')
	Return(.f.)
EndIf

Select crTmpCimp
If Reccount() > 0
	Select Distinct Impres, nTpImpres, TpBarra, MaxCols ;
	  From crTmpCimp ;
	 Order By Impres ;
	  Into Cursor crSigCdmp
EndIf

lnImpres = 1
lnCimp = 0
If lnPrinters # 0
	Select CrSigCdmp
	Scan For ( Empty( CrSigCdmp.nTpImpres ) Or ( CrSigCdmp.nTpImpres = 1 ) ) And ;
			 ( CrSigCdmp.TpBarra = 0 ) And ( CrSigCdmp.MaxCols > 39 )
		Dimension laImpOk( lnImpres )
		lnOk = AScan( laPrinters, ( Upper( CrSigCdmp.Impres ) ) )
		If lnOk # 0
			laImpOk( lnImpres ) = laPrinters( lnOk + 1 )
			lnImpres = lnImpres + 1
		EndIf
		lnCImp = lnCImp + 1
	EndScan
Else
	lcErro = 'Seu Computador Não Apresenta Impressora Configurada. Contate o Suporte Técnico.'
EndIf
lnImpres = lnImpres - 1

If lnCImp = 0
	lcErro = 'Sistema Não Apresenta Impressora(s) de Boleto Configurada(s). Contate o Suporte Técnico.'
EndIf
ThisForm.VnQtdimp = lnCimp	

If Empty(lcErro) And lnImpres # 0
	With Thisform.Opt_Impressora
		lnTop = 10
		lnHeight = 15
		.ButtonCount = lnImpres
		For lnI = 1 to lnImpres
			lcI = AllTrim( Str( lnI ) )
			With .Option&lcI.
				.AutoSize = .F.
				.Caption  = ' \<' + lcI + '. ' + laImpOk( lnI )
				.FontSize = 8
				.Tag      = Upper( laImpOk( lnI ) )
				.Top      = lnTop
				.Width    = 184
			EndWith
			lnTop = lnTop + 20
			lnHeight = lnHeight + 20
		EndFor
		.Enabled = ( lnImpres > 1 )
		.Height = lnHeight
	EndWith
	With Thisform.btnReport
		.Cmd_Imprimir.SetFocus
	EndWith
Else
	If Empty(lcErro)
		lcErro = 'Não Existe Impressora de Boleto Configurada Nesse Computador. ' + ;
				 'Contate o Suporte Técnico.'
	EndIf
	With Thisform
		.Opt_Impressora.Visible = .F.
		.Lbl_Impressora.Visible = .F.
		.Lbl_Erro.Top = .Opt_Impressora.Top
		.Lbl_Erro.Visible = .T.
		.Lbl_Erro.Caption = lcErro
		.Lbl_Erro.Height  = .Opt_Impressora.Height
		.Height = .btnReport.Top + .btnReport.Height + 20
	EndWith
EndIf

With Thisform
	With .Opt_Impressora
		lnBottom1 = .Top + .Height
	EndWith
	
	With .btnReport
		lnBottom2 = .Top + .Height
		If lnBottom1 > lnBottom2
			.Top = .Top + Int ( ( lnBottom1 - lnBottom2 ) / 2 )
			Thisform.Height = Thisform.Opt_Impressora.Top + Thisform.Opt_Impressora.Height + 20
		Else
			Thisform.Height = Thisform.btnReport.Top + Thisform.btnReport.Height + 20
		EndIf
		.Cmd_Imprimir.SetFocus
	EndWith
EndWith

ENDPROC

PROCEDURE Cmd_Imprimir.Click
*!*	Local lnImpres, lcImp, lcCursor

*!*	If Thisform.VnQtdimp > 0
*!*		With Thisform
*!*			lnImpres  = .Opt_Impressora.Value
*!*			lcImp     = .Opt_Impressora.Buttons(lnImpres).Tag
*!*			lcCursor  = .vnCursor
*!*		EndWith
*!*		Select (lcCursor)
*!*		Replace cImps With lcImp
*!*	Endif
*!*	Thisform.Release

Local lnTpBols, lcTitEnv, lcEmps, lcDopes, lnNumes, lnValDevs, lnTpVias, lnQtdBols, lcCursor

With Thisform
	lnTpBols  = .vnTpBol
	lcTitEnv  = .vcTitEnvelope
	lcEmps    = .vcEmp
	lcDopes   = .vcDope
	lnNumes   = .vnNum
	lnValDevs = .vnDev
	lnTpVias  = .vnVias
	lnQtdBols = .vnQBols
	lnImpres  = .Opt_Impressora.Value
	lcImp     = .Opt_Impressora.Buttons(lnImpres).Tag
	lcCursor  = .vnCursor
EndWith

Do SigBlBol With lnTpBols, lcTitEnv, lcEmps, lcDopes, lnNumes, lnValDevs, lnTpVias, lnQtdBols, lcImp, lcCursor

* Parâmetros :
* 01. nTpbol, Tipo de Boleto
* 02. TitEnvelope, Título
* 03. pEmp, Empresa
* 04. pDop, Nome da Operação de Estoque / Operação de Produção
* 05. pNum, Número da Movimentação de Estoque / Movimentação de Produção
* 06. pDev, Valor de Devolução
* 07. pVias, Número da Via
* 08. qBols, Quantidade de Boletos
* 09. Nome da Impressora
* 10. lcCursor, Nome do Cursor ou Tabela Para Impressão

Thisform.Release

ENDPROC
