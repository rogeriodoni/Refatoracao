PROCEDURE Init
LParameters pform,datses

ThisForm.cntSombra.lblSombra.Caption = ThisForm.Caption
ThisForm.cntSombra.lblTitulo.Caption = ThisForm.Caption

Set Datasession To DatSes
Thisform.parentform = pform
ThisForm.poDataMgr = CreateObject('fSqlConector', ThisForm.Name)
	
lcSql = [Select a.cDopes, a.cnivels, a.nCors, a.padraos, a.Mercs, a.opers, b.abrevs, b.CadProds, b.Desmembs from SigCdMop a, SigCdOpe b, SigOpCdc c, SigOpCdi d ]+;
	[Where a.cDopes = ']+TprMvCab.Dopes+[' And a.cnivels = b.dopes And b.Abrevs <> ' ' And ]+;
	[b.Dopes = c.Dopes And b.Dopes = d.Dopes And c.OpeInatvs <> 1 ]
If Thisform.PodataMgr.SqlExecute(lcSql,'CsTmpPed') < 1
	=Messagebox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (CsTmpPed)')
	Return 0
Endif
Select CsTmpPed
Index On nCors  Tag nCors
Index On Abrevs Tag Abrevs

Select crTpmMvItn
lcCod = crTpmMvItn.Cpros
lcCor = crTpmMvItn.CodCors
lcTam = crTpmMvItn.CodTams

Select * from crTpmMvItn Where Cpros = lcCod And CodCors = lcCor And CodTams = lcTam And Empty(Abrevis) Into Cursor csItemAbr ReadWrite

Select csItemAbr
Go Top
Count For Not Empty(Cpros) to lnQtde

ThisForm.getQtde.Value = lnQtde
ThisForm.getQtde.Refresh

ThisForm.get_Peso.Value = crTpmMvItn.Pesos
ThisForm.get_Peso.Refresh

lnOldSel = Select()

ThisForm.Say7.Visible = .F.
ThisForm.getTamanho.Visible = .F.
ThisForm.getTamanho.Value = lcTam

If (Thisform.PodataMgr.SqlExecute([Select a.* From SigCdUni a join sigcdpro b on a.Cunis = b.Cunis Where b.Cpros = '] + lcCod + [' ], 'crTmpUni') < 1)
	=Messagebox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (crTmpUni)')
	Return 0
Endif

If (Thisform.PodataMgr.SqlExecute([Select a.* From SigCdGrp a Join SigCdPro b on a.CGrus = b.CGrus Where b.Cpros = '] + lcCod + [' ], 'crTmpGru') < 1)
	=Messagebox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (crTmpGru)')
	Return 0
Endif

If Empty(lcTam)
	Select crTmpGru	
	If Inlist(crTmpGru.TipoEstos,3,4)
		ThisForm.Say7.Visible = .T.
		ThisForm.getTamanho.Visible = .T.
	Endif
EndIf 
Select(lnOldSel)


ThisForm.Refresh

*!*	* Fim **************************************************************************************
ENDPROC

PROCEDURE btnGravar.Click
Local lcPro, lcTam, lcCor, lcAbr, lnQtd, lnRep, lnRecc

lcPro = ThisForm.get_prod.Value
lcTam = ThisForm.getTam.Value
lcCor = ThisForm.getCor.Value
lcAbr = ThisForm.GetPedido.Value
lnQtd = ThisForm.GetQtde.Value
lnRep = ThisForm.getRepetir.Value
lnPes = ThisForm.Get_Peso.Value

lcTamanho = ThisForm.getTamanho.Value

lnRecc = 0

_CarregaItens = Inlist( crSigCdOpe.ItAutos,1,3,4,5 ) And Thisform.parentform.Op_escolha = 'INSERIR'
_ChecaQtde    = Inlist( crSigCdOpe.ChkQtdss,1,3,4 )
_ChecaOper    = ( Empty(crSigOpCdc.ChkOpers) Or crTpmMvItn.Opers = crSigOpCdc.ChkOpers )
_ValidaGrade  = Thisform.parentform.Op_escolha = 'INSERIR' And crSigOpCdi.ValQtdSup = 1

Select TmpOperacao
_SubNiv = .F.
Scan For !IsEmpty( Codigos )
	_SubNiv = .T.
EndScan

Select csItemAbr
lnRecc = Reccount([csItemAbr])

If lnRecc > 0

	For I=1 To lnRep
		Select crTpmMvItn
		Select cItens from crTpmMvItn Where Cpros = lcPro And CodCors = lcCor And CodTams = lcTam And Empty(Abrevis) Into Cursor csItemAbr ReadWrite
		If Reccount([csItemAbr]) > 0
		
			lnItem = csItemAbr.cItens
			
			Select crTpmMvItn
			Update crTpmMvItn Set Abrevis = lcAbr, Pesos = lnPes, CodTams = lcTamanho Where cItens = lnItem 
			
			Select crTpmMvItn
			Select * from crTpmMvItn Where cItens = lnItem And cItem2 = 0 Into Cursor csTmpItn ReadWrite 
			
			Select crTpmMvItn
			
			If Used([CsTmpPed]) And !Empty(CsTmpPed.Opers)	
				Update crTpmMvItn Set opers = CsTmpPed.opers Where citem2=lnItem 
			Endif
			
			If crTpmMvItn.citem2=0 And Reccount('crTpmMvItn')>0
				Select crTpmMvItn
				Update crTpmMvItn Set AbrevIs = lcAbr Where citem2 = lnItem
			EndIf 
			
*!*				If !Empty(lcTamanho)
*!*					Select csTmpItn
*!*					If  ( csTmpItn.Qtds > 1 ) And ( crTmpUni.Desmembs = 'S' ) And ( CsTmpPed.Desmembs = 1 )
*!*						For i=1 To csTmpItn.Qtds
*!*							Insert Into crTpmMvIts ( CItens, CPros, Emps, Dopes, Numes, Qtds, QtdEmbs, Pesos, CodTams, CodBarras ) ;
*!*								Values ( csTmpItn.CItens, csTmpItn.CPros, csTmpItn.Emps, csTmpItn.Dopes, csTmpItn.Numes, 1, 1, csTmpItn.Pesos, csTmpItn.CodTams, csTmpItn.codbarras )
*!*						Endfor
*!*					Else
*!*						Insert Into crTpmMvIts ( CItens, CPros, Emps, Dopes, Numes, Qtds, QtdEmbs, Pesos, CodTams, CodBarras ) ;
*!*							Values ( csTmpItn.CItens, csTmpItn.CPros, csTmpItn.Emps, csTmpItn.Dopes, csTmpItn.Numes, csTmpItn.Qtds, 1, csTmpItn.Pesos, csTmpItn.CodTams, csTmpItn.codbarras )
*!*					Endif
*!*				EndIf
			
		EndIf 
	EndFor 
EndIf 
ThisForm.ParentForm.Enabled = .t.

ThisForm.Release
* Fim **************************************************************************************
ENDPROC

PROCEDURE Valid
If Not Empty(This.Value)
	Select CsTmpPed
	If Not Seek( This.Value, 'CsTmpPed', 'Abrevs' )
		loBrowse = CreateObject( 'fwbuscasel', 'CsTmpPed', 'Abrevs', This.Value, 'Seleção')

		This.Value = CsTmpPed.Abrevs
	EndIf
	Select crTpmMvItn
	This.Refresh
	If Type('This.Value') <> Type('ThisForm.AntValu') Or This.Value <> ThisForm.AntValu
		Replace UsuLibSal With ' ' in crTpmMvItn
	EndIf
	If Not ThisForm.parentform.ValidaEstoque(CsPedidos.cNivels)
		This.Value = ''
		This.Refresh
		Return .f.
	EndIf
	Replace ChkAbrevs With .f. in crTpmMvItn
Else
	ThisForm.parentform.Calc_valite()
EndIf
This.Refresh
ENDPROC

PROCEDURE Valid
Local loLista
If Not Empty(this.Value)
	loLista = CreateObject('fwbuscaext', Thisform.PodataMgr.pnIdConn, 'SigCdTam', ;
						   'crListaRemota', 'Cods', Padr(this.Value,4), 'Seleção', .T.)
	If Not loLista.plAchouRegistro
		loLista.mAddColuna('Cods', '', 'Código',.t.,'Cods')
		loLista.mAddColuna('Descs', '', 'Descrição',.t.,'Descs')
		loLista.Show()
		This.Value = CrListaRemota.Cods
	EndIf
Else
	This.Value = ''
EndIf 	

This.Refresh
ENDPROC
