PROCEDURE consulta.Click
Local lcQuery, lcMoe

ThisForm.Enabled = .f.

Zap In TmpOper

lcQuery = [Select MoedaPs,MascNums ] + ;
		    [From SigCdPam ] + ;
		   [Where Not cIdChaves = '] + fUniqueIds() + [']

If (ThisForm.poDataMgr.SqlExecute(lcQuery, 'crSigCdPam') < 1)
	=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (crSigCdPam)')
	Return 0
EndIf

Wait Window 'Aguarde!!! Selecionando Destino...' NoWait

lcQuery = [Select a.Emps, a.Dopes, a.Numes, a.Datas, a.Valos, b.CMoes ] + ;
		    [From SigMvCab a, SigCdOpe b ] + ;
		   [Where a.GrupoDs = '] + Padr(ThisForm.Origem.get_grupo.Value, 10) + [' And ] + ;
		   		 [a.ContaDs = '] + Padr(ThisForm.Origem.get_conta.Value, 10) + [' And ] + ;
		   		 [a.Dopes = b.Dopes]

If (ThisForm.poDataMgr.SqlExecute(lcQuery, 'crSigMvCab') < 1)
	=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (crSigMvCab - Destino)')
	Return 0
EndIf

Select crSigMvCab
Scan
	lcMoe = Iif(Empty(crSigMvCab.CMoes), crSigCdPam.MoedaPs, crSigMvCab.CMoes)
	
	Insert Into TmpOper (Emps, Dopes, MascNum, Tipo, Datas, Valos, Moeda) ;
				 Values (crSigMvCab.Emps, crSigMvCab.Dopes, Alltrim(fGerMascara(crSigMvCab.Numes)), ;
				 		 'Destino', crSigMvCab.Datas, crSigMvCab.Valos, lcMoe)
EndScan

Wait Window 'Aguarde!!! Selecionando Origem...' NoWait

lcQuery = [Select a.Emps, a.Dopes, a.Numes, a.Datas, a.Valos, b.CMoes ] + ;
		    [From SigMvCab a, SigCdOpe b ] + ;
		   [Where a.GrupoOs = '] + Padr(ThisForm.Origem.get_grupo.Value, 10) + [' And ] + ;
		   		 [a.ContaOs = '] + Padr(ThisForm.Origem.get_conta.Value, 10) + [' And ] + ;
		   		 [a.Dopes = b.Dopes]

If (ThisForm.poDataMgr.SqlExecute(lcQuery, 'crSigMvCab') < 1)
	=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (crSigMvCab - Origem)')
	Return 0
EndIf

Select crSigMvCab
Scan
	lcMoe = Iif(Empty(crSigMvCab.CMoes), crSigCdPam.MoedaPs, crSigMvCab.CMoes)
	
	Insert Into TmpOper (Emps, Dopes, MascNum, Tipo, Datas, Valos, Moeda) ;
				 Values (crSigMvCab.Emps, crSigMvCab.Dopes, Alltrim(fGerMascara(crSigMvCab.Numes)), ;
				 		 'Origem', crSigMvCab.Datas, crSigMvCab.Valos, lcMoe)
EndScan

Wait Clear

Select TmpOper
Set Order To Num
Go Top
If Eof()
	=MessageBox('Nenhuma Movimentação Foi Encontrada!!!', 0+48, '')
	ThisForm.Enabled = .t.
	ThisForm.Origem.Get_Grupo.SetFocus
Else
	Do Form SigOpClp with ThisForm,ThisForm.datasessionid
EndIf
* Fim **************************************************************************************

ENDPROC

			This.Parent.get_grupo.Value = crSigCdCli.Grupos
		EndIf
		This.Parent.get_cpf.Value = crSigCdCli.Cpfs
	EndIf

	This.Parent.get_cpf.Value  = ''
EndIf

PROCEDURE Valid
Local loLista

If Not Empty(This.Value)
	lnVer = 0
	lcCgc = StrTran(StrTran(StrTran(This.Value,'.',''),'-',''),'/','')
	If (Len(Alltrim(lcCgc)) <> 14)
		lcCgc1 = TransForm(lcCgc,'@R 999.999.999-99')
		If (Len(Alltrim(lcCgc)) = 11)
			lnVer = Iif(fValidarCPF(lcCgc1),1,2)
		EndIf
	Else
		lcCgc1 = TransForm(lcCgc,'@R 99.999.999/9999-99')
		lnVer  = Iif(fValidarCNPJ(lcCgc1),1,2)
	EndIf
	
	If (lnVer = 2)
		=MessageBox('CPF/CNPJ Incorreto!!!', 48, '')
		Return 0
	EndIf

	This.Value = lcCgc1
	This.Refresh
	
	loLista = CreateObject('fwBuscaExt', ThisForm.poDataMgr.pnIdConn, 'SigCdCli', ;
						   'crListaRemota', 'Cpfs', This.Value, 'Seleção', 1000)

	If Not loLista.plAchouRegistro
		loLista.mAddColuna('Cpfs',  '', 'CPF/CNPJ')
		loLista.mAddColuna('IClis', '', 'Código')
		loLista.mAddColuna('RClis', '', 'Descrição')
		loLista.mAddColuna('Inativas ', .f., .f., .f., .f., .f., .t.)
		loLista.gridProc.SetAll('DynamicForeColor', [Iif(crListaRemota.Inativas == 1, Rgb(255, 0, 0), Rgb(0,0,0))], 'Column')
		
		loLista.Show()
	EndIf
	
	This.Value = crListaRemota.Cpfs
	This.Parent.Get_Conta.Value  = crListaRemota.IClis
	This.Parent.Get_DConta.Value = crListaRemota.RClis
	Use In crListaRemota
Else
	This.Value = ''
	This.Parent.Get_Conta.Value  = ''
	This.Parent.Get_DConta.Value = ''
EndIf
This.Refresh
This.Parent.Get_Conta.Refresh
This.Parent.Get_DConta.Refresh
* Fim **************************************************************************************

ENDPROC
