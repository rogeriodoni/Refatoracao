PROCEDURE msv_procurar
Select crSigIdPcp
Set Near On
Index On tipos+grupos tag TpGru
Set Order To TpGru
Seek Thisform.pagina.dados.get_tipos.Value+Thisform.pagina.dados.get_Grupo.Value
Set Near Off
=DoDefault()
ENDPROC

------------------------------------------------------------
* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
lcQryTpCp = [Select * From SigIdPcp ]
With ThisForm
	.poDataMgr = CreateObject('fSqlConector', ThisForm.Name)

	If (.poDataMgr.pnIdconn > 0)
		With .poDataMgr
			.AddCursor('SigIdPcp', 'cIdChaves', 'crSigIdPcp', '', ThisForm.Pagina.Lista.Grade, lcQryTpCp)
			.pfSqlTabela(1).pColuna('Tipos', '', '', 'Código', 30, .t.)
			.pfSqlTabela(1).pColuna('descs', '', '', 'Descrição', 80, .t.)
			.pfSqlTabela(1).pColuna('limites', '', '', 'Prod. Mínima', 105, .t.)
		EndWith

		=DoDefault()

		lcQuery = [Select codigos,Descrs ] + ;
		    [ From SigCdGcr] + ;
		    [ Where gerbals=1 ] + ;
		    [ Order By codigos,Descrs ]
		If (ThisForm.poDataMgr.SqlExecute(lcQuery, 'crSigCdGcr') < 1)
			=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (crSigCdGcr)')
			Return 0
		EndIf

		Select crSigIdPcp

		ThisForm.Pagina.Lista.Grade.Column1.SetFocus
	Else
		=Messagebox('Impossivel Efetuar Conexão com Servidor de Banco de Dados...', 0+48, 'CreateObject')
		Return .f.
	EndIf
EndWith
* Fim **************************************************************************************


ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Lista.Grupo_op.Click
------------------------------------------------------------
PROCEDURE Pagina.Lista.Grupo_op.Click
LPARAMETERS opcao
If !dodefault( Opcao )
	ThisForm.mAtivapagina1
	Return
Endif

Select crSigIdPcp

* posiciona o cursor de acordo com a opção
Do Case
	Case Inlist( ThisForm.pcEscolha, 'INSERIR', 'PROCURAR' )
		thisform.pagina.dados.get_Tipos.setfocus	
	Case ThisForm.pcEscolha = 'ALTERAR'
		thisform.pagina.dados.get_descs.setfocus
	Case Inlist( ThisForm.pcEscolha, 'CONSULTAR', 'EXCLUIR' )
		thisform.pagina.dados.grupo_salva.salva.setfocus
EndCase

If ThisForm.poDataMgr.CursorQuery('SigCdMoe', 'crSigCdMoe', 'CMoes', crSigIdPcp.Moedas)
	thisform.pagina.dados.get_Desc_Moe.value = crSigCdMoe.Dmoes
else
	thisform.pagina.dados.get_Desc_Moe.value = ''
EndIf

ThisForm.Pagina.Dados.Refresh
ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
If ThisForm.pcEscolha = 'INSERIR'
	IF IsEmpty(ThisForm.Pagina.Dados.Get_Tipos.Value)
    	MessageBox( 'Obrigatório Informar o Código...', 48, '' )
		ThisForm.Pagina.Dados.Get_Tipos.SetFocus
		Return
	Endif
	IF IsEmpty(ThisForm.Pagina.Dados.Get_Grupo.Value)
    	MessageBox( 'Obrigatório Informar o Grupo...', 48, '' )
		ThisForm.Pagina.Dados.Get_Grupo.SetFocus
		Return
	EndIf
	lcQuery = [Select Tipos,Grupos ] + ;
		    [ From SigIdPcp] + ;
		    [ Where Tipos=']+ThisForm.pagina.dados.get_tipos.Value+[']+;
		    [ and  Grupos=']+ThisForm.pagina.dados.get_grupo.Value+[']
	If (ThisForm.poDataMgr.SqlExecute(lcQuery, 'crPesqSigIdPcp') < 1)
		=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (crPesqSigIdPcp)')
		Return 0
	EndIf
	Select crPesqSigIdPcp
	Go Top
	If not Eof()
		MessageBox( 'Código já cadastrado...', 48, '' )
		ThisForm.Pagina.Dados.Get_Grupo.SetFocus
		Return
	EndIf
	Replace cIdChaves With fUniqueIds() in crSigIdPcp
Endif
=DoDefault()
ThisForm.mAtivapagina1

ENDPROC
