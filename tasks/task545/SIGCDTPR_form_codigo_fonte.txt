* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
Lparameters pTip

Local lcQryTron

If (Type('pTip')<>'C')
	=Messagebox('O Parâmetro Passado Não é Válido!!!', 0+48, 'Init - SigPrTro')
	Return .f.
EndIf

ThisForm.TpTrans = Padr(pTip, 6)

lcQryTron = [Select * From SigPrTro Where TpTrans = '] + ThisForm.TpTrans + [' Order BY RecTrans,Tps,ordems]

With ThisForm
	.poDataMgr = CreateObject('fSqlConector', ThisForm.Name)

	If (.poDataMgr.pnIdconn > 0)
		With .poDataMgr
			.AddCursor('SigPrTro', 'cIdChaves', 'crSigPrTro', '', ThisForm.Pagina.Lista.Grade, lcQryTron)
			.pfSqlTabela(1).pColuna('Tps',     '', '', 'Cod', 	     031, .f.)
			.pfSqlTabela(1).pColuna('RecTrans','', '', 'T/R', 	     031, .f.)
			.pfSqlTabela(1).pColuna('Dbf',     '', '', 'Tabela',    066, .f.)
			.pfSqlTabela(1).pColuna('Obs',     '', '', 'Descrição', 275, .f.)
			.pfSqlTabela(1).pColuna('CProcs',  '', '', 'Processo',  105, .f.)
			.pfSqlTabela(1).pColuna('Indice',  '', '', 'Índice',    115, .f.)
		EndWith

		=DoDefault()

		ThisForm.Pagina.Lista.Grade.Column1.SetFocus
	Else
		=Messagebox('Impossível Efetuar Conexão Com o Servidor de Banco de Dados...', 0+48, 'CreateObject')
		Return .f.
	EndIf
EndWith

ENDPROC

* PROCEDURE Pagina.Lista.Grupo_op.Click
------------------------------------------------------------
PROCEDURE Pagina.Lista.Grupo_op.Click
Lparameters Opcao

If (Type('Opcao') = 'N')
	This.Value = Opcao
EndIf

Select crSigPrTro

If Not DoDefault(Opcao)
	ThisForm.mAtivapagina1
	Return 0
EndIf

If (ThisForm.pcEscolha = 'INSERIR')
	Replace TpTrans With ThisForm.Tptrans In crSigPrTro
	ThisForm.Pagina.Dados.getTpTrans.Refresh 
EndIf

If Inlist(ThisForm.pcEscolha, 'CONSULTAR', 'EXCLUIR')
	ThisForm.Pagina.Dados.Grupo_Salva.Salva.Setfocus
Else
	ThisForm.Pagina.Dados.getObs.Setfocus
EndIf

Select crSigPrTro

ThisForm.Pagina.Dados.getTpTrans.Enabled = .f.

ThisForm.Pagina.Dados.Refresh

ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
If InList(ThisForm.pcEscolha, 'INSERIR', 'ALTERAR')
	If Empty(ThisForm.Pagina.Dados.getObs.Value)
		=MessageBox('Obrigatório Informar a Descrição do Transporte!!!', 48, '')
		ThisForm.Pagina.Dados.getObs.SetFocus
		Return 0
	EndIf

	If Empty(ThisForm.Pagina.Dados.getTps.Value)
		=MessageBox('Obrigatório Informar o Código do Transporte!!!', 48, '')
		ThisForm.Pagina.Dados.getTps.SetFocus
		Return 0
	EndIf

	If Empty(ThisForm.Pagina.Dados.getRecTrans.Value)
		=MessageBox('Obrigatório Informar o Tipo do Transporte!!!', 48, '')
		ThisForm.Pagina.Dados.getRecTrans.SetFocus
		Return 0
	EndIf

	If Not InList(ThisForm.Pagina.Dados.getRecTrans.Value, 'T', 'R')
		=MessageBox('O Tipo do Transporte Tem Que Ser T ou R!!!', 48, '')
		ThisForm.Pagina.Dados.getRecTrans.SetFocus
		Return 0
	EndIf

	If Empty(ThisForm.Pagina.Dados.getDbf.Value)
		=MessageBox('Obrigatório Informar a Tabela do Transporte!!!', 48, '')
		ThisForm.Pagina.Dados.getDbf.SetFocus
		Return 0
	EndIf

	If (ThisForm.pcEscolha = 'INSERIR')
		Replace cIdChaves With fUniqueIds(), ;
				TpTrans   With ThisForm.TpTrans In crSigPrTro
	EndIf

	lcQuery = [Select * ] + ;
				[From SigPrTro ] + ;
			   [Where TpTrans = '] + ThisForm.TpTrans + [' And ] + ;
			   		 [Tps = '] + crSigPrTro.Tps + [' And ] + ;
			   		 [RecTrans = '] + crSigPrTro.RecTrans + [' And ] + ;
			   		 [Cprocs = '] + CrSigPrTro.cprocs + [' And ] +;
			   		 [Not cIdChaves = '] + crSigPrTro.cIdChaves + [']

	If (ThisForm.poDataMgr.SqlExecute(lcQuery, 'LocalTron') < 1)
		=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (LocalTron)')
		Return 0
	EndIf

	If (Reccount('LocalTron') > 0)
		=MessageBox('Código do Transporte Já Cadastrado!!!', 48, '')
		ThisForm.Pagina.Dados.getTps.SetFocus
		Return 0
	EndIf
EndIf

=DoDefault()

ThisForm.mAtivaPagina1

ENDPROC

			'TRANSFERE', 'SITUACAO', 'CHDEVOL', 'BALANCO', 'CPF', 'ETIQUETA','COMPLEMENTO',;

		=MessageBox('Os Processos Válidos são: COPIA, PRODUTO, CLIENTE, PROETI, TRANSFERE, ' + Chr(13) + ;

					'EMPCLIENTE,DATATRANS,APAGA,SIGMVHST,SIGMVCCR,SALDO,TRANSFEREC,CADCLIENTES', 0+16, 'Atenção!!!')
		Return 0
	EndIf
ELSE
	If Not Empty(This.Value) And Not InList(This.Value, 'INSERIR', 'COPIA', 'APAGA', 'ATUALIZA', 'TRANSFERE',;
		 'BAIXA VALPAGS', 'SITUCHDEV', 'INTERATIVO', 'CURSOR', 'COMANDO','ETIQUETA','EXCLUSIVO','TRANSF ETQ','ATUACPF')
		=MessageBox('Os Processos Válidos são: INSERIR,COPIA,APAGA,ATUALIZA,TRANSFERE,TRANSF ETQ, ' + Chr(13) + ;
					'BAIXA VALPAGS,SITUCHDEV,ETIQUETA,INTERATIVO,EXCLUSIVO,CURSOR,COMANDO ', 0+16, 'Atenção!!!')
		Return 0
	EndIf
EndIf

If InList(This.Value,'COMPLEMENTO','COMPLECLI') 
	If  Asc(This.Parent.GetTps.Value) < 65 Or Asc(This.Value) > 90
		MessageBox('Código do Transporte deve ser Letra',16,'')
		Return .f.
	Endif
EndIf

If This.Value = 'ATUACPF' And This.Parent.GetDbf.Value <> 'SigCdCli'
	MessageBox('Tipo AtuaCpf é utilizado somente para o Arquivo SigCdCli !!!',16,'')
	Return .f.
EndIf
