* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
With ThisForm
	.poDataMgr = CreateObject([fSqlConector], .Name)

	If (.poDataMgr.pnIdconn > 0)
		With .poDataMgr
			.AddCursor([SigCdTpc], [Tipos], [crSigCdTpc], [], ThisForm.Pagina.Lista.Grade)
			.pfSqlTabela(1).pColuna([Tipos],  [], [], [Tipos],  150, .t.)
			.pfSqlTabela(1).pColuna([Descrs], [], [], [Descrs], 290, .t.)

			.AddCursor([SigCdCom], [cIdChaves], [crSigCdCom])
		EndWith

		=DoDefault()

		.Pagina.Lista.Grade.Column1.SetFocus
	Else
		=Messagebox([Impossível Efetuar Conexão Com o Servidor de Banco de Dados...], 0+48, [CreateObject])
		Return .f.
	EndIf
EndWith
* Fim **************************************************************************************
ENDPROC

------------------------------------------------------------
* PROCEDURE pagina.lista.grupo_op.Click
------------------------------------------------------------
PROCEDURE pagina.lista.grupo_op.Click
LParameters lnOpcao

If (Type([lnOpcao]) = [N])
	This.Value = lnOpcao
EndIf

Select crSigCdTpc

If Not DoDefault(lnOpcao)
	ThisForm.mAtivaPagina1
	Return .f.
EndIf

Zap In crGrdCdCom

lcQuery = [Select * ] + ;
		    [From SigCdCom ] + ;
		   [Where Tipos = '] + crSigCdTpc.Tipos + [' ] + ;
		   [Order By CGrus]

If (ThisForm.poDataMgr.SqlExecute(lcQuery, [LocalCdCom]) < 1)
	=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (LocalCdCom)])
	Return .f.
EndIf

Select LocalCdCom
Scan
	lcQuery = [Select DGrus ] + ;
			    [From SigCdGrp ] + ;
			   [Where CGrus = '] + LocalCdCom.CGrus + [']

	If (ThisForm.poDataMgr.SqlExecute(lcQuery, [LocalGru]) < 1)
		=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (LocalGru)])
		Return .f.
	EndIf
	Go Top In LocalGru

	Insert Into crGrdCdCom (CGrus, DGrus) Values (LocalCdCom.CGrus, LocalGru.DGrus)
EndScan
Go Top In crGrdCdCom
If Eof([crGrdCdCom]) And InList(ThisForm.pcEscolha, [INSERIR], [ALTERAR])
	Append Blank In crGrdCdCom
	Go Top In crGrdCdCom
EndIf

loDad = ThisForm.Pagina.Dados

Do Case
	Case Inlist(ThisForm.pcEscolha, [PROCURAR], [INSERIR])
		loDad.getTipos.Setfocus	

	Case (ThisForm.pcEscolha = [ALTERAR])
		loDad.getDescrs.Setfocus	

	Case Inlist(ThisForm.pcEscolha, [CONSULTAR], [EXCLUIR])
		loDad.Grupo_Salva.Salva.Setfocus
EndCase

Select crSigCdTpc

loDad.Refresh
* Fim **************************************************************************************

------------------------------------------------------------
* PROCEDURE pagina.dados.grupo_salva.salva.Click
------------------------------------------------------------
PROCEDURE pagina.dados.grupo_salva.salva.Click
Local loDad, lcPro

loDad = ThisForm.Pagina.Dados

Select crSigCdTpc

If (ThisForm.pcEscolha = [INSERIR])
	If Empty(loDad.getTipos.Value)
		=MessageBox([Código de Tipo Não Informado...], 0+48, [Atenção!!!])
		loDad.getTipos.SetFocus
		Return .f.
	EndIf

	lcQuery = [Select * ] + ;
			    [From SigCdTpc ] + ;
			   [Where Tipos = '] + loDad.getTipos.Value + [']

	If (ThisForm.poDataMgr.SqlExecute(lcQuery, [LocalTpCom]) < 1)
		=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (LocalTpCom)])
		loDad.getTipos.SetFocus
		Return .f.
	EndIf
	Select LocalTpCom
	Go Top In LocalTpCom
	If Not Eof([LocalTpCom])
		=MessageBox([Código de Tipo Já Cadastrado...], 0+48, [Atenção!!!])
		loDad.getTipos.SetFocus
		Return .f.
	EndIf
EndIf

If InList(ThisForm.pcEscolha, [INSERIR], [ALTERAR])
	If Empty(crSigCdTpc.Descrs)
		=MessageBox([Descrição Inválida!!!], 16, [Atenção!!!])
		ThisForm.Pagina.Dados.getDescrs.SetFocus
		Return .f.
	EndIf
EndIf

If InList(ThisForm.pcEscolha, [EXCLUIR], [ALTERAR])
	lcQuery = [Delete From SigCdCom ] + ;
			   [Where Tipos = '] + crSigCdTpc.Tipos + [']

	If (ThisForm.poDataMgr.SqlExecute(lcQuery, []) < 1)
		=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (Delete - LocalCdCom)])
		Return .f.
	EndIf
EndIf

Zap In crSigCdCom

Select Distinct CGrus ;
  From crGrdCdCom ;
 Where Not Empty(CGrus) ;
 Order By CGrus ;
  Into Cursor crGravar

Select crGravar
Scan
	Insert Into crSigCdCom (Tipos, CGrus, cIdChaves) ;
					Values (crSigCdTpc.Tipos, crGravar.CGrus, fUniqueIds())
EndScan
Go Top In crSigCdCom

Select crSigCdTpc

=DoDefault()

ThisForm.mAtivaPagina1
* Fim **************************************************************************************
ENDPROC
