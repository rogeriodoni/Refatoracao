------------------------------------------------------------
* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
Lparameters pGru, pSgr, pCar, pTit, pEsc, pTipo

Private pKey

ThisForm.Grupos  = Padr(pGru, Iif(pTipo = 0, 3, 20 ))
ThisForm.SubGrp  = Padr(pSgr, 06)
ThisForm.Cargos  = Padr(pCar, 20)
ThisForm.Titulos = Padr(pTit, 20)

pKey = ThisForm.Grupos + ThisForm.Cargos + ThisForm.Titulos + ThisForm.SubGrp 

lcQryCFTeR = [Select * From SigCdFtr Where GruCarTits = ?pKey Order By cIdChaves]

With ThisForm
	.poDataMgr = CreateObject('fSqlConector', ThisForm.Name)

	If (.poDataMgr.pnIdconn > 0)
		With .poDataMgr
			.AddCursor('SigCdFtr', 'cIdChaves', 'crSigCdFtr', '', ThisForm.Pagina.Lista.Grade, lcQryCFTeR)
			.pfSqlTabela(1).pColuna('Resps', '', '', 'Respostas', ThisForm.Pagina.Lista.Grade.Width - 65)
			.pfSqlTabela(1).pColuna('nResps','', '', 'Fixa', 31 )
		EndWith

		ThisForm.Pagina.Lista.Grade.Column2.ControlSource = [Iif(crSigCdFtr.nResps = 1, 'Sim', 'Não')] 

		.plAcInserir = InList(pEsc, [INSERIR], [ALTERAR])
		.plAcAlterar = InList(pEsc, [INSERIR], [ALTERAR])
		.plAcExcluir = InList(pEsc, [INSERIR], [ALTERAR])

		=DoDefault()

		.Caption = [Ficha Técnica - ] + Alltrim(.Cargos) + [ - ] + Alltrim(.Titulos)

		.Pagina.Lista.Grade.Column1.SetFocus
	Else
		=Messagebox('Impossível Efetuar Conexão Com o Servidor de Banco de Dados...', 0+48, 'CreateObject')
		Return .f.
	EndIf
EndWith
* Fim **************************************************************************************
ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.lista.grupo_op.Click
------------------------------------------------------------
PROCEDURE Pagina.lista.grupo_op.Click
LParameters lnOpcao

If (Type('lnOpcao') = 'N')
	This.Value = lnOpcao
EndIf

Select crSigCdFtr

If Not DoDefault(lnOpcao)
	ThisForm.mAtivaPagina1
	Return 0
EndIf

loDad = ThisForm.Pagina.Dados

If (ThisForm.pcEscolha = 'INSERIR')
	Replace CGrus 		With ThisForm.Grupos, ;
			SGrus 		With ThisForm.SubGrp, ;
			CCars 		With ThisForm.Cargos, ;
			CTits 		With ThisForm.Titulos, ;
			GruCarTits 	With ThisForm.Grupos + ThisForm.Cargos + ThisForm.Titulos + ThisForm.SubGrp, ;
			cIdChaves 	With fUniqueIds() In crSigCdFtr
EndIf

If Inlist(ThisForm.pcEscolha, 'CONSULTAR', 'EXCLUIR')
	loDad.Grupo_Salva.Salva.Setfocus
Else
	loDad.getResps.Setfocus	
EndIf

loDad.Refresh
* Fim **************************************************************************************
ENDPROC
