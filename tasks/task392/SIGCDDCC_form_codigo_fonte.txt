PROCEDURE Init
LParameters poForm, pnDataSes, pEmpresa

Set DataSession to pnDataSes

Thisform.ParentForm          = poForm
Thisform.ParentForm.Enabled  = .F.
ThisForm.pcescolha = ThisForm.parentform.pcEscolha
ThisForm.Podatamgr = ThisForm.ParentForm.Podatamgr
ThisForm.pEmps     = Iif(Type('pEmpresa') = [C],pEmpresa,_Empr)

With Thisform
	.cmdInserir.Visible = fChecaAcesso('SigMvEtd','INSERIR')
	.cmdInserir.Enabled = fChecaAcesso('SigMvEtd','INSERIR') And InList(ThisForm.pcEscolha,'INSERIR','ALTERAR')
	.cmdExcluir.Visible = fChecaAcesso('SigMvEtd','EXCLUIR')
	.cmdExcluir.Enabled = fChecaAcesso('SigMvEtd','EXCLUIR') And InList(ThisForm.pcEscolha,'INSERIR','ALTERAR')
EndWith

Select [] as agrupar,Sum(Valos) as valos From xEestDsc group by 1 Into cursor selecao
ThisForm.Get_total.Value = Selecao.Valos
ThisForm.Get_total.Refresh

Select xEestDsc
Go Top
With Thisform.GradeFPagto
	.RecordSource          = 'XEestDsc'
	.Column1.ControlSource = 'xEestDsc.MotDscs'
	.Column2.ControlSource = 'xEestDsc.Descs'
	.Column3.ControlSource = 'xEestDsc.Valos'
	.Column4.ControlSource = 'xEestDsc.Obs'
	.Readonly = Not InList(ThisForm.pcEscolha,'INSERIR','ALTERAR')
	.Refresh
EndWith

THisForm.GradeFPagto.Refresh
ENDPROC

PROCEDURE Valid
If Not IsEmpty(This.Value)
	lcSql = [Select Nascs, DtnCons From SigCdCli Where Iclis = ']+TprMvCab.ContaDs+[']
	If (ThisForm.poDataMgr.SqlExecute(lcSql, [LocalCli]) < 1)
		=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (LocalCli)])
		Return 0
	EndIf
	lcQuery = [Select a.*, b.Emps, c.Dopes ] + ;
				[From SigCdFs a Left Join SigCdFsI b On a.Codigos = b.Codigos ] + ;
				[Left Join SigCdFsO c On a.Codigos = c.Codigos ] + ;
			   [Where a.Tipos = 'D' And a.TpDescs <> 2 ] + ;
			   [Order By a.Codigos]

	If (ThisForm.poDataMgr.SqlExecute(lcQuery, [LocalMdsc2]) < 1)
		=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (LocalMdsc2)])
		Return 0
	EndIf

	Select 1 as Flag,* From LocalMdsc2 Where (IsEmpty(PerFins) Or PerFins >= TprMvCab.Datas) And ;
	(IsEmpty(PerInis) Or PerInis >= TprMvCab.Datas) Into Cursor LocalMdSc1 ReadWrite
	Select LocalMdSc1
	Scan
		If LocalMdSc1.ChkDtAnivs = 1
			If NOt ( ( Not IsEmpty(LocalCli.Nascs) And month(TprMvCab.Datas) = month(LocalCli.Nascs)) Or ;
				( NOt IsEmpty(LocalCli.DtnCons) And month(TprMvCab.Datas) = month(LocalCli.DtnCons) ) )
				Replace Flag With 0 In LocalMdSc1
			Endif
		Endif
	EndScan

	Select Distinct Codigos, Descrs From LocalMdsc1 Where Nvl(Emps,ThisForm.pEmps) = ThisForm.pEmps ;
	And Nvl(Dopes,TprMvCab.Dopes) = TprMvCab.Dopes Into Cursor TmpMdsc ReadWrite
	Select TmpMdsc
	Index on Codigos Tag Codigos

	If !Seek( This.Value, 'TmpMdsc', 'Codigos' )
		loBrowse = CreateObject( 'fwBuscaSel', 'TmpMdsc', 'codigos', This.Value, 'Cadastro de Motivos' )
		loBrowse.AddColuna( 'Codigos', '', 'Código', .t., 'Codigos' )
		loBrowse.AddColuna( 'Descrs', '', 'Descricao' )
		loBrowse.Show
		This.Value = TmpMdsc.Codigos
	Endif
	Replace Descs With TmpMdsc.Descrs in xEestDsc
Else
	Replace Descs With ' ' in xEestDsc
EndIf
ThisForm.GradeFPagto.Column2.Refresh

ENDPROC
