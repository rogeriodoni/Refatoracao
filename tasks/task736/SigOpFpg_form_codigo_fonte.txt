* PROCEDURE verificacheque
------------------------------------------------------------
PROCEDURE verificacheque
LParameter wbanco, wagenc, wconta, wnunch

Local _Order, wBanco, wAgenc, wConta, wNunch

lcTpLeChq=CrCfCom.TpLeChqs

select xpar
_order  = order()
_recant = recno()

goto top

sd = .t.
correto = .t.
if correto
	banco = wbanco
  	agencia = wagenc
  	numero = wconta
  	conta = wnunch
endif

if correto
	if !fChkDigitos(banco)
    	correto = .f.
	endif
endif

if correto
	if !fChkDigitos(agencia)
    	correto = .f.
	endif
endif

if correto
	if !fChkDigitos(numero)
    	correto = .f.
	endif
endif

if correto
	if !fChkDigitos(conta)
    	correto = .f.
	endif
endif

if !correto
	clear typeahead
  	If lcTpLeChq = 'S'
    	Messagebox('Leitura de cheque incorreta!',0+48,'')
	Else
    	fMensagemFixa('Leitura de cheque incorreta!',0+48,'','X ')
	Endif
	sd=.f.
endif

if sd
	set order to chq
	seek wbanco+wagenc+wconta+wnunch
	if !eof() and _recant = recno()
    	skip
	endif
	if !eof() and wbanco = bancos and wagenc = agencias and;
		wconta = contas and wnunch = numeros
		select xpar
		set order to (_order)
		if _recant <> 0
			go _recant
		endif
		If lcTpLeChq = 'S'
        	Messagebox('Cheque já lançado!!',0+48,'')
		Else
        	fMensagemFixa('Cheque já lançado !!',0+48,'','X ')
		Endif
     	sd = .f.
  	endif
endif

if sd
	lcBanco = wBanco
	lcAgenc = wAgenc
	lcConta = wConta
	lcCheque= wNunch
	lcwhere = [Select * From SigChe Where Bancos=?lcBanco And Agencias=?lcAGenc And ]+;
			  [nContas=?lcConta And Ncheques=?lcCheque ]

	lnErro = ThisForm.Parentform.PoDataMgr.SqlExecute(lcwhere,'CrCheque')

	If lnErro > 0

		Select CrCheque
		Go Top
	  	If Reccount() > 0
	     	select xpar
	     	set order to (_order)
	     	if _recant <> 0
	        	go _recant
	     	endif
	     	If Not Seek(wbanco+wagenc+wconta+wnunch,'TempChq','Cheque')
		     	If lcTpLeChq = 'S'
		        	MessageBox('Cheque já cadastrado !!',0+48,'')
		     	Else
		        	fMensagemFixa('Cheque já cadastrado !!',0+48,'','X ')
		     	Endif
		     	sd = .f.
			Else
				If xPar.Valos != CrCheque.Valors
			     	If lcTpLeChq = 'S'
			        	MessageBox('Valor do Cheque Incorreto !!',0+48,'')
			     	Else
			        	fMensagemFixa('Valor do Cheque Incorreto !!',0+48,'','X ')
			     	Endif
			     	sd = .f.
				Endif	     
			Endif	     
	  	endif
	Else
		MessageBox('Favor reinicializar o processo.',16,'Falha na Conexão')
		sd = .f.
	Endif
Endif

if !sd
	wait window('Passe o cheque pelo leitor,ou'+chr(13)+;
           '<enter> para digitar manualmente !') nowait
endif

select xpar
set order to (_order)
if _recant <> 0
	go _recant
endif

return sd

ENDPROC

		Thisform.AvisoPortaS.Visible = .F.
	Endif
Else
	Thisform.AvisoPortaT.Visible = .T.
Endif

With Thisform
	.Get_TotForma.Value = ctotal
	.Get_Fpg.Value = ''
	.Get_NPar.Value = 0
	.Get_VPar.Value = 0
	.fweditdata_Venc.Value = {}
	.Get_PEncargo.Value = 0
	.Get_VEncargo.Value = 0
	.Get_Cotacao.Value = 0
	.Refresh
	
Endwith

ThisForm.Apaga.Visible     = Inlist( ThisForm.Op_Escolha, 'INSERIR', 'ALTERAR' )

vCond = [Inlist(crSigOpFp.Situas, "A", " ", "R")]

Select * From crSigOpFp Where &vCond. Into Cursor csSigOpFp
Index On Fpags Tag FPagsGru For !IsEmpty(grupods) .and. !IsEmpty(contads)

Select xPar
Go Top

ENDPROC

		con=substr(Leitura,23,10)
		che=substr(Leitura,14,6)
		Replace ;
			xPar.bancos   With ban ;
			xPar.agencias With age ;
			xPar.contas   With con ;
			xPar.numeros  With che
		With Thisform.CntChequeFP
			.get_bco.value=ban
			.get_bco.refresh
			.get_age.value=age
			.get_age.refresh
			.get_conta.value=con
			.Get_Conta.refresh
			.get_num.value=che
			.Get_Num.refresh
		EndWith
		Thisform.leitor=.t.
	else
		Thisform.leitor=.f. 	
	endif

	if not IsEmpty(this.value)
		if not fChkDigitos(xPar.bancos) 
			messagebox('Banco Inválido !',0+48,'')
			Return
		endif
	endif
endif
Select xPar
Set Order To

ENDPROC
