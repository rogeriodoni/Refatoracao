* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
LPARAMETERS lcacesso

With ThisForm
	.poDatamgr = CreateObject('fSqlConector')
	With .poDatamgr
		.AddCursor('SigFiNtb','CidChaves','crSigFiNtb','',ThisForm.pagina.lista.grade)
		.pfSqlTabela(1).pColuna('Imps','','','Impressora',160,.t.)
		.pfSqlTabela(1).pColuna('ordems','','','Ordem',80,.t.)
		.pfSqlTabela(1).pColuna('descrs','','','Descrição',337,.t.)
		.pfSqlTabela(1).pColuna("Iif(CrSigFiNtb.nfixos=1, 'SIM', 'NÃO')",'','','Fixo',50,.t.)
		.AddCursor('SigCdOpe','Dopes','crSigCdOpe','','')
	EndWith
	.poDataMgr.Requery('crSigCdOpe')
	= DoDefault()
	.LockScreen = .f.
	With ThisForm.Pagina.Dados.Cmb_Imp
		.Clear
		.AddItem( 'DATAREGIS   ' )
		.AddItem( 'SWEDA       ' )
		.AddItem( 'BEMATECH    ' )
		.AddItem( 'BEMA8501    ' )
		.AddItem( 'NDATAREGIS  ' )
		.AddItem( 'EPSON       ' )
		.AddItem( 'DARUMA      ' )
		.AddItem( 'NSWEDA      ' )
		.AddItem( 'Em Branco   ' )
	EndWith
EndWith
ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Lista.Grupo_op.Click
------------------------------------------------------------
PROCEDURE Pagina.Lista.Grupo_op.Click
lparameters opcao

If Type( 'Opcao' ) = 'N'
	This.Value = Opcao
Endif
If This.Value = 4
	Select crSigCdOpe
	Scan
		If crSigCdOpe.TrnFis = CrSigFiNtb.Ordems
			Messagebox('Codigo de Operação sendo utilizada, Impossível Exclusão !!',0+48,'')
			return
			exit
		Endif
	EndScan
Endif
select CrSigFiNtb
If !dodefault( Opcao )
	ThisForm.mAtivapagina1
	Return
Endif

Do Case
	Case Inlist( thisform.pcEscolha, 'INSERIR', 'PROCURAR' )
		ThisForm.Pagina.Dados.cmb_Imp.SetFocus
	Case thisform.pcEscolha = 'ALTERAR'
		ThisForm.Pagina.Dados.cmb_Imp.SetFocus
	Case Inlist( thisform.pcEscolha, 'CONSULTAR', 'EXCLUIR' )
		ThisForm.Pagina.Dados.Grupo_Salva.Salva.SetFocus
Endcase
Select CrSigFiNtb

ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
If Inlist( thisform.pcEscolha, 'INSERIR','ALTERAR')
	If ThisForm.pcEscolha = 'INSERIR'
		Select CrSigFiNtb
		Replace cidchaves with fUniqueIds()
	Endif
	lcImp = ThisForm.Pagina.Dados.Cmb_Imp.Value
	lnOrd = Thisform.Pagina.Dados.get_ordem.Value
	lcOpc = Thisform.Pagina.Dados.Get_opernf.Value
	lcSql = [Select * From SigFiNtb Where Imps=']+lcImp+[' And Ordems = ]+Str(lnOrd,2)+[ ] +;
			[And Not Cidchaves = ']+CrSigFiNtb.CidChaves+[']
	If ThisForm.PodataMgr.SqlExecute(lcSql,'CrTrib') > 0 And (Reccount('CrTrib') > 0)
		Messagebox( 'Impressora / Ordem já cadastrada !', 48, '' )
		Thisform.pagina.dados.get_ordem.setfocus
		Return
	Endif

EndIf

=DoDefault()

ThisForm.Refresh
Thisform.mAtivapagina1

ENDPROC

	thisform.mAtivapagina1
endif
ENDPROC
