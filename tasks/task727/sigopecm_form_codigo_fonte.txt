------------------------------------------------------------
* PROCEDURE processamento
------------------------------------------------------------
PROCEDURE processamento
Local lcArq, lcChr, lcGru, lcDes, lcUni

If (MessageBox([Confirma a Geração dos Arquivos?], 32+4, [Geração de Arquivos]) <> 6)
	Return .f.
EndIf

lcArq = Alltrim(ThisForm.getArqTXT.Value)

If Empty(lcArq)
	=MessageBox([O Arquivo a Ser Gerado Não Foi Informado!!!], 0+16, [Atenção!!!])
	ThisForm.btnGetArqO.SetFocus
	Return .f.
EndIf

=StrToFile([], lcArq, 0)

If Not File(lcArq)
	=MessageBox([O Arquivo Não Pode Ser Criado!!!], 0+16, [Atenção!!!])
	ThisForm.btnGetArqO.SetFocus
	Return .f.
EndIf

Select Distinct CGrus From crImagens Into Cursor crGrus Order By CGrus

lcChr = Chr(13) + Chr(10)

lnQtd = Reccount([crGrus])
lnCnt = 0

ThisForm.lblResults.Caption = [Processando ] + Alltrim(Str(lnQtd, 10)) + [ Grupo] + Iif(lnQtd = 1, [], [s]) + [ / ] + ;
							  [Produtos Gerados : ] + Alltrim(Str(lnCnt, 10))

Create Cursor crGrupos (Grupo n(8))
Index On Grupo Tag Grupo

Select crGrus
Scan
	lcQuery = [Select CGrus, DGrus, Ids ] + ;
			    [From SigCdGrp ] + ;
			   [Where CGrus = '] + crGrus.CGrus + [' And ] + ;
			   		 [Ids > 0]

	If (ThisForm.poObjForm.poDataMgr.SqlExecute(lcQuery, [LocalGrupo]) < 1)
		=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (LocalGrupo)])
		Return .f.
	EndIf
	Select LocalGrupo
	Go Top In LocalGrupo
	If Not Eof([LocalGrupo])
		lcGru = Padl(LocalGrupo.Ids, 8, [0])
		lcDes = Padr(Alltrim(LocalGrupo.DGrus), 40)
		lcUni = Padr([], 4)
		
		* td01 - Tabela de Grupos
		lcStr = [td01] + ;			&& Tipo - "td01" - Constante - Caracter(4)
				lcGru + ;			&& ID - Numérico (8)
				lcDes + ;			&& Descrição - Caracter(40)
				lcUni + lcChr		&& Unidade - Caracter(4)

		=StrToFile(lcStr, lcArq, .t.)

		If Not Seek(LocalGrupo.Ids, [crGrupos], [Grupo])
			Insert Into crGrupos (Grupo) Values (LocalGrupo.Ids)
		EndIf

		Select Distinct CPros ;
		  From crImagens ;
		 Where CGrus = crGrus.CGrus ;
		  Into Cursor crPros ;
		 Order By CPros

		Select crPros
		Scan
			lcQuery = [Select CPros, DPros, CUnis, DtIncs, Situas, PesoMs, Icms, ] + ;
							 [AliqIpis, ClFiscals, SitTrIcms, Pvens, FigJpgs ] + ;
					    [From SigCdPro ] + ;
					   [Where CPros = '] + crPros.Cpros + [']

			If (ThisForm.poObjForm.poDataMgr.SqlExecute(lcQuery, [LocalProds]) < 1)
				=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (LocalProds)])
				Return .f.
			EndIf
			Select LocalProds
			Go Top In LocalProds
			If Not Eof([LocalProds])
				lcRef = Padr(Chrtran(LocalProds.Cpros, [ */\:?"'<>|], []), 25)
				lcDes = Padr(Alltrim(LocalProds.DPros), 60)
				lcUni = Padr(Alltrim(LocalProds.CUnis), 3)
				lcDat = Dtoc(Ttod(LocalProds.DtIncs))
				lcSit = Iif(LocalProds.Situas = 2, [i], [a])
				lcPes = Strtran(Padl(LocalProds.PesoMs, 8, [0]), [.], [,])
				lcIpi = Padl(Int(LocalProds.AliqIpis), 2, [0])
				lcIcm = Padl(Int(LocalProds.Icms), 2, [0])
				lcClf = Padr(Chrtran(LocalProds.ClFiscals, [ .], []), 8)
				lcTri = Padr(LocalProds.SitTrIcms, 3)
				lcVal = Strtran(Padl(LocalProds.PVens, 8, [0]), [.], [,])
				lcPre = Padr([00000000] + lcVal + [0000^], 400)

				lcGvd = [] && Colocar as coleções
				
				lcQuery = [Select b.Descrs, b.Ordens ] + ;
						    [From SigPrCar a, SigCrRap b ] + ;
						   [Where a.CPros = '] + LocalProds.CPros + [' And ] + ;
						   		 [a.Codigos = b.Codigos]

				If (ThisForm.poObjForm.poDataMgr.SqlExecute(lcQuery, [LocalCaracs]) < 1)
					=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (LocalCaracs)])
					Return .f.
				EndIf
				Select LocalCaracs
				Scan
					If Not Empty(LocalCaracs.Ordens)
						If Not Seek(LocalCaracs.Ordens, [crGrupos], [Grupo])
							Insert Into crGrupos (Grupo) Values (LocalCaracs.Ordens)
							lcGr1 = Padl(LocalCaracs.Ordens, 8, [0])
							lcDe1 = Padr(Alltrim(LocalCaracs.Descrs), 40)
							lcUn1 = Padr([], 4)

							* td01 - Tabela de Grupos
							lcStr = [td01] + ;			&& Tipo - "td01" - Constante - Caracter(4)
									lcGr1 + ;			&& ID - Numérico (8)
									lcDe1 + ;			&& Descrição - Caracter(40)
									lcUn1 + lcChr		&& Unidade - Caracter(4)

							=StrToFile(lcStr, lcArq, .t.)
						EndIf
						lcGvd = lcGvd + Alltrim(Str(LocalCaracs.Ordens, 8)) + [^]
					EndIf
				EndScan
				lcGvd = Padr(Alltrim(lcGvd), 60)

				* --> Vamos usar o último campo da tabela td03 para isso. O campo se chama "Material/Complemento"... 
				* as coleções vc envia na td01, como se fossem grupos, e neste campo mencionado vc me envia de quais 
				* grupos o item faz parte... usa ^ como separador... ex. Anel código 001 faz parte das coleções 
				* (td01: 1 = Brilhante e 4 = Primavera)... vc adiciona no campo 1^4... qtas necessárias.. 
				* no campo id_grupo da tabela td03 vc coloca somente a coleção principal do item dai.
				
				* td03 - Tabela de Produtos
				lcStr = [td03] + ;			&& Tipo - "td03" - Constante - Caracter(4)
						Space(8) + ;		&& RESERVADO - Caracter(8)
						lcRef + ;			&& Referência - Caracter(25)
						lcDes + ;			&& Descrição - Caracter(60)
						lcUni + ;			&& Unidade - Caracter(3)
						lcGru + ;			&& ID Grupo - Numérico(8)
						[00000000] + ;		&& ID Subgrupo - Numérico(8)
						lcDat + ;			&& Data do produto - Data(10)
						[ ] + ;				&& Alternativo - Caracter(1)
						[001] + ;			&& Multiplicador - Numérico(3)
						Padr(lcRef, 20) + ;	&& Nome Foto Sem Extenção - Caracter(20)
						lcSit + ;			&& Status Produto - a=ativo / i=inativo / e=problema estoque / s=saindo de linha - Caracter(1)
						[00000000] + ;		&& Desconto Máximo em porcentagem - Numérico(8)
						[00] + ;			&& Tabela - Caracter(2)
						lcPes + ;			&& Peso - Numérico(8)
						[ ] + ;				&& Status Foto - Caracter(1)
						[ ] + ;				&& Lançamento - Caracter(1)
						lcIpi + ;			&& % do IPI - Numérico(2)
						[00000000] + ;		&& Valor IPI - Numérico(8)
						lcIcm + ;			&& % ICMS - Numérico(2)
						[00000000] + ;		&& Valor ICMS - Numérico(8)
						lcClf + ;			&& Classificação Fiscal - Caracter(8)
						lcTri + ;			&& Situação Tributária - Caracter(3)
						Space(8) + ;		&& RESERVADO - Caracter(8)
						[00000000] + ;		&& Estoque - Numérico(8)
						lcPre + ;			&& Lista (8) + Preço (8) + Qtde Minima (4) + ^ - Caracter(400)
						Space(20) + ;		&& Código de Barras - Caracter(20)
						Space(64) + ;		&& RESERVADO - Caracter(64)
						Space(400) + ;		&& Características dos produtos - O Separador de linhas é o "#13" - Caracter(400)
						lcGvd + lcChr		&& Material/Complemento - Caracter(60)

				=StrToFile(lcStr, lcArq, .t.)

				lnCnt = lnCnt + 1

				ThisForm.lblResults.Caption = [Processando ] + Alltrim(Str(lnQtd, 10)) + [ Grupo] + Iif(lnQtd = 1, [], [s]) + [ / ] + ;
											  [Produtos Gerados : ] + Alltrim(Str(lnCnt, 10))
				
				* Exportando as Imagens
				If Not Empty(LocalProds.FigJpgs) And Not IsNull(LocalProds.FigJpgs)
					lcImg = Alltrim(JustPath(lcArq)) + [\] + Alltrim(lcRef) + [.Jpg]
					lcFoto=Strconv(Strtran(Strtran(Strtran(LocalProds.FigJpgs,[data:image/png;base64,],[]),[data:image/jpeg;base64,],[]),[data:image/jpg;base64,],[]),14)	
					=StrToFile(lcFoto, lcImg)
				EndIf
			EndIf
		EndScan
	EndIf
EndScan

If (lnCnt > 0)
	=MessageBox([Arquivo de Produtos Gerado Com Sucesso!!!], 0+64, [Processamento Finalizado])
Else
	=MessageBox([Atenção!!! Arquivo Gerado Sem Produtos!!!], 0+48, [Processamento Finalizado])
EndIf

ThisForm.Cancela.SetFocus
* Fim **************************************************************************************
ENDPROC
