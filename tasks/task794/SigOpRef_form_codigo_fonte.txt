* PROCEDURE auxmontadescs
------------------------------------------------------------
PROCEDURE auxmontadescs
Private pGru, pSGr

Local lnCPro

Select crSigCdPro
pGru = crSigCdPro.CGrus
pSGr = crSigCdPro.SGrus

Select crSigCdPrd
Go Top In crSigCdPrd
Locate For CGrus = pGru
If Not Eof([crSigCdPrd])
	ThisForm.MontaDescrs
	
	Return .t.
EndIf

Go Top In crSigCdPam
lnCPro      = crSigCdPam.CodProds

lnMontaGrDs = 0

If Not Empty(pGru)

	lStrQuery = [Select * From SigCdGrp Where CGrus = '] + pGru + [']
	If ThisForm.poDataMgr.SqlExecute(lStrQuery,'crSigCdGrp') < 1
		MessageBox('Favor reinicializar o processo !!!',16,'Falha na Conexão (CrSigCdGrp)')
		Return 0
	EndIf
	
	lStrQuery = [Select * From SigCdPsg Where Cgrus = '] + pGru + [' And codigos = '] + pSGr + [']
	If ThisForm.poDataMgr.SqlExecute(lStrQuery,'crSigCdPsg') < 1
		MessageBox('Favor reinicializar o processo !!!',16,'Falha na Conexão (CrSigCdPsg)')
		Return 0
	EndIf

	Go Top In crSigCdGrp
	Go Top In crSigCdPsg
	
	lnCPro      = Iif(Empty(crSigCdGrp.CodProds), crSigCdPam.CodProds, crSigCdGrp.CodProds)
	lnMontaGrds = Nvl(crSigCdGrp.MontaGrDs,0)
EndIf

If (lnMontaGrDs <> 5)
	* 1 - Composição
	* 2 - Grupo + Composição
	* 3 - SubGrupo + Composição
	* 4 - Grupo + SubGrupo + Composição
	* 5 - Manual
	* 6 - Grupo + Fornecedor (4 Dig.) + Ref.Forneedor
	* 7 - Grupo + Cor + Composição
	* 8 - Grupo + Pedras + Ouro
	Do Case
		
		Case (crSigCdGrp.MontaGrDs = 1) && Composição
			pGru = ''

		Case (crSigCdGrp.MontaGrDs = 2) && Grupo + Composição
			pGru = Alltrim(Iif(Eof('crSigCdGrp'), '', crSigCdGrp.DGrus))

		Case (crSigCdGrp.MontaGrDs = 3) && SubGrupo + Composição
			pGru = Alltrim(Iif(Not Eof('crSigCdPsg'), crSigCdPsg.Descricaos, Iif(Eof('crSigCdGrp'), '', crSigCdGrp.DGrus)))
			
		Case (crSigCdGrp.MontaGrDs = 4) && Grupo + SubGrupo + Composição
			pGru = Alltrim(Iif(Eof('crSigCdGrp'), '', Alltrim(crSigCdGrp.DGrus)) + ' ' + Iif(Eof('crSigCdPsg'), '', Alltrim(crSigCdPsg.Descricaos)))

		Case (crSigCdGrp.MontaGrDs = 6) && Grupo + Fornecedor (4 digitos) + Referencia Fornecedor
			pGru = Alltrim(Iif(Eof('crSigCdGrp'), '', Alltrim(crSigCdGrp.DGrus)) + ' ' +;
					Iif(Eof('crSigCdCli'), '', Alltrim(Left(Alltrim(crSigCdCli.Rclis), 4))) + ' ' +;
					Alltrim(crSigCdPro.reffs))

		Case (crSigCdGrp.MontaGrDs = 7) && Grupo + Cor + Composição
			ThisForm.poDataMgr.CursorQuery('SigCdCor', 'LocalCor', 'Cods', crSigCdPro.CodCors, 'Cods,Descs')
			pGru = Alltrim(Iif(Eof('crSigCdGrp'), '', Alltrim(crSigCdGrp.DGrus)) + ' ' + Iif(Eof('LocalCor'), '', Alltrim(LocalCor.Descs)))

	EndCase

	lcDescr = fGerarDescrs(pGru, crSigCdPro.CPros, 3, ThisForm.poDataMgr, .f., .f., .f.)

	If Empty(lcDescr)
		lcDescr = [PRODUTO SEM COMPOSIÇÃO]
	EndIf
	
	If (crSigCdGrp.MontaGrDs = 8) && Grupo + Pedras + Ouro
		Replace DPros      With lcDescr, ;
				ObsCompras With lcDescr In crSigCdPro
	Else
		Replace DPros With Subs(lcDescr, 01, 40) In crSigCdPro
	EndIf
Else
	If !Empty(thisform.getnewdpro.Value)
		Replace DPros With Subs(thisform.getnewdpro.Value, 01, 40) In crSigCdPro
	EndIf
EndIf
* Fim **************************************************************************************
ENDPROC

------------------------------------------------------------
* PROCEDURE montadescrs
------------------------------------------------------------
PROCEDURE montadescrs
Local lcDes

If ThisForm.mSv_Inserir()
	Select Distinct CGrus, Linguas, TpMontas ;
	  From crSigCdPrd ;
	 Where CGrus = crSigCdPro.CGrus ;
	 Order By TpMontas ;
	  Into Cursor LocalMontas

	Select LocalMontas
	Scan
		lcDes = fGerDescritivo(ThisForm.poDataMgr, crSigCdPro.CPros, LocalMontas.TpMontas)

		If Not lcDes = Chr(254)
			If Not Empty(lcDes) And (LocalMontas.Linguas = 2) && Descrição Em Inglês
				Select crSigCdDic
				Scan
					lcDes = Strtran(lcDes, Alltrim(crSigCdDic.Expressao), Alltrim(crSigCdDic.Traducao))
				EndScan
			EndIf

			Do Case
				Case (LocalMontas.TpMontas = 1) && Descrição do Produto	
					Replace DPros With lcDes In crSigCdPro

				Case (LocalMontas.TpMontas = 2) && Descrição de Compra	
					Replace DscCompras With lcDes In crSigCdPro

				Case (LocalMontas.TpMontas = 3) && Descrição Completa	
					Replace DPro3s With lcDes In crSigCdPro

				Case (LocalMontas.TpMontas = 4) && Observação da Compra	
					Replace ObsCompras With lcDes In crSigCdPro

				Case (LocalMontas.TpMontas = 5) && Descrição Fiscal		
					Replace DescFis With lcDes In crSigCdPro

				Case (LocalMontas.TpMontas = 6) && Descrição ECF		
					Replace DescEcfs With lcDes In crSigCdPro
			EndCase
		EndIf
	EndScan
EndIf
* Fim **************************************************************************************
ENDPROC

* PROCEDURE msv_inserir
------------------------------------------------------------
PROCEDURE msv_inserir
Private pPro, llTaOk

pPro = crSigCdPro.CPros

llTaOk = .t.
* Incluindo os Registros
If Not ThisForm.poDataMgr.Update('crSigCdPro')
	=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (msv_Inserir - crSigCdPro)')
	llTaOk = .f.
EndIf
If llTaOk And Not ThisForm.poDataMgr.Update('crSigPrCpo')
	=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (msv_Inserir - crSigPrCpo)')
	llTaOk = .f.
EndIf
If llTaOk And Not ThisForm.poDataMgr.Update('crSigCdPrf')
	=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (msv_Alterar - crSigCdPrf)')
	llTaOk = .f.
EndIf
If llTaOk And (ThisForm.poDataMgr.Commit() < 1)
	=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (msv_Inserir - Commit)')
	llTaOk = .f.
EndIf

If Not llTaOk
	ThisForm.poDataMgr.RollBack()
EndIf
Return llTaOk
* Fim **************************************************************************************
ENDPROC

PROCEDURE Init
LParameters pPro, pSit, pAut, pIFx, pCad, pDop, pGru

ThisForm.cntSombra.lblSombra.Caption = ThisForm.Caption
ThisForm.cntSombra.lblTitulo.Caption = ThisForm.Caption

With ThisForm
	.poDataMgr = CreateObject([fSqlConector], .Name)

	If (.poDataMgr.pnIdconn > 0)
		With .poDataMgr
			.AddCursor([SigCdPro],   [CPros],     [crSigCdPro])
			.AddCursor([SigPrCpo], [cIdChaves], [crSigPrCpo])
			.AddCursor([SigCdPrf],  [cIdChaves], [crSigCdPrf])
		EndWith

		.poDataMgr.CursorQuery([SigCdPam], [crSigCdPam], .f., .f., [CodProds, DigProds, TpCbars, CUnis])
		If Not Used([crSigCdPam])
			=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (CursorQuery - crSigCdPam)])
			Return .f.
		EndIf

		.poDataMgr.CursorQuery([SigCdPro], [csTmpPro], [CPros], pPro, [CPros, CGrus, Dpros])
		If Not Used([csTmpPro])
			=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (CursorQuery - csTmpPro)])
			Return .f.
		EndIf

		lcGrp = Iif(Type([pGru]) = [C], pGru, csTmpPro.Cgrus)
		.poDataMgr.CursorQuery([SigCdGrp], [csTmpGru], [CGrus], lcGrp, [CGrus, DigGprs])
		If Not Used([csTmpGru])
			=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (CursorQuery - csTmpGru)])
			Return .f.
		EndIf

		lcQuery = [Select * From SigCdPrd]
		If (.poDataMgr.SqlExecute(lcQuery, [crSigCdPrd]) < 1)
			=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (crSigCdPrd)])
			Return .f.
		EndIf
		Select crSigCdPrd
		Index On CGrus + Str(TpMontas, 2) + Str(COrds, 2) Tag CGrus
		Go Top In crSigCdPrd

		lcQuery = [Select Str(Len(Ltrim(Rtrim(a.Expressao))), 10) as COrdems, a.Expressao, a.Traducao ] + ;
					[From SigCdDic a ] + ;
				   [Where a.Idioma = 'INGLES    ' ] + ;
				   [Order By 1 Desc, 2 Asc]

		If (.poDataMgr.SqlExecute(lcQuery, [crSigCdDic]) < 1)
			=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (crSigCdDic)])
			Return .f.
		EndIf

		.IdentFixa  = Iif(Type([pIFx]) <> [N], 0, pIFx)
		.Situacaoit = pSit
		.Newseq 	= 0
		.AntValue 	= []
		.SaiAuto 	= pAut
		.CadPro  	= Iif(Type([pCad]) = [N], pCad, 0)
		.Dopes  	= Iif(Type([pDop]) <> [C], [], pDop)

		.GetOldCPro.Value  = csTmpPro.Cpros
		.GetOldCGrus.Value = csTmpPro.Cgrus
		.GetNewCGrus.Value = csTmpGru.Cgrus
		.getOldDpro.value = csTmpPro.Dpros		
		.Get_Qtd.Value     = 1

		If (csTmpGru.DigGprs <> 1) Or .SaiAuto
			.AcesGru = .f.
			.GetNewCGrus.Valid()
			.GetNewCGrus.ReadOnly = .t.
		Else
			.AcesGru = .t.
			.GetNewCGrus.ReadOnly = .f.
			.GetNewCGrus.Setfocus
		EndIf
		If (.Situacaoit = 1)
			.Opc_situacao.Value = 1
			.Opc_situacao.Enabled = .f.
		EndIf
		If (.Situacaoit = 2)
			.Opc_situacao.Value = 2
			.Opc_situacao.Enabled = .f.
		EndIf
		If .SaiAuto 
			.cntBotoes.btnConfirmar.Click()
		EndIf

		=DoDefault()
	Else
		=Messagebox([Impossível Efetuar Conexão Com o Servidor de Banco de Dados...], 0+48, [CreateObject])
		Return .f.
	EndIf
EndWith
* Fim **************************************************************************************
ENDPROC

PROCEDURE btnConfirmar.Click
Local lcThisRec, lcNewPros, lcLocPros, lCreated

lcLocPros = Padr(Thisform.getOldCpro.Value, 14)
lcNewPros = Padr(Thisform.getNewCpro.Value, 14)

lcLocGrus = Padr(Thisform.getOldCGrus.Value, 03)
lcNewGrus = Padr(Thisform.getNewCGrus.Value, 03)

ThisForm.RetCodproduto = ' '

lCreated = .f.

ThisForm.poDataMgr.CursorQuery('SigCdPro', 'crTmp', 'CPros', lcNewPros)
If (Reccount('crTmp') > 0)
	=MessageBox('A Nova Referência Informada Já Está Cadastrada!!!', 0+48, '')
	Thisform.getNewCpro.SetFocus
	Use In crTmp
	Return 0
EndIf
Use In crTmp

Go Top In crSigCdPam
If Empty(lcNewPros) Or fVerificarBarras(lcNewPros, crSigCdPam.TpCBars)
	=MessageBox('Código de Produto Inválido!!!', 0+48, '')
	Return 0
EndIf

If Not ThisForm.SaiAuto And MessageBox('Será Gerado Um Novo Produto Com a Referência : [' + lcNewPros + ']' + Chr(13) + ;
			  'Com a Situação : ['+ Iif(Thisform.Opc_situacao.Value=1,'ATIVA','NÃO ATIVA') + ']' + Chr(13) + ;
			  Iif(Not Empty(lcNewGrus) And (lcLocGrus <> lcNewGrus), 'O Grupo Gerado Será o [' + lcNewGrus + ']', '') + Chr(13) + ;
              'Confirma a Geração ?', 4+32+256, 'Solicitação de Confirmação...') <> 6
	Wait Window 'Operação Cancelada Pelo Usuário...' timeOut 2
	Wait Clear
	Return 0
EndIf

Select crSigPrCpo
Zap

* Gerando a Composição
If ThisForm.poDataMgr.CursorQuery('SigPrCpo', 'crTmp', 'CPros', lcLocPros)
	Select crTmp
	Scan
		Select crTmp
		Scatter Memvar
		m.CPros  	= lcNewPros
		m.DtMovs 	= DateTime()
		m.cIdChaves = fUniqueIds()

		Insert Into crSigPrCpo From Memvar
	EndScan
EndIf
Use In crTmp

Select crSigCdPrf
Zap

* Gerando as Linhas
If ThisForm.poDataMgr.CursorQuery('SigCdPrf', 'crTmp', 'Produtos', lcLocPros)
	Select crTmp
	Scan
		Select crTmp
		Scatter Memvar Memo
		m.FigProcs  = Iif(IsEmpty(m.FigProcs), Null, m.FigProcs)
		m.Obs       = Iif(IsEmpty(m.Obs),      Null, m.Obs)
		m.Produtos  = lcNewPros
		m.cIdChaves = fUniqueIds()

		Insert Into crSigCdPrf From Memvar
	EndScan
EndIf
Use In crTmp

lcSql = [Select * from SigCdPro where Cpros = ']+lcLocPros+[']
ThisForm.Podatamgr.Sqlexecute(lcSql,'CrTmp')

* Gerando o Produto
Select CrTmp
Scatter Memvar Memo
Select crSigCdPro
Zap

m.Cpros  = lcNewPros
If Not Empty(lcNewGrus) And (lcLocGrus <> lcNewGrus)
	m.CGrus = lcNewGrus
EndIf
m.Situas = Thisform.Opc_situacao.Value
m.nIdentFixa = ThisForm.IdentFixa
If ThisForm.CadPro = 5
	ThisForm.poDataMgr.CursorQuery([SigOpCdd],[crSigOpCdd],[Dopes],ThisForm.Dopes,[FinPads,nLimpaFoto,nSeriaPads])
	m.CodFinP = crSigOpCdd.FinPads
	If InList(crSigOpCdd.nSeriaPads,1,2)
		m.Encoms = crSigOpCdd.nSeriaPads
	EndIf
	If crSigOpCdd.nLimpaFoto = 1
		m.FigJpgs = []
	EndIf		
EndIf

Insert Into crSigCdPro From Memvar

Select crSigCdPro
ThisForm.Inicio = fGerUniqueKey('SigCdPro')

Cbar     = Transform(ThisForm.Inicio, '@L 9999999999999')
Cbar1    = fDigVerificador(Cbar)
Cbar2    = Cbar + Transform(Cbar1, '@L 9')
lcProEqs = Thisform.GetOldCPro.Value

If ThisForm.CadPro = 5
	lcProEqs = Space(14)
EndIf

Set Decimals To 0
Replace Cbars With Val(Cbar2) In crSigCdPro
Set Decimals To 2

Replace Datas     With DateTime(), ;
		DtIncs    With DateTime(), ;
		Flagctabs With .t., ;
		Transps   With .f., ;
		UsuIncs   With Usuar,;
		CProEqs   With lcProEqs In crSigCdPro

ThisForm.Auxmontadescs()  &&Monta Descricao do Produto

ThisForm.RetCodproduto = ' '
If ThisForm.mSv_Inserir()
	ThisForm.retcodproduto = Padr(lcNewPros,14) + Str(ThisForm.Get_Qtd.Value,2)
EndIf
ThisForm.Release
* Fim **************************************************************************************
ENDPROC
