------------------------------------------------------------
* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
LPARAMETERS lcacesso

With ThisForm
	.poDatamgr = CreateObject('fSqlConector')
	With .poDatamgr
		.AddCursor('SigFiNz','nordems','crSigFiNz','',ThisForm.pagina.lista.grade)
		.pfSqlTabela(1).pColuna('nordems','','','Ordem',80,.t.)
		.pfSqlTabela(1).pColuna('cdescrs','','','Descrição',337,.t.)
	EndWith
	= DoDefault()
	.LockScreen = .f.
EndWith
ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Lista.Grupo_op.Click
------------------------------------------------------------
PROCEDURE Pagina.Lista.Grupo_op.Click
lparameters opcao
select CrSigFiNZ
If !dodefault( Opcao )
	ThisForm.mAtivapagina1
	Return
Endif

Do Case
	Case Inlist( thisform.pcEscolha, 'INSERIR', 'PROCURAR' )
		ThisForm.Pagina.Dados.Get_Ordem.SetFocus
	Case thisform.pcEscolha = 'ALTERAR'
		ThisForm.Pagina.Dados.Get_Descr.SetFocus
	Case Inlist( thisform.pcEscolha, 'CONSULTAR', 'EXCLUIR' )
		ThisForm.Pagina.Dados.Grupo_Salva.Salva.SetFocus
Endcase
Select CrSigFiNZ

ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
if Inlist( thisform.pcEscolha, 'INSERIR')
	If ThisForm.poDataMgr.ChkRegister('SigFiNz','nordems',thisform.pagina.dados.get_ordem.value)
		messagebox('Ordem já cadastrada ! ',0+48,'')
		thisform.pagina.dados.get_ordem.setfocus
		Return
	Endif
endif
if Inlist( thisform.pcEscolha, 'INSERIR', 'ALTERAR')

	if IsEmpty(thisform.pagina.dados.get_descr.value)
		messagebox('Descrição Inválida',0+48,'')
		thisform.pagina.dados.get_descr.setfocus
		return
	endif
	
Endif

=DoDefault()

ThisForm.Refresh
Thisform.mAtivapagina1

ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Dados.Grupo_Salva.Cancelar.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Cancelar.Click
DoDefault()

ThisForm.Refresh

if thisform.plCancelar
	thisform.mAtivapagina1
endif
ENDPROC
