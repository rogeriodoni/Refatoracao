PROCEDURE Valid
Local lcQuery, lcUser

If Not Empty(This.Value)
	lcUser = Padr(Alltrim(ThisForm.Get_Usuário.Value), 10)
	lcQuery = [Select a.Usuarios, a.Senhas, a.CCargs From SigCdUsu a Where Not a.cAtivos='N' and a.usuarios = '] + lcUser + [' order by 1]
	ThisForm.poDataMgr.SQLExecute(lcQuery, 'crUsuar')

	If Not Used('crUsuar') Or Reccount('crUsuar')==0
		=Messagebox([Usuário Inválido], 48, [Atenção!!!])
		Return 0
	Else

			=Messagebox([Usuário Sem Acesso Para Alterar Cotação], 48, [Atenção!!!])
			Return 0
		EndIf
	EndIf
EndIf

PROCEDURE btnConfirmar.Click
If !Empty( ThisForm.Get_Usuário.Value )
	Local lcUser,lcQuery

	lcUser = Padr( Alltrim( ThisForm.Get_Usuário.value ), 10 )
	lcQuery = [select a.usuarios, a.senhas, a.cCargs from SigCdUsu a where Not a.cAtivos='N' and a.usuarios='] + lcUser + [' order by 1]
	ThisForm.poDataMgr.SQLExecute(lcQuery, 'crUsuar')

	If Not Used('crUsuar') Or Reccount('crUsuar')==0
		Messagebox( 'Usuário Inválido !!', 48, '' )
		ThisForm.Get_Usuário.SetFocus
		Return
	Else
		If !ThisForm.PodataMgr.Cursorquery('SigCdCrg','CrCarg','cCargs',crUsuar.Ccargs) Or crCarg.AltCots # 'S'
			Messagebox( 'Usuário sem acesso para alterar cotação !!', 48, '' )
			ThisForm.Get_Usuário.SetFocus
			Return
		Endif
	Endif
Else
	Messagebox( 'Usuário Inválido !!', 48, '' )
	ThisForm.Get_Usuário.SetFocus
	Return
Endif

If Padr( Alltrim( ThisForm.Get_Senha.Value ), 10 ) # fCriptografar( crUsuar.Senhas )
	Messagebox( 'Senha incorreta !!', 48, '' )
	ThisForm.Get_Senha.SetFocus
	Return
Endif

If ThisForm.Get_Cotação.Value = 0
	Messagebox( 'Cotação Inválida !!', 48, '' )
	ThisForm.Get_Cotação.SetFocus
	Return
Endif

ThisForm.Cotação = fBuscarCotacao(ThisForm.Get_Moeda.Value, ThisForm.Data, ThisForm.podatamgr )

VarCot = ( Round( ThisForm.Get_Cotação.Value / ThisForm.Cotação, 2 ) - 1 ) * 100

If Abs( VarCot ) > crCarg.VarCots
	Messagebox( 'Cotação Inválida !!', 48, '' )
	ThisForm.Get_Cotação.SetFocus
	Return
Endif

ThisForm.Retorna = "OK" + Chr( 13 ) + ;
					Str( Thisform.Get_Cotação.Value, 15, 4 ) + Chr( 13 ) + ;
					ThisForm.Get_Usuário.Value
Set Decimals to 2
ThisForm.Release
* Fim **************************************************************************************
ENDPROC
