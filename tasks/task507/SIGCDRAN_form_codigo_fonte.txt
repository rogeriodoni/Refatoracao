PROCEDURE inicdesc
iTPan = ThisForm.Pagina.Dados.Gettpan.Value
If Thisform.podatamgr.cursorquery('SigPrTpa','crTmpSigPrTpa','codigos',iTPan)
  ThisForm.Pagina.Dados.Getdtpan.Value = crTmpSigPrTpa.Descs
Else
  ThisForm.Pagina.Dados.Getdtpan.Value = ''
EndIf
iContas = thisform.pagina.dados.getcdconta.Value
If thisform.podatamgr.cursorquery('SigCdCli','crTmpSigCdCli','iclis',iContas)
  thisform.pagina.dados.getdsconta.Value = crTmpSigCdCli.rclis
Else
  thisform.pagina.dados.getdsconta.Value = ''
EndIf
ENDPROC

------------------------------------------------------------
* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
With ThisForm
	.poDataMgr = CreateObject('fSqlConector', ThisForm.Name)

	If (.poDataMgr.pnIdconn > 0)
        lcQryRGan = [Select * From SigInAna]
		With .poDataMgr
			.AddCursor('SigInAna', 'codigos', 'crSigInAna', '', ThisForm.Pagina.Lista.Grade, lcQryRGan)
			.pfSqlTabela(1).pColuna('Codigos', '', '', 'Códigos', 80, .t.)
			.pfSqlTabela(1).pColuna('Datas', '', '', 'Data', 80, .t.)
			.pfSqlTabela(1).pColuna('resps', '', '', 'Responsável', 80, .t.)
			.pfSqlTabela(1).pColuna('codtpans', '', '', 'Tipo Análise', 150, .F.)
			.pfSqlTabela(1).pColuna('nlaudos', '', '', 'N. Laudo', 80, .F.)
		EndWith

		*!*	Criação de tabelas temporárias

		=DoDefault()

		ThisForm.Pagina.Lista.Grade.Column1.SetFocus
	Else
		=Messagebox('Impossivel Efetuar Conexão com Servidor de Banco de Dados...', 0+48, 'CreateObject')
		Return .f.
	EndIf
EndWith
* Fim **************************************************************************************

ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Lista.Grupo_op.Click
------------------------------------------------------------
PROCEDURE Pagina.Lista.Grupo_op.Click
LPARAMETERS opcao

Local Codigos,llFechado

If Type('m.Opcao')=='N'
	This.Value = m.Opcao 
Endif

If ( DoDefault(This.Value) )

	ThisForm.LockScreen = .t.

    If not Empty(Thisform.Pagina.Dados.gettpan.Value)
      ThisForm.podatamgr.cursorquery('SigPrTpa','csTmpSigPrTpa','codigos',Thisform.Pagina.Dados.gettpan.Value)
	  Thisform.Pagina.Dados.getDtpan.Value = csTmpSigPrTpa.Descs
	EndIf
	                                                
	lcConta = Padr(Thisform.Pagina.Dados.getCdConta.Value,10)	                                               
    ThisForm.podatamgr.cursorquery('SigCdCli','csTmpSigCdCli','iclis',lcConta)
	Thisform.Pagina.Dados.getDtpan.Value = csTmpSigCdCli.Rclis

	ThisForm.Pagina.Dados.GetDtPan.Refresh
	ThisForm.Pagina.Dados.getDsConta.Refresh

    ThisForm.Pagina.Dados.Getdtpan.Value = ''
    thisform.pagina.dados.getdsconta.Value = ''
	
	Do Case
		Case InList(ThisForm.pcEscolha,'INSERIR','ALTERAR')
		
			Thisform.Pagina.Dados.Container1.Enabled = .T.
			Thisform.Pagina.Dados.Container2.Enabled = .T.

			If ThisForm.pcEscolha=='INSERIR'
	
				Select crSigInAna
				
				m.Codigos = fGerUniqueKey('SigInAna')
				m.Emps = _Empr
				Gather Memvar Fields Codigos,Emps
				
			EndIf
			
			Thisform.Pagina.Dados.getDatas.Value = Iif(ThisForm.pcEscolha=='INSERIR',Date(),This.Parent.Parent.Dados.getDatas.Value)
			Thisform.Pagina.Dados.getLaudo.SetFocus()
			
		Case ThisForm.pcEscolha=='CONSULTAR'
	
			Thisform.Pagina.Dados.Container1.Enabled = .F.
			Thisform.Pagina.Dados.Container2.Enabled = .F.
			  
		Case ThisForm.pcEscolha=='EXCLUIR'
		
			Thisform.Pagina.Dados.Grupo_Salva.Salva.SetFocus
			Thisform.Pagina.Dados.Container1.Enabled = .F.
			Thisform.Pagina.Dados.Container2.Enabled = .F.
			  
		Case ThisForm.pcEscolha=='PROCURAR'
			  
			Thisform.Pagina.Dados.getCodigos.SetFocus
			  
	EndCase
	If InList(ThisForm.pcEscolha,'CONSULTAR','ALTERAR','EXCLUIR')
	  ThisForm.inicdesc()
	EndIf

	Select crSigInAna
	
	ThisForm.Refresh
	ThisForm.LockScreen = .f.
	
Else

	ThisForm.mAtivapagina1
	
EndIf
ENDPROC

PROCEDURE Valid
if not IsEmpty(this.value)
  If this.value = 0
    ThisForm.Pagina.Dados.Getdtpan.Value = ''
  Else
	loLista = CreateObject('fwBuscaExt', ThisForm.poDataMgr.pnIdConn, 'SigPrTpa', ;
						   'crListaRemota','CODIGOS', This.Value, 'Seleção', .T.)
	If Not loLista.plAchouRegistro
		loLista.mAddColuna('Codigos', '99', 'Código')
		loLista.mAddColuna('Descs', '', 'Descrição')
		loLista.Show()
	EndIf
	If Lastkey() = 27
		ThisForm.Pagina.Dados.Gettpan.Value = ''
		ThisForm.Pagina.Dados.Getdtpan.Value = ''
	Else
		ThisForm.Pagina.Dados.Gettpan.Value = crListaRemota.Codigos
		ThisForm.Pagina.Dados.Getdtpan.Value = crListaRemota.Descs
	Endif
	ThisForm.Pagina.Dados.Gettpan.Refresh()
	ThisForm.Pagina.Dados.Getdtpan.Refresh()
	Use In crListaRemota
  EndIf
Else
  ThisForm.Pagina.Dados.Getdtpan.Value = ''
endif
return(.t.)

ENDPROC

------------------------------------------------------------
* PROCEDURE When
------------------------------------------------------------
PROCEDURE When
Return(InList(ThisForm.pcEscolha,'INSERIR','ALTERAR'))
ENDPROC

* Classe: fwget (Base: textbox)
* Registro: 27
================================================================================

------------------------------------------------------------
* PROCEDURE Valid
------------------------------------------------------------
PROCEDURE Valid
If !Empty(This.Value)
	set decimals to 0
	ndope = Val( Left( Str( This.value, 10 ), 4) )
	
    ThisForm.poDatamgr.CursorQuery('SigCdOpe','crSigCdOpe','nDopes',ndope)

	Dope = crSigCdOpe.Dopes
	If crSigCdOpe.Ctrllotes # 1
		Messagebox('Operação '+Dope+' não Configurada para gerar No. Lote !!', 48, '')
		Return 0
	Endif
	Nume = Val( Right( Str( This.Value, 10), 6 ) )
	set decimals to 2
	Declare laCampo(2,2)
	LaCampo(1,1)='Dopes'
	LaCampo(2,1)='Numes'
	LaCampo(2,1)=Dope
	LaCampo(2,2)=Nume
	If !Thisform.poDataMgr.ChkRegister('SigMvCab',@laCampo)
		Messagebox('Lote Nº '+Str(This.Value,10)+' não Existe !!', 48, '')
		Return 0
	EndIf
Endif

ENDPROC

------------------------------------------------------------
* PROCEDURE When
------------------------------------------------------------
PROCEDURE When
Return ( Empty(This.Parent.GetOp.Value) And Empty(This.Parent.Getnfund.Value) )
ENDPROC
