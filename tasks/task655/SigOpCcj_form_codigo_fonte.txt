------------------------------------------------------------
* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
Lparameters pForms, pIClis, pICpfs, pRClis, pCpRet
* Parâmetros : pForms - Formulário de Origem
*              pIClis - Campo do Formulário Anterior De Código da Conta
*              pICpfs - Campo do Formulário Anterior De Cpf da Conta
*              pRClis - Campo do Formulário Anterior De Descrição da Conta
*			   pCpRet - Propriedade Para Receber o Retorno Deste Formulário

ThisForm.cntSombra.lblSombra.Caption = ThisForm.Caption
ThisForm.cntSombra.lblTitulo.Caption = ThisForm.Caption

If (Type('pForms') <> 'O')
	=Messagebox('O Formulário Passado Não É Valido...', 0+48, 'Parâmetros Inválidos!')
	Return .f.
Else
	ThisForm.ParentForm = pForms
EndIf

If (Type('pIClis') <> 'C') Or Empty(pIClis)
	=Messagebox('O Campo de Código da Conta Passado Não É Valido...', 0+48, 'Parâmetros Inválidos!')
	Return .f.
EndIf

If (Type('pICpfs') <> 'C') Or Empty(pICpfs)
	=Messagebox('O Campo de Cpf da Conta Passado Não É Valido...', 0+48, 'Parâmetros Inválidos!')
	Return .f.
EndIf

If (Type('pRClis') <> 'C') Or Empty(pRClis)
	=Messagebox('O Campo de Descrição da Conta Passado Não É Valido...', 0+48, 'Parâmetros Inválidos!')
	Return .f.
EndIf

If (Type('pCpRet') <> 'C') Or Empty(pCpRet)
	=Messagebox('O Campo Para Retorno da Função Passado Não É Valido...', 0+48, 'Parâmetros Inválidos!')
	Return .f.
EndIf

With ThisForm
	.poDataMgr = .ParentForm.poDataMgr

	.pIClis = pIClis
	.pICpfs = pICpfs
	.pRClis = pRClis
	.pCpRet = pCpRet

	=DoDefault()
EndWith
* Fim **************************************************************************************
ENDPROC

------------------------------------------------------------
* PROCEDURE When
------------------------------------------------------------
PROCEDURE When
Return Empty(ThisForm.getConjuges.Value)
ENDPROC

------------------------------------------------------------
* PROCEDURE Valid
------------------------------------------------------------
PROCEDURE Valid
Local loLista

If Not Empty(This.Value)
	loLista = CreateObject('fwBuscaExt', ThisForm.poDataMgr.pnIdConn, 'SigCdCli', ;
						   'crListaRemota', 'RgConjuges', This.Value, 'Seleção', 200)

	If Not loLista.plAchouRegistro
		loLista.mAddColuna('RGConjuges', '', 'RG')
		loLista.mAddColuna('Conjuges',   '', 'Nome')
		loLista.mAddColuna('IClis',      '', 'Código')
		loLista.mAddColuna('Inativas ', .f., .f., .f., .f., .f., .t.)
		loLista.gridProc.SetAll('DynamicForeColor', [Iif(crListaRemota.Inativas == 1, Rgb(255, 0, 0), Rgb(0,0,0))], 'Column')
		loLista.Show()
	EndIf
	
	ThisForm.getConjuges.Value   = crListaRemota.Conjuges
	ThisForm.getRgConjuges.Value = crListaRemota.Rgs

	If Not Empty(crListaRemota.IClis)
		ThisForm.poDataMgr.CursorQuery('SigCdCli', 'LocalCli', 'IClis', crListaRemota.IClis, 'IClis, Cpfs, RClis')
		ThisForm.getIClis.Value = LocalCli.IClis
		ThisForm.getCpfs.Value  = LocalCli.Cpfs
		ThisForm.getRClis.Value = LocalCli.RClis
		Use In LocalCli
	EndIf
	Use In crListaRemota
Else
	ThisForm.getConjuges.Value   = ''
	ThisForm.getRgConjuges.Value = ''
	ThisForm.getIClis.Value      = ''
	ThisForm.getCpfs.Value       = ''
	ThisForm.getRClis.Value      = ''
EndIf
ThisForm.getConjuges.Refresh
ThisForm.getRgConjuges.Refresh
ThisForm.getIClis.Refresh
ThisForm.getCpfs.Refresh
ThisForm.getRClis.Refresh
* Fim **************************************************************************************
ENDPROC
