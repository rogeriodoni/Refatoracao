PROCEDURE Init
Parameters ThisParent, _DataSession

ThisForm.cntSombra.lblSombra.Caption = ThisForm.Caption
ThisForm.cntSombra.lblTitulo.Caption = ThisForm.Caption

ThisForm.ThisParent = ThisParent
Set DataSession to _DataSession

*!*	This.Caption = 'Cotação do ' + Alltrim(CrSigCdMoe.cmoes)

*!*	Select * From CrSigCdCot Into Cursor TmpCot ReadWrite
*!*	Index On cidchaves Tag CidChaves
*!*	Index On cmoes + Dtos(datas) + horas Tag Cotacaos
*!*	Go Bottom

*!*	Zap In CrSigCdCot

lcQuery = [Select DtIncs, FaixaInis, FaixaFins, sum(Case When EmpDopNums = '' Then 0 Else 1 End) as nUsadas, sum(Case When EmpDopNums <> '' Then 0 Else 1 End) as nLivres ] +;
			[from SigCdSed Where cEmps = ?crSigCdEmp.cEmps Group By DtIncs, FaixaInis, FaixaFins Order By DtIncs, FaixaInis, FaixaFins]
If Not (ThisForm.ThisParent.poDatamgr.SqlExecute(lcQuery,'crSedex') > 0)
	MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (crSedex)')
	Return .f.
EndIf			

thisform.grade.RecordSource = 'crSedex'
thisform.grade.column1.ControlSource = 'crSedex.FaixaInis'
thisform.grade.column2.ControlSource = 'crSedex.FaixaFins'
thisform.grade.column3.ControlSource = 'crSedex.nLivres'
thisform.grade.column4.ControlSource = 'crSedex.nUsadas'
thisform.grade.column4.ControlSource = 'crSedex.nUsadas'
thisform.grade.column5.ControlSource = 'crSedex.DtIncs'

****************  Acesso dos Botões  *****************
*!*	ThisForm.Inserir.Visible = fChecaAcesso('SIGPRCOT','INSERIR')
*!*	ThisForm.Delete.Visible  = fChecaAcesso('SIGPRCOT','EXCLUIR')

------------------------------------------------------------
* PROCEDURE Click
------------------------------------------------------------
PROCEDURE Click
If Empty(thisform.getfaixainicial.Value) or Empty(thisform.getfaixafinal.value)
	Messagebox([Faixa inicial e final são obrigatórias.],48,[Aviso])
	Return .f.
Endif

lcFaixaIni = Substr(fSubstTxt(thisform.getfaixainicial.Value, [], [N]),1,8)
lcFaixaFin = Substr(fSubstTxt(thisform.getfaixafinal.Value, [], [N]),1,8)
lnFaixaIni = Val(lcFaixaIni)
lnFaixaFin = Val(lcFaixaFin)

*** Checa se já tem etiqueta gerada pra faixa informada; checa sem o dígito verificador
lcQuery = [Select cBarSedexs from SigCdSed where nNumEtiqs Between ?lcFaixaIni And ?lcFaixaFin And cEmps = ?crSigCdEmp.cEmps]
If Not (ThisForm.ThisParent.poDatamgr.SqlExecute(lcQuery,'LocalSed') > 0)
	=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (LocalSed)')
	Return .f.
EndIf

Select cBarSedexs from crSigCdSed where nNumEtiqs Between lnFaixaIni And lnFaixaFin And cEmps = crSigCdEmp.cEmps into cursor LocalSed2

If !Eof('LocalSed') or !Eof('LocalSed2')
	Messagebox([Já existe etiquetas geradas na faixa informada.],48,[Aviso])
	Return .f.
Endif

For x = lnFaixaIni to lnFaixaFin
	Insert into crSigCdSed(cEmps,DtIncs,nNumEtiqs,FaixaInis,FaixaFins) values (crSigCdEmp.cEmps,Date(),x,thisform.getfaixainicial.Value,thisform.getfaixafinal.Value)
	lcBarEtiq = Substr(thisform.getfaixainicial.Value,1,2) + Str(x,8) + Substr(thisform.getfaixainicial.Value,13,2)	
	lcBarEtiq = fBarSedex(lcBarEtiq)
	replace cBarSedexs With lcBarEtiq in crSigCdSed
Next

Select crSigCdSed
Insert into crSedex(DtIncs,FaixaInis,FaixaFins,nLivres,nUsadas) values (Date(),thisform.getfaixainicial.Value,thisform.getfaixafinal.Value,(lnFaixaFin - lnFaixaIni + 1),0)
thisform.grade.Refresh()



ENDPROC

Return(INLIST(Thisform.thisparent.pcEscolha,'INSERIR','ALTERAR'))
