				   Alltrim(ThisForm.ParentForm.Origem.get_Conta.Value) + ' - ' + ;
				   Alltrim(ThisForm.ParentForm.Origem.get_dConta.Value)

PROCEDURE Click
Select TmpOper
If Not Eof()
	lnRecno = Recno()
Else
	lnRecno = 0
EndIf

Set Order To Num
ThisForm.fwgrade.Refresh

If (lnRecno <> 0)
	Go lnRecno
Else
	Go Top
EndIf
ThisForm.fwgrade.Refresh
* Fim **************************************************************************************

ENDPROC
