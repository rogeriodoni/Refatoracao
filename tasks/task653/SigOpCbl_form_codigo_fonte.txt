lcQryCfgBl = [Select * From SigCnFBl]
ThisForm.cntSombra.lblSombra.Caption = ThisForm.Caption
ThisForm.cntSombra.lblTitulo.Caption = ThisForm.Caption

With ThisForm
	.poDataMgr = CreateObject('fSqlConector', ThisForm.Name)

	If (.poDataMgr.pnIdconn > 0)
		With .poDataMgr
			.AddCursor('SigCnFBl', 'cIdChaves', 'crSigCnFBl', '','', lcQryCfgBl)
		EndWith
		
		.poDataMgr.ReQuery('crSigCnFBl')

		Select crSigCnFBl
		Go TOP

		If Eof()
		  Append Blank 
		EndIf

		=DoDefault()
		ThisForm.Text40.value = ''

		If Not Empty(crSigCnFBl.ctamfolha) 
			iPosSeparador = At('/',crSigCnFBl.ctamfolha)
			If iPosSeparador = 1
				lcTamFolhaTmp = Substr(crSigCnFBl.ctamfolha,2)
				lcDescricao   = Substr(crSigCnFBl.ctamfolha,At('/',lcTamFolhaTmp)+1)
				ThisForm.Text40.Value = lcDescricao
			EndIf
			If iPosSeparador = 0
				ThisForm.Text40.value = ''
			Else
				If At('/',substr(crSigCnFBl.ctamfolha,iPosSeparador+1)) > 0
					ThisForm.Text40.value = Substr(substr(crSigCnFBl.ctamfolha,iPosSeparador+1),At('/',substr(crSigCnFBl.ctamfolha,iPosSeparador+1))+1)
				Else
					ThisForm.Text40.value = ''
				EndIf
			EndIf 
		EndIf 

		ThisForm.Refresh
		ThisForm.Text1.SetFocus
	Else
		=Messagebox('Impossivel Efetuar Conexão com Servidor de Banco de Dados...', 0+48, 'CreateObject')
		Return .f.
	EndIf
EndWith
* Fim **************************************************************************************

ENDPROC

PROCEDURE cmdOk.Click

Local dDatas,cHoras,cUsuarios,ctamfolha,lcTamFolha

If MessageBox('Confirma a Configuração do Boleto?',32+4,goSistema.NomeSistema)==6

	m.dDatas = Date()
	m.cHoras = Time()
	m.cUsuarios = m.Usuar

	If Not Empty(crSigCnFBl.ctamfolha) 
		If At('/',crSigCnFBl.ctamfolha) = 0
			lcTamFolha = Alltrim(ThisForm.Text30.value)+'/'+SubStr(crSigCnFBl.ctamfolha,1,10)+'/'+SubStr(crSigCnFBl.ctamfolha,11)
		Else
			lcLocalBarraTamFolha = At('/',Substr(crSigCnFBl.ctamfolha,At('/',crSigCnFBl.ctamfolha)+1))
			If lcLocalBarraTamFolha > 0
				lcStringTamFolha = Substr(crSigCnFBl.ctamfolha,At('/',crSigCnFBl.ctamfolha)+1)
				lcCodigo    = Alltrim(Substr(lcStringTamFolha,1,At('/',lcStringTamFolha)-1))
				lcDescricao = Alltrim(Substr(lcStringTamFolha,At('/',lcStringTamFolha)+1))
				lcTamFolha = Alltrim(ThisForm.Text30.value)+'/'+lcCodigo+'/'+lcDescricao
			Else
				lcTamFolha = Alltrim(ThisForm.Text30.value)+'/'
			EndIf
		EndIf
	Else
		lcTamFolha = Alltrim(ThisForm.Text30.value)+'/'
	EndIf
	m.ctamfolha = lcTamFolha

	Select crSigCnFBL
	Gather Memvar Fields dDatas,cHoras,cUsuarios,ctamfolha

	If Not ThisForm.poDataMgr.Update('crSigCnFBL')
		=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (Update - crSigCnFBL)')
		Return 0
	EndIf
	lcQuery = [Select * ] + ;
			    [From SigCqMat]

	If (ThisForm.poDataMgr.SqlExecute(lcQuery, 'csSigCqMat') < 1)
		=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (SigCqMat)')
		Return 0
	EndIf

	Select csSigCqMat

	If (Not Eof('csSigCqMat') And Not IsEmpty(m.lcTamFolha))
		lcQuery = [Update SigCqMat Set ctamanho=']+m.lcTamFolha+[']

		If (ThisForm.poDataMgr.SqlExecute(lcQuery) < 1)
			=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (SigCqMat)')
			Return 0
		EndIf
	EndIf
	If (ThisForm.poDataMgr.Commit() < 1)
		=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (Commit - SigCnFBL)')
		Return 0
	EndIf
    ThisForm.Release 
Else
  ThisForm.Text1.SetFocus
EndIf

ThisForm.LockScreen = .f.

ENDPROC
