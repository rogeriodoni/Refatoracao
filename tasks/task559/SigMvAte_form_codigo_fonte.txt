PROCEDURE Valid
LOCAL lnReg, lcCodVen
PRIVATE pcGrVen, pcCdVen, pcDsVen

pcGrVen = TprMvCab.grvends

If Not Empty(This.Value)
	If Not fAcessoContas(Usuar, pcGrVen, 'C', This.Value, thisform.grade.column2.text1, thisform.grade.column3.text1,.t.)
		=Messagebox('Acesso Negado!!!', 0+48, '')
		This.Value = ''
		This.Refresh
		replace codvends with '' in localven
		replace desvends with '' in localven
		Return 0
	EndIf
Else
	replace codvends with '' in localven
	replace desvends with '' in localven
EndIf

SELECT localVen
lnReg = RECNO()
lcCodVen = codvends
LOCATE for codvends = lcCodven and RECNO() <> lnReg
IF FOUND()
	=Messagebox('Vendedor já incluido!!!', 0+48, 'Aviso')
	GO lnReg
	RETURN .f.
ENDIF
GO lnReg
RETURN .t.
 
ENDPROC
