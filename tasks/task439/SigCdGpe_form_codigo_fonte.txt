* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
LParameters lcAcesso

With Thisform
	.poDatamgr = CreateObject('fSqlConector',ThisForm.Name)

	If Not (.poDatamgr.pnIdconn > 0)
		Messagebox('Impossivel Efetuar Conexão com Servidor de Banco de Dados.',(0+48),'Erro')
		Return(.f.)
	Endif

	With .poDatamgr
		.AddCursor('SigCdGpe','DGOpes','crSigCdGpe','DGOpes',ThisForm.pagina.lista.grade)
		.pfSqlTabela(1).pColuna('DgOpes','','','Grupo Operacional',150,.t.)
		.AddCursor('SigCdGpo','CidChaves','crSigCdGpo','DGOpes',.F.)
		.AddCursor('SigCdPrg','pkChaves','crSigCdPrg','Programas',.f.)

		ThisForm.pagina.dados.pgfDados.pgConfig.Gradeo.RecordSource='crSigCdGpo'
		ThisForm.pagina.dados.pgfDados.pgConfig.gradeo.Column1.ControlSource="crSigCdGpo.Dopers"
		ThisForm.pagina.dados.pgfDados.pgConfig.gradeo.Column2.ControlSource="crSigCdGpo.confops"
	EndWith
Endwith	

=DoDefault()


ENDPROC

------------------------------------------------------------
PROCEDURE msv_alterar
Local m.lnErro1

m.lnErro1 = ThisForm.poDatamgr.pnIdConn

With ThisForm
	If m.lnErro1>0 
		m.lnErro1 = Iif(.poDataMgr.UpDate('crSigCdGpe'),1,-1)
	EndIf 
	If m.lnErro1>0 
		m.lnErro1 = Iif(.poDataMgr.UpDate('crSigCdGpo'),1,-1)
	EndIf 
	If m.lnErro1>0 
		m.lnErro1 = Iif(.poDataMgr.UpDate('crSigCdPrg'),1,-1)
	EndIf 
	If m.lnErro1>0 
		m.lnErro1 = .poDataMgr.Commit()
	EndIf 
	If Not m.lnErro1>0 
		.poDataMgr.RollBack()
	EndIf 
EndWith 
If Not m.lnErro1>0 
	= MessageBox('Erro na Gravação dos Dados do Cadastro de Operações.',48)
EndIf
