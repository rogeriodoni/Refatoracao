------------------------------------------------------------
* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
LParameters pEDN, pOpe

ThisForm.Empresa = Alltrim(Substr(pEDN, 01, 03))
ThisForm.Operacao = Alltrim(Substr(pEDN, 04, 20))
ThisForm.Numero = Alltrim(Substr(pEDN, 24, 06))

With Thisform
	.poDataMgr = CreateObject([fSqlConector], .Name)

	If (.poDataMgr.pnIdconn > 0)
		.Caption = [Títulos Gerados - Movimentação : ] + ;
				   ThisForm.Empresa + [ / ] + ;
				   ThisForm.Operacao + [ / ] + ;
				   ThisForm.Numero
		
		.cntSombra.lblSombra.Caption = .Caption
		.cntSombra.lblTitulo.Caption = .Caption
		
		lcQuery = [Select a.Titulos, a.Vencs, a.Valors, a.Moedas, a.Hists, a.Grupos, ] + ;
						 [b.Descrs as DGrupos, a.Contas, c.RClis as DContas, a.Dopcs, a.Numcs ] + ;
					[From SigMvCcr a ] + ;
					[Left Join SigCdGcr b On a.Grupos = b.Codigos ] + ;
					[Left Join SigCdCli c On a.Contas = c.IClis ] + ;
				   [Where a.EmpDopNums = '] + pEDN + [' And ] + ;
				   		 [a.Opers = '] + pOpe + [' ] + ;
				   [Order By a.Vencs, a.Titulos, a.Grupos, a.Contas]

		If (.poDataMgr.SqlExecute(lcQuery, [LocalTits]) < 1)
			=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (LocalTits)])
			Return .f.
		EndIf
		Select LocalTits
		Goto Top In LocalTits
		With .grdTitulos
			.RecordSource          = [LocalTits]
			.Column1.ControlSource = [LocalTits.Titulos]
			.Column2.ControlSource = [Ttod(LocalTits.Vencs)]
			.Column3.ControlSource = [LocalTits.Valors]
			.Column4.ControlSource = [LocalTits.Moedas]
			.Column5.ControlSource = [LocalTits.Hists]
			.Column6.ControlSource = [LocalTits.Grupos]
			.Column7.ControlSource = [LocalTits.Contas]
			.Refresh
			.AfterRowColChange()
		EndWith
		=DoDefault()
	Else
		=Messagebox([Impossível Efetuar Conexão Com o Servidor de Banco de Dados...], 0+48, [CreateObject])
		Return .f.
	EndIf
EndWith
* Fim **************************************************************************************
ENDPROC
