PROCEDURE msv_procurar

	ThisForm.poDataMgr.LocateCursor('crSigCdFrm','formas',ThisForm.Pagina.Dados.Get_formas.Value)

------------------------------------------------------------
* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
LPARAMETERS lcacesso

With ThisForm
	.poDatamgr = CreateObject('fSqlConector')
	With .poDatamgr
		.AddCursor('SigCdFrm','formas','crSigCdFrm','',ThisForm.pagina.lista.grade)
		.pfSqlTabela(1).pColuna('Formas','','','Formas',94,.t.)
	EndWith
	= DoDefault()
	.LockScreen = .f.
EndWith

ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Lista.Grupo_op.Click
------------------------------------------------------------
PROCEDURE Pagina.Lista.Grupo_op.Click
LPArameters Opcao

If !dodefault( Opcao )
	ThisForm.mAtivapagina1
	Return
Endif

Select CrSigCdFrm
Do Case
	Case Inlist( thisform.pcEscolha, 'INSERIR', 'PROCURAR' )
		ThisForm.Pagina.Dados.Get_Formas.SetFocus
	Case Inlist( thisform.pcEscolha, 'ALTERAR','EXCLUIR', 'CONSULTAR' )
		ThisForm.Pagina.Dados.Get_InFO.SetFocus
Endcase


ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
If Inlist( thisform.pcEscolha, 'INSERIR')
	If Empty( CrSigCdFrm.Formas )
		Messagebox( 'Forma de Pagamento Inválida.', 48 , '' )
		ThisForm.Pagina.Dados.Get_Formas.SetFocus
		Return
	Endif
	if ThisForm.poDataMgr.ChkRegister('SigCdFrm','Formas',ThisForm.Pagina.Dados.Get_Formas.Value)
		Messagebox( 'Forma de Pagamento já cadastrada.', 48,'' )
		ThisForm.Pagina.Dados.Get_Formas.SetFocus
		Return
	Endif
Endif

=DoDefault()
Thisform.mAtivapagina1
ENDPROC
