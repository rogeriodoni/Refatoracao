* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
lParameters pObj, pLin, pCnx

ThisForm.cntSombra.lblSombra.Caption = ThisForm.Caption
ThisForm.cntSombra.lblTitulo.Caption = ThisForm.Caption

This.ParentForm = pObj
This.poDataMgr  = pCnx

This.LockScreen = .t.

Select LocalBusLin
Go Top In LocalBusLin

This.ParentForm.Enabled = .f.

This.poDataMgr.CursorQuery([SigCdLin], [LocalLin], [Linhas], pLin, [FigJpgs, Descs])
If Not Used([LocalLin])
	=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (CursorQuery - LocalLin)])
	Return .f.
EndIf
Go Top In LocalLin
This.lblLinha.Caption = []
If Not Eof([LocalLin])
	This.lblLinha.Caption = LocalLin.Descs
EndIf

* Figura no Campo Memo do Cadastro de Linhas
lcArq = Sys(2023) + [\] + goSistema.Sys2015Tmp + [SigOpPrL.jpg]
This.ImgLinhas.Visible = .f.
This.ImgLinhas.Picture = []
If Not Empty(LocalLin.FigJpgs) And Not IsNull(LocalLin.FigJpgs)
	If (StrToFile(LocalLin.FigJpgs, lcArq) > 0)
		This.ImgLinhas.Picture = (lcArq)
		This.ImgLinhas.Visible = .t.
	EndIf
EndIf
Use In LocalLin

This.Grade.RecordSource          = [LocalBusLin]
This.Grade.Column1.ControlSource = [LocalBusLin.Selecao]
This.Grade.Column2.ControlSource = [LocalBusLin.DPros]
This.Grade.Column3.ControlSource = [LocalBusLin.PVens]
This.Grade.Column4.ControlSource = [LocalBusLin.MoeVs]
This.Grade.Column5.ControlSource = [LocalBusLin.Sqtds]
This.Grade.Column6.ControlSource = [LocalBusLin.CPros]
This.Grade.Column7.ControlSource = [LocalBusLin.CGrus]

This.Refresh
This.Grade.Column1.SetFocus
This.Grade.Column1.fwcheckbox1.SetFocus
This.LockScreen = .f.

ThisForm.Grade.AfterRowColChange()
* Fim **************************************************************************************
ENDPROC
