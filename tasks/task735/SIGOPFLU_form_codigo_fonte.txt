------------------------------------------------------------
PROCEDURE processamento
Local lcWhere,ldIni,ldFin,lcEmp,lcTit,lcSub

Private lcGru,ldDataI,ldDataF,lcCon

ldIni = ThisForm.GetDtInicial.Value
ldFin = ThisForm.GetDtFinal.Value
lcGru = Padr(ThisForm.GetGrupo.Value, 10)
lcCon = Padr(ThisForm.GetConta.Value, 10)

lcEmp     = Alltrim(CrSigCdEmp.CEmps + ' - ' + CrSigCdEmp.Razas)
lcTit     = Alltrim('Relatório de FollowUp de ' + Proper(lcGru))
lcSub     = Alltrim('Período : ' + Dtoc(ldIni) + ' à ' + Dtoc(ldFin))
llImpData = (ThisForm.ChkData.Value = 1)

Create Cursor CsCabecalho (Empresa c(80), Titulo c(80), SubTitulo c(80), ImpData l)
Insert Into CsCabecalho (Empresa, Titulo, SubTitulo, ImpData) Values (lcEmp, lcTit, lcSub, llImpData)

*******************************************************************************************

ldDataI = fDtoSQL(ldIni)
ldDataF = fDtoSQL(ldFin,'23:59:59')

lStrQuery = [Select a.IClis,a.Datas,a.Hists,a.Obs,b.Cpfs,b.RClis ]+;
			[From SigCdClh a, SigCdCli b ]+;
			[Where a.IClis = b.IClis And a.Datas BetWeen ?ldDataI And ?ldDataF And a.Grupos = ?lcGru ]+;
				Iif(!Empty(lcCon),[And a.IClis = ?lcCon ],'')+;
			[Order By a.IClis,a.Datas,a.Hists,b.Cpfs,b.RClis ]
	
If ThisForm.poDataMgr.SqlExecute(lStrQuery,'CsFwUp') < 1
	Return(.F.)
EndIf

Select CsFwUp
Go Top
ENDPROC

------------------------------------------------------------
* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
LParameters pDataSession, pGrupo, pConta

Private pcCliente,pcConta,pcGrupo,pcEmp

pcCliente = pConta
pcGrupo   = pgrupo
pcConta   = pConta
pcEmp     = _Empr

With ThisForm
	.poDataMgr = CreateObject('fSqlConector','SigOpFlu')
	If .poDataMgr.pnIdconn < 1
		MessageBox('Impossível efetuar conexão com Servidor de Banco de Dados...',48,'CreateObject')
		Return 0
	Else
		=DoDefault()
	EndIf
EndWith

If ThisForm.poDataMgr.SqlExecute([Select CEmps,Razas From SigCdEmp Where CEmps = ?pcEmp ],'CrSigCdEmp') < 1
	MessageBox('Favor reinicializar o processo !!!',16,'Falha na Conexão (CrSigCdEmp)')
	Return(.F.)
EndIf
Select CrSigCdEmp
Go Top

If ThisForm.poDataMgr.SqlExecute([Select Grupos,IClis,RClis From SigCdCli Where IClis = ?pcCliente ],'CrSigCdCli') < 1
	MessageBox('Favor reinicializar o processo !!!',16,'Falha na Conexão (CrSigCdCli)')
	Return(.F.)
EndIf
Select CrSigCdCli
Go Top

If ThisForm.poDataMgr.SqlExecute([Select Codigos,Descrs From SigCdGcr Where Codigos = ?pcGrupo ],'CrSigCdGcr') < 1
	MessageBox('Favor reinicializar o processo !!!',16,'Falha na Conexão (CrSigCdGcr)')
	Return(.F.)
EndIf
Select CrSigCdGcr
Go Top

ThisForm.GetDtInicial.Value = Date()
ThisForm.GetDtFinal.Value   = Date()
ThisForm.GetConta.Value     = pConta
ThisForm.GetDConta.Value    = CrSigCdCli.RClis
ThisForm.GetGrupo.Value     = pGrupo
ThisForm.GetDGrupo.Value    = CrSigCdGcr.Descrs

ThisForm.GetGrupo.Enabled   = Empty(ThisForm.GetGrupo.Value)
ThisForm.GetDGrupo.Enabled  = Empty(ThisForm.GetGrupo.Value)
ENDPROC

PROCEDURE btnReport.Click
If This.Value <> 3 && Se Não Clicou em Saída
	If Empty(ThisForm.GetDtInicial.Value)
		MessageBox('Data Inicial Inválida !!!',48,'Campo Obrigatório')
		ThisForm.GetDtInicial.SetFocus
		Return 0
	EndIf

	If Empty(ThisForm.GetDtFinal.Value)
		MessageBox('Data Final Inválida !!!',48,'Campo Obrigatório')
		ThisForm.GetDtFinal.SetFocus
		Return 0
	EndIf

	If ThisForm.GetDtFinal.Value < ThisForm.GetDtInicial.Value
		MessageBox('Data Final não pode ser maior que a Inicial !!!',48,'Seleção Inválida')
		ThisForm.GetDtInicial.SetFocus
		Return 0
	EndIf

	If Empty(ThisForm.GetGrupo.Value)
		MessageBox('Grupo Inválido !!!',48,'Campo Obrigatório')
		ThisForm.GetGrupo.SetFocus
		Return 0
	EndIf
EndIf

frmrelatorio::btnReport.Click
ENDPROC
