PROCEDURE Init
With ThisForm
	.poDataMgr = CreateObject([fSqlConector], .Name)
	
	If (.poDataMgr.pnIdconn > 0)
		With .poDataMgr
			.AddCursor([SigOpPed], [cIdChaves], [crSigOpPed], [], ThisForm.Pagina.Lista.Grade)
			.pfSqlTabela(1).pColuna([Ndopes], [], [], [Código]	   , 31, .t.)
			.pfSqlTabela(1).pColuna([Dopes ], [], [], [Operação]  ,150, .t.)
			.pfSqlTabela(1).pColuna([tPagos], [], [], [Pagamentos], 80)
			.pfSqlTabela(1).pColuna([tUtils], [], [], [Utilizados],108)
			
		EndWith

		=DoDefault()
		
		.Pagina.Lista.Grade.Columns(3).ControlSource = [Iif(crSigOpPed.tPagos = 1,'Pagos',Iif(crSigOpPed.tpagos = 2,'Não Pagos','Todos'))]
		.Pagina.Lista.Grade.Columns(4).ControlSource = [Iif(crSigOpPed.tUtils= 1,'Utilizados',Iif(crSigOpPed.tUtils= 2,'Não Utilizados','Todos'))]
		.Pagina.Lista.Grade.Column1.SetFocus
	Else
		=Messagebox([Impossível Efetuar Conexão Com o Servidor de Banco de Dados...], 0+48, [CreateObject])
		Return .f.
	EndIf
EndWith
* Fim **************************************************************************************



ENDPROC

------------------------------------------------------------
* PROCEDURE pagina.lista.grupo_op.Click
------------------------------------------------------------
PROCEDURE pagina.lista.grupo_op.Click
LParameters lnOpcao

If (Type('lnOpcao') = 'N')
	This.Value = lnOpcao
EndIf

Select crSigOpPed

If Not DoDefault(lnOpcao)
	ThisForm.mAtivapagina1
	Return 0
EndIf

Replace cIdChaves With Iif(Empty(cIdChaves), fUniqueIds(), cIdChaves)

loDad = ThisForm.Pagina.Dados

Do Case
	Case Inlist(ThisForm.pcEscolha, 'INSERIR','PROCURAR','ALTERAR')
		loDad.getnopes.Setfocus

	Case Inlist(ThisForm.pcEscolha, 'CONSULTAR', 'EXCLUIR')
		loDad.Grupo_Salva.Salva.Setfocus

EndCase

Select crSigOpPed

loDad.Refresh
* Fim **************************************************************************************
ENDPROC

------------------------------------------------------------
* PROCEDURE pagina.dados.grupo_salva.salva.Click
------------------------------------------------------------
PROCEDURE pagina.dados.grupo_salva.salva.Click
Local loDad

loDad = ThisForm.Pagina.Dados

If InList(ThisForm.pcEscolha, 'INSERIR', 'ALTERAR')
	If Empty(loDad.GetNopes.Value)
		=MessageBox('Obrigatório Informar a Operação!!!', 48, '')
		loDad.GetNopes.SetFocus
		Return 0
	EndIf
		
	If loDad.OptPagos.Value < 1
		=MessageBox('Obrigatório Informar a Pagamento!!!', 48, '')
		loDad.OptPagos.option1.SetFocus
		Return 0	
	EndIf 
	
	If loDad.OptUtils.Value < 1
		=MessageBox('Obrigatório Informar o Utilização!!!', 48, '')
		loDad.OptUtils.option1.SetFocus
		Return 0	
	EndIf

	If (Thisform.pcEscolha = 'INSERIR' )
		lcQuery = [Select * ] + ;
				    [From SigOpPed ] + ;
				   [Where Dopes = '] + Padr(loDad.GetDopes.Value, 20) + [']

		If (ThisForm.poDataMgr.SqlExecute(lcQuery, [crBusca]) < 1)
			=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (crBusca)])
			Return 0
		EndIf
		Select crBusca
		Go Top In crBusca
		If Not Eof([crBusca])
			=MessageBox([Operação já Cadastrada!!!], 48, [Atenção!!!])
			loDad.getnopes.SetFocus
			Return 0
		EndIf

	EndIf

	If (Thisform.pcEscolha = [EXCLUIR])
		Delete In crSigOpPed
	EndIf 
EndIf
Select crSigOpPed

=DoDefault()

ThisForm.mAtivaPagina1
* Fim ************************************************************************************** 
ENDPROC

PROCEDURE Valid
Local loLista

If Not Empty(This.Value)
	loLista = CreateObject('fwBuscaExt', ThisForm.poDataMgr.pnIdConn, 'SigCdOpe', ;
						   'crListaRemota', 'Ndopes', This.Value, 'Seleção', 1000)

	If Not loLista.plAchouRegistro
		loLista.mAddColuna('Ndopes', '9999', 'Código')
		loLista.mAddColuna('Dopes', '', 'Descrição')
		loLista.Show()
	EndIf
	
	This.Parent.Getnopes.Value = crListaRemota.Ndopes
	This.Parent.GetDopes.Value = crListaRemota.Dopes

	Use In crListaRemota
Else
	This.Parent.Getnopes.Value = ''
	This.Parent.GetDopes.Value = ''
EndIf

This.Parent.Getnopes.Refresh
This.Parent.GetDopes.Refresh

* Fim **************************************************************************************

ENDPROC

------------------------------------------------------------
* PROCEDURE When
------------------------------------------------------------
PROCEDURE When
Return (InList(Thisform.pcEscolha ,[INSERIR],[PROCURAR]))
ENDPROC
