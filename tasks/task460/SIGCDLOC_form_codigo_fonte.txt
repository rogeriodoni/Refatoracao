PROCEDURE Init
LParameters lcAcesso

With ThisForm
	.poDataMgr = CreateObject('fSqlConector', .Name)
	
	With .poDataMgr
		.AddCursor('SigCdLoc', 'Cods', 'crSigCdLoc', '', ThisForm.Pagina.Lista.Grade)
		.pfSqlTabela(1).pColuna('Cods',     '', '', 'Código',  080, .t.)
		.pfSqlTabela(1).pColuna('cIdConns', '', '', 'Conexão', 150, .f.)
		.pfSqlTabela(1).pColuna('Drives',   '', '', 'Caminho Para Tabelas', 400, .f.)
	EndWith
EndWith

=DoDefault()

ThisForm.Pagina.Lista.Grade.Column1.SetFocus
* Fim **************************************************************************************

ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Lista.Grupo_op.Click
------------------------------------------------------------
PROCEDURE Pagina.Lista.Grupo_op.Click
Lparameters Opcao

If (Type('Opcao') = 'N')
	This.Value = Opcao
EndIf

Select crSigCdLoc

If Not DoDefault(Opcao)
	ThisForm.mAtivapagina1
	Return 0
EndIf

Do Case
	Case Inlist(ThisForm.pcEscolha, 'PROCURAR', 'INSERIR')
		ThisForm.Pagina.Dados.getCods.Setfocus	

	Case (ThisForm.pcEscolha = 'ALTERAR')
		ThisForm.Pagina.Dados.getConexao.Setfocus	

	Case Inlist(ThisForm.pcEscolha, 'CONSULTAR', 'EXCLUIR')
		ThisForm.Pagina.Dados.Grupo_Salva.Salva.Setfocus
EndCase

Select crSigCdLoc

ThisForm.Pagina.Dados.Refresh
* Fim **************************************************************************************

ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
If (ThisForm.pcEscolha = 'INSERIR')
   If Empty(ThisForm.Pagina.Dados.getCods.Value)
	  =MessageBox('Obrigatório Informar o Código!!!', 48, '')
	  ThisForm.Pagina.Dados.getCods.SetFocus
	  Return 0
   EndIf

   If Empty(ThisForm.Pagina.Dados.getConexao.Value)
	  =MessageBox('Obrigatório Informar a Conexão!!!', 48, '')
	  ThisForm.Pagina.Dados.getConexao.SetFocus
	  Return
   Endif
Endif

=DoDefault()

ThisForm.mAtivapagina1
* Fim **************************************************************************************

ENDPROC

Local loLista

If Not Empty(This.Value)
	loLista = CreateObject('fwBuscaExt', ThisForm.poDataMgr.pnIdConn, 'SigConn', ;
						   'crListaRemota', 'cIdConns', This.Value, 'Seleção', .t.)

	If Not loLista.plAchouRegistro
		loLista.mAddColuna('cIdConns', '', 'Conexão')

	EndIf

Else
	This.Value = ''
EndIf

PROCEDURE Valid
Local lcDestino

If Not Empty(This.Value) 
	If Right(Alltrim(This.Value), 1) <> '\'
		This.Value = Alltrim(This.Value) + '\'
	EndIf

	lcDestino = Alltrim(This.Value) + 'SigCdLoc.DBF'
	
	If Not File(lcDestino)
		=Messagebox('O Diretório de Destino "' + Alltrim(This.Value) + '" Não É Um Diretório Válido!!!' + Chr(13) + ;
					'O Mesmo Pode Não Existir, Não Estar Acessível ou Não Conter o Arquivo "SigCdLoc.DBF"!!!', 0+48, ;
					'Diretório de Destino Inválido ou Inexistente')
		Return 0
	EndIf 
EndIf
