------------------------------------------------------------
* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
LParameters _Acesso

WITH Thisform

	.poDatamgr = CreateObject('fSqlConector')

	If Not .poDatamgr.pnIdconn>0
		Messagebox('Impossivel Efetuar Conexão com Servidor de Banco de Dados.',(0+48),goSistema.TituloSistema)
		Return(.f.)
	Endif

	With .poDatamgr
		.AddCursor('SigEtOco','Cods','crSigEtOco','Cods',ThisForm.pagina.lista.grade)
		.pfSqlTabela(1).pColuna('Cods','','','Código',50,.t.)
		.pfSqlTabela(1).pColuna('Descrs','','','Descrição',290,.t.)
	EndWith
Endwith	

=DoDefault()

ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Lista.Grupo_op.Click
------------------------------------------------------------
PROCEDURE Pagina.Lista.Grupo_op.Click
LPARAMETERS opcao

If !DoDefault( Opcao )
	ThisForm.mAtivapagina1
	Return
Endif

Do Case
	Case ThisForm.pcEscolha = 'INSERIR'
		ThisForm.Pagina.Dados.getCods.SetFocus

	Case ThisForm.pcEscolha = 'PROCURAR'
		ThisForm.Pagina.Dados.getCods.SetFocus

	Case Inlist( ThisForm.pcEscolha, 'EXCLUIR', 'CONSULTAR' )
		ThisForm.Pagina.Dados.Grupo_Salva.Salva.SetFocus

	Case ThisForm.pcEscolha = 'ALTERAR'
		ThisForm.Pagina.Dados.getDescrs.SetFocus
Endcase
* Fim ********************************************************************
ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
If Inlist(ThisForm.pcEscolha, 'INSERIR', 'ALTERAR') .And. IsEmpty(Descrs)
	=MessageBox('Descrição Inválida!!!', 48, '')
	ThisForm.Pagina.Dados.getDescrs.SetFocus
	Return
Endif
=DoDefault()
ThisForm.mAtivapagina1
* Fim ********************************************************************
ENDPROC

ENDPROC

PROCEDURE Valid
If (ThisForm.pcEscolha = 'INSERIR')
	If IsEmpty(This.Value)
		Thisform.Pagina.Dados.getCods.Refresh
	Else
		If ThisForm.podatamgr.ChkRegister('SigEtOco', 'Cods', This.Value)
			=MessageBox('Código Já Cadastrado!!!', 48, '')
			This.Value = ''
			Return 0
		Endif
	Endif
