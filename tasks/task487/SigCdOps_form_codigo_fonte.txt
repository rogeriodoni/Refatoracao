------------------------------------------------------------

PROCEDURE Init
LParameters _Acesso

Private lcProg, pDop, pStI, pStF

pDop   = []
pStI   = 0
pStF   = 1
lcProg = []

With Thisform
	.poDataMgr = CreateObject([fSqlConector], .Name, .t.)

	If Not .poDatamgr.pnIdconn>0
		Messagebox('Impossivel Efetuar Conexão com Servidor de Banco de Dados.',(0+48),goSistema.TituloSistema)
		Return(.f.)
	Endif

	lcQryOpe = [Select a.*, b.OpeInatvs as Inativas, Left(Right(d.ConfigGers, 89), 30) as DopMenus ] + ;
				 [From SigCdOpe a ] + ;
		    	 [Left Join SigOpCdc b On a.Dopes = b.Dopes ] + ;
		    	 [Left Join SigOpCdd d On a.Dopes = d.Dopes ] + ;
				[Where b.OpeInatvs In (?pStI, ?pStF)]
	
	With .poDatamgr
		.AddCursor([SigCdOpe], [Dopes], [crSigCdOpe], [], ThisForm.Pagina.Lista.Grade, lcQryOpe)
		.pfSqlTabela(1).pColuna('Dopes','','','Descrição',150,.t.)
		.pfSqlTabela(1).pColuna('NDopes','','','Nº OP',39,.t.)
		.pfSqlTabela(1).pColuna('Abrevs','','','Reduzida',57,.t.)
		.pfSqlTabela(1).pColuna('CMoes','','','Moeda',40,.f.)
		.pfSqlTabela(1).pColuna('Nivels','','','Sub-Nivel',150,.t.)
		.pfSqlTabela(1).pColuna('Menus','','','Menu',80,.t.)
