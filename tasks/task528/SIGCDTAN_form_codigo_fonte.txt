------------------------------------------------------------
* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
LPARAMETERS lcacesso

With ThisForm
	.poDatamgr = CreateObject('fSqlConector')
	With .poDatamgr
		.AddCursor('SigPrTpa','codigos','crSigPrTpa','',ThisForm.pagina.lista.grade)
		.pfSqlTabela(1).pColuna('codigos','','','Código',80,.t.)
		.pfSqlTabela(1).pColuna('descs','','','Descrição',337,.t.)
	EndWith
	= DoDefault()
	.LockScreen = .f.
EndWith
ENDPROC

------------------------------------------------------------
* PROCEDURE msv_procurar
------------------------------------------------------------
PROCEDURE msv_procurar
Local lcBusca

If .Not.IsEmpty(ThisForm.Pagina.Dados.Get_cod.Value)
	ThisForm.poDataMgr.LocateCursor('crSigPrTpa','codigos',ThisForm.Pagina.Dados.Get_cod.Value)
EndIf

ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Lista.Grupo_op.Click
------------------------------------------------------------
PROCEDURE Pagina.Lista.Grupo_op.Click
LParameters opcao

If !dodefault( opcao )
	ThisForm.mAtivapagina1
	Return
Endif

Do Case
	Case Inlist( ThisForm.pcEscolha, 'INSERIR', 'PROCURAR' )
		ThisForm.Pagina.Dados.Get_Cod.SetFocus
	Case ThisForm.pcEscolha = 'ALTERAR'
		ThisForm.Pagina.Dados.Get_Desc.SetFocus
Other
	ThisForm.Pagina.Dados.Grupo_Salva.Salva.SetFocus
Endcase

ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
If ThisForm.pcEscolha = 'INSERIR'
	If IsEmpty( ThisForm.Pagina.Dados.Get_Cod.Value )
		Messagebox( 'Codigo não pode ficar em branco!', 48, '' )
		ThisForm.Pagina.Dados.Get_Cod.SetFocus
		Return
	Endif
	If ThisForm.poDataMgr.ChkRegister('SigPrTpa','codigos',ThisForm.Pagina.Dados.Get_Cod.Value)
		Messagebox( 'Tipo de Análise já Cadastrada!', 48, '' )
		ThisForm.Pagina.Dados.Get_Cod.SetFocus
		Return
	Endif

Endif

=DoDefault()

ThisForm.mAtivapagina1

ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Dados.Grupo_Salva.Cancelar.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Cancelar.Click
=DoDefault()
If ThisForm.plCancelar
	ThisForm.mAtivapagina1
Endif

ENDPROC
