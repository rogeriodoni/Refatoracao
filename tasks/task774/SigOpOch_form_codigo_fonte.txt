PROCEDURE Init
LParameters poForm, pnNoper, pcOper

ThisForm.cntSombra.lblSombra.Caption = ThisForm.Caption
ThisForm.cntSombra.lblTitulo.Caption = ThisForm.Caption

With ThisForm
	.poDataMgr = CreateObject('fSqlConector')
EndWith

lcSql = [Select * From SigMvCcr where Nopers = ]+Str(pnNoper)+[ And Opers = ']+pcOper+[']
Thisform.PodataMgr.SqlExecute(lcSql,'ContaCC')

lcSql = [Select * From SigCqOch Where Nopers = ]+Str(pnNoper)+[ Order by Datars,Datas ]
Thisform.PodataMgr.SqlExecute(lcSql,'Movi')

Select *, Padr(Nvl(ObsAudts,' '),50) as ObsAudt2 From Movi Into Cursor Movimento Order by datars,datas

Select ContaCc
Go Top

With ThisForm.grd_lancamento
	.RecordSource = "Contacc"
	.Col_data.ControlSource = "Datas"
	.Col_saldo.ControlSource = "Saldos"
	.Col_hist.ControlSource = "Hists"
	.col_valor.ControlSource = "Valors"
	.col_oper.ControlSource = "Opers"
	.col_emp.ControlSource = "Emps"
	.Col_Saldo.InputMask = '999,999,999.99'
	.Col_Saldo.Text1.InputMask = '999,999,999.99'
	.SetAll('DynamicForeColor','Iif(Concs, RGB(0,0,255), RGB(0,0,0) )','Column')
	.SetAll('DynamicBackColor','Iif(!IsEmpty(DtAudits), RGB(220,255,220), RGB(255,255,255) )','Column') && Verde Claro
EndWith

With ThisForm.GridFollowUp
	.RecordSource = 'Movimento'
	.Column1.ControlSource = 'Movimento.Datars'
	.Column2.ControlSource = 'Movimento.TpLancs'
	.Column3.ControlSource = 'Movimento.datas'
	.Column4.ControlSource = 'Movimento.Usuars'
	.Column5.ControlSource = 'Movimento.ObsAudt2'
	.Refresh
EndWith
ENDPROC
