------------------------------------------------------------
* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
LPARAMETERS lcacesso

With ThisForm
	.poDatamgr = CreateObject('fSqlConector')
	With .poDatamgr
		.AddCursor('SigCdMtl','cidchaves','crSigCdMtl','',ThisForm.pagina.lista.grade)
		.pfSqlTabela(1).pColuna('tipos','','','Tipo',30,.t.)
		.pfSqlTabela(1).pColuna('codigos','','','Código',75,.t.)
		.pfSqlTabela(1).pColuna('descs','','','Descrição',150,.t.)
	EndWith
	= DoDefault()
	.LockScreen = .f.
EndWith

ENDPROC

------------------------------------------------------------
* PROCEDURE msv_procurar
------------------------------------------------------------
PROCEDURE msv_procurar
Local lcBusca

If .Not.IsEmpty(ThisForm.Pagina.Dados.Getcodigos.Value)
	ThisForm.poDataMgr.LocateCursor('crSigCdMtl','codigos',ThisForm.Pagina.Dados.Getcodigos.Value)
Else
	If .Not.IsEmpty(ThisForm.Pagina.Dados.Getdescrs.Value)
		ThisForm.poDataMgr.LocateCursor('crSigCdMtl','descs',ThisForm.Pagina.Dados.Getdescrs.Value)
	EndIf
EndIf

ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Lista.Grupo_op.Click
------------------------------------------------------------
PROCEDURE Pagina.Lista.Grupo_op.Click
LParameters opcao

If !dodefault( opcao )
	ThisForm.mAtivaPagina1
	Return
Endif

Do Case
	Case Inlist( ThisForm.pcEscolha, 'INSERIR')
		ThisForm.Pagina.Dados.Gettipos.SetFocus
	Case Inlist( ThisForm.pcEscolha, 'PROCURAR')
		ThisForm.Pagina.Dados.Getcodigos.SetFocus
	Case ThisForm.pcEscolha = 'ALTERAR'
		ThisForm.Pagina.Dados.GetDescrs.SetFocus
Other
	ThisForm.Pagina.Dados.Grupo_Salva.Salva.SetFocus
Endcase

ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
Local laCampos(2,2)

If ThisForm.pcEscolha = 'INSERIR'
   If IsEmpty(ThisForm.Pagina.Dados.getCodigos.Value)
	  MessageBox( 'Obrigatório Informar o Código!!!', 48, '' )
	  ThisForm.Pagina.Dados.getCodigos.SetFocus
	  Return
   Endif
   	m.laCampos(1,1) = 'tipos'
	m.laCampos(1,2) = thisform.pagina.dados.gettipos.value
	m.laCampos(2,1) = 'codigos'
	m.laCampos(2,2) = thisform.pagina.dados.getcodigos.value

	If ThisForm.poDataMgr.ChkRegister('SigCdMtl',@m.laCampos)
		Messagebox( 'Tipo e Código já Cadastrado!', 48, '' )
		ThisForm.Pagina.Dados.Gettipos.SetFocus
		Return
	Endif
	Select CrSigCdMtl
	Replace cidchaves with fUniqueIds()

Endif
=DoDefault()
ThisForm.mAtivapagina1

ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Dados.Grupo_Salva.Cancelar.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Cancelar.Click
=DoDefault()
If ThisForm.plCancelar
	ThisForm.mAtivapagina1
Endif

ENDPROC
