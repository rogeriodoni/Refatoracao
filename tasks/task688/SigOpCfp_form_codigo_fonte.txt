PROCEDURE mctrlbotoes
LParameters plHabilita

Local lnI

If Type('plHabilita') # 'L'
	plHabilita = .F.
Endif

With Thisform.ParentForm.Pagina.Lista
	For lnI=1 to .Grupo_Op.ButtonCount
		If .Grupo_Op.Buttons(lnI).BaseClass = 'Commandbutton' And .Grupo_Op.Buttons(lnI).Visible
			.Grupo_Op.Buttons(lnI).Enabled = plHabilita
		EndIf
	EndFor	
	.Grupo_Saida.Sair.Enabled = plHabilita
EndWith



ENDPROC

PROCEDURE Init
LParameters poForm, pnDataSes, pcEscolha

Local lcOperacao

Set DataSession to pnDataSes

Thisform.ParentForm.Enabled  = .F.
lcOperacao = crSigCdOpe.Dopes

	.Left    = Thisform.ParentForm.Left + ;
			   Thisform.ParentForm.Pagina.Lista.Grade.Left + ;
			   Thisform.ParentForm.Pagina.Lista.Grade.Columns(1).Width + ;
			   Thisform.ParentForm.Pagina.Lista.Grade.Columns(3).Width + 16

	.cmdInserir.Visible = Iif( Type( 'Thisform.PcEscolha' ) # 'C', .T., ;
						  InList( Thisform.PcEscolha, 'INSERIR', 'ALTERAR' ) )
	.cmdInserir.Enabled = Iif( Type( 'Thisform.PcEscolha' ) # 'C', .T., ;
						  InList( Thisform.PcEscolha, 'INSERIR', 'ALTERAR' ) )
	.cmdExcluir.Visible = Iif( Type( 'Thisform.PcEscolha' ) # 'C', .T., ;
						  InList( Thisform.PcEscolha, 'INSERIR', 'ALTERAR' ) )
	.cmdExcluir.Enabled = Iif( Type( 'Thisform.PcEscolha' ) # 'C', .T., ;
						  InList( Thisform.PcEscolha, 'INSERIR', 'ALTERAR' ) )

PROCEDURE Valid
LOCAL xBrox

If !IsEmpty( This.Value )
	If !SigCdOpe.podatamgr.ChkRegister( 'SigCdObs', 'Codigos', This.Value )
		xBrox = CreateObject('fwBuscaExt',SigCdOpe.poDatamgr.pnIdconn,'SigCdObs','crSigCdObs','Codigos',This.Value,'Cadastro de Observações',.t.,'Codigos')

		If  Not xBrox.plAchouRegistro
			xBrox.mAddColuna('Codigos', '999999', 'Código', .t.)
			xBrox.mAddColuna('Observas', Replicate('x',40), 'Observação',.f.,[Padr( MLine( Nvl(crSigCdObs.Observas,''), 1 ), 40 )])
			xBrox.Show()
		EndIf 

		This.Value = Iif( Lastkey() == 27, 0, crSigCdObs.Codigos )
		Use in crSigCdObs
	Endif
	This.refresh
Endif

If Not Empty(This.Value) And Not Empty(TmpCfoOp.ObsInis) And ;
   (TmpCfoOp.ObsInis > This.Value)
	=MessageBox('A Observação Inicial Não Pode Ser Maior do Que a Padrão!!!', 0+16, '')
	Return 0
EndIf

If Not Empty(This.Value) And Not Empty(TmpCfoOp.ObsFins) And ;
   (TmpCfoOp.ObsFins < This.Value)
	=MessageBox('A Observação Final Não Pode Ser Menor do Que a Padrão!!!', 0+16, '')
	Return 0
EndIf
ENDPROC

PROCEDURE Click
If Thisform.cmdInserir.Visible Or Thisform.cmdExcluir.Visible
	**If ( Thisform.lHouveExcl Or Thisform.lHouveIns )	
	
		Select TmpCfoOp
		LOCATE
		Delete For Cfops = Space(10)
		Replace all dopes     With CrSigCdOpe.Dopes,;
				    CidChaves With fUniqueIds() FOR EMPTY(cIdChaves)

		Select Cfops, sum(1) From TmpCfoOp Group by 1 having sum(1) > 1 Into Cursor SecondKey
		Select SecondKey
		If Reccount() > 0
			MessageBox('Existem Cfops em Duplicidade. Favor Acertar!!!',0+16,'Atenção')
			Return .f.
		EndIf
		
		Select CrSigCdOop
		Zap
		Append From Dbf('TmpCfoOp')
		
		lcSql = [Delete From SigCdOop Where Dopes = ']+CrSigCdOpe.dopes+[']

		m.lnErro1 = SigCdOpe.poDatamgr.pnIdConn

		With SigCdOpe
			If m.lnErro1 > 0
				m.lnErro1 = SigCdOpe.PodataMgr.Sqlexecute(lcSql,'')
			Endif
			If m.lnErro1>0 
				m.lnErro1 = Iif(.poDataMgr.UpDate('crSigCdOop'),1,-1)
			EndIf 
			If m.lnErro1>0 
				m.lnErro1 = .poDataMgr.Commit()
			EndIf 
			If Not m.lnErro1>0 
				.poDataMgr.RollBack()
			EndIf 
		EndWith 
		If Not m.lnErro1>0 
			= MessageBox('Erro na Gravação dos Dados do Cadastro de Operações.',48)
		EndIf 
	**EndIf
ENDIF

Thisform.ParentForm.Enabled = .T.
Thisform.mCtrlBotoes(.T.)
ThisForm.Release
ENDPROC
