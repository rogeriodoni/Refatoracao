PROCEDURE checacampos
Lparameters pCpo, pTab, pKey

Local lcRet

lcRet = pCpo

If Not Empty(pCpo) 
	ThisForm.poDataMgr.CursorQuery(pTab, [LocalBus], pKey, pCpo)
	If Not Used([LocalBus])
		=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (CursorQuery - ] + pTab + [)])
		lcRet = Iif(Type([pCpo]) = [N], 0, [])
	EndIf

	If Not Empty(lcRet)
		Go Top In LocalBus
		If Eof([LocalBus])
			lcRet = Iif(Type([pCpo]) = [N], 0, [])
		EndIf
		Use In LocalBus
	EndIf
EndIf

Return lcRet
* Fim **************************************************************************************
ENDPROC

PROCEDURE exportacfgnf
Local loXml, lcArq, lcKey

loXml = CreateObject([XMLAdapter])
lcArq = Alltrim(ThisForm.pgCopias.pgExportar.getArqXML.Value)
lcKey = Padr(ThisForm.PgCopias.pgExportar.getOperacao.Value, 10)

ThisForm.poDataMgr.CursorQuery([SigCnFnf], [XMLCfgnf], [Series], lcKey)
If Not Used([XMLCfgnf])
	=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (CursorQuery - XMLCfgnf)])
	Return .f.
Else
	Select XMLCfgnf
	Go Top In XMLCfgnf
	If Eof([XMLCfgnf])
		=MessageBox([A Serie Não Foi Encontrada!!!], 16, [Atenção!!!])
		Return .f.
	EndIf
	Index On series Tag series
EndIf

With loXml
	.RespectCursorCP = .t.
	.RespectNesting  = .t.
	.UTF8Encoded     = .t. 	
	.FormattedOutput = .t.
	.XMLName         = Strconv([series], 12)

	* Adiciona as Tabelas a Serem Enviadas Para o XML
	.AddTableSchema([XMLCfgnf])

	* Gera a String com o XML
	.ToXML([lcStr])

	* Grava a String com o XML Em Um Arquivo
	=StrToFile(lcStr, lcArq, 0)

	Use In XMLCfgnf

	loXml = Null
EndWith
* Fim **************************************************************************************
ENDPROC

		llRet = Thisform.ExportaFti()

		=MessageBox([O Programa Não Está Adaptado Para Exportar Dados] + Chr(13) + ;
					[Do Formulário ] + lcFrm + [!!!], 0+16, [Atenção!!!])
		llRet = .f.
EndCase

Return llRet

ENDPROC

PROCEDURE importaopp
Local loXml, lcArq, lcKey, llOks

loXml = CreateObject([XMLAdapter])
lcArq = Alltrim(ThisForm.pgCopias.pgImportar.getArqXML.Value)
lcKey = Padr(ThisForm.pgCopias.pgImportar.getOperacao.Value, 20)

* Carrega Um Arquivo XML e Joga Seu Conteúdo Em Memória
loXml.LoadXML(lcArq, .t.)

If (loXml.Tables.Count < 1)
	=MessageBox([O Arquivo XML Não é Válido Ou Não Possui Dados!!!], 0+16, [Atenção!!!])
	Return .f.
EndIf

* Corre Cada Um Dos Cursores Carregados Em Memória do XML e Joga em Seus Cursores
For I = 1 To loXml.Tables.Count
	loXml.Tables(I).ToCursor()
Next

If Not Used([XMLopp])
	=MessageBox([O Arquivo XML Não é Válido Para Essa Importação!!!], 0+16, [Atenção!!!])
	Return .f.
ELSE
	If Empty(lcKey)
		lcKey = Padr(XMLopp.dopps, 20)
		ThisForm.pgCopias.pgImportar.getOperacao.Value = lcKey
		ThisForm.pgCopias.pgImportar.getOperacao.Refresh
	EndIf
EndIf

llOks = .t.

ThisForm.poDataMgr.CursorQuery([SigCdOpd], [Localopp], [dopps], lcKey)
If Not Used([Localopp])
	=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (CursorQuery - Localopp)])
	llOks = .f.
Else
	Select Localopp
	Go Top In Localopp
	If Not Eof([Localopp])
		=MessageBox([A Dopps "] + Alltrim(lcKey) +  [" Já Está Cadastrada No Sistema!!!], 16, [Atenção!!!])
		ThisForm.pgCopias.pgImportar.getOperacao.Value = []
		ThisForm.pgCopias.pgImportar.getOperacao.Refresh
		llOks = .f.
	EndIf
EndIf

If llOks
	Zap In crSigCdMpd
	
	Select XMLopp
	Go Top In XMLopp
	If Not Eof([XMLopp])
		Scatter MemVar Memo
		m.dopps = ThisForm.pgCopias.pgImportar.getOperacao.Value
		m.ndopps = fGerUniqueKey('SIGCDOPD')
		m.cIdChaves = fUniqueIds()

		Select crSigCdMpd
		Append Blank In crSigCdMpd
		lnFim = AFields(laCpo, [XMLopp])
		For I = 1 To lnFim
			lcNom = Alltrim(laCpo(I, 1))
			lvIns = m.&lcNom.
			
			If Not IsNull(lvIns) And (Type([crSigCdMpd.&lcNom.]) = Type([lvIns]))
				Replace &lcNom. With lvIns In crSigCdMpd
			EndIf
		Next
	EndIf

	If llOks And Not ThisForm.poDataMgr.Update([crSigCdMpd])
		=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (Update - crSigCdMpd)])
		llOks = .f.
	EndIf

	If llOks And (ThisForm.poDataMgr.Commit() < 1)
		=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (Commit)])
		llOks = .f.
	EndIf

	If Not llOks
		ThisForm.poDataMgr.RollBack()
	EndIf
EndIf

loXml = Null

Return llOks
* Fim **************************************************************************************
ENDPROC

------------------------------------------------------------
* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
Lparameters pFrm, pKey, pTip, pLbl, pWid, pAut, pCon
* Parâmetros: pFrm - Formulário Que Realizou a Chamada			
*			  pKey - Chave de Procura							
*			  pTip - Tipo de Processo: [E]xportar ou [I]mportar	
*			  pLbl - Título do Campo de Digitação de Dados		
*			  pWid - Quantidade Máxima de Caracteres da Chave	
*			  pAut - Informa se é uma Geração Automática de Bkp.
*			  pCon - Conexão Quando Passada					    
Local lcLbl, lnWid

ThisForm.cntSombra.lblSombra.Caption = ThisForm.Caption
ThisForm.cntSombra.lblTitulo.Caption = ThisForm.Caption

If (Type([pKey])<>[C])
	=MessageBox([O Segundo Parâmetro Passado Não é Válido!!!], 0+16, [Atenção!!!])
	Return .f.
EndIf

If (Type([pTip])<>[C]) Or Not InList(pTip, [E], [I])
	=MessageBox([O Tipo Passado Não é Válido!!! Use "E" ou "I"!!!], 0+16, [Atenção!!!])
	Return .f.
EndIf

lcLbl = Iif(Type([pLbl])=[C], pLbl, [Chave])
lnWid = Iif(Type([pWid])=[N], pWid, 20)

With ThisForm
	.ParentForm 	= pFrm
	.DataSessionId 	= pFrm.DataSessionId
	.Automatica 	= pAut And (pTip = [E]) And (Alltrim(Upper(.ParentForm.Name)) = [SIGCDOPE])
	If .Automatica
		.poDataMgr 	= pCon
	Else
		.poDataMgr 	= pFrm.poDataMgr
	EndIf

	If .Automatica And (.poDataMgr.pnIdconn > 0)
		With .poDataMgr
			.Addcursor([SigOpBkp], [pkChaves], [crSigOpBkp])
		EndWith
	EndIf

	.pgCopias.pgExportar.lblTitulo.Caption = Alltrim(lcLbl + [ :])
	.pgCopias.pgImportar.lblTitulo.Caption = Alltrim(lcLbl + [ :])

	lcArq = Chrtran(Alltrim(pKey), [ "'!@#$%¨&*()-+=´`{}~^/?;:.>,<|\], [________________________________])
	
	.ArqXML = [C:\] + lcArq + [.XML]

	If .Automatica
		ltDat = .poDataMgr.GetDatetime()
		lcUsu = Alltrim(Chrtran(Alltrim(goSistema.Usuario), [ "'!@#$%¨&*()-+=´`{}~^/?;:.>,<|\], [________________________________]))
		lcDat = Alltrim(Chrtran(Ttoc(ltDat), [ :/], [___]))
		
		.ArqXML = Sys(2023) + [\] + lcArq + [-] + lcUsu + [-] + lcDat + [.XML]

		Zap In crSigOpBkp

		Insert Into crSigOpBkp (Usuars, DtAlts, Dopes, Redes, pkChaves) ;
					    Values (goSistema.Usuario, ltDat, pKey, Sys(0), fUniqueIds())
	EndIf
	
	.pgCopias.pgExportar.getOperacao.Value = pKey
	.pgCopias.pgExportar.getOperacao.Width = (lnWid * 7) + 10
	.pgCopias.pgExportar.getArqXML.Value   = .ArqXML
	.pgCopias.pgExportar.Enabled 		   = (pTip = [E])

	.pgCopias.pgImportar.getOperacao.Value = pKey
	.pgCopias.pgImportar.getOperacao.Width = (lnWid * 7) + 10
	.pgCopias.pgImportar.getArqXML.Value   = []
	.pgCopias.pgImportar.Enabled		   = (pTip = [I])

	.pgCopias.ActivePage = Iif(pTip = [E], .pgCopias.pgExportar.PageOrder, .pgCopias.pgImportar.PageOrder)
	
	If .Automatica
		.Processar()

		lcNom = Alltrim(.pgCopias.pgExportar.getArqXML.Value)
		If File(lcNom)
			lcArq = FileToStr(lcNom)

			Replace Xmls With lcArq In crSigOpBkp 
			
			.poDataMgr.Update([crSigOpBkp])
			* Nota: o Commit Será Dado na Conexão do Form Que Acionou a Rotina 

			Delete File lcNom
		EndIf

		If (.poDataMgr.Commit() < 1)
			=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (Commit)])
			.poDataMgr.RollBack()
			.Release
			Return .f.
		EndIf

		.Release
		Return .f.
	EndIf
EndWith	
* Fim **************************************************************************************
ENDPROC
