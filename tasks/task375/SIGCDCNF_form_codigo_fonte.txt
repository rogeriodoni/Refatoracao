------------------------------------------------------------
* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init

LParameters _acesso

Private pEmisDe,pEmisAte,pSeries,pNFisDe,pNFisAte,pCancelas,pItras,pDopes,pNumes,pIclis,pCpros,pFPags,lcDtIni,lcDtFim,;
		pItras,pDopes,pNumes,pIclis,pCpros,pFpags,pNEmps,pOpE,pOpS

ThisForm.poDataMgr = CreateObject('fSqlConector')

thisform.pagina.lista.cntFiltros.getEmp.Value = _empr

thisform.pemps = _empr
pcemps     = _empr	
pSeries    = ''
pEmisDe    = Null
pEmisAte   = Null
pOpE	   = [E]
pOpS       = [S]
pNFisDe    = ''
pNFisAte   = ''
pCancelas  = .F.
pItras     = ''
pDopes     = ''
pNumes     = 0
pIclis     = ''
pCpros     = ''
pFPags     = ''
pNEmps     = ''
llCfg      = .F.
If ThisForm.poDataMgr.SqlExecute([Select * From SigCnFNF Where emps = ?_Empr ],'CrSigCnFNF') < 1
	MessageBox('Favor reinicializar o processo !!!',16,'Falha na Conexão(CrSigCnFNF)')
	Return(.F.)
EndIf
Select CrSigCnFNF
Go Top
If !Eof()
	llCfg = .T.
EndIf

lcLeftCfg = Iif(llCfg,[Left Join SigCnFNF g On g.emps+g.series = n.emps+n.series ],[Left Join SigCnFNF g On g.series = n.series ])

lcQryNFis = [Select n.basecicms,n.datatrans,n.descsers,n.despaces,n.dopes,n.emis,n.emps,n.espes,n.fretes,n.icms,]+;
				[n.impress,n.ipincreds,n.ipis,n.marcas,n.nfis,n.numeros,n.numes,n.obs2s,n.obss,n.operas,n.pbrus,]+;
				[n.perciss,n.pesos,n.placas,n.pliqs,n.qtdes,n.regs,n.series,n.totnotas,n.totprods,n.ufplacas,]+;
				[n.valicms,n.valipis,n.vals,n.vlfretes,n.vlseguros,n.vtotiss,n.vtotsers,n.dtsaidas,n.especienfs,]+;
				[n.basecic2s,n.basecic3s,n.basecipis,n.basecip2s,n.basecip3s,n.cgcs,n.importas,n.correcs,n.cancelas,]+;
				[n.cfis,n.clfis,n.clifors,n.codcons,n.codtrans,n.conds,n.datans,c.rclis,f.descricaos,g.ncdser,]+;
				[n.empdopnums,n.cidchaves ]+;
			[From SigMvNfi n ] + lcLeftCfg +;
			[Left Join SigCdCfo f On f.codigos = n.cfis ]+;
			[Left Join SigCdCli c On c.iclis = n.clifors ]+;
			[Where n.emps = ?pcemps And n.series = ?pSeries And n.emis between ?pEmisDe And ?pEmisAte And ]+;
				[n.Operas In (?pOpE, ?pOpS) And ] + ;
				[n.cancelas = ?pcancelas And n.nfis between ?pNFisDe And ?pNFisAte ]+ Iif(llCfg,'',[And g.emps = ']+Space(3)+[' ])+;
			[Order By n.nfis,n.emis ]
lcQryTra   = [Select * From SigPrTrp Where itras = ?pItras ]
lcQryOpe   = [Select ndopes,dopes,digobs,digdoc,dtemis,dtentrs,casas,altdescs,origems,destinos,imagems,confes,valpres,cmoes ]+;
			   [From SigCdOpe Where dopes = ?pDopes Order By dopes ]
lcQryEest  = [Select * From SigMvCab Where emps = ?pNEmps And dopes = ?pDopes And numes = ?pNumes ]
lcQryEesti = [Select * From SigMvItn Where emps = ?pNEmps And dopes = ?pDopes And numes = ?pNumes ]
lcQryPar   = [Select * From SigMvPar Where emps = ?pNEmps And dopes = ?pDopes And numes = ?pNumes ]
lcQryCli   = [Select * From SigCdCli Where iclis = ?pIclis ]
lcQryPro   = [Select * From SigCdPro Where cpros = ?pCpros ]
lcQryFPag  = [Select * From SigOpFp Where fpags = ?pFPags ]

With ThisForm.poDataMgr
	.AddCursor('SigMvNfi' ,'cidchaves','CrSigMvNfi' ,'',ThisForm.Pagina.Lista.Grade,lcQryNFis)
	.pfSqlTabela(1).pColuna('nfis'    ,'','','Nota Fiscal' ,64 ,.T.)
	.pfSqlTabela(1).pColuna('emis'    ,'','','Dt. Emissão' ,80 ,.F.)
	.pfSqlTabela(1).pColuna('totnotas','','','Valor Nota'  ,136,.F.)
	.pfSqlTabela(1).pColuna('rclis'   ,'','','Destinatário',306,.F.)
	.AddCursor('SigPrTrp','itras'    ,'CrSigPrTrp','','',lcQryTra)
	.AddCursor('SigCdOpe','dopes'    ,'CrSigCdOpe','','',lcQryOpe)
	.AddCursor('SigMvCab','cidchaves','CrSigMvCab','','',lcQryEest)
	.AddCursor('SigMvItn','cidchaves','CrSigMvItn','','',lcQryEesti)
	.AddCursor('SigMvPar','cidchaves','CrSigMvPar','','',lcQryPar)
	.AddCursor('SigCdCli','iclis'    ,'CrSigCdCli','','',lcQryCli)
	.AddCursor('SigCdPro','cpros'    ,'CrSigCdPro','','',lcQryPro)
	.AddCursor('SigOpFp' ,'fpags'    ,'CrSigOpFp' ,'','',lcQryFPag)
	.AddCursor('SigCdSer','empcods'  ,'CrSigCdSer','','')	
EndWith

ThisForm.poDataMgr.Requery('CrSigCdSer')

If ThisForm.poDataMgr.SqlExecute([Select * From SigCdPam ],'CrSigCdPam') < 1
	MessageBox('Favor reinicializar o processo !!!',16,'Falha na Conexão(CrSigCdPam)')
	Return(.F.)
EndIf
Select CrSigCdPam
Go Top

If ThisForm.poDataMgr.SqlExecute([Select * From SigCnFNF Where emps = ?_Empr ],'CrSigCnFNF') < 1
	MessageBox('Favor reinicializar o processo !!!',16,'Falha na Conexão(CrSigCnFNF)')
	Return(.F.)
EndIf
Select CrSigCnFNF
Go Top

With ThisForm.Pagina.Lista.Grade
	.RecordSource = 'CrSigMvNfi'
	.Column1.ControlSource = 'CrSigMvNfi.nfis'
	.Column2.ControlSource = 'CrSigMvNfi.emis'
	.Column3.ControlSource = 'CrSigMvNfi.totnotas'
	.Column4.ControlSource = 'CrSigMvNfi.rclis'
	.Refresh
EndWith

ThisForm.plAcinserir  = .F.
ThisForm.plAcalterar  = .F. 
ThisForm.plAcexcluir  = .F. 

DoDefault(_acesso)

Select CrSigCdSer
Go Top
With ThisForm.Pagina.Lista.cntFiltros.Get_Serie
	.Value = CrSigCdSer.Cods
	.Refresh
EndWith
With ThisForm.Pagina.Lista.cntFiltros
	.DtIni.Value = DATE()
	.DtFim.Value = DATE()
	.Refresh
EndWith

ThisForm.Pagina.Lista.cntFiltros.getEmp.SetFocus 

* Checando Acessos de Notas de Entrada/Saída
ThisForm.Pagina.Lista.cntFiltros.optFiltro.Value = 3
ThisForm.Pagina.Lista.cntFiltros.optFiltro.Enabled = .t.
llAcE = fChecaAcesso([SIGNFCOR], [ENTRADA])
llAcS = fChecaAcesso([SIGNFCOR], [SAIDA])
Do Case 
	Case llAcE
		ThisForm.Pagina.Lista.cntFiltros.optFiltro.Value = 1

	Case llAcS
		ThisForm.Pagina.Lista.cntFiltros.optFiltro.Value = 2

	Otherwise
		ThisForm.Pagina.Lista.cntFiltros.optFiltro.Enabled = .f.
EndCase

ThisForm.Pagina.Lista.Grade.Setall('DynamicForeColor','IIf(CrSigMvNfi.Correcs,Rgb(255,0,0),Rgb(0,0,0))','Column')
ThisForm.Pagina.Lista.Grade.Setall('DynamicForeColor','IIf(CrSigMvNfi.Correcs,Rgb(255,0,0),Rgb(0,0,0))','Text')

With ThisForm.Pagina.Lista
	.InserirCarta.Enabled = (fChecaAcesso(ThisForm.Name,[CORRECAO]))
	.InserirCarta.Visible = (.InserirCarta.Enabled)

	.AlteraCarta.Enabled = (fChecaAcesso(ThisForm.Name,[ALTCARTA]))
	.AlteraCarta.Visible = (.AlteraCarta.Enabled)

	.ImpCarta.Enabled = (fChecaAcesso(ThisForm.Name,[IMPCARTA]))
	.ImpCarta.Visible = (.ImpCarta.Enabled)
EndWith 

ENDPROC
