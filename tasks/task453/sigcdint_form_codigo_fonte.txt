PROCEDURE fazcontra
Select CrSigMvCcr
NOper  = CrSigMvCcr.NOpers
VOper  = CrSigMvCcr.VOpers
Oper   = CrSigMvCcr.SOpers
Grupo  = CrSigMvCcr.SGrupos
Conta  = CrSigMvCcr.SContas
Moed   = CrSigMvCcr.SMoedas
Cot    = CrSigMvCcr.SCotacaos
Valor  = CrSigMvCcr.SValors
SOper  = CrSigMvCcr.Opers
SGrupo = CrSigMvCcr.Grupos
SConta = CrSigMvCcr.Contas
SMoed  = CrSigMvCcr.Moedas
SCot   = CrSigMvCcr.Cotacaos
SValor = CrSigMvCcr.Valors
NLanc  = CrSigMvCcr.NLancs
VLanc  = CrSigMvCcr.VLancs
Data   = Nvl(CrSigMvCcr.Datas,Ctod(Space(08)))
Venc   = Nvl(CrSigMvCcr.Vencs,Ctod(Space(08)))
Dataa  = Nvl(CrSigMvCcr.DatAlts,Ctod(Space(08)))
ValPag = CrSigMvCcr.ValPags
Emp    = CrSigMvCcr.Emps
Hist   = CrSigMvCcr.Hists
Dope   = CrSigMvCcr.Dopes
Nume   = CrSigMvCcr.Numes
DopC   = CrSigMvCcr.DopCs
NumC   = CrSigMvCcr.NumCs
Nota   = CrSigMvCcr.Nfs
Titulo = CrSigMvCcr.Titulos
GrupoE = CrSigMvCcr.GrupEms
ContaE = CrSigMvCcr.ContEms
GrupoA = CrSigMvCcr.GrupAges
ContaA = CrSigMvCcr.ContAges
ValLiq = CrSigMvCcr.ValLiqs

Insert Into CrSigMvCcr (VOpers,NOpers, NLancs, VLancs, Opers, Grupos, Contas, Moedas, Cotacaos, Valors, Emps, Datas,;
						UsuAlts, DatAlts, SGrupos, SContas, SMoedas, SCotacaos, SValors, SOpers, Autos, Hists, Dopes,;
						Numes, ValPags, Vencs, DopCs, NumCs, Nfs, Titulos, GrupEms, ContEms, GrupAges, ContAges, ValLiqs) ;
				Values (VOper, NOper, NLanc, VLanc, Oper, Grupo, Conta, Moed, Cot, Valor, Emp, Data, Usuar, DataA,;
						SGrupo, SConta, SMoed, SCot, SValor, SOper, .T., Hist, Dope, Nume, ValPag, Venc, DopC, NumC,;
						Nota, Titulo, GrupoE, ContaE, GrupoA, ContaA, ValLiq)

Select CrSigMvCcr
Replace CrSigMvCcr.Datas      With Iif(IsEmpty(CrSigMvCcr.Datas)    ,Null,CrSigMvCcr.Datas) ;
		CrSigMvCcr.DatAlts    With Iif(IsEmpty(CrSigMvCcr.DatAlts)  ,Null,CrSigMvCcr.DatAlts) ;
		CrSigMvCcr.DataConcs  With Iif(IsEmpty(CrSigMvCcr.DataConcs),Null,CrSigMvCcr.DataConcs) ;
		CrSigMvCcr.DataTrans  With Iif(IsEmpty(CrSigMvCcr.DataTrans),Null,CrSigMvCcr.DataTrans) ;
		CrSigMvCcr.DatExcs    With Iif(IsEmpty(CrSigMvCcr.DatExcs)  ,Null,CrSigMvCcr.DatExcs) ;
		CrSigMvCcr.DtAudits   With Iif(IsEmpty(CrSigMvCcr.DtAudits) ,Null,CrSigMvCcr.DtAudits) ;
		CrSigMvCcr.DtEmis     With Iif(IsEmpty(CrSigMvCcr.DtEmis)   ,Null,CrSigMvCcr.DtEmis) ;
		CrSigMvCcr.Vencs      With Iif(IsEmpty(CrSigMvCcr.Vencs)    ,Null,CrSigMvCcr.Vencs) ;
		CrSigMvCcr.CIdChaves  With Dtos(Datas)+Transform(fGerUniqueKey(Dtos(Datas)),"@L 999999")+ThisForm.Sigkey  ;
		CrSigMvCcr.GruConMoes With CrSigMvCcr.Grupos + CrSigMvCcr.Contas + CrSigMvCcr.Moedas ;
		CrSigMvCcr.EmpDopNums With CrSigMvCcr.Emps + CrSigMvCcr.Dopes + Str(CrSigMvCcr.Numes,6) ;
		CrSigMvCcr.EmpDopNcs  With CrSigMvCcr.Emps + CrSigMvCcr.DopCs + Str(CrSigMvCcr.NumCs,6) ;
		CrSigMvCcr.Pagos      With Iif(CrSigMvCcr.Valors = CrSigMvCcr.ValPags,'0','1')
ENDPROC

PROCEDURE Click
Local _Arquivo, lcArqBal, lcArqEti, lcArqHis, lcArqTra

Private lnCodBal

Select CrSigCdBal
If Eof()
	Return(.T.)
EndIf

* Checa se o Inventário foi Encerrado
If !CrSigCdBal.Encerras
	MessageBox('Não é possível Tratar um Inventário que não foi Encerrado !!!',48,'Inventário Não Encerrado')
	Return(.T.)
EndIf

* Checa se o Tratamento do Inventário foi Encerrado
If Type('CrSigCdBal.Tratas') = 'L'
	If CrSigCdBal.Tratas
		MessageBox('Não é possível Tratar um Tratamento que foi Encerrado!!!',48,'Tratamento Encerrado')
		Return(.T.)
	EndIf
EndIf

lnCodBal = CrSigCdBal.Codigos

Wait Window 'Verificando Tratamento...' NoWait
lStrQuery = [Select * From SigIvTrT Where Emps = ?_Empr And Codigos = ?lnCodBal Order By CIdChaves ]
If ThisForm.poDataMgr.SqlExecute(lStrQuery,'Tratamento') < 1
	MessageBox('Favor reinicializar o processo !!!',16,'Falha na Conexão (Tratamento)')
	Return(.F.)
EndIf
Select Tratamento
Index On CItens Tag CItens			   
Index On Produto + Str(Barras,14) Tag CPros
Go Top
If Eof()
	If !CrSigCdBal.Tratas Or MessageBox('Não Existem Registros de Tratamento do Inventário [' + Alltrim(Str(lnCodBal,4)) + ']'+;
		Chr(13) + 'Deseja Criar Um Agora?',4+32+256,'') = 6

		* Checa se os Arquivos de Inventário Existem
		Wait Window 'Selecionando Registros (Balanco)...' NoWait
		lStrQuery = [Select a.*,b.DPros,b.PCuss ]+;
					[From SigIvTrB a, SigCdPro b ]+;
					[Where a.Emps = ?_Empr And a.Codigos = ?lnCodBal And b.CPros = a.CPros ]+;
					[Order By a.CIdChaves,b.DPros,b.PCuss ]
		If ThisForm.poDataMgr.SqlExecute(lStrQuery,'Balanco') < 1
			MessageBox('Favor reinicializar o processo !!!',16,'Falha na Conexão (Balanco)')
			Return(.F.)
		EndIf
		Select Balanco
		Go Top
		If Eof()
			MessageBox('Inventário [' + Alltrim(Str(lnCodBal,4)) + '] Não Encontrado !!!',48,'')
			Return(.T.)
		EndIf

		Wait Window 'Selecionando Registros (Etiquetas)...' NoWait
		lStrQuery = [Select a.Emps,a.Codigos,a.CBars,a.CPros,a.Qtds,a.pesos,a.Datas,a.CodCors,a.CodTams,a.Grupos,a.Contas,]+;
						[a.CIdChaves,b.DPros,b.PCuss ]+;
					[From SigIvTrE a, SigCdPro b ]+;
					[Where a.Emps = ?_Empr And a.Codigos = ?lnCodBal And b.CPros = a.CPros ]+;
					[Order By a.Emps,a.Codigos,a.CBars,a.CPros,a.Qtds,a.Datas,a.CodCors,a.CodTams,a.Grupos,a.Contas,]+;
						[a.CIdChaves,b.DPros,b.PCuss ]
		If ThisForm.poDataMgr.SqlExecute(lStrQuery,'Etiqueta') < 1
			MessageBox('Favor reinicializar o processo !!!',16,'Falha na Conexão (Etiqueta)')
			Return(.F.)
		EndIf
		Select Etiqueta
		Go Top
		If Eof()
			If !(MessageBox('Etiquetas do Inventário [' + Alltrim(Str(lnCodBal,4)) + '] Não Encontradas !!!' + Chr(13) + Chr(10) +;
							'Deseja Continuar ?',4+32+256,'') = 6)
				Return(.T.)
			EndIf
		EndIf

		Wait Window 'Selecionando Registros (Histórico)...' NoWait
		lStrQuery = [Select a.*,a.CIdChaves,b.DPros,b.PCuss ]+;
					[From SigIvTrH a ]+;
					[Left Join SigCdPro b On b.CPros = a.CPros ]+;
					[Where a.Emps = ?_Empr And a.Codigos = ?lnCodBal ]+;
					[Order By a.CIdChaves ]
		If ThisForm.poDataMgr.SqlExecute(lStrQuery,'Historico') < 1
			MessageBox('Favor reinicializar o processo !!!',16,'Falha na Conexão (Historico)')
			Return(.F.)
		EndIf
		Select Historico
		Go Top
		If Eof()
			If !(MessageBox('Histórico do Inventário [' + Alltrim(Str(lnCodBal,4)) + '] Não Encontrado !!!' + Chr(13) + Chr(10) +;
							'Deseja Continuar ?',4+32+256,'') = 6)
				Return(.T.)
			EndIf
		EndIf

		*-[ Seleciona os Produtos com diferença de saldo no Inventário ]-*
		Select a.CPros As Produto, ;
		a.Saldos As Estoque, Iif(Empty(a.Relidos),a.Lidos,a.Relidos) As Apurado,;
		a.Saldops As EstPeso, Iif(Empty(a.Relidops),a.Lidops,a.Relidops) As ApurPeso,;
		a.DPros, a.PCuss As CCustos ;
		From Balanco a ;
		Where (Iif(Empty(a.Relidos), a.Lidos, a.Relidos) # a.Saldos) or (Iif(Empty(a.Relidops), a.Lidops, a.Relidops) # a.Saldops) ;
		Into Cursor TmpDifSaldo
		
		*-[ Seleciona os Produtos que não tiveram as Barras Lidas ]-*
		*-[ (Estão no Etiqueta, mas não estão no Histórico)       ]-*
		Select a.CPros as Produto, a.CBars as Barras, ;
		Sum(Val(Str(a.Qtds,12,3))) as Estoque, 000000000.00 as Apurado, ;
		Sum(Val(Str(a.Pesos,12,3))) as EstPeso, 000000000.00 as ApurPeso, ;
		a.DPros, a.PCuss as CCustos ;
		  From Etiqueta a ;
		 Group By a.CPros, a.CBars, a.DPros, a.PCuss ;
		 Where a.CPros + Str(a.CBars,14) Not In ;
			  (Select b.CPros + Str(b.CBars,14) as ProBar From Historico b Where Not Empty(CBars)) ;
		  Into Cursor TmpBarNaoLidas

		*-[ Seleciona as Etiquetas que foram Lidas mas não pertencem ao Estoque ]-*
		*-[ (Estão no Histórico, mas não estão no Etiqueta)                     ]-*
		Select a.CPros As Produto,a.CBars As Barras,000000000.00 As Estoque,(a.Qtds) As Apurado,000000000.00 As EstPeso,(a.Pesos) As ApurPeso,a.Obs,;
			a.Ocors As Ocorrencia,a.PesReals,a.Usuars,Nvl(a.DPros,Space(40)) As DPros,Nvl(a.PCuss,00000000.00) As CCustos ;
		From Historico a ;
		Where (!Empty(a.Ocors) And a.CBars # 99999999999999) Or (!Empty(a.CBars) And a.CBars # 99999999999999 And ;
			a.CPros + Str(a.CBars,14) Not In ;
			(Select b.CPros + Str(b.CBars,14) As ProBar From Etiqueta b)) ;
		Union All ;
		Select a.CPros As Produto,a.CBars As Barras,000000000.00 As Estoque,(a.Qtds) As Apurado,000000000.00 As EstPeso,(a.Pesos) As ApurPeso,a.Obs,a.Ocors,;
			a.PesReals,a.Usuars,Nvl(a.DPros,Space(40)) As DPros,Nvl(a.PCuss,000000.00) As CCustos ;
		From Historico a ;
		Where a.CBars = 99999999999999 ;
		Into Cursor TmpBarLidas
		
		lnCItens = 1

		* Passando os Dados Para a Tabela (Tratamento)
		Select TmpDifSaldo
		Go Top
		Scan
			Scatter MemVar Memo
			m.Tipos     = 'S'
			m.Barras    = 0
			m.Falta     = Iif(m.Apurado > m.Estoque,m.Apurado - m.Estoque,0)
			m.Sobra     = Iif(m.Apurado < m.Estoque,m.Estoque - m.Apurado,0)
			m.FaltaPeso = Iif(m.ApurPeso > m.EstPeso,m.ApurPeso - m.EstPeso,0)
			m.SobraPeso = Iif(m.ApurPeso < m.EstPeso,m.EstPeso - m.ApurPeso,0)
			m.CItens    = lnCItens
			m.Emps      = _Empr
			m.Codigos   = lnCodBal
			m.CIdChaves = fUniqueIds()
			Insert Into Tratamento From MemVar
			
			lnCItens = lnCItens + 1

			Select TmpDifSaldo
		EndScan
		
		Select TmpBarNaoLidas
		Go Top
		Scan
			Scatter MemVar Memo
			m.Tipos     = 'N'
			m.Falta     = Iif(m.Apurado > m.Estoque,m.Apurado - m.Estoque,0)
			m.Sobra     = Iif(m.Apurado < m.Estoque,m.Estoque - m.Apurado,0)
			m.FaltaPeso = Iif(m.ApurPeso > m.EstPeso,m.ApurPeso - m.EstPeso,0)
			m.SobraPeso = Iif(m.ApurPeso < m.EstPeso,m.EstPeso - m.ApurPeso,0)
			m.CITens    = lnCItens
			m.Emps      = _Empr
			m.Codigos   = lnCodBal
			m.CIdChaves = fUniqueIds()
			Insert Into Tratamento From MemVar
			
			lnCItens = lnCItens + 1

			Select TmpBarNaoLidas
		EndScan

		Select TmpBarLidas
		Go Top
		Scan
			Scatter MemVar Memo
			m.Tipos     = 'L'
			m.Falta     = Iif(m.Apurado > m.Estoque,m.Apurado - m.Estoque,0)
			m.Sobra     = Iif(m.Apurado < m.Estoque,m.Estoque - m.Apurado,0)
			m.FaltaPeso = Iif(m.ApurPeso > m.EstPeso,m.ApurPeso - m.EstPeso,0)
			m.SobraPeso = Iif(m.ApurPeso < m.EstPeso,m.EstPeso - m.ApurPeso,0)
			m.CItens    = lnCItens
			m.Emps      = _Empr
			m.Codigos   = lnCodBal
			m.CIdChaves = fUniqueIds()
			Insert Into Tratamento From MemVar

			lnCItens = lnCItens + 1
			
			Select TmpBarLidas
		EndScan

		Select Tratamento
		
		Wait Window 'Aguarde! Atualizando Fisicamente o Arquivo (Tratamento)...' NoWait
		If !ThisForm.poDataMgr.Update('Tratamento')
			MessageBox('Favor reinicializar o processo !!!',16,'Falha na Conexão (Balanco)')
			Thisform.PodataMgr.Rollback()
			Return .t.
		Else
			ThisForm.poDataMgr.Commit()
		EndIf
		Wait Clear

	Else
		Wait Clear
		Return(.T.)
	EndIf
EndIf

Wait Clear

Select CsTratamento
Go Top

This.Parent.Enabled = .F.
