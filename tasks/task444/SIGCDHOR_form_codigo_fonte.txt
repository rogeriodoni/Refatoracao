PROCEDURE ajustatela
ThisForm.pcEscolha = [CONSULTAR]

Select Mes, Ano From crMesAno Into Cursor LocalMesAno

If (Reccount('LocalMesAno') < 1)
	For lnObj = 1 to 42
		lcObj = Padl(lnObj, 2, '0')
		ThisForm.cntCalendario.chk&lcObj..ForeColor = Rgb(0,0,0)
		ThisForm.cntCalendario.chk&lcObj..Caption = [  00   000:00]
	Next

	With ThisForm
		.grdMesAno.Enabled 		 	 = .f.
		.cntCalendario.Enabled   	 = .f.
		.cntDados.Visible 		 	 = .t.
		.cntDados.btnData.Caption    = [00/00/0000]
		.cntDados.getMes.Value		 = []
		.cntDados.getAno.Value		 = []
		.cntDados.getQtds.Value		 = 0
		.cntDados.getQtHoras.Value	 = 0
		.cntDados.getTotHoras.Value	 = 0
		.cntDados.getMes.Enabled 	 = InList(ThisForm.pcEscolha, [INSERIR], [COPIARP])
		.cntDados.getAno.Enabled 	 = InList(ThisForm.pcEscolha, [INSERIR], [COPIARP])
		.cntDados.getQtds.Enabled 	 = InList(ThisForm.pcEscolha, [INSERIR], [ALTERAR])
		.cntDados.getQtHoras.Enabled = InList(ThisForm.pcEscolha, [INSERIR], [ALTERAR])
	EndWith

	With ThisForm.cmdEscolha
		.btnInserir.Enabled = .t.
		.btnAlterar.Enabled = .f.
		.btnExcluir.Enabled = .f.
		.btnCopiarP.Enabled = .f.
	EndWith

	With ThisForm.cmdSalva
		.btnConfirmar.Enabled = .f.
		.btnCancelar.Enabled  = .f.
		.btnSair.Enabled      = .t.
	EndWith
Else
	With ThisForm
		.grdMesAno.Enabled 		 	 = .t.
		.cntCalendario.Enabled   	 = .t.
		.cntDados.Visible 		 	 = .t.
		.cntDados.getMes.Enabled 	 = InList(ThisForm.pcEscolha, [INSERIR], [COPIARP])
		.cntDados.getAno.Enabled 	 = InList(ThisForm.pcEscolha, [INSERIR], [COPIARP])
		.cntDados.getQtds.Enabled 	 = InList(ThisForm.pcEscolha, [INSERIR], [ALTERAR])
		.cntDados.getQtHoras.Enabled = InList(ThisForm.pcEscolha, [INSERIR], [ALTERAR])
	EndWith

	With ThisForm.cmdEscolha
		.btnInserir.Enabled = .t.
		.btnAlterar.Enabled = .t.
		.btnExcluir.Enabled = .t.
		.btnCopiarP.Enabled = .t.
	EndWith

	With ThisForm.cmdSalva
		.btnConfirmar.Enabled = .f.
		.btnCancelar.Enabled  = .f.
		.btnSair.Enabled      = .t.
	EndWith
EndIf

ThisForm.grdMesAno.Refresh
ThisForm.grdMesAno.Column1.SetFocus
* Fim **************************************************************************************
ENDPROC

PROCEDURE montacalendario
Lparameters pMes, pAno, pTru

Private pDtI, pDtF

ldDia = Ctod('01/' + pMes + '/' + pAno)
lnDow = 1

pDtI = fDtoSQL(ldDia)
pDtF = fDtoSQL(Gomonth(ldDia, 1) - 1, '23:59:59')

loObj = ThisForm.cntCalendario

lcQuery = [Select * ] + ;
		    [From SigCdThs ] + ;
		   [Where Setors = '] + Padr(ThisForm.lpFases, 10) + [' And ] + ;
		   		 [UniPrdts = '] + Padr(ThisForm.lpUnidades, 10) + [' And ] + ;
		   		 [Datas BetWeen ?pDtI And ?pDtF]

If (ThisForm.poDataMgr.SqlExecute(lcQuery, 'TmpCtHr') < 1)
	=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (MontaCalendario - TmpCtHr)')
	Return 0
EndIf
Select TmpCtHr
Index On Dtos(Datas) Tag Datas

If pTru
	lcQuery = [Select * ] + ;
			    [From SigCdUpd ] + ;
			   [Where Codigos = '] + Padr(ThisForm.lpFases, 10) + [' And ] + ;
			   		 [UniPrdts = '] + Padr(ThisForm.lpUnidades, 10) + [']

	If (ThisForm.poDataMgr.SqlExecute(lcQuery, 'LocalUPrdt') < 1)
		=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (MontaCalendario - LocalUPrdt)')
		Return 0
	EndIf
	Go Top In LocalUPrdt

	lnHds = LocalUPrdt.HorDds
	lnDds = Iif(LocalUPrdt.DdSems < 1, 7, LocalUPrdt.DdSems)
	lnQds = Iif(LocalUPrdt.Qtdes  < 1, 1, LocalUPrdt.Qtdes)
	
	For I = 1 To Day(pDtF)
		ldDay = Ctod(Padl(I, 2, '0') + '/' + pMes + '/' + pAno)

		llZer = fChkFeriado(ThisForm.poDataMgr, ldDay)

		lnHor = Iif(llZer Or Dow(ldDay, 2) > lnDds, 0, lnHds)
		lnQts = Iif(llZer Or Dow(ldDay, 2) > lnDds, 0, lnQds)

		If Seek(Dtos(ldDay), 'TmpCtHr', 'Datas')
			Replace Qtds    With lnQts, ;
					QtHoras With lnHor In TmpCtHr
		Else
			Insert Into TmpCtHr (Setors, UniPrdts, Datas, Qtds, QtHoras) ;
						 Values (ThisForm.lpFases, ThisForm.lpUnidades, ldDay, lnQts, lnHor)
		EndIf
	Next
EndIf

* Alimentando o Cursor Com as Datas
Zap In crCalendario

If (Reccount('TmpCtHr') > 1)
	Append Blank In crCalendario
	Go Top In crCalendario

	lnDia = 0
	For lnObj = 1 to 42
		lnDia = lnDia + 1
		If (lnDia > 7)
			lnDia = 1
		EndIf

		lcObj = Padl(lnObj, 2, '0')

		Replace Data&lcObj.    With Ctod(''), ;
				QtHoras&lcObj. With 0, ;
				Qtds&lcObj.    With 0 In crCalendario

		loObj.chk&lcObj..Caption = ''

		If (lnDia = Dow(ldDia)) And (Month(ldDia) = Int(Val(pMes)))
			lnHor = 0
			lnQth = 0
			lnQtd = 0
			If Seek(Dtos(ldDia), 'TmpCtHr', 'Datas')
				lnHor = Val(Strtran(fMtoH(fHtoM(Str(TmpCtHr.QtHoras, 5, 2)) * TmpCtHr.Qtds), ':', '.'))
				lnQth = TmpCtHr.QtHoras
				lnQtd = TmpCtHr.Qtds
			EndIf

			Replace Data&lcObj.    With ldDia, ;
					QtHoras&lcObj. With lnQth, ;
					Qtds&lcObj.    With lnQtd In crCalendario

			loObj.chk&lcObj..ForeColor = Iif(lnHor > 0, Rgb(0,0,128), Rgb(216,39,70))
				
			lcHrr = Padl(Alltrim(Str(lnHor, 6, 2)), 6, '0')
			loObj.chk&lcObj..Caption = [  ] + Substr(Dtoc(ldDia), 1, 2) + [   ] + lcHrr && [  00   000:00]

			ldDia = ldDia + 1
		EndIf
	Next

	For lnObj = 1 to 42
		lcObj = Padl(lnObj, 2, '0')

		If (Day(crCalendario.Data&lcObj.) = 1)
			With ThisForm.cntDados
				.btnData.Caption 	= [01] + [/] + pMes + [/] + pAno
				.getMes.Value		= pMes
				.getAno.Value		= pAno

				.getMes.Enabled 	= InList(ThisForm.pcEscolha, [INSERIR], [COPIARP])
				.getAno.Enabled 	= InList(ThisForm.pcEscolha, [INSERIR], [COPIARP])
				.getQtds.Enabled 	= InList(ThisForm.pcEscolha, [INSERIR], [ALTERAR])
				.getQtHoras.Enabled	= InList(ThisForm.pcEscolha, [INSERIR], [ALTERAR])

				.getQtds.Value 	   = crCalendario.Qtds&lcObj.
				.getQtHoras.Value  = crCalendario.QtHoras&lcObj.

				lnTot = Val(Strtran(fMtoH(fHtoM(Str(crCalendario.QtHoras&lcObj., 5, 2)) * crCalendario.Qtds&lcObj.), ':', '.'))
				.getTotHoras.Value = lnTot
			EndWith

			lnObj = 42
		EndIf
	Next
Else
	For lnObj = 1 to 42
		lcObj = Padl(lnObj, 2, '0')
		loObj.chk&lcObj..ForeColor = Rgb(216,39,70)
		loObj.chk&lcObj..Caption = [  00   000:00]
	Next

	With ThisForm.cmdEscolha
		.btnInserir.Enabled = .t.
		.btnAlterar.Enabled = .f.
		.btnExcluir.Enabled = .f.
		.btnCopiarP.Enabled = .f.
	EndWith

	With ThisForm.cmdSalva
		.btnConfirmar.Enabled = .f.
		.btnCancelar.Enabled  = .f.
		.btnSair.Enabled      = .t.
	EndWith
EndIf

loObj.Enabled = .t.
loObj.Refresh
* Fim **************************************************************************************
ENDPROC

* PROCEDURE montagrademesano
------------------------------------------------------------
PROCEDURE montagrademesano
* Apanhando os Meses/Anos Para a Grade
Zap In crMesAno

lcQuery = [Select Distinct Datas ] + ;
		    [From SigCdThs ] + ;
		   [Where Setors = '] + Padr(ThisForm.lpFases, 10) + [' And ] + ;
		   		 [UniPrdts = '] + Padr(ThisForm.lpUnidades, 10) + [' ] + ;
		   [Order By Datas]

If (ThisForm.poDataMgr.SqlExecute(lcQuery, 'LocalCtHr1') < 1)
	=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (Init - LocalCtHr1)')
	Return .f.
EndIf

Select Distinct Dtos(Nvl(Datas, Ctod(''))) as CharData ;
  From LocalCtHr1 ;
  Into Cursor LocalCtHr2
  
Select Distinct Substr(CharData, 5, 2) as Mes, Substr(CharData, 1, 4) as Ano ;
  From LocalCtHr2 ;
  Into Cursor LocalCtHr3

Select LocalCtHr3
Scan
	Insert Into crMesAno (Mes, Ano) Values (LocalCtHr3.Mes, LocalCtHr3.Ano)
EndScan

Set Near On 
=Seek(Substr(Dtos(Date()), 1, 6), 'crMesAno', 'Ordem')
Set Near Off

ThisForm.grdMesAno.Column1.SetFocus
ThisForm.grdMesAno.Refresh
* Fim **************************************************************************************
ENDPROC

------------------------------------------------------------
PROCEDURE m_confirmar
Local lcHrr, loObj, lnHor, lnQth, lnQtd, lcObj, ldDat

llUpd = .f.
llTOk = .t.

Zap In crSigCdThs

Do Case
	Case (ThisForm.pcEscolha = [INSERIR]) && INSERIR
		pMes = Padl(ThisForm.cntDados.getMes.Value, 2, [0])
		pAno = Padl(ThisForm.cntDados.getAno.Value, 4, [0])
		
		ldDat = Ctod([01/] + pMes + [/] + pAno)

		pDtI = fDtoSQL(ldDat)
		pDtF = fDtoSQL(Gomonth(ldDat, 1) - 1, '23:59:59')

		lcQuery = [Select cIdChaves ] + ;
				    [From SigCdThs ] + ;
				   [Where Setors = '] + Padr(ThisForm.lpFases, 10) + [' And ] + ;
				   		 [UniPrdts = '] + Padr(ThisForm.lpUnidades, 10) + [' And ] + ;
				   		 [Datas BetWeen ?pDtI And ?pDtF]

		If (ThisForm.poDataMgr.SqlExecute(lcQuery, 'LocalCtHr') < 1)
			=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (m_Confirmar - Inserir - LocalCtHr)')
			llTOk = .f.
		EndIf
		If llTOk 
			If (Reccount('LocalCtHr') > 0)
				=MessageBox([Existem Lançamentos Nesse Período!!! Verifique!!!], 0+32, [Atenção!!!])
				llTOk = .f.
			Else
				lcQuery = [Select * ] + ;
						    [From SigCdUpd ] + ;
						   [Where Codigos = '] + Padr(ThisForm.lpFases, 10) + [' And ] + ;
						   		 [UniPrdts = '] + Padr(ThisForm.lpUnidades, 10) + [']

				If (ThisForm.poDataMgr.SqlExecute(lcQuery, 'LocalUPrdt') < 1)
					=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (m_Confirmar - Inserir - LocalUPrdt)')
					llTOk = .f.
				EndIf
				If llTOk 
					Go Top In LocalUPrdt

					lnHds = LocalUPrdt.HorDds
					lnDds = Iif(LocalUPrdt.DdSems < 1, 7, LocalUPrdt.DdSems)
					lnQds = Iif(LocalUPrdt.Qtdes  < 1, 1, LocalUPrdt.Qtdes)

					For I = 1 To Day(pDtF)
						ldDay = Ctod(Padl(I, 2, '0') + '/' + pMes + '/' + pAno)

						llZer = fChkFeriado(ThisForm.poDataMgr, ldDay)

						lnHor = Iif(llZer Or Dow(ldDay, 2) > lnDds, 0, lnHds)
						lnQts = Iif(llZer Or Dow(ldDay, 2) > lnDds, 0, lnQds)

						pDtI = fDtoSQL(ldDay)
						pDtF = fDtoSQL(Gomonth(ldDay, 1) - 1, '23:59:59')

						Insert Into crSigCdThs (Setors, UniPrdts, Qtds, QtHoras, Datas, cIdChaves) ;
									   Values (ThisForm.lpFases, ThisForm.lpUnidades, lnQts, lnHor, pDtI, fUniqueIds())
						llUpd = .t.
					Next
				EndIf
			EndIf

			If llTOk And llUpd And Not Empty(ldDat) And Not Seek(Substr(Dtos(ldDat), 1, 6), 'crMesAno', 'Ordem')
				Insert Into crMesAno (Mes, Ano) Values (Substr(Dtos(ldDat), 5, 2), Substr(Dtos(ldDat), 1, 4))
			EndIf
		EndIf
		
	Case (ThisForm.pcEscolha = [ALTERAR]) && ALTERAR
		ldDat = Date()

		For lnObj = 1 to 42
			lcObj = Padl(lnObj, 2, '0')

			If Not Empty(crCalendario.Data&lcObj.)
				ldDia = crCalendario.Data&lcObj.
				ldDat = ldDia
				lnHor = Val(Strtran(fMtoH(fHtoM(Str(crCalendario.QtHoras&lcObj., 5, 2)) * crCalendario.Qtds&lcObj.), ':', '.'))
				lnQth = crCalendario.QtHoras&lcObj.
				lnQtd = crCalendario.Qtds&lcObj.

				pDtI = fDtoSQL(ldDia)
				pDtF = fDtoSQL(ldDia, '23:59:59')

				lcQuery = [Select * ] + ;
						    [From SigCdThs ] + ;
						   [Where Setors = '] + Padr(ThisForm.lpFases, 10) + [' And ] + ;
						   		 [UniPrdts = '] + Padr(ThisForm.lpUnidades, 10) + [' And ] + ;
						   		 [Datas BetWeen ?pDtI And ?pDtF]

				If (ThisForm.poDataMgr.SqlExecute(lcQuery, 'LocalCtHr') < 1)
					=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (m_Confirmar - Alterar - LocalCtHr)')
					llTOk = .f.
				EndIf
				If llTOk
					If (Reccount('LocalCtHr') > 0)
						Go Top In LocalCtHr
						Scatter Memvar Memo
						Insert Into crSigCdThs From Memvar
						Replace Qtds 	With lnQtd, ;
								QtHoras With lnQth In crSigCdThs
					Else
						Insert Into crSigCdThs (Setors, UniPrdts, Qtds, QtHoras, Datas, cIdChaves) ;
									   Values (ThisForm.lpFases, ThisForm.lpUnidades, lnQtd, lnQth, pDtI, fUniqueIds())
					EndIf
					llUpd = .t.
				EndIf
			EndIf
		Next

	Case (ThisForm.pcEscolha = [EXCLUIR]) && EXCLUIR
		ldDat = Ctod([01/] + Padl(ThisForm.cntDados.getMes.Value, 2, [0]) + [/] + Padl(ThisForm.cntDados.getAno.Value, 4, [0]))

		pDtI = fDtoSQL(ldDat)
		pDtF = fDtoSQL(Gomonth(ldDat, 1) - 1, '23:59:59')

		lcQuery = [Delete From SigCdThs ] + ;
				   [Where Setors = '] + Padr(ThisForm.lpFases, 10) + [' And ] + ;
				   		 [UniPrdts = '] + Padr(ThisForm.lpUnidades, 10) + [' And ] + ;
				   		 [Datas BetWeen ?pDtI And ?pDtF]

		If (ThisForm.poDataMgr.SqlExecute(lcQuery, 'LocalCtHr') < 1)
			=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (m_Confirmar - Excluir - LocalCtHr)')
			llTOk = .f.
		EndIf

		If llTOk 
			llUpd = .t.

			Delete From crMesAno ;
			 Where Mes = Substr(Dtos(ldDat), 5, 2) And ;
				   Ano = Substr(Dtos(ldDat), 1, 4)
		EndIf

	Case (ThisForm.pcEscolha = [COPIARP]) && COPIARP
		* Checando a Existência do Mês de Lançamento
		pMes1 = Padl(ThisForm.cntDados.getMes.Value, 2, [0])
		pAno1 = Padl(ThisForm.cntDados.getAno.Value, 4, [0])
		
		ldDat1 = Ctod([01/] + pMes1 + [/] + pAno1)

		pDtI = fDtoSQL(ldDat1)
		pDtF = fDtoSQL(Gomonth(ldDat1, 1) - 1, '23:59:59')

		lcQuery = [Select cIdChaves ] + ;
				    [From SigCdThs ] + ;
				   [Where Setors = '] + Padr(ThisForm.lpFases, 10) + [' And ] + ;
				   		 [UniPrdts = '] + Padr(ThisForm.lpUnidades, 10) + [' And ] + ;
				   		 [Datas BetWeen ?pDtI And ?pDtF]

		If (ThisForm.poDataMgr.SqlExecute(lcQuery, 'LocalCtHr') < 1)
			=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (m_Confirmar - CopiarP - LocalCtHr - 1)')
			llTOk = .f.
		EndIf

		If llTOk 
			If (Reccount('LocalCtHr') < 1)
				pMes = Padl(crMesAno.Mes, 2, [0])
				pAno = Padl(crMesAno.Ano, 4, [0])
				
				ldDat = Ctod([01/] + pMes + [/] + pAno)
				ldDtn = ldDat

				pDtI = fDtoSQL(ldDat)
				pDtF = fDtoSQL(Gomonth(ldDat, 1) - 1, '23:59:59')

				lcQuery = [Select * ] + ;
						    [From SigCdThs ] + ;
						   [Where Setors = '] + Padr(ThisForm.lpFases, 10) + [' And ] + ;
						   		 [UniPrdts = '] + Padr(ThisForm.lpUnidades, 10) + [' And ] + ;
						   		 [Datas BetWeen ?pDtI And ?pDtF]

				If (ThisForm.poDataMgr.SqlExecute(lcQuery, 'LocalCtHr') < 1)
					=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (m_Confirmar - CopiarP - LocalCtHr - 2)')
					llTOk = .f.
				EndIf
				If llTOk 
					If (Reccount('LocalCtHr') > 0)
						Select LocalCtHr
						Scan
							ldDtn = Ctod(Substr(Dtoc(LocalCtHr.Datas), 1, 3) + pMes1 + [/] + pAno1)
							If Not Empty(ldDtn)
								pDtI = fDtoSQL(ldDtn)
								Insert Into crSigCdThs (Setors, UniPrdts, Qtds, QtHoras, Datas, cIdChaves) ;
											   Values (LocalCtHr.Setors, LocalCtHr.UniPrdts, LocalCtHr.Qtds, ;
											   		   LocalCtHr.QtHoras, pDtI, fUniqueIds())
								llUpd = .t.
							EndIf
						EndScan

						If llUpd
							Insert Into crMesAno (Mes, Ano) Values (Substr(Dtos(ldDtn), 5, 2), Substr(Dtos(ldDtn), 1, 4))
						EndIf
					EndIf
				EndIf
			Else
				=MessageBox([Existem Lançamentos Nesse Período!!! Verifique!!!], 0+32, [Atenção!!!])
				llTOk = .f.
			EndIf
		EndIf
EndCase

If llTOk
	If Not ThisForm.poDataMgr.Update('crSigCdThs')
		=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (m_Confirmar - Update - crSigCdThs)')
		llTOk = .f.
	EndIf
EndIf

If llTOk And llUpd
	If (ThisForm.poDataMgr.Commit() < 1)
		=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (m_Confirmar - Commit)')
		llTOk = .f.
	EndIf
EndIf

If Not llTOk
	ThisForm.poDataMgr.RollBack()
EndIf

ThisForm.AjustaTela

Set Near On 
=Seek(Substr(Dtos(ldDat), 1, 6), 'crMesAno', 'Ordem')
Set Near Off
* Fim **************************************************************************************
ENDPROC

Lparameters pCpo

With ThisForm.cntDados

PROCEDURE Init
Lparameters pCod, pUni

If (Type('pCod') <> 'C') Or Empty(pCod)
