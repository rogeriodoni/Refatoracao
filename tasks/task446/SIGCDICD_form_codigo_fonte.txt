PROCEDURE Valid
Local loLista

If Not Empty(This.Value)
	loLista = CreateObject('fwBuscaExt', ThisForm.poDataMgr.pnIdConn, 'SigPrTri', ;
						   'crListaRemota', 'Tipos', This.Value, 'Seleção', .t.)

	If Not loLista.plAchouRegistro
		loLista.mAddColuna('Tipos', '', 'Tipo',.t.)
		loLista.mAddColuna('Descs', '', 'Tipo',.t.)
		loLista.Show()
	EndIf
	
	ThisForm.Container1.GetTipo.Value = crListaRemota.Tipos
	ThisForm.Container1.GetDesc.Value = crListaRemota.Descs
	ThisForm.Container1.fwget2.Value = crListaRemota.Descs

	Use In crListaRemota
Else
	This.Value = ''
EndIf
This.Refresh
* Fim **************************************************************************************



ENDPROC

PROCEDURE Click
With ThisForm.Container1

	ThisForm.poDatamgr.SqlExecute([Select * From SigPrTri Where Tipos = ']+ThisForm.Container1.fwget1.Value+['],"CsSigPrTri")
	Select csSigPrTri
	If ! Eof()
		MessageBox("Novo Tipo Ja Cadastrado",16,"")
		Return
	EndIf
	
	lcTipo = .GetTipo.Value
	ThisForm.poDatamgr.Requery("cnSigPrTri")
	
	Replace all Tipos     With .fwget1.Value,;
	            Descs     With .fwget2.Value,;
	            cIdChaves With fUniqueIds()
	
	ThisForm.poDatamgr.UpDate("cnSigPrTri")
	ThisForm.poDatamgr.Commit()
	
EndWith 

Release Windows
ENDPROC
