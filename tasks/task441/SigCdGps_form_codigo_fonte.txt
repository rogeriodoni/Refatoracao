------------------------------------------------------------
* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
LParameters poForm, pnDataSes, pTpGgp, pEscolha

Set DataSession to pnDataSes

Thisform.ParentForm          = poForm
Thisform.ParentForm.Enabled  = .F.

Thisform.Escolha = pEscolha

Thisform.GravaDados = .f.
Thisform.Tipoggp = pTpGgp

ThisForm.cmdInserir.Enabled  = InList(Thisform.Escolha,'INSERIR','ALTERAR')
ThisForm.cmdExcluir.Enabled  = InList(Thisform.Escolha,'INSERIR','ALTERAR')


With Thisform
	.Caption = 'Grupos de Produtos ' + AllTrim(crSigCdOpe.Dopes)
EndWith


ThisForm.cntSombra.lblSombra.Caption = ThisForm.Caption
ThisForm.cntSombra.lblTitulo.Caption = ThisForm.Caption

lcOperacao = crSigCdOpe.Dopes

With Thisform
	.poDatamgr = CreateObject('fSqlConector')

	If Not .poDatamgr.pnIdconn>0
		Messagebox('Impossivel Efetuar Conexão com Servidor de Banco de Dados.',(0+48),'Erro')
		Return(.f.)
	Endif

	With .poDatamgr
		.AddCursor('SigCdGps','cIdChaves','CrSigCdGps','')
		.AddCursor('SigCdGgp','pkChave','CrSigCdGgp','')
	EndWith
Endwith	

Create Cursor xOpe ( Dopes c(20), Cgrus c(3), Dgrus c(30) )
Index on Dopes Tag Dopes
Set Order to

With Thisform.GradeOpe
	.RecordSource          = 'xOpe'
	.Column1.ControlSource = 'xOpe.Cgrus'
	.Column2.ControlSource = 'xOpe.Dgrus'
	.Refresh
EndWith

lcSql = [Select a.*, b.Dgrus ] + ;
		[From ]+Iif(Thisform.Tipoggp,[SigCdGgp],[SigCdGps])+[ a, SigCdGrp b ] + ;
		[Where a.Dopes = ']+CrSigCdOpe.Dopes+[' And a.Cgrus = b.Cgrus ]

ThisForm.Podatamgr.Sqlexecute(lcSql,'CsSelecao')

Select CsSelecao
Scan
	Scatter Memvar
	Select xOpe
	Append Blank
	Gather Memvar
EndScan

Select xOpe
Set Order to
Go Top

ThisForm.GradeOpe.Refresh


ENDPROC

PROCEDURE Valid
Local lcFpags, lnRecNo, loBrowse

	loLista = CreateObject('fwbuscaext', ThisForm.poDataMgr.pnIdConn, 'SigCdGrp', ;
						   'CrListaRemota', 'Cgrus', This.Value, 'Grupo de Produto', .t.)
	If Not loLista.plAchouRegistro
		loLista.mAddColuna('Cgrus', '', 'Grupo',.t.)
		loLista.mAddColuna('Dgrus', '', 'Descrição',.t.)
		loLista.Show()
	EndIf

	Replace Dgrus with CrListaRemota.Dgrus In xOpe

------------------------------------------------------------
* PROCEDURE Click
------------------------------------------------------------
PROCEDURE Click
If ThisForm.GravaDados

	Select Cgrus, Sum(1) as Qt From xOpe Group by 1 Into Cursor Selecao where cgrus <> ' ' having Sum(1) > 1
	If Reccount('Selecao') > 0
		MessageBox('Existem lancamentos de Grupos Em Duplicidade!!!',0+16,'')
		Return .f.
	EndIf
	
	Select CrSigCdGgp
	Zap
	Select CrSigCdGps
	Zap
	
	Select xOpe
	Scan
		If Not Empty(Cgrus)
			If ThisForm.Tipoggp
				Insert Into CrSigCdGgp (Dopes, Cgrus, pkchave) ;
							Values (crSigCdOpe.Dopes, xOpe.Cgrus, fUniqueIds())
			Else
				Insert Into CrSigCdGps (Dopes, Cgrus, cIdChaves) ;
							Values (crSigCdOpe.Dopes, xOpe.Cgrus, fUniqueIds())
			EndIf 
		Endif
	EndScan
	
	m.lnErro1 = ThisForm.poDatamgr.pnIdConn

	With ThisForm

		lcSql = [Delete From ]+Iif(ThisForm.Tipoggp,[SigCdGgp],[SigCdGps])+[ Where dopes = ']+CrSigCdOpe.dopes+[']
		m.lnErro1 = ( .PodataMgr.SqlExecute(lcSql,'') )

		If m.lnErro1>0 
			If ThisForm.Tipoggp
				m.lnErro1 = Iif(.poDataMgr.UpDate('crSigCdGgp'),1,-1)
			Else
				m.lnErro1 = Iif(.poDataMgr.UpDate('crSigCdGps'),1,-1)
			EndIf 
		EndIf 
		If m.lnErro1>0 
			m.lnErro1 = .poDataMgr.Commit()
		EndIf 
		If Not m.lnErro1>0 
			.poDataMgr.RollBack()
		EndIf 
	EndWith 
	If Not m.lnErro1>0 
		= MessageBox('Erro na Gravação dos Dados!!!',48)
	EndIf 
EndIf
Thisform.ParentForm.Enabled = .T.
ThisForm.Release

ENDPROC
