PROCEDURE mctrlbotoes
LParameters plHabilita

Local lnI

If Type('plHabilita') # 'L'
	plHabilita = .F.
Endif

With Thisform.ParentForm.Pagina.Lista
	For lnI=1 to .Grupo_Op.ButtonCount
		If .Grupo_Op.Buttons(lnI).BaseClass = 'Commandbutton' And .Grupo_Op.Buttons(lnI).Visible
			.Grupo_Op.Buttons(lnI).Enabled = plHabilita
		EndIf
	EndFor
	.cmdFPOper.Enabled = plHabilita
	.cmdMultiSubN.Enabled = plHabilita
	.Grupo_Saida.Sair.Enabled = plHabilita
EndWith



ENDPROC

PROCEDURE Init
LParameters poForm, pnDataSes, pcEscolha

Local lcOperacao

Set DataSession to pnDataSes

Thisform.ParentForm = poForm
Thisform.PcEscolha  = pcEscolha
Thisform.ParentForm.Enabled  = .F.
lcOperacao = crSigCdOpe.Dopes

With Thisform
	.Caption = 'C.Custo Padrao da Operação ' + AllTrim( Proper( crSigCdOpe.Dopes ) )
	.Top     = Thisform.ParentForm.Top + Thisform.ParentForm.Pagina.Lista.Grade.Top + 50
	.Left    = Thisform.ParentForm.Left + ;
			   Thisform.ParentForm.Pagina.Lista.Grade.Left + ;
			   Thisform.ParentForm.Pagina.Lista.Grade.Columns(1).Width + ;
			   Thisform.ParentForm.Pagina.Lista.Grade.Columns(3).Width + 16

	.cmdInserir.Visible = fChecaAcesso('SIGCCCPE','INSERIR') And ;
						  Iif( Type( 'Thisform.PcEscolha' ) # 'C', .T., ;
						  InList( Thisform.PcEscolha, 'INSERIR', 'ALTERAR' ) )
	.cmdInserir.Enabled = fChecaAcesso('SIGCCCPE','INSERIR')	And ;
						  Iif( Type( 'Thisform.PcEscolha' ) # 'C', .T., ;
						  InList( Thisform.PcEscolha, 'INSERIR', 'ALTERAR' ) )
	.cmdExcluir.Visible = fChecaAcesso('SIGCCCPE','EXCLUIR') And ;
						  Iif( Type( 'Thisform.PcEscolha' ) # 'C', .T., ;
						  InList( Thisform.PcEscolha, 'INSERIR', 'ALTERAR' ) )
	.cmdExcluir.Enabled = fChecaAcesso('SIGCCCPE','EXCLUIR')	And ;
						  Iif( Type( 'Thisform.PcEscolha' ) # 'C', .T., ;
						  InList( Thisform.PcEscolha, 'INSERIR', 'ALTERAR' ) )
EndWith

ThisForm.cntSombra.lblSombra.Caption = ThisForm.Caption
ThisForm.cntSombra.lblTitulo.Caption = ThisForm.Caption
Thisform.mCtrlBotoes(.F.)

lcSql = [Select a.*, b.Descrs as Dgrus From SigCcCpe a, SigCdGcr b where a.Dopes=']+CrSigCdOpe.Dopes+[' And a.Grupos = b.Codigos ]
SigCdOpe.PodataMgr.Sqlexecute(lcSql,'xCcOpe')

Select * From xCcOpe Into Cursor TmpCcOpe ReadWrite

Select TmpCcOpe
Goto Top	
With Thisform.GradeSubN
	.RecordSource          = 'TmpCcOpe'
	.Column1.ControlSource = 'TmpCcOpe.Grupos'
	.Column2.ControlSource = 'TmpCcOpe.Dgrus'
	.Column3.ControlSource = 'TmpCcOpe.GruCads'
	.Column4.ControlSource = 'TmpCcOpe.CtFils'
	.Refresh
EndWith

ENDPROC

* PROCEDURE Click
------------------------------------------------------------
PROCEDURE Click
If Thisform.cmdInserir.Visible Or Thisform.cmdExcluir.Visible
	If ( Thisform.lHouveExcl Or Thisform.lHouveIns )
	
		Select TmpCcOpe
		Delete For Grupos = '  '
		Replace all dopes     With CrSigCdOpe.Dopes,;
				    CidChaves With fUniqueIds()

		Select Grupos, sum(1) From TmpccOpe Group by 1 having sum(1) > 1 Into Cursor SecondKey
		Select SecondKey
		If Reccount() > 0
			MessageBox('Existem Grupos em Duplicidade. Favor Acertar!!!',0+16,'Atenção')
			Return .f.
		EndIf
		
		Select CrSigCcCpe
		Zap
		Append From Dbf('TmpCcOpe')
		
		lcSql = [Delete From SigCcCpe Where Dopes = ']+CrSigCdOpe.dopes+[']

		m.lnErro1 = SigCdOpe.poDatamgr.pnIdConn

		With SigCdOpe
			If m.lnErro1 > 0
				m.lnErro1 = SigCdOpe.PodataMgr.Sqlexecute(lcSql,'')
			Endif
			If m.lnErro1>0 
				m.lnErro1 = Iif(.poDataMgr.UpDate('crSigCcCpe'),1,-1)
			EndIf 
			If m.lnErro1>0 
				m.lnErro1 = .poDataMgr.Commit()
			EndIf 
			If Not m.lnErro1>0 
				.poDataMgr.RollBack()
			EndIf 
		EndWith 
		If Not m.lnErro1>0 
			= MessageBox('Erro na Gravação dos Dados do Cadastro de Operações.',48)
		EndIf 
	EndIf
ENDIF

Thisform.ParentForm.Enabled = .T.
Thisform.mCtrlBotoes(.T.)
ThisForm.Release
ENDPROC
