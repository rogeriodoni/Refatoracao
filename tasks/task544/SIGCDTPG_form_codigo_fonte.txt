* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
Parameters pcCodigos

lcQuery = [Select * From SigPrTra Where TPTrans = ']+pcCodigos+[']

ThisForm.Grid1.RecordSource = ''

With ThisForm
	.poDataMgr = CreateObject('fSqlConector', ThisForm.Name)
	
	With .poDataMgr
		* Cursor Principal Para Montagem da Lista (Tabela Grande)
		.AddCursor('SigPrTrn', 'cIdChaves', 'crSigPrTrn', '', .f.,lcQuery)
	EndWith
	
	.poDataMgr.Requery('crSigPrTrn')
	
EndWith

Select crSigPrTrn
Index On Emps+TpTrans+Tps Tag EmpTp
Set Order To EmpTp
Go Top

ThisForm.Grid1.RecordSource = 'crSigPrTrn'
ThisForm.Grid1.Column1.ControlSource = 'crSigPrTrn.Emps'
ThisForm.Grid1.Column2.ControlSource = 'crSigPrTrn.TpTrans'
ThisForm.Grid1.Column3.ControlSource = 'crSigPrTrn.Tps'

ThisForm.cmdinserir.SetFocus()
ENDPROC

PROCEDURE Valid
=fAcessoEmpresa(Usuar, 'C', This.Value, ThisForm.Grid1.Column1.Text1, '', ThisForm.poDataMgr)

If ! Empty(This.Value)
	lnReg = Recno()

	Select crSigPrTrn
	lcEmp   = Emps
	lcTra   = TpTrans
	lcTps   = Tps
	lcChave = cIdChaves
	Locate for Emps = lcEmp And TpTrans = lcTra And Tps = lcTps And cIdChaves <> lcChave
	If ! Eof()
		MessageBox('Tipo de Transferencia Cadastrada',16,'')
		Go lnReg
		This.Value = ''
		Return .f.
	Endif

	Go lnReg
Endif

ENDPROC
