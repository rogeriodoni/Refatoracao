PROCEDURE GotFocus
If Used('crGrupos')
	Select crGrupos
	Delete All
	Go Top
	ThisForm.Grid1.Refresh
Endif

If Used('crSigCdUsu')
	Select crSigCdUsu
	Delete all
	Go Top
	ThisForm.Grid2.Refresh
EndIf 

ENDPROC

------------------------------------------------------------
* PROCEDURE Valid
------------------------------------------------------------
PROCEDURE Valid
LOCAL xBrow

IF not IsEmpty(This.value)

	lcSearch = '%'+Upper(Alltrim(This.Value))+'%'
	Select Descricaos, programas, Parametros From crSigCdPrg1 Where Descricaos Like lcSearch into Cursor crSigCdPrg

	Select crSigCdPrg
	Go Top

	If Eof()
		MessageBox('Parametros não Localizado',16,'')
		Return(.f.)
	Endif

	If Reccount() > 1
		Go Top
		xBrow = CreateObject('fwBuscaInt','crSigCdPrg','Descricaos',This.Value,'Programas',.t.)
		xBrow.mAddColuna('Descricaos',Replicate('X',85),'Programa',.t.)
		xBrow.Show		
	EndIf

	If Lastkey() <> 27
		This.Value           = crSigCdPrg.Descricaos
		ThisForm.pcPrograma  = crSigCdPrg.Programas
		ThisForm.pcParametro = crSigCdPrg.Parametros
	Else
		This.Value           = ''
		ThisForm.pcPrograma  = ''
		ThisForm.pcParametro = ''
	EndIf

Else
	
	This.Value           = ''
	ThisForm.pcPrograma  = ''
	ThisForm.pcParametro = ''

Endif
ENDPROC

PROCEDURE AfterRowColChange
LPARAMETERS nColIndex

ThisForm.Grid2.RecordSource = ''

If Used('crGrupos') 
	If Empty(crGrupos.Grupos)
		lcQueryUsuario = [Select Distinct Usuarios, nComps, cCargs, Deptos, Supervis, cAtivos From SigCdUsu ]+;
						 [Where Supervis In ('S', 'I') ]+;
						 [Union ]+;
						 [Select Distinct b.Usuarios, b.nComps, b.cCargs, b.Deptos, b.Supervis, b.cAtivos From SigCdAcu a ]+;
		    			 [Inner Join SigCdUsu b On a.Usuarios = b.Usuarios Where a.Grupos = '          ' ]+;
		                 [And (a.Programas = ']+Alltrim(ThisForm.pcPrograma)+[' ]+;
		                 Iif(!Empty(ThisForm.pcParametro),[And a.Parametros = ']+Alltrim(ThisForm.pcParametro)+[') ],[)])
	Else
		lcQueryUsuario = [Select Distinct a.Usuarios, a.nComps, a.cCargs, a.Deptos, a.Supervis, a.cAtivos From SigCdUsu a ]+;
						 [Left Join SigCdAcG b on a.Usuarios = b.Usuarios ]+;
		                 [Where b.Grupos = ']+crGrupos.Grupos+[' ]
	EndIf

	If (ThisForm.poDataMgr.SqlExecute(lcQueryUsuario, [crSigCdUsu]) < 1)
		=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (crSigCdUsu)])
		Return 0
	EndIf

	Select crSigCdUsu
	Index On Usuarios Tag Usuarios
	Go Top

	ThisForm.Grid2.RecordSource = 'crSigCdUsu'
	ThisForm.Grid2.SetAll('DynamicForeColor',[Iif(crSigCdUsu.cAtivos=='N', Rgb(255,0,0), Rgb(0,0,255))],'Column')
	ThisForm.Grid2.Column1.ControlSource = 'crSigCdUsu.Usuarios'
	ThisForm.Grid2.Column2.ControlSource = 'crSigCdUsu.nComps'
	ThisForm.Grid2.Column3.ControlSource = 'crSigCdUsu.cCargs'
	ThisForm.Grid2.Column4.ControlSource = 'crSigCdUsu.Deptos'
	ThisForm.Grid2.Column5.ControlSource = 'crSigCdUsu.Supervis'
Endif	

ENDPROC

PROCEDURE Command1.Click
Private lcPrgs, lcParms

If ! Empty(ThisForm.pcPrograma)

	ThisForm.Grid1.RecordSource = ''
	ThisForm.Grid2.RecordSource = ''

	lcPrgs  = Alltrim(ThisForm.pcPrograma)
	lcParms = Alltrim(ThisForm.pcParametro)

	lcQueryAcesso  = [Select distinct Grupos From SigCdAcu Where Programas = ?lcPrgs ]+;
	                 Iif(!Empty(lcParms),[ And Parametros = ?lcParms ],[ ])+;
	                 [Order By Grupos]

	ThisForm.PoDatamgr.SqlExecute(lcQueryAcesso,'crGrupos')
	Go Top
	
	If Eof() or ! Empty(Grupos)
		Append Blank
	Endif

	Index On Grupos Tag Grupos
	Go Top

	ThisForm.Grid1.RecordSource = 'crGrupos'
	ThisForm.Grid1.Column1.ControlSource = 'crGrupos.Grupos'

	ThisForm.Grid1.SetFocus
	
EndIf 
ENDPROC
