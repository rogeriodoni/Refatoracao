PROCEDURE Init
Private pTip

pTip = Replicate(Chr(254), 9)

lcQryCtme  = [Select * From SigCdMe]
lcQryCtmeI = [Select * From SigCdMeI Where TipoEmps = ?pTip Order by Dopes]
lcQryCtmeT = [Select * From SigCdMeT Where TipoEmps = ?pTip Order by Dopes]

With ThisForm
	.poDataMgr = CreateObject('fSqlConector', ThisForm.Name)

	If (.poDataMgr.pnIdconn > 0)
		With .poDataMgr
			.AddCursor('SigCdMe', 'TipoEmps', 'crSigCdMe', '', ThisForm.Pagina.Lista.Grade, lcQryCtme)
			.pfSqlTabela(1).pColuna('Tipos', '', '', 'Tipo', 52, .t.)

			.pfSqlTabela(1).pColuna('Extensao','','','Extensao',50,.f.)

			.AddCursor('SigCdMeI', 'cIdChaves', 'crSigCdMeI', '', '', lcQryCtmeI)
			.AddCursor('SigCdMeT', 'cIdChaves', 'crSigCdMeT', '', '', lcQryCtmeT)
		EndWith
		
		lcSql = [Select LibTrEmps From SigCdPac ]
		If ThisForm.Podatamgr.Sqlexecute(lcSql,'CrTmpParac') < 1
			=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (SigCdPac)')
			Return .f.
		Endif

		.MontaGrades

		=DoDefault()

		With .Pagina.Lista
			.btnCopiar.Visible = fChecaAcesso('SIGCDTME', 'COPIAR')
		EndWith
		
		lcSql = [Select Dopes From SigOpOpe ]
		ThisForm.Podatamgr.Sqlexecute(lcSql,'CrSigOpOpe')

		Insert Into CrSigOpOpe (Dopes) Values ('PAGAMENTO')
		Insert Into CrSigOpOpe (Dopes) Values ('RECEBIMENTO')
		Insert Into CrSigOpOpe (Dopes) Values ('LOTE DE CHEQUE')
		Index On dopes Tag Dopes 
		
		.Pagina.Lista.Grade.Column1.SetFocus

	Else
		=Messagebox('Impossível Efetuar Conexão Com o Servidor de Banco de Dados...', 0+48, 'CreateObject')
		Return .f.
	EndIf
EndWith

PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
Private pTip

pTip = crSigCdMe.Tipos + crSigCdMe.Emps

If InList(Thisform.pcEscolha, 'INSERIR', 'ALTERAR')
	If Empty(crSigCdMe.Tipos)
		=Messagebox('O Código do Transporte Tem Que Ser Preenchido!!!', 0+48, 'Código do Transporte Inválido!!!')
		Return 0		
	EndIf

	If Empty(crSigCdMe.Emps)
		=Messagebox('O Código da Empresa Tem Que Ser Preenchido!!!', 0+48, 'Código da Empresa Inválido!!!')
		Return 0		
	EndIf

	Select csGrade
	Scan
		If Empty(csGrade.Dopes)
			Delete In csGrade
		EndIf
	EndScan
	Select csTitulo
	Scan
		If Empty(csTitulo.Dopes)
			Delete In csTitulo
		EndIf
	EndScan
	
	Go Top In CsGrade
	
	Select Emps, Dopes, EmpDs, Sum(1) as TT From CsGrade Where Emps = CrSigCdMe.Emps Group by Emps, dopes, EmpDs ;
	having Sum(1) > 1 Into Cursor CsSelecao
	
	Select CsSelecao
	Go Top
	If Reccount('CsSelecao') > 0
		MessageBox('Erro de Configuração. Operação: '+CsSelecao.Emps+' - '+CsSelecao.Dopes+' em Duplicidade!!!',0+16,'')
		Return .f.
	Endif
EndIf

Select crSigCdMe
If (Thisform.pcEscolha = 'INSERIR')
	ThisForm.poDataMgr.CursorQuery('SigCdMe', 'crTmp', 'TipoEmps', pTip)
	If (Reccount('crTmp') > 0)
		=Messagebox('O Tipo do Processo Já Está Cadastrado Para Essa Empresa!!!', 0+48, 'Nome do Processo Inválido!!!')
		Return 0		
	EndIf
EndIf

If InList(Thisform.pcEscolha, 'INSERIR', 'ALTERAR', 'EXCLUIR')
	Zap In crSigCdMeI
	Zap In CrSigCdMeT
	
	If InList(Thisform.pcEscolha, 'INSERIR', 'ALTERAR')
		Select csGrade
		Scan
			lcKey = Iif(Empty(csGrade.cIdChaves), fUniqueIds(), csGrade.cIdChaves)
			Insert Into crSigCdMeI (TipoEmps, Emps, Dopes, cIdChaves, chkPagos, ChkSubns, EmpDs, DopeDs, ;
									ContaEs, Apagas, nDias, Numeras, DtTrans ) ;
							Values (pTip, csGrade.Emps, csGrade.Dopes, ;
									lcKey, csGrade.ChkPagos, csGrade.ChkSubns, CsGrade.EmpDs, CsGrade.DopeDs, ;
									CsGrade.ContaEs, CsGrade.Apagas, CsGrade.nDias, CsGrade.Numeras, CsGrade.DtTrans )
		EndScan

		Select csTitulo
		Scan
			lcKey = Iif(Empty(csTitulo.cIdChaves), fUniqueIds(), csTitulo.cIdChaves)
			Insert Into crSigCdMeT (TipoEmps, Emps, Dopes, cIdChaves, cNotas ) ;
							Values (pTip, csTitulo.Emps, csTitulo.Dopes, ;
									lcKey, CsTitulo.cNotas )
		EndScan

	EndIf
EndIf

If Empty(crSigCdMe.TipoEmps)
	Replace TipoEmps With crSigCdMe.Tipos + crSigCdMe.Emps In crSigCdMe
EndIf

=DoDefault()

If InList(Thisform.pcEscolha, 'INSERIR', 'ALTERAR')

	ThisForm.poDatamgr.SqlExecute([Select nEmps From SigCdEmp Where cEmps = ']+crSigCdMe.Emps+['],'crSigCdEmp')

		lcValor = 'TMOVEST' + Alltrim(CrSigCdMe.Tipos) + CrSigCdMe.Emps
		lnValor = This.Parent.Parent.Pageframe1.Page1.Get_Transp.Value - 1

		If ThisForm.podatamgr.cursorquery('SigSySeq','csAutoriza','Valor', lcValor)
			lcQuery = [Update SigSySeq Set Conteudo=]+Str(lnValor)+[ Where Valor = ']+lcValor+[']
	    Else
	    	lcQuery = [INSERT INTO SigSySeq (Valor, Conteudo) VALUES (']+lcValor+[', ]+Str(lnValor)+')'
		EndIf
		If (ThisForm.poDataMgr.SqlExecute(lcQuery, '') < 1)
			=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (csAutoriza)')
			Return 0
		EndIf
	EndIf
	
	If This.Parent.Parent.Pageframe1.Page1.Get_Recebe.Value <> Thisform.Proxrecebe

		If ThisForm.podatamgr.cursorquery('SigSySeq','csAutoriza','Valor', lcValor)
			lcQuery = [Update SigSySeq Set Conteudo=]+Str(lnValor)+[ Where Valor = ']+lcValor+[']
	    Else
	    	lcQuery = [INSERT INTO SigSySeq (Valor, Conteudo) VALUES (']+lcValor+[', ]+Str(lnValor)+')'
		EndIf
		If (ThisForm.poDataMgr.SqlExecute(lcQuery, '') < 1)
			=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (csAutoriza)')
			Return 0
		EndIf
	EndIf

	If (ThisForm.poDataMgr.Commit() < 1)
		=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (Commit - Autoriza)')
		Return 0
	EndIf
