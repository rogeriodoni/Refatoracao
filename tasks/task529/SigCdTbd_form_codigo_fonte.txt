------------------------------------------------------------
PROCEDURE Init
LParameters poForm, pnDataSes

Set DataSession to pnDataSes

Thisform.ParentForm          = poForm
Thisform.ParentForm.Enabled  = .F.

Thisform.GravaDados = .f.

With Thisform
	.Caption = 'Grupos de Produtos ' + AllTrim(crSigCdOpe.Dopes)
EndWith

ThisForm.cntSombra.lblSombra.Caption = ThisForm.Caption
ThisForm.cntSombra.lblTitulo.Caption = ThisForm.Caption

lcOperacao = crSigCdOpe.Dopes

With Thisform
	.poDatamgr = CreateObject('fSqlConector')

	If Not .poDatamgr.pnIdconn>0
		Messagebox('Impossivel Efetuar Conexão com Servidor de Banco de Dados.',(0+48),'Erro')
		Return(.f.)
	Endif

	With .poDatamgr
		.AddCursor('SigCdTbd','cIdChaves','CrSigCdTbd','')
	EndWith
Endwith	

Create Cursor xOpe ( Dopes c(20), codigos c(10), descrs c(30) )
Index on Dopes Tag Dopes
Set Order to

With Thisform.GradeOpe
	.RecordSource          = 'xOpe'
	.Column1.ControlSource = 'xOpe.codigos'
	.Column2.ControlSource = 'xOpe.descrs'
	.Refresh
EndWith

lcSql = [Select a.*, b.descrs From SigCdTbd a, SigOpTdz b Where a.Dopes = ']+CrSigCdOpe.Dopes+[' And a.codigos = b.codigos ]
ThisForm.Podatamgr.Sqlexecute(lcSql,'CsSelecao')

Select CsSelecao
Scan
	Scatter Memvar
	Select xOpe
	Append Blank
	Gather Memvar
EndScan

Select xOpe
Set Order to
Go Top

ThisForm.GradeOpe.Refresh


ENDPROC

PROCEDURE Load
=fConfigGeral()
ENDPROC

------------------------------------------------------------
PROCEDURE Click
If ThisForm.GravaDados

	Select Codigos, Sum(1) as Qt From xOpe Group by 1 Into Cursor Selecao where codigos <> ' ' having Sum(1) > 1
	If Reccount('Selecao') > 0
		MessageBox('Existem lancamentos de Tabelas Em Duplicidade!!!',0+16,'')
		Return .f.
	EndIf
	
	Select CrSigCdTbd
	Zap
	
	Select xOpe
	Scan
		If Not Empty(codigos)
			Insert Into crSigCdTbd (Dopes, codigos, cIdChaves) ;
							Values (crSigCdOpe.Dopes, xOpe.codigos, fUniqueIds())
		Endif
	EndScan
	
	m.lnErro1 = ThisForm.poDatamgr.pnIdConn

	With ThisForm

		lcSql = [Delete From SigCdTbd Where dopes = ']+CrSigCdOpe.dopes+[']
		m.lnErro1 = ( .PodataMgr.SqlExecute(lcSql,'') )

		If m.lnErro1>0 
			m.lnErro1 = Iif(.poDataMgr.UpDate('crSigCdTbd'),1,-1)
		EndIf 
		If m.lnErro1>0 
			m.lnErro1 = .poDataMgr.Commit()
		EndIf 
		If Not m.lnErro1>0 
			.poDataMgr.RollBack()
		EndIf 
	EndWith 
	If Not m.lnErro1>0 
		= MessageBox('Erro na Gravação dos Dados!!!',48)
	EndIf 
EndIf
Thisform.ParentForm.Enabled = .T.
ThisForm.Release

ENDPROC
