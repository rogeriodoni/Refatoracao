PROCEDURE Init
LParameters lcAcesso

lcQueryDGG = [select a.*, b.descs from sigdptgg a join sigcdgpr b on a.codgrupo = b.codigos]

With ThisForm
	.LockScreen = .T.
	.poDatamgr = CreateObject('fSqlConector',ThisForm.Name)
	If (.poDataMgr.pnIdconn > 0)
		With .poDatamgr
			.AddCursor('SigCdDpt','codigos','crSigCdDpt','',ThisForm.pagina.lista.grade)
			.pfSqlTabela(1).pColuna('codigos',	   '', '', 'Código',	   080, .T.)
			.pfSqlTabela(1).pColuna('Descricaos', '', '', 'Departamento', 290, .T.)
			.pfSqlTabela(1).pColuna('Diretors',   '', '', 'Diretor',	   080, .T.)
			
			.AddCursor('SigDptGg','PkChave','crSigDptGg','','',lcQueryDGG)
		EndWith

		=DoDefault()

		.LockScreen = .F.
	Else
		=Messagebox('Impossível Efetuar Conexão Com o Servidor de Banco de Dados...', 0+48, 'CreateObject')
		Return .f.
	EndIf
EndWith


=Afields(v02,'crSigDptGg')
Create Cursor xDptGg From array v02

With ThisForm.Pagina.Dados.gradeGG
	.RecordSource = 'xDptGg'
	.column1.ControlSource = 'xDptGg.codgrupo'
	.Column2.controlsource = 'xDptGg.descs'
	.Refresh
EndWith
ENDPROC

PROCEDURE msv_alterar
Local llOks
llOks = .t.

If llOks 
	lcDelete = [Delete From SigDptGg Where CodDepto = ']+crSigCdDpt.codigos+[']
	If ThisForm.PodataMgr.SqlExecute(lcDelete,'') < 0
		=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (Delete - SigDptGg)')
		llOks = .f.
	Endif
	If llOks And Not ThisForm.poDataMgr.Update('crSigDptGg')
		=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (Update - crSigDptGg)')
		llOks = .f.
	EndIf
EndIf 
If llOks
	=DoDefault()
Else
	ThisForm.poDataMgr.Rollback()
	Return .f.
EndIf

ENDPROC

ThisForm.Pagina.Dados.GradeGG.Refresh

    Case INLIST(ThisForm.pcEscolha,'INSERIR')
		ThisForm.Pagina.Dados.OptAutos.Value=2
	Case INLIST(ThisForm.pcEscolha,'INSERIR','PROCURAR')
		ThisForm.Pagina.Dados.getcodigos.SetFocus

	Case ThisForm.pcEscolha=='ALTERAR'
		ThisForm.Pagina.Dados.GETDescricaos.SetFocus
        ThisForm.Pagina.Dados.OptAutos.Value=IIF(EMPTY(ThisForm.Pagina.Dados.OptAutos.Value),2,ThisForm.Pagina.Dados.OptAutos.Value)
	Case INLIST(ThisForm.pcEscolha,'CONSULTAR','EXCLUIR')
		This.Parent.Parent.dados.grupo_salva.salva.SetFocus 
EndCase

ENDPROC

------------------------------------------------------------
* PROCEDURE pagina.dados.grupo_salva.salva.Click
------------------------------------------------------------
PROCEDURE pagina.dados.grupo_salva.salva.Click
*Faz Validacao dos Campos, nas Seguinte Operacoes
If InList(ThisForm.pcEscolha,'INSERIR','ALTERAR')
	If Empty(crSigCdDpt.codigos)
		Messagebox( 'O Código precisa ser Informado...', 48, '' )
		ThisForm.Pagina.Dados.getcodigos.SetFocus
		Return
	EndIf
	If Empty(crSigCdDpt.descricaos)
		Messagebox( 'A Descrição precisa ser Informada...', 48, '' )
		ThisForm.Pagina.Dados.getDescricaos.SetFocus
		Return
	EndIf

	*Verifica se o registro já existe
	If ThisForm.pcEscolha = 'INSERIR' And ThisForm.poDataMgr.ChkRegister('SigCdDpt','codigos',ThisForm.Pagina.Dados.getcodigos.Value)
		Messagebox( 'Código do Departamento já Cadastrado!', 48, '' )
		ThisForm.Pagina.Dados.getcodigos.SetFocus
		Return
	ENDIF
	
	*-- Select para ver se já existe registro automático cadastrado paa uso na tela de atendimento
	If ThisForm.Pagina.Dados.OptAutos.Value = 1	
		lcQuery = [Select * from sigcddpt Where autos = 1 And Codigos <> '] + ThisForm.Pagina.Dados.getcodigos.Value + [']
		If (ThisForm.poDataMgr.SqlExecute(lcQuery, [LocalSel]) < 1)
		   =MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (LocalSel)])
		   Return .f.
		Endif
	    *- Verifica se já existe resgistro automático - Inclusão
		If Reccount('LocalSel')>0
			=MessageBox( 'Já existe registro com código automático!', 48, 'Aviso' )
			ThisForm.Pagina.Dados.OptAutos.option2.SetFocus()
			Return .f.
		Endif
	EndIf
	
	Select xDptGg
	Select CodGrupo from xDptGg into cursor Duplica group by CodGrupo having sum(1) > 1 
	Select Duplica
	If Reccount() > 0
		=MessageBox('Grande Grupo Informado mais de uma vez, Favor Verificar.!!!',0+32,'')
		Return .F.
	EndIf 
	
	Select CrSigDptGg
	Zap
	Select xDptGg
	Scan
		If Not Empty(CodGrupo)
			Select CrSigDptGg
			Append Blank
			replace CodGrupo With xDptGg.CodGrupo,;
					CodDepto With crSigCdDpt.codigos,;
					pkChave With fUniqueIds()
		EndIf
	EndScan
	
Else
	If ThisForm.pcEscolha = 'EXCLUIR'
		If ThisForm.poDataMgr.ChkRegister('SigCdUsu','deptos',ThisForm.Pagina.Dados.getcodigos.Value)
			Messagebox( 'Existem Usuários que utilizam esse Departamento!', 48, 'Exclusão não permetida' )
			ThisForm.Pagina.Dados.Grupo_Salva.Cancelar.SetFocus
			Return
		Endif
	EndIf
Endif

= DoDefault()

ThisForm.mAtivaPagina1

ENDPROC
