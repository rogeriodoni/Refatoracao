PROCEDURE Load
=fConfigGeral()

Create Cursor XCorNf (cods n(3),descs c(30),retifs m(8) NULL )
Index On cods Tag cods
ENDPROC

------------------------------------------------------------
* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
LParameters pForm,Emp,Serie,EmisDe,EmisAte,NFis
Private pcEmps,pcSeries,pcNFis

ThisForm.poDataMgr = CreateObject('fSqlConector','SigNfCor')

ThisForm.ParentForm = pForm

pcEmps    = Emp
pcSeries  = Serie
pcNFis    = NFis

lcQryCorNf = [Select * From SigNfCor ]+;
 			  [Where cemps = ?pcEmps And cseries = ?pcSeries And cnfis = ?pcNFis ]+;
		      [Order By cemps,cseries,cnfis,noccors ]

lcQryEmp   = [Select * From SigCdEmp Where cemps = ?pcEmps ]

With ThisForm.poDataMgr
	.AddCursor('SigNfCor','cidchaves','CrSigNfCor','','',lcQryCorNf)
	.AddCursor('SigCcOco','ncodigos' ,'CrSigCcOco','','')
	.AddCursor('SigCdEmp'  ,'cemps'    ,'CrSigCdEmp'  ,'','',lcQryEmp)
EndWith

ThisForm.poDataMgr.Requery('CrSigNfCor')
ThisForm.poDataMgr.Requery('CrSigCcOco')
ThisForm.poDataMgr.Requery('CrSigCdEmp')

ThisForm.Emp      = Emp
ThisForm.Serie    = Serie
ThisForm.EmisDe   = EmisDe
ThisForm.EmisAte  = EmisAte
ThisForm.pcNFisDe = NFis
ThisForm.Get_NotaIni.Value = NFis

With ThisForm.Grade
	.RecordSource = 'XCorNf'
	.Column1.ControlSource = 'XCorNf.cods'
	.Column2.ControlSource = 'XCorNf.descs'
	.Column3.ControlSource = 'XCorNf.retifs'
	.Refresh
EndWith

Select XCorNf
Zap

Select CrSigNfCor
Go Top
If !Eof()
	Select CrSigCcOco
	Scan
		Insert Into XCorNf (cods,descs,retifs) ;
		Values (CrSigCcOco.ncodigos,CrSigCcOco.cdescrs,'')
	EndScan
	Select CrSigNfCor
	Go Top
	Scan
		Select XCorNf
		Go Top
		Locate For XCorNf.cods = CrSigNfCor.noccors
		If Found()
			Replace XCorNf.retifs With CrSigNfCor.mretifs
		Endif
		Select CrSigNfCor
	EndScan
EndIf
Select XCorNf
Go Top
ThisForm.Refresh
ENDPROC

PROCEDURE KeyPress
LParameters nKeyCode, nShiftAltCtrl
Do Case
	Case nKeyCode = 13 or nKeyCode = 24
		Select XCorNf
		If !Eof()
			Skip
		EndIf
		ThisForm.Grade.Refresh
	Case nKeyCode = 5 
		Select XCorNf
		If !Bof()
			Skip -1
		EndIf
		ThisForm.Grade.Refresh
	Case nKeyCode = 9
		KeyBoard "{CTRL+TAB}"
EndCase
ENDPROC

* PROCEDURE Ok.Click
------------------------------------------------------------
PROCEDURE Ok.Click
Private lcEmps,lcSeries,lcNFis,lnOccors, pNot, pTru

If Not IsEmpty(ThisForm.Get_NotaIni.Value)
	pNot = .f.
	pTru = .t.

	llTaOk = .t.
Suspend

	lcQuery = [Delete From SigNfCor ] + ;
			   [Where CEmps = '] + Padr(ThisForm.Emp, 3) + [' And ] + ;
			   		 [CSeries = '] + Padr(ThisForm.Serie, 3) + [' And ] + ;
			   		 [CNfis = '] + Padr(ThisForm.Get_NotaIni.Value, 6) + [']

	If (ThisForm.poDataMgr.SqlExecute(lcQuery, []) < 1)
		=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (1)])
		llTaOk = .f.
	EndIf

	If llTaOk
		lcQuery = [Update SigMvNfi ] + ;
					 [Set Correcs = ?pNot ] + ;
				   [Where Emps = '] + Padr(ThisForm.Emp, 3) + [' And ] + ;
				   		 [Series = '] + Padr(ThisForm.Serie, 3) + [' And ] + ;
				   		 [Nfis = '] + Padr(ThisForm.Get_NotaIni.Value, 6) + [']

		If (ThisForm.poDataMgr.SqlExecute(lcQuery, []) < 1)
			=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (2)])
			llTaOk = .f.
		EndIf
	EndIf

	If llTaOk
		If (ThisForm.poDataMgr.Commit() < 1)
			=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (3)])
			llTaOk = .f.
		EndIf
	EndIf

	If Not llTaOk
		ThisForm.poDataMgr.RollBack()
		Return 0
	Else
		Update crSigMvNfi ;
		   Set Correcs = .f. ;
	     Where Emps = Padr(ThisForm.Emp, 3) And ;
			   Series = Padr(ThisForm.Serie, 3) And ;
			   Nfis = Padr(ThisForm.Get_NotaIni.Value, 6) 
	EndIf
		
	Zap In crSigNfCor

	Select XCorNf
	Go Top In XCorNf
	Scan While llTaOK
		If llTaOK And Not IsEmpty(XCorNf.Retifs)
			Select crSigNfCor
			Locate For crSigNfCor.nOccors = XCorNf.Cods
			If Eof([crSigNfCor])
				Insert Into crSigNfCor (cEmps, cSeries, cNfis, nOccors, mRetifs, cIdChaves) ;
								Values (ThisForm.Emp, ThisForm.Serie, ThisForm.Get_NotaIni.Value, ;
										XCorNf.Cods, XCorNf.Retifs, fUniqueIds())
			Else
				Replace mRetifs With XCorNf.Retifs In crSigNfCor
			EndIf
		EndIf
	EndScan

	Select crSigNfCor
	Go Top In crSigNfCor
	If Not Eof([crSigNfCor])
		If Not ThisForm.poDataMgr.Update([crSigNfCor])
			=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (4)])
			llTaOk = .f.
		EndIf

		If llTaOk
			Select crSigNfCor
			Go Top In crSigNfCor
			If Not Eof([crSigNfCor])
				lcQuery = [Update SigMvNfi ] + ;
							 [Set Correcs = ?pTru ] + ;
						   [Where Emps = '] + Padr(ThisForm.Emp, 3) + [' And ] + ;
						   		 [Series = '] + Padr(ThisForm.Serie, 3) + [' And ] + ;
						   		 [Nfis = '] + Padr(ThisForm.Get_NotaIni.Value, 6) + [']

				If (ThisForm.poDataMgr.SqlExecute(lcQuery, []) < 1)
					=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (5)])
					llTaOk = .f.
				EndIf
			EndIf
		EndIf

		If llTaOk 
			If (ThisForm.poDataMgr.Commit() < 1)
				=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (6)])
				llTaOk = .f.
			EndIf
		EndIf

		If Not llTaOk
			ThisForm.poDataMgr.RollBack()
			Return 0
		Else
			Update crSigMvNfi ;
			   Set Correcs = .t. ;
		     Where Emps = Padr(ThisForm.Emp, 3) And ;
				   Series = Padr(ThisForm.Serie, 3) And ;
				   Nfis = Padr(ThisForm.Get_NotaIni.Value, 6) 
		EndIf
	Else
		If MessageBox('Confirma Impressão da Correção ?',4 + 32 + 256,'')=6
			pcEmps    = ThisForm.Emp
			pcSeries  = ThisForm.Serie
			pcEmisDe  = ThisForm.EmisDe
			pcEmisAte = ThisForm.EmisAte
			pcNFisDe  = ThisForm.pcNFisDe
			pcNFisAte = ThisForm.pcNFisDe
			pEDNs     = CrSigMvNfi.Emps + CrSigMvNfi.Dopes + Str(CrSigMvNfi.Numes,6)

			Do Form SigreImc With pcEmps,pcSeries,pcEmisDe,pcEmisAte,pEDNs,pcNFisDe,pcNFisAte
		EndIf
	EndIf
	ThisForm.Release
Else
	MessageBox('Nota Fiscal Inválida !!',0+48,'')
EndIf
ENDPROC

PROCEDURE Valid
If !IsEmpty(This.Value)
	Select XCorNf
	Zap
	Select CrSigNfCor
	pcEmps    = ThisForm.Emp
	pcSeries  = ThisForm.Serie
	pcNFisDe  = This.Value
	ThisForm.poDataMgr.Requery('CrSigNfCor')
	Select CrSigNfCor
	Go Top
	If !Eof()
		Select CrSigCcOco
		Scan
			Insert Into XCorNf (cods,descs,retifs) ;
			Values (CrSigCcOco.ncodigos,CrSigCcOco.cdescrs,'')
		EndScan
		Select CrSigNfCor
		Scan
			If Seek(CrSigNfCor.noccors,'XCorNf.cods','cods')
				Replace XCorNf.retifs With CrSigNfCor.mretifs
			EndIf
			Select CrSigNfCor
		EndScan
	Else
		MessageBox('Nota sem Correção !!',0+48,'')
		This.Value = ''
		This.Refresh
		Select XCorNf
		Zap
		ThisForm.Grade.Refresh
		Return 0
	EndIf
EndIf
ENDPROC
