------------------------------------------------------------
* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
LPARAMETERS lcacesso

With ThisForm
	.poDatamgr = CreateObject('fSqlConector')
	With .poDatamgr
		.AddCursor('SigCdRec','grupos','crSigCdRec','',ThisForm.pagina.lista.grade)
		.pfSqlTabela(1).pColuna('grupos','','','Grupos',80,.t.)
		.AddCursor('SigCdGcr','Codigos','crSigCdGcr','','')		
	EndWith
	= DoDefault()
	.LockScreen = .f.
EndWith
ENDPROC

------------------------------------------------------------
* PROCEDURE msv_procurar
------------------------------------------------------------
PROCEDURE msv_procurar
Local lcBusca

If .Not.IsEmpty(ThisForm.Pagina.Dados.Get_cd_grupo.Value)
	ThisForm.poDataMgr.LocateCursor('crSigCdRec','grupos',ThisForm.Pagina.Dados.Get_cd_grupo.Value)
EndIf

ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Lista.Grupo_op.Click
------------------------------------------------------------
PROCEDURE Pagina.Lista.Grupo_op.Click
LParameters Opcao

If !dodefault( OPCAO )
	ThisForm.mAtivapagina1
	Return
Endif

Do Case 
	Case Inlist( ThisForm.pcEscolha, 'INSERIR', 'PROCURAR' )
		ThisForm.Pagina.Dados.Get_cd_grupo.SetFocus
	Case Inlist( ThisForm.pcEscolha, 'EXCLUIR', 'CONSULTAR' , 'ALTERAR')
		ThisForm.Pagina.Dados.Grupo_Salva.Salva.SetFocus
EndCase

ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
If Inlist( ThisForm.pcEscolha, 'INSERIR' )
	If IsEmpty( CrSigCdRec.Grupos )
		Messagebox( 'Grupo não pode ficar em branco!', 48, '' ) 
		ThisForm.Pagina.Dados.Get_cd_grupo.SetFocus
		Return
	Endif
	If ThisForm.poDataMgr.ChkRegister('SigCdRec','grupos',ThisForm.Pagina.Dados.Get_Cd_grupo.Value)
		=Messagebox( 'Grupo já cadastrado!', 48, '' )
		ThisForm.Pagina.Dados.Get_Cd_Grupo.SetFocus
		Return
	Endif


Endif
=DoDefault()
ThisForm.mAtivapagina1

ENDPROC

------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Cancelar.Click
=DoDefault()
If ThisForm.plCancelar
	ThisForm.mAtivapagina1
Endif

ENDPROC
