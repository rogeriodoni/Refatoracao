PROCEDURE msv_procurar
select crSigOpOco
cod=thisform.pagina.dados.getcodigo.value
set order to codigos
if not IsEmpty(cod)
	set near on
	seek cod
	set near off
endif

ENDPROC

------------------------------------------------------------
* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
LParameters lcAcesso

WITH Thisform
	.poDatamgr = CreateObject('fSqlConector',ThisForm.Name)

	If Not .poDatamgr.pnIdconn>0
		Messagebox('Impossivel Efetuar Conexão com Servidor de Banco de Dados.',(0+48),'Erro')
		Return(.f.)
	Endif

	With .poDatamgr
		.AddCursor('SigOpOco','Codigos','crSigOpOco','Codigos',ThisForm.pagina.lista.grade)
		.pfSqlTabela(1).pColuna('Codigos','','','Código',80,.t.)
		.pfSqlTabela(1).pColuna('Descrs','','','Descrição',220,.t.)
		.pfSqlTabela(1).pColuna('Opers','','','Tipo de Operação',104,.f.)
	EndWith
Endwith	

lcSql = [Select TitOpcS, TitOpcN, TitOpcA From SigCdPac]
If ThisForm.Podatamgr.Sqlexecute(lcSql,'LocalParac') < 1
	=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (LocalParac)')
	Return .f.
EndIf
Select LocalParac
Go Top
With ThisForm.Pagina.Dados.OpcaoSNA
	.option1.Caption = Iif(Not Empty(LocalParac.TitOpcs),LocalParac.TitOpcS,[Sim])
	.option2.Caption = Iif(Not Empty(LocalParac.TitOpcn),LocalParac.TitOpcN,[Nao])
	.option3.Caption = Iif(Not Empty(LocalParac.TitOpcA),LocalParac.TitOpcA,[Ambos])
EndWith

=DoDefault()
ENDPROC

* PROCEDURE Pagina.Lista.Grupo_op.Click

PROCEDURE Pagina.Lista.Grupo_op.Click
LPARAMETERS opcao
select crSigOpOco
If !dodefault( opcao )
	ThisForm.mAtivapagina1
	Return
Endif

if crSigOpOco.Opers = "DB"
   ThisForm.Pagina.Dados.opcaoOcorrencia.Value = 1
else
   ThisForm.Pagina.Dados.opcaoOcorrencia.Value = 2
endif
if thisform.pcEscolha='INSERIR'
	with thisform.pagina.dados.opcaocontab
		.value=2
		.refresh
	endwith
endif
if ThisForm.pcEscolha = 'INSERIR' or ;
	thisform.pcEscolha='PROCURAR'
    ThisForm.Pagina.Dados.getCodigo.SetFocus
endif
if ThisForm.pcEscolha = 'ALTERAR'
   ThisForm.Pagina.Dados.getDescricao.SetFocus  
endif
if thisform.pcEscolha='EXCLUIR' or ;
	thisform.pcEscolha='CONSULTAR'
	thisform.pagina.dados.grupo_salva.salva.setfocus
endif
Select crSigOpOco
ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
if thisform.pcEscolha='INSERIR'
   if IsEmpty(ThisForm.Pagina.Dados.getCodigo.Value)
      MessageBox("Código Inválido!",0+48,"")
      ThisForm.Pagina.Dados.getCodigo.SetFocus
      return
   endif
   if ThisForm.podatamgr.ChkRegister('SigOpOco', 'Codigos', crSigOpOco.Codigos)
      MessageBox("Código já cadastrado!",0+48,"")
      ThisForm.Pagina.Dados.getCodigo.SetFocus
      RETURN 
   endif   
Endif
if Inlist(thisform.pcEscolha,'INSERIR','ALTERAR')
   if IsEmpty(ThisForm.Pagina.Dados.get_grupo.Value)
      MessageBox("Grupo Inválido!",0+48,"")
      ThisForm.Pagina.Dados.get_grupo.SetFocus
      return
   endif
   if IsEmpty(ThisForm.Pagina.Dados.get_conta.Value)
      MessageBox("Conta Inválida!",0+48,"")
      ThisForm.Pagina.Dados.get_conta.SetFocus
      return
   endif

endif 
if Inlist(thisform.pcEscolha,'INSERIR','ALTERAR')
	If ThisForm.Pagina.Dados.OpcaoOcorrencia.Value = 1
		Replace crSigOpOco.Opers with "DB"
	Else
		Replace crSigOpOco.Opers with "CR"
	Endif  
Endif
DoDefault()
ThisForm.mAtivapagina1

ENDPROC
