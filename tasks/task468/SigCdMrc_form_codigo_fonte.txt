* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
LPARAMETERS _acesso

WITH Thisform
	.poDatamgr = CreateObject('fSqlConector',Thisform.Name)

	If Not .poDatamgr.pnIdconn>0
		Messagebox('Impossivel Efetuar Conexão com Servidor de Banco de Dados.',(0+48),goSistema.TituloSistema)
		Return(.f.)
	Endif

	With .poDatamgr
		.AddCursor('SigCdMrr','Balancos','crSigCdMrr','Balancos',ThisForm.pagina.lista.grade)
		.pfSqlTabela(1).pColuna('Balancos','','','Demonstrativo',150,.t.)
		.pfSqlTabela(1).pColuna('Titulos','','','Titulo',290,.f.)
		.pfSqlTabela(1).pColuna('Mapas','','','Impressão',75,.f.)
		.AddCursor('SigCdMri','CidChaves','csSigCdMri','Classes',.f.,[Select * From SigCdMri Where Balancos = ?crSigCdMrr.Balancos])
	EndWith
Endwith	

=DoDefault( _Acesso )

ThisForm.Pagina.Lista.grade.Column3.ControlSource=[IIF(crSigCdMrr.Mapas=2,"MAPA","PLANO")]

ThisForm.Pagina.Lista.Grade.SetAll('DynamicForeColor', 'Iif(crSigCdMrr.Mapas = 2, ;
									RGB(0,0,255), RGB(0,0,0))', 'Column')
ENDPROC

------------------------------------------------------------
* PROCEDURE msv_excluir
------------------------------------------------------------
PROCEDURE msv_excluir
Local lnErro1

m.lnErro1 = ThisForm.poDatamgr.pnIdConn

With ThisForm
	If m.lnErro1>0 
		m.lnErro1 = Iif(.poDataMgr.UpDate('crSigCdMrr'),1,-1)
	EndIf 
	If m.lnErro1>0 
		m.lnErro1 = Iif(.poDataMgr.UpDate('csSigCdMri'),1,-1)
	EndIf 
	If m.lnErro1>0 
		m.lnErro1 = .poDataMgr.Commit()
	EndIf 
	If Not m.lnErro1>0 
		.poDataMgr.RollBack()
	EndIf 
EndWith 
If Not m.lnErro1>0 
	= MessageBox('Erro na Eclusão dos Dados do Cadastro de Operações.',48)
EndIf 

ENDPROC

------------------------------------------------------------
* PROCEDURE msv_procurar
------------------------------------------------------------
PROCEDURE msv_procurar
SELECT crSigCdMrr
SET ORDER TO balancos
SET NEAR ON
SEEK Thisform.Pagina.dados.get_balanco.Value
SET NEAR OFF

ENDPROC

* PROCEDURE Pagina.Lista.Grupo_op.Click
------------------------------------------------------------
PROCEDURE Pagina.Lista.Grupo_op.Click
LParameters opcao

If !dodefault( OPCAO )
	ThisForm.mAtivapagina1
	Return
Endif
If Inlist(ThisForm.pcEscolha,'INSERIR','PROCURAR')
	ThisForm.Pagina.Dados.get_balanco.SetFocus
Endif
If Thisform.pcEscolha='ALTERAR'
	Thisform.Pagina.Dados.get_titulo.SetFocus
Endif
If Inlist(ThisForm.pcEscolha,'CONSULTAR','EXCLUIR')
	ThisForm.Pagina.Dados.Grupo_Salva.Salva.SetFocus
EndIf
ThisForm.Pagina.Dados.Refresh
ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
If InList(ThisForm.pcEscolha,'INSERIR','ALTERAR')
	If IsEmpty(Thisform.Pagina.Dados.Get_Balanco.Value)
		Messagebox( 'Demonstrativo Inválido', 48, '' )
		ThisForm.Pagina.Dados.Get_Balanco.SetFocus
		Return
	Endif
	If Thisform.pcEscolha='INSERIR'
		If 	ThisForm.podatamgr.ChkRegister('SigCdMrr','Balancos', Thisform.Pagina.Dados.Get_Balanco.Value)
			Messagebox( 'Demonstrativo já Cadastrado', 48, '' )
			ThisForm.Pagina.Dados.Get_Balanco.SetFocus
			Return
		Endif
	Endif
Endif
If InList(ThisForm.pcEscolha,'EXCLUIR')
	ThisForm.POdatamgr.Requery("csSigCdMri")
	SELECT CsSigCdMri
	DELETE ALL
	SELECT crSigCdMrr
	DELETE
Endif
=DoDefault()

ThisForm.mAtivapagina1

ENDPROC

=DoDefault()

If ThisForm.plCancelar
	ThisForm.mAtivapagina1
Endif

ENDPROC
