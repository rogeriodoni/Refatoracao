	.Column1.Sparse = .f.

	.Column2.ControlSource = 'crSigOpOco.Codigos'
	.Column2.FontName = 'Courier New'

	.Column2.ReadOnly = .t.

	.Column3.ControlSource = 'crSigOpOco.Descrs'
	.Column3.FontName = 'Courier New'
	.Column3.Movable = .f.
	.Column3.Resizable = .f.
