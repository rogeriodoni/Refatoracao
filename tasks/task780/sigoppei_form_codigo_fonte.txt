* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
LPARAMETERS poform, pnDatSes

ThisForm.cntSombra.lblSombra.Caption = ThisForm.Caption
ThisForm.cntSombra.lblTitulo.Caption = ThisForm.Caption

Set DataSession to pnDatSes
thisform.parentform = poform

thisform.grade.Visible = .f.
thisform.gettotitens.Visible = .f.
thisform.gettotbaixa.Visible = .f.
thisform.gettotdisp.Visible = .f.
thisform.getCProd.Visible = .f.
thisform.getDProd.Visible = .f.

If Used([xBaixa2]) &&and !Eof([xBaixa])

	If Reccount([xBaixa2])>0
		Select *, Space(40) as Dpros from xBaixa2 ;
		union all ;
		Select cpros, Space(3) as CodCors, Space(3) as CodCors, qtds, qtbaixas, qtprods, Space(40) as Dpros ;
			From xBaixa Where cpros not in(Select cpros from xBaixa2 ) ;
		Into Cursor TmpItepen ReadWrite
	Else
		Select *, Space(40) as Dpros, Space(3) as CodTams, Space(3) as CodCors from xBaixa Into Cursor TmpItepen ReadWrite
	EndIf 
	
	Select TmpItePen
	Scan
		lcsql = [Select Dpros from SigCdPro Where CPros = ']+TmpItePen.Cpros+[']
		ThisForm.Parentform.ParentForm.PodataMgr.Sqlexecute(lcsql,'TmpPro')
		
		replace Dpros with TmpPro.Dpros in TmpItePen
	EndScan
	
	ThisForm.Grade.RecordSource = [TmpItepen]
	ThisForm.Grade.Column1.ControlSource = [TmpItepen.Cpros]
	ThisForm.Grade.Column2.ControlSource = [TmpItepen.Dpros]
	ThisForm.Grade.Column3.ControlSource = [TmpItepen.Qtds]
	ThisForm.Grade.Column4.ControlSource = [TmpItepen.QtBaixas]
	ThisForm.Grade.Column5.ControlSource = [TmpItepen.CodTams]
	ThisForm.Grade.Column6.ControlSource = [TmpItepen.CodCors]
	ThisForm.Grade.Column7.ControlSource = [TmpItepen.Qtds - TmpItepen.QtBaixas]
	
	Select TmpItepen
	Sum qtds to lnItn
	Go top
	Sum qtBaixas to lnBaixa
	Go top
	thisform.gettotitens.Value = lnItn
	thisform.gettotbaixa.Value = lnBaixa
	thisform.gettotdisp.Value = (lnItn-lnBaixa)
	thisform.gettotitens.Visible = .t.
	thisform.gettotbaixa.Visible = .t.	
	thisform.gettotdisp.Visible = .t.	
	thisform.grade.Visible = .t.
	thisform.getCProd.Visible = .f.
	thisform.getDProd.Visible = .f.
Else
	Return .f.
EndIf

thisform.op_escolha = poform.op_escolha
thisform.Refresh()
ENDPROC
