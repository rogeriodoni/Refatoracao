* PROCEDURE msv_procurar
------------------------------------------------------------
PROCEDURE msv_procurar
Local lcBusca

If .Not.IsEmpty(ThisForm.Pagina.Dados.get_codigo.Value)
	ThisForm.poDataMgr.LocateCursor('CrSigCdCst','Codigos',ThisForm.Pagina.Dados.get_codigo.Value)
EndIf

ENDPROC

------------------------------------------------------------
* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
LPARAMETERS lcacesso

With ThisForm
	.poDatamgr = CreateObject('fSqlConector')
	With .poDatamgr
		.AddCursor('SigCdCst','codigos','crSigCdCst','',ThisForm.pagina.lista.grade)
		.pfSqlTabela(1).pColuna('codigos','','','Código',70,.t.)
		.pfSqlTabela(1).pColuna('descrs','','','Descrição',240,.t.)
		.pfSqlTabela(1).pColuna('descr2s','','','Descrição 2',240,.t.)
		.pfSqlTabela(1).pColuna('cMalads','','','Mala Direta',70,.t.)
	EndWith
	= DoDefault()
	.LockScreen = .f.
EndWith
ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Lista.Grupo_op.Click
------------------------------------------------------------
PROCEDURE Pagina.Lista.Grupo_op.Click
LPARAMETERS opcao

select CrSigCdCst
If !dodefault( Opcao )
	ThisForm.mAtivapagina1
	Return
Endif
if thisform.pcEscolha='CONSULTAR' or ;
	thisform.pcEscolha='EXCLUIR'
	thisform.pagina.dados.grupo_salva.salva.setfocus
endif
if thisform.pcEscolha='ALTERAR'
	thisform.pagina.dados.Get_Descr.setfocus
endif
if thisform.pcEscolha='INSERIR' or ;
	thisform.pcEscolha='PROCURAR'
	thisform.pagina.dados.Get_Codigo.setfocus
endif

ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
if thisform.pcEscolha='INSERIR' or thisform.pcEscolha='ALTERAR'
	if IsEmpty(thisform.pagina.dados.Get_Codigo.value)
		messagebox('Dados Inválidos',0+48,'')
		thisform.pagina.dados.Get_Codigo.setfocus
		return
	endif
	If .Not.InList(ThisForm.pagina.dados.getMalad.Value,'S','N')
		messagebox('Especificação da Mala Direta Inválida, Apenas [S]im ou [N]ão.',0+48,'')
		ThisForm.pagina.dados.getMalad.SetFocus
		return
	endif
endif
if thisform.pcEscolha='INSERIR' 
	If ThisForm.poDataMgr.ChkRegister('SigCdCst','codigos',ThisForm.Pagina.Dados.Get_Codigo.Value)
		MessageBox('Código já cadastrado ! ',0+48,'')
		ThisForm.pagina.dados.get_Codigo.SetFocus
		Return
	endif
endif

=DoDefault()
Thisform.mAtivapagina1
ENDPROC

	thisform.mAtivapagina1
endif
ENDPROC
