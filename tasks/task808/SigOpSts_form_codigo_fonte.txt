PROCEDURE mctrlbotoes
LParameters plHabilita

Local lnI

If Type('plHabilita') # 'L'
	plHabilita = .F.
Endif

With Thisform.ParentForm.Pagina.Lista
	For lnI=1 to .Grupo_Op.ButtonCount
		If .Grupo_Op.Buttons(lnI).BaseClass = 'Commandbutton' And .Grupo_Op.Buttons(lnI).Visible
			.Grupo_Op.Buttons(lnI).Enabled = plHabilita
		EndIf
	EndFor
	.cmdFPOper.Enabled = plHabilita
	.Grupo_Saida.Sair.Enabled = plHabilita
EndWith



ENDPROC

* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
LParameters poForm, pnDataSes, pcEscolha

Local lcOperacao

Set DataSession to pnDataSes

Thisform.ParentForm = poForm
Thisform.PcEscolha  = pcEscolha
Thisform.ParentForm.Enabled  = .F.
lcOperacao = crSigCdOpe.Dopes

With Thisform
	.Caption = 'Status das Operações ' + AllTrim( Proper( crSigCdOpe.Dopes ) )
	.Top     = Thisform.ParentForm.Top + Thisform.ParentForm.Pagina.Lista.Grade.Top + 50
	.Left    = Thisform.ParentForm.Left + ;
			   Thisform.ParentForm.Pagina.Lista.Grade.Left + ;
			   Thisform.ParentForm.Pagina.Lista.Grade.Columns(1).Width + ;
			   Thisform.ParentForm.Pagina.Lista.Grade.Columns(3).Width + 16

	.cmdInserir.Visible = fChecaAcesso('SIGOPSTS','INSERIR') And ;
						  Iif( Type( 'Thisform.PcEscolha' ) # 'C', .T., ;
						  InList( Thisform.PcEscolha, 'INSERIR', 'ALTERAR' ) )
	.cmdInserir.Enabled = fChecaAcesso('SIGOPSTS','INSERIR')	And ;
						  Iif( Type( 'Thisform.PcEscolha' ) # 'C', .T., ;
						  InList( Thisform.PcEscolha, 'INSERIR', 'ALTERAR' ) )
	.cmdExcluir.Visible = fChecaAcesso('SIGOPSTS','EXCLUIR') And ;
						  Iif( Type( 'Thisform.PcEscolha' ) # 'C', .T., ;
						  InList( Thisform.PcEscolha, 'INSERIR', 'ALTERAR' ) )
	.cmdExcluir.Enabled = fChecaAcesso('SIGOPSTS','EXCLUIR')	And ;
						  Iif( Type( 'Thisform.PcEscolha' ) # 'C', .T., ;
						  InList( Thisform.PcEscolha, 'INSERIR', 'ALTERAR' ) )
	.shape1.Left = IIF(.cmdInserir.visible, .cmdInserir.Left - 5, IIF(.cmdExcluir.visible , .cmdExcluir.left - 5, .cmdSair.Left - 5))
	.shape1.width = IIF(.cmdInserir.visible, 3, IIF(.cmdExcluir.visible, 2, 1)) * .cmdsair.width + 16
EndWith

ThisForm.cntSombra.lblSombra.Caption = ThisForm.Caption
ThisForm.cntSombra.lblTitulo.Caption = ThisForm.Caption

Thisform.mCtrlBotoes(.F.)

lcSql = [Select a.*,b.Descs From SIGOPSTS a, SigStats b Where a.Dopes = ']+CrSigCdOpe.Dopes+[' And a.Status = b.cods ]
ThisForm.ParentForm.PodataMgr.Sqlexecute(lcsql,'LocalStats')

Select LocalStats
Goto Top	
With Thisform.GradeSubN
	.RecordSource          = 'LocalStats'
	.Column1.ControlSource = 'LocalStats.status'
	.Column2.ControlSource = 'LocalStats.Descs'
	.Refresh
EndWith

ENDPROC

------------------------------------------------------------
* PROCEDURE Click
------------------------------------------------------------
PROCEDURE Click
If Thisform.cmdInserir.Visible Or Thisform.cmdExcluir.Visible
	If ( Thisform.lHouveExcl Or Thisform.lHouveIns )
		Select LocalStats
		Go Bottom
		Go Top
		
		Select status,sum(1) as TT From LocalStats GROUP by 1 Into cursor Totais
		Select Totais
		Scan
			If TT > 1
				MessageBox('Status '+Alltrim(Totais.status)+' Digitado em Duplicidade!!!!',0+16,'Atenção')
				Return .f.
			EndIf
		EndScan
		
		Select CrSIGOPSTS
		Zap
		Select LocalStats
		Scan
			Scatter Memvar
			If Empty(m.status)
				Loop
			Endif
			Insert Into CrSIGOPSTS (Dopes,status,CidChaves) Values ;
				(CrSigCdOpe.Dopes,m.status,fUniqueIds())
		EndScan
		
		m.lnErro1 = ThisForm.ParentForm.poDatamgr.pnIdConn

		With ThisForm.ParentForm
			If m.lnErro1>0 
				lcDelete = [Delete From SIGOPSTS Where Dopes = ']+CrSigCdOpe.Dopes+[']
				m.lnErro1 = Iif(.poDataMgr.SqlExecute(lcDelete,'')>0,1,-1)
			EndIf 
			If m.lnErro1>0 
				m.lnErro1 = Iif(.poDataMgr.UpDate('crSIGOPSTS'),1,-1)
			EndIf 
			If m.lnErro1>0 
				m.lnErro1 = .poDataMgr.Commit()
			EndIf 
			If Not m.lnErro1>0 
				.poDataMgr.RollBack()
			EndIf 
		EndWith 
		If Not m.lnErro1>0 
			= MessageBox('Erro na Gravação dos Dados do Cadastro de Status.',48)
		EndIf 
	EndIf
ENDIF

Thisform.ParentForm.Enabled = .T.
Thisform.mCtrlBotoes(.T.)
ThisForm.Release
ENDPROC
