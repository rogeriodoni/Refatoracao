PROCEDURE alimentalista
Local lcCpo, llAcPend, llAcApro, llAcTodo
llAcTodo = fChecaAcesso(ThisForm.Tits,'TODOS')
llAcPend = Iif(llAcTodo,.t.,fChecaAcesso(ThisForm.Tits,'PENDENTE'))
llAcApro = Iif(llAcTodo,.t.,fChecaAcesso(ThisForm.Tits,'APROVADO'))

Select crSigSyCrs
Scan
	If (llAcPend And ! ChkApro) or (llAcApro And ChkApro) or llAcTodo
		Select crSigSyCit
		Scan
			If (crSigSyCit.Listas = 1) && Se o Campo Aparece na Lista
				If Seek(Str(crSigSyCrs.Cods, 6) + crSigSyCit.Campos, 'LocalCadRs', 'CodCampos')
					lcCpo = Alltrim(crSigSyCit.Campos)
					Replace &lcCpo. With Iif(crSigSyCit.Tipos=[N],Val(LocalCadRs.Resps),LocalCadRs.Resps) In crSigSyCrs
				EndIf
			EndIf
		EndScan
	Else
		Delete
	EndIf 
EndScan
* Fim **************************************************************************************
ENDPROC

------------------------------------------------------------
* PROCEDURE apanharespostas
------------------------------------------------------------
PROCEDURE apanharespostas
Local lcVal, lcObj

lcQuery = [Select Cods, Campos, Resps ] + ;
		    [From SigSyCrs ] + ;
		   [Where Tits = '] + ThisForm.Tits + [' ] + ;
		   [Order By Cods, Campos]

If (ThisForm.poDataMgr.SqlExecute(lcQuery, 'LocalCadRs') < 1)
	=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (LocalCadRs)')
	Return .f.
EndIf
Select LocalCadRs
Index On Str(Cods, 6) + Campos Tag CodCampos

Return .t.
* Fim **************************************************************************************
ENDPROC

PROCEDURE gravadados
* Rotina Para Gravação dos Dados da Ficha Técnica no Cursor Para Atualizar a Base
If ThisForm.CriouForm And Used('crGrvCadRs')
	Zap In crGrvCadRs

	lnCod = ThisForm.Pagina.Dados.getCods.Value

	Select crSigSyCit
	Scan
		lcObj = [memFicha] + Alltrim(Str(crSigSyCit.Ordems))

		lcVal     = ThisForm.poWindowCad.&lcObj..Value
		If Type('ThisForm.poWindowCad.&lcObj..InputMask') <> 'U'
			lcPicture = ThisForm.poWindowCad.&lcObj..InputMask
			lcPicture = Strtran(lcPicture,',','')
		Else
			lcPicture = ''
		EndIf
		
		If (crSigSyCit.Tipos=[M])
			lcVal = Alltrim(lcVal)
		Else
			If (crSigSyCit.Tipos=[D])
				lcVal = Dtoc(lcVal)
			Else
				lcVal = Alltrim(Iif(Empty(lcPicture),Str(lcVal),Transform(lcVal,lcPicture)))
			EndIf
		EndIf

		If Not Empty(lcVal)
			Insert Into crGrvCadRs (Tits, Campos, Cods, cIdChaves, Resps, Pergs) ;
							Values (ThisForm.Tits, crSigSyCit.Campos, lnCod, fUniqueIds(), lcVal, crSigSyCit.Descs)
		EndIf
	EndScan

	llGrv = .t.
	
	lcVal = ThisForm.poWindowCad.memFichaEmps.Value
	If Not Empty(lcVal)
		Insert Into crGrvCadRs (Tits, Campos, Cods, cIdChaves, Resps, Pergs) ;
						Values (ThisForm.Tits, [Emps], lnCod, fUniqueIds(), lcVal, [Empresa])
	Else
		llGrv = .f.
	EndIf

	lcVal = ThisForm.poWindowCad.memFichaDopes.Value
	If Not Empty(lcVal)
		Insert Into crGrvCadRs (Tits, Campos, Cods, cIdChaves, Resps, Pergs) ;
						Values (ThisForm.Tits, [Dopes], lnCod, fUniqueIds(), lcVal, [Operação])
	Else
		llGrv = .f.
	EndIf

	lcVal = ThisForm.poWindowCad.memFichaNumes.Value
	If Not Empty(lcVal)
		Insert Into crGrvCadRs (Tits, Campos, Cods, cIdChaves, Resps, Pergs) ;
						Values (ThisForm.Tits, [Numes], lnCod, fUniqueIds(), lcVal, [Código])
	Else
		llGrv = .f.
	EndIf

	lcVal = ThisForm.poWindowCad.memFichaNumes.Value
	If Not Empty(lcVal)
		Insert Into crGrvCadRs (Tits, Campos, Cods, cIdChaves, Resps, Pergs) ;
						Values (ThisForm.Tits, [Baixa], lnCod, fUniqueIds(), Usuar, [Baixa])
	Else
		llGrv = .f.
	EndIf

	lcVal = ThisForm.poWindowCad.memFichaAprova.Value
	If Not Empty(lcVal)
		Insert Into crGrvCadRs (Tits, Campos, Cods, cIdChaves, Resps, Pergs) ;
						Values (ThisForm.Tits, [Aprova], lnCod, fUniqueIds(), lcVal, [Aprovado])
	EndIf

	lcVal = ThisForm.poWindowCad.memFichaDatas.Value
	If Not Empty(lcVal)
		Insert Into crGrvCadRs (Tits, Campos, Cods, cIdChaves, Resps, Pergs) ;
						Values (ThisForm.Tits, [Datas], lnCod, fUniqueIds(), lcVal, [Data])
	EndIf

	Replace All ChkSubn With llGrv In crGrvCadRs

	Return .t.
Else
	Return .f.
EndIf
* Fim *********************************************************************************
ENDPROC

		Endif
		
		ThisForm.poWindowCad.pcGet[Alen(ThisForm.poWindowCad.pcGet,1),1] = lcObj
		ThisForm.poWindowCad.pcGet[Alen(ThisForm.poWindowCad.pcGet,1),2] = crSigSyCit.Obrigatorios
		ThisForm.poWindowCad.pcGet[Alen(ThisForm.poWindowCad.pcGet,1),3] = crSigSyCit.Campos

			If Empty(crSigSyCit.Files)
				ThisForm.poWindowCad.AddObject(lcObj, [fweditboxs])
				ThisForm.poWindowCad.&lcObj..plScroll      = .T.
				ThisForm.poWindowCad.&lcObj..pcOper       = ThisForm.pcEscolha
				If Empty(crSigSyCit.Pictures)
					ThisForm.poWindowCad.&lcObj..Width     = lnWid
				Else
					lnWids = Fontmetric(7,ThisForm.poWindowCad.&lcObj..FontName,ThisForm.poWindowCad.&lcObj..FontSize)*Len(Alltrim(crSigSyCit.Pictures))
					ThisForm.poWindowCad.&lcObj..Width     = lnWids
					ThisForm.poWindowCad.&lcObj..MaxLength = Len(Alltrim(crSigSyCit.Pictures))
				Endif
			Else
				ThisForm.poWindowCad.AddObject(lcObj, [LocEdit])
				ThisForm.poWindowCad.&lcObj..pcFiles      = crSigSyCit.Files
				ThisForm.poWindowCad.&lcObj..pcListas     = crSigSyCit.Fields

------------------------------------------------------------
* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
Lparameters pPrg

Private pTru, pNot
Public  laObjs[1]

ThisForm.Tits = Padr(pPrg, 20)

Declare ThisForm.pcObj[1]

With ThisForm
	.CriouForm = .f.

	.poDataMgr = CreateObject('fSqlConector', ThisForm.Name)

	pTru = .t.
	pNot = .t.

	If (.poDataMgr.pnIdconn > 0)
		If Not ThisForm.ApanhaRespostas()
			Return .f.
		EndIf

		ThisForm.poDataMgr.CursorQuery('SigSyCit', 'crSigSyCit', 'Tits', ThisForm.Tits)
		Select crSigSyCit
		Index On Str(Ordems, 2) Tag Ordems
		lcQryCadRs = [Select Distinct ChkSubn, ChkApro, Cods]
		Scan
			If (crSigSyCit.Listas = 1) && Se o Campo Aparece na Lista
				lcQryCadRs = lcQryCadRs + [, ]+Iif(crSigSyCit.Tipos = [N],[],[']) + ;
							 Iif(crSigSyCit.Tipos = [M], Space(50), Iif(crSigSyCit.Tipos = [N], [9999999999.99] , Space(12))) + ;
							 Iif(crSigSyCit.Tipos = [N],[],['])+[ as ] + Alltrim(crSigSyCit.Campos)
			EndIf
		EndScan
		lcQryCadRs = lcQryCadRs + ;
					  [ From SigSyCrs ] + ;
					  [Where Tits = '] + ThisForm.Tits + [' And ] + ;
					  	   [(ChkSubn = ?pTru Or ChkSubn = ?pNot) ] + ;
					  [Order By Cods]

		With .poDataMgr
			.AddCursor('SigSyCrs', 'Cods', 'crSigSyCrs', '', ThisForm.Pagina.Lista.Grade, lcQryCadRs)
			.pfSqlTabela(1).pColuna('Cods', '', '', 'Código', 52, .t.)
			Select crSigSyCit
			Scan
				If (crSigSyCit.Listas = 1) && Se o Campo Aparece na Lista
					.pfSqlTabela(1).pColuna(Alltrim(crSigSyCit.Campos), '', Alltrim(crSigSyCit.Pictures), Alltrim(crSigSyCit.Descs), ;
									  Iif(crSigSyCit.Tipos = [D], 80, Iif(crSigSyCit.Tipos = [N], 100, ;
									  Iif(!Empty(crSigSyCit.Pictures),Fontmetric(6,'Courier New',9)*Len(Alltrim(crSigSyCit.Pictures)),400))), .t.)
				EndIf
			EndScan

			* Adicionando um Cursor Apenas Para Gravação
			.AddCursor('SigSyCrs', 'cIdChaves', 'crGrvCadRs')
		EndWith

		=DoDefault(ThisForm.Tits)
		
		.SqlDados

		.AlimentaLista

		.CriarForm

		.Pagina.Lista.Grade.SetAll('DynamicForeColor','Iif(crSigSyCrs.ChkSubn, RGB(0,0,255), Iif(crSigSyCrs.ChkApro, RGB(0,128,0), RGB(0,0,0)))','Column')

		.Pagina.Lista.Grade.Column1.SetFocus
	Else
		=Messagebox('Impossível Efetuar Conexão Com o Servidor de Banco de Dados...', 0+48, 'CreateObject')
		Return .f.
	EndIf

	ThisForm.Pagina.Lista.Commandgroup1.Command1.Visible = fChecaAcesso(ThisForm.Tits,'APROVACAO')
	ThisForm.Pagina.Lista.Commandgroup1.Command2.Visible = fChecaAcesso(ThisForm.Tits,'BAIXA')

EndWith
* Fim **************************************************************************************
ENDPROC

EndIf

	Endif

	Replace Cods With fGerUniqueKey(ThisForm.Tits + _Empr) In crSigSyCrs
EndIf

	loDad.Grupo_Salva.Salva.Setfocus
Else
	loDad.getCods.Setfocus
EndIf

loDad.Refresh

If Not ThisForm.MontaObjetos()
	Return .f.
EndIf

If Inlist(ThisForm.pcEscolha, 'PROCURAR')
	loDad.getCods.Enabled = .t.
EndIf 

ThisForm.CarregaDados
* Fim **************************************************************************************
ENDPROC

------------------------------------------------------------
PROCEDURE Pagina.dados.grupo_salva.salva.Click
Local loDad, llTaOk, lcQuery

loDad = ThisForm.Pagina.Dados

ThisForm.Mostrarform()

If InList(ThisForm.pcEscolha, 'INSERIR', 'ALTERAR')

	For nPos = 1 To Alen(ThisForm.poWindowCad.pcGet,1)
		If ThisForm.poWindowCad.pcGet[nPos,2] = 1
			lcGets = ThisForm.poWindowCad.pcGet[nPos,1]
			If Empty(ThisForm.poWindowCad.&lcGets..Value)
				MessageBox('Campo de Preenchimento Obrigatório, não Preenchido!!!', 48, '')
				ThisForm.Mostrarform()
				ThisForm.poWindowCad.&lcGets..SetFocus 
				Return 0
			EndIf
		EndIf
	Next
				
	If Not Empty(ThisForm.poWindowCad.memFichaNumes.Value)
		llNum = .t.
		For I = 1 to Len(Alltrim(ThisForm.poWindowCad.memFichaNumes.Value))
			If Not IsDigit(Substr(Alltrim(ThisForm.poWindowCad.memFichaNumes.Value), I, 1))
				llNum = .f.
			EndIf
		Next
		If Not llNum
			=MessageBox('O Número da Operação Digitado Não é Válido!!!', 48, '')
			loDad.Grupo_Salva.Salva.SetFocus 
			ThisForm.MostrarForm()
			Return 0
		EndIf
	EndIf
	
	If (Empty(ThisForm.poWindowCad.memFichaEmps.Value)  And ;
		Not Empty(ThisForm.poWindowCad.memFichaDopes.Value) And Not Empty(ThisForm.poWindowCad.memFichaNumes.Value))
		=MessageBox('É Necessário Preencher a Empresa Ao Informar a Operação e o Número!!!', 48, '')
		loDad.Grupo_Salva.Salva.SetFocus 
		ThisForm.MostrarForm()
		Return 0
	EndIf

	If (Not Empty(ThisForm.poWindowCad.memFichaEmps.Value)  And ;
		Not Empty(ThisForm.poWindowCad.memFichaDopes.Value) And ;
		Not Empty(ThisForm.poWindowCad.memFichaNumes.Value))

		If ! crSigSyCrs.ChkApro
			MessageBox('Baixa não Aprovada !!!',16,'')
			Return 0
		Endif

		lcEdn = Padr(ThisForm.poWindowCad.memFichaEmps.Value,  03) + ;
				Padr(ThisForm.poWindowCad.memFichaDopes.Value, 20) + ;
				Padl(Alltrim(ThisForm.poWindowCad.memFichaNumes.Value), 6)

		ThisForm.poDataMgr.CursorQuery('SigMvCab', 'LocalEest', 'EmpDopNums', lcEdn, 'EmpDopNums')
		If Eof('LocalEest')
			=MessageBox('A Operação Com o Número Digitado Não Foi Encontrada na Empresa!!!', 48, '')
			loDad.Grupo_Salva.Salva.SetFocus 
			ThisForm.MostrarForm()
			Return 0
		EndIf
	EndIf
EndIf

llTaOk = ThisForm.GravaDados()

If llTaOk
	=DoDefault()
	ThisForm.OcultarForm()
	ThisForm.DesmontaObjetos
Else
	ThisForm.MostrarForm()
	ThisForm.poDataMgr.RollBack()
	Return 0
EndIf

ThisForm.mAtivaPagina1
* Fim **************************************************************************************
ENDPROC

------------------------------------------------------------
* PROCEDURE Command1.Click
------------------------------------------------------------
PROCEDURE Command1.Click
Private llOk, lnCod
llOK = .t.
If ! crSigSyCrs.ChkApro
	If MessageBox('Confirma Aprovação?',33,'') = 1

		ThisForm.Aprova = Usuar
			
		ThisForm.pcTime = Time() 

		Zap In crGrvCadRs

		lnCod = crSigSyCrs.Cods

		lcVal = ThisForm.Aprova
		If Not Empty(lcVal)
			Insert Into crGrvCadRs (Tits, Campos, Cods, cIdChaves, Resps, Pergs, chkApro) ;
							Values (ThisForm.Tits, [Aprova], lnCod, fUniqueIds(), lcVal, [Aprovado], .T.)
		EndIf
		Insert into LocalCadRs (Cods, Campos, Resps) Values (lnCod,'Aprova',lcVal)

		lcVal = Ttoc(Datetime())
		If Not Empty(lcVal)
			Insert Into crGrvCadRs (Tits, Campos, Cods, cIdChaves, Resps, Pergs, chkApro) ;
							Values (ThisForm.Tits, [Datas], lnCod, fUniqueIds(), lcVal, [Data], .T.)
		EndIf
		Insert into LocalCadRs (Cods, Campos, Resps) Values (lnCod,'Datas',lcVal)

		ThisForm.poDatamgr.SqlExecute([UpDate SigSyCrs Set ChkApro = ?llOk Where Cods = ?lnCod])
		
		If Not ThisForm.poDataMgr.Update('crGrvCadRs')
			=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (mSv_Alterar - crGrvCadRs)')
		EndIf

		ThisForm.poDataMgr.Commit()

		Replace crSigSyCrs.ChkApro With .t.
		ThisForm.Pagina.Lista.Grade.Refresh
		
		   
   		This.Caption = 'Desaprova'

		ThisForm.Pagina.Lista.Grupo_op.alterar.Enabled = .f.
		ThisForm.Pagina.Lista.Grupo_op.Excluir.Enabled = .f.

	EndIf
	
Else

	If MessageBox('Confirma Desaprovação?',33,'') = 1

		ThisForm.Aprova = Usuar

		ThisForm.pcTime = Time() 

		lnCod = crSigSyCrs.Cods
		llOk  = .F.
		
		ThisForm.poDatamgr.SqlExecute([Delete From SigSyCrs Where Cods = ?lnCod And Campos = 'Datas' And Tits = ']+ThisForm.Tits+['])
		ThisForm.poDatamgr.SqlExecute([Delete From SigSyCrs Where Cods = ?lnCod And Campos = 'Aprova' And Tits = ']+ThisForm.Tits+['])
		ThisForm.poDatamgr.SqlExecute([UpDate SigSyCrs Set ChkApro = ?llOk Where Cods = ?lnCod])
			
		ThisForm.poDataMgr.Commit()
		
		Select crSigSyCrs
		Replace crSigSyCrs.ChkApro With .f.
		ThisForm.Pagina.Lista.Grade.Refresh
		
		Select LocalCadRs 

		Locate for Cods = lnCod And Campos = 'Datas'
		Delete
		
		Locate for Cods = lnCod And Campos = 'Aprova'
		Delete

		This.Caption = 'Aprova'

		ThisForm.Pagina.Lista.Grupo_op.alterar.Enabled = .t.
		ThisForm.Pagina.Lista.Grupo_op.Excluir.Enabled = .t.

	EndIf
	
Endif


ENDPROC

* PROCEDURE Command2.Click
------------------------------------------------------------
PROCEDURE Command2.Click
If ! crSigSyCrs.ChkSubn
	If ! crSigSyCrs.ChkApro 
		MessageBox('Baixa não Aprovada !!!',16,'')
		Return
	EndIf

	DO Form SigOpCbx With ThisForm,crSigSyCrs.Cods To pcBaixado

	If pcBaixado = 'B'
		Select crSigSyCrs
		Replace crSigSyCrs.ChkSubn With .t.
		ThisForm.Pagina.Lista.Grade.Refresh
		ThisForm.Pagina.Lista.Grupo_op.alterar.Enabled = .f.
		ThisForm.Pagina.Lista.Grupo_op.Excluir.Enabled = .f.
		This.Parent.Command1.Enabled = .f.
		This.Caption = 'Canc Baixa'
	EndIf
	
Else

	If MessageBox('Confirma Cancelamento da Baixa',33,'') = 1
	
		lnCod = crSigSyCrs.Cods
		llOk  = .F.
		
		ThisForm.poDatamgr.SqlExecute([Delete From SigSyCrs Where Cods = ?lnCod And Campos = 'Emps' And Tits = ']+ThisForm.Tits+['])
		ThisForm.poDatamgr.SqlExecute([Delete From SigSyCrs Where Cods = ?lnCod And Campos = 'Dopes' And Tits = ']+ThisForm.Tits+['])
		ThisForm.poDatamgr.SqlExecute([Delete From SigSyCrs Where Cods = ?lnCod And Campos = 'Numes' And Tits = ']+ThisForm.Tits+['])
		ThisForm.poDatamgr.SqlExecute([Delete From SigSyCrs Where Cods = ?lnCod And Campos = 'Baixa' And Tits = ']+ThisForm.Tits+['])
		ThisForm.poDatamgr.SqlExecute([UpDate SigSyCrs Set ChkSubn = ?llOk Where Cods = ?lnCod])
			
		ThisForm.poDataMgr.Commit()
		
		Select crSigSyCrs
		Replace crSigSyCrs.ChkSubn With .f.
		ThisForm.Pagina.Lista.Grade.Refresh
		
		Select LocalCadRs 

		Locate for Cods = lnCod And Campos = 'Emps'
		Delete
		
		Locate for Cods = lnCod And Campos = 'Dopes'
		Delete

		Locate for Cods = lnCod And Campos = 'Numes'
		Delete

		Locate for Cods = lnCod And Campos = 'Baixa'
		Delete

		This.Caption = 'Baixa'

	Endif

Endif
ENDPROC
