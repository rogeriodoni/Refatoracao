* PROCEDURE Init

PROCEDURE Init
ThisForm.cntSombra.lblSombra.Caption = ThisForm.Caption
ThisForm.cntSombra.lblTitulo.Caption = ThisForm.Caption

lcQryBloqueio = [Select * From SigSyBlq]
With ThisForm
	.poDataMgr = CreateObject('fSqlConector', ThisForm.Name)

	If (.poDataMgr.pnIdconn > 0)
		With .poDataMgr
			.AddCursor('SigSyBlq', 'cIdChaves', 'crBloqueio', '', '', lcQryBloqueio)
		EndWith

		.poDataMgr.Requery('crBloqueio')

		Go Top In crBloqueio
		If Eof('crBloqueio')
			Insert Into crBloqueio (cIdChaves) Values (fUniqueIds())
		EndIf

		=DoDefault()
	Else
		=Messagebox('Impossivel Efetuar Conexão com Servidor de Banco de Dados...', 0+48, 'CreateObject')
		Return .f.
	EndIf
EndWith
* Fim **************************************************************************************

ENDPROC
