------------------------------------------------------------
* PROCEDURE msv_procurar

PROCEDURE msv_procurar
Local laCampos(3,2)

m.laCampos(1,1) = 'clacods'
m.laCampos(1,2) = thisform.pagina.dados.get_codigo.value
m.laCampos(2,1) = 'clagrupos'
m.laCampos(2,2) = thisform.pagina.dados.get_grupo.value
m.laCampos(3,1) = 'clacontas'
m.laCampos(3,2) = thisform.pagina.dados.get_conta.value
ThisForm.poDataMgr.LocateCursor('crSigCdCle',@m.laCampos)

ENDPROC

------------------------------------------------------------
* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
LPARAMETERS lcacesso

With ThisForm
	.poDatamgr = CreateObject('fSqlConector')
	With .poDatamgr
		.AddCursor('SigCdCle','cidchaves','crSigCdCle','clacods,clagrupos,clacontas',ThisForm.pagina.lista.grade)
		.pfSqlTabela(1).pColuna('clacods','','','Código',80,.t.)
		.pfSqlTabela(1).pColuna('clagrupos','','','Grupo',80,.t.)
		.pfSqlTabela(1).pColuna('clacontas','','','Contas',80,.t.)
		.pfSqlTabela(1).pColuna('setors','','','Setor',80,.t.)
		.AddCursor('SigCdGcr','codigos','crSigCdGcr','','')
		.AddCursor('SigCdCli','iclis','crSigCdCli','','')
	EndWith
	= DoDefault()
	.LockScreen = .f.
EndWith
ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Lista.Grupo_op.Click
------------------------------------------------------------
PROCEDURE Pagina.Lista.Grupo_op.Click
LParameters opcao

If !dodefault( opcao )
	ThisForm.mAtivapagina1
	Return
Endif

Do Case
	Case Inlist( ThisForm.pcEscolha, 'INSERIR', 'PROCURAR' )
		ThisForm.Pagina.Dados.Get_Codigo.SetFocus
	Case ThisForm.pcEscolha = 'ALTERAR'
		ThisForm.Pagina.Dados.Get_Conta.SetFocus
Other
	ThisForm.Pagina.Dados.Grupo_Salva.Salva.SetFocus
Endcase

ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click

Local laCampos(3,2)

If ThisForm.pcEscolha = 'INSERIR'
	If IsEmpty( thisform.pagina.dados.get_grupo.value )
		Messagebox( 'Grupo não pode ficar em branco!', 48, '' )
		ThisForm.Pagina.Dados.Get_grupo.SetFocus
		Return
	Endif
	m.laCampos(1,1) = 'clacods'
	m.laCampos(1,2) = thisform.pagina.dados.get_codigo.value
	m.laCampos(2,1) = 'clagrupos'
	m.laCampos(2,2) = thisform.pagina.dados.get_grupo.value
	m.laCampos(3,1) = 'clacontas'
	m.laCampos(3,2) = thisform.pagina.dados.get_conta.value

	If ThisForm.poDataMgr.ChkRegister('SigCdCle',@m.laCampos)
		Messagebox( 'Codigo, Grupo e Conta já Cadastrado!', 48, '' )
		ThisForm.Pagina.Dados.Get_Grupo.SetFocus
		Return
	Endif
	Select CrSigCdCle
	Replace cidchaves with fUniqueIds()

Endif

=DoDefault()

ThisForm.mAtivapagina1

ENDPROC

		m.loLista.mAddColuna('Descrs',Replicate('x',60),'Descrição')

	This.Refresh
	m.llRetorno = (.Not.IsEmpty(This.Value))
	
EndIf
m.llRetorno = .t.

Return(m.llRetorno)

		m.loLista.mAddColuna('rclis',Replicate('x',60),'Descrição')
		m.loLista.mAddColuna('grupos','','Grupo')
		m.loLista.mAddColuna('Inativas ', .f., .f., .f., .f., .f., .t.)
		m.loLista.gridProc.SetAll('DynamicForeColor', [Iif(crSigCdCli.Inativas == 1, Rgb(255, 0, 0), Rgb(0,0,0))], 'Column')
		m.loLista.Show()
	EndIf
	lcod = CrSigCdCli.iclis
	If !fAcessoContas(Usuar,Grupo,'C',lcod)
		Messagebox('Acesso Negado !!',0+48,'')
		lnom = ''
		lcod = ''
    	m.llRetorno = .f.
    Else
    	m.llRetorno = .t.
	Endif

	this.parent.get_conta.value = Iif(LastKey()==27,Space(15),crSigCdCli.iclis)
	This.Refresh
	
EndIf

Return(m.llRetorno)
