PROCEDURE Init
lcQryCrt = [Select * From SigPrCrt Where Emps = ']+_Empr+[' Or Emps = ' ' ]

With ThisForm
	.poDataMgr = CreateObject('fSqlConector', ThisForm.Name)

	If (.poDataMgr.pnIdconn > 0)
		With .poDataMgr
			.AddCursor('SigPrCrt', 'cods', 'crSigPrCrt', '', ThisForm.Pagina.Lista.Grade, lcQryCrt)
			.pfSqlTabela(1).pColuna('Cods', '', '', 'Código', 80, .t.)
			.pfSqlTabela(1).pColuna('Descs', '', '', 'Descrição', 220, .t.)
			.pfSqlTabela(1).pColuna('Fators', '', '', 'Fator', 80)
			.pfSqlTabela(1).pColuna('seqs', '', '', 'Ordem', 40)
		EndWith

		=DoDefault()
		Select crSigPrCrt
		Index On cods tag cods
		
		ThisForm.Pagina.Lista.Grade.Column1.SetFocus
	Else
		=Messagebox('Impossivel Efetuar Conexão com Servidor de Banco de Dados...', 0+48, 'CreateObject')
		Return .f.
	EndIf
EndWith
* Fim **************************************************************************************


ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Lista.Grupo_op.Click
------------------------------------------------------------
PROCEDURE Pagina.Lista.Grupo_op.Click
LParameters opcao

If !dodefault( opcao )
	ThisForm.mAtivapagina1
	Return
Endif

Do Case
	Case Inlist( ThisForm.pcEscolha, 'INSERIR', 'PROCURAR' )
		ThisForm.Pagina.Dados.Get_Codigo.SetFocus
	Case ThisForm.pcEscolha = 'ALTERAR'
		ThisForm.Pagina.Dados.Get_Descricao.SetFocus
Other
	ThisForm.Pagina.Dados.Grupo_Salva.Salva.SetFocus
Endcase

ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
If ThisForm.pcEscolha = 'INSERIR'
	If IsEmpty( ThisForm.pagina.dados.get_codigo.Value )
		Messagebox( 'Codigo não pode ficar em branco!', 48, '' )
		ThisForm.Pagina.Dados.Get_Codigo.SetFocus
		Return
	EndIf
	If ThisForm.poDataMgr.CursorQuery('SigPrCrt', 'crPesqSigPrCrt', 'cods', ThisForm.pagina.dados.get_codigo.Value)
		Messagebox( 'Código já Cadastrado!', 48, '' )
		ThisForm.Pagina.Dados.Get_Codigo.SetFocus
		Return
	Endif
Endif
=DoDefault()
ThisForm.mAtivapagina1

ENDPROC

Endif

ENDPROC
