------------------------------------------------------------
* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
LParameters lcAcesso

SqlChec = [Select a.*, b.Razas From SigFiChc a Inner Join SigCdEmp b on a.Emps = b.cEmps]

WITH Thisform

	.poDatamgr = CreateObject('fSqlConector',ThisForm.Name)

	If Not .poDatamgr.pnIdconn>0
		Messagebox('Impossivel Efetuar Conexão com Servidor de Banco de Dados.',(0+48),'Erro')
		Return(.f.)
	Endif

	With .poDatamgr
		.AddCursor('SigFiChc','cIdChaves','crSigFiChc','Emps,Tipos',ThisForm.pagina.lista.grade, SqlChec)
		.pfSqlTabela(1).pColuna('Emps','','','Emp.',50,.t.)
		.pfSqlTabela(1).pColuna('Razas','','','Razao Social',150,.t.)
		.pfSqlTabela(1).pColuna('Tipos','','','Rede',80,.t.)
	EndWith
Endwith	

=DoDefault()


ENDPROC

------------------------------------------------------------
* PROCEDURE msv_procurar
------------------------------------------------------------
PROCEDURE msv_procurar
Declare laChec[2,2]
laChec[1,1]="Emps" 
laChec[1,2]=ThisForm.Pagina.Dados.GetEmpresa.Value
laChec[2,1]="Tipos"
laChec[2,2]=ThisForm.Pagina.Dados.cmbTCons.Value

ThisForm.poDataMgr.LocateCursor('crSigFiChc',@laChec)

*Select crSigFiChc
*Locate for Emps = ThisForm.Pagina.Dados.GetEmpresa.Value And Tipos = ThisForm.Pagina.Dados.cmbTCons.Value


ENDPROC

------------------------------------------------------------
* PROCEDURE pagina.lista.grupo_op.Click
------------------------------------------------------------
PROCEDURE pagina.lista.grupo_op.Click
LParameters lnOpcao

If Type( 'lnOpcao' ) = 'N'
	This.Value = lnOpcao
Endif

If !dodefault( lnOpcao )
	Thisform.mAtivapagina1
	Return
Endif

If InList(ThisForm.pcEscolha,"INSERIR","PROCURAR")
	ThisForm.Pagina.Dados.GetEmpresa.SetFocus
EndIf

If InList(ThisForm.pcEscolha,"ALTERAR")
	If crSigFiChc.Tipos = "CHEQUE PRE" 
		ThisForm.Pagina.Dados.GetContrato.SetFocus
	Else
		ThisForm.Pagina.Dados.GetLogon.SetFocus
	Endif
EndIf

If InList(ThisForm.pcEscolha,"EXCLUIR","CONSULTAR")
	ThisForm.Pagina.Dados.Grupo_salva.Salva.SetFocus
EndIf

DoDefault()
ENDPROC

------------------------------------------------------------
* PROCEDURE pagina.dados.grupo_salva.Salva.Click
------------------------------------------------------------
PROCEDURE pagina.dados.grupo_salva.Salva.Click
If InList(ThisForm.pcEscolha,"INSERIR","ALTERAR")

	If Empty(ThisForm.Pagina.Dados.CmbTCons.Value)
		MessageBox("Tipo da Consulta Invalida",16,"")
		ThisForm.Pagina.Dados.CmbTCons.SetFocus
		Return
	EndIf

	If Empty(ThisForm.Pagina.Dados.GetEmpresa.Value)
		MessageBox("Empresa Invalida",16,"")
		ThisForm.Pagina.Dados.GetEmpresa.SetFocus
		Return
	EndIf

	If ThisForm.Pagina.Dados.CmbTCons.Value = "CHEQUE PRE" 

		If Empty(ThisForm.Pagina.Dados.GetContrato.Value)
			MessageBox("Contrato Invalido",16,"")
			ThisForm.Pagina.Dados.GetContrato.SetFocus
			Return
		EndIf
		
		If Empty(ThisForm.Pagina.Dados.GetRede.Value)
			MessageBox("Rede Invalida",16,"")
			ThisForm.Pagina.Dados.GetRede.SetFocus
			Return
		EndIf
		
		If Empty(ThisForm.Pagina.Dados.GetLoja.Value)
			MessageBox("Loja Invalida",16,"")
			ThisForm.Pagina.Dados.GetLoja.SetFocus
			Return
		EndIf
		
		If Empty(ThisForm.Pagina.Dados.GetIP.Value)
			MessageBox("IP Invalido",16,"")
			ThisForm.Pagina.Dados.GetIP.SetFocus
			Return
		EndIf
		If Empty(ThisForm.Pagina.Dados.GetPorta.Value)
			MessageBox("Porta Invalida",16,"")
			ThisForm.Pagina.Dados.GetPorta.SetFocus
			Return
		EndIf

	EndIf

	If ThisForm.Pagina.Dados.CmbTCons.Value <> "CHEQUE PRE" 

		If Empty(ThisForm.Pagina.Dados.GetLogon.Value)
			MessageBox("Logon Invalido",16,"")
			ThisForm.Pagina.Dados.GetLogon.SetFocus
			Return
		EndIf
		
		If Empty(ThisForm.Pagina.Dados.GetSenha.Value)
			MessageBox("Senha Invalido",16,"")
			ThisForm.Pagina.Dados.GetSenha.SetFocus
			Return
		EndIf

	EndIf
	
	If ! InList(ThisForm.Pagina.Dados.fwget1.Value, 'CP', 'CH') And ThisForm.Pagina.Dados.CmbTCons.Value = "SERASA" 
		MessageBox('Tipo de Consulta Invalida, Informar CP ou CH !!!', 16, '')
		ThisForm.Pagina.Dados.fwget1.SetFocus
		Return 
	Endif

	If InList(ThisForm.pcEscolha,"INSERIR")
		SqlChec = [Select * From SigFiChc Where Emps = ']+ThisForm.Pagina.Dados.GetEmpresa.Value+[' ]+;
    		                           [ And Tipos = ']+ThisForm.Pagina.Dados.CmbTCons.Value+[']
    	ThisForm.poDatamgr.SqlExecute(SqlChec,"csSigFiChc" )
    	Select csSigFiChc
    	If ! Eof()
			MessageBox("Acesso Ja Cadastrado",16,"")
			ThisForm.Pagina.Dados.GetEmpresa.SetFocus
