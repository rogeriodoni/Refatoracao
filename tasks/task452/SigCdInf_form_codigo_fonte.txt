* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
LPARAMETERS lcacesso

With ThisForm
	.poDatamgr = CreateObject('fSqlConector',Thisform.Name)
	
	If Not .poDatamgr.pnIdconn > 0
		Messagebox('Impossivel Efetuar Conexão com Servidor de Banco de Dados.',(0+48),'Erro')
		Return(.F.)
	Endif
	
	lcInfos = Space(1)
	lcSqlEmInf = [Select * From SigEmInf Where Infos = ?lcInfos]

	
	With .poDatamgr
		.AddCursor('SigPrInf','Infos','crSigPrInf','',ThisForm.pagina.lista.grade)
		.pfSqlTabela(1).pColuna('Infos','','','Codigo',60,.t.)
		.pfSqlTabela(1).pColuna('Descrs','','','Descrição',120,.t.)
		.AddCursor('SigEmInf','pkchave','crSigEmInf','','',lcSqlEmInf)
	EndWith
	
	*-- Montando a grade de Empresas
	=Afields(v01,'crSigEmInf')
	Create Cursor xEmInf From array v01

	With ThisForm.Pagina.Dados.Grade
		.RecordSource = 'xEmInf'
		.column1.ControlSource = 'xEmInf.Emps'
		.Column2.controlsource = 'xEmInf.percemps'
		.Refresh
	EndWith
	
	= DoDefault()
	.LockScreen = .f.
EndWith

ENDPROC

PROCEDURE msv_alterar
llOk = .t.

lcInfos = crSigPrInf.Infos
If (ThisForm.poDataMgr.SQLExecute([Delete From SigPrInf Where Infos = ?lcInfos], '') < 1)
	=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (msv_excluir - 1)')
	llOk = .f.
EndIf

If llOk
	If (ThisForm.poDataMgr.SQLExecute([Delete From SigEmInf Where Infos = ?lcInfos], '') < 1)
		=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (msv_excluir - 2)')
		llOk = .f.
	EndIf
EndIf

If llOk And Not Thisform.poDataMgr.Update('crSigPrInf')
	=Messagebox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (Update - crSigPrInf)')
	llOk = .F.
Endif

	=Messagebox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (Update - crSigEmInf)')
	llOk = .F.
EndIf

If llOk
	ThisForm.poDataMgr.Commit()
Else
	Thisform.poDataMgr.Rollback()
	Return .F.
Endif

PROCEDURE Pagina.Lista.Grupo_op.Click
LPArameters Opcao

Select CrSigPrInf
If !dodefault( Opcao )
	ThisForm.mAtivapagina1
	Return
Endif

Select xEmInf
Zap

lcSql = [Select a.* From SigEmInf a Where a.Infos = ']+CrSigPrInf.Infos+[']
=ThisForm.PodataMgr.SqlExecute(lcsql,'TmpInfos')

Select TmpInfos
Scan
	Select xEmInf
	Append Blank
	replace Infos with TmpInfos.Infos,;
			Emps with TmpInfos.Emps,;
			PercEmps with TmpInfos.PercEmps
EndScan

Select xEmInf
If Reccount() = 0
	Append Blank
EndIf

ThisForm.Pagina.Dados.Grade.Refresh

Select CrSigPrInf
Do Case
	Case Inlist( thisform.pcEscolha, 'INSERIR', 'PROCURAR' )
		ThisForm.Pagina.Dados.Get_Info.SetFocus
	Case Inlist( thisform.pcEscolha, 'ALTERAR' )
			ThisForm.Pagina.Dados.Get_Descr.SetFocus
	Case Inlist( thisform.pcEscolha, 'EXCLUIR', 'CONSULTAR' )
			ThisForm.Pagina.Dados.gRUpo_salva.salva.SetFocus
Endcase


ENDPROC

PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
If Inlist( thisform.pcEscolha, 'INSERIR')
	If Empty( CrSigPrInf.Infos)
		Messagebox( 'Código Inválido.', 48 , '' )
		ThisForm.Pagina.Dados.Get_Info.SetFocus
		Return
	Endif
	if ThisForm.poDataMgr.ChkRegister('SigPrInf','Infos',ThisForm.Pagina.Dados.Get_Info.Value)
		Messagebox( 'Código já cadastrada.', 48,'' )
		ThisForm.Pagina.Dados.Get_Info.SetFocus
		Return
	Endif
EndIf

If InList(ThisForm.pcescolha,'INSERIR','ALTERAR')
	lcInfos = CrSigPrInf.Infos
	
	Select CrSigEmInf
	Zap

	Select xEmInf
	Scan
		If Not Empty(Emps) &&And PercEmps > 0
			Select CrSigEmInf
			Append Blank
			replace Emps With xEmInf.Emps,;
					Infos With lcInfos,;
					PercEmps With xEmInf.PercEmps,;
					pkChave With fUniqueIds()
		EndIf
	EndScan
EndIf
Set Step On 
=DoDefault()
Thisform.mAtivapagina1
