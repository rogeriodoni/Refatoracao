PROCEDURE mctrlbotoes
LParameters plHabilita

Local lnI

If Type('plHabilita') # 'L'
	plHabilita = .F.
Endif

With Thisform.ParentForm.Pagina.Lista
	For lnI=1 to .Grupo_Op.ButtonCount
		If .Grupo_Op.Buttons(lnI).BaseClass = 'Commandbutton' And .Grupo_Op.Buttons(lnI).Visible
			.Grupo_Op.Buttons(lnI).Enabled = plHabilita
		EndIf
	EndFor
	.cmdFPOper.Enabled = plHabilita
	.Grupo_Saida.Sair.Enabled = plHabilita
EndWith

PROCEDURE Init
LParameters poForm, pnDataSes, pcEscolha

Local lcOperacao

Set DataSession to pnDataSes

Thisform.ParentForm = poForm
Thisform.PcEscolha  = pcEscolha
Thisform.ParentForm.Enabled  = .F.
lcOperacao = crSigCdOpe.Dopes

With Thisform
	.Caption = 'Tabelas de Desconto da Operações ' + AllTrim( Proper( crSigCdOpe.Dopes ) )
	.Top     = Thisform.ParentForm.Top + Thisform.ParentForm.Pagina.Lista.Grade.Top + 50
	.Left    = Thisform.ParentForm.Left + ;
			   Thisform.ParentForm.Pagina.Lista.Grade.Left + ;
			   Thisform.ParentForm.Pagina.Lista.Grade.Columns(1).Width + ;
			   Thisform.ParentForm.Pagina.Lista.Grade.Columns(3).Width + 16

	.cmdInserir.Visible = fChecaAcesso('SIGOPTDO','INSERIR') And ;
						  Iif( Type( 'Thisform.PcEscolha' ) # 'C', .T., ;
						  InList( Thisform.PcEscolha, 'INSERIR', 'ALTERAR' ) )
	.cmdInserir.Enabled = fChecaAcesso('SIGOPTDO','INSERIR')	And ;
						  Iif( Type( 'Thisform.PcEscolha' ) # 'C', .T., ;
						  InList( Thisform.PcEscolha, 'INSERIR', 'ALTERAR' ) )
	.cmdExcluir.Visible = fChecaAcesso('SIGOPTDO','EXCLUIR') And ;
						  Iif( Type( 'Thisform.PcEscolha' ) # 'C', .T., ;
						  InList( Thisform.PcEscolha, 'INSERIR', 'ALTERAR' ) )
	.cmdExcluir.Enabled = fChecaAcesso('SIGOPTDO','EXCLUIR')	And ;
						  Iif( Type( 'Thisform.PcEscolha' ) # 'C', .T., ;
						  InList( Thisform.PcEscolha, 'INSERIR', 'ALTERAR' ) )
	.shape1.Left = IIF(.cmdInserir.visible, .cmdInserir.Left - 5, IIF(.cmdExcluir.visible , .cmdExcluir.left - 5, .cmdSair.Left - 5))
	.shape1.width = IIF(.cmdInserir.visible, 3, IIF(.cmdExcluir.visible, 2, 1)) * .cmdsair.width + 16

ThisForm.cntSombra.lblSombra.Caption = ThisForm.Caption

Thisform.mCtrlBotoes(.F.)

lcSql = [Select a.*,b.Descrs From SigOpTdo a, SigOpTdz b Where a.Dopes = ']+CrSigCdOpe.Dopes+[' And a.Tabds = b.Codigos ]

	.RecordSource          = 'LocalTabdO'
	.Column1.ControlSource = 'LocalTabdO.Tabds'
	.Column2.ControlSource = 'LocalTabdO.Descrs'

------------------------------------------------------------
* PROCEDURE Click
------------------------------------------------------------
PROCEDURE Click
If Thisform.cmdInserir.Visible Or Thisform.cmdExcluir.Visible
	If ( Thisform.lHouveExcl Or Thisform.lHouveIns )
		Select LocalTabdO
		Go Bottom
		Go Top
		
		Select Tabds,sum(1) as TT From LocalTabdO GROUP by 1 Into cursor Totais
		Select Totais
		Scan
			If TT > 1
				MessageBox('Tabela de Desconto '+Alltrim(Totais.Tabds)+' Digitada em Duplicidade!!!!',0+16,'Atenção')
				Return .f.
			EndIf
		EndScan
		
		Select CrSigOpTdo
		Zap
		Select LocalTabdO
		Scan
			Scatter Memvar
			If Empty(m.Tabds)
				Loop
			Endif
			Insert Into CrSigOpTdo (Dopes,Tabds,CidChaves) Values ;
				(CrSigCdOpe.Dopes,m.Tabds,fUniqueIds())
		EndScan
		
		m.lnErro1 = ThisForm.ParentForm.poDatamgr.pnIdConn

		With ThisForm.ParentForm
			If m.lnErro1>0 
				lcDelete = [Delete From SigOpTdo Where Dopes = ']+CrSigCdOpe.Dopes+[']
				m.lnErro1 = Iif(.poDataMgr.SqlExecute(lcDelete,'')>0,1,-1)
			EndIf 
			If m.lnErro1>0 
				m.lnErro1 = Iif(.poDataMgr.UpDate('crSigOpTdo'),1,-1)
			EndIf 
			If m.lnErro1>0 
				m.lnErro1 = .poDataMgr.Commit()
			EndIf 
			If Not m.lnErro1>0 
				.poDataMgr.RollBack()
			EndIf 
		EndWith 
		If Not m.lnErro1>0 
			= MessageBox('Erro na Gravação dos Dados do Cadastro de Tabelas de Desconto.',48)
		EndIf 
	EndIf
ENDIF

Thisform.ParentForm.Enabled = .T.
Thisform.mCtrlBotoes(.T.)
ThisForm.Release
ENDPROC
