PROCEDURE msv_procurar
Local lcBusca

If Not Empty(ThisForm.Pagina.Dados.getCodigo.Value)
	ThisForm.poDataMgr.LocateCursor([crSigCdEve], [Codigo], ThisForm.Pagina.Dados.getCodigo.Value)
EndIf
* Fim **************************************************************************************

------------------------------------------------------------
* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
With ThisForm
	.poDataMgr = CreateObject([fSqlConector], .Name)

	If (.poDataMgr.pnIdconn > 0)
		With .poDataMgr
			.AddCursor([SigCdEve], [pkChaves], [crSigCdEve], [], ThisForm.Pagina.Lista.Grade)
			.pfSqlTabela(1).pColuna([Codigo], [], [], [Código],    080, .t.)
			.pfSqlTabela(1).pColuna([Nome],   [], [], [Descrição], 400, .t.)
			.pfSqlTabela(1).pColuna([Grupo],  [], [], [Grupo], 	   100, .t.)
			.pfSqlTabela(1).pColuna([Conta],  [], [], [Conta], 	   100, .t.)
		EndWith

		=DoDefault()

		.Pagina.Lista.Grade.Column1.SetFocus
	Else
		=Messagebox([Impossível Efetuar Conexão Com o Servidor de Banco de Dados...], 0+48, [CreateObject])
		Return .f.
	EndIf
EndWith
* Fim **************************************************************************************
ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Lista.Grupo_op.Click
------------------------------------------------------------
PROCEDURE Pagina.Lista.Grupo_op.Click
Parameters lnOpcao

If Not DoDefault(lnOpcao)
	Thisform.mAtivaPagina1
	Return .F.
Endif

Do Case
Case Inlist(Thisform.pcEscolha, [INSERIR], [PROCURAR])
	Thisform.Pagina.Dados.getCodigo.SetFocus

Case Inlist(Thisform.pcEscolha, [ALTERAR])
	Thisform.Pagina.Dados.getNome.SetFocus

Otherwise
	Thisform.Pagina.Dados.Grupo_Salva.Salva.SetFocus
EndCase

This.Parent.Parent.Dados.getDGrupo.Value = []
This.Parent.Parent.Dados.getDConta.Value = []

Thisform.podatamgr.CursorQuery('SigCdGcr','crSigCdGcr', 'Codigos', crSigCdEve.Grupo,[descrs])
Select crSigCdGcr
If !Eof()
	This.Parent.Parent.Dados.getDGrupo.Value = crSigCdGcr.descrs
Endif

Thisform.podatamgr.CursorQuery('SigCdCli','crSigCdCli', 'Iclis', crSigCdEve.Conta,[rclis])
Select crSigCdCli
If !Eof()
	This.Parent.Parent.Dados.getDConta.Value = crSigCdCli.rclis
Endif

* Fim **************************************************************************************

ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
If InList(ThisForm.pcEscolha, [INSERIR], [ALTERAR])
	If Empty(ThisForm.Pagina.Dados.getCodigo.Value)
		=Messagebox([Código Inválido!!!], 48, [Atenção!!!])
		ThisForm.Pagina.Dados.getCodigo.SetFocus()
		Return .f.
	EndIf

	If Empty(ThisForm.Pagina.Dados.getNome.Value)
		=Messagebox([Descrição Inválida!!!], 48, [Atenção!!!])
		ThisForm.Pagina.Dados.getNome.SetFocus()
		Return .f.
	EndIf
	
	If (ThisForm.pcEscolha = [INSERIR])
		lcQuery = [Select Codigo ] + ;
				    [From SigCdEve ] + ;
				   [Where Codigo = '] + Padr(ThisForm.Pagina.Dados.getCodigo.Value, 10) + [']

		If (ThisForm.poDataMgr.SqlExecute(lcQuery, [LocalBus]) < 1)
			=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (LocalBus 1)])
			Return .f.
		EndIf
		Go Top In LocalBus
		If Not Eof([LocalBus])
			=Messagebox([Código Já Cadastrado!!!], 48, [Atenção!!!])
			ThisForm.Pagina.Dados.getCodigo.SetFocus()
			Return .f.
		EndIf
	EndIf
EndIf

If (ThisForm.pcEscolha = [INSERIR])
	Replace pkChaves With fUniqueIds() in crSigCdEve 
EndIf 


=DoDefault()

ThisForm.mAtivaPagina1
* Fim **************************************************************************************
ENDPROC
