* PROCEDURE InteractiveChange

PROCEDURE InteractiveChange
m.lcArqTxt = ''
m.lcArqTxt = PutFile([Arquivo :],[],[TXT])

If Not Empty(m.lcArqTxt)
	If Not fVerificaPasta(m.lcArqTxt,2)
		MessageBox('Local ou Arquivo Invalido !!!',16,'')
		Replace nMarcas  With 0
		Replace cNomArqs With ''
		This.Value = 0
		This.Parent.Refresh
		Return(.f.)
	EndIf
Else
	MessageBox('Destino Invalido !!!',16,'')
	Replace nMarcas  With 0
	Replace cNomArqs With ''
	This.Value = 0
	This.Parent.Refresh
	Return(.f.)
Endif

lnReg = Recno()
Locate for cNomArqs = m.lcArqTxt
If ! eof()
	MessageBox('Nome Arquivo Ja Informado !!!',16,'')
	Go lnReg
	Replace nMarcas  With 0
	Replace cNomArqs With ''
	This.Value = 0
	This.Parent.Refresh
	Return(.f.)
EndIf
Go lnReg

Replace nMarcas  With 1
Replace cNomArqs With m.lcArqTxt
This.Value = 1

This.Parent.Refresh
ENDPROC
