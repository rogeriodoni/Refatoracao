PROCEDURE msv_procurar
Local lcBusca

If Not Empty(ThisForm.Pagina.Dados.getCods.Value)
	ThisForm.poDataMgr.LocateCursor([crSigCdFip], [Cods], ThisForm.Pagina.Dados.getCods.Value)
EndIf

If Not Empty(ThisForm.Pagina.Dados.getDescs.Value)
	ThisForm.poDataMgr.LocateCursor([crSigCdFip], [Descs], ThisForm.Pagina.Dados.getDescs.Value)
EndIf
* Fim **************************************************************************************
ENDPROC

------------------------------------------------------------
* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
With ThisForm
	.poDataMgr = CreateObject([fSqlConector], .Name)

	If (.poDataMgr.pnIdconn > 0)
		With .poDataMgr
			.AddCursor([SigCdFip], [Cods], [crSigCdFip], [], ThisForm.Pagina.Lista.Grade)
			.pfSqlTabela(1).pColuna([Cods],  [], [], [Códigos],   060, .t.)
			.pfSqlTabela(1).pColuna([Descs], [], [], [Descrição], 150, .t.)
		EndWith

		=DoDefault()

		.Pagina.Lista.Grade.Column1.SetFocus
	Else
		=Messagebox([Impossível Efetuar Conexão Com o Servidor de Banco de Dados...], 0+48, [CreateObject])
		Return .f.
	EndIf
EndWith
* Fim **************************************************************************************
ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Lista.Grupo_op.Click
------------------------------------------------------------
PROCEDURE Pagina.Lista.Grupo_op.Click
LParameters Opcao

If Not DoDefault(Opcao)
	ThisForm.mAtivapagina1
	Return .f.
EndIf

Do Case
	Case Inlist(ThisForm.pcEscolha, [INSERIR], [PROCURAR])
		 ThisForm.Pagina.Dados.getCods.SetFocus
		 
	Case (ThisForm.pcEscolha = [ALTERAR])
		 ThisForm.Pagina.Dados.getDescs.SetFocus

	Otherwise
		 ThisForm.Pagina.Dados.Grupo_Salva.Salva.SetFocus
EndCase

=DoDefault()
* Fim **************************************************************************************
ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
If InList(ThisForm.pcEscolha, [INSERIR], [ALTERAR])
	If (ThisForm.pcEscolha = [INSERIR])
		If Empty(crSigCdFip.Cods) Or (At(['], crSigCdFip.Cods) + At(["], crSigCdFip.Cods) > 0)
			=MessageBox([Código Inválido!!!], 48, [Atenção!!!]) 
			ThisForm.Pagina.Dados.getCods.SetFocus
			Return .f.
		EndIf
		
		If ThisForm.poDataMgr.ChkRegister([SigCdFip], [Cods], ThisForm.Pagina.Dados.getCods.Value)
			=MessageBox([Código Já Cadastrado!!!], 48, [Atenção!!!]) 
			ThisForm.Pagina.Dados.getCods.SetFocus
			Return .f.
		EndIf
	EndIf
	
	If Empty(crSigCdFip.Descs) Or (At(['], crSigCdFip.Descs) + At(["], crSigCdFip.Descs) > 0)
		=MessageBox([Descrição Inválida!!!], 48, [Atenção!!!]) 
		ThisForm.Pagina.Dados.getDescs.SetFocus
		Return .f.
	EndIf
EndIf

If (ThisForm.pcEscolha = [EXCLUIR])
	lcQuery = [Select CPros, DPros ] + ;
			    [From SigCdPro ] + ;
			   [Where CodFinP = '] + crSigCdFip.Cods + [']

	If (ThisForm.poDataMgr.SqlExecute(lcQuery, [LocalBusca]) < 1)
		=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (LocalBusca)])
		Return .f.
	EndIf
	Go Top In LocalBusca
	If Not Eof([LocalBusca])
		If (Reccount([LocalBusca])>1)
			lcMsg = [Existem ] + Alltrim(Str(Reccount([LocalBusca]))) + [ Produtos Utilizando Esse Modelo!!! Para Excluir o] + Chr(13) + ;
					[Modelo É Necessário Alterar o Cadastro de Produtos Primeiro!!!]
		Else
			lcMsg = [O Produto ] + Alltrim(LocalBusca.CPros) + [ - ] + Alltrim(LocalBusca.DPros) + [ Está Utilizando Esse Modelo!!! ] + Chr(13) + ;
					[Para Excluir o Modelo É Necessário Alterar o Cadastro do Produto Primeiro!!!]
		EndIf
		=MessageBox(lcMsg, 48, [Atenção!!!]) 
								
		ThisForm.Pagina.Dados.Grupo_salva.Cancelar.SetFocus
		Return .f.
	EndIf
EndIf

=DoDefault()

ThisForm.mAtivaPagina1
* Fim **************************************************************************************
ENDPROC

	ThisForm.mAtivaPagina1
EndIf
* Fim **************************************************************************************
ENDPROC
