* PROCEDURE checkcp
------------------------------------------------------------
PROCEDURE checkcp
Lparameters pFase, pUnidade, pMinutos, pEnvelope, pPrioridade

Select Envelopes
If Not Seek(pEnvelope)
	Append Blank
	Replace Nenvs With pEnvelope
	If pPrioridade = 999998
		Replace Datas With ThisForm.pagina.dados.getGeracao.Value
	Else
		Replace Datas With Date()
	Endif
EndIf

xSaldo = 0
DO While xSaldo < pMinutos
	Select CrSigCdPcp
	Sum Saldos to xSaldo For Fases = pFase And UniPrdts = pUnidade And Dtos(Datas)>=Dtos(Envelopes.Datas)
	If xSaldo < pMinutos
		If Not ThisForm.GeraDia(pFase,pUnidade)
			Return .f.
		Endif
	EndIf
EndDo
Return .T.

ENDPROC

------------------------------------------------------------
* PROCEDURE geradia
------------------------------------------------------------
PROCEDURE geradia
Lparameters pFase, pUnidade

llData  = ThisForm.Ddhoje
llVolta = .t.
Select CrTmpUprdt
Scan
	If Seek(CrTmpUprdt.Codigos + CrTmpUprdt.UniPrdts + Dtos(llData),'CrTmpCtHr')
		MinDds = fHtoM(Str(CrTmpCtHr.QtHoras,5,2)) * CrTmpCtHr.Qtds
	Else
		MinDds = fHtoM(Str(CrTmpUprdt.Hordds,5,2)) * CrTmpUprdt.Qtdes
		lnDow = Iif(Dow(llData)=1,8,Dow(llData))
		If Not Between(lnDow,2,CrTmpUprdt.ddSems + 1) Or fChkFeriado(ThisForm.PodataMgr,llData)
			minDds = 0
		EndIf
		If ( CrTmpUprdt.Qtdes = 0 Or CrTmpUprdt.Hordds = 0 ) And CrTmpUprdt.UniPrdts = pUnidade
			llVolta = .f.
		Endif
	Endif
	
	Select CrSigCdPcp
	Append Blank
	Replace Fases     With CrTmpUprdt.codigos,;
			UniPrdts  With CrTmpUprdt.UniPrdts,;
			Datas     With llData,;
			Cidchaves With fUniqueIds(),;
			Codigos   With CrSigCdPcz.codigos,;
			Cemps     With CrSigCdPcz.emps,;
			Minutos   With Mindds,;
			Saldos    With MinDds
EndScan
Thisform.Ddhoje = ThisForm.Ddhoje + 1
Return (llVolta)

ENDPROC

* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
lcQryPcp = [Select * From SigCdPcz ]

With ThisForm

	.poDataMgr = CreateObject('fSqlConector', ThisForm.Name)

	If (.poDataMgr.pnIdconn > 0)
		With .poDataMgr
			.AddCursor('SigCdPcz', 'Codigos', 'crSigCdPcz', '', ThisForm.Pagina.Lista.Grade, lcQryPcp)
			.pfSqlTabela(1).pColuna('Codigos',  '', '', 'Codigo', 	 080, .t.)
			.pfSqlTabela(1).pColuna('Titulos', '', '', 'Titulo',    350, .t.)
			.pfSqlTabela(1).pColuna('Datas', '', '', 'Data Geração',    150, .t.)
			.AddCursor('SigOpPic','CidChaves','CrSigOpPic','','')
			.AddCursor('SigPdMvf','CidChaves','CrSigPdMvf','','')
			.AddCursor('SigCdNec','CidChaves','CrSigCdNec','','')
			.AddCursor('SigCdPco','CidChaves','CrSigCdPco','','')
			.Addcursor('SigCdPcp','CidChaves','CrSigCdPcp','','')
			.Addcursor('SigCdPcg','CidChaves','CrSigCdPcg','','')
		EndWith

		=DoDefault()

		lcQuery = [Select Dopes, NDopes, Globalizas, Reservas ] + ;
				    [From SigCdOpe ] + ;
				   [Where Globalizas In (1, 2)]

		If (ThisForm.poDataMgr.SqlExecute(lcQuery, 'TmpOper') < 1)
			=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (TmpOper)')
			Return 0
		EndIf
		Select TmpOper
		Index On Dopes tag Dopes
		Go Top
		If Eof()
			=Messagebox('Nenhuma Operação Configurada Para Processar Ordem de Produção!!!', 32, '')
			Return .f.
		EndIf

		lcCampos = [DopEmphs, DopReqcs, DopPedcs, DopComps, TransfRes, GrPadClis, DoppPads, DopTrfCps, ] + ;
				   [GrPadVens, PrevProds, GrupoEsts, ContaEsts, GruReservs, ConReservs, AgrupEmph, grTrans, ]+;
				   [DoppServs, mascnums, GruEstps, ConEstps, transfencs, Ouros, GruConfs, ConConfs, GlobAutos ] 

		ThisForm.poDataMgr.CursorQuery('SigCdPam', 'crSigCdPam', .f., .f., lcCampos)

		ThisForm.Pagina.Lista.Grade.Column1.SetFocus
	Else
		=Messagebox('Impossível Efetuar Conexão Com o Servidor de Banco de Dados...', 0+48, 'CreateObject')
		Return .f.
	EndIf
EndWith

Thisform.Pagina.lista.processa.Visible   = fChecaAcesso('SIGCDPCP','MAPA')
Thisform.Pagina.Lista.previsao.Visible   = fChecaAcesso('SIGCDPCP','PREVISAO')
ThisForm.Pagina.Lista.prioridade.Visible = fChecaAcesso('SIGCDPCP','PRIORIZA')

ENDPROC

PROCEDURE Pagina.Lista.Grupo_op.Click
Lparameters Opcao

If Not DoDefault(Opcao)
	ThisForm.mAtivaPagina1
	Return 0
EndIf

Thisform.pagina.dados.mensagem.Caption = ' '

Do Case
	Case InList(Thisform.pcEscolha, 'INSERIR', 'PROCURAR')
		If InList(Thisform.pcEscolha, 'INSERIR')
			Replace Codigos With fGerUniqueKey(ThisForm.Name),;
					Datas   With Datetime(),;
					DtLancs With Date()
			ThisForm.pagina.Dados.GETCODIGOS.Refresh
			Thisform.Pagina.Dados.Grupo_salva.Salva.Enabled = .f.
		EndIf
		ThisForm.Pagina.Dados.getDataei.Value  = Ttod(CrSigCdPcz.EmissaoI)
		ThisForm.Pagina.Dados.getDataef.Value  = Ttod(CrSigCdPcz.Emissaof)
		ThisForm.Pagina.Dados.getDatapi.Value  = Ttod(CrSigCdPcz.Previs)
		ThisForm.Pagina.Dados.getDatapf.Value  = Ttod(CrSigCdPcz.Prevfs)
		ThisForm.Pagina.Dados.getGeracao.Value = Ttod(CrSigCdPcz.dtLancs)
		
		ThisForm.Pagina.Dados.Refresh
		ThisForm.Pagina.Dados.processar.Enabled  = .t.
		Thisform.Pagina.Dados.Get_Titulo.Setfocus

	Case InList(Thisform.pcEscolha, 'CONSULTAR', 'EXCLUIR')
		
		ThisForm.Pagina.Dados.getDataei.Value = Ttod(CrSigCdPcz.EmissaoI)
		ThisForm.Pagina.Dados.getDataef.Value = Ttod(CrSigCdPcz.Emissaof)
		ThisForm.Pagina.Dados.getDatapi.Value = Ttod(CrSigCdPcz.Previs)
		ThisForm.Pagina.Dados.getDatapf.Value = Ttod(CrSigCdPcz.Prevfs)
		
		ThisForm.Pagina.Dados.Refresh
	
		ThisForm.Pagina.Dados.processar.Enabled  = .f.
		Thisform.Pagina.Dados.Grupo_salva.Salva.Enabled = .t.
		Thisform.Pagina.Dados.Grupo_Salva.Salva.Setfocus
EndCase

Select CrSigCdPcz

ENDPROC

	lcOpCompara = Iif(OpPendente.Dopps = lcOperacao, lcOperacao, lcServico)


	Select crSigCdOpd
	llEnvelope = (crSigCdOpd.Nivels <> lcOpCompara Or CrSigCdOpd.ChkDests = 1)
	
	If OpPendente.Priors = 999998

		Select Nops, qtds, Space(10) as ContaOs, Space(10) as ContaDs, Datetime() as PrazoEnts, ;
		EmpDopNums, 0 as globalizas, 0 as Servicos, Emps, Dopes, Numes, cpros, 999998 as Priors From CrSigOpPic Where Nops = Oppendente.Numps ;
		Into Cursor CrTmpOpi ReadWrite

		Select CrTmpOpi
		Scan
			lcSql = [Select a.ContaOs, a.ContaDs, a.PrazoEnts, b.Globalizas, b.Servicos From SigMvCab a, SigCdOpe b ]+;
					[Where a.empdopNums = ']+CrTmpOpi.EmpdopNums+[' And a.Dopes = b.dopes ]
			ThisForm.Podatamgr.Sqlexecute(lcsql,'TprMvCab')
			Go top In TprMvCab
			Replace ContaOs    With TprMvCab.Contaos,;
					ContaDs    With TprMvCab.ContaDs,;
					PrazoEnts  With TprMvCab.PrazoEnts,;
					Servicos   With TprMvCab.Servicos,;
					Globalizas With TprMvCab.Globalizas in CrTmpOpi
		EndScan

	Else

		lcQuery = [Select a.Nops, Sum(a.Qtds) as Qtds, b.ContaOs, b.ContaDs, b.prazoents, b.empdopNums, ]+;
					[b.Emps, b.Dopes, b.Numes, c.Globalizas, c.Servicos, a.cpros, b.Priors ] + ;
				    [From SigOpPic a, SigMvCab b, SigCdOpe c ] + ;
				   [Where a.Nops = ?pNop And ] + ;
				   		 [a.EmpDopNums = b.EmpDopNums And b.Dopes = c.Dopes ] + ;
				   [Group by a.Nops, b.ContaOs, b.ContaDs, b.prazoents, b.empdopnums, b.emps, b.dopes, ]+;
				   [b.numes, c.globalizas, c.Servicos, a.cpros, b.Priors ]
			   		  
		If (ThisForm.poDataMgr.SqlExecute(lcQuery, 'crTmpOpi') < 1)
			=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (crTmpOpi)')
			Return 0
		EndIf

	EndIf
	
	Select crTmpOpi
	Scan
		_Qtde = Qtds

		If (_Qtde <> 0)

			Select crTmpMFas
			lcOp  = Transform(Val(Left(Str(crTmpMFas.Nops, 10), 6)), '@L 999999')
			lcSeq = Transform(Val(Right(Str(crTmpMFas.Nops, 10), 4)), '@L 9999')

			If llEnvelope
				lnEnv = crtmpMFas.Nenvs
				If (crtmpMFas.Nenvs = crtmpMFas.Nops)
					lnEnv = crtmpMFas.Numps
				EndIf
				Else
				lnEnv = 0
			EndIf
			DO Case
				Case OpPendente.Priors=999998
					m.Priors = 999998

				Case lnEnv = 0 And CrTmpOpi.Priors=0
					m.Priors = 999991
					
				Case CrTmpOpi.Priors > 0
					m.Priors = CrTmpOpi.Priors
					
				Otherwise
					m.Priors = 	999990
			EndCase	
			
			lnEnv = Iif(lnEnv = 0,CrTmpMfas.Nops,lnEnv)
			
			lcCliente = Iif(CrTmpOpi.Globalizas=1 Or CrTmpOpi.Servicos=1, CrTmpOpi.ContaOs, CrTmpOpi.ContaDs)

			lcGrMfas = CrTmpMfas.GrupoDs
			lnOrMfas = CrTmpMfas.Ordems
			
			If ( CrTmpMfas.Grupods = crSigCdPam.GrTrans And Not IsEmpty(crSigCdPam.GrTrans) )
				lcGrMfas = CrTmpMfas.GrupoOs
				lnOrMfas = CrTmpMfas.OrdemOs
			Endif
			
			lcSql = [Select * from SigCdPrf Where produtos = ']+CrTmpMfas.CodPds+[' Order by produtos,Grupos,Ordems,Ordem2]
			ThisForm.Podatamgr.Sqlexecute(lcsql,'crTmpProf')
			
			Select CrTmpProf
			Index On Str(Ordems,2) + Str(Ordem2,2) + Grupos Tag Ordem
			Index On Grupos + Str(ordems,2) + Str(Ordem2,2) tag Grupo
			
			If Seek(lcGrMfas + Str(lnOrMfas,2),'CrTmpProf','Grupo')
				Set Order To Ordem
				=Seek(Str(lnOrMfas,2))
			Else
				Set Order To Grupo descending
				Set Near On
				=Seek(lcGrMfas + Str(lnOrMfas,2))
				Set Near Off
				If Grupos = lcGrMfas
					Set Order To Ordem
					=Seek(Str(Ordems,2))
				Else
					Set Order To Grupo
					=Seek(lcGrMfas)
					Set Order To Ordem
					=Seek(Str(Ordems,2))
				EndIf
			EndIf
			
			lnSeq = 1
			
			DO While Not Eof()
				Select CrSigCdPco
				Append Blank
				Replace Priors   With m.Priors,;
						Nenvs    With lnEnv,;
						Nops     with CrTmpMfas.Nops,;
						Fases    With CrTmpProf.Grupos,;
						Uniprdts With CrTmpProf.UniPrdts,;
						Minutos  With (CrTmpProf.Minutos * _Qtde),;
						Emps     With CrTmpOpi.Emps,;
						Dopes    With CrTmpOpi.Dopes,;
						Numes    with CrTmpOpi.Numes,;
						Contas   with lcCliente,;
						Codigos  With CrSigCdPcz.codigos,;
						prazoEnts with CrTmpOpi.PrazoEnts,;
						Ordems   With CrTmpProf.Ordems,;
						Cidchaves with fUniqueIds(),;
						Seqs      With lnSeq,;
						Cpros     With CrTmpOpi.Cpros,;
						Qtds      With crTmpOpi.Qtds,;
						cEmps     With CrSigCdPcz.Emps
				If CrTmpProf.Minutos = 0
					Replace Seqs With 0
				Else
					lnSeq  = lnSeq + 1
				Endif
				xOrdem = CrTmpProf.Ordems
				Select CrTmpProF
				DO While Ordems = xOrdem And Not Eof()
					Skip
				Enddo
			Enddo
			
		EndIf

	EndScan

EndScan
oProg.Complete()

Select CrSigCdPco
Index On Nenvs Tag Nenvs

Select Nenvs, Min(Priors) as Priors From CrSigCdPco Where Priors < 999990 Into Cursor TmpEnv Group by Nenvs
Select TmpEnv
Scan
	Select CrSigCdPco
	=Seek(TmpEnv.nenvs)
	Replace Priors With TmpEnv.Priors While Nenvs = TmpEnv.Nenvs
EndScan

Select CrSigCdPcp
Index On Fases + UniPrdts + Dtos(datas) Tag Fase
Zap

Select CrSigCdPcg
Index On Fases + UniPrdts + Str(nenvs,10) + Str(Seqs,2) + Dtos(datas) Tag Fase
Zap

Thisform.Ddhoje = Date()

lcSql = [Select * from SigCdUpd Order by Codigos,UniPrdts ]
=ThisForm.Podatamgr.Sqlexecute(lcsql,'CrTmpUPrdt')
Select CrTmpUprdt
Index On Codigos + UniPrdts Tag codigos

llDtHoje = fDtoSQL(Date())

lcSql = [Select * from SigCdThs Where Datas >= ?llDtHoje Order by Datas]
=ThisForm.Podatamgr.Sqlexecute(lcsql,'CrTmpCtHr')
Select CrTmpCtHr
Index On Setors + UniPrdts + Dtos(Datas) Tag Setor

Select CrSigCdPco
Index On Str(PRIORS,8)+Str(Seqs,3)+Str(NENVS,12)+Str(NOPS,12) TAG SKGROUP1

Create Cursor Envelopes (Nenvs n(10), Datas d)
Index On Nenvs Tag Nenvs

Select Seqs, Priors, nenvs, Fases, UniPrdts, Sum(minutos) as Minutos, Date() as uData ;
  From CrSigCdPco ;
 Where Minutos > 0 And Seqs > 0 ;
 Group by Seqs, Priors, nenvs, Fases, UniPrdts ;
 Order by Seqs, Priors, nenvs, Fases, UniPrdts ;
  Into Cursor TmpPcpOp

lnContador = 1
Select TmpPcpOp
oProg = CreateObject('fwprogressbar', 'Gerando Capacidade de Produção ...', Reccount())
oProg.Show
Scan
	oProg.Update(.t.)

	Scatter Memvar
	
	If Not ThisForm.CheckCp(m.Fases, m.Uniprdts, m.minutos, m.nenvs, m.Priors )
		Loop
	Endif
	
	Select Envelopes
	=Seek(m.nenvs)

	Select CrSigCdPcp
	=Seek(m.Fases + m.UniPrdts + Dtos(Envelopes.Datas))
	_Baixar = m.minutos
	_Baixado= 0
	Scan While Fases + UniPrdts = m.Fases + m.UniPrdts And _Baixar > 0
	
		If Saldos = 0
			Loop
		Endif

		If CrSigCdPcp.Saldos >= _Baixar
			Replace CrSigCdPcp.Saldos With CrSigCdPcp.Saldos - _Baixar
			_Baixado = _Baixar
			_Baixar   = 0
		Else
			_Baixar = ( _Baixar - CrSigCdPcp.Saldos )
			_Baixado = CrSigCdPcp.Saldos
			Replace CrSigCdPcp.Saldos With 0
		EndIf
		
		Replace Datas With CrSigCdPcp.Datas in Envelopes

		Select CrSigCdPcg
		Set Order To Fase
		If Not Seek(m.Fases + m.UniPrdts + Str(m.Nenvs,10) + Str(m.Seqs,2) + Dtos(CrSigCdPcp.Datas))
			Append Blank
			Replace Fases     With m.fases,;
					UniPrdts  With m.Uniprdts,;
					Datas     With CrSigCdPcp.Datas,;
					Nenvs     With TmpPcpOp.Nenvs,;
					Seqs      With TmpPcpOp.Seqs,;
					Codigos   With CrSigCdPcz.Codigos,;
					CidChaves With fUniqueIds(),;
					Ordems    With lncontador,;
					Cemps     with CrSigCdPcz.Emps,;
					Priors    With TmpPcpOp.Priors
			lnContador = lnContador + 1
		EndIf
		Replace Minutos With Minutos + _Baixado
		
	EndScan

EndScan
oProg.Complete()

	ThisForm.Pagina.Dados.Mensagem.Caption = 'Executando Gravação Fisica dos arquivos '
	lnErro = ThisForm.Podatamgr.Commit()

		MessageBox('Nao foi possivel completar o Processamento (commit). tente novamente!!',16,'Falha na Conexão')
		Return 0
	Endif
Else
	ThisForm.Pagina.Dados.Mensagem.Caption = ' '
	ThisForm.Podatamgr.RollBack()
	MessageBox('Nao foi possivel completar o Processamento (inserir). tente novamente!!',16,'Falha na Conexão')
	Return 0
EndIf
