PROCEDURE Load
=fConfigGeral()

Create Cursor csEmpresas (ImpEmps l(1), CPros c(14), Pecas c(10), Promos c(25), Datas t Null, cIdChaves c(20))
Index On CPros + Promos Tag CPros
* Fim **************************************************************************************
ENDPROC

------------------------------------------------------------
* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
LParameters pAnt


ThisForm.cntSombra.lblSombra.Caption = ThisForm.Caption
ThisForm.cntSombra.lblTitulo.Caption = ThisForm.Caption
Private pPro

=DoDefault()

ThisForm.ParentForm = pAnt

Select crSigPrPmi
Scan
	Insert Into csEmpresas (ImpEmps, CPros, Pecas, Promos, Datas, cIdChaves) ;
					Values (.f., crSigPrPmi.CPros, crSigPrPmi.Pecas, crSigPrPmi.Promos, ;
							Nvl(crSigPrPmi.Datas, Ctod('')), crSigPrPmi.cIdChaves)
EndScan
Select csEmpresas

* Colocar as Empresas Que Possuem Transporte na Grade
With ThisForm.GridEmps
	.ColumnCount  = 4
	.RecordSource = 'csEmpresas'
	.ReadOnly     = .f.

	.Column1.ControlSource   = 'csEmpresas.ImpEmps'
	.Column1.Movable         = .f.
	.Column1.Resizable       = .f.
	.Column1.ReadOnly        = (ThisForm.ParentForm.pcEscolha <> 'CONSULTAR')
	.Column1.Sparse          = .f.
	.Column1.Check1.AutoSize = .t.

	.Column2.ControlSource = 'csEmpresas.Pecas'
	.Column3.ControlSource = 'csEmpresas.Promos'
	.Column4.ControlSource = 'csEmpresas.Datas'

	For I=2 To .ColumnCount
		lcCol = Alltrim(Str(I))
		.Column&lcCol..Movable   = .f.
		.Column&lcCol..Resizable = .f.
		.Column&lcCol..ReadOnly  = .t.
	Next
EndWith

ThisForm.btnRetirar.Visible = (ThisForm.ParentForm.pcEscolha <> 'CONSULTAR')
ThisForm.cmdRetirar.Visible = (ThisForm.ParentForm.pcEscolha <> 'CONSULTAR')

Go Top In csEmpresas

ThisForm.GridEmps.Refresh
* Fim ******************************************************************************************
ENDPROC

------------------------------------------------------------
* PROCEDURE Click
------------------------------------------------------------
PROCEDURE Click
Private pChv

If fChecaAcesso('SigCdPrm', 'EXCLUIR') Or fChecaAcesso('SigCdPrm', 'ALTERAR')
	Select cIdChaves ;
	  From csEmpresas ;
	 Where ImpEmps ;
	  Into Cursor csApagar

	If (RecCount('csApagar') < 1)
		=MessageBox('Não Existem Promoções Selecionadas Para Retirar o Produto!!!', 0+48, '')
		ThisForm.GridEmps.Column1.SetFocus
		Return 0
	Else
		If (MessageBox('Confirma a Retirada do Produto das Promoções Selecionadas ?', 4+32+256, '') = 6)
			Select csApagar
			Scan
				pChv = csApagar.cIdChaves
		
				If (ThisForm.ParentForm.poDataMgr.SQLExecute([Delete From SigPrPmi Where cIdChaves = ?pChv], '') < 1)
					=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (Excluindo Promoção)')
					Return 0
				EndIf
				Delete From crSigPrPmi Where cIdChaves = pChv
				Delete From csEmpresas Where cIdChaves = pChv
			EndScan
		Endif
	Endif
Else
	=MessageBox('Você Não Tem Acesso Para Alterar/Excluir Promoções!!!', 0+48, '')
	ThisForm.GridEmps.Column1.SetFocus
	Return 0
Endif
ThisForm.GridEmps.Refresh
* Fim ******************************************************************************************
ENDPROC
