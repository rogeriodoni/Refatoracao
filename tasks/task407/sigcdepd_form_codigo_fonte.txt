* PROCEDURE carregaitens
------------------------------------------------------------
PROCEDURE carregaitens
Private pTru, pNop

Select xNensi
Zap

ThisForm.poDataMgr.CursorQuery('SigCdOpd', 'LocalOpp', 'Dopps', TmpNens.Dopps, 'Dopps,NDopps,TNivels,Artigos,Nivels,BxAutos ')

Select TmpOperacao
Scan

	_Operacao  = Right(Str(TmpOperacao.Codigos, 12), 10)
	_nOperacao = Val(_Operacao)
	_Nivel 	   = TmpNens.Emps + LocalOpp.Nivels + Str(_nOperacao, 10)

	_Posicao 	   = TmpNens.Emps + TmpNens.Dopps + Str(TmpNens.Numps, 10)
	_BaixaSubnivel = (LocalOpp.BxAutos = 1)

	ThisForm.poDataMgr.CursorQuery('SigCdNec', 'LocalNens', 'EmpDNps', _Nivel)
	ThisForm.poDataMgr.CursorQuery('SigPdMvf', 'LocalMFas', 'EmpDNps', _Nivel)

	Select LocalMFas
	Scan
		_Nops = Nops
		If _BaixaSubNivel
			_CheckBaixa = LocalOpp.Nivels
			pTru = .t.
			lcSql = [Select CidChaves From SigCdNec Where EmpDnPs = '] + TmpNens.Emps + _CheckBaixa + Str(_Nops, 10) + [']
			If (ThisForm.poDataMgr.SqlExecute(lcSql, 'LocalNens2') < 1)
				=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (Marca - Update - SigCdNec)')
				Return 0
			EndIf
			Select LocalNens2
			Scan
				lcQuery = [Update SigCdNec Set ChkSubn = ?pTru Where CidChaves = '] +LocalNens2.CidChaves + [']
				If (ThisForm.poDataMgr.SqlExecute(lcQuery, '') < 1)
					=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (Marca - Update - SigCdNec)')
					Return 0
				EndIf
			EndScan
		EndIf

		ThisForm.poDataMgr.CursorQuery('SigCdPro', 'LocalPro', 'CPros', LocalMfas.CodPds, 'MatPrincs, Cgrus')
		ThisForm.PodataMgr.CursorQuery('SigCdGrp', 'LocalGru', 'CGrus', LocalPro.Cgrus,[Mercs])
		ThisForm.PodataMgr.CursorQuery('SigCdGpr','LocalGgrp','Codigos',LocalGru.Mercs,[MatPrincs])

		_MatP = IIF(!EMPTY(LocalPro.MatPrincs),LocalPro.MatPrincs,IIF(!EMPTY(LocalGgrp.Matprincs),LocalGgrp.MatPrincs,CrSigCdPam.Ouros))

		ThisForm.poDataMgr.CursorQuery('SigPrCpo', 'LocalCompo', 'CPros', LocalMfas.CodPds, 'Mats, Qtds, Cats')
		Select LocalCompo
		Scan
			ThisForm.poDataMgr.CursorQuery('SigCdPro', 'LocalPro', 'CPros', LocalCompo.Mats, 'CUnis, CGRus, DPros, Varias')
			ThisForm.poDataMgr.CursorQuery('SigCdUni', 'LocalUni', 'CUnis', LocalPro.CUnis, 'Fators')

			If (LocalCompo.Mats = _MatP) And (LocalMFas.Pesos <> 0)
				_QtAnt = LocalMFas.Pesos
			Else
				_QtAnt = LocalMFas.Qtds * LocalCompo.Qtds
			Endif

			If (LocalUni.Fators = 0)
				_PsAnt = LocalMFas.Qtds * LocalCompo.Qtds * LocalMFas.Pesos
			Else
				_PsAnt = LocalMFas.Qtds * LocalCompo.Qtds * LocalUni.Fators
			EndIf

			Insert Into xNensi (Emps, Dopps, Numps, Nops, NEnvs, CMats, CDescs, CUnis, CGrus,;
								QtFaseAnt, PsFaseAnt, Visivel, Rateio, FlagIncs, Cats, Qtds, Pesos) ;
						Values (TmpNens.Emps, TmpNens.Dopps, TmpNens.Numps, xMFas.Nops, xMFas.Nops , ;
								LocalCompo.Mats, LocalPro.Dpros, LocalPro.CUnis, LocalPro.CGrus, _QtAnt, ;
								_PsAnt, .f., Iif(LocalPro.Varias = 1, 2, 3), .t., LocalCompo.Cats, ;
								_QtAnt, _PsAnt)

			If Not Seek(_Operacao + LocalCompo.Mats, 'xNensi', 'NEnvCMat')
				Insert Into xNensi (Nops, Nenvs, CMats, CDescs, CUnis, CGrus, Visivel, Rateio, FlagIncs, Cats) ;
							Values (xMFas.Nops,xMFas.Nops, LocalCompo.Mats, LocalPro.DPros, ;
									LocalPro.CUnis, LocalPro.CGrus, .t., 1, .t., LocalCompo.Cats)
			EndIf
			Replace QtFaseAnt With xNensi.QtFaseAnt + _QtAnt, ;
					Qtds 	  With xNensi.Qtds + _QtAnt In xNensi

			If (LocalUni.Fators = 0)
				Replace PsFaseAnt With xNensi.PsFaseAnt + _PsAnt, ;
						Pesos 	  With 0 In xNensi
			Else
				Replace PsFaseAnt With xNensi.PsFaseAnt + _PsAnt, ;
						Pesos 	  With xNensi.Pesos + _PsAnt In xNensi
			EndIf
		EndScan
	EndScan
EndScan
Return .t.
ENDPROC

* PROCEDURE criacursores
------------------------------------------------------------
PROCEDURE criacursores
Select * From CrSigCdNec Where 0=1 Into Cursor TmpNens ReadWrite

Select *, 000000.000 as PsFaseAnt, 0 as Rateio, .f. as Visivel, 000000.000 as QtFaseAnt ;
From CrSigCdNei Where 0=1 Into Cursor xNensi ReadWrite
Index On Recno() For Visivel Tag Editar
Index On Cats Tag Cats
Index On Nops Tag Nops
Index On Str( Nops, 10 ) + Str( Recno(), 10 ) For Rateio # 1 Tag Gravar
Index On Str( Nops, 10 ) + Str( Rateio, 1 ) Tag Rateio
Index On Str( Nops, 10 ) + CMats Tag NopCMat
Index On Str( NEnvs, 10 ) + CMats For Visivel Tag NEnvCMat
Set Order to Editar

Select * From CrSigPdMvf Where 0=1 Into Cursor xMFas ReadWrite
Index on Nenvs Tag Nenvs
Index on Nops Tag Nops

Select * From CrSigPdMvf Where 0=1 Into cursor xMFas2 ReadWrite
Index on Nops Tag Nops

Create Cursor TmpOperacao ( Codigos n(12) )
Index on Codigos Tag Codigos 
Set Order to 

Create Cursor xOpi ( Nops n(10), Oper c(31), Qtds n(10,3), CPros c(14), Indiv l, Amarra n(10), Nenvs n(10), Emps c(3), Dopes c(20), Numes n(6) )
Index on Nops tag Nops
Set Order to

Create Cursor xPesa ( cBars n(14), Oper c(31), CPros c(14), Qtds n(10,3), Peso n(10,3), Amarra n(10), Nenvs n(10), Emps c(3), Dopes c(20),;
					  Numes n(6), Nops n(10), Conta c(10), Obs c(10), Cors c(4), Tams c(2), MatPs c(10), Peso2 n(10,3), pesoms n(10,3), ;
					  Usulibs c(10), Pesofixs n(10,3), Especial L)
Index On Nops Tag Nops
Index on Amarra Tag Amarra
Set Order to 

With ThisForm.Pagina.Dados.GradeOperação
	.ColumnCount            =  1
	.RecordSource           = 'TmpOperacao'
	.Column1.ControlSource  = 'TmpOperacao.Codigos'
	.Column1.InputMask 		= '999999999999'
EndWith
* Fim **************************************************************************************

ENDPROC

* PROCEDURE excluirop
------------------------------------------------------------
PROCEDURE excluirop
Private pZer, pDay, pNop
Local _ChaveDel1, _ChaveDel2, _ChaveEest, llTaOk

pZer = 0
pDay = Datetime()

_Nops = Int(crSigCdNec.Numps / 10000)

ThisForm.poDataMgr.CursorQuery('SigMvCab', 'LocalEest', 'Nops', _Nops, 'Emps, Dopes, Nums, Nops, cIdChaves')
Select LocalEest
Scan
	_ChaveEest = LocalEest.Emps + LocalEest.Dopes + Str(LocalEest.Numes, 6)

	* Excluindo do SigMvCab
	If (LocalEest.Dopes = crSigCdPam.TransfRes)
		lcQuery = [Delete From SigMvCab ] + ;
				   [Where cIdChaves = '] + LocalEest.cIdChaves + [']
	Else
		lcQuery = [Update SigMvCab ] + ;
					 [Set Nops = ?pZer, ] + ;
					 	 [DtAlts = ?pDay ] + ;
				   [Where cIdChaves = '] + LocalEest.cIdChaves + [']
	EndIf

	If (ThisForm.poDataMgr.SqlExecute(lcQuery, '') < 1)
		=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (ExcluirOp - SigMvCab)')
		Return 0
	EndIf

	* Excluindo do SigMvItn
	If (LocalEest.Dopes = crSigCdPam.TransfRes)
		lcQuery = [Delete From SigMvItn ] + ;
				   [Where EmpDopNums = '] + _ChaveEest + [']
	Else
		lcQuery = [Update SigMvItn ] + ;
					 [Set QtProds = ?pZer, ] + ;
					 	 [QtBxProds = ?pZer, ] + ;
					 	 [DtAlts = ?pDay ] + ;
				   [Where EmpDopNums = '] + _ChaveEest + [']
	EndIf

	If (ThisForm.poDataMgr.SqlExecute(lcQuery, '') < 1)
		=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (ExcluirOp - SigMvItn)')
		Return 0
	EndIf

	* Excluindo do SigMvHst
	lcCpo = 'Emps, Grupos, Estos, CPros, Datas, CodCors, CodTams, cIdChaves'
	ThisForm.poDataMgr.CursorQuery('SigMvHst', 'LocalHis', 'EmpDopNums', _ChaveEest, lcCpo)
	Select LocalHis
	Scan
		=fRecalculaP(LocalHis.Emps, LocalHis.Grupos, LocalHis.Estos, LocalHis.CPros, LocalHis.Datas, LocalHis.CodCors, LocalHis.CodTams, ThisForm.poDataMgr)
		=fRecalculaC(LocalHis.Emps, LocalHis.CPros, LocalHis.Datas, ThisForm.poDataMgr)

		If (LocalEest.Dopes = crSigCdPam.TransfRes)
			lcQuery = [Delete From SigMvHst ] + ;
					   [Where cIdChaves = '] + LocalHis.cIdChaves + [']

			If (ThisForm.poDataMgr.SqlExecute(lcQuery, '') < 1)
				=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (ExcluirOp - SigMvHst (1))')
				Return 0
			EndIf
		EndIf
	EndScan
EndScan

ThisForm.poDataMgr.CursorQuery('SigCdNec', 'LocalNens', 'Nops', _Nops, 'Emps, Dopps, Numps, Nops, cIdChaves')
Select LocalNens
Scan
	_ChaveDel1 = LocalNens.Emps + LocalNens.Dopps + Str(LocalNens.Numps, 10)
	_ChaveDel2 = LocalNens.Emps + LocalNens.Dopps + Str(LocalNens.Numps, 06)

	* Excluindo do SigCdNec
	lcQuery = [Delete From SigCdNec ] + ;
			   [Where cIdChaves = '] + LocalNens.cIdChaves + [']

	If (ThisForm.poDataMgr.SqlExecute(lcQuery, '') < 1)
		=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (ExcluirOp - SigCdNec)')
		Return 0
	EndIf

	* Excluindo do SigCdNei
	lcQuery = [Delete From SigCdNei ] + ;
			   [Where EmpDNps = '] + _ChaveDel1 + [']

	If (ThisForm.poDataMgr.SqlExecute(lcQuery, '') < 1)
		=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (ExcluirOp - SigCdNei)')
		Return 0
	EndIf

	* Excluindo do SigOpPic
	pNop = LocalNens.Numps
	lcQuery = [Delete From SigOpPic ] + ;
			   [Where Nops = ?pNop]

	If (ThisForm.poDataMgr.SqlExecute(lcQuery, '') < 1)
		=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (ExcluirOp - SigOpPic)')
		Return 0
	EndIf

	* Excluindo do SigMvHst
	lcCpo = 'Emps, Grupos, Estos, CPros, Datas, CodCors, CodTams'
	ThisForm.poDataMgr.CursorQuery('SigMvHst', 'LocalHis', 'EmpDopNums', _ChaveDel2, lcCpo)
	Select LocalHis
	Scan
		=fRecalculaP(LocalHis.Emps, LocalHis.Grupos, LocalHis.Estos, LocalHis.CPros, LocalHis.Datas, LocalHis.CodCors, LocalHis.CodTams, ThisForm.poDataMgr)
		=fRecalculaC(LocalHis.Emps, LocalHis.CPros, LocalHis.Datas, ThisForm.poDataMgr)

		lcQuery = [Delete From SigMvHst ] + ;
				   [Where EmpDopNums = '] + _ChaveEest + [']

		If (ThisForm.poDataMgr.SqlExecute(lcQuery, '') < 1)
			=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (ExcluirOp - SigMvHst (2))')
			Return 0
		EndIf
	EndScan

	* Excluindo do SigPdMvf
	lcQuery = [Delete From SigPdMvf ] + ;
			   [Where EmpDNps = '] + _ChaveDel1 + [']

	If (ThisForm.poDataMgr.SqlExecute(lcQuery, '') < 1)
		=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (ExcluirOp - SigPdMvf)')
		Return 0
	EndIf
EndScan

llTaOK = .t.
If Not fRecalculaP(.t., ThisForm.poDataMgr)
	=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (SigOpClP)')
	llTaOk = .f.
EndIf
If llTaOk And Not fRecalculaC(.t., .f., .f., ThisForm.poDataMgr)
	=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (SigOpClC)')
	llTaOk = .f.
EndIf
If llTaOk
	If (ThisForm.poDataMgr.Commit() < 1)
		=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (Commit)')
		llTaOk = .f.
	EndIf
EndIf
If Not llTaOk
	ThisForm.Podatamgr.Rollback()
Endif

ENDPROC

PROCEDURE Init
Lparameters laNop, lcCnx

Private pIds, pOpI, pOpF, pBar


ThisForm._Dopp = Padr('ENCERRAMENTO DIRETO', 20)
thisform.cEmps = _Empr
thisform.lAutomatico = Iif(Type([lcCnx])=[C],.t.,.f.)

pIds = Replicate(Chr(254), 20)
pOpI = thisform.cEmps + ThisForm._Dopp + Str(0, 10)
pOpF = thisform.cEmps + ThisForm._Dopp + Str(9999999999, 10)
pBar = 0

lcQryNens  = [Select * From SigCdNec  Where EmpDNps BetWeen ?pOpI And ?pOpF]
lcQryNensi = [Select * From SigCdNei Where cIdChaves = ?pIds]
lcQryMFas  = [Select * From SigPdMvf  Where cIdChaves = ?pIds]
lcQryHis   = [Select * From SigMvHst   Where cIdChaves = ?pIds]
lcQryEti   = [Select * From SigOpEtq   Where CBars = ?pBar]

ThisForm.Tiposubnivel = ' '
With ThisForm
	.poDataMgr = CreateObject('fSqlConector', Iif(Type([lcCnx])=[C],lcCnx,ThisForm.Name))

	If (.poDataMgr.pnIdconn > 0)
		With .poDataMgr
			.AddCursor('SigCdNec', 'cIdChaves', 'crSigCdNec', '', ThisForm.Pagina.Lista.Grade, lcQryNens)
			.pfSqlTabela(1).pColuna('Numps',    '', '', 'Código',        070, .t.)
			.pfSqlTabela(1).pColuna('Datas',    '', '', 'Data',          073, .t.)
			.pfSqlTabela(1).pColuna('TotPesos', '', '', 'Total de Peso', 070)
			.pfSqlTabela(1).pColuna('GrupoOs',  '', '', 'Gr. Origem',    070)
			.pfSqlTabela(1).pColuna('ContaOs',  '', '', 'Ct. Origem',    070)
			.pfSqlTabela(1).pColuna('GrupoDs',  '', '', 'Gr. Destino',   070)
			.pfSqlTabela(1).pColuna('ContaDs',  '', '', 'Ct. Destino',   070)
			.pfSqlTabela(1).pColuna('Usuars',   '', '', 'Usuário', 	  070)

			.AddCursor('SigCdNei', 'cIdChaves', 'crSigCdNei', '', '', lcQryNensi)
			.AddCursor('SigPdMvf', 'cIdChaves', 'crSigPdMvf', '', '', lcQryMFas)
			.AddCursor('SigMvHst', 'cIdChaves', 'crSigMvHst', '', '', lcQryHis)
			.AddCursor('SigOpEtq', 'CBars', 	'crSigOpEtq', '', '', lcQryEti)
			.AddCursor('SigEtPos', 'CidChaves', 'crSigEtPos', '', '')
			.AddCursor('SigMvCab', 'cIdChaves', 'crSigMvCab', '', '')
			.AddCursor('SigMvItn', 'cIdChaves', 'crSigMvItn', '', '')
		EndWith

		ThisForm.poDataMgr.CursorQuery('SigCdOpd', 'crSigCdOpd', 'Dopps', ThisForm._Dopp)
		If Eof('crSigCdOpd')
			ThisForm.Release
			Return .f.
		EndIf

		lcCpo = 'DoppPads, TpPesas, CuniPPads, Ouros, TransfRes, GrupoRecs, GrupoPags, ContaRecs, ContaPags, TransfEncs, mascnums'
		ThisForm.poDataMgr.CursorQuery('SigCdPam', 'crSigCdPam', .f., .f., lcCpo)

		ThisForm.poDataMgr.CursorQuery('SigCdPac', 'crSigCdPac')
		ThisForm.SigKey = CrSigCdPac.SigKeys

		ThisForm.poDataMgr.CursorQuery('SigCdGrp', 'crSigCdGrp', .f., .f., 'CGrus, CEstoqs')
		Select crSigCdGrp
		Index On CGrus Tag CGrus
		
		ThisForm.poDataMgr.CursorQuery('SigCdUni', 'crSigCdUni', .f., .f., 'CUnis, Fators')
		Select crSigCdUni
		Index On CUnis Tag CUnis
		
		ThisForm.poDataMgr.CursorQuery('SigCdGcr', 'TempGccr', .f., .f., 'Codigos, Descrs, GerBals')
		Select TempGccr
		Index On Codigos Tag BalCodigo
		Index On Codigos Tag Codigos
		Set Order To Codigos

		=DoDefault('P' + Padl(crSigCdOpd.nDopps, 8, '0'))

		ThisForm.Titulo = Proper(Alltrim(crSigCdOpd.TNivels)) + ' '
		ThisForm.Artigo = Lower(Alltrim(crSigCdOpd.Artigos))

		ThisForm.Pagina.Dados.GradeOperação.Column1.Header1.Caption = ThisForm.Titulo

		ThisForm.CriaCursores

		Go Bottom In crSigCdNec

		If !thisform.lAutomatico
			ThisForm.Pagina.Lista.Grade.Column1.SetFocus()
		Endif
	Else
		=Messagebox('Impossivel Efetuar Conexão com Servidor de Banco de Dados...', 0+48, 'CreateObject')
		Return .f.
	EndIf
Endwith

*** Tiago - 08/10/2013 - Finaliza automaticamente as OPs que foram passadas por parâmetro
If Type('laNop',1) = [A]
	llOk = .f.

	thisform.cEmps = laNop[1,2]	
	ThisForm.Pagina.Lista.Grupo_Op.Click( 1 )		
	
	For lnCnt = 1 To Alen(laNop,1)
		ThisForm._OldCodigo = 0
		thisform.pagina.dados.gradeOperação.column1.text1.Value = laNop[lnCnt,1]
		thisform.pagina.dados.gradeOperação.column1.text1.Valid()
	Next

	Select TmpOperacao
	Go top
	Sum 1 To _TotEnv For NOT Empty(codigos)

	IF _TotEnv > 0
		thisform.pagina.dados.grupo_Salva.salva.Click()
	Else
		thisform.pagina.dados.grupo_Salva.cancelar.Click()
	endif

	thisform.Release()
	Return .f.
Else
	Dimension laNop[1] && Linha apenas para não dar erro na geração do exe	
Endif
***
* Fim **************************************************************************************

ENDPROC

* PROCEDURE msv_excluir

PROCEDURE msv_excluir
Private pNot, pEmp, pDop, pNum
Local _ChaveDel, lcGrupos, lcContas, ldDatas, lcMoedas, lcSGrupos, lcSContas, lcSMoedas

Select TmpNens
Go Top
_ChaveDel1 = Emps + Dopps + Str(Numps, 10)
_ChaveDel2 = Emps + Dopps + Str(Numps, 06)

pEmp = Emps
pDop = Dopps
pNum = Numps
pDat = Datetime()

* Excluindo do SigCdNei
lcQuery = [Delete From SigCdNei ] + ;
		   [Where EmpDNps = '] + _ChaveDel1 + [']

If (ThisForm.poDataMgr.SqlExecute(lcQuery, '') < 1)
	=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (mSv_Excluir - SigCdNei)')
	Return .f.
EndIf

* Excluindo do SigMvHst
lcCpo = 'Emps, Grupos, Estos, CPros, Datas, CodCors, CodTams, cIdChaves'
ThisForm.poDataMgr.CursorQuery('SigMvHst', 'LocalHis', 'EmpDopNums', _ChaveDel2, lcCpo)
Select LocalHis
Scan
	=fRecalculaP(LocalHis.Emps, LocalHis.Grupos, LocalHis.Estos, LocalHis.CPros, LocalHis.Datas, LocalHis.CodCors, LocalHis.CodTams, ThisForm.poDataMgr)
	=fRecalculaC(LocalHis.Emps, LocalHis.CPros, LocalHis.Datas, ThisForm.poDataMgr)

	lcQuery = [Delete From SigMvHst ] + ;
			   [Where cIdChaves = '] + LocalHis.cIdChaves + [']

	If (ThisForm.poDataMgr.SqlExecute(lcQuery, '') < 1)
		=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (mSv_Excluir - SigMvHst (3))')
		Return .f.
	EndIf
EndScan

* Excluindo do SigMvCcr
lcCpo = 'cIdChaves, Grupos, Contas, Datas, Moedas'
ThisForm.poDataMgr.CursorQuery('SigMvCcr', 'LocalMccr', 'EmpDopNums', _ChaveDel2, lcCpo)
Select LocalMccr
Scan
	=fRecalculaS( LocalMccr.Grupos,  LocalMccr.Contas,  LocalMccr.Datas, LocalMccr.Moedas, ThisForm.Podatamgr )
	=fRecalculaS( LocalMccr.SGrupos, LocalMccr.SContas, LocalMccr.Datas, LocalMccr.SMoedas, ThisForm.Podatamgr )

	lcQuery = [Delete From SigMvCcr ] + ;
			   [Where cIdChaves = '] + LocalMccr.cIdChaves + [']

	If (ThisForm.poDataMgr.SqlExecute(lcQuery, '') < 1)
		=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (mSv_Excluir - SigMvCcr)')
		Return .f.
	EndIf

		Scan For Rateio = 2 And NEnvs = _Nenv And Not FlagIncs

		Scan For Rateio = 2 And NEnvs = _Nenv And Not FlagIncs

					Qtds  with Qtds + _Pesagem

			ThisForm.Pagina.Dados.Operacao.SetFocus

	EndCase
EndIf

		Delete All For Nenvs = _nOperacao

		Delete All For Nenvs = _nOperacao

		Delete All For Nenvs = _nOperacao

		Delete All For Nenvs = _nOperacao

		EndScan

			Insert Into TmpOperacao (Codigos) Values (0)

Ok = .t.

			Select LocalCompo
			Scan
				ThisForm.poDataMgr.CursorQuery('SigCdPro', 'LocalPro', 'CPros', LocalCompo.Mats, 'CUnis, CGrus, DPros, Varias')
				=Seek(LocalPro.CUnis, 'crSigCdUni', 'CUnis')

				If (LocalCompo.Mats = _MatP) And (LocalMFas.Pesos <> 0)
					_QtAnt = LocalMFas.Pesos
				Else
					_QtAnt = LocalMFas.Qtds * LocalCompo.Qtds
				EndIf

				If (crSigCdUni.Fators = 0)
					_PsAnt = LocalMFas.Qtds * LocalCompo.Qtds * LocalMFas.Pesos
				Else
					_PsAnt = LocalMFas.Qtds * LocalCompo.Qtds * crSigCdUni.Fators
				EndIf

				Insert Into xNensi (Emps, Dopps, Numps, Nops, NEnvs, CMats, CDescs, CUnis, CGrus, QtFaseAnt, ;
									PsFaseAnt, Visivel, Rateio, FlagIncs, Cats, Qtds, Pesos) ;
							Values (TmpNens.Emps, TmpNens.Dopps, TmpNens.Numps, xMFas.Nops, _nOperacao, ;
									LocalCompo.Mats, LocalPro.Dpros, LocalPro.CUnis, LocalPro.CGrus, _QtAnt, ;
									_PsAnt, .f., Iif(LocalPro.Varias=1, 2, 3), .t., LocalCompo.Cats, _QtAnt, _PsAnt)

				If Not Seek(_Operacao + LocalCompo.Mats, 'xNensi', 'NEnvCMat')
					Insert Into xNensi (Nops, Nenvs, CMats, CDescs, CUnis, CGrus, Visivel, Rateio, FlagIncs, Cats) ;
								Values (_nOperacao, _nOperacao, LocalCompo.Mats, LocalPro.DPros, LocalPro.CUnis, ;
										LocalPro.CGrus, .t., 1, .t., LocalCompo.Cats)
				EndIf

				Replace QtFaseAnt With xNensi.QtFaseAnt + _QtAnt, ;
						Qtds 	  With xNensi.Qtds + _QtAnt In xNensi

				If (crSigCdUni.Fators = 0)
					Replace PsFaseAnt With xNensi.PsFaseAnt + _PsAnt, ;
							Pesos 	  With 0 In xNensi
				Else
					Replace PsFaseAnt With xNensi.PsFaseAnt + _PsAnt, ;
							Pesos 	  With xNensi.Pesos + _PsAnt In xNensi
				EndIf
			EndScan
			Select TmpNens
		EndScan

		With ThisForm.pagina.dados
			Select TmpNens
			ThisForm.poDataMgr.CursorQuery('SigCdCli', 'LocalCli', 'IClis', TmpNens.ContaOs, 'RClis')
			If Not Eof('LocalCli')
				.Origem.get_dconta.Value = LocalCli.RClis
			Else
				.Origem.get_dconta.Value = ''
			EndIf

			ThisForm.poDataMgr.CursorQuery('SigCdCli', 'LocalCli', 'IClis', TmpNens.ContaDs, 'RClis')
			If Not Eof('LocalCli')
				.Destino.get_dconta.Value = LocalCli.RClis
			Else
				.Destino.get_dconta.Value = ''
			EndIf

			ThisForm.poDataMgr.CursorQuery('SigCdCli', 'LocalCli', 'IClis', TmpNens.Vends, 'RClis')
			If Not Eof('LocalCli')
				.Responsavel.get_dconta.Value = LocalCli.RClis
			Else
				.Responsavel.get_dconta.Value = ''
			EndIf
			
			.Refresh
		EndWith

		Select TmpOperacao
		Skip
		If Eof()
			Append Blank
			Keyboard '{DNARROW}'
		EndIf
		ThisForm.Pagina.Dados.Refresh
		Return .t.
	Else
		Return 0
	EndIf
	Select TmpNens
	Return 0
EndIf
Select xNensi
Set Order To

Select TmpNens
ThisForm.Pagina.Dados.Refresh
* Fim *********************************************************************************

ENDPROC
