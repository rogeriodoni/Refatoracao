ENDPROC

PROCEDURE When
Select CrInvH
Go Bottom
ThisForm.Grade.Refresh
If ThisForm.ParentForm.TipoLeitora = 'S'
	If Empty(ThisForm.ParentForm.LeituraSerial)
		If !ThisForm.ParentForm.ReadComm.Ole_Comm.PortOpen And !ThisForm.ParentForm.ReadComm.PortaError
			ThisForm.ParentForm.ReadComm.Ole_Comm.PortOpen = .T.
		EndIf
	EndIf
Else
	Set Confirm On
EndIf
ENDPROC

------------------------------------------------------------
* PROCEDURE Valid
------------------------------------------------------------
PROCEDURE Valid
If ThisForm.ParentForm.TipoLeitora = 'S'
	If Empty(ThisForm.ParentForm.LeituraSerial)
		If !ThisForm.ParentForm.ReadComm.Ole_Comm.PortOpen And !ThisForm.ParentForm.ReadComm.PortaError
			ThisForm.ParentForm.ReadComm.Ole_Comm.PortOpen = .T.
		EndIf
	Else
		This.Value = ThisForm.ParentForm.LeituraSerial
		This.Refresh
		ThisForm.ParentForm.LeituraSerial = ''
	EndIf
EndIf

If !Empty(This.Value)
	lcConsulta = Sys(2023) + '\TempCon.jpg'

	ThisForm.ExcluiItem.Visible = ThisForm.Excluir
	ThisForm.Shape2.Visible     = ThisForm.Excluir
	ThisForm.Shape3.Visible     = !ThisForm.Excluir

	lStrQuery = [Select CBars,CPros,Pesos ]+;
				[From SigOpEtq ]+;
				[Where CBars = ]+This.Value+[ ]
	If ThisForm.ParentForm.poDataMgr.SqlExecute(lStrQuery,'TmpBar') < 1
		MessageBox('Favor reinicializar o processo !!!',16,'Falha na Conexão (TmpBar)')
		Return(.F.)
	EndIf
	Select TmpBar
	Go Top
	If !Eof()
		ThisForm.ParentForm.poDataMgr.CursorQuery('SigCdPro','TmpPro','CPros',TmpBar.CPros,'CPros,DPros,Linhas,FigJpgs')
		ThisForm.ParentForm.poDataMgr.CursorQuery('SigCdLin','TmpLinha','Linhas',TmpPro.Linhas,'Descs')
		ThisForm.Get_Produto.Value   = TmpPro.CPros
		ThisForm.Get_Descricao.Value = TmpPro.DPros
		ThisForm.Get_Peso.Value      = TmpBar.Pesos
		ThisForm.Get_Linha.Value     = TmpLinha.Descs
	
		With ThisForm
			.ImgJpg.Visible = .F.
			.ImgJpg.Picture = ''
			Clear Resources
			If !Empty(TmpPro.FigJpgs) And !IsNull(TmpPro.FigJpgs)
				* Figura no Campo Memo
				lcFoto=Strconv(Strtran(Strtran(Strtran(TmpPro.FigJpgs,[data:image/png;base64,],[]),[data:image/jpeg;base64,],[]),[data:image/jpg;base64,],[]),14)
				If (StrToFile(lcFoto,lcConsulta) > 0)
					.ImgJpg.Visible = .T.
					.ImgJpg.Picture = (lcConsulta)
				EndIf
				If File(lcConsulta)
					Delete File (lcConsulta)
				EndIf
			Else
				.ImgJpg.Visible = .F.
				.ImgJpg.Picture = ''
			EndIf
			.Refresh
		EndWith
	EndIf

	ThisForm.Grade.RecordSource = ''
	
	If !Seek(This.Value,'CrInvH','CBars')
		If Seek(This.Value,'Historico','CBars')
			Select Historico
			Scatter MemVar Memo
			Insert Into CrInvH From MemVar
		Else
			lStrQuery = [Select * From SigIvTrH ]+;
						[Where Emps = ']+CrSigCdBal.Emps+[' And Codigos = ]+Alltrim(Str(CrSigCdBal.Codigos,6))+[ ]+;
							[And CBars = ]+This.Value+[ ]
			If ThisForm.ParentForm.poDataMgr.SqlExecute(lStrQuery,'CrInvH') < 1
				MessageBox('Favor reinicializar o processo !!!',16,'Falha na Conexão (CrInvH)')
				Return(.F.)
			EndIf
			Select CrInvH
			Index On CBars Tag CBars
			Index On CPros Tag CPros
			Go Top
		EndIf
	EndIf
	Select CrInvH
	Update CrInvH Set Obs = Nvl(Obs,'')
	ThisForm.AcertaGrade()
	Select CrInvH
	Go Top
EndIf

If LastKey() # 27 And !ThisForm.Excluir
	ThisForm.Get_Barra.Value = ''
	ThisForm.Refresh
	If ThisForm.ParentForm.TipoLeitora = 'S'
		If Empty(ThisForm.ParentForm.LeituraSerial)
			If !ThisForm.ParentForm.ReadComm.Ole_Comm.PortOpen And !ThisForm.ParentForm.ReadComm.PortaError
				ThisForm.ParentForm.ReadComm.Ole_Comm.PortOpen = .T.
			EndIf
		Else
			This.Value = ThisForm.ParentForm.LeituraSerial
			This.Refresh
			ThisForm.ParentForm.LeituraSerial = ''
		EndIf
	Endif
	Return 0
EndIf
Return(.T.)
ENDPROC

PROCEDURE Valid
If !Empty(This.Value)
	m.loLista = CreateObject('fwBuscaExt',ThisForm.ParentForm.poDataMgr.pnIdConn,'SigCdPro','CrProds','CPros',This.Value,'Produtos',1000)
	If !(m.loLista.plAchouRegistro)
		m.loLista.mAddColuna('CPros','','Código')
		m.loLista.mAddColuna('DPros','','Descrição')
		m.loLista.Show()
	EndIf
	If LastKey() <> 27 And Used('CrProds')
		This.Value                   = Evaluate('CrProds.CPros')
		ThisForm.Get_Descricao.Value = Evaluate('CrProds.DPros') 
	Else
		This.Value                   = ''
		ThisForm.Get_Descricao.Value = ''
	EndIf

	lcConsulta = Sys(2023) + '\TempCon.jpg'

	ThisForm.ExcluiItem.Visible = ThisForm.Excluir
	ThisForm.Shape2.Visible     = ThisForm.Excluir
	ThisForm.Shape3.Visible     = !ThisForm.Excluir

	lStrQuery = [Select a.CPros,a.Linhas,a.FigJpgs,b.Descs ]+;
				[From SigCdPro a ]+;
				[Left Join SigCdLin b On b.Linhas = a.Linhas ]+;
				[Where CPros = ']+This.Value+[' ]
	If ThisForm.ParentForm.poDataMgr.SqlExecute(lStrQuery,'TmpPro') < 1
		MessageBox('Favor reinicializar o processo !!!',16,'Falha na Conexão (TmpPro)')
		Return(.F.)
	EndIf
	Select TmpPro
	Go Top
	If !Eof()
		ThisForm.Get_Produto.Value   = TmpPro.CPros
		ThisForm.Get_Peso.Value      = 0
		ThisForm.Get_Linha.Value     = Nvl(TmpPro.Descs,'')
	
		With ThisForm
			.ImgJpg.Visible = .F.
			.ImgJpg.Picture = ''
			Clear Resources
			If !Empty(TmpPro.FigJpgs) And !IsNull(TmpPro.FigJpgs)
				* Figura no Campo Memo
				lcFoto=Strconv(Strtran(Strtran(Strtran(TmpPro.FigJpgs,[data:image/png;base64,],[]),[data:image/jpeg;base64,],[]),[data:image/jpg;base64,],[]),14)	
				If (StrToFile(lcFoto,lcConsulta) > 0)
					.ImgJpg.Visible = .T.
					.ImgJpg.Picture = (lcConsulta)
				EndIf
				If File(lcConsulta)
					Delete File (lcConsulta)
				EndIf
			Else
				.ImgJpg.Visible = .F.
				.ImgJpg.Picture = ''
			EndIf
			.Refresh
		EndWith
	EndIf
	ThisForm.Grade.RecordSource = ''

	If !Seek(This.Value,'CrInvH','CPros')
		Select * From Historico Where CPros = This.Value Into Cursor TmpHistorico
		Select TmpHistorico
		Go Top
		If !Eof()
			Scan
				Scatter MemVar Memo
				Insert Into CrInvH From MemVar
				Select TmpHistorico
			EndScan
		Else
			lStrQuery = [Select * From SigIvTrH ]+;
						[Where Emps = ']+CrSigCdBal.Emps+[' And Codigos = ]+Alltrim(Str(CrSigCdBal.Codigos,6))+[ ]+;
							[And CPros = ']+This.Value+[' ]
			If ThisForm.ParentForm.poDataMgr.SqlExecute(lStrQuery,'CrInvH') < 1
				MessageBox('Favor reinicializar o processo !!!',16,'Falha na Conexão (CrInvH)')
				Return(.F.)
			EndIf
			Select CrInvH
			Index On CBars Tag CBars
			Index On CPros Tag CPros
			Go Top
		EndIf
	EndIf
	Select CrInvH
	Update CrInvH Set Obs = Nvl(Obs,'')
	ThisForm.AcertaGrade()
	Select CrInvH
	Go Top
EndIf
Return(.T.)
ENDPROC

PROCEDURE Click
Select CrInvH
DO Case
	Case Empty(ThisForm.Get_Barra.Value) And !Empty(ThisForm.Get_Produto.Value)
		Select Historico
		If !Seek(ThisForm.Get_Produto.Value,'Historico','CPros')
			Select CrInvH
			Scatter MemVar Memo
			Insert Into Historico From MemVar
		EndIf
		Select Historico
		Go Top
		Delete From Historico Where CPros = ThisForm.Get_Produto.Value
		Select Historico
		Go Top
		Select CrInvH
		Delete From CrInvH
	Case !Empty(ThisForm.Get_Barra.Value)
		If !Seek(CrInvH.CBars,'Historico','CBars')
			Select CrInvH
			Scatter MemVar Memo
			Insert Into Historico From MemVar
		EndIf
		Select Historico
		lnRecno = Recno('Historico')
		Go Top
		Go lnRecno
		Delete In Historico
		Select CrInvH
		Delete In CrInvH
	Otherwise
		ThisForm.Grade.SetFocus
		ThisForm.Grade.Refresh
		ThisForm.Refresh
		ThisForm.Get_Barra.Value = ''
		ThisForm.Get_Barra.SetFocus
		Return
EndCase

Select CrInvH
Go Bottom
ThisForm.Grade.SetFocus
ThisForm.Grade.Refresh
ThisForm.Refresh
ThisForm.Get_Barra.Value = ''
If Eof('CrInvH')
	ThisForm.ExcluiItem.Visible = .F.
	ThisForm.Shape2.Visible     = .F.
	ThisForm.Shape3.Visible     = .T.
	ThisForm.Get_Barra.SetFocus
EndIf
Return
ENDPROC
