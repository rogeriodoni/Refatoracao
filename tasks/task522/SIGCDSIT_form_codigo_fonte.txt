* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
With ThisForm
	.poDataMgr = CreateObject([fSqlConector], .Name)

	If (.poDataMgr.pnIdconn > 0)
		With .poDataMgr
			.AddCursor([SigCdSit], [cIdChaves], [crSigCdSit], [], ThisForm.Pagina.Lista.Grade)
			.pfSqlTabela(1).pColuna([Cods],  [], [], [Código],    080, .t.)
			.pfSqlTabela(1).pColuna([Descs], [], [], [Descrição], 300, .t.)
		EndWith

		=DoDefault()

		.Pagina.Lista.Grade.Column1.SetFocus
	Else
		=Messagebox([Impossível Efetuar Conexão Com o Servidor de Banco de Dados...], 0+48, [CreateObject])
		Return .f.
	EndIf
EndWith
* Fim **************************************************************************************
ENDPROC

------------------------------------------------------------
* PROCEDURE msv_procurar
------------------------------------------------------------
PROCEDURE msv_procurar
Local lcBusca

If Not Empty(ThisForm.Pagina.Dados.getCods.Value)
	ThisForm.poDataMgr.LocateCursor([crSigCdSit], [Cods], ThisForm.Pagina.Dados.getCods.Value)
EndIf
* Fim **************************************************************************************
ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Lista.Grupo_op.Click
------------------------------------------------------------
PROCEDURE Pagina.Lista.Grupo_op.Click
Parameters lnOpcao

If Not DoDefault(lnOpcao)
	ThisForm.mAtivaPagina1
	Return .f.
EndIf

Do Case
	Case InList(ThisForm.pcEscolha, [INSERIR], [PROCURAR])
		 ThisForm.Pagina.Dados.getCods.SetFocus
		 
	Case InList(ThisForm.pcEscolha, [ALTERAR])
		 ThisForm.Pagina.Dados.getDescs.SetFocus

	Otherwise
		 ThisForm.Pagina.Dados.Grupo_Salva.Salva.SetFocus
EndCase
* Fim **************************************************************************************
ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
If InList(ThisForm.pcEscolha, [INSERIR], [ALTERAR])
	If Empty(ThisForm.Pagina.Dados.getCods.Value)
		=Messagebox([Código Inválido!!!], 48, [Atenção!!!])
		ThisForm.Pagina.Dados.getCods.SetFocus()
		Return .f.
	EndIf

	If Empty(ThisForm.Pagina.Dados.getDescs.Value)
		=Messagebox([Descrição Inválida!!!], 48, [Atenção!!!])
		ThisForm.Pagina.Dados.getDescs.SetFocus()
		Return .f.
	EndIf
	
	If (ThisForm.pcEscolha = [INSERIR])
		lcQuery = [Select Cods ] + ;
				    [From SigCdSit ] + ;
				   [Where Cods = '] + Padr(ThisForm.Pagina.Dados.getCods.Value, 3) + [']

		If (ThisForm.poDataMgr.SqlExecute(lcQuery, [LocalBus]) < 1)
			=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (LocalBus 1)])
			Return .f.
		EndIf
		Go Top In LocalBus
		If Not Eof([LocalBus])
			=Messagebox([Código Já Cadastrado!!!], 48, [Atenção!!!])
			ThisForm.Pagina.Dados.getCods.SetFocus()
			Return .f.
		EndIf
	EndIf
EndIf

If (ThisForm.pcEscolha = [INSERIR])
	Replace cIdChaves With fUniqueIds() in crSigCdSit 
EndIf 


=DoDefault()

ThisForm.mAtivaPagina1
* Fim **************************************************************************************
ENDPROC
