* PROCEDURE Init

PROCEDURE Init
LParameters lnTipoLeitura, llOnLine

Wait Window 'Aguarde! Totalizando Locais...' NoWait
ThisForm.cntSombra.lblSombra.Caption = ThisForm.Caption
ThisForm.cntSombra.lblTitulo.Caption = ThisForm.Caption

Select TmpLocal
Zap

If Not llOnLine

	lnTetQLoad  = RecCount('Etiqueta')
	lnTetQLoad  = Iif(lnTetQLoad = 0,1,lnTetQLoad)
	lnTetQReadL = 0
	lnTetQReadR = 0

	Select Historico
	Set Order To
	Scan
		If !Seek(Historico.Localis,'TmpLocal')
			Insert Into TmpLocal(Localis) Values (Historico.Localis)
		EndIf
		If Historico.Tipos = 1
			Replace TmpLocal.Leitura With TmpLocal.Leitura + Historico.Qtds
		Else
			If Historico.Tipos = 2
				Replace TmpLocal.ReLeitura With TmpLocal.ReLeitura + Historico.Qtds
			EndIf
		EndIf
	EndScan

	Select TmpLocal
	Scan
		If TmpLocal.ReLeitura > 0
			lnTetQReadL = lnTetQReadL + TmpLocal.ReLeitura
		Else
			lnTetQReadL = lnTetQReadL + TmpLocal.Leitura
		EndIf
	EndScan

	Select Historico
	Go Bottom

Else

	lnTetQLoad  = RecCount('csEtiqueta')
	lnTetQLoad  = Iif(lnTetQLoad = 0,1,lnTetQLoad)
	lnTetQReadL = 0
	lnTetQReadR = 0

	Select csHistorico
	Set Order To
	Scan
		If !Seek(csHistorico.Localis,'TmpLocal')
			Insert Into TmpLocal(Localis) Values (csHistorico.Localis)
		EndIf
		If csHistorico.Tipos = 1
			Replace TmpLocal.Leitura With TmpLocal.Leitura + csHistorico.Qtds
		Else
			If csHistorico.Tipos = 2
				Replace TmpLocal.ReLeitura With TmpLocal.ReLeitura + csHistorico.Qtds
			EndIf
		EndIf
	EndScan

	Select TmpLocal
	Scan
		If TmpLocal.ReLeitura > 0
			lnTetQReadL = lnTetQReadL + TmpLocal.ReLeitura
		Else
			lnTetQReadL = lnTetQReadL + TmpLocal.Leitura
		EndIf
	EndScan

EndIf

Select TmpLocal
Go Top

lnPercLeit  = (lnTetQReadL / lnTetQLoad * 100)
lnPercRLeit = (lnTetQReadR / lnTetQLoad * 100)

If CrSigCdPam.PerLocals
	ThisForm.lb_TotLIdo.Caption    = Str(lnTetQReadL,5) 
	ThisForm.lb_TotEstoque.Caption = Str(lnTetQLoad,5)
	ThisForm.Leitura.Caption = Transform(lnPercLeit,'@Z 9999.99%')
EndIf

Wait Clear

ThisForm.Grade.Refresh
ThisForm.Grade.SetFocus
ENDPROC
