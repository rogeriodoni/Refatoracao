PROCEDURE Valid
If Not Empty(This.Value)
	Select crChaves	
*!*		=TableUpdate(.T.)
	Select ChaveNota, Count(*) as qt from crChaves where ChaveNota = This.Value Group by ChaveNota Into Cursor csRegDup ReadWrite
	
	Select csRegDup
	If Reccount([csRegDup])>0 And csRegDup.qt > 1
		MessageBox([Registro já Informado.],0+48,[Aviso])
		Return .F.
	EndIf

*!*		Locate For (Empty(ChaveNota))

*!*		EndIf
