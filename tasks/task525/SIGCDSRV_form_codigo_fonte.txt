------------------------------------------------------------
* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
Lparameters lcTipo

ThisForm.pcTipo = Iif(Type([lcTipo]) <> [C], [S], lcTipo)
ThisForm.Caption= [Serviços de ] + Iif(ThisForm.pcTipo = [S], [Saídas], [Entradas])
ThisForm.Name  = [SIGCDSRV]+Iif(ThisForm.pcTipo <> [S], [E],[])

pcCodServ = []

With ThisForm
	.poDataMgr = CreateObject([fSqlConector], This.Name)	
	
	m.lcQuery = [Select * from SigCdSrv where cTipos = ']+ThisForm.pcTipo+[']
	If (.poDataMgr.pnIdconn > 0)
		With .poDataMgr
			.AddCursor([SigCdSrv], [cIdChaves], [crSigCdSrv], [], ThisForm.Pagina.Lista.Grade,m.lcQuery)
			.pfSqlTabela(1).pColuna([CodServs], [], [], [Código], 080, .t.)
			.pfSqlTabela(1).pColuna([DescServs], [], [], [Descrição], 480, .t.)
		EndWith

		lcQuery = [Select a.*, b.dPros from SigSeRvp a inner Join SigCdPro b on b.cpros = a.cpros where a.CodServs = ?pcCodServ]
		.poDataMgr.AddCursor([SigSeRvp],[cIdChaves],[crSigSeRvp],'','',lcQuery)

		lcQuery = [Select Codigos, Descrs from SigOpOco]
		If .poDataMgr.SQLExecute(lcQuery,[crSigOpOco]) < 1
			=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (crSigOpOco)])
			Return 0
		EndIf
		Select crSigOpOco
		Index on Codigos Tag Codigos
		Index on Descrs Tag Descrs
		
		ThisForm.Pagina.Dados.pgDados.pgRetencao.cntRPA.Visible = (ThisForm.pcTipo = [E])

		=DoDefault()

		.Pagina.Lista.Grade.Column1.SetFocus
	Else
		=Messagebox([Impossível Efetuar Conexão Com o Servidor de Banco de Dados...], 0+48, [CreateObject])
		Return .f.
	EndIf
EndWith
* Fim **************************************************************************************
ENDPROC

* PROCEDURE pagina.dados.grupo_salva.salva.Click
------------------------------------------------------------
PROCEDURE pagina.dados.grupo_salva.salva.Click
Local loDad

loDad = ThisForm.Pagina.Dados.pgDados.pgImpostos  

If InList(ThisForm.pcEscolha, 'INSERIR', 'ALTERAR')
	If Empty(loDad.getCodigos.Value)
		=MessageBox('Obrigatório Informar o Código!!!', 48, '')
		loDad.getCodigos.SetFocus
		Return 0
	EndIf

	If Empty(loDad.getDescricao.Value)
		=MessageBox('Obrigatório Informar a Descrição!!!', 48, '')
		loDad.getDescricao.SetFocus
		Return 0
	EndIf
		
	If (Thisform.pcEscolha = 'INSERIR' )
		lcQuery = [Select * ] + ;
				    [From SigCdSrv ] + ;
				   [Where CodServs = '] + Padr(loDad.getCodigos.Value, 10) + ['] + ;
				   [ And cTipos = ']+ThisForm.pcTipo+[']

		If (ThisForm.poDataMgr.SqlExecute(lcQuery, [crBusca]) < 1)
			=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (crBusca)])
			Return 0
		EndIf
		Select crBusca
		Go Top In crBusca
		If Not Eof([crBusca])
			=MessageBox([Código Já Cadastrado!!!], 48, [Atenção!!!])
			loDad.getCodigos.SetFocus
			Return 0
		EndIf
		Replace cTipos with ThisForm.pcTipo in crSigCdSrv
	EndIf	
EndIf

If (Thisform.pcEscolha = [EXCLUIR])
	lcQuery = [Delete from SigSeRvp where CodServs = ']+crSigCdSrv.CodServs+[']
	ThisForm.poDataMgr.SQLExecute(lcQuery)
*!*		Delete In crSigCdSrv	
EndIf 

Select crSigCdSrv

=DoDefault()

ThisForm.mAtivaPagina1
* Fim ************************************************************************************** 
ENDPROC
