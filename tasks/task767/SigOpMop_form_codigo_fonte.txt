PROCEDURE mctrlbotoes
LParameters plHabilita

Local lnI

If Type('plHabilita') # 'L'
	plHabilita = .F.
Endif

With Thisform.ParentForm.Pagina.Lista
	For lnI=1 to .Grupo_Op.ButtonCount
		If .Grupo_Op.Buttons(lnI).BaseClass = 'Commandbutton' And .Grupo_Op.Buttons(lnI).Visible
			.Grupo_Op.Buttons(lnI).Enabled = plHabilita
		EndIf
	EndFor
	.cmdFPOper.Enabled = plHabilita
	.cmdMultiSubN.Enabled = plHabilita
	.Grupo_Saida.Sair.Enabled = plHabilita
EndWith
ENDPROC

PROCEDURE Init
LParameters poForm, pnDataSes, pcEscolha

Local lcOperacao

Set DataSession to pnDataSes

Thisform.ParentForm = poForm
Thisform.PcEscolha  = pcEscolha
Thisform.ParentForm.Enabled  = .F.
lcOperacao = crSigCdOpe.Dopes

With Thisform
	.Caption = 'Geração de Multiplas Operações ' + AllTrim( Proper( crSigCdOpe.Dopes ) )
	.Top     = Thisform.ParentForm.Top + Thisform.ParentForm.Pagina.Lista.Grade.Top + 50
	.Left    = Thisform.ParentForm.Left + ;
			   Thisform.ParentForm.Pagina.Lista.Grade.Left + ;
			   Thisform.ParentForm.Pagina.Lista.Grade.Columns(1).Width + ;
			   Thisform.ParentForm.Pagina.Lista.Grade.Columns(3).Width + 16

	.cmdInserir.Visible = fChecaAcesso('SIGOPMOP','INSERIR') And ;
						  Iif( Type( 'Thisform.PcEscolha' ) # 'C', .T., ;
						  InList( Thisform.PcEscolha, 'INSERIR', 'ALTERAR' ) )
	.cmdInserir.Enabled = fChecaAcesso('SIGOPMOP','INSERIR')	And ;
						  Iif( Type( 'Thisform.PcEscolha' ) # 'C', .T., ;
						  InList( Thisform.PcEscolha, 'INSERIR', 'ALTERAR' ) )
	.cmdExcluir.Visible = fChecaAcesso('SIGOPMOP','EXCLUIR') And ;
						  Iif( Type( 'Thisform.PcEscolha' ) # 'C', .T., ;
						  InList( Thisform.PcEscolha, 'INSERIR', 'ALTERAR' ) )
	.cmdExcluir.Enabled = fChecaAcesso('SIGOPMOP','EXCLUIR')	And ;
						  Iif( Type( 'Thisform.PcEscolha' ) # 'C', .T., ;
						  InList( Thisform.PcEscolha, 'INSERIR', 'ALTERAR' ) )
	.shape1.Left = IIF(.cmdInserir.visible, .cmdInserir.Left - 5, IIF(.cmdExcluir.visible , .cmdExcluir.left - 5, .cmdSair.Left - 5))
	.shape1.width = IIF(.cmdInserir.visible, 3, IIF(.cmdExcluir.visible, 2, 1)) * .cmdsair.width + 16
EndWith

ThisForm.cntSombra.lblSombra.Caption = ThisForm.Caption
ThisForm.cntSombra.lblTitulo.Caption = ThisForm.Caption

Thisform.mCtrlBotoes(.F.)

lcSql = [Select Dopes, GerOpes, OpeInatvs From SigOpCdc ]
SigCdOpe.Podatamgr.Sqlexecute(lcSql,'LocalDcOpe')
Select LocalDcOpe
Index On Dopes Tag dopes
=Seek(CrSigCdOpe.Dopes)

WITH SigCdOpe
	With .poDatamgr
        .ReQuery('crSigCdMop')
		SELECT crSigCdMop
		GO top
	EndWith
Endwith	

Select crSigCdMop
Goto Top	
With Thisform.GradeSubN
	.RecordSource          = 'crSigCdMop'
	.Column1.ControlSource = 'crSigCdMop.cNivels'
	.Column2.ControlSource = 'crSigCdMop.nDopes'
	.Column3.ControlSource = 'crSigCdMop.nCors'
	.Column4.ControlSource = 'crSigCdMop.Padraos'
	.Column5.ControlSource = 'crSigCdMop.Ordems'
	.Column6.ControlSource = 'crSigCdMop.Mercs'
	.Column7.ControlSource = 'crSigCdMop.EmpDs'

		.Column3.ReadOnly  = .t.
	Else
		.SetAll('DynamicForeColor','iif(CrSigCdMop.ncors>0,nCors,0)','Column')
	Endif
	.Refresh
EndWith
ENDPROC

Local lcOperacao, lnRecNo, loBrowse

lcDopes = crSigCdOpe.Dopes

If Not IsEmpty(This.Value)
	xBrox = CreateObject('fwBuscaExt',SigCdOpe.poDatamgr.pnIdconn,'SigCdOpe','cpSigCdOpe','Dopes',This.Value,'Cadastro de Operações',.t.,'Dopes')
	If  Not xBrox.plAchouRegistro
		xBrox.mAddColuna('Dopes', '', 'Operação', .t.)	
		xBrox.mAddColuna('nDopes', '9999', ' # ',.f.)
		xBrox.Show()
	EndIf 
	=Seek(cpSigCdOpe.Dopes,'LocalDcOpe','Dopes')
	If LocalDcOpe.OpeInatvs = 1
		MessageBox('Esta Operação está Inativa!!!!',0+16,'Atenção...')
		This.Value = ' '
		Return .f.
	Endif
	This.Value = cpSigCdOpe.Dopes
	Replace nDopes With CpSigCdOpe.nDopes in CrSigCdMop
	This.Refresh
EndIf
Select CrSigCdMop
Return .t.

ENDPROC

* OBJETO: SIGOPMOP.cmdSair
* Classe: fwbtng (Base: commandbutton)
* Registro: 24
================================================================================

------------------------------------------------------------
* PROCEDURE Click
------------------------------------------------------------
PROCEDURE Click
If Thisform.cmdInserir.Visible Or Thisform.cmdExcluir.Visible
	If ( Thisform.lHouveExcl Or Thisform.lHouveIns )
		
		Select CrSigCdMop
		Go Bott
		Go Top
		
		Select cNivels as Dopes,Mercs,EmpDs,sum(1) as TT From CrSigCdMop GROUP by 1,2,3 Into cursor Totais
		Select Totais
		Scan
			If TT > 1
				MessageBox('Operação '+Alltrim(Totais.Dopes)+' Digitada em Duplicidade!!!!',0+16,'Atenção')
				Return .f.
			EndIf
		EndScan

		m.lnErro1 = SigCdOpe.poDatamgr.pnIdConn

		With SigCdOpe
			If m.lnErro1>0 
				m.lnErro1 = Iif(.poDataMgr.UpDate('crSigCdMop'),1,-1)
			EndIf 
			If m.lnErro1>0 
				m.lnErro1 = .poDataMgr.Commit()
			EndIf 
			If Not m.lnErro1>0 
				.poDataMgr.RollBack()
			EndIf 
		EndWith 
		If Not m.lnErro1>0 
			= MessageBox('Erro na Gravação dos Dados do Cadastro de Operações.',48)
		EndIf 
	EndIf
ENDIF

Thisform.ParentForm.Enabled = .T.
Thisform.mCtrlBotoes(.T.)
ThisForm.Release
ENDPROC
