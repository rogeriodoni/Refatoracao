PROCEDURE Init
Lparameters ocnx, __saldo

*!*	thisform.parentform = oform

lcAlias = Upper(Alias())
*!*	lcArqCb = [TprMvCab]
lcproduto = &lcAlias..CPros
dtini = Dtot(Date()) &&TprMvCab.datas
dtfin = Iif(IsNull(TprMvCab.prazoents) or Empty(TprMvCab.prazoents),TprMvCab.datas,TprMvCab.prazoents)
dtfin = fDtoSQL(dtfin,'23:59:59')

*** Seleciona as operações que tem movimentação do determinado produto no período da venda e entrega, e que não foi usada subnível
lcoper = TprMvCab.dopes
lcquery = [select b.empdopnums, b.datas, b.prazoents, a.qtds, a.qtbaixas, 000000.000 as qtdests ] + ;
			[from SigMvItn a inner join SigMvCab b on a.empdopnums = b.empdopnums where cpros = ?lcproduto and a.dopes = ?lcoper and ]+;
			[(b.datas between ?dtini and ?dtfin or b.prazoents >= ?dtini) and b.chksubn = 0 ] +;
			[order by b.datas , b.prazoents ]
ocnx.sqlexecute(lcquery,[TmpReser])

*** Seleciona as operações Pai que tem movimentação do determinado produto no período da venda e entrega
lcquery2 = [select dopes from SigCdOpe where nivels = '] + lcoper + [' and copers = 1]
ocnx.sqlexecute(lcquery2,[TmpOpe])
lcoper = tmpOpe.dopes
ocnx.sqlexecute(lcquery,[TmpLoca])

Create Cursor Tmpresul(datas d, estoque n(11,3))
index on datas to datas

Insert into Tmpresul(datas,estoque) values (Date(), __saldo)

*** Calcula o estoque por dia dos itens já reservados
Select TmpReser
Scan
	=Seek(Ttod(Tmpreser.datas),'TmpResul','datas')
	If !Eof('TmpResul')
		If TprMvCab.empdopnums = tmpreser.empdopnums
			replace estoque with estoque + (TmpReser.qtds - TmpReser.qtbaixas) for datas >= Ttod(Tmpreser.datas) in TmpResul
		Else
			replace estoque with estoque - (TmpReser.qtds - TmpReser.qtbaixas) for datas >= Ttod(Tmpreser.datas) in TmpResul
		EndIf
	Else
		dmax = Max(Ttod(dtini),TmpReser.datas)
		Select Max(datas) as datas from tmpresul where datas <= dmax into cursor Tmpdata
		lnsaldo = __saldo
		If !IsNull(tmpdata.datas)
	*!*			Set near On
			=Seek(tmpdata.datas,'Tmpresul','datas')
	*!*			Set Near Off
			If !Eof('Tmpresul')
	*!*				Skip -1 in Tmpresul
	*!*				If !Eof('Tmpresul')
					lnsaldo = Tmpresul.estoque
	*!*				EndIf
			EndIf	
		EndIf
		If TprMvCab.empdopnums = tmpreser.empdopnums
			Insert into TmpResul(datas,estoque) values (Ttod(Tmpreser.datas), lnsaldo + (TmpReser.qtds - TmpReser.qtbaixas))
		Else
			Insert into TmpResul(datas,estoque) values (Ttod(Tmpreser.datas), lnsaldo - (TmpReser.qtds - TmpReser.qtbaixas))
		EndIf
	EndIf
	
	=Seek(Ttod(TmpReser.prazoents),'TmpResul','datas')
	If !Eof('TmpResul')
		replace estoque with estoque + (TmpReser.qtds - TmpReser.qtbaixas) for datas >= Ttod(Tmpreser.prazoents) in TmpResul
	Else
		Select Max(datas) as datas from tmpresul where datas between TmpReser.datas and Ttod(TmpReser.prazoents) into cursor Tmpdata
		lnsaldo = __saldo

		If !IsNull(tmpdata.datas)
	*!*			Set near On
			=Seek(tmpdata.datas,'Tmpresul','datas')
	*!*			Set Near Off
			If !Eof('Tmpresul')
	*!*				Skip -1 in Tmpresul
	*!*				If !Eof('Tmpresul')
					lnsaldo = Tmpresul.estoque
	*!*				EndIf
			EndIf	
		EndIf
		
		Insert into Tmpresul(datas,estoque) values (Ttod(TmpReser.prazoents), lnsaldo + (TmpReser.qtds - TmpReser.qtbaixas)) &&
	EndIf
EndScan

*** Calcula o estoque por dia dos itens já locados
Select TmpLoca
Scan 
	=Seek(Ttod(TmpLoca.datas),'Tmpresul','datas')
	If !Eof('Tmpresul')
		replace estoque with estoque - (TmpLoca.qtds - TmpLoca.qtbaixas) for datas >= Ttod(TmpLoca.datas) and datas < dtini in TmpResul
	Else
		dmax = Max(Ttod(dtini),TmpLoca.datas)
		Select Max(datas) as datas from tmpresul where datas <= dmax into cursor Tmpdata
		lnsaldo = __saldo
	
		If !IsNull(tmpdata.datas)
	*!*			Set near On
			=Seek(tmpdata.datas,'Tmpresul','datas')
	*!*			Set Near Off
			If !Eof('Tmpresul')
	*!*				Skip -1 in Tmpresul
	*!*				If !Eof('Tmpresul')
					lnsaldo = Tmpresul.estoque
	*!*				EndIf
			EndIf	
		EndIf
		Insert into Tmpresul(datas,estoque) values (Ttod(TmpLoca.datas), lnsaldo - (TmpLoca.qtds - TmpLoca.qtbaixas)) &&__saldo 
	EndIf
*!*		lnsaldo = lnsaldo - TmpLoca.qtds 
	
	=Seek(Ttod(TmpLoca.prazoents),'TmpResul','datas')
	If !Eof('TmpResul')
		replace estoque with estoque + (TmpLoca.qtds - TmpLoca.qtbaixas) for datas >= Ttod(TmpLoca.prazoents) in TmpResul
	Else
		Select Max(datas) as datas from tmpresul where datas between TmpLoca.datas and Ttod(TmpLoca.prazoents) into cursor Tmpdata
		lnsaldo = __saldo
	
		If !IsNull(tmpdata.datas)
	*!*			Set near On
			=Seek(tmpdata.datas,'Tmpresul','datas')
	*!*			Set Near Off
			If !Eof('Tmpresul')
	*!*				Skip -1 in Tmpresul
	*!*				If !Eof('Tmpresul')
					lnsaldo = Tmpresul.estoque
	*!*				EndIf
			EndIf	
		EndIf	
		Insert into TmpResul(datas,estoque) values (Ttod(TmpLoca.prazoents), lnsaldo + (TmpLoca.qtds - TmpLoca.qtbaixas)) &&__saldo 
	EndIf
EndScan

*** Inseri estoque nos dias que não tiveram movimentação
Select Min(datas) as dtini, Max(datas) as dtfin from TmpResul into cursor TmpData
nsaldoant = __saldo
If Reccount('TmpResul') > 0
	For x = 0 to (TmpData.dtfin - TmpData.dtini)
		=Seek(TmpData.dtini+x,'Tmpresul','datas')
		If !Eof('Tmpresul')
			nsaldoant = Tmpresul.estoque
		Else
			Insert into Tmpresul(datas,estoque) values (TmpData.dtini+x, nsaldoant)
		EndIf
	Next 
Else
	Insert into Tmpresul(datas,estoque) values (Ttod(TprMvCab.datas), nsaldoant)
EndIf

Select * from Tmpresul where datas between dtini and dtfin into cursor Tmpresul order by datas

Select Tmpresul
index on datas to datas
Set near On
=Seek(Ttod(TprMvCab.datas),'Tmpresul','datas')
Set Near Off
If Bof() or Eof()
	Go bottom
EndIf
thisform.nsaldo = TmpResul.estoque
thisform.grid1.RecordSource = [TmpResul]
thisform.grid1.column1.ControlSource = [TmpResul.datas]
thisform.grid1.column2.ControlSource = [TmpResul.estoque]

thisform.Refresh()

ENDPROC
