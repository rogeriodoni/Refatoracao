PROCEDURE Init
LParameters pArq, pEsc, pObj

With ThisForm
	.poDataMgr = pObj
	.Arquivo   = Alltrim(pArq)
	.pcEscolha = pEsc
	
	lcArq = .Arquivo
	
	.Caption = [Impostos Gerados Na Movimentação: ] + ;
			   Alltrim(Left(&lcArq..EmpDopNums, 3)) + [ / ] + ;
			   Alltrim(Substr(&lcArq..EmpDopNums, 4, 20)) + [ / ] + ;
			   Alltrim(Right(&lcArq..EmpDopNums, 6))
	
	.cntSombra.lblSombra.Caption = .Caption
	.cntSombra.lblTitulo.Caption = .Caption
	
	Select &lcArq.
	Goto Top In &lcArq.
	With .grdTitulos
		.RecordSource          = lcArq
		.Column1.ControlSource = lcArq + [.Impostos]
		.Column2.ControlSource = lcArq + [.ValBases]
		.Column3.ControlSource = lcArq + [.Aliqs]
		.Column4.ControlSource = lcArq + [.ValImps]
		.Column5.ControlSource = lcArq + [.Grupos]
		.Column6.ControlSource = lcArq + [.Contas]
		.Column7.ControlSource = lcArq + [.ValTits]
		.Column8.ControlSource = [Ttod(] + lcArq + [.Vencs)]
		
		.Refresh
		.AfterRowColChange()
	EndWith
	
	If InList(.pcEscolha, [INSERIR], [ALTERAR])
		.grdTitulos.Column7.SetFocus
	Else
		.cmdSair.SetFocus
	EndIf
EndWith
* Fim **************************************************************************************
ENDPROC

------------------------------------------------------------
* PROCEDURE AfterRowColChange

PROCEDURE AfterRowColChange
Lparameters nColIndex

lcArq = ThisForm.Arquivo
	
ThisForm.getDGrupos.Value = []
If Not Empty(&lcArq..Grupos)
	lcQuery = [Select Descrs ] + ;
				[From SigCdGcr ] + ;
			   [Where Codigos = '] + Padr(&lcArq..Grupos, 10) + [' ] + ;
			   [Order By Codigos]

	If (ThisForm.poDataMgr.SqlExecute(lcQuery, [crfOpFin_Bus]) < 1)
		=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (crfOpFin_Bus)])
		Return .f.
	EndIf
	Go Top In crfOpFin_Bus
	lcDgr = Iif(Eof([crfOpFin_Bus]), [], crfOpFin_Bus.Descrs)
	ThisForm.getDGrupos.Value = lcDgr
EndIf

ThisForm.getDContas.Value = []
If Not Empty(&lcArq..Contas)
	lcQuery = [Select RClis ] + ;
				[From SigCdCli ] + ;
			   [Where IClis = '] + Padr(&lcArq..Contas, 10) + [' ] + ;
			   [Order By IClis]

	If (ThisForm.poDataMgr.SqlExecute(lcQuery, [crfOpFin_Bus]) < 1)
		=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (crfOpFin_Bus)])
		Return .f.
	EndIf
	Go Top In crfOpFin_Bus
	lcDct = Iif(Eof([crfOpFin_Bus]), [], crfOpFin_Bus.RClis)
	ThisForm.getDContas.Value = lcDct
EndIf

If Used([crfOpFin_Bus])
	Use In crfOpFin_Bus
EndIf

ThisForm.getDGrupos.Refresh
ThisForm.getDContas.Refresh 
* Fim **************************************************************************************
ENDPROC

PROCEDURE Click
Local lcArq, llOks

lcArq = ThisForm.Arquivo
llOks = .t.
	
Select &lcArq.
Go Top In &lcArq.
Do While Not Eof() And llOks
	If (Empty(&lcArq..ValTits) Or IsEmpty(&lcArq..Vencs))
		If (MessageBox([O ] + Alltrim(&lcArq..Impostos) + [ NÃO Está Com Valor Título / Vencimento] + Chr(13) + ;
					   [Informado e NÃO Será Lançado! Deseja Corrigir?], 4+32+256, [Lançamento de Título Cancelado]) = 6)
			llOks = .f.
		EndIf	
	EndIf
	If llOks
		Skip In &lcArq.
	EndIf
EndDo

If Not llOks
	ThisForm.grdTitulos.Column7.SetFocus
	Return .f.
Else
	ThisForm.Release()
EndIf
* Fim **************************************************************************************
ENDPROC
