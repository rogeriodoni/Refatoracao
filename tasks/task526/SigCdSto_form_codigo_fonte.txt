* PROCEDURE chkdtvalid
------------------------------------------------------------
PROCEDURE chkdtvalid
Lparameters pTip

Local lcStr, loPag

loPag = ThisForm.Pagina.Dados
lcStr = CrSigCdSts.DdVals

If (pTip = [SALVAR])
	lcStr = Str(loPag.chkEntregaDo.Value, 1) + ;
			Str(loPag.chkEntrega02.Value, 1) + ;
			Str(loPag.chkEntrega03.Value, 1) + ;
			Str(loPag.chkEntrega04.Value, 1) + ;
			Str(loPag.chkEntrega05.Value, 1) + ;
			Str(loPag.chkEntrega06.Value, 1) + ;
			Str(loPag.chkEntregaSa.Value, 1) + ;
			Str(loPag.chkEntregaFe.Value, 1)

	Replace ddVals With lcStr In CrSigCdSts
Else
	loPag.chkEntregaDo.Value = Val(Substr(lcStr, 1, 1))
	loPag.chkEntrega02.Value = Val(Substr(lcStr, 2, 1))
	loPag.chkEntrega03.Value = Val(Substr(lcStr, 3, 1))
	loPag.chkEntrega04.Value = Val(Substr(lcStr, 4, 1))
	loPag.chkEntrega05.Value = Val(Substr(lcStr, 5, 1))
	loPag.chkEntrega06.Value = Val(Substr(lcStr, 6, 1))
	loPag.chkEntregaSa.Value = Val(Substr(lcStr, 7, 1))
	loPag.chkEntregaFe.Value = Val(Substr(lcStr, 8, 1))
EndIf

loPag.chkEntregaDo.Refresh 
loPag.chkEntrega02.Refresh 
loPag.chkEntrega03.Refresh 
loPag.chkEntrega04.Refresh 
loPag.chkEntrega05.Refresh 
loPag.chkEntrega06.Refresh 
loPag.chkEntregaSa.Refresh 
loPag.chkEntregaFe.Refresh 

ENDPROC

------------------------------------------------------------
* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
With ThisForm
	.poDataMgr = CreateObject([fSqlConector], .Name)

	If (.poDataMgr.pnIdconn > 0)
		With .poDataMgr
			.AddCursor([SigCdSts], [Cods], [crSigCdSts], [], ThisForm.Pagina.Lista.Grade)
			.pfSqlTabela(1).pColuna([Cods],  [], [], [Código],    080, .t.)
			.pfSqlTabela(1).pColuna([Descs], [], [], [Descrição], 220, .t.)
		EndWith

		=DoDefault()

		.Pagina.Lista.Grade.Column1.SetFocus
	Else
		=Messagebox([Impossível Efetuar Conexão Com o Servidor de Banco de Dados...], 0+48, [CreateObject])
		Return .f.
	EndIf
EndWith

ENDPROC

ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Lista.Grupo_op.Click
------------------------------------------------------------
PROCEDURE Pagina.Lista.Grupo_op.Click
Parameters lnOpcao

If Not DoDefault(lnOpcao)
	ThisForm.mAtivaPagina1
	Return .f.
EndIf

ThisForm.ChkDtvalid('Leitura')
Do Case
	Case InList(ThisForm.pcEscolha, [INSERIR], [PROCURAR])
		 ThisForm.Pagina.Dados.getCods.SetFocus
		 
	Case InList(ThisForm.pcEscolha, [ALTERAR])
		 ThisForm.Pagina.Dados.getDescs.SetFocus

	Otherwise
		 ThisForm.Pagina.Dados.Grupo_Salva.Salva.SetFocus
EndCase
* Fim **************************************************************************************
ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
If InList(ThisForm.pcEscolha, [INSERIR], [ALTERAR])
	If Empty(ThisForm.Pagina.Dados.getCods.Value)
		=Messagebox([Código Inválido!!!], 48, [Atenção!!!])
		ThisForm.Pagina.Dados.getCods.SetFocus
		Return .f.
	EndIf
	If Empty(ThisForm.Pagina.Dados.getDescs.Value)
		=Messagebox([Descrição Inválida!!!], 48, [Atenção!!!])
		ThisForm.Pagina.Dados.getDescs.SetFocus
		Return .f.
	EndIf
	If (ThisForm.pcEscolha = [INSERIR])
		lcQuery = [Select Cods ] + ;
				    [From SigCdSts ] + ;
				   [Where Cods = '] + Padr(ThisForm.Pagina.Dados.getCods.Value, 3) + [']

		If (ThisForm.poDataMgr.SqlExecute(lcQuery, [LocalBus]) < 1)
			=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (LocalBus 1)])
			Return .f.
		EndIf
		Go Top In LocalBus
		If Not Eof([LocalBus])
			=Messagebox([Código Já Cadastrado!!!], 48, [Atenção!!!])
			ThisForm.Pagina.Dados.getCods.SetFocus
			Return .f.
		EndIf
	EndIf
	ThisForm.Chkdtvalid([SALVAR])
EndIf
=DoDefault()
ThisForm.mAtivaPagina1

ENDPROC
