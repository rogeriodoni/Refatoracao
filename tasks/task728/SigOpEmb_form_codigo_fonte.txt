* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
lParameters poForm, pCnx

ThisForm.cntSombra.lblSombra.Caption = ThisForm.Caption
ThisForm.cntSombra.lblTitulo.Caption = ThisForm.Caption

ThisForm.ParentForm = poForm
thisform.parentform.pesoemb = 0

ThisForm.poDataMgr = pCnx

Select crSigMvIte
Set Order to
Set Order to CItens
Set Key to crTpmMvItn.CItens
Go Top

lcQuery = [Select Descs, Cods, Pesos as PesoEmbs From SigCdemb Order By Cods]
If (ThisForm.poDataMgr.SqlExecute(lcQuery, [LocalCdEmb]) < 1) 
	=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (LocalEmb)])
	Return .f.
EndIf

Select a.Cods, a.Descs, a.PesoEmbs, Nvl(b.EmpDopNums,'') as EmpDopNums, Nvl(b.Emps,'') as Emps, Nvl(b.Dopes,'') as Dopes, Nvl(b.Numes,0) as Numes, ;
		Nvl(b.Qtds,0) as Qtds, Nvl(b.Pesos,0) as Pesos, Nvl(b.CidChaves,'') as CidChaves, Nvl(b.CodEmbs,'') as CodEmbs ;
	From LocalCdEmb a Left Join crSigMvIte b On a.Cods = b.CodEmbs And b.Citens = crTpmMvItn.Citens Order By a.Cods Into Cursor LocalEmb ReadWrite

This.Grade.RecordSource          = [LocalEmb]
This.Grade.Column1.ControlSource = [LocalEmb.Qtds]
This.Grade.Column2.ControlSource = [LocalEmb.Descs]
This.Grade.Column3.ControlSource = [LocalEmb.PesoEmbs]
This.Grade.Column4.ControlSource = [LocalEmb.Pesos]

Select LocalEmb
Go Top

This.Refresh()
This.Grade.Column1.SetFocus()
* Fim **************************************************************************************

ENDPROC

PROCEDURE btnConfirmar.Click
LOCAL lnTotP

Select LocalEmb
Go Top In LocalEmb
lnTotP = 0

Select CrSigMvIte
Set Order To
Set Order To Citens
Go Top
Delete For cItens = crTpmMvItn.Citens in CrSigMvIte

**** Calcula peso total das embalagens ****
Select LocalEmb
Scan
	If LocalEmb.Qtds > 0
		lnTotP = lnTotP + Localemb.Qtds * Localemb.PesoEmbs
		
		Scatter Memo Memvar
		m.Citens = crTpmMvItn.Citens
		m.CodEmbs = m.Cods
		Insert into crSigMvIte From Memvar
	Endif
Endscan
thisform.parentform.pesoemb = lnTotP
ThisForm.Release()
* Fim **************************************************************************************
ENDPROC
