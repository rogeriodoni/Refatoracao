PROCEDURE msv_procurar
Local lcCodigos, lcDescs

lcCodigos = Alltrim(ThisForm.Pagina.Dados.GetCodigo.Value)
lcDescs   = Alltrim(ThisForm.Pagina.Dados.GetDescricao.Value)

Select CrSigCdTif
Set Near On

Do Case

	Case !Empty(lcCodigos)
		Set Order To Codigos
		Seek lcCodigos
		ThisForm.Busca = 1

	Case !Empty(lcDescs)
		Set Order To Descricaos
		Seek lcDescs
		ThisForm.Busca = 2

EndCase

Set Near Off
ENDPROC

PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
Select CrSigCdTif
Go Top

lcGCodigo = This.Parent.Parent.GetCodigo.Value

If ThisForm.pcEscolha = 'INSERIR'
	If Empty(lcGCodigo)
		Messagebox('Código Inválido',48,'Campo Obrigatório')
		ThisForm.Pagina.Dados.GetCodigo.SetFocus
		Return 0
	EndIf
	
	If Used('TmpTif')
		Use In TmpTif
	EndIf
	lStrQuery = [Select Codigos From SigCdTif Where Codigos = ']+lcGCodigo+[' ]
	If ThisForm.poDataMgr.SqlExecute(lStrQuery,'TmpTif') < 1
		MessageBox('Favor reinicializar o processo !!!',16,'Falha na Conexão (TmpTif)')
		Return(.F.)
	EndIf
	Select TmpTif
	Go Top
	If !Eof()
		MessageBox('Código Já Cadastrado !!!',48,'')
		ThisForm.Pagina.Dados.GetCodigo.SetFocus
		Return
	EndIf
Endif

=DoDefault()
ThisForm.mAtivapagina1
ENDPROC
