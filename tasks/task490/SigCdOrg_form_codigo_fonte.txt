------------------------------------------------------------
* PROCEDURE msv_procurar
------------------------------------------------------------
PROCEDURE msv_procurar
Local lcBusca

If .Not.IsEmpty(ThisForm.Pagina.Dados.Getcodigo.Value)
	ThisForm.poDataMgr.LocateCursor('crSigCdOrg','codigos',ThisForm.Pagina.Dados.Getcodigo.Value)
EndIf


ENDPROC

------------------------------------------------------------
* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
LPARAMETERS lcacesso

With ThisForm
	.poDatamgr = CreateObject('fSqlConector',ThisForm.Name)
	With .poDatamgr
		.AddCursor('SigCdOrg','codigos','crSigCdOrg','',ThisForm.pagina.lista.grade)
		.pfSqlTabela(1).pColuna('Codigos','','','Código',50,.t.)
		.pfSqlTabela(1).pColuna('Descricaos','','','Descriçãos',290,.t.)
	EndWith
	= DoDefault()
	.LockScreen = .f.
EndWith

ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Lista.Grupo_op.Click
------------------------------------------------------------
PROCEDURE Pagina.Lista.Grupo_op.Click
parameters lnOpcao

If !dodefault( lnOpcao )
	ThisForm.mAtivapagina1
	Return
Endif

do case
	case thisform.pcEscolha='INSERIR' or ;
		thisform.pcEscolha='PROCURAR'
		thisform.Pagina.Dados.getcodigo.setfocus
	case thisform.pcEscolha='ALTERAR'
		thisform.pagina.dados.getdescricao.setfocus
	case thisform.pcEscolha='CONSULTAR' or ;
		thisform.pcEscolha='EXCLUIR'
		thisform.pagina.dados.grupo_salva.salva.setfocus
endcase

ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click

local lcRetorno

m.lcRetorno = .t.

if inlist(thisform.pcEscolha,'INSERIR','ALTERAR')

   with thisform.Pagina.Dados
      if IsEmpty(.getCodigo.value)
	      = messagebox('Código em branco !!!',0+64,goSistema.NomeSistema)
         .getCodigo.setfocus
		  return
	  else
		  If ThisForm.poDataMgr.ChkRegister('SigCdOrg','Codigos',ThisForm.Pagina.Dados.GetCodigo.Value) and thisform.pcEscolha = 'INSERIR'
	         = messagebox('Código já cadastrado !!!',0+64,goSistema.NomeSistema)
	         .getCodigo.setfocus
			  return
	      else
		      if IsEmpty(.getDescricao.value)
		         = messagebox('Descrição Inválida !!!',0+64,goSistema.NomeSistema)
		         .getDescricao.setfocus
				  return
			  endif
	      endif
	  endif
	endwith
ENDIF

IF inlist(thisform.pcEscolha,'EXCLUIR')
	IF ThisForm.poDataMgr.ChkRegister('SigCdPro','origmercs',ThisForm.Pagina.Dados.GetCodigo.Value) OR ThisForm.poDataMgr.ChkRegister('SigCdPrc','origmercs',ThisForm.Pagina.Dados.GetCodigo.Value)
		MESSAGEBOX('Origem não pode ser excluida pois está sendo usada em Produtos!',0+64,goSistema.NomeSistema)
		RETURN .f.
	ENDIF
ENDIF

=dodefault()
select CrSigCdOrg
ThisForm.mAtivapagina1

ENDPROC

* PROCEDURE Pagina.Dados.Activate
------------------------------------------------------------
PROCEDURE Pagina.Dados.Activate

=dodefault()

if (thisform.pcEscolha=='INSERIR'.or.!eof('CrSigCdOrg'))

	if inlist(thisform.pcEscolha,'INSERIR','ALTERAR','PROCURAR')
		thisform.Pagina.Dados.getCodigo.setfocus
	else
	   thisform.Pagina.Dados.Grupo_salva.Salva.setfocus
	endif   

else

   thisform.Pagina.Lista.enabled = .t.
   thisform.Pagina.activepage = 1
   thisform.Pagina.Lista.Grade.setfocus
   
endif

ENDPROC
