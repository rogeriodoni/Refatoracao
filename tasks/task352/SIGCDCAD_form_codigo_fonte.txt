PROCEDURE Init
Lparameters pTip, pTam
Private pIds

pIds = Replicate(Chr(254), 20)

lcQryCdCad = [Select * From SigCdCad Where TipoCads = ?pIds]

With ThisForm
	.Name      = 'SIGCDCAD'		&& mena - trocado para liberar os botoes Alltrim(Upper(pTip))
	.TipoCads  = Alltrim(Upper(pTip))
	.poDataMgr = CreateObject([fSqlConector], .Name)
	.Caption   = Proper(.TipoCads)
	
	* Aqui podem entrar algumas excessões 
	If (Padr(Upper(.TipoCads), 20) = Padr([APONTAMTO], 20))
		.Caption = [Avaliação]
	EndIf
	
	pIds = .TipoCads

	If (Type([pTam])<>[N]) Or Not Between(pTam, 1, 20) 
		pTam = 20
	EndIf
		
	If (.poDataMgr.pnIdconn > 0)
		With .poDataMgr
			.AddCursor([SigCdCad], [pkChaves], [crSigCdCad], [], ThisForm.Pagina.Lista.Grade, lcQryCdCad)
			.pfSqlTabela(1).pColuna([CodCads], [], [], [Código], (pTam * 7) + 10, .t.)
			.pfSqlTabela(1).pColuna([DesCads], [], [], [Descrição], 300, .t.)
		EndWith

		=DoDefault()

		.Pagina.Dados.lblCodCads.Value = Alltrim(.Caption) + [ :]
		.Pagina.Dados.getCodCads.MaxLength = pTam
		.Pagina.Dados.getCodCads.Width = (pTam * 7) + 10

		.Pagina.Lista.Grade.Column1.SetFocus
	Else
		=Messagebox([Impossível Efetuar Conexão Com o Servidor de Banco de Dados...], 0+48, [CreateObject])
		Return .f.
	EndIf
ENDWITH
* Fim **************************************************************************************
ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Lista.Grupo_op.Click
------------------------------------------------------------
PROCEDURE Pagina.Lista.Grupo_op.Click
LParameters Opcao
Private pIds

pIds = ThisForm.TipoCads

If Not DoDefault(Opcao)
	ThisForm.mAtivaPagina1
	Return .f.
EndIf

If (ThisForm.pcEscolha = [INSERIR])
	Replace TipoCads With pIds, ;
			pkChaves With fUniqueIds(), ;
			Usuars   With Usuar, ;
			DtIncs   With Datetime() In crSigCdCad
EndIf

If (ThisForm.pcEscolha = [ALTERAR])
	Replace UsuAlts With Usuar, ;
			DtAlts  With Datetime() In crSigCdCad
EndIf

Do Case
	Case Inlist(ThisForm.pcEscolha, [INSERIR], [PROCURAR])
		ThisForm.Pagina.Dados.getCodCads.SetFocus

	Case (ThisForm.pcEscolha = [ALTERAR])
		ThisForm.Pagina.Dados.getDesCads.SetFocus

	Case Inlist(ThisForm.pcEscolha, [EXCLUIR], [CONSULTAR])
		ThisForm.Pagina.Dados.Grupo_Salva.Salva.SetFocus
EndCase
* Fim **************************************************************************************

ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
If (ThisForm.pcEscolha = [INSERIR])
	If Empty(crSigCdCad.CodCads)
		=Messagebox([O Código Não Pode Ficar Em Branco!!!], 48, [Atenção!!!]) 
		ThisForm.Pagina.Dados.getCodCads.SetFocus
		Return .f.
	EndIf
	
	lcQuery = [Select * ] + ;
			    [From SigCdCad ] + ;
			   [Where TipoCads = '] + Padr(ThisForm.TipoCads, 20) + [' And ] + ;
			   		 [CodCads = '] + crSigCdCad.CodCads + [']

	If (ThisForm.poDataMgr.SqlExecute(lcQuery, [LocalBus]) < 1)
		=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (LocalBus)])
		Return .f.
	EndIf
	Go Top In LocalBus
	If Not Eof([LocalBus])
		=Messagebox([O Código Já Está Cadastrado!!!], 48, [Atenção!!!]) 
		ThisForm.Pagina.Dados.getCodCads.SetFocus
		Return .f.
	EndIf
EndIf

=DoDefault()

ThisForm.mAtivaPagina1
* Fim **************************************************************************************
ENDPROC
