------------------------------------------------------------
* PROCEDURE msv_procurar
------------------------------------------------------------
PROCEDURE msv_procurar
=DoDefault()
If .Not. Empty(ThisForm.Pagina.Dados.Get_Cods.Value)
	ThisForm.pCol = 1
	ThisForm.poDataMgr.LocateCursor('crSigCcCco','cods',ThisForm.Pagina.Dados.Get_Cods.Value)
Else
	If .Not.IsEmpty(ThisForm.Pagina.Dados.Get_Descs.Value)
		ThisForm.pCol = 2
		ThisForm.poDataMgr.LocateCursor('crSigCcCco','descs',ThisForm.Pagina.Dados.Get_Descs.Value)
	EndIf
EndIf



ENDPROC

------------------------------------------------------------
* PROCEDURE mativapagina1
------------------------------------------------------------
PROCEDURE mativapagina1

=DoDefault()

ThisForm.Pagina.Lista.Grade.SetFocus
Do Case
	Case (ThisForm.pCol = 1)
		ThisForm.Pagina.Lista.Grade.Column1.SetFocus

	Case (ThisForm.pCol = 2)
		ThisForm.Pagina.Lista.Grade.Column2.SetFocus
EndCase
ThisForm.Pagina.Lista.Grade.Refresh
ENDPROC

------------------------------------------------------------
* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
&&=fConfigGeral()

lcQryCCoCo = [Select * From SigCcCCo]

ThisForm.pCol = 1

With ThisForm
	.poDataMgr = CreateObject('fSqlConector', ThisForm.Name)

	If (.poDataMgr.pnIdconn > 0)
		With .poDataMgr
			.AddCursor('SigCcCCo', 'Cods', 'crSigCcCCo', '', ThisForm.Pagina.Lista.Grade, lcQryCCoCo)
			.pfSqlTabela(1).pColuna('Cods', '', '', 'Códigos',  150, .t.)
			.pfSqlTabela(1).pColuna('descs','', '','Descrição', 220, .t.)
		EndWith

		=DoDefault()

		ThisForm.Pagina.Lista.Grade.Column1.SetFocus
	Else
		=Messagebox('Impossivel Efetuar Conexão com Servidor de Banco de Dados...', 0+48, 'CreateObject')
		Return .f.
	EndIf
EndWith
* Fim **************************************************************************************


ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Lista.Grupo_op.Click
------------------------------------------------------------
PROCEDURE Pagina.Lista.Grupo_op.Click
LParameters opcao

If !dodefault( opcao )
	ThisForm.mAtivaPagina1
	Return
Endif

Do Case
	Case Inlist( ThisForm.pcEscolha, 'PROCURAR', 'INSERIR' )
		thisform.pagina.dados.Get_Cods.setfocus	
	Case ThisForm.pcEscolha = 'ALTERAR'
		thisform.pagina.dados.Get_Descs.setfocus
	Case Inlist( ThisForm.pcEscolha, 'CONSULTAR', 'EXCLUIR' )
		thisform.pagina.dados.grupo_salva.salva.setfocus
Endcase

ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click

If ThisForm.pcEscolha = 'INSERIR'
   If Empty(ThisForm.Pagina.Dados.Get_Cods.Value)
	    MessageBox( 'Obrigatório Informar o Código.', 48, '' )
        ThisForm.Pagina.Dados.Get_Cods.SetFocus
		Return
	Endif
	If ThisForm.poDataMgr.ChkRegister('SigCcCco','cods',ThisForm.Pagina.Dados.Get_Cods.Value)
		MessageBox( 'Código já cadastrado...', 48, '' )
		ThisForm.Pagina.Dados.Get_Cods.SetFocus
		Return
	Endif
Endif
If ThisForm.pcEscolha = 'INSERIR' or ThisForm.pcEscolha = 'ALTERAR'
  If IsEmpty(ThisForm.Pagina.Dados.Get_Descs.Value)
    MessageBox( 'A descrição não pode estar vazia.', 48, '' )
    ThisForm.Pagina.Dados.Get_Descs.SetFocus
	Return
  EndIf
Endif

= DoDefault()

ThisForm.mAtivaPagina1
ENDPROC

If ThisForm.plCancelar
	ThisForm.mAtivapagina1
Endif

ENDPROC
