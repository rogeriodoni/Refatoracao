* PROCEDURE montagrade
------------------------------------------------------------
PROCEDURE montagrade
Private pKey

pKey = crSigCdSet.Setores

With ThisForm.Pagina.Dados.grdLocalizacoes
	.RecordSource = ''

	ThisForm.poDataMgr.Requery('crSigCdSei')
	Select crSigCdSei
	Scan
		ThisForm.poDataMgr.CursorQuery('SigPrLcl', 'LocalPrLoc', 'Codigos', crSigCdSei.Locals, 'Descricaos')
		If Not Eof('LocalPrLoc')
			Replace Descricaos With LocalPrLoc.Descricaos In crSigCdSei
		EndIf
	EndScan
	Go Top In crSigCdSei

	.RecordSource          = 'crSigCdSei'
	.Column1.ControlSource = 'crSigCdSei.Locals'
	.Column2.ControlSource = 'crSigCdSei.Descricaos'
	.Refresh
EndWith
* Fim **************************************************************************************
ENDPROC

------------------------------------------------------------
* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
Private pKey

pKey = Replicate(Chr(254), 10)

lcQrySetor = [Select * From SigCdSet]
lcQrySetoI = [Select *, Space(20) as Descricaos From SigCdSei Where Setores = ?pKey]

With ThisForm
	.poDataMgr = CreateObject('fSqlConector', ThisForm.Name)

	If (.poDataMgr.pnIdconn > 0)
		With .poDataMgr
			.AddCursor('SigCdSet', 'Setores', 'crSigCdSet', '', ThisForm.Pagina.Lista.Grade, lcQrySetor)
			.pfSqlTabela(1).pColuna('Setores', '', '', 'Setor',     080, .t.)
			.pfSqlTabela(1).pColuna('Descrs',  '', '', 'Descrição', 290, .t.)

			.AddCursor('SigCdSei', 'cIdChaves', 'crSigCdSei', '', '', lcQrySetoI)
		EndWith

		=DoDefault()

		ThisForm.Pagina.Lista.Grade.Column1.SetFocus
	Else
		=Messagebox('Impossível Efetuar Conexão Com o Servidor de Banco de Dados...', 0+48, 'CreateObject - ' + ThisForm.Name)
		Return .f.
	EndIf
EndWith
* Fim **************************************************************************************
ENDPROC

PROCEDURE msv_procurar
loDad = ThisForm.Pagina.Dados

Select crSigCdSet

	Case Not Empty(loDad.getSetores.Value)
		Locate For Setores = loDad.getSetores.Value

		Locate For Descrs = loDad.getDescrs.Value

* PROCEDURE pagina.lista.grupo_op.Click
------------------------------------------------------------
PROCEDURE pagina.lista.grupo_op.Click
LParameters lnOpcao

If (Type('lnOpcao') = 'N')
	This.Value = lnOpcao
EndIf

Select crSigCdSet

If Not DoDefault(lnOpcao)
	ThisForm.mAtivapagina1
	Return 0
EndIf

ThisForm.MontaGrade

loDad = ThisForm.Pagina.Dados

Do Case
	Case Inlist(ThisForm.pcEscolha, 'PROCURAR', 'INSERIR')
		loDad.getSetores.Setfocus	

	Case (ThisForm.pcEscolha = 'ALTERAR')
		loDad.grdLocalizacoes.Column1.Setfocus	

	Case Inlist(ThisForm.pcEscolha, 'CONSULTAR', 'EXCLUIR')
		loDad.Grupo_Salva.Salva.Setfocus
EndCase

loDad.cmdApagar.Visible = InList(ThisForm.pcEscolha, 'ALTERAR', 'INSERIR')

If InList(ThisForm.pcEscolha, 'ALTERAR', 'INSERIR')
	* Insere Um Setor Na Grade
	Go Bottom In crSigCdSei
	Insert Into crSigCdSei (Setores, cIdChaves) Values (crSigCdSet.Setores, fUniqueIds())
EndIf

Select crSigCdSet

loDad.Refresh
* Fim **************************************************************************************
ENDPROC

------------------------------------------------------------
* PROCEDURE pagina.dados.grupo_salva.salva.Click
------------------------------------------------------------
PROCEDURE pagina.dados.grupo_salva.salva.Click
Private pKey

pKey = crSigCdSet.Setores

loDad = ThisForm.Pagina.Dados

Delete From crSigCdSei Where Empty(Locals)

If InList(ThisForm.pcEscolha, 'INSERIR', 'ALTERAR')
	If Empty(loDad.getSetores.Value)
		=MessageBox('Obrigatório Informar o Setor!!!', 48, '')
		loDad.getSetores.SetFocus
		Return 0
	EndIf

	If Empty(loDad.getDescrs.Value)
		=MessageBox('Obrigatório Informar a Descrição!!!', 48, '')
		loDad.getDescrs.SetFocus
		Return 0
	EndIf

	Select * ;
	  From crSigCdSei ;
	  Into Cursor csItens

	If (Reccount('csItens') < 1)
		=MessageBox('Não Foram Cadastrados Grupos!!!', 48, '')
		loDad.grdLocalizacoes.Column1.SetFocus
		Return 0
	EndIf

	If (Thisform.pcEscolha = 'INSERIR' )
		If ThisForm.poDataMgr.CursorQuery('SigCdSet', 'LocalSetor', 'Setores',loDad.getSetores.Value )
			MessageBox( 'Código já cadastrado...', 48, '' )
			loDad.getSetores.SetFocus
			Return 0
		EndIf
	EndIf
	
	Select crSigCdSei
	Scan
		If Empty(crSigCdSei.Setores)
			Replace Setores With crSigCdSet.Setores In crSigCdSei
		EndIf
	EndScan
EndIf

If (ThisForm.pcEscolha = 'EXCLUIR')
	ThisForm.MontaGrade
	Delete From crSigCdSei
EndIf

If Not ThisForm.poDataMgr.Update('crSigCdSei')
	=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (Update - crSigCdSei)')
	Return 0
EndIf

Select crSigCdSet

=DoDefault()

ThisForm.mAtivaPagina1
* Fim ************************************************************************************** 
ENDPROC

PROCEDURE Valid
Local loLista

If Not Empty(This.Value)
	loLista = CreateObject('fwBuscaExt', ThisForm.poDataMgr.pnIdConn, 'SigPrLcl', ;
						   'crListaRemota', 'Codigos', This.Value, 'Seleção', .t.)

	If Not loLista.plAchouRegistro
		loLista.mAddColuna('Codigos',    '', 'Código')
		loLista.mAddColuna('Descricaos', '', 'Descrição')
		loLista.Show()
	EndIf
	
	This.Value = crListaRemota.Codigos
	ThisForm.Pagina.Dados.grdLocalizacoes.Column2.Text1.Value = crListaRemota.Descricaos
	Use In crListaRemota
Else
	This.Value = ''
EndIf
This.Refresh

Select CrSigCdSei
If Not Empty(This.Value)
	&& checa Duplicidade de Local na Grade
	Skip 
	Skip -1
	Select Count(1) As Qt From crSigCdSei ;
		Where Locals = This.Value ;
		Into Cursor TmpTotSetores
	If TmpTotSetores.Qt > 1
		MessageBox('Código já Cadastrado!!!',0+16,'')
		This.Value = ' '
		ThisForm.Pagina.Dados.grdLocalizacoes.Column2.Text1.Value = ' '
		ThisForm.Pagina.Dados.grdLocalizacoes.Refresh
		Return 0
	Else
		&& checa se Local já não está cadastrado em outro Setor
		lcLocals = This.Value
		lcSetors = ThisForm.Pagina.Dados.GetSetores.Value
		lcSql = [Select Locals,Setores From SigCdSei Where Locals = ']+lcLocals+[' And Setores <> ']+lcSetors+[']
		If ThisForm.Podatamgr.Sqlexecute(lcSql,'LocalSetoI') < 1
			=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (LocalSetoI)')
			Return 0
		EndIf
		If (Reccount('LocalSetoI') > 0)
			MessageBox('O Local '+lcLocals+' já está Cadastrado no Setor '+LocalSetoI.Setores+'!!!',0+16,'')
			This.Value = ' '	
			ThisForm.Pagina.Dados.grdLocalizacoes.Column2.Text1.Value = ' '
			ThisForm.Pagina.Dados.grdLocalizacoes.Refresh
			Return 0
		Endif

		&& Insere Um Local Na Grade Se Não Existir Uma Em Branco
		Select * ;
		  From crSigCdSei ;
		 Where Empty(Locals) ;
		  Into Cursor TmpInsSetores

		If (Reccount('TmpInsSetores') = 0)
			Go Bottom In crSigCdSei

			Insert Into crSigCdSei (Setores, cIdChaves) Values (crSigCdSet.Setores, fUniqueIds())
			ThisForm.Pagina.Dados.grdLocalizacoes.Column1.Refresh
		EndIf
		Select CrSigCdSei
		If This.Value <> ThisForm.Antvalue Or (Reccount('TmpInsSetores') = 0) Or Lastkey() = 13
			Keyboard '{DnArrow}'
		Endif
	Endif
EndIf
* Fim **************************************************************************************
ENDPROC
