PROCEDURE hinivel
Parameters pnNivel, pcEmpDopNums, pccPros

Local lcSqlFile, lcEmpDopNums

m.loTree1 = ThisForm.TreeView

lcSqlFile = 'crHiEstPe'+Alltrim(Str(pnNivel))

If Type('pccPros') = [C]
	lcSqlEstPe = [Select a.Codigos, a.Emps, a.Dopes, a.Numes, a.EmpSubns, ] + ;
				 [b.Datas, b.Datars, b.Notas, b.ChkPagos, b.EmpDopNums, b.Usuars, b.DtAlts, b.Usuals, ] + ;
				 [Sum(c.Qtds) As Qtds ] + ;
	             [From SigMvPec a ] + ;
	             [Inner Join SigMvCab b on a.EmpDopNums = b.EmpDopNums ] + ;
	             [Left  Join SigMvItn c on a.EmpDopNums = c.EmpDopNums ]+;
	             [Where a.EmpDopNums = '] + pcEmpDopNums + [' ] + ;
	             Iif(Type('pccPros') = [C], [And c.Cpros = '] + pccPros + [' ], [And 0 = 1 ]) + ;
			     [Group By a.Codigos, a.Emps, a.Dopes, a.Numes, a.EmpSubns, ] + ;
				 [b.Datas, b.Datars, b.Notas, b.ChkPagos, b.EmpDopNums, b.Usuars, b.DtAlts, b.Usuals ]+;
			     [Union all ]+;
			     [Select a.Codigos, a.Emps, a.Dopes, a.Numes, a.Emps As EmpSubns, ] + ;
			     [b.Datas, b.Datars, b.Notas, b.ChkPagos, b.EmpDopNums, b.Usuars, b.DtAlts, b.Usuals, ] + ;
			     [Sum(c.Qtds) As Qtds ] + ;
	             [From SigOpDev a ] + ;
	             [Inner Join SigMvCab b on a.EmpDopNums = b.EmpDopNums ] + ;
	             [Left  Join SigMvItn c on a.EmpDopNums = c.EmpDopNums ]+;
	             [Where a.EmpDopNums = '] + pcEmpDopNums + [' ] + ;
	             Iif(Type('pccPros') = [C] And pnNivel = 3, [ And c.Cpros = '] + pccPros + [' ],[]) + ;
			     [Group By a.Codigos, a.Emps, a.Dopes, a.Numes, ] + ;
			     [b.Datas, b.Datars, b.Notas, b.ChkPagos, b.EmpDopNums, b.Usuars, b.DtAlts, b.Usuals ] + ;
				 [Order By b.Datas, b.Datars, b.EmpDopNums, 14]
Else
	lcSqlEstPe = [Select a.Codigos, a.Emps, a.Dopes, a.Numes, a.EmpSubns, ] + ;
				 [b.Datas, b.Datars, b.Notas, b.ChkPagos, b.EmpDopNums, b.Usuars, b.DtAlts, b.Usuals, ] + ;
				 [Sum(c.Qtds) As Qtds ] + ;
	             [From SigMvPec a ] + ;
	             [Inner Join SigMvCab b on a.EmpDopNums = b.EmpDopNums ] + ;
	             [Left  Join SigMvItn c on a.EmpDopNums = c.EmpDopNums ]+;
	             [Where a.EmpDopNums = '] + pcEmpDopNums + [' ] + ;
			     [Group By a.Codigos, a.Emps, a.Dopes, a.Numes, a.EmpSubns, ] + ;
				 [b.Datas, b.Datars, b.Notas, b.ChkPagos, b.EmpDopNums, b.Usuars, b.DtAlts, b.Usuals ] + ;
			     [Union all ]+;
			     [Select a.Codigos, a.Emps, a.Dopes, a.Numes, a.Emps As EmpSubns, ] + ;
			     [b.Datas, b.Datars, b.Notas, b.ChkPagos, b.EmpDopNums, b.Usuars, b.DtAlts, b.Usuals, ] + ;
			     [Sum(c.Qtds) As Qtds ] + ;
	             [From SigOpDev a ] + ;
	             [Inner Join SigMvCab b on a.EmpDopNums = b.EmpDopNums ] + ;
	             [Left  Join SigMvItn c on a.EmpDopNums = c.EmpDopNums ]+;
	             [Where a.EmpDopNums = '] + pcEmpDopNums + [' ] + ;
			     [Group By a.Codigos, a.Emps, a.Dopes, a.Numes, ] + ;
			     [b.Datas, b.Datars, b.Notas, b.ChkPagos, b.EmpDopNums, b.Usuars, b.DtAlts, b.Usuals ] + ;
				 [Order By b.Datas, b.Datars, b.EmpDopNums, 14]
EndIf 

If Thisform.PoDataMgr.SqlExecute(lcSqlEstPe,lcSqlFile) < 1
	MessageBox('Favor reinicializar o processo.',16,'Falha na Conexão ' + lcSqlFile)
	RETURN .F.
EndIf

Scan

	lnDopes = Alltrim(Str(Codigos/1000000))	

	lcSqlOpe = [Select Dopes From SigCdOpe Where nDopes = ] + lnDopes
	If Thisform.PoDataMgr.SqlExecute(lcSqlOpe,'crTmpOPe') < 1
		MessageBox('Favor reinicializar o processo.',16,'Falha na Conexão (crTmpOpe)')
		Return .F.
	EndIf

	Select &lcSqlFile.

	lnNumes = Val(Right(Alltrim(Str(Codigos)),6))

	lcEmpDopNums = EmpSubns + crTmpOpe.Dopes + Str(lnNumes,6)
	
	nPos = Ascan(ThisForm.pcEmpDopNums, pcEmpDopNums + lcEmpDopNums + Iif(Type('pccPros') = [C], pccPros, Space(14)))

*	If nPos = 0 Or ThisForm.pcEmpDopNums[Int(nPos/2)+1,2] = pnNivel
	If nPos = 0 And ! lcEmpDopNums = ThisForm.pcEmpDopNums[1,1]

		Declare ThisForm.pcEmpDopNums[ALen(ThisForm.pcEmpDopNums,1)+1,2]
		ThisForm.pcEmpDopNums[ALen(ThisForm.pcEmpDopNums,1),1] = pcEmpDopNums + lcEmpDopNums + Iif(Type('pccPros') = [C], pccPros, Space(14))
		ThisForm.pcEmpDopNums[ALen(ThisForm.pcEmpDopNums,1),2] = pnNivel

		lcQryBx = [Select a.Emps, a.Dopes, a.Numes, a.Datas, a.Datars, a.Notas, a.ChkPagos, a.Usuars, a.DtAlts, a. Usuals, ] + ;
		          [c.EmpDopNums, Sum(c.Qtds) As Qtds ]+;
		          [From SigMvCab a ] + ;
		          [Left Join SigMvItn c On a.EmpDopNums = c.EmpDopNums ] + ;
		          [Where ] + Iif(pnNivel = 3 And Type('pccPros') = [C],;
		          [c.Cpros = '] + pccPros + [' And ],[]) + ;
				  [a.EmpDopNums = '] + lcEmpDopNums + [' ]+;
				  [Group By a.Emps, a.Dopes, a.Numes, a.Datas, a.Datars, a.Notas, a.ChkPagos, a.Usuars, a.DtAlts, a. Usuals, c.EmpDopNums ]
		
		If Thisform.PoDataMgr.SqlExecute(lcQryBx,'crTprMvCab') < 1
			MessageBox('Favor reinicializar o processo.',16,'Falha na Conexão ' + lcSqlFile)
			RETURN .F.
		EndIf

		If ! Eof()
			lcDtt = Left(Ttoc(Nvl(Datas, Ctot(''))), 10) + [ (] + ;
			Substr(Ttoc(Nvl(Datars, Ctot(''))), 12, 5) + [)]

			lcDados   = '(-) ' + ;
			            Alltrim(Emps)  + ' - ' + ;
			            Alltrim(Dopes) + ' - ' + ;
			            Alltrim(Transform(Numes,'999999')) + ' - ' + ;
		    		    Alltrim(Notas) + ' - ' + ;
		    		    lcDtt + ' - ' + ;
		    		    Alltrim(Usuars) + ;
		    		    Iif(!IsNull(Qtds),' - Qtde : ' + Alltrim(Transform(Qtds,'999,999.99')),'')  + ' - ' + ;
		    		    Dtoc(Ttod(Nvl(DtAlts, Ctot('')))) + ' - ' + ;
		    		    Alltrim(UsuAls)  

			m.lnNode1 = m.loTree1.AddPictureNode(lcDados, 0, pnNivel, Iif(ChkPagos,3,1), 0, 0)
			m.loTree1.NodeHeader(m.lnNode1) = .f.
			m.loTree1.NodeExpanded(m.lnNode1) = .f.
			m.loTree1.NodeShowChecks(m.lnNode1) = .f.
			m.loTree1.NodeFontBold(m.lnNode1) = 1
			m.loTree1.NodeCargo(m.lnNode1) = Nvl(lcEmpDopNums,'')
			lcDados1 = lcDados

			If ! Upper(Substr(lcEmpDopNums, 4, 20)) = [NF - TRF SIMBOLICA] And ThisForm.pnNiveis <= 50
				ThisForm.pnNiveis = ThisForm.pnNiveis + 1
				ThisForm.SubNivel(pnNivel + 1, lcEmpDopNums, pccPros)
				ThisForm.HiNivel(pnNivel + 1, lcEmpDopNums, pccPros)
				ThisForm.pnNiveis = ThisForm.pnNiveis - 1
			EndIf
		Endif
	
	Endif

EndScan

ENDPROC

* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
Parameters plModal, pcEmps, pcDopes, pcNumes

Private lcEmpDopNums, lccPros

Declare ThisForm.paEmpDopNums[1]
ThisForm.paEmpDopNums[1] = ''

Declare ThisForm.pcEmpDopNums[1,2]
ThisForm.pcEmpDopNums[1,1] = ''

lcEmpDopNums = ''

ThisForm.GetDtI.Value = Ctod('')
ThisForm.GetDtF.Value = Ctod('')
ThisForm.lacoficina = .f.

lcQryEest  = [Select * From SigMvCab Where EmpDopNums = ?lcEmpDopNums]
lcQryEesti = [Select EmpdopNums, cPros, dPros, Sum(Qtds) As Qtds From SigMvItn Where EmpDopNums = ?lcEmpDopNums ] + ;
             [Group by EmpdopNums, cPros, dPros Order By EmpdopNums, cPros, dPros]

With ThisForm
	.poDataMgr  = CreateObject('fSqlConector', ThisForm.Name)
	With .poDataMgr
		.AddCursor('SigMvCab',  'cIdChaves', 'crSigMvCab',  '', .f., lcQryEest)
		.AddCursor('SigMvItn', 'cIdChaves', 'crSigMvItn', '', .f., lcQryEesti)
	EndWith
EndWith

ThisForm.GetCEmp.Value  = []
ThisForm.GetDopes.Value = []
ThisForm.GetNumes.Value = 0

ThisForm.WindowType = Iif(plModal,1,0)

ThisForm.Automatico = Not Empty(pcEmps) And Not Empty(pcDopes) And Not Empty(pcNumes)

If ! IsEmpty(pcEmps) 
	ThisForm.poDatamgr.SqlExecute([Select * From SigCdEmp Where cEmps = ']+Alltrim(pcEmps)+['],'crSigCdEmp')
	ThisForm.GetCEmp.Value  = Alltrim(pcEmps)
Endif
ThisForm.GetCEmp.ReadOnly   = ! IsEmpty(pcEmps) And ! IsEmpty(pcDopes) And ! IsEmpty(pcNumes) 

If ! IsEmpty(pcDopes) 
	ThisForm.GetDopes.Value = pcDopes
Endif
ThisForm.GetDopes.ReadOnly  = ! IsEmpty(pcEmps) And ! IsEmpty(pcDopes) And ! IsEmpty(pcNumes) 

If ! IsEmpty(pcNumes) 
	ThisForm.GetNumes.Value = pcNumes
Endif
ThisForm.GetNumes.ReadOnly  = ! IsEmpty(pcEmps) And ! IsEmpty(pcDopes) And ! IsEmpty(pcNumes) 

Local loTree1

loTree1 = ThisForm.TreeView

loTree1.AddImage(LoadPicture('\vbmp\Form4.Ico'))
loTree1.AddImage(LoadPicture('\vbmp\depend3.bmp'))
loTree1.AddImage(LoadPicture('\vbmp\OHist.Ico'))

If ThisForm.Automatico
	ThisForm.GetCEmp.Enabled  = .f.
	ThisForm.GetDopes.Enabled = .f.
	ThisForm.GetNumes.Enabled = .f.
	ThisForm.GetDtI.Enabled = .F.
	ThisForm.GetDtF.Enabled = .F.
	ThisForm.MakeTree()
EndIf
ENDPROC

PROCEDURE makelist
Local lcEmp, lcDopes, lcQryEest
Private  lcDtIni, lcDtFim

lcEmp 	= ThisForm.GetcEmp.Value
lcDopes = ThisForm.GetDopes.Value
lcDtIni = fDtoSQL(ThisForm.GetDtI.Value)
lcDtFim = fDtoSQL(ThisForm.GetDtF.Value,'23:59:59')

m.loTree1 = ThisForm.TreeView

m.loTree1.ClearNodes
m.loTree1.TipsDisplay = .t.
m.loTree1.ExpandOnDClick 
m.loTree1.SmallIcons = .t.
m.loTree1.PicAlign = 0
m.loTree1.SelectedStyle = 2
m.loTree1.PictureType = 1

lcQryEest = [Select a.Emps, a.Dopes, a.Numes, a.Notas, a.Datas, a.Datars, a.ChkPagos, Sum(b.qtds) as qtds ] + ;
			  [From SigMvCab a ] + ;
		      [Left Join SigMvItn b on a.EmpDopNums = b.EmpDopNums ] + ;
		     [Where a.Emps = ']+lcEmp+[' And ] + ;
		    	   [a.Dopes = ']+lcDopes+[' And ] + ;
		    	   [a.Datas Between ?lcDtIni And ?lcDtFim ] + ;
		     [Group By a.Datas, a.Datars, a.Emps, a.Dopes, a.Numes, a.Notas, a.ChkPagos ] + ; 
		     [Order By a.Datas, a.Datars, a.Emps, a.Dopes, a.Numes, a.Notas, a.ChkPagos]
		  
ThisForm.poDatamgr.SqlExecute(lcQryEest,'crSigMvCab')

If Eof()
	MessageBox('Não Foi Localizada Nenhuma Operação',16,'')
	Return .f.
Endif

Scan

	lcDados   = Alltrim(Emps)  + ' - ' +Alltrim(Dopes) + ' - ' + Alltrim(Transform(Numes,'999999')) + ' - ' + ;
    		    Alltrim(Notas) + ' - ' +Dtoc(Ttod(Nvl(Datas, Ctot('')))) + Iif(! isNull(Qtds),' - Qtde : '+Alltrim(Transform(Qtds,'999,999.99')),[])
	m.lnNode1 = m.loTree1.AddPictureNode(lcDados, 0, 1, Iif(ChkPagos,3,1), 0, 0)
	m.loTree1.NodeHeader(m.lnNode1) = .f.
	m.loTree1.NodeExpanded(m.lnNode1) = .f.
	m.loTree1.NodeShowChecks(m.lnNode1) = .f.
	m.loTree1.NodeFontBold(m.lnNode1) = 1
	m.loTree1.NodeCargo(m.lnNode1) = Nvl(Emps+Dopes+Alltrim(Transform(Numes,'999999')),'')
 
 EndScan 
ENDPROC

lcTmpSub = 'crTmpSub'+Alltrim(Str(pnNivel))

If Thisform.PoDataMgr.SqlExecute(lcSql,'CrxOpe') < 1
	MessageBox('Favor reinicializar o processo.',16,'Falha na Conexão (CrXOpe)')
	RETURN .F.
EndIf

lcCodigo = Alltrim(Str(nDopes))+Transform(Val(Right(lcEmpDopNums,6)),"@L 999999")

lcQryBx = [Select a.Codigos, a.Emps, a.Dopes, a.Numes, b.Datas, b.Datars, b.Notas, b.ChkPagos, c.EmpDopNums, a.EmpSubns, Sum(c.Qtds) As Qtds ]+;
          [From SigMvPec a, SigMvCab b, SigMvItn c ]+;
          [Where a.codigos = ]+lcCodigo+[ And ]+;
          [a.EmpDopnums = b.EmpDopNums And ]+;
          [a.EmpDopnums = c.EmpDopNums And ]+;
          [c.Cpros = ']+pccPros+[']+;
		  [Group By a.Codigos, a.Emps, a.Dopes, a.Numes, b.Datas, b.Datars, b.Notas, b.ChkPagos, c.EmpDopNums, a.EmpSubns ]+;
		  [Union all ]+;
		  [Select a.Codigos, a.Emps, a.Dopes, a.Numes, b.Datas, b.Datars, b.Notas, b.ChkPagos, a.EmpDopNums, a.EmpSubns, 0 As Qtds ]+;
	  	  [From SigMvPec a, SigMvCab b ]+;
		  [Where a.codigos = ]+lcCodigo+[ And ]+;
		  [a.EmpDopnums = b.EmpDopNums  And ]+;
		  [a.EmpSubns = ']+lcEmps+[' ]+;
		  [Group By a.Codigos, a.Emps, a.Dopes, a.Numes, b.Datas, b.Datars, b.Notas, b.ChkPagos, a.EmpDopNums, a.EmpSubns  ]+;

		  [From SigOpDev a, SigMvCab b, SigMvItn c ]+;
		  [Where a.codigos = ]+lcCodigo+[ And ]+;
		  [a.EmpDopnums = b.EmpDopNums And ]+;
		  [a.EmpDopnums = c.EmpDopNums And ]+;
		  [c.Cpros = ']+pccPros+[']+;
		  [Group By a.Codigos, a.Emps, a.Dopes, a.Numes, b.Datas, b.Datars, b.Notas, b.ChkPagos, c.EmpDopNums ]

ThisForm.poDatamgr.SqlExecute(lcQryBx ,lcBxAlias)

If Ascan(ThisForm.paOperacoes,EmpdopNums) <> 0
	Return
Endif

Select &lcBxAlias
Select a.codigos, a.Emps, a.Dopes, a.Numes, a.Datas, a.Datars, a.Notas, a.ChkPagos, a.EmpDopNums, EmpSubns, sum(qtds) as qtds From &lcBxAlias a ;
	Group By a.codigos, a.Emps, a.Dopes, a.Numes, a.Datas, a.Datars, a.Notas, a.ChkPagos, a.EmpDopNums, EmpSubns Into Cursor &lcTmpSub ReadWrite

Select &lcTmpSub

Go Top

	Scan 
		lcEDN = EmpDopNums

		lcQryBx = [Select usuars, DtAlts, UsuAls ] + ;
		          [From SigMvCab ] + ;
		          [Where EmpDopNums = ']+lcEDN+[']

		Select &lcTmpSub

			ThisForm.paOperacoes[Len(ThisForm.paOperacoes)]=EmpDopNums
			lcDados   = Alltrim(Emps)  + ' - ' +Alltrim(Dopes) + ' - ' + Alltrim(Transform(Numes,'999999')) + ' - ' + ;
		    		    Alltrim(Notas) + ' - ' +Dtoc(Ttod(Nvl(Datas, Ctot('')))) + ' - ' + Alltrim(csSigMvCab.Usuars) + ;
			    		' - ' + Dtoc(Ttod(Nvl(csSigMvCab.DtAlts, Ctot('')))) + ' - ' + Alltrim(csSigMvCab.UsuAls) 
		   	If lcDados <> lcDados1
				m.lnNode1 = m.loTree1.AddPictureNode(lcDados, 0, pnNivel, Iif(ChkPagos,3,1), 0, 0)
				m.loTree1.NodeHeader(m.lnNode1) = .f.
				m.loTree1.NodeExpanded(m.lnNode1) = .f.
				m.loTree1.NodeShowChecks(m.lnNode1) = .f.
				m.loTree1.NodeFontBold(m.lnNode1) = 1
				m.loTree1.NodeCargo(m.lnNode1) = lcEDN
				lcDados1 = lcDados
			Endif
	*		ThisForm.SubNivel(pnNivel+1, lcEDN, pccPros)
	*		Select &lcTmpSub
			ThisForm.NoHiPro(pnNivel+1, EmpSubns, EmpDopNums, pccPros)
			If pnNivel <> 3
				Select &lcTmpSub
				ThisForm.SubNivel(pnNivel+1, lcEDN, pccPros)
				Select &lcTmpSub
				ThisForm.HINivel(pnNivel+1, EmpSubns, Int(codigos/1000000), Val(Right(Alltrim(Str(Codigos)),6)), pccPros)
				Select &lcTmpSub
				ThisForm.NoProd(pnNivel+1, EmpSubns, codigos, pccPros)

* PROCEDURE subnivel

PROCEDURE subnivel
Parameters pnNivel, pcEmpDopNums, pccPros

Local lcSqlOpe, lcSqlEstPe, lcCodigo, lcSqlFile

m.loTree1 = ThisForm.TreeView

lcSqlOpe = [Select nDopes From SigCdOpe Where Dopes = ']+Substr(pcEmpDopNums,4,20)+[']
If Thisform.PoDataMgr.SqlExecute(lcSqlOpe,'crTmpOpe') < 1
	MessageBox('Favor reinicializar o processo.',16,'Falha na Conexão (crTmpOpe)')
	RETURN .F.
EndIf

lcSqlFile = 'crSubEstPe'+Alltrim(Str(pnNivel))

lcCodigo = Alltrim(Str(crTmpOpe.nDopes * 1000000 + Val(Right(pcEmpDopNums,6))))

If Type('pccPros') = [C]
	lcSqlEstPe = [Select a.Codigos, a.Emps, a.Dopes, a.Numes, a.EmpSubns, ] + ;
				 [b.Datas, b.Datars, b.Notas, b.ChkPagos, b.EmpDopNums, b.Usuars, b.DtAlts, b.Usuals, ] + ;
				 [Sum(c.Qtds) As Qtds ] + ;
	             [From SigMvPec a ] + ;
	             [Inner Join SigMvCab b on a.EmpDopNums = b.EmpDopNums ] + ;
	             [Left  Join SigMvItn c on a.EmpDopNums = c.EmpDopNums ]+;
	             [Where a.Codigos = ] + lcCodigo + [ And a.EmpSubns = '] + Left(pcEmpDopNums,3) + [' ] + ;
	             Iif(Type('pccPros') = [C],[And c.Cpros = ']+pccPros+[' ], [And 0 = 1 ]) + ;
			     [Group By a.Codigos, a.Emps, a.Dopes, a.Numes, a.EmpSubns, ] + ;
				 [b.Datas, b.Datars, b.Notas, b.ChkPagos, b.EmpDopNums, b.Usuars, b.DtAlts, b.Usuals ]+;
			     [Union all ]+;
			     [Select a.Codigos, a.Emps, a.Dopes, a.Numes, a.Emps As EmpSubns, ] + ;
			     [b.Datas, b.Datars, b.Notas, b.ChkPagos, b.EmpDopNums, b.Usuars, b.DtAlts, b.Usuals, ] + ;
			     [Sum(c.Qtds) As Qtds ] + ;
	             [From SigOpDev a ] + ;
	             [Inner Join SigMvCab b on a.EmpDopNums = b.EmpDopNums ] + ;
	             [Left  Join SigMvItn c on a.EmpDopNums = c.EmpDopNums ]+;
	             [Where a.Codigos = ] + lcCodigo + ;
	             Iif(Type('pccPros') = [C] And pnNivel = 3, [ And c.Cpros = '] + pccPros + [' ],[]) + ;
			     [Group By a.Codigos, a.Emps, a.Dopes, a.Numes, ] + ;
			     [b.Datas, b.Datars, b.Notas, b.ChkPagos, b.EmpDopNums, b.Usuars, b.DtAlts, b.Usuals ] + ;
				 [Order By b.Datas, b.Datars, b.EmpDopNums, 14]
Else
	lcSqlEstPe = [Select a.Codigos, a.Emps, a.Dopes, a.Numes, a.EmpSubns, ] + ;
				 [b.Datas, b.Datars, b.Notas, b.ChkPagos, b.EmpDopNums, b.Usuars, b.DtAlts, b.Usuals, ] + ;
				 [Sum(c.Qtds) As Qtds ] + ;
	             [From SigMvPec a ] + ;
	             [Inner Join SigMvCab b on a.EmpDopNums = b.EmpDopNums ] + ;
	             [Left  Join SigMvItn c on a.EmpDopNums = c.EmpDopNums ]+;
	             [Where a.Codigos = ] + lcCodigo + [ And a.EmpSubns = '] + Left(pcEmpDopNums,3) + [' ] + ;
			     [Group By a.Codigos, a.Emps, a.Dopes, a.Numes, a.EmpSubns, ] + ;
				 [b.Datas, b.Datars, b.Notas, b.ChkPagos, b.EmpDopNums, b.Usuars, b.DtAlts, b.Usuals ] + ;
			     [Union all ]+;
			     [Select a.Codigos, a.Emps, a.Dopes, a.Numes, a.Emps As EmpSubns, ] + ;
			     [b.Datas, b.Datars, b.Notas, b.ChkPagos, b.EmpDopNums, b.Usuars, b.DtAlts, b.Usuals, ] + ;
			     [Sum(c.Qtds) As Qtds ] + ;
	             [From SigOpDev a ] + ;
	             [Inner Join SigMvCab b on a.EmpDopNums = b.EmpDopNums ] + ;
	             [Left  Join SigMvItn c on a.EmpDopNums = c.EmpDopNums ]+;
	             [Where a.Codigos = ] + lcCodigo + ;
			     [Group By a.Codigos, a.Emps, a.Dopes, a.Numes, ] + ;
			     [b.Datas, b.Datars, b.Notas, b.ChkPagos, b.EmpDopNums, b.Usuars, b.DtAlts, b.Usuals ] + ;
				 [Order By b.Datas, b.Datars, b.EmpDopNums, 14]
Endif

If Thisform.PoDataMgr.SqlExecute(lcSqlEstPe,lcSqlFile) < 1
	MessageBox('Favor reinicializar o processo.',16,'Falha na Conexão ' + lcSqlFile)
	RETURN .F.
EndIf

Scan

	nPos = Ascan(ThisForm.pcEmpDopNums, lcCodigo + EmpDopNums + Iif(Type('pccPros') = [C], pccPros, Space(6)))

	If nPos = 0 And ! EmpDopNums = ThisForm.pcEmpDopNums[1,1]

		Declare ThisForm.pcEmpDopNums[ALen(ThisForm.pcEmpDopNums,1)+1,2]
		ThisForm.pcEmpDopNums[ALen(ThisForm.pcEmpDopNums,1),1] = lcCodigo + EmpDopNums + Iif(Type('pccPros') = [C], pccPros, Space(6))
		ThisForm.pcEmpDopNums[ALen(ThisForm.pcEmpDopNums,1),2] = pnNivel
