------------------------------------------------------------
* PROCEDURE msv_procurar
------------------------------------------------------------
PROCEDURE msv_procurar
Local lcGCodigo, lcGDescrs

lnGCodigo = ThisForm.Pagina.Dados.Get_Codigos.Value
lcGDescrs = Alltrim(ThisForm.Pagina.Dados.Get_Descrs.Value)

Select CrSigInTgo
Set Near On

Do Case

	Case !Empty(lnGCodigo)
		Set Order To Codigos
		Seek lnGCodigo
		ThisForm.Busca = 1

	Case !Empty(lcGDescrs)
		Set Order To Descs
		Seek lcGDescrs
		ThisForm.Busca = 2

EndCase

Set Near Off
ENDPROC

------------------------------------------------------------
* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
LParameters lcAcesso

ThisForm.Busca = 1

With ThisForm
	.poDatamgr = CreateObject('fSqlConector',ThisForm.Name)
	With .poDatamgr
		.AddCursor('SigInTgo','Codigos','CrSigInTgo','',ThisForm.Pagina.Lista.Grade)
		.pfSqlTabela(1).pColuna('Codigos','','','Codigos'  , 80,.T.)
		.pfSqlTabela(1).pColuna('Descs' ,'','','Descrição',279,.T.)
	EndWith
	=DoDefault()
	.LockScreen = .F.
EndWith

ThisForm.mAtivaPagina1

ThisForm.LockScreen = .T.
=fCarregaGrade(ThisForm.Pagina.Lista.Grade)
ThisForm.LockScreen = .F.
ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Lista.Grupo_op.Click
------------------------------------------------------------
PROCEDURE Pagina.Lista.Grupo_op.Click
LParameters Opcao

Select CrSigInTgo

If !DoDefault(Opcao)
	ThisForm.mAtivaPagina1
	Return 0
EndIf


Do Case
	Case InList(Thisform.pcEscolha,'INSERIR','PROCURAR')
		Thisform.Pagina.Dados.Get_Codigos.SetFocus
		
	Case InList(Thisform.pcEscolha,'ALTERAR')
		Thisform.Pagina.Dados.Get_Descrs.SetFocus

	OtherWise
		Thisform.Pagina.Dados.Grupo_Salva.Salva.SetFocus
EndCase
ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
If InList(ThisForm.pcEscolha,'INSERIR')
	If Empty(CrSigInTgo.Codigos)
		Messagebox('Código Inválido !!!',48,'Campo Obrigatório')
		ThisForm.Pagina.Dados.Get_Codigos.SetFocus
		Return 0
	EndIf
	
	If Empty(CrSigInTgo.Descs)
		Messagebox('Descrição Inválida !!!',48,'Campo Obrigatório')
		ThisForm.Pagina.Dados.Get_Descrs.SetFocus
		Return 0
	EndIf

	If ThisForm.poDataMgr.ChkRegister('SigInTgo','Codigos',ThisForm.Pagina.Dados.Get_Codigos.Value)
		Messagebox('Código Já Cadastrado !!!',48,'')
		ThisForm.Pagina.Dados.Get_Codigos.SetFocus
		Return 0
	EndIf
EndIf

If InList(ThisForm.pcEscolha,'INSERIR','ALTERAR')
	For x = 1 to 2
		_Prog = Allt( CrSigInTgo.Codigos )
		_Desc = 'Menu - Indústria - Gerar Industrialização' + Iif(x=1,'',' (S)') +' por Tipo - ' + Allt( CrSigInTgo.Codigos )

		DECLARE laProgs[2,2]
		laProgs[1,1]="Programas"
		laProgs[1,2]=Iif(x=1,"SIGPRGLOT","SIGIDISVT")
		laProgs[2,1]="Parametros"
		laProgs[2,2]=_Prog

		If !ThisForm.podatamgr.ChkRegister('SigCdPrg',@laProgs)
			lcQuery = [Insert Into SigCdPrg (descricaos, parametros, programas, barraforms, barrapict,pkchaves) ]+;
					  [ Values ( ] + ;
					  [ '] + _Desc + [']+;
					  [,'] + _Prog + [']+;
					  [,']+ Iif(x=1,"SIGPRGLOT","SIGIDISVT") +[']+;
					  [,' ',' ']+;
					  [,']+fUniqueIds()+[')]
			If (ThisForm.poDataMgr.SqlExecute(lcQuery, '') < 1)
				=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (crInsPrograma)')
				Return 0
			EndIf
		Endif
	Next
EndIf

=DoDefault()
ThisForm.mAtivaPagina1
ENDPROC

PROCEDURE Valid
*!*	IF not EMPTY(this.value)
*!*		SELECT nivel
*!*		SET ORDER TO dopps
*!*		SET NEAR ON
*!*		IF not SEEK(this.value)
*!*		Set Near On
*!*			xBrow = CreateObject('fwBuscaInt', 'nivel', 'dopps', This.Value, 'Seleção')
*!*			xBrow.mAddColuna('dopps', '', 'Código', .t.)
*!*			xBrow.mAddColuna('ndopps', '999', '', .F.)
*!*			xBrow.Show
*!*		ENDIF
*!*		SET NEAR OFF
*!*		IF LASTKEY()<>27
*!*			this.value = EVALUATE("dopps")
*!*			this.refresh
*!*		ENDIF
*!*	ENDIF

if not IsEmpty(this.value)
	loLista = CreateObject('fwBuscaExt', ThisForm.poDataMgr.pnIdConn, 'SigCdOpd', ;
						   'crListaRemota', 'Dopps', This.Value, 'Operações de Produção', .T.)
	If Not loLista.plAchouRegistro
		loLista.mAddColuna('Dopps', '', 'Operação')
		loLista.Show()
	EndIf
	If Lastkey() == 27 
		This.Value = ''
	Else
		This.Value = crListaRemota.Dopps
	EndIf
	Use In crListaRemota
EndIf
ENDPROC
