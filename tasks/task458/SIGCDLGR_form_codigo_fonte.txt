------------------------------------------------------------
PROCEDURE Init
LParameters lcAcesso

thisform.pagina.lista.getDtIni.Value = DATE()
thisform.pagina.lista.getDtfim.Value = DATE()
lcDatas = DATE()

PRIVATE LcSerie , LdDtini , LdDtfim
Lcserie = ""
LdDtini = ""
LdDtfim = ""

lcQuery = [select b.cmarcas, a.*, ?lcDatas As tdtData  from SigFiGfd a inner join SigFiMpm b on a.cmodelos = b.cModelos ] + ;
          [where a.cserifab = ?Lcserie and a.cdatas between ?Lddtini and ?Lddtfim]

With Thisform
	.poDatamgr = CreateObject('fSqlConector',ThisForm.Name)
	If Not .poDatamgr.pnIdconn>0
		Messagebox('Impossivel Efetuar Conexão com Servidor de Banco de Dados.',(0+48),'Erro')
		Return(.f.)
	Endif

	With .poDatamgr
		.AddCursor('SigFiGfd','cIdChaves','crSigFiGfd','cDatas',ThisForm.pagina.lista.grade, lcQuery)
		.pfSqlTabela(1).pColuna('tdtdata','','','Data',75,.t.)
		.pfSqlTabela(1).pColuna('cDoctos','','','Tipo',30,.t.)
		.pfSqlTabela(1).pColuna('cMarcas','','','Marca',150,.t.)
		.pfSqlTabela(1).pColuna('cModelos','','','Modelo',150,.t.)
		.pfSqlTabela(1).pColuna('cCoos','','','COO',50,.t.)
	EndWith
Endwith	

Thisform.LockScreen = .T.

=DoDefault()

replace ALL tdtdata WITH CTOD(LEFT(cDatas,2) + [/] + SUBSTR(cDatas,3,2) + [/] + Right(cDatas,4))

Thisform.Pagina.Lista.Grade.Refresh

Thisform.LockScreen = .F.

* PROCEDURE pagina.lista.grupo_op.Click
------------------------------------------------------------
PROCEDURE pagina.lista.grupo_op.Click
LPARAMETERS opcao

If Type( 'Opcao' ) = 'N'
	This.Value = Opcao
ENDIF

IF EMPTY(ThisForm.Pagina.Lista.getSerieFab.Value) .or. EMPTY(thisform.pagina.Lista.getDtini.value) .or. EMPTY(thisform.pagina.lista.getdtfim.value ) then
   Messagebox('Impossivel Efetuar Manutenção / Consulta sem escolher Série e Datas' ,16,'')
   return
Endif

IF Inlist( This.Value, 3, 4 )
   IF crSigFiGfd.cIncMans <> [M] then
      Messagebox('Registro não Incluído manualmente , Não pode ser Alterado' ,16,'')
      thisform.pagina.lista.SetFocus 
      return   
   Endif   
Endif

lcDatas = DATE()

Lcserie = ThisForm.pagina.lista.GetSerieFab.Value 
LdDtini = StrTran(DTOC(ThisForm.pagina.lista.GetDtini.Value),[/],[])
LdDtfim = StrTran(DTOC(ThisForm.pagina.lista.GetDtfim.Value),[/],[])

If !dodefault( Opcao )
	Thisform.mAtivapagina1
	Return
ENDIF

Do Case

	Case Inlist( ThisForm.pcEscolha, 'INSERIR')
		ThisForm.Podatamgr.sqlExecute([Select cModelos From SigFiMpf Where cImpfabs = '] +  fDecriptografar(Thisform.Pagina.Lista.GetSerieFab.Value) + ['], [crTmpFab])
		ThisForm.Pagina.Dados.Get_codigo.Value = Thisform.Pagina.Lista.GetSerieFab.Value 
		Thisform.Pagina.Dados.GetModelo.Value  = crTmpFab.cModelos
		ThisForm.Pagina.Dados.Getdtmov.Value   = DATE()
		Thisform.Pagina.Dados.Gethrmov.Value   = StrTran(TIME(), [:], [])
		Thisform.pagina.dados.Getcoo.SetFocus

	Case Inlist( ThisForm.pcEscolha, 'PROCURAR', 'ALTERAR' )
		Thisform.pagina.dados.Getcoo.SetFocus

	Case Inlist( ThisForm.pcEscolha, 'CONSULTAR', 'EXCLUIR' )
		thisform.pagina.dados.grupo_salva.salva.setfocus

Endcase

IF Inlist( ThisForm.pcEscolha, 'ALTERAR', 'CONSULTAR', 'EXCLUIR' )
	ThisForm.Pagina.Dados.Getdtmov.Value   = CTOD(LEFT(cDatas,2) + [/] + SUBSTR(cDatas,3,2) + [/] + Right(cDatas,4))
Endif
	

ENDPROC
