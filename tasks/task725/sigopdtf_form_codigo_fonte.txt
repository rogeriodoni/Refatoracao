PROCEDURE errotef
PARAMETERS pnRetornos

If Type("lnRetorno") <> "N" and (lnRetorno > -1 Or lnRetorno < -5)
	ThisForm.RetornoFalha("Operacao Cancelada pelo Usuario", "0,00")
Endif

IF pnRetornos = -1
	ThisForm.RetornoFalha("Modulo Nao Iniciado", "0,00")
ENDIF
IF pnRetornos = -2
	ThisForm.RetornoFalha("Operacao Cancelada pelo Usuario", "0,00")
ENDIF
IF pnRetornos = -3
	ThisForm.RetornoFalha("Fornecida uma Modalidade Invalida", "0,00")
ENDIF
IF pnRetornos = -4
	ThisForm.RetornoFalha("Falta Memoria para Rodar a Funcao", "0,00")
ENDIF
IF pnRetornos = -5
	ThisForm.RetornoFalha("Sem Comunicacao com o SiTef", "0,00")
ENDIF

ENDPROC

 	IF UPPER(ALLTRIM(Buffer))$"DIGITE A SENHA" or ProximoComando = 23
		ThisForm.Container1.Label1.Caption="Digite a Senha"
	Endif
	IF ProximoComando = 3
		lsMRet=ALLTRIM(Buffer)
	Endif
	IF TipoCampo = 100
		lsTipTran=Buffer
	Endif
	IF TipoCampo = 105
		lsDataHora=Buffer
	Endif
	IF TipoCampo = 121
		lsCupom=STRTRAN(Buffer,CHR(0),' ')
	Endif
	IF TipoCampo = 122 
		lsCupom=lsCupom+STRTRAN(Buffer,CHR(0),' ')
	Endif
	IF TipoCampo = 131
		ThisForm.lsCartao=Left(Buffer, 5)
	Endif
	IF TipoCampo = 132
		Thisform.pcBandeira=Left(Buffer,5)
	Endif
	IF TipoCampo = 134
		lsNsu=ALLTRIM(STR(VAL(Buffer)))
	Endif
	IF TipoCampo = 135
		lsAutoriza=Buffer
	ENDIF
	IF TipoCampo = 136
		lcQuery = [Select ChkBins From SigOpFp Where FPags = '] + ThisForm.pcOpers + [']
		ThisForm.poDatamgr.SqlExecute(lcQuery, [csTmpFPag])
		If csTmpFPag.ChkBins = 1
			lcQuery = [Select CodBins, DataIs, DataFs From SigOpFpb Where FPags = '] + ThisForm.pcOpers + [' ] + ;
											       [And CodBins = '] + Alltrim(Strtran(Buffer, Chr(0), '')) + [']				
			ThisForm.poDatamgr.SqlExecute(lcQuery, [csTmpFPagb]) 
			If Eof() Or ;
				Not Between(Dtos(Date()),Dtos(Nvl(csTmpFPagb.Datais,Ctod('01/01/1900'))),Dtos(Nvl(csTmpFPagb.DataFs,Ctod('31/12/9999'))))
				lnRetorno=ContinuaFuncaoSiTefInterativo(@ProximoComando, @TipoCampo,@TamanhoMinimo, @TamanhoMaximo, @Buffer, Len(Buffer), -1)
				DO Form SigTfDss With "Erro na Forma de Pagamento","",'Cartão Invalido para esta','Condição de Pagamento',"","VM",ThisForm.Left,ThisForm.Top 
				RELEASE WINDOWS
				RETURN .f.
			EndIf
		Endif
	ENDIF
	IF lnRetorno <> 0
		lsFinaliza=Buffer
	Endif
	IF ProximoComando <> 21 .and. ProximoComando <> 30 .and. ProximoComando <> 34
		lsMensagem=Buffer
		Buffer=SPACE(2000)
		lnContinua=0
	ENDIF
Enddo 
IF lnRetorno <> 10000
	IF lnRetorno = 0
		ThisForm.MontaRetorno(lsTipTran, lsDataHora, lsCupom, Thisform.pcBandeira, lsNsu, lsAutoriza, lsFinaliza, lnValPago,lsMRet)
	ENDIF
	RELEASE WINDOWS
Endif
ThisForm.Container1.Label1.Caption="Digite Resp."+CHR(13)+"Parcelamento"
ThisForm.GetDigitos.Enabled=.f.
ThisForm.Text1.Enabled=.t.
IF  lsDebCred<>"P" .and. ! DCD
	ThisForm.Container1.Label1.Caption="Digite o Tipo"+CHR(13)+"Venda"
	ThisForm.optiongroup1.Enabled=.t.
	ThisForm.getDatas.Enabled=.t.
	ThisForm.Text1.Enabled=.f.
Else
	ThisForm.optiongroup1.Enabled=.f.
	ThisForm.getDatas.Enabled=.f.
	ThisForm.Text1.Enabled=.f.
Endif

ENDPROC
