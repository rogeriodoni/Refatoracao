* PROCEDURE Pagina.Lista.Grupo_op.Click
------------------------------------------------------------
PROCEDURE Pagina.Lista.Grupo_op.Click
Lparameters Opcao

If Not DoDefault(Opcao)
	ThisForm.mAtivaPagina1
	Return 0
EndIf

Select crSigPrTrn
Do Case
	Case Inlist(ThisForm.pcEscolha, 'PROCURAR', 'INSERIR')
		ThisForm.Pagina.Dados.getCodigos.SetFocus

	Case (ThisForm.pcEscolha = 'ALTERAR')
		ThisForm.Pagina.Dados.getDescrs.SetFocus	

	Case Inlist(ThisForm.pcEscolha, 'CONSULTAR', 'EXCLUIR')
		ThisForm.Pagina.Dados.Grupo_Salva.Salva.SetFocus
EndCase

If InList(Thisform.pcEscolha, 'INSERIR', 'ALTERAR', 'CONSULTAR')

	lcValor = 'TRANSPORTE ' + CrSigPrTrn.Extensao + _Empr
	lcSql = [Select * From SigSySeq Where Valor = ']+lcValor+[']
	If ThisForm.PodataMgr.SQLExecute(lcSql,'CsAutoriza') < 1
		MessageBox('Favor reinicializar o processo !!!',16,'Falha na Conexão (CsAutoriza)')
		Return(.F.)
	Endif
	Select CsAutoriza
	Go Top
	If Reccount() > 0
		This.Parent.Parent.Dados.Get_Transp.Value = csAutoriza.Conteudo + 1
	Else
		This.Parent.Parent.Dados.Get_Transp.Value = 1
	EndIf
	This.Parent.Parent.Dados.Get_Transp.Refresh()
	ThisForm.Proxtransp = This.Parent.Parent.Dados.Get_Transp.Value

	lcEmp = StrTran(_Empr,' ','_') 
	lcChave = Padr('RECEPCAO' + lcEmp + crSigPrTrn.codigos, 25)
	lcQuery = [Select Conteudo From SigSySeq Where Valor = '] + lcChave + [']
	If (ThisForm.poDataMgr.SqlExecute(lcQuery, 'LocalAuto') < 1)
		=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (Processamento - LocalAuto)')
		Return 0
	EndIf
	Select LocalAuto
	Go Top
	If Reccount() > 0
		This.Parent.Parent.Dados.Get_Recebe.Value = LocalAuto.Conteudo + 1
	Else
		This.Parent.Parent.Dados.Get_Recebe.Value = 1
	EndIf
	This.Parent.Parent.Dados.Get_Recebe.Refresh()
	ThisForm.Proxrecebe = This.Parent.Parent.Dados.Get_Recebe.Value
EndIf

Select crSigPrTrn
ThisForm.Pagina.Dados.Refresh

ENDPROC

	    Else

		EndIf

			=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (csAutoriza)')
			Return 0
		EndIf
	EndIf

	If (ThisForm.poDataMgr.Commit() < 1)

	=MessageBox('O Novo Transporte Informado Já Está Cadastrado!!!', 0+48, '')
	This.Parent.getNew.SetFocus
	Use In crTmp

Use In crTmp

If MessageBox('Será Gerado Um Novo Transporte Com o Código : [' + lcNew + ']' + Chr(13) + ;
              'Confirma a Geração ?', 4+32+256, 'Solicitação de Confirmação...') <> 6
	Wait Window 'Operação Cancelada Pelo Usuário...' NoWait

Zap In crSigPrTro

If ThisForm.poDataMgr.CursorQuery('SigPrTro', 'crTmp', 'TpTrans', lcOld)
	Select crTmp
	Scan
		Select crTmp
		Scatter Memvar Memo
		m.TpTrans  	= lcNew
		m.cIdChaves = fUniqueIds()

		Insert Into crSigPrTro From Memvar
	EndScan
EndIf
Use In crTmp

* Gerando o Transporte
Select crSigPrTrn
Scatter Memvar Memo
Zap
m.Codigos  = lcNew
Insert Into crSigPrTrn From Memvar

ThisForm.pcEscolha = 'INSERIR'

ThisForm.mSv_Inserir
