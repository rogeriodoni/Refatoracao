			=Messagebox([Impossível Efetuar Conexão Com o Arquivo SigCdOpt...], 0+48, [AddCursor])

		EndIf

	Else
		=Messagebox([Impossível Efetuar Conexão Com o Servidor de Banco de Dados...], 0+48, [CreateObject])
		Return .f.
	EndIf
EndWith

------------------------------------------------------------
* PROCEDURE Pagina.Lista.Grupo_op.Click
------------------------------------------------------------
PROCEDURE Pagina.Lista.Grupo_op.Click
LParameters Opcao

local LcVersql

If !dodefault( Opcao )
	ThisForm.mAtivapagina1
	Return
Endif

Do Case
	Case Inlist( ThisForm.pcEscolha, 'INSERIR', 'PROCURAR' )
		ThisForm.Pagina.Dados.Get_Codigo.SetFocus
	Case ThisForm.pcEscolha = 'ALTERAR'
 		ThisForm.Pagina.Dados.Get_Descr.SetFocus
	Case Inlist( ThisForm.pcEscolha, 'CONSULTAR', 'EXCLUIR' )
		ThisForm.Pagina.Dados.Grupo_Salva.Salva.SetFocus
Endcase

With ThisForm.Pagina.Dados
	
	LcVersql = .t.

	If Thisform.PoDatamgr.SQLExecute([Select Rclis from SigCdCli where Iclis = '] + CrSigCdOpt.ContaOs + [' ],'CrSigCdCli') > 0
		.Get_DContaOs.Value=CrSigCdCli.Rclis
		.Get_DContaOs.refresh
	else
		LcVersql  = .f.	
	Endif

	If Thisform.PoDatamgr.SQLExecute([Select Descrs from SigCdGcr where Codigos = '] + CrSigCdOpt.GrupoOs + [' ],'CrSigCdGcr') > 0
		.Get_DGrupoOs.Value=CrSigCdGcr.Descrs
		.Get_DGrupoOs.refresh
	Else
		LcVersql  = .f.	
	Endif	

	If Thisform.PoDatamgr.SQLExecute([Select Rclis from SigCdCli where Iclis = '] + CrSigCdOpt.Condeps + [' ],'CrSigCdCli') > 0
		.Get_Dconta.Value=CrSigCdCli.Rclis
		.get_dconta.refresh
	else
		LcVersql  = .f.	
	Endif

	If Thisform.PoDatamgr.SQLExecute([Select Descrs from SigCdGcr where Codigos = '] + CrSigCdOpt.Grudeps + [' ],'CrSigCdGcr') > 0
		.Get_Ds_Grupo.Value=CrSigCdGcr.Descrs
		.get_Ds_grupo.refresh
	Else
		LcVersql  = .f.	
	Endif	

	if Thisform.PoDatamgr.SQLExecute([Select Descs from SIGCCCCO where Cods = '] + CrSigCdOpt.Ccentrals + [' ],'CrSIGCCCCO') > 0
		.Get_Dcentral.Value=CrSIGCCCCO.descs
		.get_dcentral.refresh
	Else	
		LcVersql  = .f.	
	Endif
	
	If !LcVersql
		MessageBox('Favor reinicializar o processo.',16,'Falha na Conexão')
		Return(.F.)
	Endif	
	
EndWith
ENDPROC

PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
Local lnRecNo, lcOperacao, lcOpEntradas

Select CrSigCdOpt
lcOperacao = CrSigCdOpt.Operacaos

If ThisForm.pcEscolha = 'INSERIR'
	If IsEmpty(CrSigCdOpt.Operacaos)
		=MessageBox( 'Código Inválido!', 48, '' )
		ThisForm.Pagina.Dados.Get_Codigo.SetFocus
		Return
	Endif

	If ThisForm.poDataMgr.ChkRegister('SigCdOpt','operacaos',ThisForm.Pagina.Dados.Get_Codigo.Value)
		MessageBox( 'Código já Cadastrado!', 48, '' )
		ThisForm.Pagina.Dados.Get_Codigo.SetFocus
		Return
	Endif
Endif

If InList(ThisForm.pcEscolha,'INSERIR','ALTERAR')
	If CrSigCdOpt.Parcelas = 'S' And CrSigCdOpt.Tipos # 'CR'
		MessageBox('Tipo de Operação Inválida.' + Chr(13) + ;
					'Operação com Parcelas permite apenas CR.', 48, ;
					'Movimentação de Cheques Com Parcelas')
		ThisForm.Pagina.Dados.Get_Tipo.SetFocus
		Return
	EndIf
	
	If CrSigCdOpt.Parcelas = 'S' And IsEmpty(CrSigCdOpt.OpEntradas)
		MessageBox('Obrigatório Informar Operação de Entrada de Cheques.',48,'Movimentação de Cheques Com Parcelas')
		ThisForm.Pagina.Dados.Get_OpEntrada.SetFocus
		Return
	EndIf

	If CrSigCdOpt.Parcelas = 'S' And Not IsEmpty(CrSigCdOpt.OpEntradas)
		lcOpEntradas = CrSigCdOpt.OpEntradas
		If Used('csOpEntradas')
			Use In csOpEntradas
		EndIf
		
		lnQueryOk = ThisForm.poDataMgr.SqlExecute([Select * From SigCdOpt Where operacaos = '] + lcOpEntradas + [' ],'csOpEntradas')
		
		If lnQueryOk < 1
			MessageBox('Favor reinicializar o processo.',16,'Falha na Conexão')
			Return(.F.)
		Endif		
		
		Goto Top In csOpEntradas
		Do Case
			Case CrSigCdOpt.Operacaos = lcOpEntradas Or Eof('csOpEntradas')
				=MessageBox('Operação de Entrada de Cheques Inválida.',48,'Movimentação de Cheques Com Parcelas')
				ThisForm.Pagina.Dados.Get_OpEntrada.SetFocus
				Return
				
			Case CrSigCdOpt.Tipos = csOpEntradas.Tipos
				=MessageBox('Tipo da Operação de Entrada de Cheques Inválido.',48,'Movimentação de Cheques Com Parcelas')
				ThisForm.Pagina.Dados.Get_OpEntrada.SetFocus
				Return
		
		EndCase		
	EndIf
	
	If InList(crSigCdOpt.Comiss, 2, 3) And Empty(crSigCdOpt.Descontos)
		=MessageBox('Percentual de Desconto/Recuperação Inválido!!!',48,'Atenção!!!')
		ThisForm.Pagina.Dados.getDescontos.SetFocus 
		Return
	EndIf
EndIf
	
=DoDefault()

ThisForm.mAtivapagina1

ENDPROC
