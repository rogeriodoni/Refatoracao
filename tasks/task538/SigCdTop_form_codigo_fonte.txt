------------------------------------------------------------
* PROCEDURE msv_procurar
------------------------------------------------------------
PROCEDURE msv_procurar
Local lcGCodigo, lcGDescrs

lnGCodigo = ThisForm.Pagina.Dados.Get_Codigos.Value
lcGDescrs = Alltrim(ThisForm.Pagina.Dados.Get_Descrs.Value)

Select CrSigCdTom
Set Near On

Do Case

	Case !Empty(lnGCodigo)
		Set Order To Codigos
		Seek lnGCodigo
		ThisForm.Busca = 1

	Case !Empty(lcGDescrs)
		Set Order To Descrs
		Seek lcGDescrs
		ThisForm.Busca = 2

EndCase

Set Near Off
ENDPROC

* PROCEDURE mativapagina1
------------------------------------------------------------
PROCEDURE mativapagina1
=DoDefault()

lcCol = Padr(ThisForm.Busca,1)
ThisForm.Pagina.Lista.Grade.SetFocus
ThisForm.Pagina.Lista.Grade.Column&lcCol..SetFocus
ThisForm.Pagina.Lista.Grade.Refresh
ENDPROC

------------------------------------------------------------
* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
LParameters lcAcesso

ThisForm.Busca = 1

With ThisForm
	.poDatamgr = CreateObject('fSqlConector',ThisForm.Name)
	With .poDatamgr
		.AddCursor('SigCdTom','Codigos','CrSigCdTom','',ThisForm.Pagina.Lista.Grade)
		.pfSqlTabela(1).pColuna('Codigos','','','Codigos'  , 44,.T.)
		.pfSqlTabela(1).pColuna('Descrs' ,'','','Descrição',279,.T.)
	EndWith
	=DoDefault()
	.LockScreen = .F.
EndWith

ThisForm.mAtivaPagina1

ThisForm.LockScreen = .T.
=fCarregaGrade(ThisForm.Pagina.Lista.Grade)
ThisForm.LockScreen = .F.
ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Lista.Grupo_op.Click
------------------------------------------------------------
PROCEDURE Pagina.Lista.Grupo_op.Click
LParameters Opcao

Select CrSigCdTom

If !DoDefault(Opcao)
	ThisForm.mAtivaPagina1
	Return 0
EndIf

*!*	If InList(ThisForm.pcEscolha,'EXCLUIR')
*!*		If !(CrSigCdTom.lAlts)
*!*			MessageBox('Registro Automático !!! Impossível Excluir ...',64,'')
*!*			ThisForm.mAtivaPagina1
*!*			Return 0
*!*		EndIf
*!*	EndIf

Do Case
	Case InList(Thisform.pcEscolha,'INSERIR','PROCURAR')
		If ThisForm.pcEscolha = 'INSERIR'
			Replace CrSigCdTom.lAlts with .t.
		Endif
		Thisform.Pagina.Dados.Get_Codigos.SetFocus
		
	Case InList(Thisform.pcEscolha,'ALTERAR')
		Thisform.Pagina.Dados.Get_Descrs.SetFocus

	OtherWise
		Thisform.Pagina.Dados.Grupo_Salva.Salva.SetFocus
EndCase
ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
If InList(ThisForm.pcEscolha,'INSERIR', 'ALTERAR')
	If (CrSigCdTom.Codigos >= 90)
		Messagebox('Os Tipos Devem Ser Cadastrados Com Códigos Inferiores a 90!!!',48,'Campo Inválido')
		ThisForm.Pagina.Dados.Get_Codigos.SetFocus
		Return 0
	EndIf
EndIf

If InList(ThisForm.pcEscolha,'INSERIR')
	If Empty(CrSigCdTom.Codigos)
		Messagebox('Tipo Inválido !!!',48,'Campo Obrigatório')
		ThisForm.Pagina.Dados.Get_Codigos.SetFocus
		Return 0
	EndIf
	
	If Empty(CrSigCdTom.Descrs)
		Messagebox('Descrição Inválida !!!',48,'Campo Obrigatório')
		ThisForm.Pagina.Dados.Get_Descrs.SetFocus
		Return 0
	EndIf

	If ThisForm.poDataMgr.ChkRegister('SigCdTom','Codigos',ThisForm.Pagina.Dados.Get_Codigos.Value)
		Messagebox('Tipo Já Cadastrado !!!',48,'')
		ThisForm.Pagina.Dados.Get_Codigos.SetFocus
		Return 0
	EndIf
	
*!*		Replace CrSigCdTom.lAlts With (CrSigCdTom.Codigos > 9)
EndIf

=DoDefault()
ThisForm.mAtivaPagina1
ENDPROC
