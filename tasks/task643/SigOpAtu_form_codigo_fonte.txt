PROCEDURE Valid
Select CrOpera
Go Top
Set Near On
If !Seek(This.Value)
	m.loLista = CreateObject('fwBuscaInt','CrOpera','Dopes',This.Value,'Operações')
	m.loLista.mAddColuna('Dopes','','Código')
	m.loLista.Show()
	This.Value = Evaluate('CrOpera.dopes')
Else
	This.Value = Evaluate('CrOpera.dopes')
EndIf
Set Near Off
This.Refresh
ENDPROC
