------------------------------------------------------------
* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
Lparameters lcacesso

With Thisform
	.poDatamgr = Createobject('fSqlConector')
	If Not .poDatamgr.pnIdconn>0
		Messagebox('Impossivel Efetuar Conexão com Servidor de Banco de Dados.',(0+48),'Erro')
		Return(.f.)
	Endif


	lcQuery = [Select a.*, b.Rclis from SigCdRst a inner join SigCdCli b on a.Iclis = b.Iclis]	
	With .poDatamgr
		.AddCursor('SigCdRst','skChaves','crSigCdRst','',Thisform.pagina.lista.grade,lcQuery)
		.pfSqlTabela(1).pColuna('Iclis','','','Conta',80,.T.)
		.pfSqlTabela(1).pColuna('Rclis','','','Nome',290,.T.)
		.AddCursor('SigCdRsi','pkChaves','crSigCdRsi','','')
	Endwith
Endwith

Select *, Space(40) As munici, Space(2) As uf From crSigcdRsi  Into Cursor crMuni Readwrite

Select crMuni

With Thisform.pagina.Dados.gradMuni
	.RecordSource = 'crMuni'
	.column1.ControlSource = 'crMuni.uf'
	.Column2.ControlSource = 'crMuni.munici'
	.Refresh
Endwith

Select *, Space(40) As dpros From crSigcdRsi  Into Cursor crProd Readwrite

Select crProd

With Thisform.pagina.Dados.GradProd
	.RecordSource = 'crProd'
	.column1.ControlSource = 'crProd.cpros'
	.Column2.ControlSource = 'crProd.dpros'
	.Refresh
Endwith

= DoDefault()
Thisform.LockScreen = .F.

ENDPROC

------------------------------------------------------------
* PROCEDURE msv_alterar
------------------------------------------------------------
PROCEDURE msv_alterar
llOks = .t.

pChave= crSigCdRst.skChaves

If llOKs
	If (ThisForm.poDataMgr.SQLExecute([Delete From SigCdRst Where skChaves = ?pChave], '') < 1)
		=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (msv_excluir - 1)')
		llOKs = .f.
	EndIf
EndIf

If llOKs
	If (ThisForm.poDataMgr.SQLExecute([Delete From SigCdRsi Where skChaves = ?pChave], '') < 1)
		=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (msv_excluir - 2)')
		llOKs = .f.
	EndIf
EndIf

If Not Thisform.poDataMgr.Update('crSigcdRst')
	=Messagebox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (Update - crSigcdRst )')
	llOks = .F.
Endif

If Not Thisform.poDataMgr.Update('crSigcdRsi')
	=Messagebox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (Update - crSigcdRsi )')
	llOks = .F.
EndIf

If llOks
	ThisForm.poDataMgr.Commit()
Else
	Thisform.poDataMgr.Rollback()
	Return .F.
Endif
ENDPROC

------------------------------------------------------------
* PROCEDURE msv_excluir
------------------------------------------------------------
PROCEDURE msv_excluir
Private pChave, llOK

pChave= crSigCdRst.skChaves
llOK = .t.

Select crSigCdRst
Go top
Delete For skChaves = pChave

Select crSigCdRsi
Go top
Delete For skChaves = pChave

* Excluindo os Registros
If llOK
	If Not ThisForm.poDataMgr.Update('crSigCdRst')
		=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (msv_excluir - 3)')
		llOK = .f.
	EndIf
EndIf
If llOK
	If Not ThisForm.poDataMgr.Update('crSigCdRsi')
		=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (msv_excluir - 4)')
		llOK = .f.
	EndIf
EndIf

If llOK
	ThisForm.poDataMgr.Commit()
Else
	ThisForm.poDataMgr.Rollback()
	Return 0
EndIf
NoDefault
* Fim *********************************************************************************
ENDPROC

	[c.descs as munici, d.estados as uf From SigCdRsi a ] + ;
	[left Join SigCdRst b On a.skChaves = b.skchaves ] +  ;

	[left Join SigCdRst b On a.skChaves = b.skchaves ] +  ;
	[left join sigcdpro c on c.cpros = a.cpros Where a.skchaves = '] + lcSK + [' and a.tipos='P']
