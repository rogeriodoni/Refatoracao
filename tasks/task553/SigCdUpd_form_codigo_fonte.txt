PROCEDURE msv_procurar
Select crSigCdUpd
If ! Empty(ThisForm.Pagina.Dados.GetCodigos.Value)
	If Empty(ThisForm.Pagina.Dados.GetUnidade.Value)
		Locate for Codigos >= ThisForm.Pagina.Dados.GetCodigos.Value
	Else
		Locate for Codigos >= ThisForm.Pagina.Dados.GetCodigos.Value And UniPrdts >= ThisForm.Pagina.Dados.GetUnidade.Value
	Endif
EndIf
If ! Empty(ThisForm.Pagina.Dados.GetUnidade.Value)
	Locate for UniPrdts >= ThisForm.Pagina.Dados.GetUnidade.Value
EndIf
* Fim **************************************************************************************
ENDPROC

------------------------------------------------------------
* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
lcQryUprdt = [Select a.*, b.Descrs From SigCdUpd a Inner Join SigCdGcr b on a.Codigos = b.Codigos]

With ThisForm

	.poDataMgr = CreateObject('fSqlConector', ThisForm.Name)

	If (.poDataMgr.pnIdconn > 0)
		With .poDataMgr
			.AddCursor('SigCdUpd', 'cIdChaves', 'crSigCdUpd', '', ThisForm.Pagina.Lista.Grade, lcQryUprdt)
			.pfSqlTabela(1).pColuna('Codigos',  '', '', 'Fase', 	 080, .t.)
			.pfSqlTabela(1).pColuna('UniPrdts', '', '', 'Unid. Produtiva',    150, .t.)
			.pfSqlTabela(1).pColuna('Descrs', '', '', 'Descrição',    150, .t.)
			.pfSqlTabela(1).pColuna('Ordems', '', '', 'Ordem',    60, .t.)
		EndWith

		=DoDefault()

		ThisForm.Pagina.Lista.Grade.Column1.SetFocus
	Else
		=Messagebox('Impossível Efetuar Conexão Com o Servidor de Banco de Dados...', 0+48, 'CreateObject')
		Return .f.
	EndIf
EndWith
* Fim **************************************************************************************
ENDPROC

* PROCEDURE Pagina.Lista.Grupo_op.Click

PROCEDURE Pagina.Lista.Grupo_op.Click
Lparameters Opcao

If Not DoDefault(Opcao)
	ThisForm.mAtivaPagina1
	Return 0
EndIf

Do Case
	Case InList(Thisform.pcEscolha, 'INSERIR', 'PROCURAR')
		If InList(Thisform.pcEscolha, 'INSERIR')
			Replace cIdChaves With fUniqueIds()
		Endif
		Thisform.Pagina.Dados.getCodigos.Setfocus

	Case (Thisform.pcEscolha = 'ALTERAR')
		Thisform.Pagina.Dados.getCodigos.Setfocus

	Case InList(Thisform.pcEscolha, 'CONSULTAR', 'EXCLUIR')
		Thisform.Pagina.Dados.Grupo_Salva.Salva.Setfocus
EndCase

Select crSigCdUpd
* Fim **************************************************************************************
ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
If Inlist(ThisForm.Pagina.Lista.Grupo_op.Value, 1, 3)

	If Empty(Thisform.Pagina.Dados.getCodigos.Value)
		=Messagebox('Código Inválido!!!', 0+64, '')
		Thisform.Pagina.Dados.getCodigos.Setfocus
		Return 0
	EndIf

	If Inlist(ThisForm.Pagina.Lista.Grupo_op.Value, 1)
		lcQryUprdt = [Select * From SigCdUpd Where Codigos = ']+Thisform.Pagina.Dados.getCodigos.Value+[' And ]+;
												  [UniPrdts = ']+Thisform.Pagina.Dados.GetUnidade.Value+[']
		ThisForm.poDatamgr.SqlExecute(lcQryUprdt,'csSigCdUpd')
		Select csSigCdUpd
		Go Top
		If ! Eof()
			=Messagebox('Unidade Ja Cadastrada!!!', 0+64, '')
			Thisform.Pagina.Dados.getUnidade.Setfocus
			Return 0
		EndIf
	Endif

EndIf

Select crSigCdUpd

=DoDefault()

If InList(ThisForm.pcEscolha,'INSERIR','ALTERAR')
	lnOrdem  = ThisForm.Pagina.Dados.GET_ORDEM.Value
	lcfase   = ThisForm.Pagina.Dados.GetCodigos.Value
	lcUpdate = [Update SigCdUpd Set Ordems = ]+Str(lnOrdem,2)+[ Where Codigos = ']+lcFase+[']
	Thisform.Podatamgr.Sqlexecute(lcUpdate,'')
	ThisForm.Podatamgr.Commit()
EndIf

ThisForm.mAtivaPagina1
* Fim **************************************************************************************
ENDPROC
