PROCEDURE calcsaldo
Update LocalDados Set SaldoDia = 0, SaldoTot = 0, Debitos = 0, Creditos = 0

loFil = ThisForm.pgfPrincipal.pgFiltros
lnSdD = 0
lnSD1 = 0
lnSC1 = 0
lnSD2 = 0
lnSC2 = 0
lnSdT = loFil.getSaldo.Value
lcMsd = loFil.getMoeSaldo.Value
lnOrd = ThisForm.pgfPrincipal.pgDados.cmbOrdem.Value
llPri = .t.

Select * From LocalDados Where 0=1 Into Cursor LocalTotais ReadWrite

Delete From LocalDados Where (Nopers > 9999997) Or (SInis = 1)

Select LocalDados
Go Top In localDados
ldDat = Ttod(LocalDados.DtEmis) 		&& Emissão   
Do Case
	Case (lnOrd = 2) 					&& Vencimento
		ldDat = Ttod(LocalDados.Vencs)					
	Case (lnOrd = 3) 					&& Lançamento
		ldDat = Ttod(LocalDados.Datas)
	Case (lnOrd = 4) 					&& Acerto    
		ldDat = LocalDados.DtBaixa
EndCase

lcMoe = LocalDados.Moedas

If Not Empty(lnSdT) && Inserindo Linha de Saldo Inicial
	If (lcMsd <> lcMoe)
		lnCt1 = fBuscarCotacao(lcMsd, ldDat, ThisForm.poDataMgr)
		lnCt2 = fBuscarCotacao(lcMoe, ldDat, ThisForm.poDataMgr)
		lnSdT = (lnSdT / lnCt1) * lnCt2
	EndIf
	lnSdD = lnSdT
	
	Scatter Memvar
	m.Nopers   = 0
	m.Valors   = 0
	m.DtEmis   = ldDat
	m.Vencs    = ldDat
	m.Datas    = ldDat
	m.DtBaixa  = ldDat
	m.Hists    = [Saldo Inicial :]
	m.Debitos  = 0
	m.Creditos = 0
	m.SaldoDia = lnSdT
	m.SaldoTot = lnSdT
	m.Moedas   = lcMoe
	m.OrdHs    = [1]
	m.Titulos  = []

	Insert Into LocalTotais From Memvar		
	Replace SInis With 1 In LocalTotais
EndIf

Scan
	ldRla = Ttod(LocalDados.DtEmis) 		&& Emissão   
	Do Case
		Case (lnOrd = 2) 					&& Vencimento
			ldRla = Ttod(LocalDados.Vencs)					
		Case (lnOrd = 3) 					&& Lançamento
			ldRla = Ttod(LocalDados.Datas)
		Case (lnOrd = 4) 					&& Acerto    
			ldRla = LocalDados.DtBaixa
	EndCase
			
	If (lcMoe <> LocalDados.Moedas)
		Scatter Memvar
		m.Nopers   = 9999999
		m.Valors   = 0
		m.DtEmis   = ldDat
		m.Vencs    = ldDat
		m.Datas    = ldDat
		m.DtBaixa  = ldDat
		m.Hists    = [Saldo da Moeda ] + lcMoe + [:]
		m.Debitos  = lnSD1
		m.Creditos = lnSC1
		m.SaldoDia = lnSdD
		m.SaldoTot = lnSdT
		m.Moedas   = lcMoe
		m.OrdHs    = Chr(254)
		m.Titulos  = Replicate(Chr(254), 10)
		Insert Into LocalTotais From Memvar
		lnSD1 = 0
		lnSC1 = 0

		If InList(ThisForm.Ordem, [Titulos], [Hists])
			lnSdT = Iif(lcMoe <> LocalDados.Moedas, 0, lnSdT)
			lnSdD = 0
		EndIf
	EndIf

	If Not InList(ThisForm.Ordem, [Titulos], [Hists])
		If ((ldDat <> ldRla) Or (lcMoe <> LocalDados.Moedas)) And (InList(lnOrd, 1, 2, 3, 4))
			Scatter Memvar
			m.Nopers   = 9999998
			m.Valors   = 0

					lcCpp = [SContas]

					lcCpp = [ContAges]
			EndCase

					Case Not Empty(TmpCliente.Grupo) And Not Empty(TmpCliente.Conta)

					Case Empty(TmpCliente.Grupo) And Not Empty(TmpCliente.Conta)

* Preenchendo a Data da Baixa
Select LocalDados
Scan
	If (LocalDados.Pagos = [0])
		lnNop = LocalDados.Nopers

		lcQuery = [Select b.DatarCs ] + ;
				    [From SigCdPit a, SigCdPgr b ] + ;
				   [Where a.Nopers = ] + Str(lnNop, 7) + [ And ] + ;
				   		 [a.EmpDopNums = b.EmpDopNums ] + ;
				  [Order By b.DatarCs Desc]

		If (ThisForm.poDataMgr.SqlExecute(lcQuery, [ConsPgt]) < 1)
			=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (ConsPgt 1)])
			Return .f.
		EndIf
		Go Top In ConsPgt
		If Not Eof([ConsPgt])
			Replace DtBaixa With ConsPgt.DataRcs In LocalDados
		Else
			lcQuery = [Select Datas as DataRcs ] + ;
						[From SigTtCcr ] + ;
					   [Where NOpers = ] + Str(lnNop, 7) + [ ] + ;
					   [Order By Datas Desc]

			If (ThisForm.poDataMgr.SqlExecute(lcQuery, [ConsPgt]) < 1)
				=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (ConsPgt 2)])
				Return .f.
			EndIf
			Go Top In ConsPgt
			If Not Eof([ConsPgt])
				Replace DtBaixa With ConsPgt.DataRcs In LocalDados
			EndIf
		EndIf
	EndIf
EndScan

	Scan
		If (LocalDados.Moedas <> lcMdO)

				Case (loFil.optCotEm.Value = 2) 		&& Vencimento	
					ltDat = LocalDados.Vencs
				Case (loFil.optCotEm.Value = 3) 		&& Lançamento	
					ltDat = LocalDados.Datas
				Case (loFil.optCotEm.Value = 4) 		&& Data Atual	
					ltDat = Datetime()
				Case (loFil.optCotEm.Value = 5)			&& Acerto		
					If Not Empty(LocalDados.DtBaixa)
						ltDat = LocalDados.DtBaixa
					EndIf
			EndCase
			If Empty(ltDat)
				ltDat = LocalDados.Datas
			EndIf
			
			lnCtO = fBuscarCotacao(lcMdO, ltDat, ThisForm.poDataMgr)
			lnCtD = fBuscarCotacao(LocalDados.Moedas, ltDat, ThisForm.poDataMgr)
			lnVlD = (LocalDados.Valors   / lnCtO) * lnCtD
			lnVlO = (LocalDados.ValOcurs / lnCtO) * lnCtD
			lnVlP = (LocalDados.ValPrev  / lnCtO) * lnCtD
			lnVlA = (LocalDados.ValPags  / lnCtO) * lnCtD

			Replace Moedas   With lcMdO, ;
					Valors   With lnVlD, ;
					ValOcurs With lnVlO, ;
					ValPrev  With lnVlP, ;
					ValPags  With lnVlA In LocalDados
		EndIf
	EndScan
EndIf

* PROCEDURE AfterRowColChange
------------------------------------------------------------
PROCEDURE AfterRowColChange
Lparameters nColIndex

loDad = ThisForm.pgfPrincipal.pgDados

llTot = (LocalDados.Nopers > 9999997) Or (LocalDados.SInis = 1)

loDad.getDContEms.Value  = []
loDad.getDContas.Value   = []
loDad.getDSContas.Value  = []
loDad.getDContAges.Value = []

If llTot
	loDad.getNFs.Value 		  = []
	loDad.getTitBans.Value 	  = []
	loDad.getEspecieNfs.Value = []
	loDad.getHist2s.Value 	  = []
	loDad.getEmps.Value 	  = []
	loDad.getValOcurs.Value   = []
	loDad.getGrupEms.Value 	  = []
	loDad.getContEms.Value 	  = []
	loDad.getGrupos.Value 	  = []
	loDad.getContas.Value 	  = []
	loDad.getSGrupos.Value 	  = []
	loDad.getSContas.Value 	  = []
	loDad.getGrupAges.Value   = []
	loDad.getContAges.Value   = []
EndIf

loDad.Botoes.Botao1.Value = Iif(Not llTot And LocalDados.TitCancs = 1, [ X], [])
loDad.Botoes.Botao2.Value = Iif(Not llTot And LocalDados.Pagos=[0], [ X], [])
loDad.Botoes.Botao3.Value = Iif(Not llTot And LocalDados.Concs, [ X], [])
loDad.Botoes.Botao4.Value = Iif(Not llTot And Not Empty(LocalDados.Prorrogs), [ X], [])

If Not llTot 
	loDad.getNFs.Value 		  = LocalDados.NFs
	loDad.getTitBans.Value 	  = LocalDados.TitBans
	loDad.getEspecieNfs.Value = LocalDados.EspecieNfs
	loDad.getHist2s.Value 	  = LocalDados.Hist2s
	loDad.getEmps.Value 	  = LocalDados.Emps
	loDad.getValOcurs.Value   = LocalDados.ValOcurs
	loDad.getGrupEms.Value 	  = LocalDados.GrupEms
	loDad.getContEms.Value 	  = LocalDados.ContEms
	loDad.getGrupos.Value 	  = LocalDados.Grupos
	loDad.getContas.Value 	  = LocalDados.Contas
	loDad.getSGrupos.Value 	  = LocalDados.SGrupos
	loDad.getSContas.Value 	  = LocalDados.SContas
	loDad.getGrupAges.Value   = LocalDados.GrupAges
	loDad.getContAges.Value   = LocalDados.ContAges

	* Carregando as Descrições das Contas
	For I = 1 To 4
		lcCta = Iif(I=1, [ContEms], Iif(I=2, [Contas], Iif(I=3, [SContas], [ContAges])))
		
		If Not Empty(LocalDados.&lcCta.)
			If Not Seek(LocalDados.&lcCta., [TmpContas], [IClis])
				lcQuery = [Select IClis, RClis ] + ;
						    [From SigCdCli ] + ;
						   [Where IClis = '] + LocalDados.&lcCta. + [']

				If (ThisForm.poDataMgr.SqlExecute(lcQuery, [LocalCli]) < 1)
					=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (LocalCli)])
					Return .f.
				EndIf
				Go Top In LocalCli
				Insert Into TmpContas (IClis, RClis) Values (LocalCli.IClis, LocalCli.RClis)
			EndIf
			loDad.getD&lcCta..Value = TmpContas.RClis
		EndIf
		loDad.getD&lcCta..Refresh
	Next
EndIf

loDad.getNFs.Refresh
loDad.getTitBans.Refresh
loDad.getEspecieNfs.Refresh
loDad.getHist2s.Refresh
loDad.getEmps.Refresh
loDad.getValOcurs.Refresh
loDad.getGrupEms.Refresh
loDad.getContEms.Refresh
loDad.getGrupos.Refresh
loDad.getContas.Refresh
loDad.getSGrupos.Refresh
loDad.getSContas.Refresh
loDad.getGrupAges.Refresh
loDad.getContAges.Refresh
loDad.Botoes.Botao1.Refresh
loDad.Botoes.Botao2.Refresh
loDad.Botoes.Botao3.Refresh
loDad.Botoes.Botao4.Refresh
* Fim **************************************************************************************
ENDPROC

------------------------------------------------------------
* PROCEDURE InteractiveChange
------------------------------------------------------------
PROCEDURE InteractiveChange
Do Case
	Case (This.Value = 1) && Emissão
		ThisForm.Ordem = [DtEmis]
	Case (This.Value = 2) && Vencimento
		ThisForm.Ordem = [Vencs]
	Case (This.Value = 3) && Lançamento
		ThisForm.Ordem = [Datas]
	Case (This.Value = 4) && Baixa
		ThisForm.Ordem = [DtBaixa]
	Case (This.Value = 5) && Titulo
		ThisForm.Ordem = [Titulos]
	Case (This.Value = 6) && Historico
		ThisForm.Ordem = [Hists]
	Case (This.Value = 7) && Emitente
		ThisForm.Ordem = [ContEms]
	Case (This.Value = 8) && Portador
		ThisForm.Ordem = [Contas]
EndCase

ThisForm.OrdenaGrade

ThisForm.pgfPrincipal.pgDados.grdDados1.SetFocus
* Fim **************************************************************************************
ENDPROC
