PROCEDURE msv_procurar
Local lcBusca

If Not Empty(ThisForm.Pagina.Dados.Get_Codigo.Value)
	ThisForm.poDataMgr.LocateCursor([crSigCdBlq], [Codigos], ThisForm.Pagina.Dados.Get_Codigo.Value)
EndIf
* Fim **************************************************************************************
ENDPROC

------------------------------------------------------------
* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
With ThisForm
	.poDataMgr = CreateObject([fSqlConector], .Name)

	If (.poDataMgr.pnIdconn > 0)
		With .poDataMgr
			.AddCursor([SigCdBlq], [cIdChaves], [crSigCdBlq], [], ThisForm.Pagina.Lista.Grade)
			.pfSqlTabela(1).pColuna([codigos],  [], [], [Codigo],    080, .t.)
			.pfSqlTabela(1).pColuna([dtInicial],  [], [], [Data Inicial],   074, .t.)
			.pfSqlTabela(1).pColuna([dtFinal], [], [], [Data Final], 074, .t.)
			.pfSqlTabela(1).pColuna([UsuIncs],  [], [], [Usuario],   080, .t.)
			.pfSqlTabela(1).pColuna([DtIncs],  [], [], [Data],   080, .t.)
			.pfSqlTabela(1).pColuna([UsuAlts],  [], [], [Alteração],   080, .t.)
			.pfSqlTabela(1).pColuna([DtAlts],  [], [], [Data],   080, .t.)
			
		EndWith

		=DoDefault()

		.Pagina.Lista.Grade.Column1.SetFocus
	Else
		=Messagebox([Impossível Efetuar Conexão Com o Servidor de Banco de Dados...], 0+48, [CreateObject])
		Return .f.
	EndIf
EndWith
* Fim **************************************************************************************
ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Lista.Grupo_op.Click
------------------------------------------------------------
PROCEDURE Pagina.Lista.Grupo_op.Click
Parameters lnOpcao

If Not DoDefault(lnOpcao)
	ThisForm.mAtivaPagina1
	Return .f.
EndIf

Do Case
	Case ThisForm.pcEscolha = 'INSERIR'
	
		Replace Codigos With fGerUniqueKey('BLOQUEIO'),;
				CidChaves With fUniqueIds() ,;
				UsuIncs With Usuar ,;
				DtIncs With Datetime() In CrSigCdBlq
				
		This.Parent.Parent.Dados.get_codigo.Refresh		
		This.Parent.Parent.Dados.GetDtinicial.SetFocus
		 
	Case InList(ThisForm.pcEscolha, [PROCURAR])
		 ThisForm.Pagina.Dados.codigos.SetFocus
		 
	Case InList(ThisForm.pcEscolha, [ALTERAR])
		 ThisForm.Pagina.Dados.GetDtinicial.SetFocus

	Otherwise
		 ThisForm.Pagina.Dados.Grupo_Salva.Salva.SetFocus
EndCase

* Fim **************************************************************************************
ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
If InList(ThisForm.pcEscolha, [INSERIR], [ALTERAR])
	If Empty(ThisForm.Pagina.Dados.Get_Codigo.Value)
		=Messagebox([Código Inválido!!!], 48, [Atenção!!!])
		ThisForm.Pagina.Dados.Get_Codigo.SetFocus()
		Return .f.
	EndIf

	If Empty(ThisForm.Pagina.Dados.getDtInicial.Value)
		=Messagebox([Data Inicial Inválida!!!], 48, [Atenção!!!])
		ThisForm.Pagina.Dados.getDtInicial.SetFocus()
		Return .f.
	EndIf
	
	If Empty(ThisForm.Pagina.Dados.getDtFinal.Value)
		=Messagebox([Data Final Inválida!!!], 48, [Atenção!!!])
		ThisForm.Pagina.Dados.getDtFinal.SetFocus()
		Return .f.
	EndIf
	
	If ThisForm.Pagina.Dados.getDtInicial.Value > ThisForm.Pagina.Dados.getDtFinal.Value
		=Messagebox([Data Inicial Maior que a Data Final!!!], 48, [Atenção!!!])
		ThisForm.Pagina.Dados.getDtInicial.SetFocus()
		Return .f.
	EndIf 
	
EndIf

If ThisForm.pcEscolha = [ALTERAR]
	Replace UsuAlts With Usuar ,;
			DtAlts With Datetime() In CrSigCdBlq
EndIf 

=DoDefault()

ThisForm.mAtivaPagina1
* Fim **************************************************************************************
ENDPROC
