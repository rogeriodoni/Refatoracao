* PROCEDURE gerareg2
------------------------------------------------------------
PROCEDURE gerareg2
Local lnAliIcm, lnUltVlrCompra

Select crReg2

Select cPros, Max(cGrus) as cGrus, Sum(Sqtds) as Sqtds, Max(dPros) as dPros, Max(cUnis) as cUnis, Max(ClFiscals) as ClFiscals ;
	 from CsTemporario Where Ps = 1 AND Ps2 = 1 Having Sum(Sqtds) > 0 Group by cPros Into cursor crProds

ThisForm.TotalImposto = 0

pDtSaldo = fDtoSQL(ThisForm.getDtEst.Value,[23:59:59]) 

Select crProds
Scan
	lnAliIcm = 0
	ThisForm.poDataMgr.CursorQuery([SigCdPro],[LocalPro],[cPros],crProds.cPros,[Valors,Icms,CustoFS])	
	ThisForm.poDataMgr.CursorQuery([SigCdClf],[LocalClf],[Codigos],crProds.ClFiscals,[IndIVAST,TpSubSt,TipoMercs])
	
	** Query para pegar o valor de compra da última entrada realizada
	lnUltVlrCompra = LocalPro.CustoFS
	lcQuery = [Select top 1 Units From SigMvItn a Inner Join SigCdOpe o on o.Dopes = a.Dopes ] + ;
				[Inner Join SigCdTom t on t.Codigos = o.TipoOps ] + ;
				[Where a.cPros = ']+crProds.cPros+[' and t.GerGdmis = 2 ] + ;
				[And a.DtAlts < ?pDtSaldo Order by a.DtAlts Desc]
	If ThisForm.poDataMgr.SQLExecute(lcQuery,[crUltVlr]) < 1 
		=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (crUltVlr)')
		Return .f.
	Endif				
	If Used([crUltVlr]) and not Eof([crUltVlr])
		lnUltVlrCompra = ROUND(crUltVlr.Units,2)
	Endif

	** Trazer a alíquota do produto
	Declare laCampo(2,2)
	laCampo(1,1) = [Origems]
	laCampo(1,2) = [SP]
	laCampo(2,1) = [Destinos]
	laCampo(2,2) = [SP]
	If ThisForm.poDataMgr.CursorQuery([SigPrTri],[crSigPrTri],@laCampo) 
		lnAliIcm = crSigPrTri.Aliquotas
	Endif	
	IF lnAliIcm = 0
		laCampo(1,1) = [Codigos]
		laCampo(1,2) = [SP]
		laCampo(2,1) = [Destinos]
		laCampo(2,2) = [SP]		
		If ThisForm.poDataMgr.CursorQuery([SigCdUfd],[crSigCdUfd],@laCampo)
			lnAliIcm = crSigCdUfd.Aliqs
		ENDIF
		IF lnAliIcm = 0
			laCampo(1,1) = [Grupos]
			laCampo(1,2) = crProds.cGrus
			laCampo(2,1) = [Ests]
			laCampo(2,2) = [SP]
			If ThisForm.poDataMgr.CursorQuery([SigCdcFi],[crSigCdcFi],@laCampo)
				lnAliIcm = crSigCdcFi.Icms
			ENDIF
			IF lnAliIcm = 0
				If( LocalPro.Icms > 0 )
					lnAliIcm = LocalPro.Icms
				Else
					If ThisForm.poDataMgr.CursorQuery([SigCdUfs],[crSigCdUfs],[Estados],[SP])
						lnAliIcm = crSigCdUfs.AIcms
					Endif
				Endif				
			Endif
		Endif			
	Endif
	
	lnIVAST    = LocalClf.IndIVAST
	lnPrecoFin = LocalPro.Valors
	If LocalClf.TpSubSt == 2
		lnIVAST = 0
	Else
		lnPrecoFin = 0		
	EndIf

	** Cálculo do Imposto	
	lnVlrImp = 0
	If( LocalClf.TpSubSt == 2 )
		If( crSigCdEmp.RegTribs = [01] )
			lnVlrImp = (crProds.Sqtds*lnUltVlrCompra) * (lnAliIcm/100)
		EndIf
		If( crSigCdEmp.RegTribs = [04] )
			lnVlrImp = ((crProds.Sqtds*LocalPro.Valors)-(crProds.Sqtds*lnUltVlrCompra)) * (lnAliIcm/100)
		Endif
	Else
		If( crSigCdEmp.RegTribs = [01] )
			lnVlrImp = ((crProds.Sqtds*lnUltVlrCompra) * (lnAliIcm/100)) + ((crProds.Sqtds*lnUltVlrCompra) * (lnIVAST/100) * (lnAliIcm/100))
		EndIf
		If( crSigCdEmp.RegTribs = [04] )
			lnVlrImp = ((crProds.Sqtds*lnUltVlrCompra) * (lnIVAST/100) * (lnAliIcm/100))
		EndIf	
	Endif

	ThisForm.TotalImposto = ThisForm.TotalImposto + Round(lnVlrImp,2)
		
	Append Blank in crReg2
	Replace TipoReg		With [2],;
			CodMerc		With crProds.cPros,;			
			Qtde		With Str(crProds.Sqtds,12,3),;
			VlrUnit		With Str(lnUltVlrCompra,12,2),;
			VlrTot		With Str(crProds.Sqtds*lnUltVlrCompra,12,2),;
			AliqIcms	With Str(lnAliIcm,5,2),;	
			IVASt		With Str(lnIVAST,7,2),;
			PrecoFin	With Str(lnPrecoFin,12,2),;	
			ImpDev		With Str(lnVlrImp,12,2),;
			CodTpMerc	With Transform(LocalClf.TipoMercs,'@L 99'),;
			DescMerc	With fLimpaTexto(crProds.dPros),;
			Unid		With crProds.cUnis in crReg2
EndScan

** Acerto da quantidade de registro do Registro 1
Select crReg2
Locate

Replace QtdRegT2 with Str(Reccount([crReg2]),5) in crReg1

Return
ENDPROC

* PROCEDURE Init

PROCEDURE Init
Lparameters lcEmp

With ThisForm
	.poDataMgr = CreateObject('fSqlConector', .Name)

	If (.poDataMgr.pnIdconn > 0)
	
		If Not .poDataMgr.CursorQuery([SigCdPac], [crSigCdPac], .F., .F., [DirCat44])
			=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (crSigCdPac)')
			Return .f.
		EndIf

		lcEmp = Iif(Type([lcEmp]) == 'C' and not Empty(lcEmp),lcEmp,_Empr)
		=Seek(lcEmp,[crSigCdEmp],[cEmps])
		
		.TxtDest.Value = [C:\]

		If Not Empty(crSigCdPac.DirCat44)
			.TxtDest.Value = crSigCdPac.DirCat44
		EndIf

		.GetDtEst.Value = Date()
		.GetNumParc.Value = 1
	
		=DoDefault()
	Else
		=Messagebox('Impossível Efetuar Conexão Com o Servidor de Banco de Dados...', 0+48, 'CreateObject')
		Return .f.
	EndIf
EndWith
* Fim **************************************************************************************
ENDPROC

PROCEDURE Click
Local lcArqDest, ldDtIni, ldDtFim

If Empty(ThisForm.Txtdest.Value)
	=MessageBox('Destino de Arquivo Inválido!!!' + Chr(13) + ;
				'Verifique a Configuração da Empresa.', 48, 'Destino Inválido...')

	Return .f.
EndIf

	=MessageBox('Data do Estoque não informada!',48,'Data do Estoque inválida...')

	=MessageBox('Finalidade do arquivo não informada!',48,'Finalidade inválida...')

	Return(.F.)
Endif

	=MessageBox('Operação Cancelada Pelo Usuário!!!', 64, '')
	Return
EndIf

Set Point To [,]

** Gera registro Tipo 1
ThisForm.GeraReg1()

** Gera registros Tipo 2
ThisForm.GeraReg2()

** Gera registros Tipo 3
ThisForm.GeraReg3()

lcNomFile = [CAT44]+Strtran(Dtoc(ThisForm.GetDtEst.Value),[/],[])+[.TXT]

nHandle = Fcreate(xNmArq)

lnQtCampos1 = Fcount([crReg1])
lnQtCampos2 = Fcount([crReg2])
lnQtCampos3 = Fcount([crReg3])

Select crReg1
m.lcLinha = []
For i = 1 To lnQtCampos1
	lcCampo = Fields(i)
	m.lcLinha = m.lcLinha + Iif(not Empty(m.lcLinha),[|],[]) + Alltrim(&lcCampo)
Next i

Select crReg2
Scan
	m.lcLinha = []
	For i = 1 To lnQtCampos2
		lcCampo = Fields(i)
		m.lcLinha = m.lcLinha + Iif(not Empty(m.lcLinha),[|],[]) + Alltrim(&lcCampo)

Select crReg3

	lcCampo = Fields(i)
	m.lcLinha = m.lcLinha + Iif(not Empty(m.lcLinha),[|],[]) + Alltrim(&lcCampo)

* Fim **************************************************************************************
