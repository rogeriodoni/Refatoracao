PROCEDURE Init
LParameters _Tipo, _Titulo, _Titulo2, _Documento, _Data, _Empresa, _Fpagto, _Valor

If Type("_Documento")#"N"
	_Documento = 0
Endif
If Type("_Data")#"D"
	_Data = Date()
Endif
If Type("_Empresa")#"C"
	_Empresa = ' '
Endif
If Type("_FPagto")#"C"
	_FPagto = ' '
Endif
If Type("_Valor")#"N"
	_Valor = 0
Endif

ThisForm.zsen    = .F.
ThisForm.NroDoc  = _documento
ThisForm.DatDoc  = _Data
ThisForm.Tipo    = _Tipo
Thisform.usulibs = Usuar
ThisForm.Financ  = .f.
ThisForm.CancItem= .f.
ThisForm.Libfpag = .f.
ThisForm.Libfpag2= .f.
ThisForm.Empresa = _Empresa
ThisForm.Fpagto  = _Fpagto
ThisForm.ValFpag = _Valor
ThisForm.LibOpe  = .f.
Thisform.LibProducao  = .f.
ThisForm.Saldoestoque = .f.
ThisForm.LibValida    = .f.
ThisForm.LiberaTitulo = .f.
ThisForm.LibQtMin 	  = .f.
ThisForm.LibLimitador = .f.
ThisForm.MovEstoque   = .f.

Thisform.poDatamgr = CreateObject('fSqlConector','SigOpSen')

ThisForm.Podatamgr.Cursorquery('SigCdPam','CrParam',.f.,.f.,[LiblCreds])
ThisForm.Podatamgr.Cursorquery('SigCdPac','CrParac',.f.,.f.,[SenhDEmps])

Do Case
	Case _Tipo = 'DTENTREGA'
		This.Caption = 'Prazo de Entrega'
		This.Frase.Caption = Alltrim( _titulo ) + ' fora do prazo de entrega!'
		ThisForm.Financ = .f.

	Case _Tipo = 'CHDEVOLVIDO'
		This.Caption = 'Cheque Devolvido'
		This.Frase.Caption = 'A conta ' + Alltrim( _Titulo2 ) + ' possui '
		If _Titulo = 1
			This.Frase.Caption = This.Frase.Caption + '1 Cheque Devolvido Pendente!'
		Else
			This.Frase.Caption = This.Frase.Caption + Alltrim( Str( _Titulo ) ) + ' Cheques Devolvidos Pendentes!'
		Endif
		ThisForm.Financ = .t.

	Case _Tipo = 'LIMITE'
		This.Caption = 'Limite Excedido'
		This.Frase.Caption = 'O Limite da conta ' + Alltrim( _Titulo ) + ' foi excedido! '
		ThisForm.Financ = .t.

	Case _Tipo = 'ALTERAIMPOSTO'
		This.Caption = Alltrim( _Titulo )
		This.Frase.Caption = Alltrim( _Titulo2 )
		ThisForm.Financ = .t.

	Case _Tipo = 'CANCITEM'
		This.Caption = Alltrim( _Titulo )
		This.Frase.Caption = Alltrim( _Titulo2 )
		ThisForm.CancItem = .t.

	Case _Tipo = 'LIBFPAG'
		This.Caption = Alltrim( _Titulo )
		This.Frase.Caption = Alltrim( _Titulo2 )
		ThisForm.LibFpag = .t.

	Case _Tipo = 'LIBPRDZ' Or _Tipo = 'LIBPRDZ2' Or _Tipo = 'LIBPESO'
		This.Caption = Alltrim( _Titulo )
		This.Frase.Caption = Alltrim( _Titulo2 )
		ThisForm.LibProducao = .t.

	Case _Tipo = 'SDINSUF'
		This.Caption = Alltrim( _Titulo )
		This.Frase.Caption = Alltrim( _Titulo2 )
		ThisForm.Saldoestoque = .t.

	Case _Tipo = 'MAXIMO'
		This.Caption = 'Valor Máximo Excedido :'+_Titulo2
		This.Frase.Caption = 'Valor da Condição de Pagamento ' + Alltrim( _Titulo ) + ' foi excedido! '
		ThisForm.Financ = .t.

	Case _Tipo = 'DTVALIDA'
		This.Caption = 'Validade da Movimentação'
		This.Frase.Caption = Alltrim( _titulo ) + ' fora da Validade!!!'
		ThisForm.LibValida = .t.

	Case _Tipo = 'LIBTITULO'
		This.Caption = 'Liberar Título '
		This.Frase.Caption = Alltrim( _titulo )
		ThisForm.LiberaTitulo = .t.

	Case _Tipo = 'LIBERAFPG'
		This.Caption = Alltrim( _Titulo )
		This.Frase.Caption = Alltrim( _Titulo2 )
		ThisForm.LibFpag2 = .t.

	Case _Tipo = 'LIBOPE'
		This.Caption = Alltrim( _Titulo )
		This.Frase.Caption = Alltrim( _Titulo2 )
		ThisForm.LibOpe = .t.
		
	Case _Tipo = 'LIBQTMIN'
		This.Caption = Alltrim( _Titulo )
		This.Frase.Caption = Alltrim( _Titulo2 )
		ThisForm.LibQtMin = .T. 		

	Case _Tipo = 'LIBLIMITADOR'
		This.Caption = Alltrim( _Titulo )
		This.Frase.Caption = Alltrim( _Titulo2 )
		ThisForm.LibLimitador = .T.

	Case _Tipo = 'MOVESTOQUE'
		This.Caption = Alltrim( _Titulo )
		This.Frase.Caption = Alltrim( _Titulo )
		ThisForm.MovEstoque = .T.

	Case _Tipo = 'VALIDACESSO'
		This.Caption = Alltrim( _Titulo )
		This.Frase.Caption = Alltrim( _Titulo )
		ThisForm.MovEstoque = .T.
Endcase

ThisForm.cntSombra.lblSombra.Caption = ThisForm.Caption
ThisForm.cntSombra.lblTitulo.Caption = ThisForm.Caption

This.Refresh
This.Frase.Refresh 

If ( crParam.LiblCreds And _Tipo = 'LIMITE' ) Or _Tipo = 'LIBPRDZ2' Or _Tipo = 'MAXIMO'
	ThisForm.usuario.Value = Usuar
	ThisForm.Say3.Caption  = 'Senha Do Dia : '
	ThisForm.Say3.Left = 142
EndIf

ThisForm.Top  = ThisForm.Top  + 5
ThisForm.Left = ThisForm.Left + 5

ENDPROC

PROCEDURE Valid
local lcod,lfan,codcor,xobj,totite

usual=''
sd = .t.
If IsEmpty( This.Value )
	zsen = .f.
	Thisform.zsen=.F.
	Thisform.release
	Return .t.
EndIf
Do Case
	Case ThisForm.Cancitem
		lcQuery = [Select a.* From SigCdUsu a, SigCdCrg b ]+;
				  [Where Not a.cAtivos='N' And a.Ccargs = b.ccargs And b.CancItens = "S" and not a.Senhas = ']+Space(6)+[']

	Case ThisForm.Financ
		lcQuery = [Select a.* From SigCdUsu a, SigCdCrg b ]+;
				  [Where Not a.cAtivos='N' And a.Ccargs = b.ccargs And b.limites = "S" and not a.Senhas = ']+Space(6)+[']

	Case ThisForm.Libfpag 
		lcQuery = [Select a.* From SigCdUsu a, SigCdCrg b ]+;
				  [Where Not a.cAtivos='N' And a.Ccargs = b.ccargs And b.LibFpags = "S" and not a.Senhas = ']+Space(6)+[']

	Case ThisForm.Libfpag2
		lcQuery = [Select a.* From SigCdUsu a, SigCdCrg b ]+;
				  [Where Not a.cAtivos='N' And a.Ccargs = b.ccargs And b.LibFpgs = "S" and not a.Senhas = ']+Space(6)+[']

	Case Thisform.LibProducao 
		If Thisform.Tipo = 'LIBPESO'
			lcQuery = [Select a.* From SigCdUsu a ]+;
					  [Where Not a.cAtivos='N' And a.LibPess = "S" and not a.Senhas = ']+Space(6)+[']
		Else
			lcQuery = [Select a.* From SigCdUsu a ]+;
					  [Where Not a.cAtivos='N' And a.LibProds = "S" and not a.Senhas = ']+Space(6)+[']
		Endif

	Case ThisForm.Saldoestoque 
		lcQuery = [Select a.* From SigCdUsu a, SigCdCrg b ]+;
				  [Where Not a.cAtivos='N' And a.Ccargs = b.ccargs And b.LibSdIns = "S" and not a.Senhas = ']+Space(6)+[']

	Case Thisform.LibValida
		lcQuery = [Select a.* From SigCdUsu a ]+;
				  [Where Not a.cAtivos='N' And a.LibvOpes = 'S' And not a.Senhas = ']+Space(6)+[']

	Case ThisForm.LibOpe
		lcQuery = [Select a.* From SigCdUsu a, SigCdCrg b ]+;
				  [Where Not a.cAtivos='N' And a.Ccargs = b.ccargs And b.LibOpes = "S" and not a.Senhas = ']+Space(6)+[']
				  
	Case ThisForm.LibQtMin
		lcQuery = [Select a.* From SigCdUsu a ]+;
				  [Where Not a.cAtivos='N' And a.LibQtMins = "S" and not a.Senhas = ']+Space(6)+[']

	Case ThisForm.LibLimitador
		lcQuery = [Select a.* From SigCdUsu a ]+;
				  [Where Not a.cAtivos='N' And a.LibLmPss = "S" and not a.Senhas = ']+Space(6)+[']

	Case ThisForm.movestoque
		lcQuery = [Select a.* From SigCdUsu a ]+;
				  [Where Not a.cAtivos='N' And not a.Senhas = ']+Space(6)+[']

	Otherwise
		lcQuery = [Select a.* From SigCdUsu a, SigCdCrg b ]+;
				  [Where Not a.cAtivos='N' And a.Ccargs = b.ccargs And b.AltCots = "S" and not a.Senhas = ']+Space(6)+[']
EndCase

If (ThisForm.poDataMgr.SqlExecute(lcQuery, 'TmpUsu') < 1)
	=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (TmpUsu)')
	Return 0
EndIf

Select TmpUsu
index on usuarios tag usuarios
if !IsEmpty(this.value)
	select tmpusu
	set order to usuarios
	set near on
	if not seek(this.value)
		xBrow=CreateObject('fwBuscaSel','tmpusu','usuarios',this.value,'Seleção')
		xBrowCol=xBrow.AddColuna('usuarios','','Código',.t.,'usuarios')
		xBrowCol=xBrow.AddColuna('ccargs','','cargo',.F.)
		xBrow.Show
	endif
	set near off
	select tmpusu
	this.value = evaluate('usuarios')
	this.refresh
	usual=this.value
endif
thisform.usulibs = this.value
If IsEmpty(usual)
	thisform.zsen=.F.
	thisform.release
Else
	If lastkey()<>27
		usual = upper(usual)
		select TmpUsu
		set order to usuarios
		seek usual
		zsen = fCriptografar(TmpUsu.senhas)
		If IsEmpty(TmpUsu.senhas)
			zsen = .t.
			thisform.say3.visible=.f.
			thisform.senha.visible=.f.
			thisform.zsen=.t.
			thisform.release
		else
			thisform.say3.visible=.t.
			thisform.senha.visible=.t.
		endif
	else
		zsen = .f.
	endif
EndIf

ENDPROC

PROCEDURE Valid
local senha,resp,lcQuery

senha = Chr(254)
lcQuery = [select a.usuarios, a.senhas from SigCdUsu a where Not a.cAtivos='N' and a.usuarios='] + ThisForm.Usuario.Value + [' order by 1]
ThisForm.poDataMgr.SQLExecute(lcQuery, 'TmpUsu')

If Used('TmpUsu')
	senha=fCriptografar(TmpUsu.senhas)
EndIf 

Thisform.zsen=.F.

If ( Not crparam.LiblCreds Or (ThisForm.Tipo # 'LIMITE' And ThisForm.tipo # 'LIBPRDZ2' ) ) And ThisForm.tipo # 'MAXIMO'

	If IsEmpty(This.Value) Or Padr(this.value,10)#Padr(senha,10)
		if messagebox('Deseja digitar novamente ? (S/N)',4+256,'Senha Conflitante...' ) = 7
			thisform.zsen=.F.
			thisform.release
		else
			return 0
		endif
	Else
		If ThisForm.Tipo = 'LIBPRDZ' Or ThisForm.Tipo = 'LIBPRDZ2' Or ThisForm.Tipo = 'LIBPESO' Or ThisForm.Tipo = 'LIBQTMIN'
		Else
			If Thisform.Tipo = 'LIBTITULO'
				=fGravarLog('T', 'SigPgTit', ThisForm.Tipo, ThisForm.Frase.Caption, Usuar )
			Else
				If Used('TprMvCab')
					=fGravarLog('T', TprMvCab.Dopes, Thisform.Tipo, TprMvCab.Emps + '-' + ;
									AllTrim(TprMvCab.Dopes) + '-' + AllTrim(Str(TprMvCab.Numes,6)) + ;
									Iif(ThisForm.Tipo = 'LIBERAFPG','('+Alltrim(Thisform.Frase.Caption)+')',' '),Thisform.UsuLibs)
				Endif
			Endif
		Endif
		Thisform.zsen=.T.
		Thisform.release
	EndIf
Else
	lnNum = 0
	If CrParac.SenhDEmps
		lcSql = [Select nemps From SigCdEmp Where CEmps = ']+ThisForm.Empresa+[']
		ThisForm.Podatamgr.Sqlexecute(lcSql,'CrEmp')
		lnNum = CrEmp.nEmps

		lcSql = [Select nfpags From SigOpFp Where Fpags = ']+ThisForm.Fpagto+[']
		ThisForm.Podatamgr.Sqlexecute(lcSql,'CrfPagto')
		lnNum = lnNum + CrFPagto.NfPags + ( ThisForm.Valfpag * 100 )

	Endif
	lcSenha = fGerPswDiario( ThisForm.NroDoc, ThisForm.DatDoc, lnNum )
	
	If alltrim(This.Value) # lcSenha
		if messagebox('Deseja digitar novamente ? (S/N)',4+256,'Senha Conflitante...' ) = 7
			thisform.zsen=.F.
			thisform.release
		else
			return 0
		endif
	Else
		Thisform.zsen=.T.
		Thisform.usulibs = lcSenha
		Thisform.release
	EndIf

Endif
ENDPROC
