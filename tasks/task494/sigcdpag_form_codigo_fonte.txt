------------------------------------------------------------
PROCEDURE msv_alterar
Local lnErro1

m.lnErro1 = ThisForm.poDatamgr.pnIdConn

With ThisForm
	If m.lnErro1>0 
		m.lnErro1 = Iif(.poDataMgr.UpDate('crSigOpFp'),1,-1)
	EndIf
	If m.lnErro1>0 
		m.lnErro1 = Iif(.poDataMgr.UpDate('crSigOpFpI'),1,-1)
	EndIf 
	If m.lnErro1>0 
		m.lnErro1 = Iif(.poDataMgr.UpDate('crSigOpFpP'),1,-1)
	EndIf 
	If m.lnErro1>0 
		m.lnErro1 = .poDataMgr.Commit()
	EndIf 
	If Not m.lnErro1>0 
		.poDataMgr.RollBack()
	EndIf 
EndWith 
If Not m.lnErro1>0 
	= MessageBox('Erro na Alteração dos Dados do Cadastro de Operações.',48)
EndIf 

ENDPROC

		m.lnErro1 = .poDataMgr.Commit()
	EndIf 
	If Not m.lnErro1>0 
		.poDataMgr.RollBack()
	EndIf 
EndWith

	= MessageBox('Erro na Alteração dos Dados do Cadastro de Operações.',48)
EndIf 

ENDPROC

* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
Local lcMensa 

Select crSigOpFp

lcMensa = []

If Inlist( ThisForm.pcescolha, 'INSERIR', 'ALTERAR')
	If !Empty(crSigOpFp.Bancos)  .or. !Empty(crSigOpFp.Agencias) .or. !Empty(crSigOpFp.NContas)
		Do Case
			Case Empty(crSigOpFp.Bancos)
				 Messagebox( 'Banco Inválido ! ',0+48,'')
				 ThisForm.Pagina.Dados.Dados.Page1.Get_bancos.SetFocus
				 Return
			Case Empty(crSigOpFp.Agencias)
				 Messagebox( 'Agência Inválida ! ',0+48,'')
				 Thisform.Pagina.Dados.Dados.Page1.Get_Agencias.SetFocus
				 Return
			Case Empty(crSigOpFp.NContas)
				 Messagebox( 'Número da Conta Inválido ! ',0+48,'')
				 Thisform.Pagina.Dados.Dados.Page1.Get_NContas.SetFocus
				 Return
		EndCase
	Endif

	If (crSigOpFp.valpends <> 2) And (Not Empty(crSigOpFp.Grupods) Or Not Empty(crSigOpFp.Contads)) And (crSigOpFp.OriDes <> 4)
		 =Messagebox([Condição Configurada Para Valor Pendente Não] + Chr(13) + ;
		 			 [Pode Ter Grupo/Conta de Depósito Preenchidos], 0+16, [Atenção!!!])
		 Thisform.Pagina.Dados.Dados.ActivePage = 1
		 ThisForm.Pagina.Dados.get_Grupo.SetFocus
		 Return .f.
	EndIf
	
	If (crSigOpFp.EmiChqs # 1) .and. (!Empty(crSigOpFp.Bancos)  .or. ;
		!Empty(crSigOpFp.Agencias) .or. !Empty(crSigOpFp.NContas))
		Replace Bancos With Space(3), Agencias With Space(4), NContas With Space(10)
	Endif

	If (crSigOpFp.Infos # 'C') .and. (crSigOpFp.EmiChqs = 1)
		Do Case
			Case crSigOpFp.Infos = 'T'
				lcMensa = 'Cartão de Crédito'
			Case crSigOpFp.Infos = 'N'
				lcMensa = 'Carnê'
			Case crSigOpFp.Infos = 'O'
				lcMensa = 'Outras Formas'
			Case crSigOpFp.Infos = 'R'
				lcMensa = 'Conta Corrente'
			Otherwise
				lcMensa = ["] + crSigOpFp.Infos + ["]
		EndCase
		lcMensa = 'Não Se Pode Emitir Cheque a Partir de ' + Alltrim(lcMensa) + '!'
		Messagebox(lcMensa, 0+48, '')
		Thisform.Pagina.Dados.Dados.Page1.Get_Info.SetFocus
		Return
	EndIf
	If crSigOpFp.valpends <> 1 And (Empty(crSigOpFp.GrupoDs) Or (Empty(crSigOpFp.ContaDs) And Not InList(crSigOpFp.OriDes,1,2)))
		Messagebox('Para Valor Pendente = [Não] é obrigatório o preenchimento do Grupo/Conta ou Conta Mov.Estoque', 0+48, '')
		ThisForm.Pagina.Dados.Get_grupo.SetFocus
		Return
	Endif

	IF Inlist( thisform.pcescolha, 'ALTERAR')
		_Codigo = crSigOpFp.nFPags
		If _Codigo > 0 And ThisForm.podatamgr.CursorQuery('SigOpFp','Crfpag','nfpags', _codigo) And Reccount('CrFpag') > 1
			Messagebox( 'Acesso Rapido da Forma Pagamento já cadastrada ! ',0+48,'')
			Thisform.Pagina.Dados.Get_nfpag.SetFocus
			Return
		Endif
		ThisForm.msv_alterar()
	Endif
Endif

If Inlist( thisform.pcescolha, 'INSERIR')
	If Empty( crSigOpFp.fpags )
		Messagebox( 'Forma Pagamento Inválida ! ',0+48,'')
		Thisform.Pagina.Dados.Get_Forma.SetFocus
		Return
	Endif
	_Codigo = crSigOpFp.FPags
	If ThisForm.podatamgr.ChkRegister('SigOpFp','fpags', _codigo)
		Messagebox( 'Forma Pagamento já cadastrada ! ',0+48,'')
		Thisform.Pagina.Dados.Get_Forma.SetFocus
		Return
	Endif
	
	_Codigo = crSigOpFp.nFPags
	If _codigo > 0 and ThisForm.podatamgr.ChkRegister('SigOpFp','nfpags', _codigo)
		Messagebox( 'Acesso Rapido da Forma Pagamento já cadastrada ! ',0+48,'')
		Thisform.Pagina.Dados.Get_nfpag.SetFocus
		Return
	Endif

	If empty(crSigOpFp.Infos) 
		Messagebox( 'Tipo de Forma Pagamento não pode ficar em Branco ! ',0+48,'')
		Thisform.Pagina.Dados.Dados.Page1.Get_Info.SetFocus
		Return
	Endif
	
	If empty(crSigOpFp.Moefpgs) 
		Messagebox( 'Moeda não pode ficar em branco ! ',0+48,'')
*!*		Thisform.Pagina.Dados.Dados.Page1.Get_modfpg.SetFocus		&& mena - substituido pela linha abaixo
		Thisform.Pagina.Dados.Get_modfpg.SetFocus
		Return
	Endif

	If empty(crSigOpFp.Formas) 
		Messagebox( 'Forma de Pagamento não pode ficar em branco ! ',0+48,'')
		Thisform.Pagina.Dados.Get_formas.SetFocus
		Return
	ENDIF
	
	ThisForm.msv_inserir()
Endif
If Inlist( thisform.pcescolha, 'EXCLUIR')
	ThisForm.msv_excluir()
Endif
Select crSigOpFp
=DoDefault()
Thisform.mAtivaPagina1
ENDPROC
