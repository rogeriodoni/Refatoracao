PROCEDURE carregaprzent
Local lcCta, ldDat

If (InList(crSigCdOpe.DtEntrs, 1, 4, 6) And (InList(ThisForm.pcEscolha,'INSERIR','ALTERAR')) And ;
   (IsEmpty(TprMvCab.PrazoEnts) Or Dtos(TprMvCab.PrazoEnts) = Dtos(Date())))

	lcCta = Iif(crSigCdOpe.COpers = 1, TprMvCab.ContaDs, TprMvCab.ContaOs)
	
	If IsEmpty(TprMvCab.PrazoEnts)
		ldDat = Date()
		If CrSigOpCdc.NMinEnts = CrSigOpCdc.nMaxEnts
			ldDat = Date() + CrSigOpCdc.nMinEnts
			Replace PrazoEnts With ldDat In TprMvCab
		Endif
	Else
		ldDat = Iif(Type('TprMvCab.PrazoEnts')='T', Ttod(TprMvCab.PrazoEnts), TprMvCab.PrazoEnts)
	EndIf

	ThisForm.Parentform.poDataMgr.CursorQuery('SigCdCli', 'LocalCli', 'IClis', lcCta, 'PrazoEnts')
	If Used('LocalCli') And Not Eof('LocalCli') And (LocalCli.PrazoEnts > 0)
		ldDat = ldDat + LocalCli.PrazoEnts

		Replace PrazoEnts With ldDat In TprMvCab
	EndIf
EndIf
* Fim **************************************************************************************
ENDPROC

------------------------------------------------------------
* PROCEDURE return
------------------------------------------------------------
PROCEDURE return
Local llOk

If ThisForm.Direta
	Select CrSigMvCab
	Zap
	Select TprMvCab
	Scatter Memvar memo
	Select CrSigMvCab
	Append Blank
	Gather Memvar Memo
	
	lcSql = [Select a.*, b.TpVencs,b.Entradas,b.dias,b.tVens From SigMvPar a, SigOpFp b where a.EmpDopNums = ']+TprMvCab.EmpDopNums+[' ]+;
			[And a.Fpags = b.fpags ]
	If Thisform.parentForm.PodataMgr.Sqlexecute(lcSql,'LocalPar') < 1
		MessageBox('Favor reinicializar o processo.',16,'Falha na Conexão ( LocalPar )')
		Return .f.
	EndIf

	Select LocalPar
	Index on Fpags + cnidTefs + Str(Parcs,2) Tag FpagV
	Scan
		If LocalPar.TpVencs = 2
			DVenc  = ( Ttod(TprMvCab.PrazoEnts) + LocalPar.Entradas )
			ndia   = LocalPar.dias
			tven   = LocalPar.tvens
			_lcFpg = LocalPar.Fpags
			_lcIdT = LocalPar.CnIdTefs
			Scan While Fpags = _lcFpg And cnIdTefs = _lcIdT
				lnFer = 0
				Verd  = .t.
				lcQuery = [Select * From SigCdFer ]
				If ThisForm.Parentform.podatamgr.sqlexecute(lcquery,'CrSigCdFer') < 1
					=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (crSigCdFer)')
					Return 0
				EndIf
				Select CrSigCdFer
				Index On Dtos(Datas) Tag Datas
				Seek Dtos(DVenc)
				Do While Verd .and. !Eof()
					lnFer = lnFer + 1
					Skip
					If DVenc+lnFer # crSigCdFer.Datas .or. ;
					   Dow(DVenc+lnFer)=1 .or. ;
					   Dow(DVenc+lnFer)=7
					   Verd = .f.
					Endif
				Enddo
				DVenc = DVenc + lnFer
				Replace Vencs With DVenc In LocalPar
				If TVen='F'
					DVenc = GoMonth( DVenc,1 )
				Else
					DVenc = DVenc + NDia
				Endif

				ldVenc = LocalPar.Vencs
				lcSql = [Update SigMvPar Set Vencs = ?ldVenc Where CidChaves = ']+LocalPar.cidchaves+[']
				If ThisForm.ParentForm.PodataMgr.Sqlexecute(lcSql,'') < 1
					MessageBox('Favor reinicializar o processo.',16,'Falha na Conexão (Update SigMvPar )')
					Return .f.
				EndIf
			EndScan
			Skip -1
		Endif
	EndScan

	llOk = ThisForm.Parentform.PoDataMgr.UpDate('CrSigMvCab')
	
	llOk = llOk And ThisForm.ParentForm.Podatamgr.Commit() > 0 
	If Not llOk
		ThisForm.ParentForm.Podatamgr.RollBack()
	Endif

	Select TprMvCab
	Zap
	
Endif
ThisForm.parentform.enabled = .T.
thisform.parentform.validok = thisform.validok
ThisForm.parentform.pagina.dados.dados.setfocus
ThisForm.release

ENDPROC

PROCEDURE Valid
Private pIni, pFim

loCnx = ThisForm.ParentForm.poDataMgr

If Not Empty(This.Value) And Not (This.Tag = Str(This.Value, 10))
	loCnx.CursorQuery('SigCdOpe', 'LocalOpe', 'Dopes', TprMvCab.Dopes, 'ObsPads, ObsInis, ObsFins')
	pIni = Iif(Not Eof('LocalOpe') And Not Empty(LocalOpe.ObsInis), LocalOpe.ObsInis, 000)
	pFim = Iif(Not Eof('LocalOpe') And Not Empty(LocalOpe.ObsFins), LocalOpe.ObsFins, 999)
	
	If not Empty(crSigMvNfi.Cfis) and (crSigOpCdd.ChkCFOP <> 2)
		lcQuery = [Select ObsInis, ObsFins from SigCdOop where Dopes = ']+crSigCdOpe.Dopes+[' and Cfops = ']+crSigMvNfi.Cfis+['] 
		loCnx.SQLExecute(lcQuery,[crTmpCfoOp])
		If not Eof([crTmpCfoOp])
			pIni = Iif( not Empty(crTmpCfoOp.ObsInis), crTmpCfoOp.ObsInis, pIni )
			pFim = Iif( not Empty(crTmpCfoOp.ObsFins), crTmpCfoOp.ObsFins, pFim )
		Endif
	Endif	

	lcQuery = [Select Codigos, Descrs ] + ;
			    [From SigCdObs ] + ;
			   [Where Codigos BetWeen ?pIni And ?pFim]

	If (loCnx.SqlExecute(lcQuery, 'crLocalObs') < 1)
		=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (crLocalObs)')
		Return 0
	EndIf
	Select crLocalObs
	Index On Codigos Tag Codigos
	Index On Descrs Tag Descrs

	Set Near On
	If Not Seek(This.Value, 'crLocalObs', 'Codigos')
		m.loBrowse = CreateObject('fwBuscaSel', 'crLocalObs', 'Codigos', This.Value, 'Observações')
		m.loBrowse.AddColuna('Codigos', '999', 'Cód.', .t., 'Codigos')
		m.loBrowse.AddColuna('Descrs', '', 'Descrição', .t., 'Descrs')
		m.loBrowse.Show
	Endif
	Set Near Off

	If (Lastkey()=27)
		Return 0
