		Return(.F.)
	Else
		If ThisForm.Check1.Value = 1
			MessageBox([Arquivo C:\SAIDA_ESP.TXT Gerado com sucesso !!!], 64, [])
		Else
			Create Cursor crMfd (Linha C(63))
			Select crMfd 
			Append From c:\SAIDA_ESP.TXT SDF
			lcCupom = ''
			Scan 
				lcCupom = lcCupom + Linha
			EndScan

			ThisForm.Edit1.Value = lcCupom
			ThisForm.Edit1.Refresh

			Erase c:\SAIDA_ESP.TXT

			If Used('crMfd')
				USE in crMfd
			EndIf
		EndIf 
	Endif

	FecharPorta(.t.)
		

EndIf

ThisForm.cntAguarde.Visible = .F.
ThisForm.CntAguarde.Refresh



ENDPROC
