* PROCEDURE Pagina.Lista.Grupo_op.Click
------------------------------------------------------------
PROCEDURE Pagina.Lista.Grupo_op.Click
LParameters opcao

If !dodefault( opcao )
	ThisForm.mAtivapagina1
	Return
Endif

ThisForm.AntGrupo = ThisForm.Pagina.Dados.get_Grupo.Value
ThisForm.AntConta = ThisForm.Pagina.Dados.get_Conta.Value

If Inlist( ThisForm.pcEscolha, 'INSERIR', 'ALTERAR', 'PROCURAR' )
	ThisForm.Pagina.Dados.Get_Grupo.SetFocus
Else
	ThisForm.Pagina.Dados.Grupo_Salva.Salva.SetFocus
Endif

ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
If InList(ThisForm.pcEscolha,'INSERIR','ALTERAR')
	If Empty( thisform.pagina.dados.get_grupo.value )
		Messagebox( 'Grupo não pode ficar em branco!', 48, '' )
		ThisForm.Pagina.Dados.Get_grupo.SetFocus
		Return
	Endif
	If Empty( thisform.pagina.dados.get_conta.value ) And crSigCdCeg.TpCads > 2 &&(crSigCdCeg.TpCads <> 1)
		Messagebox( 'Conta não pode ficar em branco!', 48, '' )
		ThisForm.Pagina.Dados.Get_Conta.SetFocus
		Return
	EndIf
	
	lcGrupo = thisform.pagina.dados.get_grupo.value
	lcConta = thisform.pagina.dados.get_conta.value
	
	If (ThisForm.pcescolha = 'INSERIR') And (ThisForm.AntGrupo <> lcGrupo) Or (ThisForm.AntConta <> lcConta)
		lcQuery = [Select * From SigCdCeg Where Grupos = ']+lcGrupo+[' And Contas = ']+lcConta+[']
		If Thisform.PodataMgr.SqlExecute(lcQuery,'TmpCeg') > 0 And Reccount('TmpCeg') > 0
			Messagebox( 'Codigo, Grupo e Conta já Cadastrado!', 48, '' )
			ThisForm.Pagina.Dados.Get_Grupo.SetFocus
			Return
		EndIf
	EndIf
	
	If (ThisForm.pcescolha = 'INSERIR')
		Select CrSigCdCeg
		Replace cidchaves with fUniqueIds()
	EndIf
Endif

=DoDefault()

ThisForm.mAtivapagina1

ENDPROC

local lcod,lfan,codcor,xobj,grupo,retorno

m.llRetorno = .t.

Return(m.llRetorno)
