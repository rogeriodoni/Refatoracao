------------------------------------------------------------
* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
LPARAMETERS lcacesso

With ThisForm
	.poDatamgr = CreateObject('fSqlConector')
	With .poDatamgr
		.AddCursor('SigCcOco','ncodigos','crSigCcOco','',ThisForm.pagina.lista.grade)
		.pfSqlTabela(1).pColuna('ncodigos','','','Código',80,.t.)
		.pfSqlTabela(1).pColuna('cdescrs','','','Descrição',200,.t.)
		.AddCursor('SigNfCor','noccors','crSigNfCor','','')
	EndWith
	= DoDefault()
	.LockScreen = .f.
EndWith

ENDPROC

------------------------------------------------------------
* PROCEDURE msv_procurar
------------------------------------------------------------
PROCEDURE msv_procurar
Local lcBusca

If .Not.IsEmpty(Thisform.Pagina.Dados.Get_ncodigo.value)
	ThisForm.poDataMgr.LocateCursor('crSigCcOco','ncodigos',Thisform.Pagina.Dados.Get_ncodigo.value)
EndIf

ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Lista.Grupo_op.Click
------------------------------------------------------------
PROCEDURE Pagina.Lista.Grupo_op.Click
LPARAMETERS opcao
If Type( 'Opcao' ) = 'N'
	This.Value = Opcao
Endif
If This.Value=4
	If ThisForm.poDataMgr.ChkRegister('SigNfCor','noccors',ThisForm.Pagina.Dados.Get_nCodigo.Value)
		Messagebox( 'Ocorrência com Movimentação, Exclusão Inválida !', 48, '' )
		ThisForm.Pagina.Dados.Grupo_Salva.Cancelar.SetFocus
		Return
	Endif
Endif

Select CrSigCcOco
If !dodefault( Opcao )
	ThisForm.mAtivapagina1
	Return
Endif
If Inlist(Thisform.pcEscolha,'INSERIR','PROCURAR')
	Thisform.Pagina.Dados.Get_ncodigo.SetFocus
Endif
If Thisform.pcEscolha='ALTERAR'
	Thisform.Pagina.DAdos.Get_cdescr.SetFocus
Endif
If Inlist(Thisform.pcEscolha,'CONSULTAR','EXCLUIR')
	Thisform.Pagina.Dados.Grupo_Salva.Salva.SetFocus
Endif

ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
If Thisform.pcEscolha='INSERIR'
	If IsEmpty(Thisform.Pagina.Dados.Get_NCodigo.Value)
		Messagebox('Código Inválido !!',0+48,'')
		Thisform.Pagina.Dados.Get_NCodigo.SetFocus
		Return
	Endif
	
	If ThisForm.poDataMgr.ChkRegister('SigCcOco','ncodigos',ThisForm.Pagina.Dados.Get_Ncodigo.Value)
		Messagebox( 'Código já Cadastrado !!', 48, '' )
		ThisForm.Pagina.Dados.Get_Ncodigo.SetFocus
		Return
	Endif
Endif
If Inlist(Thisform.pcEscolha,'INSERIR','ALTERAR')
	If IsEmpty(Thisform.Pagina.Dados.Get_cDescr.Value)
		Messagebox('Descrição Inválida !!',0+48,'')
		Thisform.Pagina.Dados.Get_cDescr.SetFocus
		Return
	Endif	
Endif
=DoDefault()

ThisForm.Refresh
Thisform.mAtivapagina1

ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Dados.Grupo_Salva.Cancelar.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Cancelar.Click
=DoDefault()

If ThisForm.plCancelar
	ThisForm.mAtivapagina1
Endif


ENDPROC
