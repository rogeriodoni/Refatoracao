------------------------------------------------------------
* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
With ThisForm
	.poDataMgr = CreateObject('fSqlConector', ThisForm.Name)

	If (.poDataMgr.pnIdconn > 0)
		With .poDataMgr
			.AddCursor('SigCdUsu', 'Usuarios', 'crSigCdUsu', '', ThisForm.Pagina.Lista.Grade)
			.pfSqlTabela(1).pColuna('Usuarios', '', '', 'Usuário', 080, .t.)
			.pfSqlTabela(1).pColuna('NComps',   '', '', 'Nome',    310, .t.)
			.pfSqlTabela(1).pColuna('CCargs',   '', '', 'Cargo',   080, .t.)
			.pfSqlTabela(1).pColuna('Deptos',   '', '', 'Depto.',  080, .t.)
			.pfSqlTabela(1).pColuna('Setores',  '', '', 'Setor', 	080, .t.)
		EndWith

		=DoDefault()

		Select crSigCdUsu
		Locate For Usuarios = Usuar
		.Pagina.Lista.Grade.SetAll('DynamicForeColor',;
			        [Iif(crsigcdusu.cAtivos=='N', Rgb(255,0,0), Rgb(0,0,255))],'Column')
		.Pagina.Lista.Grade.Column1.SetFocus
		
		With Thisform.Pagina.Lista.cmdGLimSenha
			.Enabled = (fChecaAcesso(ThisForm.Name, 'LIBSENHA'))
			.Visible = (fChecaAcesso(ThisForm.Name, 'LIBSENHA'))
			.cmdLimSenha.Enabled = (fChecaAcesso(ThisForm.Name, 'LIBSENHA'))
			.cmdLimSenha.Visible = (fChecaAcesso(ThisForm.Name, 'LIBSENHA'))
		EndWith 
		
	Else
		=Messagebox('Impossível Efetuar Conexão Com o Servidor de Banco de Dados...', 0+48, 'CreateObject')
		Return .f.
	EndIf
EndWith
* Fim **************************************************************************************
ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Lista.Grupo_op.Click
------------------------------------------------------------
PROCEDURE Pagina.Lista.Grupo_op.Click
Parameters Opcao

If (Type('Opcao') = 'N')
	This.Value = Opcao
EndIf

Select crSigCdUsu

If Not DoDefault(Opcao)
	ThisForm.LockScreen = .f.
	ThisForm.mAtivapagina1
	Return 0
EndIf

Do Case
	Case InList(ThisForm.pcEscolha, 'INSERIR', 'PROCURAR')
		Replace cAtivos With 'S' In crSigCdUsu
		ThisForm.Pagina.Dados.getUsuarios.SetFocus

	Case Inlist(ThisForm.pcEscolha, 'EXCLUIR', 'CONSULTAR')
		ThisForm.Pagina.Dados.Grupo_Salva.Salva.SetFocus

	Case Inlist(ThisForm.pcEscolha, 'ALTERAR')
		 ThisForm.Pagina.Dados.getNComps.SetFocus
EndCase

Select crSigCdUsu

ThisForm.Pagina.Dados.getRClis.Value = []
If Not Empty(crSigCdUsu.IClis) And InList(ThisForm.pcEscolha, [ALTERAR], [EXCLUIR], [CONSULTAR])
	lcQuery = [Select RClis ] + ;
			    [From SigCdCli ] + ;
			   [Where IClis = '] + crSigCdUsu.IClis + [']

	If (ThisForm.poDataMgr.SqlExecute(lcQuery, [LocalBus]) < 1)
		=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (LocalBus)])
		Return .f.
	EndIf
	Go Top In LocalBus
	If Not Eof([LocalBus]) And Not Empty(LocalBus.RClis)
		ThisForm.Pagina.Dados.getRClis.Value = LocalBus.RClis
	EndIf
EndIf
ThisForm.Pagina.Dados.getRClis.Refresh 

ThisForm.Pagina.Dados.optAtivo.Value = Iif(InList(crSigCdUsu.CAtivos, [S], [ ]), 1, 2)

ThisForm.Pagina.Dados.Activate
* Fim **************************************************************************************
ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
If InList(Thisform.pcEscolha, 'INSERIR', 'ALTERAR')
	If (Thisform.pcEscolha = 'INSERIR')
		If Empty(crSigCdUsu.Usuarios)
			=MessageBox('Usuário Inválido!!!', 0+48, '')
			ThisForm.Pagina.Dados.getUsuarios.SetFocus	
			Return 0
		EndIf
		
		If ThisForm.poDataMgr.CursorQuery('SigCdUsu', 'crTmp', 'Usuarios', crSigCdUsu.Usuarios)
			=MessageBox('Usuário Já Cadastrado!!!', 0+48, '')
			ThisForm.Pagina.Dados.getUsuarios.SetFocus	
			Return 0
		EndIf
	EndIf
EndIf

If (Thisform.pcEscolha = 'EXCLUIR')
	If ThisForm.poDataMgr.CursorQuery('SigCdAcG', 'crTmp', 'Usuarios', crSigCdUsu.Usuarios)
		=MessageBox('Usuário Possui Acessos de Grupo Cadastrados!!!', 0+48, 'Exclusão Não Permitida!!!')
		ThisForm.Pagina.Dados.Grupo_Salva.Cancelar.SetFocus 
		Return 0
	EndIf

	If ThisForm.poDataMgr.CursorQuery('SigCdAcU', 'crTmp', 'Usuarios', crSigCdUsu.Usuarios)
		=MessageBox('Usuário Possui Acessos de Programas Cadastrados!!!', 0+48, 'Exclusão Não Permitida!!!')
		ThisForm.Pagina.Dados.Grupo_Salva.Cancelar.SetFocus 
		Return 0
	EndIf

	If ThisForm.poDataMgr.CursorQuery('SigCdAcB', 'crTmp', 'Usuarios', crSigCdUsu.Usuarios)
		=MessageBox('Usuário Possui Acessos de Ícones Cadastrados!!!', 0+48, 'Exclusão Não Permitida!!!')
		ThisForm.Pagina.Dados.Grupo_Salva.Cancelar.SetFocus 
		Return 0
	EndIf
EndIf

=DoDefault()

ThisForm.mAtivaPagina1
* Fim **************************************************************************************
ENDPROC
