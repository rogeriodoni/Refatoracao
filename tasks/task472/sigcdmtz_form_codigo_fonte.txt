* PROCEDURE atualizatotal
------------------------------------------------------------
PROCEDURE atualizatotal
Parameters pcChave
Local lnOldSel, lnTotal

lnOldSel = Select()

*-- Atualiza valor Total da linha
lnTotal = 0

Select csSigCdMtI
Locate For .F.
Select (val_abr+val_ago+val_dez+val_fev+val_jan+val_jul+val_jun+val_mai+val_nov+val_out+val_set+val_mar) As Total From csSigCdMtI Where cIdChaves = pcChave Into Cursor crTmpTot Readwrite

Select crTmpTot
lnTotal = crTmpTot.Total

Select csSigCdMtI
Replace Total With lnTotal For cIdChaves = pcChave In csSigCdMtI

*-- Atualiza valor Total
If Used([crTmpTot])
	use in crTmpTot
EndIf 
lnTotal = 0

Select csSigCdMtI
Locate For .F.
Select Sum(Total) As Total From csSigCdMtI Into Cursor crTmpTot Readwrite

Select crTmpTot
lnTotal = Nvl(crTmpTot.Total,0)

ThisForm.Pagina.Dados.getValorTotal.Value = lnTotal
ThisForm.Pagina.Dados.getValorTotal.Refresh

Select(lnOldSel)
Return .T.
ENDPROC

* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click


If Inlist( ThisForm.pcEscolha,[INSERIR],[ALTERAR] )

	lcCodigo = thisform.pagina.dados.getCodigo.value
	lcAno 	 = This.parent.parent.getAno.Value 
	lcGrupo  = This.parent.parent.Get_sgrupo.Value
	lcConta  = This.parent.parent.Get_sconta.Value
	lcMoeda  = This.parent.parent.getMoeda.Value

	If ThisForm.pcEscolha=[INSERIR]
		
		lcQuery = [Select * From SigCdMtz Where Ano = ']+lcAno+[' And sGrupos = ']+lcGrupo+[' And sContas = ']+lcConta+[' ]
		If ThisForm.Podatamgr.SqlExecute(lcQuery,[crChkMtz])<1
			MessageBox([Favor Reiniciar a Transação.],0+16,[Falha de Comunicação (crChkMtz)])
			Return .F.
		EndIf 
		If Reccount([crChkMtz])>0
			Messagebox([Código já cadastrado.],64,[Aviso])
			thisform.pagina.dados.getAno.Setfocus()
			Return .f.
		Endif
		
	EndIf 

	If Empty(lcAno)
		Messagebox([Ano obrigatório.],64,[Aviso])
		thisform.pagina.dados.getAno.Setfocus()
		Return .f.
	EndIf
	
	If Empty(lcGrupo)
		Messagebox([Grupo de Centro de Custo Obrigatório.],64,[Aviso])
		thisform.pagina.dados.Get_sgrupo.Setfocus()
		Return .f.
	EndIf 
	
	If Empty(lcConta)
		Messagebox([Conta de Centro de Custo Obrigatório.],64,[Aviso])
		thisform.pagina.dados.Get_sconta.Setfocus()
		Return .f.
	EndIf 
	
	If Empty(lcMoeda)
		Messagebox([Moeda obrigatório.],64,[Aviso])
		thisform.pagina.dados.getMoeda.Setfocus()
		Return .f.
	EndIf
	
	Select csSigCdMtI
	Select Count(*) as Qtds, Grupos, Contas ;
		From csSigCdMtI ;
		Where Not Empty(Contas) And Not Empty(Grupos) ;
		Group by Grupos, Contas ;
		Having Count(*) >= 2 ;
		Into Cursor csRegDup ReadWrite 
	
	Select csRegDup
	If Reccount([csRegDup])>0
		MessageBox([Há registros duplicados na grade de Contas, Verifique.],0+48,[Aviso])
		Return .F.
	EndIf 
	
	
	If ThisForm.pcEscolha=[INSERIR]
		Replace CidChaves With fUniqueIds() in crSigCdMtz
		Replace DtIncs With Datetime() in crSigCdMtz
		Replace UsuIncs With Usuar in crSigCdMtz
	EndIf 	

	Select crSigCdMtI
	Go Top
	
	Select csSigCdMtI
	Scan For (Not Empty(Grupos) And Not Empty(Contas))
		Scatter Memvar
		
		Select crSigCdMtI
		Append Blank 
		
		Gather Memvar 
		Select csSigCdMtI
	EndScan 
	
	Select crSigCdMtI
	Go Top
	Delete For (Empty(Grupos) And Empty(Contas))
	Go Top
	
	Replace All Codigo With crSigCdMtz.Codigo For Empty(Codigo) In crSigCdMtI
	Replace CidChaves  With fUniqueIds() For Empty(CidChaves) In crSigCdMtI

	If Inlist( ThisForm.pcEscolha,[ALTERAR])
		Replace DtAlts With Datetime() in crSigCdMtz
		Replace UsuAlts With Usuar in crSigCdMtz
	Endif
Endif

DoDefault()

Thisform.mAtivapagina1
ENDPROC
