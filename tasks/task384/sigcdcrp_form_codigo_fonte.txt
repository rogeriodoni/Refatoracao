PROCEDURE msv_procurar

* Fim **************************************************************************************

------------------------------------------------------------
* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
Lparameters pcTipo

** Tipo ( = 0 - Carac Produtos, = 1 - Carac Contas, = 2 - Movimentações ) para manter compatibilidade
ThisForm.pnTipo = Iif(Type([pcTipo])<>[C],0,Iif(pcTipo == 'C',1,2))

Do case
Case( ThisForm.pnTipo = 1 )
	ThisForm.Caption = [Características de Contas]
	ThisForm.Name    = [SIGCDCRPC]

Case( ThisForm.pnTipo = 2 )
	ThisForm.Caption = [Características de Movimentações]
	ThisForm.Name    = [SIGCDCRPO]
	
OtherWise
	ThisForm.Caption = [Características de Produtos]
	ThisForm.Name    = [SIGCDCRP]
EndCase

With ThisForm
	.poDataMgr = CreateObject('fSqlConector', ThisForm.Name)

	lcQuery = [Select a.*, ]+Iif(ThisForm.pnTipo=2,[b.Descs],[b.Dgrus])+[ as Dgrus ] + ;
			[from SigCrRaP a ] + ;
			[Left Join ]+Iif(ThisForm.pnTipo=2,[SigCdGpr b on a.Cgrus = b.Codigos ],[SigCdGrp b on a.Cgrus = b.Cgrus ]) + ;
			[where a.nTipos = ]+Str(ThisForm.pnTipo,1)
				
	If (.poDataMgr.pnIdconn > 0)
		With .poDataMgr
			.AddCursor('SigCrRaP', 'cidchaves', 'crSigCrRaP', '', ThisForm.Pagina.Lista.Grade,lcQuery)
			.pfSqlTabela(1).pColuna('Codigos', '', '', 'Código',    150, .t.)
			.pfSqlTabela(1).pColuna('Descrs',  '', '', 'Descrição', 290, .t.)
			.pfSqlTabela(1).pColuna('Cgrus',   '', '', 'Grupo', 80, .t.)
			.pfSqlTabela(1).pColuna('Ordens',  '', '', 'Ordem', 60, .t.)
		EndWith

		=DoDefault()

		Select crSigCrRaP
		Go Top
		Replace all Dgrus With Nvl(Dgrus,'')
		
		ThisForm.Pagina.Lista.Grade.Column1.SetFocus
	Else
		=Messagebox('Impossível Efetuar Conexão Com o Servidor de Banco de Dados...', 0+48, 'CreateObject')
		Return .f.
	EndIf
EndWith
* Fim **************************************************************************************
ENDPROC

PROCEDURE Pagina.Lista.Grupo_op.Click
Lparameters Opcao

If Not DoDefault(Opcao)

		Thisform.Pagina.Dados.getDescrs.Setfocus

	Case InList(Thisform.pcEscolha, 'CONSULTAR', 'EXCLUIR')
		Thisform.Pagina.Dados.Grupo_Salva.Salva.Setfocus
EndCase

Select crSigCrRaP
* Fim **************************************************************************************

------------------------------------------------------------
* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
If Inlist(Thisform.pcEscolha, [INSERIR], [ALTERAR])

	If Empty(Thisform.Pagina.Dados.getCodigos.Value)
		=Messagebox('Código Inválido!!!', 0+64, '')
		Thisform.Pagina.Dados.getCodigos.SetFocus
		Return 0
	Else
		If (Thisform.Pagina.Lista.Grupo_op.Value = 1)
			lcQuery = [Select * From SigCrRaP ] + ;
				[Where Codigos = '] + Thisform.Pagina.Dados.getCodigos.Value + [' ] + ;
				[And nTipos = ]+Alltrim(Str(Thisform.pnTipo))
			If (Thisform.poDataMgr.SqlExecute(lcQuery, 'crBusca') < 1)
				=Messagebox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (crBusca)')
				Return 0
			Endif
			If Not Eof('crBusca')
				=Messagebox('Característica Já Cadastrada!!!', 0+64, '')
				Thisform.Pagina.Dados.getCodigos.SetFocus
				Return 0
			Endif
			Replace nTipos With Thisform.pnTipo In crSigCrRaP
		Endif
	Endif

	lcQuery = [Select * ] + ;
		[From SigCrRaP ] + ;
		[Where Ordens = ] + Alltrim(Str(Thisform.Pagina.Dados.get_ordem.Value)) + [ ] + ;
		[And Not Codigos = '] + Thisform.Pagina.Dados.getCodigos.Value + [' ] + ;
		[And nTipos = ]+Alltrim(Str(Thisform.pnTipo))

	If (Thisform.poDataMgr.SqlExecute(lcQuery, 'crBusca') < 1)
		=Messagebox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (crBusca)')
		Return 0
	Endif
	Go Top In crBusca
	If Not Eof('crBusca')
		=Messagebox('Ordem da Característica Já Cadastrada!!!', 0+64, '')
		Thisform.Pagina.Dados.getCodigos.SetFocus
		Return 0
	Endif

	If Empty(Thisform.Pagina.Dados.GetDescrs.Value)
		=Messagebox('Descricao Inválida!!!', 0+64, '')
		Thisform.Pagina.Dados.GetDescrs.SetFocus
		Return 0
	Endif
Endif

If Thisform.pcEscolha = 'INSERIR'
	Replace CidChaves With fUniqueIds() In crSigCrRaP
Endif

If Inlist(Thisform.pcEscolha, 'EXCLUIR')
	If Thisform.pnTipo = 0
		lcQuery = [Select * ] + ;
			[From SigPrCar ] + ;
			[Where Codigos = '] + Thisform.Pagina.Dados.getCodigos.Value + [' ]

		If (Thisform.poDataMgr.SqlExecute(lcQuery, 'LocalCar') < 1)
			=Messagebox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (LocalCar)')
			Return 0
		Endif

		If (Reccount('LocalCar') > 0)
			=Messagebox('Característica Sendo Utilizada Em Produtos!!!', 16, '')
			Return 0
		Endif
	Endif
Endif

Select crSigCrRaP

=DoDefault()

Thisform.mAtivaPagina1
* Fim **************************************************************************************

ENDPROC

			This.Parent.Get_Ds_Grupo.Value = ''
		EndIf
	Else
		m.loLista = CreateObject('fwbuscaext',ThisForm.poDataMgr.pnIdConn,'SigCdGrp','CrGrupos','CGrus',This.Value,'Grupos',.T.)
		If !(m.loLista.plAchouRegistro)
			m.loLista.mAddColuna('CGrus','XXXXXX','Código')
			m.loLista.mAddColuna('DGrus','','Descrição')
			m.loLista.mAddColuna('Mercs','','Gde.Grupo')
			m.loLista.Show()
		EndIf
		If LastKey() <> 27 And Used('CrGrupos')
			This.Value                  = Evaluate('CrGrupos.CGrus')
			This.Parent.Get_Ds_Grupo.Value = Evaluate('CrGrupos.DGrus')
		Else
			This.Value                  = ''
			This.Parent.Get_Ds_Grupo.Value = ''
		EndIf

	EndIf 
EndIf
This.Refresh
This.Parent.Get_Ds_Grupo.Refresh
ENDPROC
