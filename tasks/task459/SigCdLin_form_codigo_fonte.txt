* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
Private pLin, pGru

ThisForm.Busca = 1

pLin = Space(10)
pGru = Space(03)

lcQryLinF = [Select a.*, b.Descrs From SigCdLnf a, SigCdGcr b Where a.Linhas = ?pLin And b.Codigos = a.Grupos]
lcQryLinI = [Select * From SigCdLni Where Linhas = ?pLin]
lcQryGru  = [Select * From SigCdGrp Where CGrus = ?pGru]
lcQryPCo  = [Select * From SigPrPrc Where Linhas = ?pLin]

With ThisForm
	.poDataMgr = CreateObject('fSqlConector', ThisForm.Name)

	If (.poDataMgr.pnIdconn > 0)
		With .poDataMgr
			lcQryLin = [Select a.* from SigCdLin a]
			.AddCursor('SigCdLin', 'Linhas', 'crSigCdLin', '', ThisForm.Pagina.Lista.Grade,lcQryLin)
			.pfSqlTabela(1).pColuna('Linhas',  '', '', 'Linha',        080, .t.)
			.pfSqlTabela(1).pColuna('Descs',   '', '', 'Descrição',    290, .t.)

			.AddCursor('SigCdLnf', 'cIdChaves', 'crSigCdLnf', '','',lcQryLinF)
			.AddCursor('SigCdLni', 'cIdChaves', 'crSigCdLni', '','',lcQryLinI)
			.AddCursor('SigCdGrp',  'CGrus',     'crSigCdGrp',  '','',lcQryGru)
			.AddCursor('SigPrPrc','cIdChaves', 'crSigPrPrc','','',lcQryPco)
		EndWith
		
		=DoDefault()

		.poDataMgr.CursorQuery('SigCdPam', 'crSigCdPam', .f., .f., 'GesInd')

		.poDataMgr.CursorQuery('SigCdGcr', 'crSigCdGcr', 'GerBals', 1)
		Select crSigCdGcr
		Index On Codigos Tag Codigos

		lcQuery = [Select a.* From SigCdUsu a ]+;
				  [Where Not a.cAtivos='N' ]
		.poDataMgr.SqlExecute(lcQuery,'TmpUsu')
		Select TmpUsu
		Index On Usuarios Tag Usuarios

		If (crSigCdPam.GesInd = 1)
			.Pagina.Dados.PageFrame1.Page2.Enabled = .t.
			.Pagina.Dados.PageFrame1.Page2.Caption = 'Produção: Processos'
		Else
			.Pagina.Dados.PageFrame1.Page2.Enabled = .f.
			.Pagina.Dados.PageFrame1.Page2.Caption = ''
		EndIf

		lcQuery = [Select Distinct a.CPros, a.DPros, a.CGrus ] + ;
					[From SigCdPro a, SigCdGrp b ] + ;
				   [Where a.CGrus = b.CGrus And ] + ;
				   		 [b.MtPrimas In (1, 2) ] + ;
				   [Order By a.CPros]

		If (.poDataMgr.SqlExecute(lcQuery, 'crMtPrima') < 1)
			=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (crMtPrima)')
			Return .f.
		EndIf
		Select crMtPrima
		Index On DPros Tag DPros
		Index On CPros Tag CPros
		Go Top

		If (ThisForm.poDataMgr.getDriver() = 'foxpro')
			lcDiretorio = Addbs(JustPath(ThisForm.poDataMgr.GetDatabase()))
			Use &lcDiretorio.\SigCdCli Alias TmpCli In 0
		Else
			lcQuery = [Select IClis, RClis, Grupos, Inativas ] + ;
					    [From SigCdCli ] + ;
				   	   [Where GerBals = 1]

			If (ThisForm.poDataMgr.SqlExecute(lcQuery, 'TmpCli') < 1)
				=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (TmpCli)')
				Return 0
			EndIf
			Select TmpCli
			Index On IClis Tag BalCodigo
			Index On RClis Tag BalNome
		EndIf
		
		.LockScreen = .t.
		=fCarregaGrade(.Pagina.Lista.Grade)
		.LockScreen = .f.
	Else
		=Messagebox('Impossível Efetuar Conexão Com o Servidor de Banco de Dados...', 0+48, 'CreateObject')
		Return .f.
	EndIf
EndWith
* Fim **************************************************************************************
ENDPROC

		llOks = .f.
	EndIf

		=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (Update - crSigCdLnf)')

	EndIf
EndIf
If llOks
	If Not ThisForm.poDataMgr.Update('crSigCdLni')
		=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (Update - crSigCdLni)')
		llOks = .f.
	EndIf
EndIf

If llOks
	lcQuery = [Delete From SigPrPrc Where Linhas = '] + lcLin + [']
	If (ThisForm.poDataMgr.SqlExecute(lcQuery, '') < 1)
		=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (Delete - crSigPrPrc)')
		llOks = .f.
	EndIf
EndIf
If llOks
	If Not ThisForm.poDataMgr.Update('crSigPrPrc')
		=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (Update - crSigPrPrc)')
		llOks = .f.
	EndIf
EndIf

If llOks
	=DoDefault()
Else
	ThisForm.poDataMgr.Rollback()
	Return .f.
EndIf
* Fim **************************************************************************************
ENDPROC

* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
Private pLin

pLin = crSigCdLin.Linhas

With ThisForm.Pagina.Dados.PageFrame1
	If InList(ThisForm.pcEscolha,'INSERIR','ALTERAR')
		Do Case
			Case Empty(.Page1.GetLinhas.Value)
				=MessageBox('Linha de Produção Inválida!!!', 48, '')
				.Page1.GetLinhas.SetFocus
				.ActivePage = 1
				Return .f.

			Case Empty(.Page1.getDescs.Value)
				=MessageBox('Descrição Inválida!!!', 48, '')
				.Page1.getDescs.SetFocus
				.ActivePage = 1
				Return .f.				
		EndCase

		If (ThisForm.pcEscolha = 'INSERIR')
			If ThisForm.poDataMgr.CursorQuery('SigCdLin', , 'Linhas', .Page1.GetLinhas.Value)
				=Messagebox('Linha de Produção Já Cadastrada!!!', 0+64, '')
				.Page1.GetLinhas.SetFocus
				.ActivePage = 1
				Return .f.
			EndIf
		EndIf

		If (crSigCdPam.GesInd = 1)
			Select cQSigCdLnf
			Go Top
			If Eof()
				=MessageBox('Nenhum Processo Cadastrado!!!', 48, '')
				.ActivePage = 2 
				.Page2.SetFocus
				.Page2.GradFase.Column1.SetFocus
				Return .f.
			Else
				If (cqSigCdLnf.Ordems <> 1)
		            =MessageBox('Inconsistência na Ordem dos Processos!!!', 48, '')
					.ActivePage = 2 
					.Page2.SetFocus
					.Page2.GradFase.Column1.SetFocus
					Return .f.
		      	EndIf
			EndIf

			lnOrdem = 1
			Scan
				If Empty(cqSigCdLnf.Grupos)
					=MessageBox('Existe Processo em Branco!!!', 48, '')
					.Page2.GradFase.Column2.SetFocus
					Return .f.
				EndIf
				If (cqSigCdLnf.Ordems < lnOrdem)
					Loop
				Endif
				If (cqSigCdLnf.Ordems <> lnOrdem)
					=MessageBox('Inconsistência na Ordem dos Processos!!!', 48, '')
					.ActivePage = 2
					.Page2.SetFocus
                    .Page2.GradFase.Column1.SetFocus
					Return .f.
				EndIf
				lnOrdem = cqSigCdLnf.Ordems + 1
			EndScan
		EndIf
		Select crSigCdLin
	EndIf

	If (crSigCdPam.GesInd = 1) And (ThisForm.pcEscolha = 'EXCLUIR')
		Delete From crSigCdLnf
		Delete From CrSigPrPrc
	EndIf

	If (ThisForm.pcEscolha = 'EXCLUIR')
		Delete From crSigCdLni
	EndIf
      
	Select crSigCdLnf
	Zap
	Select CqSigCdLnf
	Scan
		Scatter Memvar Memo
		m.cidchaves = fUniqueIds()
		Select CrSigCdLnf
		Append Blank
		Gather Memvar Memo
	EndScan

	Select crSigCdLni
	Scan
		Replace cIdchaves With Iif(Empty(cIdChaves), fUniqueIds(), cIdChaves), ;
				Linhas    With Iif(Empty(Linhas), pLin, Linhas)
	EndScan

	Select CrSigPrPrc
	Scan
		If Empty(Mat1s) And Empty(Mat2s) And Empty(Cgrus) and Empty(mercs)
			Delete
		Endif
		Replace cIdchaves With Iif(Empty(cIdChaves), fUniqueIds(), cIdChaves), ;
				Linhas    With pLin
	EndScan

	=DoDefault()

	ThisForm.mAtivaPagina1
EndWith
* Fim *********************************************************************************
ENDPROC
