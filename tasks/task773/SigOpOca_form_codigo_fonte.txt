PROCEDURE cmd_ok.Click
Replace ObsAudts With ThisForm.Memo_Obs.Value in CrSigCqOch

If ThisForm.pTipo
	lctipo   = 'Auditado'
	lcUpdate = [Update SigMvCcr Set DtAudits = ?llData, Auditors = ']+Usuar+[ ' Where Grupos = ']+crSigMvCcr.Grupos+[' And Contas = ']+;
				crSigMvCcr.Contas+[' And Moedas = ']+crSigMvCcr.Moedas+[ ' And Nopers = ]+Str(crSigMvCcr.Nopers)
Else
	lctipo   = 'DesAuditado'
	llData = .Null.
	lcUpdate = [Update SigMvCcr Set DtAudits = ?llData, Auditors = '          ' Where Grupos = ']+crSigMvCcr.Grupos+[' And Contas = ']+;
				crSigMvCcr.Contas+[' And Moedas = ']+crSigMvCcr.Moedas+[ ' And Nopers = ]+Str(crSigMvCcr.Nopers)
Endif

If ThisForm.Formpai.PodataMgr.Update('CrSigCqOch')
	lnErro = ThisForm.FormPai.poDataMgr.SqlExecute(lcUpdate,'')
Else
	lnErro = 0
Endif
If lnErro > 0
	ThisForm.FormPai.poDataMgr.Commit()
		
	Select CrSigMvCcr
	If Thisform.pTipo
		Replace DtAudits With Datetime(), ;
		        Auditors With Usuar
	Else
		Replace DtAudits With {} ,;
		        Auditors With " "
	Endif
EndIf
ThisForm.Release

ENDPROC
