PROCEDURE Click
*!*	If ThisForm.Get_FrtI.Value <> ThisForm.Get_FrtT.Value
*!*		MessageBox('A Soma do Valor do Frete Informado não confere com Total do Frete!!!',0+16,'')
*!*		Return .f.
*!*	Endif
*!*	If ThisForm.Get_Despai.Value <> ThisForm.Get_Despat.Value
*!*		MessageBox('A Soma do Valor das Despesas Acessórias Informado não confere com Total !!!',0+16,'')
*!*		Return .f.
*!*	Endif
*!*	If ThisForm.Get_SegI.Value <> ThisForm.Get_SegT.Value
*!*		MessageBox('A Soma do Valor do Seguro Informado não confere com Total !!!',0+16,'')
*!*		Return .f.
*!*	Endif

Update crTpmMvItn Set PrzEnts = Ctod(''), ValFrts = 0, DespAces = 0, ValSegs = 0

Select CsPrazE
Scan
	=Seek(csPrazE.Dopes, [crTodosOpe2], [Dopes])
	lnUti = Nvl(crTodosOpe2.chkUteis, 0)
	If Not IsEmpty(csPrazE.Prazos)
		lcHra = Time(csPrazE.Prazos)
		ldDat = Ttod(csPrazE.Prazos)

		If (lnUti = 1) && Se Valida Dias Úteis
			ldNew = Ttod(csPrazE.Prazos)
			llOks = .f.
			Do While Not llOks
				ldNew = fChkDiaUteis(TprMvCab.Datas, ldNew, csPrazE.NMinEnts, ThisForm.ParentForm.poDataMgr)
				llOks = (ldNew = ldDat)
				ldDat = ldNew
			EndDo

			If (Ttod(csPrazE.Prazos) <> ldDat)
				lcMsg = [A Data Selecionada Para Entrega Não Coincidiu Com os Dias Úteis] + Chr(13) + ;
						[Necessários Para Entrega e Será Alterada Para : ] + Dtoc(ldDat)
				=MessageBox(lcMsg, 0+16, [Atenção!!!])
			EndIf
		EndIf
		
		Select csPrazE
		lcMsg = fChkDiasEnts(ldDat, Dopes, ThisForm.ParentForm.poDataMgr)
		If Not Empty(lcMsg) && Se a Data Não É Válida Para Entregas
			llOks = .f.
			Do While Not llOks
				ldDat = ldDat + 1
				lcRet = fChkDiasEnts(ldDat, CsPrazE.Dopes, ThisForm.ParentForm.poDataMgr)
				llOks = Empty(lcRet)
			EndDo

			lcMsg = lcMsg + Chr(13) + [A Data da Operação ]+Alltrim(CsPrazE.Dopes)+[ Será Postergada Para : ] + Dtoc(ldDat)
			=MessageBox(lcMsg, 0+16, [Atenção!!!])
		EndIf
		Replace Prazos With fDtoSql(ldDat, lcHra) In CsPrazE
	EndIf

	If InList(CsPrazE.DtEntrs, 1, 4, 6)
		If IsEmpty( CsPrazE.Prazos )
			Messagebox( 'Prazo de Entrega da Operacao '+Alltrim(CsPrazE.Dopes)+' deve ser informado...', 32, '' )
			ThisForm.Grd.SetFocus
			Return
		EndIf
		If (CsPrazE.Prazos < TprMvCab.Datas)
			Messagebox( 'Prazo de Entrega da Operacao '+Alltrim(CsPrazE.Dopes)+' deve ser maior que a Data da Movimentação...', 32, '' )
			ThisForm.Grd.SetFocus
			Return
		EndIf
		If CsPrazE.NMinEnts > 0
			ldEnt = (Ttod(fDtoSql(TprMvCab.Datas)) + CsPrazE.NMinEnts)
			If (CsPrazE.Prazos < ldEnt)
				Messagebox('Prazo de Entrega Mínimo da Operacao '+Alltrim(CsPrazE.Dopes)+' é ' + Dtoc(ldEnt) + '...', 32, '' )
				ThisForm.Grd.SetFocus
				Return
			EndIf
		EndIf
		If CsPrazE.NMaxEnts > 0
			ldEnt = (Ttod(fDtoSql(TprMvCab.Datas)) + CsPrazE.NMaxEnts)
			If (Ttod(CsPrazE.Prazos) > ldEnt)
				Messagebox('Prazo de Entrega Máximo da Operacao '+Alltrim(CsPrazE.Dopes)+' é ' + Dtoc(ldEnt) + '...', 32, '' )
				ThisForm.Grd.SetFocus
				Return
			EndIf
		EndIf
	EndIf
	If CsPrazE.LocEntObs = 1 And InList(CsPedidos.DtEntrs, 3, 4, 5, 6) And CsPrazE.LocalEnts = 0
		Messagebox('Local de Entrega da Operacao '+Alltrim(CsPrazE.Dopes)+' não Informado!!!',0+32, 'Obrigatório' )
		ThisForm.Grd.SetFocus
		Return
	EndIf

	* Checagens de Valores Mínimos
	=Seek(csPrazE.Dopes, [crTodosOpe2], [Dopes])
	lcMsg = []
	If Not Empty(crTodosOpe2.MinFretes) And (CsPrazE.VlFretes < crTodosOpe2.MinFretes)
		lcMsg = lcMsg + Iif(Empty(lcMsg), [], [ / ]) + [Frete]
	EndIf

	If Not Empty(crTodosOpe2.MinDesps) And (CsPrazE.DespAces < crTodosOpe2.MinDesps)
		lcMsg = lcMsg + Iif(Empty(lcMsg), [], [ / ]) + [Desp. Acessórias]
	EndIf

	If Not Empty(lcMsg)
		=MessageBox([Existem Valores Na Operação Abaixo do Mínimo : ] + lcMsg, 0+64, [ERRO])
		Do Form SigOpSen With [LIBOPE], [Libera Valores na Operação Abaixo do Limite], lcMsg To _Retorno
		If (Left(_Retorno, 1) <> [*])
			=MessageBox([Autorização Não Confirmada!!!], 0+48, [Atenção!!!])
			Return .f.
		EndIf
	EndIf

	Update crTpmMvItn Set PrzEnts = CsPrazE.Prazos, ValFrts = CsPrazE.VlFretes, LocalEnts = CsPrazE.LocalEnts, ;
			ValSegs = CsPrazE.VlSeguros, ContaEs = CsPrazE.ContaEs, DespAces = CsPrazE.DespAces ;
			Where AbrevIs = CsPrazE.Abrevs

EndScan

ThisForm.Release
ENDPROC

PROCEDURE Valid
Private pIni, pFim

If Not Empty(This.Value) And (This.peAntvalue <> This.Value)
	ThisForm.Parentform.Podatamgr.CursorQuery('SigCdOpe', 'LocalOpe', 'Dopes', TprMvCab.Dopes, 'ObsPads, ObsInis, ObsFins')
	pIni = Iif(Not Eof('LocalOpe') And Not Empty(LocalOpe.ObsInis), LocalOpe.ObsInis, 000)
	pFim = Iif(Not Eof('LocalOpe') And Not Empty(LocalOpe.ObsFins), LocalOpe.ObsFins, 999)

	lcQuery = [Select Codigos, Descrs ] + ;
			    [From SigCdObs ] + ;
			   [Where Codigos BetWeen ?pIni And ?pFim]

	If (Thisform.parentForm.Podatamgr.SqlExecute(lcQuery, 'crLocalObs') < 1)
		=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (crLocalObs)')
		Return 0
	EndIf
	Select crLocalObs
	Index On Codigos Tag Codigos
	Index On Descrs Tag Descrs

	Set Near On
	If Not Seek(This.Value, 'crLocalObs', 'Codigos')
		m.loBrowse = CreateObject('fwbuscasel', 'crLocalObs', 'Codigos', This.Value, 'Observações')
		m.loBrowse.AddColuna('Codigos', '999', 'Cód.', .t., 'Codigos')
		m.loBrowse.AddColuna('Descrs', '', 'Descrição', .t., 'Descrs')
		m.loBrowse.Show
	Endif
	Set Near Off

	If (Lastkey()=27)
		Return 0
	EndIf

	This.Value = crLocalObs.Codigos
	
	Thisform.ParentForm.Podatamgr.Cursorquery('SigCdObs','CrSigCdObs','Codigos',This.Value)
		
	Select crSigCdObs
	ThisForm.cntObservacao.Get_CodObs.Value = crSigCdObs.Codigos
	If ("#" $ crSigCdObs.Observas) && Completa a observação com a nota do subnível.
		ObsPadrao = Substr(crSigCdObs.Observas, 1, At("#", crSigCdObs.Observas) - 1)

		Select TmpOperacao
		Scan
			ObsPadrao = ObsPadrao + " " + Notas
		Endscan

		ObsPadrao = ObsPadrao + Substr(crSigCdObs.Observas,At("#",crSigCdObs.Observas)+1)
		
		ThisForm.cntObservacao.get_Obs.Value = ObsPadrao
	Else
		ThisForm.cntObservacao.get_Obs.Value = crSigCdObs.Observas
	Endif
	This.refresh
EndIf
ENDPROC
