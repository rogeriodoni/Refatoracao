------------------------------------------------------------
* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
LPARAMETERS lcacesso

With ThisForm
	.poDatamgr = CreateObject('fSqlConector')
	With .poDatamgr
		.AddCursor('SigCdTam','cods','crSigCdTam','',ThisForm.pagina.lista.grade)
		.pfSqlTabela(1).pColuna('cods','','','Código',50,.t.)
		.pfSqlTabela(1).pColuna('descs','','','Descrição',150,.t.)
		.pfSqlTabela(1).pColuna('varias','','','Variação de Preço',100,.t.)
	EndWith
	= DoDefault()
	.LockScreen = .f.
EndWith


ENDPROC

------------------------------------------------------------
* PROCEDURE msv_procurar
------------------------------------------------------------
PROCEDURE msv_procurar
Local lcBusca

If !IsEmpty(ThisForm.Pagina.Dados.getcods.Value)
	ThisForm.poDataMgr.LocateCursor('crSigCdTam','cods',ThisForm.Pagina.Dados.Getcods.Value)
Else
	If .Not.IsEmpty(ThisForm.Pagina.Dados.Getdescs.Value)
		ThisForm.poDataMgr.LocateCursor('crSigCdTam','descs',ThisForm.Pagina.Dados.Getdescs.Value)
	EndIf
EndIf

ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Lista.Grupo_op.Click
------------------------------------------------------------
PROCEDURE Pagina.Lista.Grupo_op.Click
parameters lcOpcao

If !dodefault( lcopcao )
	ThisForm.mAtivapagina1
	Return
Endif

do case
case thisform.pcEscolha='INSERIR' or ;
	thisform.pcEscolha='PROCURAR'
	thisform.Pagina.Dados.getcods.setfocus
case thisform.pcEscolha='ALTERAR'
	thisform.pagina.dados.getdescs.setfocus
case thisform.pcEscolha='CONSULTAR' or ;
	thisform.pcEscolha='EXCLUIR'
	thisform.pagina.dados.grupo_salva.salva.setfocus
endcase
	

ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
if inlist(thisform.pcEscolha,'INSERIR')
    if IsEmpty(thisform.Pagina.Dados.getcods.value)
       messagebox('Tamanho Inválido !!',0+48,'')
       thisform.Pagina.Dados.getcods.setfocus
		return 	
	endif	
	
	if ThisForm.poDataMgr.ChkRegister('SigCdTam','Cods',ThisForm.Pagina.Dados.GetCods.Value)
       messagebox('Tamanho já Cadastrado !!',0+48,'')
       thisform.Pagina.Dados.getcods.setfocus
      	return
	endif			
endif
DoDefault()
Thisform.mAtivapagina1

ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Dados.Grupo_Salva.Cancelar.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Cancelar.Click
DoDefault()
if thisform.plCancelar
	thisform.mAtivapagina1
endif
ENDPROC
