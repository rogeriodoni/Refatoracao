* PROCEDURE sqldados

PROCEDURE sqldados
Private pDtI, pDtF, pGru, pCta

With this.pagina.lista
	pGru = thisform.antgru
	pCta = Alltrim(.getConta.Value)
	pDtI = fDtoSQL(.getDtInicial.Value)
	pDtF = fDtoSQL(.getDtFinal.Value, '23:59:59')

	ThisForm.poDataMgr.Requery('crSigCdVis')
	
	Select crSigCdVis
	index on pkidchave tag pkidchave
*!*		Go Top
EndWith

ThisForm.LockScreen = .t.
=fCarregaGrade(ThisForm.Pagina.Lista.Grade)
ThisForm.LockScreen = .f.
* Fim **************************************************************************************
ENDPROC

------------------------------------------------------------
* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
Private pGru, pCta, pDtI, pDtF

pDtI = Date()
pDtF = Date()

With ThisForm
	.poDataMgr = CreateObject('fSqlConector', ThisForm.Name)

	** Selecionando Grupo Padrao de Representantes **********************************************
	lcQryPar = [Select GrPadVens From SigCdPam]	

	If (.poDataMgr.SqlExecute(lcQryPar, 'crSigCdPam') < 1)
		=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (LocalMccr)')
		Return 0
	EndIf
	
	pGru = Alltrim(crSigCdPam.GrPadVens)
	thisform.antgru = pGru
	** Fim ***************************************************************************************
	
	** Verifica liberação de senha ***************************************************************
	libConta = fChecaAcesso('SIGCDVIS','ALTCONTA') 
	If not libConta
	
		Do Form SigOpCks With 'VALIDACESSO','Validação de Acesso' To _Retorno
		llOk = ( Left( _Retorno, 1 ) = "*" )
		If !llOk
			MessageBox('Autorização não Confirmada!!!',0+48,'')
			pCta = []
			Return .f.
		Else 	
			ThisForm.UsuVendedor = Substr( _Retorno, 2 )
		
			lcQryCta = [Select iclis,rclis From SigCdCli ] + ;
					[Where rclis = '] + thisform.Usuvendedor + [' AND Grupos = '] + pGru + [']
					
			If (.poDataMgr.SqlExecute(lcQryCta, 'crSigCdCli') < 1)
				=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (LocalMccr)')
				Return 0
			EndIf 
			*** Verifica se existe conta cadastrada para esse usuario **************************
			If Empty(crSigCdCli.rclis)&& = []
				=MessageBox('Conta não cadastrada no Cadastro de Contas de Representantes', 0+256, 'Cadastro de Visitas')
				Return .f.				
			EndIf 
			** Fim *****************************************************************************
			
			pCta = Alltrim(crSigCdCli.iclis)
			pDCta = Alltrim(crSigCdCli.rclis)
		EndIf 
	Else
		ThisForm.UsuVendedor = usuar
		pCta = []
		pDCta = []
	EndIf 
	** Fim ******************************************************************************************
	
	** Query Principal da Grid ********************************************************************************
	lcQryVis = [Select a.*, b.rclis as descri, c.rclis as repr, ] + ;
		[case when a.vendas=1 then 'Sim' else 'Não' end as descvendas, ] + ;
		[case when a.pedidos=1 then 'Sim' else 'Não' end as descpedidos ] + ;
	      [From SigCdVis a ] + ;
	      [Inner Join SigCdCli b on a.iclis=b.iclis ] + ;
	      [Inner Join SigCdCli c on a.Contas=c.iclis ] + ;	      
		 [Where a.Grupos = ?pGru And ] + ;
		 	[(a.Contas = ?pCta Or ?pCta='') And ] +;
		 	   [a.Datas Between ?pDtI And ?pDtF ] + ;
		 [Order By a.pkidchave]			 
	** Fim ***************************************************************************************************
				 
	If (.poDataMgr.pnIdconn > 0)
		.Pagina.Lista.getConta.Value = pCta
		.Pagina.Lista.getdtinicial.Value = pDtI
		.Pagina.Lista.getDtFinal.Value   = pDtF
		.Pagina.Lista.getDesCta.Value = pDCta

		.Pagina.Lista.getConta.enabled = libConta
		.Pagina.Lista.getDesCta.enabled = libConta
		
		
		With .poDataMgr
			.AddCursor('SigCdVis', 'pkIdChave', 'crSigCdVis', '', ThisForm.Pagina.Lista.Grade, lcQryVis)

* PROCEDURE msv_procurar
------------------------------------------------------------
PROCEDURE msv_procurar
Private pGru, pRep, pCta, pDta

pGru = thisform.antgru 
pRep = thisform.pagina.dados.getCodCli.Value 
pCta = thisform.pagina.dados.getCodCta.Value
pDtI = fDtoSQL(thisform.pagina.dados.getData.Value)
pDtf = Iif(Empty(thisform.pagina.dados.getData.Value),Datetime(),fDtoSQL(thisform.pagina.dados.getData.Value, '23:59:59'))

If Used([crTmp])
	Use In CrTmp
EndIf

** Cursor para Procurar ****************************************************
lcQuery = [Select a.*, b.rclis as descri, c.rclis as repr, ] + ;
		[case when a.vendas=1 then 'Sim' else 'Não' end as descvendas, ] + ;
		[case when a.pedidos=1 then 'Sim' else 'Não' end as descpedidos ] + ;
	      [From SigCdVis a ] + ;
	      [Inner Join SigCdCli b on a.iclis=b.iclis ] + ;
	      [Inner Join SigCdCli c on a.Contas=c.iclis ] + ;	      
		 [Where a.Grupos = ?pGru And ] + ;
		 	[(a.contas = ?pRep Or ?pRep='') And ] +;
		 	[(a.iclis = ?pCta Or ?pCta='') And ] +;
		 	   [a.Datas Between ?pDtI And ?pDtF ] + ;
		 [Order By a.pkidchave]	

ThisForm.poDataMgr.SqlExecute(lcQuery, [crTmp])

Select crtmp
thisform.ativaprocura = .T.
* Fim ***********************************************************************
ENDPROC

	EndIf

		[Where iclis = '] + crSigCdVis.contas + [']
		
	If (thisform.poDataMgr.SqlExecute(lcQuery, 'crSigCdCli') < 1)
		=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (LocalMccr)')
		Return 0

		[Where iclis = '] + crSigCdVis.iclis + [']

		=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (LocalMccr)')
		Return 0
	EndIf		
	thisform.pagina.dados.getdesCta.Value = crSigCdCll.rclis	
	
	lcObs = Iif(IsNull(crSigCdVis.obss),[],crSigCdVis.obss)	
	replace obss with lcObs in crSigCdVis
	
	thisform.Refresh 	
EndIf

	thisform.pagina.dados.getdesCta.Value = []
	thisform.pagina.dados.getCodCli.Value = thisform.pagina.lista.getConta.Value
	thisform.pagina.dados.getdesCli.Value = thisform.pagina.lista.getdesCta.Value
