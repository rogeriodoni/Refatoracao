PROCEDURE msv_procurar
ThisForm.poDataMgr.LocateCursor('crSigSyMov','USUARS',ThisForm.Pagina.Dados.getusuars.Value)
ENDPROC

------------------------------------------------------------
* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
LParameters lcAcesso

With ThisForm
	.LockScreen = .T.
	.poDatamgr = CreateObject('fSqlConector',ThisForm.Name)
	If (.poDataMgr.pnIdconn > 0)
		lcQuery = [select a.usuars,a.Dopes,a.Emps,b.ncomps,c.razas,a.cidchaves from SigSyMov a ]+;
				  [Left Join SigCdUsu B ON a.usuars = b.usuarios ]+;
				  [Left Join SigCdEmp C ON a.emps = c.cemps ]			  
				  
		With .poDatamgr
			.AddCursor('SigSyMov','cidchaves','crSigSyMov','',ThisForm.pagina.lista.grade,lcQuery)
			.pfSqlTabela(1).pColuna('usuars','','','Usuário'           ,080,.T.)
			.pfSqlTabela(1).pColuna('Dopes'  ,'','','Operação'			,150,.T.)
			.pfSqlTabela(1).pColuna('Emps'  ,'','','Emp'				,31,.T.)			
		EndWith
		= DoDefault()
	Else
		=Messagebox('Impossível Efetuar Conexão Com o Servidor de Banco de Dados...', 0+48, 'CreateObject')
		Return .f.
	EndIf
	.LockScreen = .F.	
EndWith
ENDPROC

------------------------------------------------------------
* PROCEDURE pagina.lista.grupo_op.Click
------------------------------------------------------------
PROCEDURE pagina.lista.grupo_op.Click
LParameters lnOpcao
lnOpcao = Iif(Type("lnOpcao")=='N',lnOpcao,This.Value)

If !dodefault( lnOpcao )
	ThisForm.mAtivapagina1
	Return
Endif


Do Case && Ativa o Foco do mouse, de acordo com o Tipo de Escolha
	Case INLIST(ThisForm.pcEscolha,'INSERIR','PROCURAR')
		ThisForm.Pagina.Dados.getUsuars.SetFocus
		ThisForm.pagina.dados.getDopes.Enabled=.f.
		ThisForm.pagina.dados.getDopes.Refresh
		
	Case INLIST(ThisForm.pcEscolha,'CONSULTAR','EXCLUIR')	
		This.Parent.Parent.dados.grupo_salva.salva.SetFocus 			
EndCase


If ThisForm.pcescolha='PROCURAR'
	ThisForm.pagina.dados.getDEmps.Enabled=.f.
	ThisForm.pagina.dados.getDUsuars.Enabled=.f.
	
	ThisForm.pagina.dados.getDEmps.Refresh
	ThisForm.pagina.dados.getDUsuars.Refresh
EndIf


If INLIST(ThisForm.pcEscolha,'ALTERAR','CONSULTAR','EXCLUIR') 
	ThisForm.pagina.dados.getEmps.SetFocus
	ThisForm.pagina.dados.getDUsuars.Enabled=.f.

	If Empty(ThisForm.pagina.dados.getUsuars.Value)
		ThisForm.pagina.dados.getDopes.Enabled=.f.
	EndIf

	If IsNull(ThisForm.pagina.dados.getDUsuars.Value)
		ThisForm.pagina.dados.getDUsuars.Value=Nvl(ThisForm.pagina.dados.getDUsuars.Value,Space(40))
	EndIf
	
	ThisForm.pagina.dados.getDUsuars.Refresh
EndIf
ENDPROC

------------------------------------------------------------
* PROCEDURE pagina.dados.grupo_salva.salva.Click
------------------------------------------------------------
PROCEDURE pagina.dados.grupo_salva.salva.Click
*Verifica se o registro já existe
If !Empty(ThisForm.pagina.dados.getUsuars.Value)
	If ThisForm.pcEscolha = 'INSERIR' And ThisForm.poDataMgr.ChkRegister('SigSyMov','usuars',ThisForm.Pagina.Dados.getUsuars.value)
		Messagebox( 'Usuário já Cadastrado!', 48, 'Atenção' )
		ThisForm.Pagina.Dados.getUsuars.SetFocus
		Return
	Else
	*Faz Validacao dos Campos, nas Seguinte Operacoes
		If InList(ThisForm.pcEscolha,'INSERIR','ALTERAR')
			If Empty(crSigSyMov.Emps)
				Messagebox( 'A Empresa precisa ser Informada...', 48, 'Atenção' )
				ThisForm.Pagina.Dados.getemps.SetFocus	
				Return
			Else
				If Empty(crSigSyMov.Dopes)
					Messagebox( 'A Operação precisar ser Informado...', 48, 'Atenção' )
					ThisForm.Pagina.Dados.getDopes.SetFocus				
					Return
				EndIf
			EndIf
		EndIf
	EndIf
Else
	If Empty(ThisForm.pagina.dados.getEmps.Value)
		Messagebox( 'A Empresa precisa ser Informada...', 48, 'Atenção' )
		ThisForm.Pagina.Dados.getemps.SetFocus
		Return
	EndIf
	lcSql = [Select * From SigSyMov Where Emps=']+ThisForm.Pagina.Dados.getEmps.Value+[' And Dopes = '  ' ]
	=ThisForm.Podatamgr.Sqlexecute(lcSql,'LocMOV')
	If ThisForm.pcEscolha = 'INSERIR' And Reccount('LocMov') > 0
		Messagebox( 'Empresa já Cadastrada com esse Usuário/Operação!', 48, 'Atenção' )
		ThisForm.Pagina.Dados.getEmps.SetFocus
		Return
	EndIf
EndIf

If ThisForm.pcEscolha = 'INSERIR'
	Replace cidChaves with fUniqueIds() IN crSigSyMov
EndIf		
= DoDefault()
ThisForm.mAtivaPagina1
ENDPROC
