PROCEDURE msv_procurar
Local lcEmp, lcTab, lcCar

lcEmp = ThisForm.Pagina.Dados.getEmpresa.Value
lcTab = ThisForm.Pagina.Dados.getTabDes.Value
lcCar = ThisForm.Pagina.Dados.getCargo.Value

Select crSigCdCmi
Set Near On
Do Case
	Case Not Empty(lcEmp) And Not Empty(lcTab) And Not Empty(lcCar)
		 Locate For Emps = lcEmp And TabDes = lcTab And Cargos = lcCar

	Case Not Empty(lcEmp) And Not Empty(lcTab)
		 Locate For Emps = lcEmp And TabDes = lcTab

	Case Not Empty(lcEmp)
		 Locate For Emps = lcEmp
EndCase
Set Near Off
* Fim **************************************************************************************
ENDPROC

* PROCEDURE Pagina.Lista.Grupo_op.Click
------------------------------------------------------------
PROCEDURE Pagina.Lista.Grupo_op.Click
Lparameters Opcao

If Not DoDefault(Opcao)
	ThisForm.mAtivapagina1
	Return 0
EndIf

Select crSigCdCmi

With ThisForm.Pagina.Dados
	.getDEmpresa.Value = ''
	.getDCargo.Value   = ''
	.getDTabDes.Value  = ''

	If (ThisForm.pcEscolha <> 'INSERIR')
		If ThisForm.poDataMgr.CursorQuery('SigCdEmp', 'crTmp', 'Cemps', .getEmpresa.Value)
			Go Top In crTmp
			.getDEmpresa.Value = Iif(Eof('crTmp'), '', crTmp.Razas)
			Use In crTmp
		EndIf

		If ThisForm.poDataMgr.CursorQuery('SigCdCrg', 'crTmp', 'Ccargs', .getCargo.Value)
			Go Top In crTmp
			.getDCargo.Value = Iif(Eof('crTmp'), '', crTmp.DCargs)
			Use In crTmp
		EndIf

		If ThisForm.poDataMgr.CursorQuery('SigOpTdz', 'crTmp', 'Codigos', .getTabDes.Value)
			Go Top In crTmp
			.getDTabDes.Value = Iif(Eof('crTmp'), '', crTmp.Descrs)
			Use In crTmp
		EndIf
	Endif
	.getDEmpresa.Refresh
	.getDCargo.Refresh
	.getDTabDes.Refresh

	Do Case
		Case InList(ThisForm.pcEscolha, 'PROCURAR', 'ALTERAR', 'INSERIR')
			.getEmpresa.SetFocus	

		Case InList(ThisForm.pcEscolha, 'CONSULTAR', 'EXCLUIR')
			.Grupo_Salva.Salva.SetFocus
	EndCase
	.Refresh
EndWith

Select crSigCdCmi

ThisForm.Pagina.Dados.Activate
* Fim **************************************************************************************
ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
Private pDtI, pDtF

With ThisForm.Pagina.Dados
	If (ThisForm.pcEscolha = 'INSERIR')
		pDtI = crSigCdCmi.DtInis
		pDtF = crSigCdCmi.DtFins

		lcSQLCmd = [Select Emps, Cargos, TabDes, DtInis, DtFins ] + ;
					 [From SigCdCmi ] + ;
					[Where Emps = ']   + crSigCdCmi.Emps   + [' And ] + ;
						  [Cargos = '] + crSigCdCmi.Cargos + [' And ] + ;
						  [FPags  = '] + crSigCdCmi.FPags  + [' And ] + ;
						  [TabDes = '] + crSigCdCmi.TabDes + [' And ] + ;
						  [DtInis = ?pDtI And ] + ;
						  [DtFins = ?pDtF]
					 
		If (ThisForm.poDataMgr.SQLExecute(lcSQLCmd, 'crTmp') < 1)
			=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão')
			Return 0
		EndIf
		Go Top In crTmp
		If Not Eof('crTmp')
			=MessageBox('Dados Já Cadastrados!!!', 48, '')
			.getEmpresa.SetFocus
			Return 0
		EndIf
	EndIf

	If Inlist(ThisForm.pcEscolha, 'INSERIR', 'ALTERAR')
		If Empty(.getTabDes.Value)
			=MessageBox('Tabela de Descontos Inválida!!!', 48, '')
			.getTabDes.SetFocus
			Return 0
		EndIf 

		If Empty(.getCargo.Value)
			=MessageBox('Cargo Inválido!!!', 48, '')
			.getCargo.SetFocus
			Return 0
		EndIf 

		If (.getComiss.Value <= 0)
			=MessageBox('Comissão Inválida!!!', 48, '')
			.getComiss.SetFocus
			Return 0
		EndIf 

		If Empty(.getDtIni.Value)
			=MessageBox('Data Inicial Inválida!!!', 48, '')
			.getDtIni.SetFocus
			Return 0
		EndIf 

		If Empty(.getDtFin.Value)
			=MessageBox('Data Final Inválida!!!', 48, '')
			.getDtFin.SetFocus
			Return 0
		EndIf 

		If (.getDtIni.Value > .getDtFin.Value)
			=MessageBox('Data Inicial Não Pode Ser Maior do Que a Final!!!', 48, '' )
			.getDtIni.SetFocus
			Return 0
		EndIf 

		If Empty(crSigCdCmi.DtInis)
			Replace DtInis With .Null. In crSigCdCmi
		EndIf
		
		If Empty(crSigCdCmi.DtFins)
			Replace DtFins With .Null. In crSigCdCmi
		EndIf
		
		If Empty(crSigCdCmi.cIdChaves)
			Replace cIdChaves With fUniqueIds() In crSigCdCmi
		EndIf
		Replace DtAlts With Datetime() In crSigCdCmi
	EndIf 
EndWith

=DoDefault()

ThisForm.mAtivapagina1
* Fim **********************************************************************************
ENDPROC

PROCEDURE Valid
Local loLista

With ThisForm.Pagina.Dados
	If Not Empty(This.Value)
		loLista = CreateObject('fwBuscaExt', ThisForm.poDataMgr.pnIdConn, 'SigOpTdz', ;
							   'crListaRemota', 'Descrs', This.Value, 'Seleção', .t.)

		If Not loLista.plAchouRegistro
			loLista.mAddColuna('Descrs',  '', 'Descrição')
			loLista.mAddColuna('Codigos', '', 'Código')
			loLista.Show()
		EndIf
		
		.getTabDes.Value  = crListaRemota.Codigos
		.getDTabDes.Value = crListaRemota.Descrs
		Use In crListaRemota
	Endif
	.getTabDes.Refresh
	.getDTabDes.Refresh
EndWith
* Fim **********************************************************************************

ENDPROC
