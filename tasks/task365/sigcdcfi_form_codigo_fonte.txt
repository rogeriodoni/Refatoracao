LPARAMETERS lcacesso

With ThisForm
	.poDatamgr = CreateObject('fSqlConector')
	With .poDatamgr
		.AddCursor('SigCdcfi','cidchaves','crSigCdcfi','grupos,ests,tpicms',ThisForm.pagina.lista.grade)

		.pfSqlTabela(1).pColuna('ests','','','Estado',40,.t.)
		.pfSqlTabela(1).pColuna('icms','','','Aliq. (%)',60,.t.)
		.pfSqlTabela(1).pColuna('tpicms','','','Tipo',40,.t.)
		.pfSqlTabela(1).pColuna('tpTribs','','','Tipo Trib.',80,.T.)
		.pfSqlTabela(1).pColuna('cfops','','','CFOP',60,.T.)
		.pfSqlTabela(1).pColuna('issqn','','','Aliq Iss',60,.T.)
		.pfSqlTabela(1).pColuna('itemls','','','Item Lista Serv',100,.T.)
		.pfSqlTabela(1).pColuna('codtrib','','','Cod Tributação',130,.T.)
		.pfSqlTabela(1).pColuna('natope','','','Nat Oper',40,.T.)
		.AddCursor('SigCdGrp','cgrus','crSigCdGrp','','')
		.AddCursor('SigCdUfs','estados','crSigCdUfs','','')
		.AddCursor('SigFiTri','tipos','crSigFiTri','','')
	EndWith
	= DoDefault()
	.LockScreen = .f.
EndWith
ENDPROC

------------------------------------------------------------
* PROCEDURE msv_procurar
------------------------------------------------------------
PROCEDURE msv_procurar
Local laCampos(2,2), Grupo, Estado


Grupo  = ThisForm.Pagina.Dados.getCgru.Value
Estado = ThisForm.Pagina.Dados.GetEstado.Value

Do Case
Case !IsEmpty(Grupo) and !IsEmpty(Estado)
	m.laCampos(1,1) = 'grupos'
	m.laCampos(1,2) = grupo
	m.laCampos(2,1) = 'ests'
	m.laCampos(2,2) = estado
	ThisForm.poDataMgr.LocateCursor('crSigCdcfi',@m.laCampos)
		
Case !IsEmpty(Grupo) and IsEmpty(Estado)
	
	ThisForm.poDataMgr.LocateCursor('crSigCdcfi','grupos',grupo)	

Case IsEmpty(Grupo) and !IsEmpty(Estado)
	
	ThisForm.poDataMgr.LocateCursor('crSigCdcfi','ests',estado)	

EndCase

thisform.Pagina.Lista.Grade.refresh

ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Lista.Grupo_op.Click
------------------------------------------------------------
PROCEDURE Pagina.Lista.Grupo_op.Click
Lparameters lnOpcao

If !dodefault( lnOpcao )
	ThisForm.mAtivapagina1
	Return
Endif

If !thisform.pcEscolha$'INSERIR PROCURAR'
	If Thisform.PoDatamgr.SQLExecute([Select Dgrus from SigCdGrp where Cgrus = '] + CrSigCdcfi.Grupos + [' ],'CrSigCdGrp') < 0
		MessageBox('Favor reinicializar o processo.',16,'Falha na Conexão')
		Return(.F.)
	Else
		This.Parent.Parent.Dados.getDgru.Value = CrSigCdGrp.Dgrus
		This.Parent.Parent.Dados.getDgru.Refresh
	Endif	
Endif

Do Case
	Case thisform.pcEscolha$'INSERIR PROCURAR'
		This.Parent.Parent.Dados.getDgru.Value = Space(40)
		This.Parent.Parent.Dados.getCgru.setfocus
	Case thisform.pcEscolha='ALTERAR'
		This.Parent.Parent.Dados.Get_Icm.setfocus
	Case thisform.pcEscolha$'CONSULTAR EXCLUIR'
		This.Parent.Parent.Dados.Grupo_Salva.Salva.setfocus
EndCase

ENDPROC

local lacampos(3,2)
