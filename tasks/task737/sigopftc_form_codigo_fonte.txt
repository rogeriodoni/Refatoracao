PROCEDURE carregafichatecnica
* Rotina Para Gravação dos Dados da Ficha Técnica no Cursor Para Atualizar a Base
Local lcVal, lcObj, loObjForm1

If ThisForm.CriouForm
	loObjForm1 = ThisForm.poWindowOpe.ScrollBarContainer.ScrollSlide

	Select crTmpFicha
	Scan
		If( crTmpFicha.cTpDados <> [D] )
			lcVal = Nvl(crTmpFicha.Resps, [])
		Else
			lcVal = Nvl(Ctod(crTmpFicha.Resps),Ctod(''))
		Endif			

		lcObj = [memFicha] + Alltrim(Str(crTmpFicha.COrds))

		loObjForm1.&lcObj..Value = lcVal
		loObjForm1.&lcObj..Refresh
	EndScan
EndIf
ENDPROC

------------------------------------------------------------
* PROCEDURE checarfichatecnica
------------------------------------------------------------
PROCEDURE checarfichatecnica
Lparameters pTip
Local lcArea

lcArea = Alias()

llOks = .f.

If ((pTip = [GRUPO_OP]) And (ThisForm.pcEscolha <> [INSERIR]))
	llOks = .t.
EndIf

If Not llOks And InList(ThisForm.pcEscolha, [INSERIR], [ALTERAR], [CONSULTAR]) And ThisForm.PrimeiraEntrada And ThisForm.CriouForm
	This.DesmontaFichaTecnica 
	ThisForm.PrimeiraEntrada = .f.
	llOks = .t.
EndIf

If llOks
	If Not Empty(ThisForm.Operacao)		
		lcQuery = [Select Distinct a.Dopes, a.TitOpes, b.CGrus as TemDados ] + ;
		   			[From SigCdOpe a ] + ;
		   			[Left Join SigCdFtc b ] + ;
		   	 		  [On a.Dopes = b.CGrus ] + ;
		   	 	   [Where a.Dopes = '] + Padr(ThisForm.Operacao, 20) + [' And b.nTipos = 2 ] + ;
		   		   [Order By a.Dopes]

		If (ThisForm.poDataMgr.SqlExecute(lcQuery, 'cnSigCdOpe') < 1)
			=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (cnSigCdOpe)')
			Return .f.
		EndIf		

		Select cnSigCdOpe
		Go Top		
		If Not This.MontaFichaTecnica(ThisForm.Operacao)
			Select &lcArea
			Return 0
		EndIf
	EndIf
EndIf
If !Empty(lcArea)
	Select &lcArea
Endif	
ENDPROC

PROCEDURE desmontafichatecnica
* Rotina Para Remover os Campos Criados no Formulário de Ficha Técnica
Local lcCar, lcObj, loObjForm1

If ThisForm.CriouForm And Used('crTmpFicha') And (ThisForm.pcEscolha <> [PROCURAR]) And ThisForm.MontouCampos
	loObjForm1 = ThisForm.poWindowOpe.ScrollBarContainer.ScrollSlide
	lcCar = []

	Select crTmpFicha
	Scan
		If (lcCar <> crTmpFicha.CCars)
			lcCar = crTmpFicha.CCars

			* Destruindo o Shape e Seu Respectivo Label
			lcObj = [sayFicha] + Alltrim(Str(crTmpFicha.COrds))
			loObjForm1.RemoveObject(lcObj)

			lcObj = [shpFicha] + Alltrim(Str(crTmpFicha.COrds))
			loObjForm1.RemoveObject(lcObj)
		EndIf

		* Destruindo o Label
		lcObj = [lblFicha] + Alltrim(Str(crTmpFicha.COrds))
		loObjForm1.RemoveObject(lcObj)

		* Destruindo o Memo
		lcObj = [memFicha] + Alltrim(Str(crTmpFicha.COrds))
		loObjForm1.RemoveObject(lcObj)
	EndScan

	If (Reccount('crTmpFicha') > 0)
		lcObj = [shpFichaFim]
		loObjForm1.RemoveObject(lcObj)

		lcObj = [sayFichaFim]
		loObjForm1.RemoveObject(lcObj)
	EndIf

	ThisForm.MontouCampos = .f.
EndIf
* Fim *********************************************************************************
ENDPROC

* PROCEDURE gravafichatecnica
------------------------------------------------------------
PROCEDURE gravafichatecnica
* Rotina Para Gravação dos Dados da Ficha Técnica no Cursor Para Atualizar a Base
Local loObjForm1

If ThisForm.CriouForm And Used('crTmpFicha')
	loObjForm1 = ThisForm.poWindowOpe.ScrollBarContainer.ScrollSlide

	Zap in crSigCdCeT

	Select crTmpFicha
	Scan
		lcObj = [memFicha] + Alltrim(Str(crTmpFicha.COrds))

		lcVal = loObjForm1.&lcObj..Value
		If( crTmpFicha.cTpDados == [D] )
			lcVal = Dtoc(lcVal)
		Endif

		If Not Empty(lcVal)
			Select crSigCdCeT
			Go Top In crSigCdCeT
			Locate For IClis + CCars + CTits = crSigCdCli.IClis + crTmpFicha.CCars + crTmpFicha.CTits
			If Eof([crSigCdCeT])
				Insert Into crSigCdCeT (IClis, CCars, CTits, Resps, CliCarTits, cIdChaves) ;
								Values (crSigCdCli.IClis, crTmpFicha.CCars, crTmpFicha.CTits, lcVal, ;
										crSigCdCli.IClis + crTmpFicha.CCars + crTmpFicha.CTits, NewIdChave())
			EndIf 
		EndIf 
	EndScan
EndIf

Return(.t.)
* Fim *********************************************************************************
ENDPROC

* PROCEDURE montafichatecnica
------------------------------------------------------------
PROCEDURE montafichatecnica
* Rotina Para Criar os Campos no Formulário de Ficha Técnica
Lparameters pDop
Local lcDopes, lcQuery, lnTop, lnLef, lnWid, lcCar, lcObj, loObjForm1, lcArea, lcPerg
lcArea = Alias()

If ThisForm.CriouForm And (ThisForm.pcEscolha <> [PROCURAR])
	lcDopes = Padr(pDop, 20)

	llFaz = .t.		
	If llFaz
		lcQuery = [Select a.CCars, a.CTits, a.CLins, a.COrds, a.Maxs, a.Mins, a.nChkCxas, a.cSelecaos, a.cTpDados, b.Resps, c.Descrs ] + ;   
					[From SigCdFtc a ] + ;  
				    [Left Join SigOpFtc b On '] + lcDopes + [' + a.CCars + a.CTits = b.DopCarTits ] + ;  
				    [Left Join SigPrCpP c on a.CTits = c.Codigos ] + ;  
				   [Where a.CGrus = '] + lcDopes + [' ] + ;
				   [And a.SGrus = Space(6) And a.nTipos = 2 ] + ;  
				   [Order By a.COrds]

		If (ThisForm.poDataMgr.SqlExecute(lcQuery, 'crTmpFicha') < 1)
			=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (crTmpFicha)')
			Return .f.
		EndIf
	EndIf

	loObjForm1 = ThisForm.poWindowOpe.ScrollBarContainer.ScrollSlide
	lnTop = 1
	lnLef = 6
	lnWid = ThisForm.poWindowOpe.Width - 14
	lcCar = []

	ThisForm.MontouCampos = .t.

	Select crTmpFicha
	Scan
		If (lcCar <> crTmpFicha.CCars)
			lcCar = crTmpFicha.CCars
			lnTop = lnTop + 1

			* Adicionando o Shape e Seu Respectivo Label
			lcObj = [sayFicha] + Alltrim(Str(crTmpFicha.COrds))
			loObjForm1.AddObject(lcObj, [Say])

			With loObjForm1.&lcObj.
				.Visible   = .t.
				.Top	   = lnTop
				.Left	   = 1
				.Caption   = Proper(lcCar)
				.FontBold  = .t.
				.FontSize  = 9
				.ForeColor = Rgb(0,0,128)
				.AutoSize  = .t.

				lnTop = lnTop + .Height
			EndWith 

			lcObj = [shpFicha] + Alltrim(Str(crTmpFicha.COrds))
			loObjForm1.AddObject(lcObj, [Shape])

			With loObjForm1.&lcObj.
				.Visible   	   = .t.
				.Top	       = lnTop
				.Left	       = 1
				.Width	       = lnWid + 9
				.Height    	   = 2
				.SpecialEffect = 0

				lnTop = lnTop + .Height + 2
			EndWith 
		EndIf
		
		* Adicionando o Label
		lcObj = [lblFicha] + Alltrim(Str(crTmpFicha.COrds))
		loObjForm1.AddObject(lcObj, [Say])

		With loObjForm1.&lcObj.
			.Visible   = .t.
			.Top	   = lnTop
			.Left	   = lnLef + 4
			.Caption   = Proper(Iif(!IsEmpty(crTmpFicha.Descrs),crTmpFicha.Descrs,crTmpFicha.CTits))
			.FontBold  = .t.
			.FontSize  = 8
			.ForeColor = Rgb(0,0,128)
			.AutoSize  = .t.
			lcPerg     = .Caption
		EndWith 
		lnTop = lnTop + 12

		* Adicionando o Memo
		lnHei = Iif(crTmpFicha.CLins = 1, 23, ((crTmpFicha.CLins - 1) * 16) + 23)

		lcObj = [memFicha] + Alltrim(Str(crTmpFicha.COrds))

		If( crTmpFicha.cTpDados <> [D] )	
			llRespFixa = .F.
			lcQryResp = [Select nResps from SigCdFtr where GruCarTits = ']+lcDopes + crTmpFicha.CCars + crTmpFicha.CTits+[']
			If ThisForm.poDataMgr.SQLExecute(lcQryResp,[crResp]) < 1
				=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (crResp)')
				Return .f.				
			EndIf
			If !Eof([crResp]) and crResp.nResps == 1
				llRespFixa = .T.
			Endif
		
			lcMsk = Replicate([x], 100)
		
			loObjForm1.AddObject(lcObj, [fweditbox], ThisForm.poDataMgr.pnIdConn, [SigCdFtr], ;
									    [Resps], lcMsk, [Respostas], [GruCarTits], ;
									    lcDopes + crTmpFicha.CCars + crTmpFicha.CTits, ;
									    ThisForm.pcEscolha, 254, .T.,lcPerg)
			With loObjForm1.&lcObj.
				.Visible   = .t.
				.Top	   = lnTop
				.Left	   = lnLef
				.Width	   = lnWid
				.Height    = lnHei
				.ReadOnly  = Not InList(ThisForm.pcEscolha, 'INSERIR', 'ALTERAR') or llRespFixa
				.pnCaixa   = crTmpFicha.nChkCxas
				.plSelecao = (crTmpFicha.cSelecaos = [S])

				If (crTmpFicha.Maxs > 0)
					.MaxLength = crTmpFicha.Maxs
				EndIf
			EndWith 
		Else
			lcMsk = [99/99/9999]
			loObjForm1.AddObject(lcObj, [fwget])
			With loObjForm1.&lcObj.
				.Visible   = .T.
				.Top       = lnTop
				.Left      = lnLef
				.Width     = 80
				.Height    = lnHei
				.ReadOnly  = Not InList(ThisForm.pcEscolha, 'INSERIR', 'ALTERAR')
				.MaxLength = 10
				.InputMask = lcMsk
				.Format    = [E]
				.Value     = Ctod([])
			EndWith						
		Endif			

		lnTop = lnTop + lnHei + 2
	EndScan

	If (Reccount('crTmpFicha') > 0)
		lcObj = [shpFichaFim]
		loObjForm1.AddObject(lcObj, [Shape])

		With loObjForm1.&lcObj.
			.Visible   	   = .t.
			.Top	       = lnTop
			.Left	       = 1
			.Width	       = lnWid + 9
			.Height    	   = 2
			.SpecialEffect = 0
		EndWith 

		lcObj = [sayFichaFim]
		loObjForm1.AddObject(lcObj, [Say])

		With loObjForm1.&lcObj.
			.Visible   = .t.
			.Top	   = lnTop + 100
			.Left	   = 1
			.Caption   = [.]
			.ForeColor = ThisForm.BackColor
			.AutoSize  = .t.
		
			lnTop = (lnTop + 100 + .Height)
		EndWith 
	EndIf
	loObjForm1.Top = 0
	loObjForm1.Height = (lnTop + 100)
 
 	If loObjForm1.ControlCount > 0
		ThisForm.poWindowOpe.ScrollBarContainer.VerticalScrollBar.SetSlideTop(0, loObjForm1.Controls(1) ) && seta scroll para o inicio da barra vertical...
	EndIf 

	This.CarregaFichaTecnica
EndIf
Select &lcArea
Return .t.
* Fim *********************************************************************************
ENDPROC

PROCEDURE validafichatecnica
Local loObjForm1

loObjForm1 = ThisForm.poWindowOpe.ScrollBarContainer.ScrollSlide

If Used('crTmpFicha')
	Select crTmpFicha
	Scan
		lcObj = [memFicha] + Alltrim(Str(crTmpFicha.COrds))

		lcVal = loObjForm1.&lcObj..Value
		If( crTmpFicha.cTpDados == [D] )
			lcVal = Dtoc(lcVal)
		Endif

		If Len(Alltrim(lcVal)) < crTmpFicha.Mins
			lcMsgs = 'O Titulo ' + Alltrim(crTmpFicha.CTits) + ;
			         ' da Caracteristica ' + Alltrim(crTmpFicha.CCars) + Chr(13) + Chr(10) +;
			         'deve conter no minimo ' + Alltrim(Str(crTmpFicha.Mins)) + ' caracteres.'
			MessageBox(lcMsgs,16,'')
			Return(.f.)
		EndIf
	EndScan
Endif	

Return(.T.)
ENDPROC
