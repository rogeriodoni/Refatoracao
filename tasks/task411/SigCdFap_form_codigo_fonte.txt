* PROCEDURE Init

PROCEDURE Init
Lparameters _Tipo

ThisForm.Tipo = Alltrim(Iif(Type('_Tipo') = 'C', _Tipo, 'F'))

lcQryFape = [Select * From SigCdFea Where Emps = '] + _Empr + [' And Tipos = '] + ThisForm.Tipo + [']

With ThisForm
	.poDataMgr = CreateObject('fSqlConector', ThisForm.Name)

	If (.poDataMgr.pnIdconn > 0)
		With .poDataMgr
			.AddCursor('SigCdFea', 'cIdChaves', 'crSigCdFea', '', ThisForm.Pagina.Lista.Grade, lcQryFape)
			.pfSqlTabela(1).pColuna('Numes',  '', '', 'Código', 	     052, .t.)
			.pfSqlTabela(1).pColuna('Datas',  '', '', 'Data',   	     075, .f.)
			.pfSqlTabela(1).pColuna('IFors',  '', '', 'Recuperadora',   085, .f.)
			.pfSqlTabela(1).pColuna('Grupos', '', '', 'Setor', 	     085, .f.)
			.pfSqlTabela(1).pColuna('Contas', '', '', 'Funcionário',    085, .f.)
			.pfSqlTabela(1).pColuna('TpMats', '', '', 'Tipo Material',  120, .f.)
			.pfSqlTabela(1).pColuna('Brutos', '', '', 'Peso Bruto (g)', 090, .f.)
		EndWith

		ThisForm.poDataMgr.CursorQuery('SigCdGcr', 'TmpGccr', 'BalFalPers', 1)
		Select TmpGccr
		Index On Codigos Tag Codigos

		lcQuery = [Select a.IClis, a.RClis ] + ;
					[From SigCdCli a, SigCdGcr b ] + ;
				   [Where a.Grupos = b.Codigos And ] + ;
				   		 [b.BalFalPers = 1]

		If (ThisForm.poDataMgr.SqlExecute(lcQuery, 'TmpCli') < 1)
			=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (TmpCli)')
			Return 0
		EndIf
		Select TmpCli
		Index On Iclis Tag Iclis

		ThisForm.LockScreen = .t.
		Thisform.Name = Alltrim(ThisForm.Name) + ThisForm.Tipo

		=DoDefault()

		Thisform.Pagina.Lista.Grade.SetAll('DynamicForeColor', 'Iif(crSigCdFea.Baixas,RGB(255,0,0),RGB(0,0,0))','Column')
		ThisForm.Pagina.Lista.Grade.SetFocus
		ThisForm.LockScreen = .f.
		ThisForm.Pagina.Lista.Grade.Column1.SetFocus
	Else
		=Messagebox('Impossível Efetuar Conexão Com o Servidor de Banco de Dados...', 0+48, 'CreateObject')
		Return .f.
	EndIf
EndWith
* Fim **************************************************************************************

ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Lista.Grupo_op.Click
------------------------------------------------------------
PROCEDURE Pagina.Lista.Grupo_op.Click
Lparameters Opcao

Local Codigos, llFechado

If (Type('m.Opcao')='N')
	This.Value = m.Opcao 
EndIf

If DoDefault(This.Value)
	ThisForm.LockScreen = .t.

	With This.Parent.Parent.Dados
		ThisForm.poDataMgr.CursorQuery('SigCdGcr', 'LocalGccr', 'Codigos', .getCdGrupo.Value, 'Descrs')
		ThisForm.poDataMgr.CursorQuery('SigCdCli',  'LocalCli',  'IClis',   .getCdCOnta.Value, 'RClis')

		.getDsGrupo.Value = Iif(Eof('LocalGccr'), '', LocalGccr.Descrs)
		.getDsConta.Value = Iif(Eof('LocalCli'),  '', LocalCli.RClis)
	EndWith

	Do Case
		Case InList(ThisForm.pcEscolha, 'INSERIR', 'ALTERAR')
			If (ThisForm.pcEscolha = 'INSERIR')
				Select crSigCdFea
				m.Numes 	= fGerUniqueKey(_Empr + 'SigCdFea')
				m.Emps  	= _Empr
