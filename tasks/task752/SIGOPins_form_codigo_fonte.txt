PROCEDURE Init
lParameters pCnx, pLista, pTipo, pCapTion, pEdn
LOCAL lcLista
public pcLista, pcTipo, pcChv, pcEDN

ThisForm.Caption = 'Seleção de ' + Iif(Type([pCaption])='C' And Not Empty(pCaption),pCaption,'Finalidades')

ThisForm.cntSombra.lblSombra.Caption = ThisForm.Caption
ThisForm.cntSombra.lblTitulo.Caption = ThisForm.Caption

pcLista  = pLista	
pcTipo = pTipo
pcChv = []
pcEDN = Iif(Type([pEdn])=[C],pEdn,[])

This.poDataMgr = pCnx

*** mena 18/02/2015 - cria lista de observações dois ***
IF VARTYPE(pcLista) = 'C'		&& se for para selecionar finalidades	

	*** seleciona a lista de observações ***
	lcQuery = [Select 0 as Selecao, Codigos, Observas, Descrs ] + ;
			    [From SigCdObs ] + ;
			   [Where Codigos BetWeen 0 And 999 ] + ;
			   [Order By Codigos]
	If (ThisForm.poDataMgr.SqlExecute(lcQuery, [LocalObs]) < 1)
		=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (LocalObs)])
		Return .f.
	EndIf

	*** QUANDO VEM DAS INFORMACOES DA OPERACAO - SIGOPINF***
	IF VARTYPE(pcTipo) <> 'C'	
		*** seleciona as observações que já estão associadas à esta operacao ***
		lcLista = ALLTRIM(pcLista)
		lcWhere = IIF(EMPTY(lcLista),' < 0 ', 'in ('+SUBSTR(lclista,2,LEN(lclista)-2)+') ')
		
		lcQuery = [Select Codigos From SigCdObs Where Codigos ] + lcWhere
		If (ThisForm.poDataMgr.SqlExecute(lcQuery, [CrObs]) < 1)
			=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (LocalObs)])
			Return .f.
		EndIf
	ENDIF
	
	*** QUANDO VEM DO CADASTRO DE GRUPOS***
	IF VARTYPE(pcTipo) = 'C' and pcTipo = 'GRP'	
		*** seleciona as observações que já estão associadas a este grupo ***
		SELECT * from crSigOpFnl into cursor crObs 
	ENDIF

	*** QUANDO VEM DA TELA DE ITENS DA OPERAÇÃO - SIGOPPRO***
	IF VARTYPE(pcTipo) = 'C' and pcTipo = 'ITN'	
		pcEDN = crTpmMvItn.Emps + crTpmMvItn.Dopes + str(crTpmMvItn.Numes,6)
		pcChv = str(crTpmMvItn.citens,6)	&& salva o item como chave. O cidchaves é refeito a cada alteração. Não dá para ser usado
		
		lcSql = [Select Cgrus From SigCdPro Where cpros = ']+crTpmMvItn.Cpros+[']
		ThisForm.PodataMgr.SqlExecute(lcSql,'crPro')

		lcQuery = [Select * From sigopFnl Where operacao = 'GRP' and codutil = '] +ALLTRIM(crPro.cgrus)+ [']
		If (ThisForm.poDataMgr.SqlExecute(lcQuery, [CrGRP]) < 1)
			=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (SigOpFnl)])
			Return .f.
		EndIf

		Select Distinct a.Selecao, a.Codigos, Left(a.Observas,100) as Observas, a.Descrs from localObs a, crGRP b where a.codigos = b.codigos into cursor localObs READWRITE

		*** seleciona as observações que já estão associadas a este Item ***

		Select * From crSigopFnl Where codutil = pcChv into cursor crObs
	ENDIF

	*** seta as observações já incluidas ***
	SELECT crObs
	GO top
	SCAN
		SELECT localObs
		LOCATE for codigos = crObs.codigos
		IF FOUND()
			replace selecao with 1
		ENDIF
		SELECT crObs	
	ENDSCAN
	lcObsPads = -1

ELSE
	lcQuery = [Select ObsPads, ObsInis, ObsFins ] + ;
				[From SigCdOpe ] + ;
				[Where Dopes = '] + TprMvCab.Dopes + [']

	If (ThisForm.poDataMgr.SqlExecute(lcQuery, [LocalOpe]) < 1)
		=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (LocalOpe)])
		Return .f.
	EndIf
	Go Top In LocalOpe

	pIni = Iif(Not Eof([LocalOpe]) And Not Empty(LocalOpe.ObsInis), LocalOpe.ObsInis, 000)
	pFim = Iif(Not Eof([LocalOpe]) And Not Empty(LocalOpe.ObsFins), LocalOpe.ObsFins, 999)

	lcQuery = [Select 0 as Selecao, Codigos, Observas, Descrs ] + ;
			    [From SigCdObs ] + ;
			   [Where Codigos BetWeen ?pIni And ?pFim ] + ;
			   [Order By Codigos]

	If (ThisForm.poDataMgr.SqlExecute(lcQuery, [LocalObs]) < 1)
		=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (LocalObs)])
		Return .f.
	EndIf
	lcObsPads = LocalOpe.ObsPads
ENDIF

This.Grade.RecordSource          = [LocalObs]
This.Grade.Column1.ControlSource = [LocalObs.Selecao]
This.Grade.Column2.ControlSource = [LocalObs.Descrs]

This.getObs.ControlSource = [LocalObs.Observas]

Select LocalObs
Go Top In LocalObs
Update LocalObs Set Selecao = 1 Where Codigos = lcObsPads
Go Top In LocalObs

This.Refresh
This.Grade.Column1.SetFocus
This.Grade.Column1.fwcheckbox1.SetFocus

ThisForm.Grade.AfterRowColChange()
* Fim **************************************************************************************
ENDPROC

PROCEDURE btnConfirmar.Click
Thisform.Grade.Refresh
Select LocalObs
Go Top In LocalObs
Thisform.Grade.Refresh

**** monta as observacoes normais ****
If Vartype(pcLista) <> 'C'
	Locate For Selecao = 1
	If Eof([LocalObs])
		=Messagebox([Nenhuma Observação Foi Selecionada!], 0+48, [Atenção])
		Go Top In LocalObs
		Thisform.Grade.Column1.SetFocus
		Return .F.
	Endif
	llPri = .T.
	If Not IsEmpty(TprMvCab.obses)
		If (Messagebox([Já Existe Conteúdo Informado No Campo de Observação] + Chr(13) + ;
				[Deseja Manter o Conteúdo e Acrescentar as Observações] + Chr(13) + ;
				[Que Foram Selecionadas Agora?], 4+32+256, [Atenção!]) <> 6)
			Replace obses With [] In TprMvCab
		Else
			llPri = .F.
		Endif
	Endif

	Select LocalObs
	Go Top
	Scan
		If (LocalObs.Selecao = 1)
			Replace obses With obses + ;
				Iif(llPri, [], Chr(13) + Iif(Thisform.chkPular.Value, Chr(13), [])) + ;
				Alltrim(LocalObs.Observas) In TprMvCab
			llPri = .F.
		Endif
	Endscan
Else
	If Vartype(pcTipo) <> 'C'
		*** Mena - 19/02/2015 monta campo codobs2 com as finalidades da operação - icaro ***
		listaObs = []	&& carrega a lista na variável publica do SIGOPINF
		Select LocalObs
		Go Top
		llPri = .T.
		Scan
			If (LocalObs.Selecao = 1)
				listaObs = Iif(llPri,','+Alltrim(Str(LocalObs.codigos))+',', Alltrim(listaObs)+Alltrim(Str(LocalObs.codigos))+',')
				llPri = .F.
			Endif
		Endscan
	Endif
	*** prepara o cursor com a finalidade do grupo ***
	If Vartype(pcTipo) = 'C' And pcTipo = 'GRP'
		*** atualiza cursor da tabela de finalidades do grupo atual ***
		Select LocalObs
		Go Top
		Scan
			If (LocalObs.Selecao = 1)
				Select crSigOpFnl
				Locate For codigos = LocalObs.codigos
				If !Found()
					lcIdChave = fUniqueIds()
					Insert Into crSigOpFnl (operacao, codigos, codutil, cidchaves, empdopnums) Values ('GRP',LocalObs.codigos, Alltrim(crSigCdGrp.cgrus),lcIdChave,'')
				Endif
			Else
				Select crSigOpFnl
				Locate For codigos = LocalObs.codigos
				If Found()
					Delete
				Endif
			Endif
			Select LocalObs
		Endscan
	Endif

	*** prepara o cursor com a finalidade do item ***
	If Vartype(pcTipo) = 'C' And pcTipo = 'ITN'
		*** atualiza cursor da tabela de finalidades do item atual ***
		Select LocalObs
		Go Top
		Scan
			If (LocalObs.Selecao = 1)
				Select crSigOpFnl
				Locate For codigos = LocalObs.codigos And codutil = pcChv
				If !Found()
					lcIdChave = fUniqueIds()
					Insert Into crSigOpFnl (operacao, codigos, codutil, cidchaves, empdopnums) Values ('ITN',LocalObs.codigos, pcChv,lcIdChave,pcEDN)
				Endif
			Else
				Select crSigOpFnl
				Locate For codigos = LocalObs.codigos And codutil = pcChv
				If Found()
					Delete
				Endif
			Endif
			Select LocalObs
		Endscan
	Endif
Endif

Thisform.Release
* Fim **************************************************************************************

ENDPROC
