------------------------------------------------------------
* PROCEDURE autentica
------------------------------------------------------------
PROCEDURE autentica
Lparameters _TpAutentica

Local Impressora, _autVal, _toPrint

Thisform.Primeiravez = .f.

If ThisForm.Parentform.Podatamgr.Cursorquery('SigMvCab','CrTprMvCab','EmpDopNums',Thisform.toSeek)
	_autVal  = padc(DTOC(Date())+"-"+Time()+" - "+left(CrTprMvCab.ecfs,3)+" - "+CrTprMvCab.notas,60," ")
else
	_autVal  = padc(DTOC(Date())+"-"+Time(),60," ")
EndIf

_toPrint = padc(alltrim(ThisForm.txtObs.Value),60," ")

If _TpAutentica = 1 &&Impressora Fiscal (Verso) / Impressora de Cheque

	If InList(CrSigFiFis.Imps,'DATAREGIS','NDATAREGIS','BEMA8501','DARUMA','EPSON')
		If CrSigFiFis.Imps = 'BEMA8501'
			oFormulario = ThisForm.ParentForm.oModemEest
			lcNoSerie = LeituraFab(oFormulario)
			ThisForm.ParentForm.poDataMgr.SqlExecute([Select cModelos From SigFiMpF Where cImpFabs = '] + fDecriptografar(lcNoSerie) + ['], [crTmpImpF])
			lcModelo = crTmpImpF.cModelos
		Else
			lcModelo = Space(10)
		EndIf

		=fMensagemFixa( "Posicione o papel na impressora para autenticação...", 48, "Autenticação", .F., .F., "\vbmp\ifiscal2.bmp" )

		If ! lcModelo = [MP-7000 TH FI]
			RetImpVer = ImpVerso(_autVal,_toPrint,Thisform.oModem,.t.)
		Else
			If Not Thisform.Primeiravez 
				RetImpVer = ReAutentica(Thisform.oModem)
			Else
				If Not Thisform.ParentForm.TemAutentica
					RetImpVer = .t.
				Else
					RetImpVer = Autentica(Thisform.oModem)
				EndIf
				Thisform.Primeiravez = .f.
			EndIf
		EndIf

	Else

		=ThisForm.parentform.PodataMgr.CursorQuery('SigCdPam','CrTmpParam',.f.,.f.,[ImpCheqs])

		=fMensagemFixa( "Posicione o Papel na impressora para autenticação...", 48, "Autenticação", .F., .F., "\vbmp\a_print4.bmp" )

		If Not fLerSettings('CrSigSyCom',ThisForm.Parentform.Podatamgr)
			Return 0
		EndIf
		Select * From crSigSyCom Where Tipos = 'Impressora de Cheque' Into Cursor CrTmpCfCom ReadWrite
		Select CrTmpCfCom
		Go Top

		IF ! ThisForm.OleComm.portopen And crTmpCfCom.Portas <> 0
			With ThisForm.OleComm
				.commport = crTmpCfCom.Portas
				.Settings = Alltrim(crTmpCfCom.Settings)
				.Inputlen = 1
			EndWith
		EndIf

		ThisForm.OleComm.PortOpen = .T.

		If CrTmpParam.ImpCheqs = 2

			Set Century Off

			_toPrint = ThisForm.TxtObs.Value

			ThisForm.OleComm.Output = Spac( 8 ) + _autVal  + Chr(10) + Chr(13)
			ThisForm.OleComm.Output = Spac( 8 ) + _toPrint + Chr(10) + Chr(13)
			ThisForm.OleComm.Output = Chr(12)

		Else

			Set Century Off

			_toPrint = ThisForm.TxtObs.Value
			
			ThisForm.OleComm.Output = Chr(27) + Chr(64) + Chr(13)
			ThisForm.OleComm.Output = Spac( 5 ) + _autVal  + Chr(13) + Chr(10)
			ThisForm.OleComm.Output = Spac( 5 ) + _toPrint + Chr(13) + Chr(10)
			ThisForm.OleComm.Output = Chr(12)

		Endif

		Set Century On

		ThisForm.OleComm.PortOpen = .F.

	EndIf

Else

	=fMensagemFixa( "Posicione o papel na impressora para autenticação...", 48, "Autenticação", .F., .F., "\vbmp\IFiscal.bmp" )
	If InList(CrSigFiFis.Imps,'SWEDA','NSWEDA') And Not Thisform.Primeiravez 
		RetImpVer = ReAutentica(Thisform.oModem)
	Else
		If Not Thisform.ParentForm.TemAutentica
			RetImpVer = .t.
		Else
			RetImpVer = Autentica(Thisform.oModem)
		Endif
		Thisform.Primeiravez = .f.
	Endif
Endif
Return .t.
ENDPROC

------------------------------------------------------------
* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
Parameters poform, _pEmps, _pDopes, _pNumes, _pTipo, _pnAutentica

ThisForm.cntSombra.lblSombra.Caption = ThisForm.Caption
ThisForm.cntSombra.lblTitulo.Caption = ThisForm.Caption

ThisForm.Parentform = poform

Thisform.oModem = ThisForm.Parentform.oModemEest
ThisForm.toSeek = _pEmps +_pDopes +Str(_pNumes,6)
Thisform.Primeiravez = .T.

lcEDN = _pEmps + _pDopes + Str(_pNumes,6)

=fLerFiscal()

Select CrSigFiFis
Goto top

If Not _pTipo

	If !ThisForm.Parentform.podatamgr.cursorquery('SigMvPar','CrTmpPar','EmpDopNums',lcEDN)
		=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (crtmppar)')
		Return 0
	EndIf

	Select CrTmpPar
	Scan
		Scatter Memvar
		If ThisForm.Parentform.podatamgr.cursorquery('SigOpFp','CrTmpFpag','Fpags',CrTmpPar.Fpags) And InList(CrTmpFpag.Autenticas,1,3)
			m.Obs = Padr(Iif(!IsEmpty(NCarnes),'# ' + AllTrim(NCarnes) + ;
					Iif(!IsEmpty(Outros),'; ',''),'') + Outros,40)
			m.AutNb = m.cnnSuParcs		
			m.Marca = .t.
			Printed = .f.
			m.Autenticas = CrTmpFpag.Autenticas
			If _pnAutentica > 0 And _pnAutentica # m.Autenticas
				Loop
			EndIf
			Insert into csPgto from Memvar
		Endif
	EndScan
Else

	If ThisForm.Parentform.podatamgr.cursorquery('SigMvPec','CrTmpEstPe','EmpDopNums',lcEDN)
	
		ThisForm.ParentForm.PodataMgr.Cursorquery('SigCdOpe','CrTmpOpe','Dopes',_pDopes,[Autenticas])

		Select CrTmpEstPe
		Scan
			Scatter Memvar
			_nDope = Val(Left(Str(Codigos,10),4))
			_nNume = Val(Right(Str(Codigos,10),6))
			If ThisForm.Parentform.podatamgr.cursorquery('SigCdOpe','CrTmpOpe2','ndopes',_ndope,[Dopes,CupFis])
				m.Obs = CrTmpOpe2.dopes+' '+Str(_nNume,6)
				m.Fpags = Str(CrTmpEstPe.Codigos)
				m.AutNb = ''
				m.Marca = .t.
				Printed = .f.
				m.Autenticas = CrTmpOpe.Autenticas
				If ( _pnAutentica > 0 And _pnAutentica # m.Autenticas ) Or InList(CrTmpOpe2.CupFis,1,3)
					Loop
				EndIf
				Insert into csPgto from Memvar
			Endif
		EndScan

	Endif

EndIf

Select csPgto
If ( Alltrim(CrSigFiFis.Imps) = 'DARUMA' And _PnAutentica = 1 ) Or ;
		(_PnAutentica <> 1 And Not Thisform.ParentForm.TemAutentica )
	Zap
Endif
Select csPgto
Go Top	
If Reccount() = 0
	ThisForm.Release
	Return .F.
Endif

With ThisForm
	With .Grd 
		.RecordSource          = 'csPgto'
		.RecordSourceType      = 1
		.Column1.ControlSource = 'csPgto.marca'
		.Column2.ControlSource = 'csPgto.Obs'
		.Column3.ControlSource = 'csPgto.fPags'
		.Refresh
	EndWith	

	If InList(CrSigFiFis.Imps,'BEMA8501','EPSON')
		.txtObs.Visible = .f.
	Else
		.txtObs.value = ""
		.Refresh
		.txtObs.SetFocus
	Endif
EndWith

ENDPROC

PROCEDURE Click
Select csPgto

	Go top
	Scan
		if !(csPgto.Printed)
			With ThisForm
				.Grd.Refresh
				.Grd.Column1.SetFocus
			EndWith
			MessageBox("Autenticação Obrigatória",16,"")
			Return
		endif
	EndScan
EndIf
ldtime = DateTime()
Do While DateTime() - ldtime <= 2
	=chrsaw(.5)
EndDo
ThisForm.Release
ENDPROC
