* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
With ThisForm
	.poDataMgr = CreateObject([fSqlConector], .Name)

	If (.poDataMgr.pnIdconn > 0)
		With .poDataMgr
			.AddCursor([SigCdEsp], [Especies], [crSigCdEsp], [], ThisForm.Pagina.Lista.Grade)
			.pfSqlTabela(1).pColuna([Especies], [], [], [Espécie],   150, .t.)
			.pfSqlTabela(1).pColuna([Descs],    [], [], [Descrição], 290, .t.)
		EndWith

		=DoDefault()

		.Pagina.Lista.Grade.Column1.SetFocus
	Else
		=Messagebox([Impossível Efetuar Conexão Com o Servidor de Banco de Dados...], 0+48, [CreateObject])
		Return .f.
	EndIf
EndWith
* Fim **************************************************************************************
ENDPROC

------------------------------------------------------------
* PROCEDURE msv_procurar
------------------------------------------------------------
PROCEDURE msv_procurar
Local lcBusca

If Not Empty(ThisForm.Pagina.Dados.getCodigo.Value)
	ThisForm.poDataMgr.LocateCursor([crSigCdEsp], [Especies], ThisForm.Pagina.Dados.getCodigo.Value)
EndIf
* Fim **************************************************************************************
ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Lista.Grupo_op.Click
------------------------------------------------------------
PROCEDURE Pagina.Lista.Grupo_op.Click
Parameters lnOpcao

If Not DoDefault(lnOpcao)
	ThisForm.mAtivaPagina1
	Return .f.
EndIf

Do Case
	Case InList(ThisForm.pcEscolha, [INSERIR], [PROCURAR])
		ThisForm.Pagina.Dados.getCodigo.SetFocus

	Case InList(ThisForm.pcEscolha, [ALTERAR])
		ThisForm.Pagina.Dados.getDescs.SetFocus

	Case InList(ThisForm.pcEscolha, [CONSULTAR], [EXCLUIR])
		ThisForm.Pagina.Dados.Grupo_salva.Salva.SetFocus

		If (ThisForm.pcEscolha = [EXCLUIR])
			lcQuery = [Select cIdChaves ] + ;
					    [From SigMvCcr ] + ;
					   [Where EspecieNfs = '] + crSigCdEsp.Especies + [']

			If (ThisForm.poDataMgr.SqlExecute(lcQuery, [LocalBus]) < 1)
				=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (LocalBus)])
				Return .f.
			EndIf
			Select LocalBus
			Go Top In LocalBus
			If Not Eof([LocalBus])
				=MessageBox([A Espécie Está Em Uso Nos Títulos e Não Pode Ser Excluída!!!], 0+16, [Atenção!!!])
				ThisForm.mAtivaPagina1
				Return .f.
			EndIf

			lcQuery = [Select Dopes ] + ;
					    [From SigCdOpe ] + ;
					   [Where Especies = '] + crSigCdEsp.Especies + [']

			If (ThisForm.poDataMgr.SqlExecute(lcQuery, [LocalBus]) < 1)
				=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (LocalBus)])
				Return .f.
			EndIf
			Select LocalBus
			Go Top In LocalBus
			If Not Eof([LocalBus])
				=MessageBox([A Espécie Está Em Uso Na Operação "] + Alltrim(LocalBus.Dopes) + [" e Não Pode Ser Excluída!!!], 0+16, [Atenção!!!])
				ThisForm.mAtivaPagina1
				Return .f.
			EndIf
		EndIf
EndCase
* Fim **************************************************************************************
ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
If InList(ThisForm.pcEscolha, [INSERIR], [ALTERAR])
	If Empty(crSigCdEsp.Especies)
		=MessageBox([Código Inválido!!!], 0+64, [Atenção!!!])
		ThisForm.Pagina.Dados.getCodigo.SetFocus
		Return .f.
	EndIf

	If Empty(crSigCdEsp.Descs)
		=MessageBox([Descrição Inválida!!!], 0+64, [Atenção!!!])
		ThisForm.Pagina.Dados.getDescs.SetFocus
		Return .f.
	EndIf

	If Empty(crSigCdEsp.Provs)
		=MessageBox([Integração Contábil Inválida!!!], 0+64, [Atenção!!!])
		ThisForm.Pagina.Dados.optProvs.option1.SetFocus 
		Return .f.
	EndIf

	If (ThisForm.pcEscolha = [INSERIR]) And ThisForm.poDataMgr.ChkRegister([SigCdEsp], [Especies], crSigCdEsp.Especies) 
		=MessageBox([Código Já Cadastrado!!!], 0+64, [Atenção!!!])
		ThisForm.Pagina.Dados.getCodigo.SetFocus
		Return .f.
	EndIf
EndIf

=DoDefault()

Select crSigCdEsp

ThisForm.mAtivaPagina1
* Fim **************************************************************************************
ENDPROC

	ThisForm.mAtivaPagina1
EndIf
* Fim **************************************************************************************
ENDPROC
