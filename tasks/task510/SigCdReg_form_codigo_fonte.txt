* PROCEDURE Init
------------------------------------------------------------
PROCEDURE Init
LPARAMETERS lcacesso

With ThisForm
	.poDatamgr = CreateObject('fSqlConector')
	With .poDatamgr
		.AddCursor('SigCdReg','regiaos','crSigCdReg','',ThisForm.pagina.lista.grade)
		.pfSqlTabela(1).pColuna('regiaos','','','Regiãos',80,.T.)
		.pfSqlTabela(1).pColuna('descs','','','Descrição',160,.T.)
		.AddCursor('SigCdRgi','pkChaves','crSigCdRgi','','')
	EndWith
ENDWITH

SELECT *, SPACE(40) as munici, SPACE(2) as uf from crSigcdRgi into cursor crMuni readwrite

SELECT crMuni

With ThisForm.Pagina.Dados.gradMuni
	.RecordSource = 'crMuni'
	.column1.ControlSource = 'crMuni.uf'
	.Column2.controlsource = 'crMuni.munici'
	.Refresh
EndWith

SELECT *, SPACE(50) as vend from crSigcdRgi into cursor crVend readwrite

SELECT crVend

With ThisForm.Pagina.Dados.gradVend
	.RecordSource = 'crVend'
	.column1.ControlSource = 'crVend.iclis'
	.Column2.controlsource = 'crVend.vend'
	.Refresh
EndWith

= DoDefault()
thisform.LockScreen = .f.

ENDPROC

PROCEDURE msv_procurar
Local lcBusca

If .Not.IsEmpty(ThisForm.Pagina.Dados.get_codigo.Value)
	ThisForm.poDataMgr.LocateCursor('crSigCdReg','regiaos',ThisForm.Pagina.Dados.get_codigo.Value)

	EndIf

ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Lista.Grupo_op.Click
------------------------------------------------------------
PROCEDURE Pagina.Lista.Grupo_op.Click
LParameters opcao
LOCAL lcSK

=DODEFAULT()

*** pega chave secundária se houver ***
lcSK = IIF(ThisForm.PcEscolha='INSERIR', '', crsigcdreg.skchaves)

*** Prepara Grade de Municipio ***
Select crMuni
Zap

lcSql = [Select distinct a.skchaves as skchaves, pkchaves, codibges, tipos, ] +;
		[c.descs as munici, d.estados as uf From SigCdRgi a ] + ;
		[left Join SigCdReg b On a.skChaves = b.skchaves ] +  ;
		[left join sigcdmun c on c.codigos = a.codibges ] +;
		[left join sigcdufs d on d.ufibges = c.ufibges ] +;
		[Where a.skchaves = '] + lcSK + [' and tipos='M']


=ThisForm.PodataMgr.SqlExecute(lcSql,'TmpMun')
Select TmpMun
Scan
	Select crMuni
	Append Blank
	replace	skchaves with tmpMun.skchaves, ;
			pkchaves with tmpMun.pkchaves, ;
			codibges with TmpMun.codibges, ;
			tipos    with TmpMun.tipos,    ;
			munici   with tmpMun.munici,     ;
			uf       with tmpMun.uf
EndScan
Select crMuni
If Reccount() = 0
	Append Blank
EndIf

ThisForm.Pagina.Dados.GradMuni.Refresh


*** Prepara Grade de Vendedor ***
Select crVend
Zap

lcSql = [Select distinct a.skchaves as skchaves, pkchaves, codibges, tipos, ] +;
		[a.iclis as iclis, c.razaos as vend From SigCdRgi a ] + ;
		[left Join SigCdReg b On a.skChaves = b.skchaves ] +  ;
		[left join sigcdcli c on c.iclis = a.iclis Where a.skchaves = '] + lcSK + [' and a.tipos='C']


=ThisForm.PodataMgr.SqlExecute(lcSql,'TmpVen')
Select TmpVen
Scan
	Select crVend
	Append Blank
	replace	skchaves with tmpVen.skchaves, ;
			pkchaves with tmpVen.pkchaves, ;
			tipos    with TmpVen.tipos,    ;
			iclis 	 with TmpVen.iclis,    ;
			vend 	 with tmpVen.vend
ENDSCAN

Select crVend
If Reccount() = 0
	Append Blank
EndIf

ThisForm.Pagina.Dados.GradVend.Refresh

SELECT crSigCdReg

If !dodefault( opcao )
	ThisForm.MAtivaPagina1
	Return
Endif

Do Case
	Case Inlist( ThisForm.PcEscolha, 'INSERIR', 'PROCURAR' )
		ThisForm.Pagina.Dados.Get_Codigo.SetFocus
	Case ThisForm.PcEscolha = 'ALTERAR'
		ThisForm.Pagina.Dados.Get_Desc.SetFocus
Other
	ThisForm.Pagina.Dados.Grupo_Salva.Salva.SetFocus
Endcase

ENDPROC

------------------------------------------------------------
* PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
------------------------------------------------------------
PROCEDURE Pagina.Dados.Grupo_Salva.Salva.Click
Local llOks
llOks = .t.

If ThisForm.PcEscolha = 'INSERIR'
	If ThisForm.poDataMgr.ChkRegister('SigCdReg','regiaos',ThisForm.Pagina.Dados.Get_Codigo.Value)
		Messagebox( 'Região já Cadastrada!', 48, '' )
		ThisForm.Pagina.Dados.Get_Codigo.SetFocus
		Return
	Endif
	replace skChaves With fUniqueIds() in crSigCdReg
Endif


**** atualiza sigcdrgi com municipios e vendedores ****
SET delete off
If InList(ThisForm.pcescolha,'INSERIR','ALTERAR')
*** Atualiza tabela de municipios ***	
	Select CrSigCdRgi
	ZAP
	Select crMuni
	GO top
	Scan
		If Not Empty(codibges) 
			Select CrSigCdRgi
			Append Blank
			replace codibges With crMuni.codibges,;
					tipos    With crMuni.tipos,;
					skChaves With crSigCdReg.skChaves,;
					PkChaves With crMuni.pkChaves
			IF DELETED('crMuni')
				delete
			ENDIF
		ENDIF
		SELECT crMuni
	EndScan

*** Atualiza tabela de vendedores ***	
	Select crVend
	GO top
	Scan
		If Not Empty(iclis) 
			Select CrSigCdRgi
			Append Blank
			replace iclis With crVend.iclis,;
					tipos    With crVend.tipos,;
					skChaves With crSigCdReg.skChaves,;
					PkChaves With crVend.pkChaves
			IF DELETED('crVend')
				delete
			ENDIF
		ENDIF
		SELECT crVend
	EndScan
ENDIF
SET delete on
********************************************************

If Not ThisForm.poDataMgr.Update('crSigCdRgi')
	=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (Update - crSigCdRgi)')
	llOks = .f.
EndIf
If !llOks
	ThisForm.poDataMgr.Rollback()
	Return .f.
EndIf

a=DoDefault()
Thisform.mAtivapagina1

ENDPROC
