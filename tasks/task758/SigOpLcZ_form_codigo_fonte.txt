PROCEDURE montagrade
Local lnOldSel

Select crItensLoc

Go top

	.RecordSource = [crItensLoc]
	.column1.ControlSource = [crItensLoc.Cpros]
	.column2.ControlSource = [crItensLoc.Dpros]
	.column3.ControlSource = [crItensLoc.Marcas]
	.column4.ControlSource = [crItensLoc.Locals]
	.Refresh()
EndWith 

ThisForm.grdItens.Column1.SetFocus
ThisForm.grdItens.Refresh
ENDPROC

PROCEDURE AfterRowColChange
LPARAMETERS nColIndex

lcArquivo2 = Sys(2023) + [\] + goSistema.Sys2015Tmp + Sys(2015) + [.jpg]

Select crItensLoc
= ThisForm.poDataMgr.CursorQuery('SigCdPro','crTPro','Cpros',crItensLoc.Cpros,'FigJpgs')
If Not IsNull(crTPro.FigJpgs) And Not IsEmpty(crTPro.FigJpgs)
	This.parent.ImgJpg.Visible = .F.
	This.parent.ImgJpg.Picture = ''
	Select crTPro
	Clear Resources
	lcFoto=Strconv(Strtran(Strtran(Strtran(crTPro.FigJpgs,[data:image/png;base64,],[]),[data:image/jpeg;base64,],[]),[data:image/jpg;base64,],[]),14)
	StrToFile(lcFoto, lcArquivo2)
	This.parent.ImgJpg.Picture = (lcArquivo2)
	This.parent.ImgJpg.Visible = .T.
Else
	This.parent.ImgJpg.Visible = .F.
	This.parent.ImgJpg.Picture = ''
EndIf

If Empty(crItensLoc.Locals) And Not Empty(crItensLoc.Marcas)
	ThisForm.cmdAcoes.btnLocalizar.Enabled = .T.
Else
	ThisForm.cmdAcoes.btnLocalizar.Enabled = .F.
EndIf 
ENDPROC

PROCEDURE btnLocalizar.Click

	Messagebox([Informar a Base da Localização.],0+48,[Atenção])
	Return .F.

	Select csLocals
	Append Blank 
	
	Replace Cpros  With crItensLoc.Cpros
	Replace Locals With lcLocaliza
