PROCEDURE processamento
Local lcEmp, llNPr, lcArq, lnOrd, llOks

lcEmp = Padr(ThisForm.getDEmpOs.Value, 3)
llNPr = (ThisForm.optProcessados.Value = 1) && 1-Não Processados; 2-Já Processados
lcArq = ThisForm.getArqNoms.Value 

Create Cursor crExcel (TIPO_REGISTRO c(3), ADQUIRENTE c(60), LOJA_FILIAL c(15), BANDEIRA c(60), TIPO_TRANSACAO c(15), ;
					   NRO_PARCELA c(2), QTD_PARCELAS c(2), TID c(40), NSU c(30), AUTORIZACAO c(20), VALOR c(14), ;
					   CLIENTE c(50), DATA_VENDA c(10), ORIGEM c(10), SEQUENCIA c(20), NOTA_FISCAL c(20), CPF_CNPJ c(20), ;
					   ESTABELECIMENTO c(20), DATA_VENCIMENTO c(10), NUMERO_CARTAO c(30), REFERENCIA1 c(60))

Insert Into crExcel (TIPO_REGISTRO, ADQUIRENTE, LOJA_FILIAL, BANDEIRA, TIPO_TRANSACAO, NRO_PARCELA, QTD_PARCELAS, ;
					 TID, NSU, AUTORIZACAO, VALOR, CLIENTE, DATA_VENDA, ORIGEM, SEQUENCIA, NOTA_FISCAL, CPF_CNPJ, ;
					 ESTABELECIMENTO, DATA_VENCIMENTO, NUMERO_CARTAO, REFERENCIA1) ; 
			 Values ([A01], [ADQUIRENTE], [LOJA_FILIAL], [BANDEIRA], [TIPO_TRANSACAO], [NRO_PARCELA], [QTD_PARCELAS], ;
			 		 [TID], [NSU], [AUTORIZAÇÃO], [VALOR], [CLIENTE], [DATA_VENDA], [ORIGEM], [SEQUENCIA], [NOTA_FISCAL], ;
			 		 [CPF_CNPJ], [ESTABELECIMENTO], [DATA_VENCIMENTO], [NUMERO_CARTAO], [REFERENCIA1])

lnOrd = 0
Select csOperacoes
Scan
	If csOperacoes.SelImp
		lnOrd = lnOrd + 1
		
		Scatter Memvar Memo

		m.SEQUENCIA = Alltrim(Str(lnOrd, 20))
		
		Insert Into crExcel From Memvar

		Insert Into crSigPcOoL (Tipos, Emps, Dopes, Numes, Datas, Usuars, Produtos, ;
								Processos, cIdChaves, EmpDopNums) ;
						Values ([SIGOPVDA], m.Emps, m.Dopes, m.Numes, Datetime(), ;
								Usuar, Iif(llNPr, [], [JÁ PROCESSADOS]), m.REFERENCIA1, ;
								fUniqueIds(), m.EmpDopNums)
	EndIf
EndScan

Select crExcel
Copy To &lcArq. Delimited

llOks = .t.

If Not File(lcArq)
	=MessageBox([Não Foi Possível Gerar o Arquivo!!! Verifique Suas Permissões!!!], 0+16, [Atenção!!!])
	llOks = .f.
EndIf

If llOks And Not ThisForm.poDataMgr.Update([crSigPcOoL])
	=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (Update - crSigPcOoL)])
	llOks = .f.
EndIf

If llOks And (ThisForm.poDataMgr.Commit() < 1)
	=MessageBox([Favor Reinicializar o Processo!!!], 16, [Falha na Conexão (Commit)])
EndIf

If Not llOks
	ThisForm.poDataMgr.RollBack()		
Else
	=Messagebox([Arquivo Gerado!!!], 0+48, [OK])
		
	ThisForm.LimpaSelecao()
	ThisForm.cmdSelecionar.Click(.t.)
EndIf

Return llOks
* Fim ********************************************************************************************
ENDPROC

PROCEDURE selecionar
Private pDtI, pDtF

Local lcEmp, llNPr, ldDtI, ldDtF, lcArq

Zap In csOperacoes

lcEmp = Padr(ThisForm.getEmpOs.Value, 3)
llNPr = (ThisForm.optProcessados.Value = 1) && 1-Não Processados; 2-Já Processados
ldDtI = ThisForm.getDtIni.Value
ldDtF = ThisForm.getDtFin.Value 
lcArq = ThisForm.getArqNoms.Value 
lcAdq = ThisForm.getAdquirs.Value 

If Empty(ldDtI)
	=MessageBox([A Data Inicial Não Pode Ficar em Branco!!!], 0+16, [Atenção!!!])
	ThisForm.getDtIni.SetFocus 
	Return .f.
EndIf

If Empty(ldDtF)
	=MessageBox([A Data Final Não Pode Ficar em Branco!!!], 0+16, [Atenção!!!])
	ThisForm.getDtFin.SetFocus 
	Return .f.
EndIf

If (ldDtI > ldDtF)
	=MessageBox([A Data Inicial Não Pode Ser Maior do Que a Data Final!!!], 0+16, [Atenção!!!])
	ThisForm.getDtIni.SetFocus 
	Return .f.
EndIf

If Not fVerificaPasta(lcArq, 2)
	=MessageBox([O Arquivo a Ser Gerado Precisa Ser Informado!!!], 0+16, [Atenção!!!])
	ThisForm.btnNomArqs.SetFocus
	Return .f.
EndIf

pDtI = fDtoSQL(ldDtI)
pDtF = fDtoSQL(ldDtF, [23:59:59])
						   
lcQuery = [Select 'A02' as TIPO_REGISTRO, p.Adquirente as ADQUIRENTE, p.Emps as LOJA_FILIAL, f.Bandeira as BANDEIRA1, ] + ;
				 [d.RClis as BANDEIRA2, f.Opers as TIPO_TRANSACAO, p.Parcs as NRO_PARCELA, p.TParcs as QTD_PARCELAS, ] + ;
				 [Space(40) as TID, p.NSU as NSU, p.AUTORIZA as AUTORIZACAO, p.Valos as VALOR, Space(50) as CLIENTE, ] + ;
				 [p.Datas as DATA_VENDA, f.CCupomNvs as ORIGEM, '' as SEQUENCIA, '' as NOTA_FISCAL, ] + ;
				 ['' as CPF_CNPJ, f.CodEstabs as ESTABELECIMENTO, p.Vencs as DATA_VENCIMENTO, '' as NUMERO_CARTAO, ] + ;
				 [p.EmpDopNums + p.FPags + Str(p.Nopers, 10) as REFERENCIA1, p.Emps, p.Dopes, p.Numes, p.EmpDopNums ] + ;
			[From SigOpFP f, SigCdCli d, SigMvPar p, SigCdOpe o ] + ;
		   [Where f.Infos = 'T' And ] + ;
		   		 [f.ContaDs = d.IClis And ] + ;
		   		 [p.FPags = f.FPags And ] + ;
		   		 Iif(Empty(lcEmp), [], [p.Emps = '] + lcEmp + [' And ]) + ;
		   		 Iif(Empty(lcAdq), [], [p.Adquirente = '] + lcAdq + [' And ]) + ;
		   		 [p.Datas BetWeen ?pDtI And ?pDtF And ] + ;
		   		 [p.Dopes = o.Dopes And ] + ;
		   		 [Not o.Vendas = 2 And ] + ;
		   		 [Not p.Vencs Is Null And ] + ;
		   		 [p.EmpDopNums ] + Iif(llNPr, [Not ], []) + ;
		   		 [In (Select Distinct b.EmpDopNums ] + ;
	 	  		  	   [From SigPcOoL b ] + ;
	 	  		  	  [Where b.Tipos = 'SIGOPVDA  ') ] + ;
		   [Order By p.Datas, p.EmpDopNums, p.FPags, p.Parcs]

If Not fExecCmds(ThisForm.poDataMgr, [crDados], lcQuery)
	Return .f.
EndIf

Select crDados
Go Top In crDados
If Eof([crDados])
	=MessageBox([Não Existem Registros Informados Com as Condições Selecionadas!!!], 0+64, [Nenhum Registro Encontrado])
	ThisForm.LimpaSelecao()
	Return .f.
EndIf

lnSeq = 0
Select crDados
Scan
	Scatter Memvar Memo
	
	m.Erros = .f.
	If (Empty(Alltrim(m.BANDEIRA1 + m.BANDEIRA2)) Or Empty(m.NRO_PARCELA) Or Empty(m.QTD_PARCELAS) Or Empty(m.NSU) Or ;
		Empty(m.AUTORIZACAO) Or Empty(m.VALOR) Or Empty(m.DATA_VENDA) Or Empty(m.DATA_VENCIMENTO) Or Empty(m.ADQUIRENTE))
		m.Erros = .t.
	EndIf
	
	m.BANDEIRA		  = Iif(Empty(m.BANDEIRA1), m.BANDEIRA2, m.BANDEIRA1)
*!*		m.ORIGEM    	  = Iif(Empty(m.ORIGEM), [TEF], m.ORIGEM) && Alterado conforme informação da Mena em 13/10/2016
	m.ORIGEM    	  = Iif(Empty(m.ORIGEM), [POS], m.ORIGEM)
	m.NRO_PARCELA     = Alltrim(Str(m.NRO_PARCELA, 3))
	m.QTD_PARCELAS    = Alltrim(Str(m.QTD_PARCELAS, 3))
	m.VALOR			  = Strtran(Alltrim(Str(m.VALOR, 12, 2)), [.], [,])
	m.DATA_VENDA	  = Dtoc(Ttod(m.DATA_VENDA))
	m.DATA_VENCIMENTO = Dtoc(Ttod(m.DATA_VENCIMENTO))
	m.SelImp = Not m.Erros

	Insert Into csOperacoes From Memvar
EndScan

ThisForm.getDtIni.Enabled = .f.
ThisForm.getDtFin.Enabled = .f.
ThisForm.optProcessados.Option1.Enabled = .f.
ThisForm.optProcessados.Option2.Enabled = .f.
ThisForm.getEmpOs.Enabled = .f.
ThisForm.getDEmpOs.Enabled = .f.
ThisForm.getArqNoms.Enabled = .f.
ThisForm.btnNomArqs.Enabled = .f.
ThisForm.cmdSelecionar.Enabled = .f.
ThisForm.cmdBtnForm.btnLimpar.Enabled = .t.
ThisForm.cmdBtnForm.btnCopiar.Enabled = .t.
ThisForm.cmdBtnGrade.btnMarcaTudo.Enabled = .t.
ThisForm.cmdBtnGrade.btnDesmarcar.Enabled = .t.
ThisForm.cmdBtnGrade.btnDocumentos.Enabled = .t.

Select csOperacoes
Go Top

ThisForm.cntGrade.GrdOperacoes.Column1.SetFocus
ThisForm.cntGrade.GrdOperacoes.Column1.Refresh
ThisForm.cntGrade.GrdOperacoes.Refresh
* Fim ********************************************************************************************
ENDPROC
