PROCEDURE btnBoleto.Click
If MessageBox('Confirma a Impressão do(s) Boleto(s)?', 4+32+256, 'Boleto') = 6
	If (Thisform.ParentForm.Tipo = 'ARQUIVADAS')
		ThisForm.ImpArquivadas = .T.
	EndIf

	Create Cursor csEmpresas (ImpEmps l(1), CEmps c(3), Razas c(40))
	Index On CEmps Tag CEmps

	lcQuery = [Select Cemps, Razas ] + ;
			    [From SigCdEmp ] + ;
			   [Where Ativas = 1 ] + ;
			   [Order By Cemps]

	If (ThisForm.ParentForm.poDataMgr.SqlExecute(lcQuery, 'LocalEmp') < 1)
		=MessageBox('Favor Reinicializar o Processo!!!', 16, 'Falha na Conexão (LocalEmp)')
		Return 0
	EndIf

	Select LocalEmp
	Scan
		lcEmD = LocalEmp.Cemps
		If Not Seek(lcEmD, 'csEmpresas', 'CEmps')
			Insert Into csEmpresas (ImpEmps, CEmps, Razas) Values (.T., lcEmD, LocalEmp.Razas)
		EndIf
	EndScan

	Select CEmps From csEmpresas Where ImpEmps Into Cursor csEmps 
	Select csEmps
	oProg = CreateObject('fwprogressbar', 'Imprimindo Boletos...', Reccount())
	oProg.Show
	Scan
		lcEmp = csEmps.Cemps
		oProg.SubTitulo.Caption = 'Boleto da Empresa : ' + lcEmp
		oProg.Update(.T.)

		Select CBars ;
		  From DbEtiqueta ;
		 Where EmpOs = lcEmp ;
		 Order By CBars ;
		  Into Cursor TmpEtiquetas

		Select TmpEtiquetas
		Go Top
		If Not Eof()
			Do SigBlBol.Prg With 10, , csEmps.Cemps, '', '', 0, 64, 1, ALLTRIM(crImpreV.impres)
		EndIf
	EndScan
	oProg.Complete
EndIf
* Fim **************************************************************************************

ENDPROC
