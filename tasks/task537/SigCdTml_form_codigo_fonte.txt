PROCEDURE iniciadesc
iGrupo = crSigOpOpt.Grupos
If not IsEmpty(iGrupo)
	thisform.podatamgr.cursorquery('SigCdGcr','crTmpgccr','codigos',iGrupo)
	thisform.pagina.dados.get_G.value = crTmpgccr.Descrs
Else
	thisform.pagina.dados.get_G.value = ''
EndIf
iConta = crSigOpOpt.Contas
If not IsEmpty(iConta)
	thisform.podatamgr.cursorquery('SigCdCli','crTmpCli','iclis',iConta)
	thisform.pagina.dados.get_C.value = crTmpcli.rclis
Else
	thisform.pagina.dados.get_C.value = ''
EndIf
