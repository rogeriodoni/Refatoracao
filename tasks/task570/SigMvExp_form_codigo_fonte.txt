------------------------------------------------------------
* PROCEDURE Init

PROCEDURE Init
Lparameters pDopes, pTipos, pNumes, pEmps, pConsulta, pCaixa, pMacro

ThisForm.poDataMgr6 = CreateObject('fSqlConector', 'SigMvExp')

ThisForm.Erro = .f.

ThisForm.pCaixa = Iif(Type('pCaixa')=[N],pCaixa,0)
ThisForm.GrupoOper = []
ThisForm.pMacro = pMacro

* Richard em 06/08/2014 - Alterando Caption Conforme Confguração da Movimentação
lcMnu = fRetDopMenu([], pDopes, ThisForm.poDataMgr6)
If Not Empty(lcMnu)
	ThisForm.pDope = Alltrim(lcMnu)
Else
	ThisForm.pDope = pDopes
EndIf

If (InList(pTipos,'C','A'))
	ThisForm.pNume = pNumes
	ThisForm.pEmp  = pEmps
EndIf
	
ThisForm.pTipo = pTipos
* Fim **************************************************************************************
ENDPROC

PROCEDURE Click
Do Case 
	Case (ThisForm.pTipo = 'E') And Not WExist('SigMvCab')
		Do Form SigMvCab With Padr(ThisForm.pDope,20), ThisForm, .F., .F., .F., .F., .f., .f., .f., .f., .f., Iif(ThisForm.pMacro,.F.,Padr(ThisForm.pDope,20))

	Case (ThisForm.pTipo = 'C') And Not WExist('SigMvCab')
		Do Form SigMvCab With Padr(ThisForm.pDope,20), ThisForm.pNume, ThisForm.pEmp, .t., ThisForm.pCaixa, ThisForm
	
	Case (ThisForm.pTipo = 'A') And Not WExist('SigMvCab')
		Do Form SigMvCab With Padr(ThisForm.pDope,20), ThisForm.pNume, ThisForm.pEmp, .F., ThisForm.pCaixa, ThisForm, .f., .f., .f., .f., .t.
		
EndCase
* Fim **************************************************************************************
ENDPROC
